موطأ مالك

كتاب الحج

٢٥١ حديثًا•صفحة ٢ من ٢
٨٥٥أنه قال من ساق بدنة تطوعا فعطبت فنحرها ثم خلى بينها وبين الناس…٨٥٦مثل ذلك .٨٥٧أنه قال من أهدى بدنة جزاء أو نذرا أو هدى تمتع فأصيبت في الطريق فعليه…٨٥٨وحدثني عن مالك، عن نافع، عن عبد الله بن عمر، أنه قال من أهدى بدنة ثم…٨٥٩أنه بلغه أن عمر بن الخطاب، وعلي بن أبي طالب، وأبا، هريرة سئلوا عن…٨٦٠أنه سمع سعيد بن المسيب، يقول ما ترون في رجل وقع بامرأته وهو محرم فلم…٨٦١أن أبا أيوب الأنصاري، خرج حاجا حتى إذا كان بالنازية من طريق مكة أضل…٨٦٢أن هبار بن الأسود، جاء يوم النحر وعمر بن الخطاب ينحر هديه فقال يا…٨٦٣وقع بأهله وهو بمنى قبل أن يفيض فأمره أن ينحر بدنة .٨٦٤أنه قال الذي يصيب أهله قبل أن يفيض يعتمر ويهدي .٨٦٥أنه سمع ربيعة بن أبي عبد الرحمن، يقول في ذلك مثل قول عكرمة عن ابن…٨٦٦كان يقول { ما استيسر من الهدى } شاة .٨٦٧أنه بلغه أن عبد الله بن عباس، كان يقول { ما استيسر من الهدى } شاة .…٨٦٨أن عبد الله بن عمر، كان يقول { ما استيسر من الهدى } بدنة أو بقرة .٨٦٩أن مولاة، لعمرة بنت عبد الرحمن يقال لها رقية أخبرته أنها، خرجت مع…٨٧٠أن رجلا، من أهل اليمن جاء إلى عبد الله بن عمر وقد ضفر رأسه فقال يا…٨٧١أن عبد الله بن عمر، كان يقول المرأة المحرمة إذا حلت لم تمتشط حتى تأخذ…٨٧٢مولى عبد الله بن جعفر أنه أخبره أنه، كان مع عبد الله بن جعفر فخرج معه…٨٧٣عرفة كلها موقف وارتفعوا عن بطن عرنة والمزدلفة كلها موقف وارتفعوا عن…٨٧٤وحدثني عن مالك، عن هشام بن عروة، عن عبد الله بن الزبير، أنه كان يقول…٨٧٥أن عبد الله بن عمر، كان يقول من لم يقف بعرفة من ليلة المزدلفة قبل أن…٨٧٦أنه قال من أدركه الفجر من ليلة المزدلفة ولم يقف بعرفة فقد فاته الحج…٨٧٧وعبيد الله، ابنى عبد الله بن عمر أن أباهما عبد الله بن عمر، كان يقدم…٨٧٨أن مولاة، لأسماء بنت أبي بكر أخبرته قالت، جئنا مع أسماء ابنة أبي بكر…٨٧٩أنه بلغه أن طلحة بن عبيد الله، كان يقدم نساءه وصبيانه من المزدلفة إلى…٨٨٠أخبرته أنها، كانت ترى أسماء بنت أبي بكر بالمزدلفة تأمر الذي يصلي لها…٨٨١معه - كيف كان يسير رسول الله صلى الله عليه وسلم في حجة الوداع حين دفع…٨٨٢أن عبد الله بن عمر، كان يحرك راحلته في بطن محسر قدر رمية بحجر .٨٨٣هذا المنحر وكل منى منحر٨٨٤قال أخبرتني عمرة بنت عبد الرحمن، أنها سمعت عائشة أم المؤمنين، تقول…٨٨٥إني لبدت رأسي وقلدت هديي فلا أحل حتى أنحر٨٨٦أن رسول الله صلى الله عليه وسلم نحر بعض هديه ونحر غيره بعضه .٨٨٧أن عبد الله بن عمر، قال من نذر بدنة فإنه يقلدها نعلين ويشعرها ثم…٨٨٨أن أباه، كان ينحر بدنه قياما . قال مالك لا يجوز لأحد أن يحلق رأسه حتى…٨٨٩اللهم ارحم المحلقين٨٩٠وحدثني عن مالك، عن عبد الرحمن بن القاسم، عن أبيه، أنه كان يدخل مكة…٨٩١أن عبد الله بن عمر، كان إذا أفطر من رمضان وهو يريد الحج لم يأخذ من…٨٩٢أن عبد الله بن عمر، كان إذا حلق في حج أو عمرة أخذ من لحيته وشاربه .٨٩٣أن رجلا، أتى القاسم بن محمد فقال إني أفضت وأفضت معي بأهلي ثم عدلت إلى…٨٩٤أنه لقي رجلا من أهله يقال له المجبر قد أفاض ولم يحلق ولم يقصر جهل ذلك…٨٩٥أنه بلغه أن سالم بن عبد الله، كان إذا أراد أن يحرم دعا بالجلمين فقص…٨٩٦أن عمر بن الخطاب، قال من ضفر رأسه فليحلق ولا تشبهوا بالتلبيد .٨٩٧أن عمر بن الخطاب، قال من عقص رأسه أو ضفر أو لبد فقد وجب عليه الحلاق .٨٩٨حدثني يحيى، عن مالك، عن نافع، عن عبد الله بن عمر، أن رسول الله صلى…٨٩٩وحدثني عن مالك، عن ابن شهاب، عن سالم بن عبد الله، أنه قال كتب عبد…٩٠٠أن عبد الله بن عمر، كان يصلي الظهر والعصر والمغرب والعشاء والصبح بمنى…٩٠١أن رسول الله صلى الله عليه وسلم صلى المغرب والعشاء بالمزدلفة جميعا .٩٠٢أنه سمعه يقول دفع رسول الله صلى الله عليه وسلم من عرفة حتى إذا كان…٩٠٣أن عبد الله بن يزيد الخطمي، أخبره أن أبا أيوب الأنصاري أخبره أنه، صلى…٩٠٤أن عبد الله بن عمر، كان يصلي المغرب والعشاء بالمزدلفة جميعا .٩٠٥أن رسول الله صلى الله عليه وسلم صلى الصلاة الرباعية بمنى ركعتين وأن…٩٠٦. أن عمر بن الخطاب، لما قدم مكة صلى بهم ركعتين ثم انصرف فقال يا أهل…٩٠٧أن عمر بن الخطاب، صلى للناس بمكة ركعتين فلما انصرف قال يا أهل مكة…٩٠٨أنه بلغه أن عمر بن الخطاب، خرج الغد من يوم النحر حين ارتفع النهار…٩٠٩أن رسول الله صلى الله عليه وسلم أناخ بالبطحاء التي بذي الحليفة فصلى…٩١٠أن عبد الله بن عمر، كان يصلي الظهر والعصر والمغرب والعشاء بالمحصب ثم…٩١١أنه قال زعموا أن عمر بن الخطاب، كان يبعث رجالا يدخلون الناس من وراء…٩١٢أن عمر بن الخطاب، قال لا يبيتن أحد من الحاج ليالي منى من وراء العقبة .٩١٣أنه قال في البيتوتة بمكة ليالي منى لا يبيتن أحد إلا بمنى .٩١٤أنه بلغه أن عمر بن الخطاب، كان يقف عند الجمرتين الأوليين وقوفا طويلا…٩١٥أن عبد الله بن عمر، كان يقف عند الجمرتين الأوليين وقوفا طويلا يكبر…٩١٦أن عبد الله بن عمر، كان يكبر عند رمى الجمرة كلما رمى بحصاة . وحدثني…٩١٧أن عبد الله بن عمر، كان يقول من غربت له الشمس من أوسط أيام التشريق…٩١٨أن الناس، كانوا إذا رموا الجمار مشوا ذاهبين وراجعين وأول من ركب…٩١٩وحدثني عن مالك، أنه سأل عبد الرحمن بن القاسم من أين كان القاسم يرمي…٩٢٠أن عبد الله بن عمر، كان يقول لا ترمى الجمار في الأيام الثلاثة حتى…٩٢١أن أبا البداح بن عاصم بن عدي، أخبره عن أبيه، أن رسول الله صلى الله…٩٢٢أنه سمعه يذكر، أنه أرخص للرعاء أن يرموا، بالليل يقول في الزمان الأول…٩٢٣أن ابنة أخ، لصفية بنت أبي عبيد نفست بالمزدلفة فتخلفت هي وصفية حتى…٩٢٤وعبد الله بن دينار، عن عبد الله بن عمر، أن عمر بن الخطاب، خطب الناس…٩٢٥وعبد الله بن دينار، عن عبد الله بن عمر، أن عمر بن الخطاب، قال من رمى…٩٢٦من كان معه هدى فليهلل بالحج مع العمرة ثم لا يحل حتى يحل منهما جميعا٩٢٨افعلي ما يفعل الحاج غير أن لا تطوفي بالبيت ولا بين الصفا والمروة حتى…٩٢٩أن صفية بنت حيى، حاضت فذكرت ذلك للنبي صلى الله عليه وسلم فقال "…٩٣٠لعلها تحبسنا ألم تكن طافت معكن بالبيت٩٣١. أن عائشة أم المؤمنين، كانت إذا حجت ومعها نساء تخاف أن يحضن قدمتهن…٩٣٢وحدثني عن مالك، عن هشام بن عروة، عن أبيه، عن عائشة أم المؤمنين، أن…٩٣٣أن أبا سلمة بن عبد الرحمن، أخبره أن أم سليم بنت ملحان استفتت رسول…٩٣٤أن عمر بن الخطاب، قضى في الضبع بكبش وفي الغزال بعنز وفي الأرنب بعناق…٩٣٥أن رجلا، جاء إلى عمر بن الخطاب فقال إني أجريت أنا وصاحب، لي فرسين…٩٣٦أن أباه، كان يقول في البقرة من الوحش بقرة وفي الشاة من الظباء شاة .٩٣٧وحدثني عن مالك، عن يحيى بن سعيد، عن سعيد بن المسيب، أنه كان يقول في…٩٣٨أن رجلا، جاء إلى عمر بن الخطاب فقال يا أمير المؤمنين إني أصبت جرادات…٩٣٩أن رجلا، جاء إلى عمر بن الخطاب فسأله عن جرادات، قتلها وهو محرم فقال…٩٤٠صم ثلاثة أيام أو أطعم ستة مساكين مدين مدين لكل إنسان أو انسك بشاة أى…٩٤١لعلك آذاك هوامك٩٤٢احلق هذا الشعر وصم ثلاثة أيام أو أطعم ستة مساكين٩٤٣قال من نسي من نسكه شيئا أو تركه فليهرق دما . قال أيوب لا أدري قال ترك…٩٤٤حدثني يحيى، عن مالك، عن ابن شهاب، عن عيسى بن طلحة، عن عبد الله بن…٩٤٥لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…٩٤٦مولى عبد الله بن عباس أن رسول الله صلى الله عليه وسلم مر بامرأة وهي…٩٤٧ما رئي الشيطان يوما هو فيه أصغر ولا أدحر ولا أحقر ولا أغيظ منه في يوم…٩٤٨أفضل الدعاء دعاء يوم عرفة وأفضل ما قلت أنا والنبيون من قبلي لا إله…٩٤٩أن رسول الله صلى الله عليه وسلم دخل مكة عام الفتح وعلى رأسه المغفر…٩٥٠أن عبد الله بن عمر، أقبل من مكة حتى إذا كان بقديد جاءه خبر من المدينة…٩٥٢إذا كنت بين الأخشبين من منى - ونفخ بيده نحو المشرق - فإن هناك واديا…٩٥٣أن عمر بن الخطاب، مر بامرأة مجذومة وهي تطوف بالبيت فقال لها يا أمة…٩٥٤أنه بلغه أن عبد الله بن عباس، كان يقول ما بين الركن والباب الملتزم .٩٥٥أنه سمعه يذكر، أن رجلا، مر على أبي ذر بالربذة وأن أبا ذر سأله أين…٩٥٦في الحج فقال أويصنع ذلك أحد وأنكر ذلك . سئل مالك هل يحتش الرجل لدابته…٩٥٧أنه كان يقول في ذلك مثل قول عائشة رضى الله تعالى عنها
