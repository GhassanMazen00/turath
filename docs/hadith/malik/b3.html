كتاب الصلاة

٧٥ حديثًا
١٤٦أنه قال كان رسول الله صلى الله عليه وسلم قد أراد أن يتخذ خشبتين يضرب…١٤٧إذا سمعتم النداء فقولوا مثل ما يقول المؤذن١٤٨لو يعلم الناس ما في النداء والصف الأول ثم لم يجدوا إلا أن يستهموا…١٤٩إذا ثوب بالصلاة فلا تأتوها وأنتم تسعون وأتوها وعليكم السكينة فما…١٥٠إني أراك تحب الغنم والبادية فإذا كنت في غنمك أو باديتك فأذنت بالصلاة…١٥١إذا نودي للصلاة أدبر الشيطان له ضراط حتى لا يسمع النداء فإذا قضي…١٥٢أنه قال ساعتان يفتح لهما أبواب السماء وقل داع ترد عليه دعوته حضرة…١٥٣أنه بلغه أن المؤذن، جاء إلى عمر بن الخطاب يؤذنه لصلاة الصبح فوجده…١٥٤أنه قال ما أعرف شيئا مما أدركت عليه الناس إلا النداء بالصلاة .١٥٥أن عبد الله بن عمر، سمع الإقامة، وهو بالبقيع فأسرع المشى إلى المسجد .١٥٦ألا صلوا في الرحال١٥٧أن عبد الله بن عمر، كان لا يزيد على الإقامة في السفر إلا في الصبح…١٥٨أن أباه، قال له إذا كنت في سفر فإن شئت أن تؤذن وتقيم فعلت وإن شئت…١٥٩وحدثني عن مالك، عن يحيى بن سعيد، عن سعيد بن المسيب، أنه كان يقول من…١٦٠إن بلالا ينادي بليل فكلوا واشربوا حتى ينادي ابن أم مكتوم١٦١إن بلالا ينادي بليل فكلوا واشربوا حتى ينادي ابن أم مكتوم١٦٢سمع الله لمن حمده ربنا ولك الحمد١٦٣أنه قال كان رسول الله صلى الله عليه وسلم يكبر في الصلاة كلما خفض ورفع…١٦٤أن رسول الله صلى الله عليه وسلم كان يرفع يديه في الصلاة .١٦٥أن أبا هريرة، كان يصلي لهم فيكبر كلما خفض ورفع فإذا انصرف قال والله…١٦٦أن عبد الله بن عمر، كان يكبر في الصلاة كلما خفض ورفع . وحدثني يحيى،…١٦٨وحدثني عن مالك، عن أبي نعيم، وهب بن كيسان عن جابر بن عبد الله، أنه…١٦٩وحدثني عن مالك، عن ابن شهاب، أنه كان يقول إذا أدرك الرجل الركعة فكبر…١٧٠حدثني يحيى، عن مالك، عن ابن شهاب، عن محمد بن جبير بن مطعم، عن أبيه،…١٧١أن أم الفضل بنت الحارث، سمعته وهو، يقرأ { والمرسلات عرفا } فقالت له…١٧٢وحدثني عن مالك، عن أبي عبيد، مولى سليمان بن عبد الملك عن عبادة بن…١٧٣أن عبد الله بن عمر، كان إذا صلى وحده يقرأ في الأربع جميعا في كل ركعة…١٧٤أنه قال صليت مع رسول الله صلى الله عليه وسلم العشاء فقرأ فيها بالتين…١٧٥حدثني يحيى، عن مالك، عن نافع، عن إبراهيم بن عبد الله بن حنين، عن…١٧٦إن المصلي يناجي ربه فلينظر بما يناجيه به ولا يجهر بعضكم على بعض…١٧٧أنه قال قمت وراء أبي بكر وعمر وعثمان فكلهم كان لا يقرأ { بسم الله…١٧٨وحدثني عن مالك، عن عمه أبي سهيل بن مالك، عن أبيه، أنه قال كنا نسمع…١٧٩أن عبد الله بن عمر، كان إذا فاته شىء من الصلاة مع الإمام فيما جهر فيه…١٨٠أنه قال كنت أصلي إلى جانب نافع بن جبير بن مطعم فيغمزني فأفتح عليه…١٨١أن أبا بكر الصديق، صلى الصبح فقرأ فيها سورة البقرة في الركعتين…١٨٢أنه سمع عبد الله بن عامر بن ربيعة، يقول صلينا وراء عمر بن الخطاب…١٨٣وربيعة بن أبي عبد الرحمن، عن القاسم بن محمد، أن الفرافصة بن عمير…١٨٤أن عبد الله بن عمر، كان يقرأ في الصبح في السفر بالعشر السور الأول من…١٨٥إني لأرجو أن لا تخرج من المسجد حتى تعلم سورة ما أنزل الله في التوراة…١٨٦وهب بن كيسان أنه سمع جابر بن عبد الله، يقول من صلى ركعة لم يقرأ فيها…١٨٧من صلى صلاة لم يقرأ فيها بأم القرآن فهي خداج هي خداج هي خداج غير تمام١٨٨أنه كان يقرأ خلف الإمام فيما لا يجهر فيه الإمام بالقراءة .١٨٩أن القاسم بن محمد، كان يقرأ خلف الإمام فيما لا يجهر فيه الإمام…١٩٠أن نافع بن جبير بن مطعم، كان يقرأ خلف الإمام فيما لا يجهر فيه…١٩١أن عبد الله بن عمر، كان إذا سئل هل يقرأ أحد خلف الإمام قال إذا صلى…١٩٢هل قرأ معي منكم أحد آنفا١٩٣إذا أمن الإمام فأمنوا فإنه من وافق تأمينه تأمين الملائكة غفر له ما…١٩٤إذا قال الإمام { غير المغضوب عليهم ولا الضالين } فقولوا آمين فإنه من…١٩٥إذا قال أحدكم آمين وقالت الملائكة في السماء آمين فوافقت إحداهما…١٩٦إذا قال الإمام سمع الله لمن حمده فقولوا اللهم ربنا لك الحمد فإنه من…١٩٧حدثني يحيى، عن مالك، عن مسلم بن أبي مريم، عن علي بن عبد الرحمن…١٩٨أنه سمع عبد الله بن عمر، وصلى، إلى جنبه رجل فلما جلس الرجل في أربع…١٩٩أنه رأى عبد الله بن عمر يرجع في سجدتين في الصلاة على صدور قدميه فلما…٢٠٠أنه أخبره أنه، كان يرى عبد الله بن عمر يتربع في الصلاة إذا جلس قال…٢٠١أن القاسم بن محمد، أراهم الجلوس في التشهد فنصب رجله اليمنى وثنى رجله…٢٠٢أنه سمع عمر بن الخطاب، وهو على المنبر يعلم الناس التشهد يقول قولوا…٢٠٣أن عبد الله بن عمر، كان يتشهد فيقول باسم الله التحيات لله الصلوات لله…٢٠٤وحدثني عن مالك، عن عبد الرحمن بن القاسم، عن أبيه، عن عائشة، زوج النبي…٢٠٥وحدثني عن مالك، عن يحيى بن سعيد الأنصاري، عن القاسم بن محمد، أنه…٢٠٦وحدثني عن مالك، أنه سأل ابن شهاب ونافعا مولى ابن عمر عن رجل، دخل مع…٢٠٧أنه قال الذي يرفع رأسه ويخفضه قبل الإمام فإنما ناصيته بيد شيطان .٢٠٨أن رسول الله صلى الله عليه وسلم انصرف من اثنتين فقال له ذو اليدين…٢٠٩يقول صلى رسول الله صلى الله عليه وسلم صلاة العصر فسلم في ركعتين فقام…٢١٠ما قصرت الصلاة وما نسيت٢١١وحدثني عن مالك، عن ابن شهاب، عن سعيد بن المسيب، وعن أبي سلمة بن عبد…٢١٢إذا شك أحدكم في صلاته فلم يدر كم صلى أثلاثا أم أربعا فليصلي ركعة…٢١٣أن عبد الله بن عمر، كان يقول إذا شك أحدكم في صلاته فليتوخ الذي يظن…٢١٤يشك في صلاته فلا يدري كم صلى أثلاثا أم أربعا فكلاهما قال ليصلي ركعة…٢١٥أن عبد الله بن عمر، كان إذا سئل عن النسيان، في الصلاة قال ليتوخ أحدكم…٢١٦حدثني يحيى، عن مالك، عن ابن شهاب، عن الأعرج، عن عبد الله ابن بحينة،…٢١٧وحدثني عن مالك، عن يحيى بن سعيد، عن عبد الرحمن بن هرمز، عن عبد الله…٢١٨ردي هذه الخميصة إلى أبي جهم فإني نظرت إلى علمها في الصلاة فكاد يفتنني٢١٩إني نظرت إلى علمها في الصلاة٢٢٠أن أبا طلحة الأنصاري، كان يصلي في حائطه فطار دبسي فطفق يتردد يلتمس…٢٢١أن رجلا، من الأنصار كان يصلي في حائط له بالقف - واد من أودية المدينة…
