كتاب الرضاع

١٨ حديثًا
١٢٧٢أن عائشة أم المؤمنين، أخبرتها أن رسول الله صلى الله عليه وسلم كان…١٢٧٣إنه عمك فأذني له١٢٧٤أنها أخبرته أن أفلح أخا أبي القعيس جاء يستأذن عليها وهو عمها من…١٢٧٥أنه كان يقول ما كان في الحولين وإن كان مصة واحدة فهو يحرم .١٢٧٦أن عبد الله بن عباس، سئل عن رجل، كانت له امرأتان فأرضعت إحداهما غلاما…١٢٧٧أن عبد الله بن عمر، كان يقول لا رضاعة إلا لمن أرضع في الصغر ولا رضاعة…١٢٧٨أن سالم بن عبد الله بن عمر، أخبره أن عائشة أم المؤمنين أرسلت به وهو…١٢٧٩أن صفية بنت أبي عبيد، أخبرته أن حفصة أم المؤمنين أرسلت بعاصم بن عبد…١٢٨٠أنه أخبره أن عائشة زوج النبي صلى الله عليه وسلم كان يدخل عليها من…١٢٨١فقال سعيد كل ما كان في الحولين وإن كانت قطرة واحدة فهو يحرم وما كان…١٢٨٢يقول لا رضاعة إلا ما كان في المهد وإلا ما أنبت اللحم والدم .١٢٨٣وحدثني عن مالك، عن ابن شهاب، أنه كان يقول الرضاعة قليلها وكثيرها تحرم…١٢٨٤أرضعيه خمس رضعات فيحرم بلبنها١٢٨٥وحدثني عن مالك، عن عبد الله بن دينار، أنه قال جاء رجل إلى عبد الله بن…١٢٨٦أن رجلا، سأل أبا موسى الأشعري فقال إني مصصت عن امرأتي، من ثديها لبنا…١٢٨٧يحرم من الرضاعة ما يحرم من الولادة١٢٨٨لقد هممت أن أنهى عن الغيلة حتى ذكرت أن الروم وفارس يصنعون ذلك فلا يضر…١٢٨٩زوج النبي صلى الله عليه وسلم أنها قالت كان فيما أنزل من القرآن عشر…
