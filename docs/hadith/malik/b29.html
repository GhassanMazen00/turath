١١٨ حديثًا
١١٥٠أنه بلغه أن رجلا، قال لعبد الله بن عباس إني طلقت امرأتي مائة تطليقة…١١٥١أنه بلغه أن رجلا، جاء إلى عبد الله بن مسعود فقال إني طلقت امرأتي…١١٥٢أن عمر بن عبد العزيز، قال له البتة ما يقول الناس فيها قال أبو بكر…١١٥٣أن مروان بن الحكم، كان يقضي في الذي يطلق امرأته البتة أنها ثلاث…١١٥٤حدثني يحيى، عن مالك، أنه بلغه أنه كتب إلى عمر بن الخطاب من العراق أن…١١٥٥أنه بلغه أن علي بن أبي طالب، كان يقول في الرجل يقول لامرأته أنت على…١١٥٦أن عبد الله بن عمر، كان يقول في الخلية والبرية إنها ثلاث تطليقات كل…١١٥٧. أن رجلا، كانت تحته وليدة لقوم فقال لأهلها شأنكم بها . فرأى الناس…١١٥٨أنه سمع ابن شهاب، يقول في الرجل يقول لامرأته برئت مني وبرئت منك إنها…١١٥٩أنه بلغه أن رجلا، جاء إلى عبد الله بن عمر فقال يا أبا عبد الرحمن إني…١١٦٠أن عبد الله بن عمر، كان يقول إذا ملك الرجل امرأته أمرها فالقضاء ما…١١٦١أنه أخبره أنه، كان جالسا عند زيد بن ثابت فأتاه محمد بن أبي عتيق…١١٦٢أن رجلا، من ثقيف ملك امرأته أمرها فقالت أنت الطلاق فسكت ثم قالت أنت…١١٦٣حدثني يحيى، عن مالك، عن عبد الرحمن بن القاسم، عن أبيه، عن عائشة أم…١١٦٤أن عائشة، زوج النبي صلى الله عليه وسلم زوجت حفصة بنت عبد الرحمن…١١٦٥أنه بلغه أن عبد الله بن عمر، وأبا، هريرة سئلا عن الرجل، يملك امرأته…١١٦٦أنه قال إذا ملك الرجل امرأته أمرها فلم تفارقه وقرت عنده فليس ذلك…١١٦٧حدثني يحيى، عن مالك، عن جعفر بن محمد، عن أبيه، عن علي بن أبي طالب،…١١٦٨أنه كان يقول أيما رجل آلى من امرأته فإنه إذا مضت الأربعة الأشهر وقف…١١٦٩أن سعيد بن المسيب، وأبا، بكر بن عبد الرحمن كانا يقولان في الرجل يولي…١١٧٠أنه بلغه أن مروان بن الحكم، كان يقضي في الرجل إذا آلى من امرأته أنها…١١٧٢فقال هو نحو إيلاء الحر وهو عليه واجب وإيلاء العبد شهران .١١٧٣طلق امرأة إن هو تزوجها فقال القاسم بن محمد إن رجلا جعل امرأة عليه…١١٧٤أنه بلغه أن رجلا، سأل القاسم بن محمد وسليمان بن يسار عن رجل تظاهر من…١١٧٥وحدثني عن مالك، عن هشام بن عروة، عن أبيه، أنه قال في رجل تظاهر من…١١٧٧أنه سمع رجلا، يسأل عروة بن الزبير عن رجل، قال لامرأته كل امرأة أنكحها…١١٧٨فقال نحو ظهار الحر . قال مالك يريد أنه يقع عليه كما يقع على الحر .…١١٧٩الولاء لمن أعتق١١٨٠أنه كان يقول في الأمة تكون تحت العبد فتعتق إن الأمة لها الخيار ما لم…١١٨١أن مولاة، لبني عدي يقال لها زبراء أخبرته أنها كانت تحت عبد وهي أمة…١١٨٢وحدثني عن مالك، أنه بلغه عن سعيد بن المسيب، أنه قال أيما رجل تزوج…١١٨٣وحدثني عن مالك، عن ابن شهاب، أنه سمعه يقول إذا خير الرجل امرأته…١١٨٤حدثني يحيى، عن مالك، عن يحيى بن سعيد، عن عمرة بنت عبد الرحمن، أنها…١١٨٥وحدثني عن مالك، عن نافع، عن مولاة، لصفية بنت أبي عبيد . أنها اختلعت…١١٨٦أن ربيع بنت معوذ بن عفراء، جاءت هي وعمها إلى عبد الله بن عمر فأخبرته…١١٨٧أنه بلغه أن سعيد بن المسيب، وسليمان بن يسار، وابن، شهاب كانوا يقولون…١١٨٨قد أنزل فيك وفي صاحبتك فاذهب فأت بها١١٨٩أن رجلا، لاعن امرأته في زمان رسول الله صلى الله عليه وسلم وانتفل من…١١٩٠أنه بلغه أن عروة بن الزبير، كان يقول في ولد الملاعنة وولد الزنا أنه…١١٩٢حدثني يحيى، عن مالك، عن ابن شهاب، عن محمد بن عبد الرحمن بن ثوبان، عن…١١٩٣وحدثني عن مالك، عن يحيى بن سعيد، عن بكير بن عبد الله بن الأشج، عن…١١٩٤وحدثني عن مالك، عن يحيى بن سعيد، عن بكير بن عبد الله بن الأشج، أنه…١١٩٥أن عبد الرحمن بن عوف، طلق امرأته البتة وهو مريض فورثها عثمان بن عفان…١١٩٦أن عثمان بن عفان، ورث نساء ابن مكمل منه وكان طلقهن وهو مريض .١١٩٧أنه سمع ربيعة بن أبي عبد الرحمن، يقول بلغني أن امرأة، عبد الرحمن بن…١١٩٨وحدثني عن مالك، عن يحيى بن سعيد، عن محمد بن يحيى بن حبان، قال كانت…١١٩٩أنه سمع ابن شهاب، يقول إذا طلق الرجل امرأته ثلاثا وهو مريض فإنها ترثه…١٢٠٠أنه بلغه أن عبد الرحمن بن عوف، طلق امرأة له فمتع بوليدة .١٢٠١أنه كان يقول لكل مطلقة متعة إلا التي تطلق وقد فرض لها صداق ولم تمس…١٢٠٢وحدثني عن مالك، عن ابن شهاب، أنه قال لكل مطلقة متعة . قال مالك وبلغني…١٢٠٤أن نفيعا، مكاتبا كان لأم سلمة زوج النبي صلى الله عليه وسلم أو عبدا…١٢٠٥أن نفيعا، مكاتبا كان لأم سلمة زوج النبي صلى الله عليه وسلم طلق امرأة…١٢٠٦أن نفيعا، مكاتبا كان لأم سلمة زوج النبي صلى الله عليه وسلم استفتى زيد…١٢٠٧أن عبد الله بن عمر، كان يقول إذا طلق العبد امرأته تطليقتين فقد حرمت…١٢٠٨أن عبد الله بن عمر، كان يقول من أذن لعبده أن ينكح فالطلاق بيد العبد…١٢٠٩أن عمر بن الخطاب، قال أيما امرأة فقدت زوجها فلم تدر أين هو فإنها…١٢١٠قال مالك وبلغني أن عمر بن الخطاب قال في المرأة يطلقها زوجها وهو غائب…١٢١١مره فليراجعها ثم يمسكها حتى تطهر ثم تحيض ثم تطهر ثم إن شاء أمسك بعد…١٢١٢أنها انتقلت حفصة بنت عبد الرحمن بن أبي بكر الصديق حين دخلت في الدم من…١٢١٣وحدثني عن مالك، عن ابن شهاب، أنه قال سمعت أبا بكر بن عبد الرحمن، يقول…١٢١٤وزيد بن أسلم، عن سليمان بن يسار، أن الأحوص، هلك بالشام حين دخلت…١٢١٥وسالم بن عبد الله، وأبي، بكر بن عبد الرحمن وسليمان بن يسار وابن شهاب…١٢١٦وحدثني عن مالك، عن نافع، عن عبد الله بن عمر، أنه كان يقول إذا طلق…١٢١٧مولى المهري أن القاسم بن محمد، وسالم بن عبد الله، كانا يقولان إذا…١٢١٨وابن، شهاب وسليمان بن يسار أنهم كانوا يقولون عدة المختلعة ثلاثة قروء .١٢١٩أنه سمع ابن شهاب، يقول عدة المطلقة الأقراء وإن تباعدت .١٢٢٠من الأنصار أن امرأته، سألته الطلاق فقال لها إذا حضت فآذنيني . فلما…١٢٢١وسليمان بن يسار، أنه سمعهما يذكران، أن يحيى بن سعيد بن العاص، طلق…١٢٢٢أن بنت سعيد بن زيد بن عمرو بن نفيل، كانت تحت عبد الله بن عمرو بن…١٢٢٣أن عبد الله بن عمر، طلق امرأة له في مسكن حفصة زوج النبي صلى الله عليه…١٢٢٤أن سعيد بن المسيب، سئل عن المرأة، يطلقها زوجها وهي في بيت بكراء على…١٢٢٥ليس لك عليه نفقة١٢٢٦أنه سمع ابن شهاب، يقول المبتوتة لا تخرج من بيتها حتى تحل وليست لها…١٢٢٧أنه قال قال عمر بن الخطاب أيما امرأة طلقت فحاضت حيضة أو حيضتين ثم…١٢٢٨أنه كان يقول الطلاق للرجال والعدة للنساء .١٢٢٩وحدثني عن مالك، عن ابن شهاب، عن سعيد بن المسيب، أنه قال عدة المستحاضة…١٢٣٠أنه بلغه أن علي بن أبي طالب، قال في الحكمين اللذين قال الله تعالى {…١٢٣١أنه بلغه أن عمر بن الخطاب، وعبد الله بن عمر، وعبد الله بن مسعود،…١٢٣٢أنه بلغه أن عبد الله بن مسعود، كان يقول فيمن قال كل امرأة أنكحها فهي…١٢٣٣أنه كان يقول من تزوج امرأة فلم يستطع أن يمسها فإنه يضرب له أجل سنة…١٢٣٤أنه سأل ابن شهاب متى يضرب له الأجل أمن يوم يبني بها أم من يوم ترافعه…١٢٣٥أمسك منهن أربعا وفارق سائرهن١٢٣٦وحميد بن عبد الرحمن بن عوف، وعبيد الله بن عبد الله بن عتبة بن مسعود،…١٢٣٧أنه تزوج أم ولد لعبد الرحمن بن زيد بن الخطاب - قال - فدعاني عبد الله…١٢٣٨قرأ { يا أيها النبي إذا طلقتم النساء فطلقوهن } لقبل عدتهن . قال مالك…١٢٣٩أنه قال كان الرجل إذا طلق امرأته ثم ارتجعها قبل أن تنقضي عدتها كان…١٢٤٠أن الرجل، كان يطلق امرأته ثم يراجعها ولا حاجة له بها ولا يريد إمساكها…١٢٤١أنه بلغه أن سعيد بن المسيب، وسليمان بن يسار، سئلا عن طلاق السكران،…١٢٤٢أنه بلغه أن سعيد بن المسيب، كان يقول إذا لم يجد الرجل ما ينفق على…١٢٤٣قد حللت فانكحي من شئت١٢٤٤يتوفى عنها زوجها وهي حامل فقال عبد الله بن عمر إذا وضعت حملها فقد حلت…١٢٤٥قد حللت فانكحي من شئت١٢٤٦قد حللت فانكحي من شئت١٢٤٧زينب بنت كعب بن عجرة أن الفريعة بنت مالك بن سنان، - وهي أخت أبي سعيد…١٢٤٨. أن عمر بن الخطاب، كان يرد المتوفى عنهن أزواجهن من البيداء يمنعهن…١٢٤٩أنه بلغه أن السائب بن خباب، توفي وإن امرأته جاءت إلى عبد الله بن عمر…١٢٥٠وحدثني عن مالك، عن هشام بن عروة، عن أبيه، أنه كان يقول في المرأة…١٢٥١أنه كان يقول لا تبيت المتوفى عنها زوجها ولا المبتوتة إلا في بيتها .١٢٥٢يقول إن يزيد بن عبد الملك فرق بين رجال وبين نسائهم وكن أمهات أولاد…١٢٥٣أنه قال عدة أم الولد إذا توفي عنها سيدها حيضة .١٢٥٤وحدثني مالك، عن يحيى بن سعيد، عن القاسم بن محمد، أنه كان يقول عدة أم…١٢٥٥أنه بلغه أن سعيد بن المسيب، وسليمان بن يسار، كانا يقولان عدة الأمة…١٢٥٦وحدثني عن مالك، عن ابن شهاب، مثل ذلك . قال مالك في العبد يطلق الأمة…١٢٥٧ما عليكم أن لا تفعلوا ما من نسمة كائنة إلى يوم القيامة إلا وهي كائنة١٢٥٨أنه كان يعزل .١٢٥٩لأبي أيوب الأنصاري أنه كان يعزل .١٢٦٠أنه كان لا يعزل وكان يكره العزل .١٢٦١أنه كان جالسا عند زيد بن ثابت فجاءه ابن قهد - رجل من أهل اليمن - فقال…١٢٦٢وحدثني عن مالك، عن حميد بن قيس المكي، عن رجل، يقال له ذفيف أنه قال…١٢٦٣لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد على ميت فوق ثلاث ليال…١٢٦٤لا يحل لامرأة تؤمن بالله واليوم الآخر تحد على ميت فوق ثلاث ليال إلا…١٢٦٥زوج النبي صلى الله عليه وسلم تقول جاءت امرأة إلى رسول الله صلى الله…١٢٦٦لا يحل لامرأة تؤمن١٢٦٧أنه بلغه أن أم سلمة، زوج النبي صلى الله عليه وسلم قالت لامرأة حاد على…١٢٦٨وسليمان بن يسار، أنهما كانا يقولان في المرأة يتوفى عنها زوجها إنها…١٢٦٩أن صفية بنت أبي عبيد، اشتكت عينيها وهي حاد على زوجها عبد الله بن عمر…١٢٧٠ما هذا يا أم سلمة١٢٧١أنه بلغه أن أم سلمة، زوج النبي صلى الله عليه وسلم كانت تقول تجمع…
