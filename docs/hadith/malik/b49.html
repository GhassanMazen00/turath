موطأ مالك

كتاب صفة النبى صلى الله عليه وسلم

٤٠ حديثًا
١٦٧١أنه سمعه يقول كان رسول الله صلى الله عليه وسلم ليس بالطويل البائن ولا…١٦٧٢أراني الليلة عند الكعبة فرأيت رجلا آدم كأحسن ما أنت راء من أدم الرجال…١٦٧٣قال خمس من الفطرة تقليم الأظافر وقص الشارب ونتف الإبط وحلق العانة…١٦٧٤وحدثني عن مالك، عن يحيى بن سعيد، عن سعيد بن المسيب، أنه قال كان…١٦٧٥وحدثني عن مالك، عن أبي الزبير، عن جابر بن عبد الله السلمي، أن رسول…١٦٧٦إذا أكل أحدكم فليأكل بيمينه وليشرب بيمينه فإن الشيطان يأكل بشماله…١٦٧٧ليس المسكين بهذا الطواف الذي يطوف على الناس فترده اللقمة واللقمتان…١٦٧٨ردوا المسكين ولو بظلف محرق١٦٧٩يأكل المسلم في معى واحد والكافر يأكل في سبعة أمعاء١٦٨٠المؤمن يشرب في معى واحد والكافر يشرب في سبعة أمعاء١٦٨١الذي يشرب في آنية الفضة إنما يجرجر في بطنه نار جهنم١٦٨٢فأبن القدح عن فيك ثم تنفس١٦٨٣أنه بلغه أن عمر بن الخطاب، وعلي بن أبي طالب، وعثمان بن عفان، كانوا…١٦٨٤أن عائشة أم المؤمنين، وسعد بن أبي وقاص، كانا لا يريان بشرب الإنسان…١٦٨٥أنه قال رأيت عبد الله بن عمر يشرب قائما .١٦٨٦أنه كان يشرب قائما .١٦٨٧حدثني عن مالك، عن ابن شهاب، عن أنس بن مالك، أن رسول الله صلى الله…١٦٨٨أتأذن لي أن أعطي هؤلاء١٦٨٩أنه سمع أنس بن مالك، يقول قال أبو طلحة لأم سليم لقد سمعت صوت، رسول…١٦٩٠طعام الاثنين كافي الثلاثة وطعام الثلاثة كافي الأربعة١٦٩١أغلقوا الباب وأوكوا السقاء وأكفئوا الإناء - أو خمروا الإناء - وأطفئوا…١٦٩٢من كان يؤمن بالله واليوم الآخر فليقل خيرا أو ليصمت ومن كان يؤمن بالله…١٦٩٣بينما رجل يمشي بطريق إذ اشتد عليه العطش فوجد بئرا فنزل فيها فشرب وخرج…١٦٩٤وحدثني عن مالك، عن وهب بن كيسان، عن جابر بن عبد الله، أنه قال بعث…١٦٩٥يا نساء المؤمنات لا تحقرن إحداكن لجارتها ولو كراع شاة محرقا١٦٩٦قاتل الله اليهود نهوا عن أكل الشحم فباعوه فأكلوا ثمنه١٦٩٧أنه بلغه أن عيسى ابن مريم، كان يقول يا بني إسرائيل عليكم بالماء…١٦٩٨وأنا أخرجني الجوع١٦٩٩أن عمر بن الخطاب، كان يأكل خبزا بسمن فدعا رجلا من أهل البادية فجعل…١٧٠٠أنه قال رأيت عمر بن الخطاب - وهو يومئذ أمير المؤمنين - يطرح له صاع من…١٧٠١فقال وددت أن عندي، قفعة نأكل منه .١٧٠٢وحدثني عن مالك، عن محمد بن عمرو بن حلحلة، عن حميد بن مالك بن خثيم،…١٧٠٣سم الله وكل مما يليك١٧٠٤يقول جاء رجل إلى عبد الله بن عباس فقال له إن لي يتيما وله إبل أفأشرب…١٧٠٥وحدثني عن مالك، عن هشام بن عروة، عن أبيه، أنه كان لا يؤتى أبدا بطعام…١٧٠٦أن عمر بن الخطاب، قال إياكم واللحم فإن له ضراوة كضراوة الخمر .١٧٠٧أن عمر بن الخطاب، أدرك جابر بن عبد الله ومعه حمال لحم فقال ما هذا…١٧٠٨أن رسول الله صلى الله عليه وسلم كان يلبس خاتما من ذهب ثم قام رسول…١٧٠٩فقال البسه وأخبر الناس، أني أفتيتك بذلك .١٧١٠لا تبقين في رقبة بعير قلادة من وتر أو قلادة إلا قطعت
