كتاب الكلام

٢٨ حديثًا
١٨١١من قال لأخيه يا كافر . فقد باء بها أحدهما١٨١٢إذا سمعت الرجل يقول هلك الناس . فهو أهلكهم١٨١٣لا يقل أحدكم يا خيبة الدهر . فإن الله هو الدهر١٨١٤أن عيسى ابن مريم، لقي خنزيرا بالطريق فقال له انفذ بسلام . فقيل له…١٨١٥إن الرجل ليتكلم بالكلمة من رضوان الله ما كان يظن أن تبلغ ما بلغت يكتب…١٨١٦أنه أخبره أن أبا هريرة قال إن الرجل ليتكلم بالكلمة ما يلقي لها بالا…١٨١٧إن من البيان لسحرا١٨١٨أنه بلغه أن عيسى ابن مريم، كان يقول لا تكثروا الكلام بغير ذكر الله…١٨١٩أنه بلغه أن عائشة، زوج النبي صلى الله عليه وسلم كانت ترسل إلى بعض…١٨٢٠أن تذكر من المرء ما يكره أن يسمع١٨٢١من وقاه الله شر اثنين ولج الجنة١٨٢٢أن عمر بن الخطاب، دخل على أبي بكر الصديق وهو يجبذ لسانه فقال له عمر…١٨٢٣لا يتناجى اثنان دون واحد١٨٢٤إذا كان ثلاثة فلا يتناجى اثنان دون واحد١٨٢٥أن رجلا، قال لرسول الله صلى الله عليه وسلم أكذب امرأتي يا رسول الله…١٨٢٦أنه بلغه أن عبد الله بن مسعود، كان يقول عليكم بالصدق فإن الصدق يهدي…١٨٢٧أنه بلغه أنه قيل للقمان ما بلغ بك ما نرى يريدون الفضل . فقال لقمان…١٨٢٨أنه بلغه أن عبد الله بن مسعود، كان يقول لا يزال العبد يكذب وتنكت في…١٨٢٩وحدثني مالك، عن صفوان بن سليم، أنه قال قيل لرسول الله صلى الله عليه…١٨٣٠إن الله يرضى لكم ثلاثا ويسخط لكم ثلاثا يرضى لكم أن تعبدوه ولا تشركوا…١٨٣١من شر الناس ذو الوجهين الذي يأتي هؤلاء بوجه وهؤلاء بوجه١٨٣٢نعم إذا كثر الخبث١٨٣٣أنه سمع عمر بن عبد العزيز، يقول كان يقال إن الله تبارك وتعالى لا يعذب…١٨٣٤وخرجت، معه حتى دخل حائطا فسمعته وهو، يقول وبيني وبينه جدار - وهو في…١٨٣٥قال مالك وبلغني أن القاسم بن محمد كان يقول أدركت الناس وما يعجبون…١٨٣٦أنه كان إذا سمع الرعد، ترك الحديث١٨٣٧لا نورث ما تركنا فهو صدقة١٨٣٨لا يقتسم ورثتي دنانير ما تركت بعد نفقة نسائي ومؤنة عاملي فهو صدقة
