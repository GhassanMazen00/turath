كتاب حسن الخلق

١٨ حديثًا
١٦٣٣أحسن خلقك للناس يا معاذ بن جبل١٦٣٤زوج النبي صلى الله عليه وسلم أنها قالت ما خير رسول الله صلى الله عليه…١٦٣٥من حسن إسلام المرء تركه ما لا يعنيه١٦٣٦بئس ابن العشيرة١٦٣٧أنه قال إذا أحببتم أن تعلموا ما للعبد عند ربه فانظروا ماذا يتبعه من…١٦٣٨أنه قال بلغني أن المرء، ليدرك بحسن خلقه درجة القائم بالليل الظامي…١٦٣٩يقول ألا أخبركم بخير، من كثير من الصلاة والصدقة قالوا بلى . قال إصلاح…١٦٤٠بعثت لأتمم حسن الأخلاق١٦٤١لكل دين خلق وخلق الإسلام الحياء١٦٤٢دعه فإن الحياء من الإيمان١٦٤٣أن رجلا، أتى إلى رسول الله صلى الله عليه وسلم فقال يا رسول الله علمني…١٦٤٤ليس الشديد بالصرعة إنما الشديد الذي يملك نفسه عند الغضب١٦٤٥لا يحل لمسلم أن يهاجر أخاه فوق ثلاث ليال يلتقيان فيعرض هذا ويعرض هذا…١٦٤٦لا تباغضوا ولا تحاسدوا ولا تدابروا وكونوا عباد الله إخوانا ولا يحل…١٦٤٧إياكم والظن فإن الظن أكذب الحديث ولا تجسسوا ولا تحسسوا ولا تنافسوا…١٦٤٨تصافحوا يذهب الغل وتهادوا تحابوا وتذهب الشحناء١٦٤٩تفتح أبواب الجنة يوم الاثنين ويوم الخميس فيغفر لكل عبد مسلم لا يشرك…١٦٥٠أنه قال تعرض أعمال الناس كل جمعة مرتين يوم الاثنين ويوم الخميس فيغفر…
