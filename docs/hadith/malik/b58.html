موطأ مالك

كتاب الصدقة

١٥ حديثًا
١٨٤١من تصدق بصدقة من كسب طيب - ولا يقبل الله إلا طيبا - كان إنما يضعها في…١٨٤٢بخ ذلك مال رابح ذلك مال رابح وقد سمعت ما قلت فيه وإني أرى أن تجعلها…١٨٤٣أعطوا السائل وإن جاء على فرس١٨٤٤يا نساء المؤمنات لا تحقرن إحداكن أن تهدي لجارتها ولو كراع شاة محرقا١٨٤٥وحدثني عن مالك، أنه بلغه عن عائشة، زوج النبي صلى الله عليه وسلم أن…١٨٤٦قال بلغني أن مسكينا، استطعم عائشة أم المؤمنين وبين يديها عنب فقالت…١٨٤٧ما يكون عندي من خير فلن أدخره عنكم ومن يستعفف يعفه الله ومن يستغن…١٨٤٨اليد العليا خير من اليد السفلى واليد العليا هي المنفقة والسفلى هي…١٨٤٩وحدثني عن مالك، عن زيد بن أسلم، عن عطاء بن يسار، أن رسول الله صلى…١٨٥٠والذي نفسي بيده لأن يأخذ أحدكم حبله فيحتطب على ظهره خير له من أن يأتي…١٨٥١وحدثني عن مالك، عن زيد بن أسلم، عن عطاء بن يسار، عن رجل، من بني أسد…١٨٥٢وعن مالك، عن العلاء بن عبد الرحمن، أنه سمعه يقول ما نقصت صدقة من مال…١٨٥٣لا تحل الصدقة لآل محمد إنما هي أوساخ الناس١٨٥٤إن الرجل ليسألني ما لا يصلح لي ولا له فإن منعته كرهت المنع وإن أعطيته…١٨٥٥وحدثني عن مالك، عن زيد بن أسلم، عن أبيه، أنه قال قال عبد الله بن…
