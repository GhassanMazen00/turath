موطأ مالك

كتاب الاستئذان

٤٥ حديثًا
١٧٦٣أن رسول الله صلى الله عليه وسلم سأله رجل فقال يا رسول الله أستأذن على…١٧٦٤الاستئذان ثلاث فإن أذن لك فادخل وإلا فارجع١٧٦٥الاستئذان ثلاث فإن أذن لك فادخل وإلا فارجع١٧٦٦إن عطس فشمته ثم إن عطس فشمته ثم إن عطس فشمته ثم إن عطس فقل إنك مضنوك١٧٦٧أن عبد الله بن عمر، كان إذا عطس فقيل له يرحمك الله . قال يرحمنا الله…١٧٦٨أن الملائكة لا تدخل بيتا فيه تماثيل أو تصاوير١٧٦٩إلا ما كان رقما في ثوب١٧٧٠فما بال هذه النمرقة١٧٧١حدثني مالك، عن عبد الرحمن بن عبد الله بن عبد الرحمن بن أبي صعصعة، عن…١٧٧٢لا ولكنه لم يكن بأرض قومي فأجدني أعافه١٧٧٣لست بآكله ولا بمحرمه١٧٧٤من اقتنى كلبا لا يغني عنه زرعا ولا ضرعا نقص من عمله كل يوم قيراط١٧٧٥من اقتنى كلبا إلا كلبا ضاريا أو كلب ماشية نقص من عمله كل يوم قيراطان١٧٧٦أن رسول الله صلى الله عليه وسلم أمر بقتل الكلاب .١٧٧٧رأس الكفر نحو المشرق والفخر والخيلاء في أهل الخيل والإبل والفدادين…١٧٧٨يوشك أن يكون خير مال المسلم غنما يتبع بها شعف الجبال ومواقع القطر يفر…١٧٧٩لا يحتلبن أحد ماشية أحد بغير إذنه أيحب أحدكم أن تؤتى مشربته فتكسر…١٧٨٠ما من نبي إلا قد رعى غنما١٧٨١أن ابن عمر، كان يقرب إليه عشاؤه فيسمع قراءة الإمام وهو في بيته فلا…١٧٨٢انزعوها وما حولها فاطرحوه١٧٨٣إن كان ففي الفرس والمرأة والمسكن١٧٨٤الشؤم في الدار والمرأة والفرس١٧٨٥أنه قال جاءت امرأة إلى رسول الله صلى الله عليه وسلم فقالت يا رسول…١٧٨٦أن رسول الله قال للقحة تحلب " من يحلب هذه " . فقام رجل فقال له رسول…١٧٨٧أن عمر بن الخطاب، قال لرجل ما اسمك فقال جمرة . فقال ابن من فقال ابن…١٧٨٨أنه قال احتجم رسول الله صلى الله عليه وسلم حجمه أبو طيبة فأمر له رسول…١٧٨٩إن كان دواء يبلغ الداء فإن الحجامة تبلغه١٧٩٠أحد بني حارثة أنه استأذن رسول الله صلى الله عليه وسلم في إجارة الحجام…١٧٩١ها إن الفتنة ها هنا إن الفتنة ها هنا من حيث يطلع قرن الشيطان١٧٩٢أنه بلغه أن عمر بن الخطاب، أراد الخروج إلى العراق فقال له كعب الأحبار…١٧٩٣حدثني مالك، عن نافع، عن أبي لبابة، أن رسول الله صلى الله عليه وسلم…١٧٩٤وحدثني مالك، عن نافع، عن سائبة، مولاة لعائشة . أن رسول الله صلى الله…١٧٩٥خذ عليك سلاحك فإني أخشى عليك بني قريظة١٧٩٦باسم الله . اللهم أنت الصاحب في السفر والخليفة في الأهل اللهم ازو لنا…١٧٩٧من نزل منزلا فليقل أعوذ بكلمات الله التامات من شر ما خلق . فإنه لن…١٧٩٨الراكب شيطان والراكبان شيطانان والثلاثة ركب١٧٩٩الشيطان يهم بالواحد والاثنين فإذا كانوا ثلاثة لم يهم بهم١٨٠٠لا يحل لامرأة تؤمن بالله واليوم الآخر تسافر مسيرة يوم وليلة إلا مع ذي…١٨٠١إن الله تبارك وتعالى رفيق يحب الرفق ويرضى به ويعين عليه ما لا يعين…١٨٠٢السفر قطعة من العذاب يمنع أحدكم نومه وطعامه وشرابه فإذا قضى أحدكم…١٨٠٣للمملوك طعامه وكسوته بالمعروف ولا يكلف من العمل إلا ما يطيق١٨٠٤أنه بلغه أن عمر بن الخطاب، كان يذهب إلى العوالي كل يوم سبت فإذا وجد…١٨٠٥أنه سمع عثمان بن عفان، وهو يخطب وهو يقول لا تكلفوا الأمة غير ذات…١٨٠٦العبد إذا نصح لسيده وأحسن عبادة الله فله أجره مرتين١٨٠٧أنه بلغه أن أمة، كانت لعبد الله بن عمر بن الخطاب رآها عمر بن الخطاب…
