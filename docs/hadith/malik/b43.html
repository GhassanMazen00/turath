٤٣ حديثًا
١٥٥٢أن في الكتاب الذي، كتبه رسول الله صلى الله عليه وسلم لعمرو بن حزم في…١٥٥٣أنه بلغه أن عمر بن الخطاب، قوم الدية على أهل القرى فجعلها على أهل…١٥٥٤أن ابن شهاب، كان يقول في دية العمد إذا قبلت خمس وعشرون بنت مخاض وخمس…١٥٥٥أن مروان بن الحكم، كتب إلى معاوية بن أبي سفيان أنه أتي بمجنون قتل…١٥٥٦وسليمان بن يسار، أن رجلا، من بني سعد بن ليث أجرى فرسا فوطئ على إصبع…١٥٥٧أن ابن شهاب، وسليمان بن يسار، وربيعة بن أبي عبد الرحمن، كانوا يقولون…١٥٥٨أنه كان يقول تعاقل المرأة الرجل إلى ثلث الدية إصبعها كإصبعه وسنها…١٥٥٩وبلغه، عن عروة بن الزبير، أنهما كانا يقولان مثل قول سعيد بن المسيب في…١٥٦٠أنه سمع ابن شهاب، يقول مضت السنة أن الرجل، إذا أصاب امرأته بجرح أن…١٥٦١أن امرأتين، من هذيل رمت إحداهما الأخرى فطرحت جنينها فقضى فيه رسول…١٥٦٢إنما هذا من إخوان الكهان١٥٦٣أنه كان يقول الغرة تقوم خمسين دينارا أو ستمائة درهم ودية المرأة الحرة…١٥٦٤أنه كان يقول في الشفتين الدية كاملة فإذا قطعت السفلى ففيها ثلثا الدية…١٥٦٥يفقأ عين الصحيح فقال ابن شهاب إن أحب الصحيح أن يستقيد، منه فله القود…١٥٦٦أن زيد بن ثابت، كان يقول في العين القائمة إذا طفئت مائة دينار . قال…١٥٦٧أنه سمع سليمان بن يسار، يذكر أن الموضحة، في الوجه مثل الموضحة في…١٥٦٨أنه قال كل نافذة في عضو من الأعضاء ففيها ثلث عقل ذلك العضو .١٥٦٩حدثني مالك، كان ابن شهاب لا يرى ذلك وأنا لا، أرى في نافذة في عضو من…١٥٧٠أن عبد الله بن الزبير، أقاد من المنقلة .١٥٧١وحدثني يحيى، عن مالك، عن ربيعة بن أبي عبد الرحمن، أنه قال سألت سعيد…١٥٧٢مولى عمر بن الخطاب أن عمر بن الخطاب، قضى في الضرس بجمل وفي الترقوة…١٥٧٣أنه سمع سعيد بن المسيب، يقول قضى عمر بن الخطاب في الأضراس ببعير بعير…١٥٧٤أنه كان يقول إذا أصيبت السن فاسودت ففيها عقلها تاما فإن طرحت بعد أن…١٥٧٥أنه أخبره أن مروان بن الحكم بعثه إلى عبد الله بن عباس يسأله ماذا في…١٥٧٦في السن خمس من الإبل١٥٧٧أنه بلغه أن سعيد بن المسيب، وسليمان بن يسار، كانا يقولان في موضحة…١٥٧٨أنه بلغه أن مروان بن الحكم، كان يقضي في العبد يصاب بالجراح أن على من…١٥٧٩أنه بلغه أن عمر بن عبد العزيز، قضى أن دية اليهودي، أو النصراني - إذا…١٥٨٠أن سليمان بن يسار، كان يقول دية المجوسي ثماني مائة درهم . قال مالك…١٥٨١أنه كان يقول ليس على العاقلة عقل في قتل العمد إنما عليهم عقل قتل…١٥٨٢مثل ذلك١٥٨٤قال مالك إن ابن شهاب قال مضت السنة في قتل العمد حين يعفو أولياء…١٥٨٥أن عمر بن الخطاب، نشد الناس بمنى من كان عنده علم من الدية أن يخبرني .…١٥٨٦أن رجلا، من بني مدلج - يقال له قتادة - حذف ابنه بالسيف فأصاب ساقه…١٥٨٧أنه بلغه أن سعيد بن المسيب، وسليمان بن يسار، سئلا أتغلظ الدية في…١٥٨٨أن رجلا، من الأنصار - يقال له أحيحة بن الجلاح - كان له عم صغير هو…١٥٨٩جرح العجماء جبار والبئر جبار والمعدن جبار وفي الركاز الخمس١٥٩٠أن عمر بن الخطاب، قتل نفرا خمسة أو سبعة برجل واحد قتلوه قتل غيلة وقال…١٥٩١أنه بلغه أن حفصة، زوج النبي صلى الله عليه وسلم قتلت جارية لها سحرتها…١٥٩٢مولى عائشة بنت قدامة أن عبد الملك بن مروان، أقاد ولي رجل من رجل قتله…١٥٩٣أنه بلغه أن مروان بن الحكم، كتب إلى معاوية بن أبي سفيان يذكر أنه أتي…١٥٩٤أنه بلغه أن أبا بكر بن محمد بن عمرو بن حزم، أقاد من كسر الفخذ .١٥٩٥أن سائبة، أعتقه بعض الحجاج فقتل ابن رجل من بني عائذ فجاء العائذي أبو…
