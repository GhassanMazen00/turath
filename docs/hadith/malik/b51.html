كتاب الشعر

١٧ حديثًا
١٧٣٠أن رسول الله صلى الله عليه وسلم أمر بإحفاء الشوارب وإعفاء اللحى .١٧٣١إنما هلكت بنو إسرائيل حين اتخذ هذه نساؤهم١٧٣٢أنه سمعه يقول سدل رسول الله صلى الله عليه وسلم ناصيته ما شاء الله ثم…١٧٣٣أنه كان يكره الإخصاء ويقول فيه تمام الخلق .١٧٣٤أنا وكافل اليتيم له أو لغيره في الجنة كهاتين إذا اتقى١٧٣٥أن أبا قتادة الأنصاري، قال لرسول الله صلى الله عليه وسلم إن لي جمة…١٧٣٦أليس هذا خيرا من أن يأتي أحدكم ثائر الرأس كأنه شيطان١٧٣٧أن عبد الرحمن بن الأسود بن عبد يغوث، قال وكان جليسا لهم وكان أبيض…١٧٣٨قل أعوذ بكلمات الله التامة من غضبه وعقابه وشر عباده ومن همزات…١٧٣٩أنه قال أسري برسول الله صلى الله عليه وسلم فرأى عفريتا من الجن يطلبه…١٧٤٠أن رجلا، من أسلم قال ما نمت هذه الليلة . فقال له رسول الله صلى الله…١٧٤١أن كعب الأحبار، قال لولا كلمات أقولهن لجعلتني يهود حمارا . فقيل له…١٧٤٢إن الله تبارك وتعالى يقول يوم القيامة أين المتحابون لجلالي اليوم…١٧٤٣سبعة يظلهم الله في ظله يوم لا ظل إلا ظله إمام عادل وشاب نشأ في عبادة…١٧٤٤إذا أحب الله العبد قال لجبريل قد أحببت فلانا فأحبه . فيحبه جبريل ثم…١٧٤٥قال الله تبارك وتعالى وجبت محبتي للمتحابين في والمتجالسين في…١٧٤٦أنه كان يقول القصد والتؤدة وحسن السمت جزء من خمسة وعشرين جزءا من…
