٣٥ حديثًا
١٥٠٠ما تجدون في التوراة في شأن الرجم١٥٠١أيشتكي أم به جنة١٥٠٢يا هزال لو سترته بردائك لكان خيرا لك١٥٠٣حدثني مالك، عن ابن شهاب، أنه أخبره . أن رجلا اعترف على نفسه بالزنا…١٥٠٤أنه أخبره أن امرأة جاءت إلى رسول الله صلى الله عليه وسلم فأخبرته أنها…١٥٠٥وزيد بن خالد الجهني، أنهما أخبراه أن رجلين اختصما إلى رسول الله صلى…١٥٠٦أن سعد بن عبادة، قال لرسول الله صلى الله عليه وسلم أرأيت لو أني وجدت…١٥٠٧يقول الرجم في كتاب الله حق على من زنى من الرجال والنساء إذا أحصن إذا…١٥٠٨أن عمر بن الخطاب، أتاه رجل وهو بالشام فذكر له أنه وجد مع امرأته رجلا…١٥٠٩حدثني مالك، عن يحيى بن سعيد، عن سعيد بن المسيب، أنه سمعه يقول لما صدر…١٥١٠أنه بلغه أن عثمان بن عفان، أتي بامرأة قد ولدت في ستة أشهر فأمر بها أن…١٥١١أن رجلا، اعترف على نفسه بالزنا على عهد رسول الله صلى الله عليه وسلم…١٥١٢أن صفية بنت أبي عبيد، أخبرته أن أبا بكر الصديق أتي برجل قد وقع على…١٥١٣إن زنت فاجلدوها ثم إن زنت فاجلدوها ثم إن زنت فاجلدوها ثم بيعوها ولو…١٥١٤أن عبدا، كان يقوم على رقيق الخمس وأنه استكره جارية من ذلك الرقيق فوقع…١٥١٥أن سليمان بن يسار، أخبره أن عبد الله بن عياش بن أبي ربيعة المخزومي…١٥١٦حدثني مالك، عن أبي الزناد، أنه قال جلد عمر بن عبد العزيز عبدا في فرية…١٥١٧أن رجلا، يقال له مصباح استعان ابنا له فكأنه استبطأه فلما جاءه قال له…١٥١٨عمرة بنت عبد الرحمن أن رجلين، استبا في زمان عمر بن الخطاب فقال أحدهما…١٥٢٠أن عمر بن الخطاب، قال لرجل خرج بجارية لامرأته معه في سفر فأصابها…١٥٢١أن رسول الله صلى الله عليه وسلم قطع في مجن ثمنه ثلاثة دراهم .١٥٢٢لا قطع في ثمر معلق ولا في حريسة جبل١٥٢٣أن سارقا، سرق في زمان عثمان أترجة فأمر بها عثمان بن عفان أن تقوم…١٥٢٤القطع في ربع دينار فصاعدا١٥٢٥وحدثني عن مالك، عن عبد الله بن أبي بكر بن حزم، عن عمرة بنت عبد…١٥٢٦أن عبدا، لعبد الله بن عمر سرق وهو آبق فأرسل به عبد الله بن عمر إلى…١٥٢٧أنه أخبره أنه، أخذ عبدا آبقا قد سرق قال فأشكل على أمره - قال - فكتبت…١٥٢٩أن صفوان بن أمية، قيل له إنه من لم يهاجر هلك . فقدم صفوان بن أمية…١٥٣٠أن الزبير بن العوام، لقي رجلا قد أخذ سارقا وهو يريد أن يذهب به إلى…١٥٣١أن رجلا، من أهل اليمن أقطع اليد والرجل قدم فنزل على أبي بكر الصديق…١٥٣٢أن أبا الزناد، أخبره أن عاملا لعمر بن عبد العزيز أخذ ناسا في حرابة…١٥٣٣لا قطع في ثمر ولا كثر١٥٣٤أن عبد الله بن عمرو بن الحضرمي، جاء بغلام له إلى عمر بن الخطاب فقال…١٥٣٥أن مروان بن الحكم، أتي بإنسان قد اختلس متاعا فأراد قطع يده فأرسل إلى…١٥٣٦وحدثني عن مالك، عن يحيى بن سعيد، أنه قال أخبرني أبو بكر بن محمد بن…
