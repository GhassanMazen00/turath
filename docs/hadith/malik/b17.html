موطأ مالك

كتاب الزكاة

٥٢ حديثًا
٥٧٨ليس فيما دون خمس ذود صدقة، وليس فيما دون خمس أواق صدقة، وليس فيما دون…٥٧٩ليس فيما دون خمسة أوسق من التمر صدقة، وليس فيما دون خمس أواقي من…٥٨٠أنه بلغه : أن عمر بن عبد العزيز، كتب إلى عامله على دمشق في الصدقة :…٥٨١له قاطعه بمال عظيم هل عليه فيه زكاة فقال القاسم : إن أبا بكر الصديق…٥٨٢أنه قال : كنت إذا جئت عثمان بن عفان أقبض عطائي سألني : هل عندك من مال…٥٨٣أن عبد الله بن عمر، كان يقول : لا تجب في مال زكاة حتى يحول عليه الحول…٥٨٤فإن زادت حتى تبلغ بزيادتها عشرين دينارا وازنة ففيها الزكاة، وليس فيما…٥٨٥واحد، أن رسول الله صلى الله عليه وسلم قطع لبلال بن الحارث المزني…٥٨٦أن رسول الله صلى الله عليه وسلم قال " في الركاز الخمس " .٥٨٧أن عائشة، زوج النبي صلى الله عليه وسلم كانت تلي بنات أخيها يتامى في…٥٨٨أن عبد الله بن عمر، كان يحلي بناته وجواريه الذهب ثم لا يخرج من حليهن…٥٨٩أنه بلغه أن عمر بن الخطاب، قال اتجروا في أموال اليتامى لا تأكلها…٥٩٠وحدثني عن مالك، عن عبد الرحمن بن القاسم، عن أبيه، أنه قال كانت عائشة…٥٩١أنه بلغه أن عائشة، زوج النبي صلى الله عليه وسلم كانت تعطي أموال…٥٩٢وحدثني عن مالك، عن يحيى بن سعيد، أنه اشترى لبني أخيه - يتامى في حجره…٥٩٣أن عثمان بن عفان، كان يقول هذا شهر زكاتكم فمن كان عليه دين فليؤد دينه…٥٩٤أن عمر بن عبد العزيز، كتب في مال قبضه بعض الولاة ظلما يأمر برده إلى…٥٩٥وحدثني عن مالك، عن يزيد بن خصيفة، أنه سأل سليمان بن يسار عن رجل، له…٥٩٦حدثني عن مالك، عن يحيى بن سعيد، عن زريق بن حيان، - وكان زريق على جواز…٥٩٧ما هو فقال هو المال الذي لا تؤدى منه الزكاة .٥٩٨وحدثني عن مالك، عن عبد الله بن دينار، عن أبي صالح السمان، عن أبي…٥٩٩أنه قرأ كتاب عمر بن الخطاب في الصدقة قال فوجدت فيه بسم الله الرحمن…٦٠٠أن معاذ بن جبل الأنصاري، أخذ من ثلاثين بقرة تبيعا ومن أربعين بقرة…٦٠١سفيان بن عبد الله أن عمر بن الخطاب، بعثه مصدقا فكان يعد على الناس…٦٠٢حدثني يحيى، عن مالك، عن يحيى بن سعيد، عن محمد بن يحيى بن حبان، عن…٦٠٣من أشجع أن محمد بن مسلمة الأنصاري، كان يأتيهم مصدقا فيقول لرب المال…٦٠٤لا تحل الصدقة لغني إلا لخمسة لغاز في سبيل الله أو لعامل عليها أو…٦٠٥أنه بلغه أن أبا بكر الصديق، قال لو منعوني عقالا لجاهدتهم عليه .٦٠٦وحدثني عن مالك، عن زيد بن أسلم، أنه قال شرب عمر بن الخطاب لبنا فأعجبه…٦٠٧أنه بلغه أن عاملا، لعمر بن عبد العزيز كتب إليه يذكر أن رجلا منع زكاة…٦٠٨فيما سقت السماء والعيون والبعل العشر وفيما سقي بالنضح نصف العشر٦٠٩وحدثني عن مالك، عن زياد بن سعد، عن ابن شهاب، أنه قال لا يؤخذ في صدقة…٦١٠حدثني يحيى، عن مالك، أنه سأل ابن شهاب عن الزيتون، فقال فيه العشر .…٦١١ليس على المسلم في عبده ولا في فرسه صدقة٦١٢أن أهل الشام، قالوا لأبي عبيدة بن الجراح خذ من خيلنا ورقيقنا صدقة .…٦١٣أنه قال جاء كتاب من عمر بن عبد العزيز إلى أبي وهو بمنى أن لا يأخذ من…٦١٤فقال وهل في الخيل من صدقة٦١٥قال بلغني أن رسول الله صلى الله عليه وسلم أخذ الجزية من مجوس البحرين…٦١٦سنوا بهم سنة أهل الكتاب٦١٧مولى عمر بن الخطاب أن عمر بن الخطاب، ضرب الجزية على أهل الذهب أربعة…٦١٨أنه قال لعمر بن الخطاب إن في الظهر ناقة عمياء . فقال عمر ادفعها إلى…٦١٩أنه بلغه أن عمر بن عبد العزيز، كتب إلى عماله أن يضعوا الجزية عمن أسلم…٦٢٠أن عمر بن الخطاب، كان يأخذ من النبط من الحنطة والزيت نصف العشر يريد…٦٢١وحدثني عن مالك، عن ابن شهاب، عن السائب بن يزيد، أنه قال كنت غلاما…٦٢٢أنه سأل ابن شهاب على أى وجه كان يأخذ عمر بن الخطاب من النبط العشر…٦٢٣لا تشتره وإن أعطاكه بدرهم واحد فإن العائد في صدقته كالكلب يعود في قيئه٦٢٤لا تبتعه ولا تعد في صدقتك٦٢٥أن عبد الله بن عمر، كان يخرج زكاة الفطر عن غلمانه الذين، بوادي القرى…٦٢٦أن رسول الله صلى الله عليه وسلم فرض زكاة الفطر من رمضان على الناس…٦٢٧أنه سمع أبا سعيد الخدري، يقول كنا نخرج زكاة الفطر صاعا من طعام أو…٦٢٨أن عبد الله بن عمر، كان لا يخرج في زكاة الفطر إلا التمر إلا مرة واحدة…٦٢٩أن عبد الله بن عمر، كان يبعث بزكاة الفطر إلى الذي تجمع عنده قبل الفطر…
