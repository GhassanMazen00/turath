٥٩ حديثًا
١٠٩١لا يخطب أحدكم على خطبة أخيه١٠٩٢لا يخطب أحدكم على خطبة أخيه١٠٩٣أنه كان يقول في قول الله تبارك وتعالى { ولا جناح عليكم فيما عرضتم به…١٠٩٤الأيم أحق بنفسها من وليها والبكر تستأذن في نفسها وإذنها صماتها١٠٩٥أنه قال قال عمر بن الخطاب لا تنكح المرأة إلا بإذن وليها أو ذي الرأى…١٠٩٦أنه بلغه أن القاسم بن محمد، وسالم بن عبد الله، كانا ينكحان بناتهما…١٠٩٧أنه بلغه أن القاسم بن محمد، وسالم بن عبد الله، وسليمان بن يسار، كانوا…١٠٩٨هل عندك من شىء تصدقها إياه١٠٩٩أنه قال قال عمر بن الخطاب أيما رجل تزوج امرأة وبها جنون أو جذام أو…١١٠٠أن ابنة عبيد الله بن عمر، - وأمها بنت زيد بن الخطاب - كانت تحت ابن…١١٠١أنه بلغه أن عمر بن عبد العزيز، كتب في خلافته إلى بعض عماله أن كل ما…١١٠٢أن عمر بن الخطاب، قضى في المرأة إذا تزوجها الرجل أنه إذا أرخيت الستور…١١٠٣أن زيد بن ثابت، كان يقول إذا دخل الرجل بامرأته فأرخيت عليهما الستور…١١٠٤أنه بلغه أن سعيد بن المسيب، كان يقول إذا دخل الرجل بالمرأة في بيتها…١١٠٥ليس بك على أهلك هوان إن شئت سبعت عندك وسبعت عندهن وإن شئت ثلثت عندك…١١٠٦أنه كان يقول للبكر سبع وللثيب ثلاث .١١٠٧أنه بلغه أن سعيد بن المسيب، سئل عن المرأة، تشترط على زوجها أنه لا…١١٠٨لا تحل لك حتى تذوق العسيلة١١٠٩وحدثني عن مالك، عن يحيى بن سعيد، عن القاسم بن محمد، عن عائشة، زوج…١١١٠أنه بلغه أن القاسم بن محمد، سئل عن رجل، طلق امرأته البتة ثم تزوجها…١١١١لا يجمع بين المرأة وعمتها ولا بين المرأة وخالتها١١١٢أنه كان يقول ينهى أن تنكح المرأة على عمتها أو على خالتها وأن يطأ…١١١٣تزوج امرأة ثم فارقها قبل أن يصيبها هل تحل له أمها فقال زيد بن ثابت لا…١١١٤واحد، أن عبد الله بن مسعود، استفتي وهو بالكوفة عن نكاح الأم، بعد…١١١٥حدثني يحيى، عن مالك، عن نافع، عن عبد الله بن عمر، أن رسول الله صلى…١١١٦ومجمع، ابنى يزيد بن جارية الأنصاري عن خنساء بنت خدام الأنصارية، أن…١١١٧أن عمر بن الخطاب، أتي بنكاح لم يشهد عليه إلا رجل وامرأة فقال هذا نكاح…١١١٨أن طليحة الأسدية، كانت تحت رشيد الثقفي فطلقها فنكحت في عدتها فضربها…١١١٩أنه بلغه أن عبد الله بن عباس، وعبد الله بن عمر، سئلا عن رجل، كانت…١١٢٠وحدثني عن مالك، عن يحيى بن سعيد، عن سعيد بن المسيب، أنه كان يقول لا…١١٢١أنه كان يقول في الرجل يطلق الأمة ثلاثا ثم يشتريها إنها لا تحل له حتى…١١٢٢أنه بلغه أن سعيد بن المسيب، وسليمان بن يسار، سئلا عن رجل، زوج عبدا له…١١٢٣كانت تحته أمة مملوكة فاشتراها وقد كان طلقها واحدة فقال تحل له بملك…١١٢٤أن عمر بن الخطاب، سئل عن المرأة، وابنتها، من ملك اليمين توطأ إحداهما…١١٢٥أن رجلا، سأل عثمان بن عفان عن الأختين، من ملك اليمين هل يجمع بينهما…١١٢٦مثل ذلك . قال مالك في الأمة تكون عند الرجل فيصيبها ثم يريد أن يصيب…١١٢٧أنه بلغه أن عمر بن الخطاب، وهب لابنه جارية فقال لا تمسها فإني قد…١١٢٨أنه قال وهب سالم بن عبد الله لابنه جارية قال لا تقربها فإني قد أردتها…١١٢٩أن أبا نهشل بن الأسود، قال للقاسم بن محمد إني رأيت جارية لي منكشفا…١١٣٠أنه وهب لصاحب له جارية ثم سأله عنها فقال قد هممت أن أهبها لابني فيفعل…١١٣١حدثني يحيى، عن مالك، عن ابن شهاب، عن سعيد بن المسيب، أنه قال المحصنات…١١٣٢وبلغه، عن القاسم بن محمد، أنهما كانا يقولان إذا نكح الحر الأمة فمسها…١١٣٣والحسن، ابنى محمد بن علي بن أبي طالب عن أبيهما، عن علي بن أبي طالب،…١١٣٤أن خولة بنت حكيم، دخلت على عمر بن الخطاب فقالت إن ربيعة بن أمية…١١٣٥أنه سمع ربيعة بن أبي عبد الرحمن، يقول ينكح العبد أربع نسوة . قال مالك…١١٣٦أنه بلغه أن نساء، كن في عهد رسول الله صلى الله عليه وسلم يسلمن بأرضهن…١١٣٧وحدثني عن مالك، عن ابن شهاب، أنه قال كان بين إسلام صفوان وبين إسلام…١١٣٨أن أم حكيم بنت الحارث بن هشام، وكانت، تحت عكرمة بن أبي جهل فأسلمت يوم…١١٣٩أن عبد الرحمن بن عوف، جاء إلى رسول الله صلى الله عليه وسلم وبه أثر…١١٤٠أنه قال لقد بلغني أن رسول الله صلى الله عليه وسلم كان يولم بالوليمة…١١٤١إذا دعي أحدكم إلى وليمة فليأتها١١٤٢أنه كان يقول شر الطعام طعام الوليمة يدعى لها الأغنياء ويترك المساكين…١١٤٣أنه سمع أنس بن مالك، يقول إن خياطا دعا رسول الله صلى الله عليه وسلم…١١٤٤إذا تزوج أحدكم المرأة أو اشترى الجارية فليأخذ بناصيتها وليدع بالبركة…١١٤٥أن رجلا، خطب إلى رجل أخته فذكر أنها قد كانت أحدثت فبلغ ذلك عمر بن…١١٤٦أن القاسم بن محمد، وعروة بن الزبير، كانا يقولان في الرجل يكون عنده…١١٤٧أن القاسم بن محمد، وعروة بن الزبير، أفتيا الوليد بن عبد الملك عام قدم…١١٤٨أنه قال ثلاث ليس فيهن لعب النكاح والطلاق والعتق .١١٤٩أنه تزوج بنت محمد بن مسلمة الأنصاري فكانت عنده حتى كبرت فتزوج عليها…
