كتاب الصيام

٥٩ حديثًا
٦٣٠لا تصوموا حتى تروا الهلال ولا تفطروا حتى تروه فإن غم عليكم فاقدروا له٦٣١الشهر تسع وعشرون فلا تصوموا حتى تروا الهلال ولا تفطروا حتى تروه فإن…٦٣٢لا تصوموا حتى تروا الهلال ولا تفطروا حتى تروه فإن غم عليكم فأكملوا…٦٣٣أنه بلغه أن الهلال، رئي في زمان عثمان بن عفان بعشي فلم يفطر عثمان حتى…٦٣٤أنه كان يقول لا يصوم إلا من أجمع الصيام قبل الفجر .٦٣٥وحفصة، زوجى النبي صلى الله عليه وسلم بمثل ذلك .٦٣٦لا يزال الناس بخير ما عجلوا الفطر٦٣٧لا يزال الناس بخير ما عجلوا الفطر٦٣٨أن عمر بن الخطاب، وعثمان بن عفان، كانا يصليان المغرب حين ينظران إلى…٦٣٩وأنا أصبح جنبا وأنا أريد الصيام فأغتسل وأصوم٦٤٠وأم سلمة زوجى النبي صلى الله عليه وسلم أنهما قالتا كان رسول الله صلى…٦٤١مولى أبي بكر بن عبد الرحمن بن الحارث بن هشام أنه سمع أبا بكر بن عبد…٦٤٢وأم سلمة زوجى النبي صلى الله عليه وسلم أنهما قالتا إن كان رسول الله…٦٤٣أن رجلا، قبل امرأته وهو صائم في رمضان فوجد من ذلك وجدا شديدا فأرسل…٦٤٤- رضى الله عنها - أنها قالت إن كان رسول الله صلى الله عليه وسلم ليقبل…٦٤٥أن عاتكة ابنة زيد بن عمرو بن نفيل، - امرأة عمر بن الخطاب - كانت تقبل…٦٤٦مولى عمر بن عبيد الله أن عائشة بنت طلحة، أخبرته أنها، كانت عند عائشة…٦٤٧أن أبا هريرة، وسعد بن أبي وقاص، كانا يرخصان في القبلة للصائم .٦٤٨أنه بلغه أن عائشة، زوج النبي صلى الله عليه وسلم كانت إذا ذكرت أن رسول…٦٤٩قال يحيى قال مالك قال هشام بن عروة قال عروة بن الزبير لم أر القبلة…٦٥٠أن عبد الله بن عباس، سئل عن القبلة، للصائم فأرخص فيها للشيخ وكرهها…٦٥١أن عبد الله بن عمر، كان ينهى عن القبلة، والمباشرة، للصائم .٦٥٢أن رسول الله صلى الله عليه وسلم خرج إلى مكة عام الفتح في رمضان فصام…٦٥٣وحدثني عن مالك، عن سمى، مولى أبي بكر بن عبد الرحمن عن أبي بكر بن عبد…٦٥٤وحدثني عن مالك، عن حميد الطويل، عن أنس بن مالك، أنه قال سافرنا مع…٦٥٥إن شئت فصم وإن شئت فأفطر٦٥٦أن عبد الله بن عمر، كان لا يصوم في السفر .٦٥٧وحدثني عن مالك، عن هشام بن عروة، عن أبيه، أنه كان يسافر في رمضان…٦٥٨أنه بلغه أن عمر بن الخطاب، كان إذا كان في سفر في رمضان فعلم أنه داخل…٦٥٩خذ هذا فتصدق به٦٦٠وحدثني عن مالك، عن عطاء بن عبد الله الخراساني، عن سعيد بن المسيب، أنه…٦٦١أنه كان يحتجم وهو صائم - قال - ثم ترك ذلك بعد فكان إذا صام لم يحتجم…٦٦٢أن سعد بن أبي وقاص، وعبد الله بن عمر، كانا يحتجمان وهما صائمان .٦٦٣أنه كان يحتجم وهو صائم ثم لا يفطر . قال وما رأيته احتجم قط إلا وهو…٦٦٤زوج النبي صلى الله عليه وسلم أنها قالت كان يوم عاشوراء يوما تصومه…٦٦٥هذا يوم عاشوراء ولم يكتب عليكم صيامه وأنا صائم فمن شاء فليصم ومن شاء…٦٦٦أنه بلغه أن عمر بن الخطاب، أرسل إلى الحارث بن هشام أن غدا، يوم…٦٦٧حدثني يحيى، عن مالك، عن محمد بن يحيى بن حبان، عن الأعرج، عن أبي…٦٦٨أنه سمع أهل العلم، يقولون لا بأس بصيام الدهر إذا أفطر الأيام التي نهى…٦٦٩إني لست كهيئتكم إني أطعم وأسقى٦٧٠إياكم والوصال إياكم والوصال٦٧١حديث ٦٧١٦٧٢مثل ذلك . قال مالك من مات وعليه نذر من رقبة يعتقها أو صيام أو صدقة أو…٦٧٣أنه بلغه أن عبد الله بن عمر، كان يسأل هل يصوم أحد عن أحد، أو يصلي أحد…٦٧٤خالد بن أسلم أن عمر بن الخطاب، أفطر ذات يوم في رمضان في يوم ذي غيم…٦٧٥أن عبد الله بن عمر، كان يقول يصوم قضاء رمضان متتابعا من أفطره من مرض…٦٧٦أن عبد الله بن عباس، وأبا، هريرة اختلفا في قضاء رمضان فقال أحدهما…٦٧٧أنه كان يقول من استقاء وهو صائم فعليه القضاء ومن ذرعه القىء فليس عليه…٦٧٨أنه سمع سعيد بن المسيب، يسأل عن قضاء، رمضان فقال سعيد أحب إلى أن لا…٦٧٩وحدثني عن مالك، عن حميد بن قيس المكي، أنه أخبره قال كنت مع مجاهد وهو…٦٨٠اقضيا مكانه يوما آخر٦٨١أنه بلغه أن أنس بن مالك، كبر حتى كان لا يقدر على الصيام فكان يفتدي .…٦٨٢أنه بلغه أن عبد الله بن عمر، سئل عن المرأة الحامل، إذا خافت على ولدها…٦٨٣مثل ذلك .٦٨٥أنه سمع عائشة، زوج النبي صلى الله عليه وسلم تقول إن كان ليكون على…٦٨٦زوج النبي صلى الله عليه وسلم أنها قالت كان رسول الله صلى الله عليه…٦٨٧الصيام جنة فإذا كان أحدكم صائما فلا يرفث ولا يجهل فإن امرؤ قاتله أو…٦٨٨والذي نفسي بيده لخلوف فم الصائم أطيب عند الله من ريح المسك إنما يذر…٦٨٩أنه قال إذا دخل رمضان فتحت أبواب الجنة وغلقت أبواب النار وصفدت…
