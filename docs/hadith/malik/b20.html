كتاب الحج

٢٥١ حديثًا•صفحة ١ من ٢
٧٠٥مرها فلتغتسل ثم لتهل٧٠٦أن أسماء بنت عميس، ولدت، محمد بن أبي بكر بذي الحليفة فأمرها أبو بكر…٧٠٧أن عبد الله بن عمر، كان يغتسل لإحرامه قبل أن يحرم ولدخوله مكة ولوقوفه…٧٠٨أن عبد الله بن عباس، والمسور بن مخرمة، اختلفا بالأبواء فقال عبد الله…٧٠٩أن عمر بن الخطاب، قال ليعلى ابن منية وهو يصب على عمر بن الخطاب ماء…٧١٠أن عبد الله بن عمر، كان إذا دنا من مكة بات بذي طوى بين الثنيتين حتى…٧١١أن عبد الله بن عمر، كان لا يغسل رأسه وهو محرم إلا من الاحتلام . قال…٧١٢لا تلبسوا القمص ولا العمائم ولا السراويلات ولا البرانس ولا الخفاف إلا…٧١٣من لم يجد نعلين فليلبس خفين وليقطعهما أسفل من الكعبين٧١٤أنه سمع أسلم، مولى عمر بن الخطاب يحدث عبد الله بن عمر أن عمر بن…٧١٥وحدثني عن مالك، عن هشام بن عروة، عن أبيه، عن أسماء بنت أبي بكر، أنها…٧١٦أن عبد الله بن عمر، كان يكره لبس المنطقة للمحرم .٧١٧أنه سمع سعيد بن المسيب، يقول في المنطقة يلبسها المحرم تحت ثيابه أنه…٧١٨أنه رأى عثمان بن عفان بالعرج يغطي وجهه وهو محرم .٧١٩أن عبد الله بن عمر، كان يقول ما فوق الذقن من الرأس فلا يخمره المحرم .٧٢٠أن عبد الله بن عمر، كفن ابنه واقد بن عبد الله ومات بالجحفة محرما وخمر…٧٢١أن عبد الله بن عمر، كان يقول لا تنتقب المرأة المحرمة ولا تلبس…٧٢٢وحدثني عن مالك، عن هشام بن عروة، عن فاطمة بنت المنذر، أنها قالت كنا…٧٢٣زوج النبي صلى الله عليه وسلم أنها قالت كنت أطيب رسول الله صلى الله…٧٢٤انزع قميصك واغسل هذه الصفرة عنك وافعل في عمرتك ما تفعل في حجك٧٢٥مولى عمر بن الخطاب أن عمر بن الخطاب، وجد ريح طيب وهو بالشجرة فقال ممن…٧٢٦واحد، من أهله أن عمر بن الخطاب، وجد ريح طيب وهو بالشجرة وإلى جنبه…٧٢٧وعبد الله بن أبي بكر، وربيعة بن أبي عبد الرحمن، أن الوليد بن عبد…٧٢٨يهل أهل المدينة من ذي الحليفة ويهل أهل الشام من الجحفة ويهل أهل نجد…٧٢٩ويهل أهل اليمن من يلملم٧٣٠أن عبد الله بن عمر، أهل من الفرع .٧٣١عنده أن عبد الله بن عمر، أهل من إيلياء .٧٣٢أنه بلغه أن رسول الله صلى الله عليه وسلم أهل من الجعرانة بعمرة .٧٣٣لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك والملك لا…٧٣٤أن رسول الله صلى الله عليه وسلم كان يصلي في مسجد ذي الحليفة ركعتين…٧٣٥أنه سمع أباه، يقول بيداؤكم هذه التي تكذبون على رسول الله صلى الله…٧٣٦وحدثني عن مالك، عن سعيد بن أبي سعيد المقبري، عن عبيد بن جريج، أنه قال…٧٣٧أن عبد الله بن عمر، كان يصلي في مسجد ذي الحليفة٧٣٨أنه بلغه أن عبد الملك بن مروان، أهل من عند مسجد ذي الحليفة حين استوت…٧٣٩أتاني جبريل فأمرني أن آمر أصحابي أو من معي أن يرفعوا أصواتهم بالتلبية…٧٤٠حدثني يحيى، عن مالك، عن أبي الأسود، محمد بن عبد الرحمن عن عروة بن…٧٤١أن رسول الله صلى الله عليه وسلم أفرد الحج .٧٤٢وحدثني عن مالك، عن أبي الأسود، محمد بن عبد الرحمن قال وكان يتيما في…٧٤٣أن المقداد بن الأسود، دخل على علي بن أبي طالب بالسقيا وهو ينجع بكرات…٧٤٤من كان معه هدى فليهلل بالحج مع العمرة ثم لا يحل حتى يحل منهما جميعا٧٤٥حدثني يحيى، عن مالك، عن محمد بن أبي بكر الثقفي، أنه سأل أنس بن مالك -…٧٤٦أن علي بن أبي طالب، كان يلبي في الحج حتى إذا زاغت الشمس من يوم عرفة…٧٤٧زوج النبي صلى الله عليه وسلم أنها كانت تترك التلبية إذا رجعت إلى…٧٤٨أن عبد الله بن عمر، كان يقطع التلبية في الحج إذا انتهى إلى الحرم حتى…٧٤٩أنه كان يقول كان عبد الله بن عمر لا يلبي وهو يطوف بالبيت .٧٥٠أنها كانت تنزل من عرفة بنمرة ثم تحولت إلى الأراك . قالت وكانت عائشة…٧٥١أن عمر بن عبد العزيز، غدا يوم عرفة من منى فسمع التكبير عاليا فبعث…٧٥٢أن عمر بن الخطاب، قال يا أهل مكة ما شأن الناس يأتون شعثا وأنتم مدهنون…٧٥٣أن عبد الله بن الزبير، أقام بمكة تسع سنين يهل بالحج لهلال ذي الحجة…٧٥٤حدثني يحيى، عن مالك، عن عبد الله بن أبي بكر بن محمد، عن عمرة بنت عبد…٧٥٥وحدثني عن مالك، عن يحيى بن سعيد، أنه قال سألت عمرة بنت عبد الرحمن عن…٧٥٦وحدثني عن مالك، عن يحيى بن سعيد، عن محمد بن إبراهيم بن الحارث التيمي،…٧٥٧أن عبد الله بن عمر، كان يقول المرأة الحائض التي تهل بالحج أو العمرة…٧٥٨أنه بلغه أن رسول الله صلى الله عليه وسلم اعتمر ثلاثا عام الحديبية…٧٥٩أن رسول الله صلى الله عليه وسلم لم يعتمر إلا ثلاثا إحداهن في شوال…٧٦٠أن رجلا، سأل سعيد بن المسيب فقال أعتمر قبل أن أحج، فقال سعيد نعم قد…٧٦١أن عمر بن أبي سلمة، استأذن عمر بن الخطاب أن يعتمر، في شوال فأذن له…٧٦٢حدثني يحيى، عن مالك، عن هشام بن عروة، عن أبيه، أنه كان يقطع التلبية…٧٦٣أنه حدثه أنه، سمع سعد بن أبي وقاص، والضحاك بن قيس، عام حج معاوية بن…٧٦٤أنه قال والله لأن أعتمر قبل الحج وأهدي أحب إلى من أن أعتمر بعد الحج…٧٦٥وحدثني عن مالك، عن عبد الله بن دينار، عن عبد الله بن عمر، أنه كان…٧٦٦أنه سمع سعيد بن المسيب، يقول من اعتمر في شوال أو ذي القعدة أو في ذي…٧٦٧العمرة إلى العمرة كفارة لما بينهما والحج المبرور ليس له جزاء إلا الجنة٧٦٨اعتمري في رمضان فإن عمرة فيه كحجة٧٦٩أن عمر بن الخطاب، قال افصلوا بين حجكم وعمرتكم فإن ذلك أتم لحج أحدكم…٧٧٠أنه بلغه أن عثمان بن عفان، كان إذا اعتمر ربما لم يحطط عن راحلته حتى…٧٧١أن رسول الله صلى الله عليه وسلم بعث أبا رافع ورجلا من الأنصار فزوجاه…٧٧٢لا ينكح المحرم ولا ينكح ولا يخطب٧٧٣أن أبا غطفان بن طريف المري، أخبره أن أباه طريفا تزوج امرأة وهو محرم…٧٧٤أن عبد الله بن عمر، كان يقول لا ينكح المحرم ولا٧٧٥أنه بلغه أن سعيد بن المسيب، وسالم بن عبد الله، وسليمان بن يسار، سئلوا…٧٧٦أن رسول الله صلى الله عليه وسلم احتجم وهو محرم فوق رأسه وهو يومئذ…٧٧٧أنه كان يقول لا يحتجم المحرم إلا مما لا بد له منه . قال مالك لا يحتجم…٧٧٨إنما هي طعمة أطعمكموها الله٧٧٩أن الزبير بن العوام، كان يتزود صفيف الظباء وهو محرم . قال مالك…٧٨٠هل معكم من لحمه شىء٧٨١دعوه فإنه يوشك أن يأتي صاحبه٧٨٢أنه سمع سعيد بن المسيب، يحدث عن أبي هريرة، أنه أقبل من البحرين حتى…٧٨٣أنه سمع أبا هريرة، يحدث عبد الله بن عمر أنه مر به قوم محرمون بالربذة…٧٨٤أن كعب الأحبار، أقبل من الشام في ركب حتى إذا كانوا ببعض الطريق وجدوا…٧٨٥إنا لم نرده عليك إلا أنا حرم٧٨٦قال رأيت عثمان بن عفان بالعرج وهو محرم في يوم صائف قد غطى وجهه بقطيفة…٧٨٧وحدثني عن مالك، عن هشام بن عروة، عن أبيه، عن عائشة أم المؤمنين، أنها…٧٨٨خمس من الدواب ليس على المحرم في قتلهن جناح الغراب والحدأة والعقرب…٧٨٩خمس من الدواب من قتلهن وهو محرم فلا جناح عليه العقرب والفأرة والغراب…٧٩٠خمس فواسق يقتلن في الحرم الفأرة والعقرب والغراب والحدأة والكلب العقور٧٩١أن عمر بن الخطاب، أمر بقتل الحيات في الحرم . قال مالك في الكلب العقور…٧٩٢حدثني يحيى، عن مالك، عن يحيى بن سعيد، عن محمد بن إبراهيم بن الحارث…٧٩٣وحدثني عن مالك، عن علقمة بن أبي علقمة، عن أمه، أنها قالت سمعت عائشة،…٧٩٤أن عبد الله بن عمر، نظر في المرآة لشكو كان بعينيه وهو محرم .٧٩٥أن عبد الله بن عمر، كان يكره أن ينزع المحرم، حلمة أو قرادا عن بعيره .…٧٩٦وحدثني عن مالك، عن محمد بن عبد الله بن أبي مريم، . أنه سأل سعيد بن…٧٩٧قال كان الفضل بن عباس رديف رسول الله صلى الله عليه وسلم فجاءته امرأة…٧٩٨أنه بلغه أن رسول الله صلى الله عليه وسلم حل هو وأصحابه بالحديبية…٧٩٩وحدثني عن مالك، عن نافع، عن عبد الله بن عمر، أنه قال حين خرج إلى مكة…٨٠٠أنه قال المحصر بمرض لا يحل حتى يطوف بالبيت ويسعى بين الصفا والمروة…٨٠١زوج النبي صلى الله عليه وسلم أنها كانت تقول المحرم لا يحله إلا البيت .٨٠٢من أهل البصرة كان قديما أنه قال خرجت إلى مكة حتى إذا كنت ببعض الطريق…٨٠٣أنه قال من حبس دون البيت بمرض فإنه لا يحل حتى يطوف بالبيت وبين الصفا…٨٠٤أن معبد بن حزابة المخزومي، صرع ببعض طريق مكة وهو محرم فسأل من يلي على…٨٠٥ألم ترى أن قومك حين بنوا الكعبة اقتصروا عن قواعد إبراهيم٨٠٦أن عائشة أم المؤمنين، قالت ما أبالي أصليت في الحجر أم في البيت .٨٠٧أنه سمع ابن شهاب، يقول سمعت بعض، علمائنا يقول ما حجر الحجر فطاف الناس…٨٠٨أنه قال رأيت رسول الله صلى الله عليه وسلم رمل من الحجر الأسود حتى…٨٠٩أن عبد الله بن عمر، كان يرمل من الحجر الأسود إلى الحجر الأسود ثلاثة…٨١٠أن أباه، كان إذا طاف بالبيت يسعى الأشواط الثلاثة يقول اللهم لا إله…٨١١أنه رأى عبد الله بن الزبير أحرم بعمرة من التنعيم - قال - ثم رأيته…٨١٢أن عبد الله بن عمر، كان إذا أحرم من مكة لم يطف بالبيت ولا بين الصفا…٨١٣أنه بلغه أن رسول الله صلى الله عليه وسلم كان إذا قضى طوافه بالبيت…٨١٤كيف صنعت يا أبا محمد في استلام الركن٨١٥أن أباه، كان إذا طاف بالبيت يستلم الأركان كلها وكان لا يدع اليماني…٨١٦أن عمر بن الخطاب، قال وهو يطوف بالبيت للركن الأسود إنما أنت حجر ولولا…٨١٧حدثني يحيى، عن مالك، عن هشام بن عروة، عن أبيه، أنه كان لا يجمع بين…٨١٨أن عبد الرحمن بن عبد القاري، أخبره أنه، طاف بالبيت مع عمر بن الخطاب…٨١٩أنه قال رأيت عبد الله بن عباس يطوف بعد صلاة العصر ثم يدخل حجرته فلا…٨٢٠أنه قال لقد رأيت البيت يخلو بعد صلاة الصبح وبعد صلاة العصر ما يطوف به…٨٢١أن عمر بن الخطاب، قال لا يصدرن أحد من الحاج حتى يطوف بالبيت فإن آخر…٨٢٢. أن عمر بن الخطاب، رد رجلا من مر الظهران لم يكن ودع البيت حتى ودع .٨٢٣أنه قال من أفاض فقد قضى الله حجه فإنه إن لم يكن حبسه شىء فهو حقيق أن…٨٢٤طوفي من وراء الناس وأنت راكبة٨٢٥أن أبا ماعز الأسلمي عبد الله بن سفيان، أخبره أنه، كان جالسا مع عبد…٨٢٦أنه بلغه أن سعد بن أبي وقاص، كان إذا دخل مكة مراهقا خرج إلى عرفة قبل…٨٢٧نبدأ بما بدأ الله به٨٢٨لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء قدير٨٢٩أنه سمع عبد الله بن عمر، وهو على الصفا يدعو يقول اللهم إنك قلت ادعوني…٨٣٠حدثني يحيى، عن مالك، عن هشام بن عروة، عن أبيه، أنه قال قلت لعائشة أم…٨٣١أن سودة بنت عبد الله بن عمر، كانت عند عروة بن الزبير فخرجت تطوف بين…٨٣٢أن رسول الله صلى الله عليه وسلم كان إذا نزل من الصفا والمروة مشى حتى…٨٣٣أن ناسا، تماروا عندها يوم عرفة في صيام رسول الله صلى الله عليه وسلم…٨٣٤أن عائشة أم المؤمنين، كانت تصوم يوم عرفة . قال القاسم ولقد رأيتها…٨٣٥حدثني يحيى، عن مالك، عن أبي النضر��، مولى عمر بن عبيد الله عن سليمان…٨٣٦أن رسول الله صلى الله عليه وسلم بعث عبد الله بن حذافة أيام منى يطوف…٨٣٧وحدثني عن مالك، عن محمد بن يحيى بن حبان، عن الأعرج، عن أبي هريرة، أن…٨٣٨أنه أخبره أنه، دخل على أبيه عمرو بن العاص فوجده يأكل قال فدعاني قال…٨٣٩أن رسول الله صلى الله عليه وسلم أهدى جملا كان لأبي جهل بن هشام في حج…٨٤٠أن رسول الله صلى الله عليه وسلم رأى رجلا يسوق بدنة فقال " اركبها " .…٨٤١وحدثني عن مالك، عن عبد الله بن دينار، أنه كان يرى عبد الله بن عمر…٨٤٢أن عمر بن عبد العزيز، أهدى جملا في حج أو عمرة .٨٤٣. أن عبد الله بن عياش بن أبي ربيعة المخزومي، أهدى بدنتين إحداهما…٨٤٤أن عبد الله بن عمر، كان يقول إذا نتجت الناقة فليحمل ولدها حتى ينحر…٨٤٥أن أباه، قال إذا اضطررت إلى بدنتك فاركبها ركوبا غير فادح وإذا اضطررت…٨٤٦أنه كان إذا أهدى هديا من المدينة قلده وأشعره بذي الحليفة يقلده قبل أن…٨٤٧أن عبد الله بن عمر، كان إذا طعن في سنام هديه وهو يشعره قال بسم الله…٨٤٨أن عبد الله بن عمر، كان يقول الهدى ما قلد وأشعر ووقف به بعرفة .٨٤٩أن عبد الله بن عمر، كان يجلل بدنه القباطي والأنماط والحلل ثم يبعث بها…٨٥٠أنه سأل عبد الله بن دينار ما كان عبد الله بن عمر يصنع بجلال بدنه حين…٨٥١أن عبد الله بن عمر، كان يقول في الضحايا والبدن الثني فما فوقه .٨٥٢أن عبد الله بن عمر، كان لا يشق جلال بدنه ولا يجللها حتى يغدو من منى…٨٥٣أنه كان يقول لبنيه يا بني لا يهدين أحدكم من البدن شيئا يستحيي أن…٨٥٤كل بدنة عطبت من الهدى فانحرها ثم ألق قلادتها في دمها ثم خل بينها وبين…
