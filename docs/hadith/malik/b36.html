كتاب الأقضية

٥٦ حديثًا
١٣٩٩إنما أنا بشر وإنكم تختصمون إلى فلعل بعضكم أن يكون ألحن بحجته من بعض…١٤٠٠أن عمر بن الخطاب، اختصم إليه مسلم ويهودي فرأى عمر أن الحق لليهودي…١٤٠١ألا أخبركم بخير الشهداء الذي يأتي بشهادته قبل أن يسألها أو يخبر…١٤٠٢وحدثني مالك، عن ربيعة بن أبي عبد الرحمن، أنه قال قدم على عمر بن…١٤٠٣أنه بلغه أن عمر بن الخطاب، قال لا تجوز شهادة خصم ولا ظنين .١٤٠٤وغيره، أنهم سئلوا عن رجل، جلد الحد أتجوز شهادته فقالوا نعم إذا ظهرت…١٤٠٥أنه سمع ابن شهاب، يسأل عن ذلك، فقال مثل ما قال سليمان بن يسار . قال…١٤٠٦أن رسول الله صلى الله عليه وسلم قضى باليمين مع الشاهد .١٤٠٧أن عمر بن عبد العزيز، كتب إلى عبد الحميد بن عبد الرحمن بن زيد بن…١٤٠٨أنه بلغه أن أبا سلمة بن عبد الرحمن، وسليمان بن يسار، سئلا هل يقضى…١٤٠٩قال يحيى قال مالك عن جميل بن عبد الرحمن المؤذن، أنه كان يحضر عمر بن…١٤١٠أن عبد الله بن الزبير، كان يقضي بشهادة الصبيان فيما بينهم من الجراح .…١٤١١من حلف على منبري آثما تبوأ مقعده من النار١٤١٢من اقتطع حق امرئ مسلم بيمينه حرم الله عليه الجنة وأوجب له النار١٤١٣أنه سمع أبا غطفان بن طريف المري، يقول اختصم زيد بن ثابت الأنصاري وابن…١٤١٤أن رسول الله صلى الله عليه وسلم قال " لا يغلق الرهن " .١٤١٥أن عبد الملك بن مروان، قضى في امرأة أصيبت مستكرهة بصداقها على من فعل…١٤١٦من غير دينه فاضربوا عنقه١٤١٧فأخبره ثم، قال له عمر هل كان فيكم من مغربة خبر فقال نعم رجل كفر بعد…١٤١٨أن سعد بن عبادة، قال لرسول الله صلى الله عليه وسلم أرأيت إن وجدت مع…١٤١٩أن رجلا، من أهل الشام - يقال له ابن خيبري - وجد مع امرأته رجلا فقتله…١٤٢٠قال يحيى قال مالك عن ابن شهاب، عن سنين أبي جميلة، رجل من بني سليم أنه…١٤٢١هو لك يا عبد بن زمعة١٤٢٢أن امرأة، هلك عنها زوجها فاعتدت أربعة أشهر وعشرا ثم تزوجت حين حلت…١٤٢٣أن عمر بن الخطاب، كان يليط أولاد الجاهلية بمن ادعاهم في الإسلام فأتى…١٤٢٤أنه بلغه أن عمر بن الخطاب، أو عثمان بن عفان قضى أحدهما في امرأة غرت…١٤٢٥أن عمر بن الخطاب، قال ما بال رجال يطئون ولائدهم ثم يعزلوهن لا تأتيني…١٤٢٦وحدثني مالك، عن نافع، عن صفية بنت أبي عبيد، أنها أخبرته أن عمر بن…١٤٢٧من أحيا أرضا ميتة فهي له وليس لعرق ظالم حق١٤٢٨أن عمر بن الخطاب، قال من أحيا أرضا ميتة فهي له . قال مالك وعلى ذلك…١٤٢٩يمسك حتى الكعبين ثم يرسل الأعلى على الأسفل١٤٣٠لا يمنع فضل الماء ليمنع به الكلأ١٤٣١لا يمنع نقع بئر١٤٣٢أن رسول الله صلى الله عليه وسلم قال " لا ضرر ولا ضرار " .١٤٣٣لا يمنع أحدكم جاره خشبة يغرزها فى جداره١٤٣٤أن الضحاك بن خليفة، ساق خليجا له من العريض فأراد أن يمر به في أرض…١٤٣٥أنه قال كان في حائط جده ربيع لعبد الرحمن بن عوف فأراد عبد الرحمن بن…١٤٣٦أيما دار أو أرض قسمت في الجاهلية فهي على قسم الجاهلية وأيما دار أو…١٤٣٧أن ناقة، للبراء بن عازب دخلت حائط رجل فأفسدت فيه فقضى رسول الله صلى…١٤٣٨أن رقيقا، لحاطب سرقوا ناقة لرجل من مزينة فانتحروها فرفع ذلك إلى عمر…١٤٣٩أكل ولدك نحلته مثل هذا١٤٤٠زوج النبي صلى الله عليه وسلم أنها قالت إن أبا بكر الصديق كان نحلها…١٤٤١أن عمر بن الخطاب، قال ما بال رجال ينحلون أبناءهم نحلا ثم يمسكونها فإن…١٤٤٢أن عمر بن الخطاب، قال من وهب هبة لصلة رحم أو على وجه صدقة فإنه لا…١٤٤٣أيما رجل أعمر عمرى له ولعقبه فإنها للذي يعطاها لا ترجع إلى الذي…١٤٤٤أنه سمع مكحولا الدمشقي، يسأل القاسم بن محمد عن العمرى، وما يقول الناس…١٤٤٥. أن عبد الله بن عمر، ورث من حفصة بنت عمر دارها قال وكانت حفصة قد…١٤٤٦اعرف عفاصها ووكاءها ثم عرفها سنة فإن جاء صاحبها وإلا فشأنك بها١٤٤٧أن أباه، أخبره أنه، نزل منزل قوم بطريق الشام فوجد صرة فيها ثمانون…١٤٤٨أن رجلا، وجد لقطة فجاء إلى عبد الله بن عمر فقال له إني وجدت لقطة…١٤٤٩أن ثابت بن الضحاك الأنصاري، أخبره . أنه، وجد بعيرا بالحرة فعقله ثم…١٤٥٠أن عمر بن الخطاب، قال وهو مسند ظهره إلى الكعبة من أخذ ضالة فهو ضال .١٤٥١أنه سمع ابن شهاب، يقول كانت ضوال الإبل في زمان عمر بن الخطاب إبلا…١٤٥٢أنه قال خرج سعد بن عبادة مع رسول الله صلى الله عليه وسلم في بعض…١٤٥٣زوج النبي صلى الله عليه وسلم أن رجلا قال لرسول الله صلى الله عليه…١٤٥٤قد أجرت في صدقتك وخذها بميراثك
