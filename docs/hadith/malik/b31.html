٩٥ حديثًا
١٢٩٠أو الوليدة. أو يتكارى الدابة. ثم يقول للذي اشترى منه، أو تكارى منه:…١٢٩١أن عمر بن الخطاب، قال من باع عبدا وله مال فماله للبائع إلا أن يشترطه…١٢٩٢أن أبان بن عثمان، وهشام بن إسماعيل، كانا يذكران في خطبتهما عهدة…١٢٩٣أن عبد الله بن عمر، باع غلاما له بثمانمائة درهم وباعه بالبراءة فقال…١٢٩٤أن عبيد الله بن عبد الله بن عتبة بن مسعود، أخبره أن عبد الله بن مسعود…١٢٩٥أنه كان يقول لا يطأ الرجل وليدة إلا وليدة إن شاء باعها وإن شاء وهبها…١٢٩٦أن عبد الله بن عامر، أهدى لعثمان بن عفان جارية ولها زوج ابتاعها…١٢٩٧أن عبد الرحمن بن عوف، ابتاع وليدة فوجدها ذات زوج فردها .١٢٩٨من باع نخلا قد أبرت فثمرها للبائع إلا أن يشترط المبتاع١٢٩٩حدثني يحيى، عن مالك، عن نافع، عن ابن عمر، أن رسول الله صلى الله عليه…١٣٠٠وحدثني عن مالك، عن حميد الطويل، عن أنس بن مالك، أن رسول الله صلى الله…١٣٠١وحدثني عن مالك، عن أبي الرجال، محمد بن عبد الرحمن بن حارثة عن أمه،…١٣٠٢أنه كان لا يبيع ثماره حتى تطلع الثريا .١٣٠٣أن رسول الله صلى الله عليه وسلم أرخص لصاحب العرية أن يبيعها بخرصها .١٣٠٤أن رسول الله صلى الله عليه وسلم أرخص في بيع العرايا بخرصها فيما دون…١٣٠٥تألى أن لا يفعل خيرا١٣٠٦أنه بلغه أن عمر بن عبد العزيز، قضى بوضع الجائحة . قال مالك وعلى ذلك…١٣٠٧أن القاسم بن محمد، كان يبيع ثمر حائطه ويستثني منه .١٣٠٨أن جده، محمد بن عمرو بن حزم باع ثمر حائط له يقال له الأفراق بأربعة…١٣٠٩محمد بن عبد الرحمن بن حارثة أن أمه، عمرة بنت عبد الرحمن كانت تبيع…١٣١٠التمر بالتمر مثلا بمثل١٣١١أكل تمر خيبر هكذا١٣١٢أينقص الرطب إذا يبس١٣١٣حدثني يحيى، عن مالك، عن نافع، عن عبد الله بن عمر، أن رسول الله صلى…١٣١٤وحدثني عن مالك، عن داود بن الحصين، عن أبي سفيان، مولى ابن أبي أحمد عن…١٣١٥وحدثني عن مالك، عن ابن شهاب، عن سعيد بن المسيب، أن رسول الله صلى الله…١٣١٦حدثني يحيى، عن مالك، عن يحيى بن سعيد، أنه قال أمر رسول الله صلى الله…١٣١٧الدينار بالدينار والدرهم بالدرهم لا فضل بينهما١٣١٨لا تبيعوا الذهب بالذهب إلا مثلا بمثل ولا تشفوا بعضها على بعض ولا…١٣١٩وحدثني عن مالك، عن حميد بن قيس المكي، عن مجاهد، أنه قال كنت مع عبد…١٣٢٠لا تبيعوا الدينار بالدينارين ولا الدرهم بالدرهمين١٣٢١أن معاوية بن أبي سفيان، باع سقاية من ذهب أو ورق بأكثر من وزنها فقال…١٣٢٢أن عمر بن الخطاب، قال لا تبيعوا الذهب بالذهب إلا مثلا بمثل ولا تشفوا…١٣٢٣أن عمر بن الخطاب، قال لا تبيعوا الذهب بالذهب إلا مثلا بمثل ولا تشفوا…١٣٢٤أنه قال قال عمر بن الخطاب الدينار بالدينار والدرهم بالدرهم والصاع…١٣٢٥أنه سمع سعيد بن المسيب، يقول لا ربا إلا في ذهب أو فضة أو ما يكال أو…١٣٢٦أنه سمع سعيد بن المسيب، يقول قطع الذهب والورق من الفساد في الأرض .…١٣٢٧الذهب بالورق ربا إلا هاء وهاء والبر بالبر ربا إلا هاء وهاء والتمر…١٣٢٨حدثني يحيى، عن مالك، عن يزيد بن عبد الله بن قسيط، أنه رأى سعيد بن…١٣٢٩من ابتاع طعاما فلا يبعه حتى يستوفيه١٣٣٠من ابتاع طعاما فلا يبعه حتى يقبضه١٣٣١وحدثني عن مالك، عن نافع، عن عبد الله بن عمر، أنه قال كنا في زمان رسول…١٣٣٢. أن حكيم بن حزام، ابتاع طعاما أمر به عمر بن الخطاب للناس فباع حكيم…١٣٣٣أنه بلغه . أن صكوكا، خرجت للناس في زمان مروان بن الحكم من طعام الجار…١٣٣٤أنه بلغه أن رجلا، أراد أن يبتاع طعاما من رجل إلى أجل فذهب به الرجل…١٣٣٥أنه سمع جميل بن عبد الرحمن المؤذن، يقول لسعيد بن المسيب إني رجل أبتاع…١٣٣٦أنه سمع سعيد بن المسيب، وسليمان بن يسار، ينهيان أن يبيع الرجل، حنطة…١٣٣٧وحدثني عن مالك، عن كثير بن فرقد، . أنه سأل أبا بكر بن محمد بن عمرو بن…١٣٣٩حدثني يحيى، عن مالك، عن نافع، عن عبد الله بن عمر، أنه قال لا بأس بأن…١٣٤٠أنه بلغه أن سليمان بن يسار، قال فني علف حمار سعد بن أبي وقاص فقال…١٣٤١أنه أخبره أن عبد الرحمن بن الأسود بن عبد يغوث فني علف دابته فقال…١٣٤٢وحدثني عن مالك، أنه بلغه عن القاسم بن محمد، عن ابن معيقيب الدوسي، مثل…١٣٤٣أنه سأل سعيد بن المسيب فقال إني رجل أبتاع الطعام يكون من الصكوك…١٣٤٤أنه بلغه أن محمد بن سيرين، كان يقول لا تبيعوا الحب في سنبله حتى يبيض…١٣٤٥أنه بلغه أن عمر بن الخطاب، قال لا حكرة في سوقنا لا يعمد رجال بأيديهم…١٣٤٦أن عمر بن الخطاب، مر بحاطب بن أبي بلتعة وهو يبيع زبيبا له بالسوق فقال…١٣٤٧أنه بلغه أن عثمان بن عفان، كان ينهى عن الحكرة، .١٣٤٨أن علي بن أبي طالب، باع جملا له يدعى عصيفيرا بعشرين بعيرا إلى أجل .١٣٤٩أن عبد الله بن عمر، اشترى راحلة بأربعة أبعرة مضمونة عليه يوفيها…١٣٥٠وحدثني عن مالك، أنه سأل ابن شهاب عن بيع الحيوان، اثنين بواحد إلى أجل…١٣٥١حدثني يحيى، عن مالك، عن نافع، عن عبد الله بن عمر، أن رسول الله صلى…١٣٥٢وحدثني عن مالك، عن ابن شهاب، عن سعيد بن المسيب، أنه قال لا ربا في…١٣٥٣أن رسول الله١٣٥٤أنه سمع سعيد بن المسيب، يقول من ميسر أهل الجاهلية بيع الحيوان باللحم…١٣٥٥وحدثني عن مالك، عن أبي الزناد، عن سعيد بن المسيب، أنه كان يقول نهي عن…١٣٥٦حدثني يحيى، عن مالك، عن ابن شهاب، عن أبي بكر بن عبد الرحمن بن الحارث…١٣٥٧حدثني يحيى، عن مالك، أنه بلغه أن رسول الله صلى الله عليه وسلم نهى عن…١٣٥٨ورجل، يسأله عن رجل، سلف في سبائب فأراد بيعها قبل أن يقبضها فقال ابن…١٣٥٩حدثني يحيى، عن مالك، أنه بلغه . أن رسول الله صلى الله عليه وسلم نهى…١٣٦٠أنه بلغه أن رجلا، قال لرجل ابتع لي هذا البعير بنقد حتى أبتاعه منك إلى…١٣٦١أنه بلغه أن القاسم بن محمد، سئل عن رجل، اشترى سلعة بعشرة دنانير نقدا…١٣٦٢حدثني يحيى، عن مالك، عن أبي حازم بن دينار، عن سعيد بن المسيب، أن رسول…١٣٦٣حدثنا يحيى، عن مالك، عن محمد بن يحيى بن حبان، وعن أبي الزناد، عن…١٣٦٤المتبايعان كل واحد منهما بالخيار على صاحبه ما لم يتفرقا إلا بيع الخيار١٣٦٥أيما بيعين تبايعا فالقول ما قال البائع أو يترادان١٣٦٦حدثني يحيى، عن مالك، عن أبي الزناد، عن بسر بن سعيد، عن عبيد أبي صالح،…١٣٦٧يكون له الدين على الرجل إلى أجل فيضع عنه صاحب الحق ويعجله الآخر فكره…١٣٦٨وحدثني مالك، عن زيد بن أسلم، أنه قال كان الربا في الجاهلية أن يكون…١٣٦٩مطل الغني ظلم وإذا أتبع أحدكم على مليء فليتبع١٣٧٠أنه سمع رجلا، يسأل سعيد بن المسيب فقال إني رجل أبيع بالدين . فقال…١٣٧١أيما رجل باع متاعا فأفلس الذي ابتاعه منه ولم يقبض الذي باعه من ثمنه…١٣٧٢أيما رجل أفلس فأدرك الرجل ماله بعينه فهو أحق به من غيره١٣٧٣أعطه إياه فإن خيار الناس أحسنهم قضاء١٣٧٤أنه قال استسلف عبد الله بن عمر من رجل دراهم ثم قضاه دراهم خيرا منها…١٣٧٥أنه بلغه أن عمر بن الخطاب، قال في رجل أسلف رجلا طعاما على أن يعطيه…١٣٧٦أنه بلغه أن رجلا، أتى عبد الله بن عمر فقال يا أبا عبد الرحمن إني…١٣٧٧أنه سمع عبد الله بن عمر، يقول من أسلف سلفا فلا يشترط إلا قضاءه .١٣٧٨أنه بلغه أن عبد الله بن مسعود، كان يقول من أسلف سلفا فلا يشترط أفضل…١٣٧٩لا يبع بعضكم على بيع بعض١٣٨٠لا تلقوا الركبان للبيع ولا يبع بعضكم على بيع بعض ولا تناجشوا ولا يبع…١٣٨١قال مالك عن نافع، عن عبد الله بن عمر، أن رسول الله صلى الله عليه وسلم…١٣٨٢إذا بايعت فقل لا خلابة١٣٨٣أنه سمع سعيد بن المسيب، يقول إذا جئت أرضا يوفون المكيال والميزان فأطل…١٣٨٤أنه سمع محمد بن المنكدر، يقول أحب الله عبدا سمحا إن باع سمحا إن ابتاع…١٣٨٥يتكارى الدابة ثم يكريها بأكثر مما تكاراها به فقال لا بأس بذلك .
