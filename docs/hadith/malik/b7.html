كتاب صلاة الليل

٣٣ حديثًا
٢٥٤ما من امرئ تكون له صلاة بليل يغلبه عليها نوم إلا كتب الله له أجر…٢٥٥زوج النبي صلى الله عليه وسلم أنها قالت كنت أنام بين يدى رسول الله صلى…٢٥٦إذا نعس أحدكم في صلاته فليرقد حتى يذهب عنه النوم فإن أحدكم إذا صلى…٢٥٧أنه بلغه أن رسول الله صلى الله عليه وسلم سمع امرأة من الليل تصلي فقال…٢٥٨أن عمر بن الخطاب، كان يصلي من الليل ما شاء الله حتى إذا كان من آخر…٢٥٩أنه بلغه أن سعيد بن المسيب، كان يقول يكره النوم قبل العشاء والحديث…٢٦٠أنه بلغه أن عبد الله بن عمر، كان يقول صلاة الليل والنهار مثنى مثنى…٢٦١زوج النبي صلى الله عليه وسلم . أن رسول الله صلى الله عليه وسلم كان…٢٦٢يا عائشة إن عينى تنامان ولا ينام قلبي٢٦٣قالت كان رسول الله صلى الله عليه وسلم يصلي بالليل ثلاث عشرة ركعة ثم…٢٦٤مولى ابن عباس أن عبد الله بن عباس، أخبره أنه، بات ليلة عند ميمونة زوج…٢٦٥أن عبد الله بن قيس بن مخرمة، أخبره عن زيد بن خالد الجهني، أنه قال…٢٦٦صلاة الليل مثنى مثنى فإذا خشي أحدكم الصبح صلى ركعة واحدة توتر له ما…٢٦٧خمس صلوات كتبهن الله عز وجل على العباد فمن جاء بهن لم يضيع منهن شيئا…٢٦٨قال كنت أسير مع عبد الله بن عمر بطريق مكة - قال سعيد - فلما خشيت…٢٦٩أنه قال كان أبو بكر الصديق إذا أراد أن يأتي، فراشه أوتر وكان عمر بن…٢٧٠أنه بلغه أن رجلا، سأل عبد الله بن عمر عن الوتر أواجب هو فقال عبد الله…٢٧١أنه بلغه أن عائشة، زوج النبي صلى الله عليه وسلم كانت تقول من خشي أن…٢٧٢أنه قال كنت مع عبد الله بن عمر بمكة والسماء مغيمة فخشي عبد الله الصبح…٢٧٣أن عبد الله بن عمر، كان يسلم بين الركعتين والركعة في الوتر حتى يأمر…٢٧٤أن سعد بن أبي وقاص، كان يوتر بعد العتمة بواحدة . قال مالك وليس على…٢٧٥أن عبد الله بن عمر، كان يقول صلاةالمغرب وتر صلاة النهار . قال مالك من…٢٧٦أن عبد الله بن عباس، رقد ثم استيقظ فقال لخادمه انظر ما صنع الناس .…٢٧٧أنه بلغه أن عبد الله بن عباس، وعبادة بن الصامت، والقاسم بن محمد، وعبد…٢٧٨أن عبد الله بن مسعود، قال ما أبالي لو أقيمت صلاة الصبح وأنا أوتر، .٢٧٩أنه قال كان عبادة بن الصامت يؤم قوما فخرج يوما إلى الصبح فأقام المؤذن…٢٨٠أو بعد الفجر . يشك عبد الرحمن أى ذلك قال .٢٨١أنه سمع أباه القاسم بن محمد، يقول إني لأوتر بعد الفجر . قال مالك…٢٨٢أن حفصة، زوج النبي صلى الله عليه وسلم أخبرته أن رسول الله صلى الله…٢٨٣أن عائشة، زوج النبي صلى الله عليه وسلم قالت إن كان رسول الله صلى الله…٢٨٤أصلاتان معا أصلاتان معا٢٨٥أنه بلغه أن عبد الله بن عمر، فاتته ركعتا الفجر فقضاهما بعد أن طلعت…٢٨٦أنه صنع مثل الذي صنع ابن عمر .
