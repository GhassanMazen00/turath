موطأ مالك

كتاب الطهارة

١١٤ حديثًا
٣٢أنه قال لعبد الله بن زيد بن عاصم وهو جد عمرو بن يحيى المازني وكان من…٣٣إذا توضأ أحدكم فليجعل في أنفه ماء ثم لينثر ومن استجمر فليوتر٣٤من توضأ فليستنثر ومن استجمر فليوتر٣٥ويل للأعقاب من النار٣٦أن أباه، حدثه أنه، سمع عمر بن الخطاب، يتوضأ بالماء لما تحت إزاره .…٣٧إذا استيقظ أحدكم من نومه فليغسل يده قبل أن يدخلها في وضوئه فإن أحدكم…٣٨أن عمر بن الخطاب، قال إذا نام أحدكم مضطجعا فليتوضأ .٣٩أن تفسير، هذه الآية { يا أيها الذين آمنوا إذا قمتم إلى الصلاة فاغسلوا…٤٠أن ابن عمر، كان ينام جالسا ثم يصلي ولا يتوضأ .٤١هو الطهور ماؤه الحل ميتته٤٢إنها ليست بنجس إنما هي من الطوافين عليكم أو الطوافات٤٣أن عمر بن الخطاب، خرج في ركب فيهم عمرو بن العاص حتى وردوا حوضا فقال…٤٤أن عبد الله بن عمر، كان يقول إن كان الرجال والنساء في زمان رسول الله…٤٥لإبراهيم بن عبد الرحمن بن عوف أنها سألت أم سلمة زوج النبي صلى الله…٤٦وحدثني عن مالك، أنه رأى ربيعة بن أبي عبد الرحمن يقلس مرارا وهو في…٤٧أن عبد الله بن عمر، حنط ابنا لسعيد بن زيد وحمله ثم دخل المسجد فصلى…٤٨. أن رسول الله صلى الله عليه وسلم أكل كتف شاة ثم صلى ولم يتوضأ .٤٩أنه أخبره أنه، خرج مع رسول الله صلى الله عليه وسلم عام خيبر حتى إذا…٥٠أنه تعشى مع عمر بن الخطاب ثم صلى ولم يتوضأ .٥١أن عثمان بن عفان، أكل خبزا ولحما ثم مضمض وغسل يديه ومسح بهما وجهه ثم…٥٢أنه بلغه أن علي بن أبي طالب، وعبد الله بن عباس، كانا لا يتوضآن مما…٥٣يتوضأ للصلاة ثم يصيب طعاما قد مسته النار أيتوضأ قال رأيت أبي يفعل ذلك…٥٤وهب بن كيسان أنه سمع جابر بن عبد الله الأنصاري، يقول رأيت أبا بكر…٥٥أن رسول الله صلى الله عليه وسلم دعي لطعام فقرب إليه خبز ولحم فأكل منه…٥٦أن أنس بن مالك، قدم من العراق فدخل عليه أبو طلحة وأبى بن كعب فقرب…٥٧أولا يجد أحدكم ثلاثة أحجار٥٨السلام عليكم دار قوم مؤمنين وإنا إن شاء الله بكم لاحقون وددت أني قد…٥٩ما من امرئ يتوضأ فيحسن وضوءه ثم يصلي الصلاة إلا غفر له ما بينه وبين…٦٠إذا توضأ العبد المؤمن فتمضمض خرجت الخطايا من فيه وإذا استنثر خرجت…٦١إذا توضأ العبد المسلم - أو المؤمن - فغسل وجهه خرجت من وجهه كل خطيئة…٦٢أنه قال رأيت رسول الله صلى الله عليه وسلم وحانت صلاة العصر فالتمس…٦٣أنه سمع أبا هريرة، يقول من توضأ فأحسن وضوءه ثم خرج عامدا إلى الصلاة…٦٤أنه سمع سعيد بن المسيب، يسأل عن الوضوء، من الغائط بالماء فقال سعيد…٦٥إذا شرب الكلب في إناء أحدكم فليغسله سبع مرات٦٦استقيموا ولن تحصوا واعملوا وخير أعمالكم الصلاة ولا يحافظ على الوضوء…٦٧أن عبد الله بن عمر، كان يأخذ الماء بأصبعيه لأذنيه .٦٨أنه بلغه أن جابر بن عبد الله الأنصاري، سئل عن المسح، على العمامة فقال…٦٩أن أباه، عروة بن الزبير كان ينزع العمامة ويمسح رأسه بالماء .٧٠وحدثني عن مالك، عن نافع، أنه رأى صفية بنت أبي عبيد امرأة عبد الله بن…٧١أن رسول الله صلى الله عليه وسلم ذهب لحاجته في غزوة تبوك - قال المغيرة…٧٢وعبد الله بن دينار، أنهما أخبراه أن عبد الله بن عمر قدم الكوفة على…٧٣أن عبد الله بن عمر، بال في السوق ثم توضأ فغسل وجهه ويديه ومسح رأسه ثم…٧٤وحدثني عن مالك، عن سعيد بن عبد الرحمن بن رقيش، أنه قال رأيت أنس بن…٧٥أنه رأى أباه يمسح على الخفين قال وكان لا يزيد إذا مسح على الخفين على…٧٦وحدثني عن مالك، أنه سأل ابن شهاب عن المسح، على الخفين كيف هو فأدخل…٧٧أن عبد الله بن عمر، كان إذا رعف انصرف فتوضأ ثم رجع فبنى ولم يتكلم .٧٨أنه بلغه أن عبد الله بن عباس، كان يرعف فيخرج فيغسل الدم عنه ثم يرجع…٧٩أنه رأى سعيد بن المسيب رعف وهو يصلي فأتى حجرة أم سلمة زوج النبي صلى…٨٠أنه قال رأيت سعيد٨١أنه رأى سالم بن عبد الله يخرج من أنفه الدم حتى تختضب أصابعه ثم يفتله…٨٢أن المسور بن مخرمة، أخبره أنه، دخل على عمر بن الخطاب من الليلة التي…٨٣أن سعيد بن المسيب، قال ما ترون فيمن غلبه الدم من رعاف فلم ينقطع عنه…٨٤إذا وجد ذلك أحدكم فلينضح فرجه بالماء وليتوضأ وضوءه للصلاة٨٥أن عمر بن الخطاب، قال إني لأجده ينحدر مني مثل الخريزة فإذا وجد ذلك…٨٦فقال إذا وجدته فاغسل فرجك وتوضأ وضوءك للصلاة .٨٧أنه سمعه ورجل، يسأله فقال إني لأجد البلل وأنا أصلي، أفأنصرف فقال له…٨٨أجده فقال انضح ما تحت ثوبك بالماء واله عنه .٨٩إذا مس أحدكم ذكره فليتوضأ٩٠أنه قال كنت أمسك المصحف على سعد بن أبي وقاص فاحتككت فقال سعد لعلك…٩١أن عبد الله بن عمر، كان يقول إذا مس أحدكم ذكره فقد وجب عليه الوضوء .٩٢أنه كان يقول من مس ذكره فقد وجب عليه الوضوء .٩٣وحدثني عن مالك، عن ابن شهاب، عن سالم بن عبد الله، أنه قال رأيت أبي…٩٤أنه قال كنت مع عبد الله بن عمر في سفر فرأيته بعد أن طلعت الشمس توضأ…٩٥أنه كان يقول قبلة الرجل امرأته وجسها بيده من الملامسة فمن قبل امرأته…٩٦أنه بلغه أن عبد الله بن مسعود، كان يقول من قبلة الرجل امرأته الوضوء .٩٧وحدثني عن مالك، عن ابن شهاب، أنه كان يقول من قبلة الرجل امرأته الوضوء…٩٨أن رسول الله صلى الله عليه وسلم كان إذا اغتسل من الجنابة بدأ بغسل…٩٩أن رسول الله صلى الله عليه وسلم كان يغتسل من إناء - هو الفرق - من…١٠٠أن عبد الله بن عمر، كان إذا اغتسل من الجنابة بدأ فأفرغ على يده اليمنى…١٠١أنه بلغه أن عائشة، سئلت عن غسل المرأة، من الجنابة فقالت لتحفن على…١٠٢أن عمر بن الخطاب، وعثمان بن عفان، وعائشة، زوج النبي صلى الله عليه…١٠٣أنه قال سألت عائشة زوج النبي صلى الله عليه وسلم ما يوجب الغسل فقالت…١٠٤أن أبا موسى الأشعري، أتى عائشة زوج النبي صلى الله عليه وسلم فقال لها…١٠٥مولى عثمان بن عفان أن محمود بن لبيد الأنصاري، سأل زيد بن ثابت عن…١٠٦أن عبد الله بن عمر، كان يقول إذا جاوز الختان الختان فقد وجب الغسل .١٠٧توضأ واغسل ذكرك ثم نم١٠٨زوج النبي صلى الله عليه وسلم أنها كانت تقول إذا أصاب أحدكم المرأة ثم…١٠٩أن عبد الله بن عمر، كان إذا أراد أن ينام، أو يطعم وهو جنب غسل وجهه…١١٠أن عطاء بن يسار، أخبره أن رسول الله صلى الله عليه وسلم كبر في صلاة من…١١١وحدثني عن مالك، عن هشام بن عروة، عن زييد بن الصلت، أنه قال خرجت مع…١١٢أن عمر بن الخطاب، غدا إلى أرضه بالجرف فوجد في ثوبه احتلاما فقال لقد…١١٣أن عمر بن الخطاب، صلى بالناس الصبح ثم غدا إلى أرضه بالجرف فوجد في…١١٤وحدثني عن مالك، عن هشام بن عروة، عن أبيه، عن يحيى بن عبد الرحمن بن…١١٥أن أم سليم، قالت لرسول الله صلى الله عليه وسلم المرأة ترى في المنام…١١٦نعم إذا رأت الماء١١٧أن عبد الله بن عمر، كان يقول لا بأس أن يغتسل بفضل المرأة ما لم تكن…١١٨أن عبد الله بن عمر، كان يعرق في الثوب وهو جنب ثم يصلي فيه .١١٩أن عبد الله بن عمر، كان يغسل جواريه رجليه ويعطينه الخمرة وهن حيض .…١٢٠حدثني يحيى، عن مالك، عن عبد الرحمن بن القاسم، عن أبيه، عن عائشة أم…١٢١حدثني يحيى، عن مالك، عن نافع، أنه أقبل هو وعبد الله بن عمر من الجرف…١٢٢أن عبد الله بن عمر، كان يتيمم إلى المرفقين . وسئل مالك كيف التيمم…١٢٣أن رجلا، سأل سعيد بن المسيب عن الرجل الجنب، يتيمم ثم يدرك الماء فقال…١٢٤لتشد عليها إزارها ثم شأنك بأعلاها١٢٥ما لك لعلك نفست١٢٦أن عبيد الله بن عبد الله بن عمر، أرسل إلى عائشة يسألها هل يباشر الرجل…١٢٧أنه بلغه أن سالم بن عبد الله، وسليمان بن يسار، سئلا عن الحائض، هل…١٢٨حدثني يحيى، عن مالك، عن علقمة بن أبي علقمة، عن أمه، مولاة عائشة أم…١٢٩أنه بلغها أن نساء، كن يدعون بالمصابيح من جوف الليل ينظرن إلى الطهر…١٣٠أنه بلغه أن عائشة، زوج النبي صلى الله عليه وسلم قالت في المرأة الحامل…١٣١وحدثني عن مالك، أنه سأل ابن شهاب عن المرأة الحامل، ترى الدم قال تكف…١٣٢وحدثني عن مالك، عن هشام بن عروة، عن أبيه، عن عائشة، زوج النبي صلى…١٣٣إذا أصاب ثوب إحداكن الدم من الحيضة فلتقرصه ثم لتنضحه بالماء ثم لتصلي…١٣٤إنما ذلك عرق وليست بالحيضة فإذا أقبلت الحيضة فاتركي الصلاة فإذا ذهب…١٣٥لتنظر إلى عدد الليالي والأيام التي كانت تحيضهن من الشهر قبل أن يصيبها…١٣٦أنها رأت زينب بنت جحش التي كانت تحت عبد الرحمن بن عوف وكانت تستحاض…١٣٧مولى أبي بكر بن عبد الرحمن أن القعقاع بن حكيم، وزيد بن أسلم، أرسلاه…١٣٨وحدثني عن مالك، عن هشام بن عروة، عن أبيه، أنه قال ليس على المستحاضة…١٣٩زوج النبي صلى الله عليه وسلم أنها قالت أتي رسول الله صلى الله عليه…١٤٠أنها أتت بابن لها صغير لم يأكل الطعام إلى رسول الله صلى الله عليه…١٤١حدثني يحيى، عن مالك، عن يحيى بن سعيد، أنه قال دخل أعرابي المسجد فكشف…١٤٢وحدثني عن مالك، عن عبد الله بن دينار، أنه قال رأيت عبد الله بن عمر…١٤٣يا معشر المسلمين إن هذا يوم جعله الله عيدا فاغتسلوا ومن كان عنده طيب…١٤٤لولا أن أشق على أمتي لأمرتهم بالسواك١٤٥أنه قال لولا أن يشق، على أمته لأمرهم بالسواك مع كل وضوء .
