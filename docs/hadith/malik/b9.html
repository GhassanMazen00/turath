كتاب قصر الصلاة فى السفر

١٠٢ حديثًا
٣٢٦أن رسول الله صلى الله عليه وسلم كان يجمع بين الظهر والعصر في سفره إلى…٣٢٧إنكم ستأتون غدا إن شاء الله عين تبوك وإنكم لن تأتوها حتى يضحى النهار…٣٢٨أن عبد الله بن عمر، قال كان رسول الله صلى الله عليه وسلم إذا عجل به…٣٢٩أنه قال صلى رسول الله صلى الله عليه وسلم الظهر والعصر جميعا والمغرب…٣٣٠أن عبد الله بن عمر، كان إذا جمع الأمراء بين المغرب والعشاء في المطر…٣٣١أنه سأل سالم بن عبد الله هل يجمع بين الظهر والعصر في السفر فقال نعم…٣٣٢أنه كان يقول كان رسول الله صلى الله عليه وسلم إذا أراد أن يسير يومه…٣٣٣حدثني يحيى، عن مالك، عن ابن شهاب، عن رجل، من آل خالد بن أسيد أنه سأل…٣٣٤زوج النبي صلى الله عليه وسلم أنها قالت فرضت الصلاة ركعتين ركعتين في…٣٣٥أنه قال لسالم بن عبد الله ما أشد ما رأيت أباك أخر المغرب في السفر…٣٣٦أن عبد الله بن عمر، كان إذا خرج حاجا أو معتمرا قصر الصلاة بذي الحليفة…٣٣٧أنه ركب إلى ريم فقصر الصلاة في مسيره ذلك . قال مالك وذلك نحو من أربعة…٣٣٨أن عبد الله بن عمر، ركب إلى ذات النصب فقصر الصلاة في مسيره ذلك . قال…٣٣٩أنه كان يسافر إلى خيبر فيقصر الصلاة .٣٤٠أن عبد الله بن عمر، كان يقصر الصلاة في مسيره اليوم التام .٣٤١أنه كان يسافر مع ابن عمر البريد فلا يقصر الصلاة .٣٤٢أنه بلغه أن عبد الله بن عباس، كان يقصر الصلاة في مثل ما بين مكة…٣٤٣أن عبد الله بن عمر، كان يقول أصلي صلاة المسافر ما لم أجمع مكثا وإن…٣٤٤أن ابن عمر، أقام بمكة عشر ليال يقصر الصلاة إلا أن يصليها مع الإمام…٣٤٥أنه سمع سعيد بن المسيب، قال من أجمع إقامة أربع ليال وهو مسافر أتم…٣٤٦أن عمر بن الخطاب، كان إذا قدم مكة صلى بهم ركعتين ثم يقول يا أهل مكة…٣٤٧مثل ذلك .٣٤٨أن عبد الله بن عمر، كان يصلي وراء الإمام بمنى أربعا فإذا صلى لنفسه…٣٤٩وحدثني عن مالك، عن ابن شهاب، عن صفوان، أنه قال جاء عبد الله بن عمر…٣٥٠أنه لم يكن يصلي مع صلاة الفريضة في السفر شيئا قبلها ولا بعدها إلا من…٣٥١أنه بلغه أن القاسم بن محمد، وعروة بن الزبير، وأبا، بكر بن عبد الرحمن…٣٥٢أن عبد الله بن عمر، كان يرى ابنه عبيد الله بن عبد الله يتنفل في السفر…٣٥٣أنه قال رأيت رسول الله صلى الله عليه وسلم يصلي وهو على حمار وهو متوجه…٣٥٤أن رسول الله صلى الله عليه وسلم كان يصلي على راحلته في السفر حيث…٣٥٥قال رأيت أنس بن مالك في السفر وهو يصلي على حمار وهو متوجه إلى غير…٣٥٦مولى عقيل بن أبي طالب أن أم هانئ بنت أبي طالب، أخبرته أن رسول الله…٣٥٧مولى عمر بن عبيد الله أن أبا مرة، مولى عقيل بن أبي طالب أخبره أنه،…٣٥٨زوج النبي صلى الله عليه وسلم أنها قالت ما رأيت رسول الله صلى الله…٣٥٩أنها كانت تصلي الضحى ثماني ركعات ثم تقول لو نشر لي أبواى ما تركتهن .٣٦٠قوموا فلأصلي لكم٣٦١وحدثني عن مالك، عن ابن شهاب، عن عبيد الله بن عبد الله بن عتبة، أنه…٣٦٢إذا كان أحدكم يصلي فلا يدع أحدا يمر بين يديه وليدرأه ما استطاع فإن…٣٦٣لو يعلم المار بين يدى المصلي ماذا عليه لكان أن يقف أربعين خيرا له من…٣٦٤أن كعب الأحبار، قال لو يعلم المار بين يدى المصلي ماذا عليه لكان أن…٣٦٥أنه بلغه أن عبد الله بن عمر، كان يكره أن يمر، بين أيدي النساء وهن…٣٦٦أن عبد الله بن عمر، كان لا يمر بين يدى أحد ولا يدع أحدا يمر بين يديه .٣٦٧حدثني يحيى، عن مالك، عن ابن شهاب، عن عبيد الله بن عبد الله بن عتبة بن…٣٦٨أنه بلغه أن سعد بن أبي وقاص، كان يمر بين يدى بعض الصفوف والصلاة قائمة…٣٦٩أنه بلغه أن علي بن أبي طالب، قال لا يقطع الصلاة شىء مما يمر بين يدى…٣٧٠أن عبد الله بن عمر، كان يقول لا يقطع الصلاة شىء مما يمر بين يدى…٣٧١أنه بلغه أن عبد الله بن عمر، كان يستتر براحلته إذا صلى .٣٧٢أن أباه، كان يصلي في الصحراء إلى غير سترة .٣٧٣أنه قال رأيت عبد الله بن عمر إذا أهوى ليسجد مسح الحصباء لموضع جبهته…٣٧٤أنه بلغه أن أبا ذر، كان يقول مسح الحصباء مسحة واحدة وتركها خير من حمر…٣٧٥أن عمر بن الخطاب، كان يأمر بتسوية الصفوف فإذا جاءوه فأخبروه أن قد…٣٧٦في أن يفرض، لي فلم أزل أكلمه وهو يسوي الحصباء بنعليه حتى جاءه رجال قد…٣٧٧أنه قال من كلام النبوة إذا لم تستحي فافعل ما شئت ووضع اليدين إحداهما…٣٧٨أنه قال كان الناس يؤمرون أن يضع الرجل اليد اليمنى على ذراعه اليسرى في…٣٧٩أن عبد الله بن عمر، كان لا يقنت في شىء من الصلاة .٣٨٠إذا أراد أحدكم الغائط فليبدأ به قبل الصلاة٣٨١أن عمر بن الخطاب، قال لا يصلين أحدكم وهو ضام بين وركيه .٣٨٢الملائكة تصلي على أحدكم ما دام في مصلاه الذي صلى فيه ما لم يحدث اللهم…٣٨٣لا يزال أحدكم في صلاة ما كانت الصلاة تحبسه لا يمنعه أن ينقلب إلى أهله…٣٨٤مولى أبي بكر أن أبا بكر بن عبد الرحمن، كان يقول من غدا أو راح إلى…٣٨٥أنه سمع أبا هريرة، يقول إذا صلى أحدكم ثم جلس في مصلاه لم تزل الملائكة…٣٨٦ألا أخبركم بما يمحو الله به الخطايا ويرفع به الدرجات إسباغ الوضوء عند…٣٨٧أنه بلغه أن سعيد بن المسيب، قال يقال لا يخرج أحد من المسجد بعد النداء…٣٨٨إذا دخل أحدكم المسجد فليركع ركعتين قبل أن يجلس٣٨٩أنه قال له ألم أر صاحبك إذا دخل المسجد يجلس قبل أن يركع قال أبو النضر…٣٩٠أن عبد الله بن عمر، كان إذا سجد وضع كفيه على الذي يضع عليه جبهته .…٣٩١أن عبد الله بن عمر، كان يقول من وضع جبهته بالأرض فليضع كفيه على الذي…٣٩٢يا أبا بكر ما منعك أن تثبت إذ أمرتك٣٩٣أن ابن عمر، لم يكن يلتفت في صلاته .٣٩٤أنه قال كنت أصلي وعبد الله بن عمر ورائي ولا أشعر فالتفت فغمزني .٣٩٥أنه قال دخل زيد بن ثابت المسجد فوجد الناس ركوعا فركع ثم دب حتى وصل…٣٩٦أنه بلغه أن عبد الله بن مسعود، كان يدب راكعا .٣٩٧قولوا اللهم صل على محمد وأزواجه وذريته كما صليت على آل إبراهيم وبارك…٣٩٨قولوا اللهم صل على محمد وعلى آل محمد كما صليت على إبراهيم وبارك على…٣٩٩قال رأيت عبد الله بن عمر يقف على قبر النبي صلى الله عليه وسلم فيصلي…٤٠٠أن رسول الله صلى الله عليه وسلم كان يصلي قبل الظهر ركعتين وبعدها…٤٠١أترون قبلتي هاهنا فوالله ما يخفى على خشوعكم ولا ركوعكم إني لأراكم من…٤٠٢أن رسول الله صلى الله عليه وسلم كان يأتي قباء راكبا وماشيا .٤٠٣ما ترون في الشارب والسارق والزاني٤٠٤اجعلوا من صلاتكم في بيوتكم٤٠٥أن عبد الله بن عمر، كان يقول إذا لم يستطع المريض السجود أومأ برأسه…٤٠٦أن عبد الله بن عمر، كان إذا جاء المسجد - وقد صلى الناس - بدأ بصلاة…٤٠٧أن عبد الله بن عمر، مر على رجل وهو يصلي فسلم عليه فرد الرجل كلاما…٤٠٨أن عبد الله بن عمر، كان يقول من نسي صلاة فلم يذكرها إلا وهو مع الإمام…٤٠٩واسع بن حبان، أنه قال كنت أصلي وعبد الله بن عمر مسند ظهره إلى جدار…٤١٠من المهاجرين لم ير به بأسا أنه سأل عبد الله بن عمرو بن العاص أأصلي في…٤١١أنه قال ما صلاة يجلس في كل ركعة منها ثم قال سعيد هي المغرب إذا فاتتك…٤١٢. أن رسول الله صلى الله عليه وسلم كان يصلي وهو حامل أمامة بنت زينب…٤١٣يتعاقبون فيكم ملائكة بالليل وملائكة بالنهار ويجتمعون في صلاة العصر…٤١٤مروا أبا بكر فليصل للناس٤١٥أليس يشهد أن لا إله إلا الله وأن محمدا رسول الله٤١٦اللهم لا تجعل قبري وثنا يعبد اشتد غضب الله على قوم اتخذوا قبور…٤١٧أين تحب أن أصلي٤١٨أنه رأى رسول الله صلى الله عليه وسلم مستلقيا في المسجد واضعا إحدى…٤١٩أن عمر بن الخطاب، وعثمان بن عفان، - رضى الله عنهما - كانا يفعلان ذلك .٤٢٠أن عبد الله بن مسعود، قال لإنسان إنك في زمان كثير فقهاؤه قليل قراؤه…٤٢١أنه قال بلغني أن أول، ما ينظر فيه من عمل العبد الصلاة فإن قبلت منه…٤٢٢زوج النبي صلى الله عليه وسلم أنها قالت كان أحب العمل إلى رسول الله…٤٢٣ألم يكن الآخر مسلما٤٢٤أنه بلغه أن عطاء بن يسار، كان إذا مر عليه بعض من يبيع في المسجد دعاه…٤٢٥أنه بلغه أن عمر بن الخطاب، بنى رحبة في ناحية المسجد تسمى البطيحاء…٤٢٦خمس صلوات في اليوم والليلة٤٢٧يعقد الشيطان على قافية رأس أحدكم إذا هو نام ثلاث عقد يضرب
