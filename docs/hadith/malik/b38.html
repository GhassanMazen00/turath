كتاب العتق والولاء

٢٥ حديثًا
١٤٦٤من أعتق شركا له في عبد فكان له مال يبلغ ثمن العبد قوم عليه قيمة العدل…١٤٦٥واحد، عن الحسن بن أبي الحسن البصري، وعن محمد بن سيرين، أن رجلا، في…١٤٦٦أن رجلا، في إمارة أبان بن عثمان أعتق رقيقا له كلهم جميعا ولم يكن له…١٤٦٧أنه سمعه يقول مضت السنة أن العبد إذا عتق تبعه ماله . قال مالك ومما…١٤٦٨أن عمر بن الخطاب، قال أيما وليدة ولدت من سيدها فإنه لا يبيعها ولا…١٤٦٩أنه بلغه أن عمر بن الخطاب، أتته وليدة قد ضربها سيدها بنار أو أصابها…١٤٧٠حدثني مالك، عن هلال بن أسامة، عن عطاء بن يسار، عن عمر بن الحكم، أنه…١٤٧١أتشهدين أن لا إله إلا الله١٤٧٢وحدثني مالك، أنه بلغه عن المقبري، أنه قال سئل أبو هريرة عن الرجل،…١٤٧٣وكان، من أصحاب رسول الله صلى الله عليه وسلم أنه سئل عن الرجل تكون…١٤٧٤أنه بلغه أن عبد الله بن عمر، سئل عن الرقبة الواجبة، هل تشترى بشرط…١٤٧٥أن أمه، أرادت أن توصي ثم أخرت ذلك إلى أن تصبح فهلكت وقد كانت همت بأن…١٤٧٦وحدثني مالك، عن يحيى بن سعيد، أنه قال توفي عبد الرحمن بن أبي بكر في…١٤٧٧أغلاها ثمنا وأنفسها عند أهلها١٤٧٨وحدثني مالك، عن نافع، عن عبد الله بن عمر، أنه أعتق ولد زنا وأمه .١٤٧٩خذيها واشترطي لهم الولاء فإنما الولاء لمن أعتق١٤٨٠لا يمنعنك ذلك فإنما الولاء لمن أعتق١٤٨١اشتريها وأعتقيها فإنما الولاء لمن أعتق١٤٨٢وحدثني مالك، عن عبد الله بن دينار، عن عبد الله بن عمر، أن رسول الله…١٤٨٣. أن الزبير بن العوام، اشترى عبدا فأعتقه ولذلك العبد بنون من امرأة…١٤٨٤أنه بلغه أن سعيد بن المسيب، سئل عن عبد له، ولد من امرأة حرة لمن…١٤٨٥حدثني مالك، عن عبد الله بن أبي بكر بن محمد بن عمرو بن حزم، عن عبد…١٤٨٦أنه أخبره أبوه، أنه كان جالسا عند أبان بن عثمان فاختصم إليه نفر من…١٤٨٧أنه بلغه أن سعيد بن المسيب، قال في رجل هلك وترك بنين له ثلاثة وترك…١٤٨٨قال يوالي من شاء فإن مات ولم يوال أحدا فميراثه للمسلمين وعقله عليهم .…
