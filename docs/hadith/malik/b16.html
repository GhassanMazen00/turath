كتاب الجنائز

٥٨ حديثًا
٥٢٠أن رسول الله صلى الله عليه وسلم غسل في قميص .٥٢١اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك بماء وسدر واجعلن في…٥٢٢حديث ٥٢٢٥٢٣زوج النبي صلى الله عليه وسلم أن رسول الله صلى الله عليه وسلم كفن في…٥٢٤أن رسول الله صلى الله عليه وسلم كفن في ثلاثة أثواب بيض سحولية .٥٢٥أنه قال بلغني أن أبا بكر الصديق، قال لعائشة وهو مريض في كم كفن رسول…٥٢٦أنه قال الميت يقمص ويؤزر ويلف في الثوب الثالث فإن لم يكن إلا ثوب واحد…٥٢٧أن رسول الله صلى الله عليه وسلم وأبا بكر وعمر كانوا يمشون أمام…٥٢٨أنه أخبره أنه، رأى عمر بن الخطاب يقدم الناس أمام الجنازة في جنازة…٥٢٩قال ما رأيت أبي قط في جنازة إلا أمامها - قال - ثم يأتي البقيع فيجلس…٥٣٠أنه قال المشى خلف الجنازة من خطإ السنة .٥٣١أنها قالت لأهلها أجمروا ثيابي إذا مت ثم حنطوني ولا تذروا على كفني…٥٣٢أنه نهى أن يتبع، بعد موته بنار . قال يحيى سمعت مالكا يكره ذلك .٥٣٣أن رسول الله صلى الله عليه وسلم نعى النجاشي للناس في اليوم الذي مات…٥٣٤إذا ماتت فآذنوني بها٥٣٥يدرك بعض التكبير على الجنازة ويفوته بعضه فقال يقضي ما فاته من ذلك .٥٣٦أخبرك أتبعها، من أهلها فإذا وضعت كبرت وحمدت الله وصليت على نبيه ثم…٥٣٧وحدثني عن مالك، عن يحيى بن سعيد، أنه قال سمعت سعيد بن المسيب، يقول…٥٣٨أن عبد الله بن عمر، كان لا يقرأ في الصلاة على الجنازة .٥٣٩مولى عبد الرحمن بن أبي سفيان بن حويطب أن زينب بنت أبي سلمة، توفيت -…٥٤٠أن عبد الله بن عمر، قال يصلى على الجنازة بعد العصر وبعد الصبح إذا…٥٤١زوج النبي صلى الله عليه وسلم أنها أمرت أن يمر عليها بسعد بن أبي وقاص…٥٤٢أنه قال صلي على عمر بن الخطاب في المسجد .٥٤٣أنه بلغه أن عثمان بن عفان، وعبد الله بن عمر، وأبا، هريرة كانوا يصلون…٥٤٤أن عبد الله بن عمر، كان إذا صلى على الجنائز يسلم حتى يسمع من يليه .٥٤٥أن عبد الله بن عمر، كان يقول لا يصلي الرجل على الجنازة إلا وهو طاهر .…٥٤٦ما دفن نبي قط إلا في مكانه الذي توفي فيه٥٤٧أنه قال كان بالمدينة رجلان أحدهما يلحد والآخر لا يلحد فقالوا أيهما…٥٤٨أنه بلغه أن أم سلمة، زوج النبي صلى الله عليه وسلم كانت تقول ما صدقت…٥٤٩أن عائشة، زوج النبي صلى الله عليه وسلم قالت رأيت ثلاثة أقمار سقطن في…٥٥٠واحد، ممن يثق به أن سعد بن أبي وقاص وسعيد بن زيد بن عمرو بن نفيل…٥٥١أنه قال ما أحب أن أدفن، بالبقيع لأن أدفن بغيره أحب إلى من أن أدفن به…٥٥٢أن رسول الله صلى الله عليه وسلم كان يقوم في الجنائز ثم جلس بعد .٥٥٣أنه بلغه أن علي بن أبي طالب، كان يتوسد القبور ويضطجع عليها . قال مالك…٥٥٤أنه سمع أبا أمامة بن سهل بن حنيف، يقول كنا نشهد الجنائز فما يجلس آخر…٥٥٥غلبنا عليك يا أبا الربيع٥٥٦إنكم لتبكون عليها وإنها لتعذب في قبرها٥٥٧لا يموت لأحد من المسلمين ثلاثة من الولد فتمسه النار إلا تحلة القسم٥٥٨لا يموت لأحد من المسلمين ثلاثة من الولد فيحتسبهم إلا كانوا جنة من…٥٥٩ما يزال المؤمن يصاب في ولده وحامته حتى يلقى الله وليست له خطيئة٥٦٠ليعز المسلمين في مصائبهم المصيبة بي٥٦١زوج النبي صلى الله عليه وسلم، أن رسول الله صلى الله عليه وسلم قال من…٥٦٢وحدثني عن مالك، عن يحيى بن سعيد، عن القاسم بن محمد، أنه قال هلكت…٥٦٣عمرة بنت عبد الرحمن أنه سمعها تقول، لعن رسول الله صلى الله عليه وسلم…٥٦٤أنه بلغه أن عائشة، زوج النبي صلى الله عليه وسلم كانت تقول كسر عظم…٥٦٥اللهم اغفر لي وارحمني وألحقني بالرفيق الأعلى٥٦٦ما من نبي يموت حتى يخير٥٦٧إن أحدكم إذا مات عرض عليه مقعده بالغداة والعشي إن كان من أهل الجنة…٥٦٨كل ابن آدم تأكله الأرض إلا عجب الذنب منه خلق وفيه يركب٥٦٩وحدثني عن مالك عن ابن شهاب عن عبد الرحمن بن كعب بن مالك الأنصاري أنه…٥٧٠قال الله تبارك وتعالى إذا أحب عبدي لقائي أحببت لقاءه وإذا كره لقائي…٥٧١قال رجل لم يعمل حسنة قط لأهله إذا مات فحرقوه ثم اذروا نصفه في البر…٥٧٢كل مولود يولد على الفطرة فأبواه يهودانه أو ينصرانه، كما تناتج الإبل…٥٧٣لا تقوم الساعة حتى يمر الرجل بقبر الرجل فيقول يا ليتني مكانه٥٧٤مستريح ومستراح منه٥٧٥ذهبت ولم تلبس منها بشىء٥٧٦إني بعثت إلى أهل البقيع لأصلي عليهم٥٧٧أن أبا هريرة، قال : أسرعوا بجنائزكم، فإنما هو خير تقدمونه إليه أو شر…
