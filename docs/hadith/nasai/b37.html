كتاب تحريم الدم

٣٩٦٦أمرت أن أقاتل المشركين حتى يشهدوا أن لا إله إلا الله وأن محمدا عبده…٣٩٦٧أمرت أن أقاتل الناس حتى يشهدوا أن لا إله إلا الله وأن محمدا رسول الله…٣٩٦٨أخبرنا محمد بن المثنى، قال حدثنا محمد بن عبد الله الأنصاري، قال…٣٩٦٩أمرت أن أقاتل الناس حتى يشهدوا أن لا إله إلا الله وأني رسول الله…٣٩٧٠أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…٣٩٧١أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوها فقد عصموا…٣٩٧٢أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…٣٩٧٣أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…٣٩٧٤أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قالها فقد عصم مني…٣٩٧٥أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوها عصموا مني…٣٩٧٦أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوها منعوا مني…٣٩٧٧أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوها منعوا مني…٣٩٧٨نقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوا لا إله إلا الله…٣٩٧٩أخبرنا محمد بن عبد الله بن المبارك، قال حدثنا الأسود بن عامر، قال…٣٩٨٠إنه أوحي إلى أن أقاتل الناس حتى يقولوا لا إله إلا الله٣٩٨١أخبرنا أحمد بن سليمان، قال حدثنا الحسن بن محمد بن أعين، قال حدثنا…٣٩٨٢يقول أتيت رسول الله صلى الله عليه وسلم في وفد ثقيف فكنت معه في قبة…٣٩٨٣أمرت أن أقاتل الناس حتى يشهدوا أن لا إله إلا الله ثم تحرم دماؤهم…٣٩٨٤كل ذنب عسى الله أن يغفره إلا الرجل يقتل المؤمن متعمدا أو الرجل يموت…٣٩٨٥لا تقتل نفس ظلما إلا كان على ابن آدم الأول كفل من دمها وذلك أنه أول…٣٩٨٦والذي نفسي بيده لقتل مؤمن أعظم عند الله من زوال الدنيا٣٩٨٧لزوال الدنيا أهون عند الله من قتل رجل مسلم٣٩٨٨قال قتل المؤمن أعظم عند الله من زوال الدنيا .٣٩٨٩قال قتل المؤمن أعظم عند الله من زوال الدنيا .٣٩٩٠قتل المؤمن أعظم عند الله من زوال الدنيا٣٩٩١أول ما يحاسب به العبد الصلاة وأول ما يقضى بين الناس في الدماء٣٩٩٢أول ما يحكم بين الناس في الدماء٣٩٩٣قال قال عبد الله أول ما يقضى بين الناس يوم القيامة في الدماء .٣٩٩٤قال أول ما يقضى بين الناس يوم القيامة في الدماء .٣٩٩٥أول ما يقضى فيه بين الناس يوم القيامة في الدماء٣٩٩٦قال أول ما يقضى بين الناس في الدماء .٣٩٩٧يجيء الرجل آخذا بيد الرجل فيقول يا رب هذا قتلني . فيقول الله له لم…٣٩٩٨يجيء المقتول بقاتله يوم القيامة فيقول سل هذا فيم قتلني فيقول قتلته…٣٩٩٩يجيء متعلقا بالقاتل تشخب أوداجه دما فيقول أى رب سل هذا فيم قتلني٤٠٠٠قال وأخبرني أزهر بن جميل البصري، قال حدثنا خالد بن الحارث، قال حدثنا…٤٠٠١أخبرنا عمرو بن علي، قال حدثنا يحيى، قال حدثنا ابن جريج، قال حدثني…٤٠٠٢قال أمرني عبد الرحمن بن أبي ليلى أن أسأل ابن عباس، عن هاتين الآيتين،…٤٠٠٣أن قوما، كانوا قتلوا فأكثروا وزنوا فأكثروا وانتهكوا فأتوا النبي صلى…٤٠٠٤أن ناسا، من أهل الشرك أتوا محمدا فقالوا إن الذي تقول وتدعو إليه لحسن…٤٠٠٥يجيء المقتول بالقاتل يوم القيامة ناصيته ورأسه في يده وأوداجه تشخب دما…٤٠٠٦أخبرنا محمد بن المثنى، قال حدثنا الأنصاري، قال حدثنا محمد بن عمرو، عن…٤٠٠٧أخبرني محمد بن بشار، عن عبد الوهاب، قال حدثنا محمد بن عمرو، عن موسى…٤٠٠٨أخبرنا عمرو بن علي، عن مسلم بن إبراهيم، قال حدثنا حماد بن سلمة، عن…٤٠٠٩من جاء يعبد الله ولا يشرك به شيئا ويقيم الصلاة ويؤتي الزكاة ويجتنب…٤٠١٠الكبائر الشرك بالله وعقوق الوالدين وقتل النفس وقول الزور٤٠١١الكبائر الإشراك بالله وعقوق الوالدين وقتل النفس واليمين الغموس٤٠١٢هن سبع أعظمهن إشراك بالله وقتل النفس بغير حق وفرار يوم الزحف٤٠١٣أن تجعل لله ندا وهو خلقك٤٠١٤أن تجعل لله ندا وهو خلقك٤٠١٥الشرك أن تجعل لله ندا وأن تزاني بحليلة جارك وأن تقتل ولدك مخافة الفقر…٤٠١٦والذي لا إله غيره لا يحل دم امرئ مسلم يشهد أن لا إله إلا الله وأني…٤٠١٧لا يحل دم امرئ مسلم إلا رجل زنى بعد إحصانه أو كفر بعد إسلامه أو النفس…٤٠١٨قال قالت عائشة يا عمار أما إنك تعلم أنه لا يحل دم امرئ إلا ثلاثة…٤٠١٩لا يحل دم امرئ مسلم إلا بإحدى ثلاث رجل كفر بعد إسلامه أو زنى بعد…٤٠٢٠إنه سيكون بعدي هنات وهنات فمن رأيتموه فارق الجماعة أو يريد تفريق أمر…٤٠٢١إنها ستكون بعدي هنات وهنات وهنات - ورفع يديه - فمن رأيتموه يريد تفريق…٤٠٢٢ستكون بعدي هنات وهنات فمن أراد أن يفرق أمر أمة محمد صلى الله عليه…٤٠٢٣أيما رجل خرج يفرق بين أمتي فاضربوا عنقه٤٠٢٤ألا تخرجون مع راعينا في إبله فتصيبوا من ألبانها وأبوالها٤٠٢٥أن نفرا، من عكل قدموا على النبي صلى الله عليه وسلم فاجتووا المدينة…٤٠٢٦قال قدم على رسول الله صلى الله عليه وسلم ثمانية نفر من عكل فذكر نحوه…٤٠٢٧قال أتى النبي صلى الله عليه وسلم نفر من عكل أو عرينة فأمر لهم -…٤٠٢٨وغيره، عن حميد الطويل، عن أنس بن مالك، أن ناسا، من عرينة قدموا على…٤٠٢٩لو خرجتم إلى ذودنا فكنتم فيها فشربتم من ألبانها وأبوالها٤٠٣٠لو خرجتم إلى ذودنا فشربتم من ألبانها٤٠٣١لو خرجتم إلى ذود لنا فشربتم من ألبانها٤٠٣٢أن أنس بن مالك، حدثهم أن ناسا أو رجالا من عكل أو عرينة قدموا على رسول…٤٠٣٣نحوه .٤٠٣٤وثابت، عن أنس، أن نفرا، من عرينة نزلوا في الحرة فأتوا النبي صلى الله…٤٠٣٥قال قدم أعراب من عرينة إلى نبي الله صلى الله عليه وسلم فأسلموا…٤٠٣٦اللهم عطش من عطش آل محمد الليلة٤٠٣٧رضى الله عنها قالت أغار قوم على لقاح رسول الله صلى الله عليه وسلم…٤٠٣٨أن قوما، أغاروا على لقاح رسول الله صلى الله عليه وسلم فأتي بهم النبي…٤٠٣٩أن قوما، أغاروا على إبل رسول الله صلى الله عليه وسلم فقطع أيديهم…٤٠٤٠وسعيد بن عبد الرحمن، وذكر، آخر عن هشام بن عروة، عن عروة بن الزبير،…٤٠٤١أخبرنا أحمد بن عمرو بن السرح، قال أخبرني ابن وهب، قال أخبرني عمرو بن…٤٠٤٢أن رسول الله صلى الله عليه وسلم لما قطع الذين سرقوا لقاحه وسمل أعينهم…٤٠٤٣قال إنما سمل النبي صلى الله عليه وسلم أعين أولئك لأنهم سملوا أعين…٤٠٤٤والحارث بن مسكين، قراءة عليه وأنا أسمع، قال حدثنا ابن وهب، قال أخبرني…٤٠٤٥أن رجلا، قتل جارية من الأنصار على حلي لها ثم ألقاها في قليب ورضخ…٤٠٤٦في قوله تعالى { إنما جزاء الذين يحاربون الله ورسوله } الآية قال نزلت…٤٠٤٧أخبرنا محمد بن المثنى، قال حدثنا عبد الصمد، قال حدثنا هشام، عن قتادة،…٤٠٤٨لا يحل دم امرئ مسلم إلا بإحدى ثلاث خصال زان محصن يرجم أو رجل قتل رجلا…٤٠٤٩إذا أبق العبد لم تقبل له صلاة حتى يرجع إلى مواليه٤٠٥٠إذا أبق العبد لم تقبل له صلاة وإن مات مات كافرا٤٠٥١قال إذا أبق العبد إلى أرض الشرك فلا ذمة له .٤٠٥٢إذا أبق العبد إلى أرض الشرك فقد حل دمه٤٠٥٣إذا أبق العبد إلى أرض الشرك فقد حل دمه٤٠٥٤قال أيما عبد أبق إلى أرض الشرك فقد حل دمه .٤٠٥٥قال أيما عبد أبق إلى أرض الشرك فقد حل دمه .٤٠٥٦قال أيما عبد أبق من مواليه ولحق بالعدو فقد أحل بنفسه .٤٠٥٧لا يحل دم امرئ مسلم إلا بإحدى ثلاث رجل زنى بعد إحصانه فعليه الرجم أو…٤٠٥٨لا يحل دم امرئ مسلم إلا بثلاث أن يزني بعد ما أحصن أو يقتل إنسانا…٤٠٥٩من بدل دينه فاقتلوه٤٠٦٠لا تعذبوا بعذاب الله أحدا٤٠٦١من بدل دينه فاقتلوه٤٠٦٢من بدل دينه فاقتلوه٤٠٦٣من بدل دينه فاقتلوه٤٠٦٤من بدل دينه فاقتلوه٤٠٦٥من بدل دينه فاقتلوه٤٠٦٦أن النبي صلى الله عليه وسلم بعثه إلى اليمن ثم أرسل معاذ بن جبل بعد…٤٠٦٧اقتلوهم وإن وجدتموهم متعلقين بأستار الكعبة٤٠٦٨. فأرسل إليه فأسلم .٤٠٦٩قال في سورة النحل { من كفر بالله من بعد إيمانه إلا من أكره } إلى قوله…٤٠٧٠أنشد الله رجلا لي عليه حق فعل ما فعل إلا قام٤٠٧١قال أغلظ رجل لأبي بكر الصديق فقلت أقتله فانتهرني وقال ليس هذا لأحد…٤٠٧٢قال تغيظ أبو بكر على رجل فقلت من هو يا خليفة رسول الله قال لم قلت…٤٠٧٣قال مررت على أبي بكر وهو متغيظ على رجل من أصحابه فقلت يا خليفة رسول…٤٠٧٤قال تغيظ أبو بكر على رجل فقال لو أمرتني لفعلت . قال أما والله ما كانت…٤٠٧٥أخبرنا معاوية بن صالح الأشعري، قال حدثنا عبد الله بن جعفر، قال حدثنا…٤٠٧٦قال أتيت على أبي بكر وقد أغلظ لرجل فرد عليه فقلت ألا أضرب عنقه…٤٠٧٧أخبرني أبو داود، قال حدثنا عفان، قال حدثنا يزيد بن زريع، قال حدثنا…٤٠٧٨لا تشركوا بالله شيئا ولا تسرقوا ولا تزنوا ولا تقتلوا النفس التي حرم…٤٠٧٩من عقد عقدة ثم نفث فيها فقد سحر ومن سحر فقد أشرك ومن تعلق شيئا وكل…٤٠٨٠قال سحر النبي صلى الله عليه وسلم رجل من اليهود فاشتكى لذلك أياما…٤٠٨١أخبرنا هناد بن السري، في حديثه عن أبي الأحوص، عن سماك، عن قابوس، عن…٤٠٨٢قال جاء رجل إلى رسول الله صلى الله عليه وسلم فقال يا رسول الله أرأيت…٤٠٨٣أن رجلا، جاء إلى رسول الله صلى الله عليه وسلم فقال يا رسول الله أرأيت…٤٠٨٤من قاتل دون ماله فقتل فهو شهيد٤٠٨٥من قاتل دون ماله فقتل فهو شهيد٤٠٨٦من قتل دون ماله مظلوما فله الجنة٤٠٨٧من قتل دون ماله فهو شهيد٤٠٨٨من أريد ماله بغير حق فقاتل فقتل فهو شهيد٤٠٨٩من قتل دون ماله فهو شهيد٤٠٩٠من قتل دون ماله فهو شهيد٤٠٩١من قاتل دون ماله فهو شهيد٤٠٩٢من قتل دون ماله فهو شهيد٤٠٩٣من قتل دون مظلمته فهو شهيد٤٠٩٤من قاتل دون ماله فقتل فهو شهيد ومن قاتل دون دمه فهو شهيد ومن قاتل دون…٤٠٩٥من قتل دون ماله فهو شهيد ومن قتل دون أهله فهو شهيد ومن قتل دون دينه…٤٠٩٦من قتل دون مظلمته فهو شهيد٤٠٩٧من شهر سيفه ثم وضعه فدمه هدر٤٠٩٨بهذا الإسناد مثله ولم يرفعه .٤٠٩٩قال من رفع السلاح ثم وضعه فدمه هدر .٤١٠٠من حمل علينا السلاح فليس منا٤١٠١أخبرنا محمود بن غيلان، قال حدثنا عبد الرزاق، قال أنبأنا الثوري، عن…٤١٠٢يخرج قوم في آخر الزمان أحداث الأسنان سفهاء الأحلام يقولون من خير قول…٤١٠٣والله لا تجدون بعدي رجلا هو أعدل مني٤١٠٤قتال المسلم كفر وسبابه فسوق٤١٠٥قال سباب المسلم فسوق وقتاله كفر .٤١٠٦أخبرنا يحيى بن حكيم، قال حدثنا عبد الرحمن بن مهدي، عن شعبة، عن أبي…٤١٠٧قال سباب المسلم فسوق وقتاله كفر .٤١٠٨سباب المسلم فسوق وقتاله كفر٤١٠٩سباب المسلم فسوق مات ميتة جاهلية ومن خرج على أمتي…٤١١٥من قاتل تحت راية عمية يقاتل عصبية ويغضب لعصبية فقتلته جاهلية
