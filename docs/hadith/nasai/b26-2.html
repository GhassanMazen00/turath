سنن النسائي

كتاب النكاح

١٩٣ حديثًا•صفحة ٢ من ٢
٣٣٤٦وسليمان بن داود، عن ابن وهب، أخبرني يونس، عن ابن شهاب، قال أخبرني…٣٣٤٧فقالت فعل رسول الله صلى الله عليه وسلم على اثنتى عشرة أوقية ونش وذلك…٣٣٤٨أخبرنا محمد بن عبد الله بن المبارك، قال حدثنا عبد الرحمن بن مهدي، قال…٣٣٤٩من قتل في سبيل الله أو مات فهو في الجنة٣٣٥٠أن رسول الله صلى الله عليه وسلم تزوجها وهي بأرض الحبشة زوجها النجاشي…٣٣٥١والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ لمحمد - عن ابن…٣٣٥٢يقول قال عبد الرحمن بن عوف رآني رسول الله صلى الله عليه وسلم وعلى…٣٣٥٣أيما امرأة نكحت على صداق أو حباء أو عدة قبل عصمة النكاح فهو لها وما…٣٣٥٤والأسود، قالا أتي عبد الله في رجل تزوج امرأة ولم يفرض لها فتوفي قبل…٣٣٥٥أنه أتي في امرأة تزوجها رجل فمات عنها ولم يفرض لها صداقا ولم يدخل بها…٣٣٥٦أخبرنا إسحاق بن منصور، قال حدثنا عبد الرحمن، قال حدثنا سفيان، عن…٣٣٥٧مثله .٣٣٥٨أخبرنا علي بن حجر، قال حدثنا علي بن مسهر، عن داود بن أبي هند، عن…٣٣٥٩أن رسول الله صلى الله عليه وسلم جاءته امرأة فقالت يا رسول الله إني قد…٣٣٦٠إن كانت أحلتها له جلدته مائة وإن لم تكن أحلتها له رجمته٣٣٦١أن رجلا، يقال له عبد الرحمن بن حنين وينبز قرقورا أنه وقع بجارية…٣٣٦٢إن كانت أحلتها له فأجلده مائة وإن لم تكن أحلتها له فأرجمه٣٣٦٣إن كان استكرهها فهي حرة وعليه لسيدتها مثلها وإن كانت طاوعته فهي له…٣٣٦٤إن كان استكرهها فهي حرة من ماله وعليه الشروى لسيدتها وإن كانت طاوعته…٣٣٦٥وعبد الله، ابنى محمد عن أبيهما، أن عليا، بلغه أن رجلا، لا يرى بالمتعة…٣٣٦٦والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - قال أنبأنا ابن…٣٣٦٧ومحمد بن بشار، ومحمد بن المثنى، قالوا أنبأنا عبد الوهاب، قال سمعت…٣٣٦٨من كان عنده من هذه النساء اللاتي يتمتع فليخل سبيلها٣٣٦٩فصل ما بين الحلال والحرام الدف والصوت في النكاح٣٣٧٠إن فصل ما بين الحلال والحرام الصوت٣٣٧١بارك الله فيكم وبارك لكم٣٣٧٢أن رسول الله صلى الله عليه وسلم رأى على عبد الرحمن أثر صفرة فقال " ما…٣٣٧٣أن عبد الرحمن بن عوف، جاء وعليه ردع من زعفران فقال رسول الله صلى الله…٣٣٧٤قال رأى رسول الله صلى الله عليه وسلم على - كأنه يعني عبد الرحمن بن…٣٣٧٥أن عليا، قال تزوجت فاطمة رضى الله عنها فقلت يا رسول الله ابن بي . قال…٣٣٧٦قال لما تزوج علي رضى الله عنه فاطمة رضى الله عنها قال له رسول الله…٣٣٧٧قالت تزوجني رسول الله صلى الله عليه وسلم في شوال وأدخلت عليه في شوال…٣٣٧٨أخبرنا محمد بن آدم، عن عبدة، عن هشام، عن أبيه، عن عائشة، قالت تزوجني…٣٣٧٩قالت تزوجني رسول الله صلى الله عليه وسلم وهي بنت ست سنين وبنى بها وهي…٣٣٨٠الله أكبر خربت خيبر إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٣٣٨١أنه سمع أنسا، يقول إن رسول الله صلى الله عليه وسلم أقام على صفية بنت…٣٣٨٢قال أقام النبي صلى الله عليه وسلم بين خيبر والمدينة ثلاثا يبني بصفية…٣٣٨٣أخبرنا علي بن حجر، قال حدثنا شريك، عن أبي إسحاق، عن عامر بن سعد، قال…٣٣٨٤رضى الله عنه قال جهز رسول الله صلى الله عليه وسلم فاطمة في خميل وقربة…٣٣٨٥فراش للرجل وفراش لأهله والثالث للضيف والرابع للشيطان٣٣٨٦أخبرنا قتيبة، قال حدثنا سفيان، عن ابن المنكدر، عن جابر، قال قال لي…٣٣٨٧ضعه - ثم قال - اذهب فادع فلانا وفلانا ومن لقيت٣٣٨٨أخبرنا أحمد بن يحيى بن الوزير، قال حدثنا سعيد بن كثير بن عفير، قال…
