سنن النسائي

كتاب الأشربة

٢٢٠ حديثًا•صفحة ٢ من ٢
٥٦٨٩أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن عيينة بن عبد الرحمن، عن…٥٦٩٠قال نبيذ البسر بحتا لا يحل .٥٦٩١فنهى عنه . قلت يا أبا عباس إني أنتبذ في جرة خضراء نبيذا حلوا فأشرب…٥٦٩٢مرحبا بالوفد ليس بالخزايا ولا النادمين٥٦٩٣أخبرنا سويد، قال أنبأنا عبد الله، عن سليمان التيمي، عن قيس بن هنان،…٥٦٩٤قال قال ابن عمر رأيت رجلا جاء إلى رسول الله صلى الله عليه وسلم بقدح…٥٦٩٥وأخبرنا زياد بن أيوب، عن أبي معاوية، قال حدثنا أبو إسحاق الشيباني، عن…٥٦٩٦أن رجلا، سأل عن الأشربة، فقال اجتنب كل شىء ينش .٥٦٩٧فقال اجتنب كل شىء ينش .٥٦٩٨قال المسكر قليله وكثيره حرام .٥٦٩٩قال كل مسكر خمر وكل مسكر حرام .٥٧٠٠حرم الله الخمر وكل مسكر حرام٥٧٠١كل مسكر حرام وكل مسكر خمر٥٧٠٢قال حدثتني رقية بنت عمرو بن سعيد، قالت كنت في حجر ابن عمر فكان ينقع…٥٧٠٣على بذنوب من زمزم٥٧٠٤أدنه مني يا أبا هريرة٥٧٠٥أخبرنا سويد، قال أنبأنا عبد الله، عن السري بن يحيى، قال حدثنا أبو…٥٧٠٦سمع سعيد بن المسيب، يقول تلقت ثقيف عمر بشراب فدعا به فلما قربه إلى…٥٧٠٧قال كان النبيذ الذي يشربه عمر بن الخطاب قد خلل . ومما يدل على صحة هذا…٥٧٠٨قال الحارث بن مسكين قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٥٧٠٩أن رجلا، من جيشان - وجيشان من اليمن - قدم فسأل رسول الله صلى الله…٥٧١٠إن الحلال بين وإن الحرام بين وإن بين ذلك أمورا مشتبهات٥٧١١دع ما يريبك إلى ما لا يريبك٥٧١٢أنه كان يكره أن يبيع الزبيب، لمن يتخذه نبيذا .٥٧١٣أخبرنا سويد، قال أنبأنا عبد الله، عن سفيان بن دينار، عن مصعب بن سعد،…٥٧١٤قال بعه عصيرا ممن يتخذه طلاء ولا يتخذه خمرا .٥٧١٥قال كتب عمر بن الخطاب إلى بعض عماله أن ارزق، المسلمين من الطلاء ما…٥٧١٦أنه قال قرأت كتاب عمر بن الخطاب إلى أبي موسى أما بعد فإنها قدمت على…٥٧١٧أن عبد الله بن يزيد الخطمي، قال كتب إلينا عمر بن الخطاب رضى الله عنه…٥٧١٨قال كان علي رضى الله عنه يرزق الناس الطلاء يقع فيه الذباب ولا يستطيع…٥٧١٩قال سألت سعيدا ما الشراب الذي أحله عمر رضى الله عنه قال الذي يطبخ حتى…٥٧٢٠أن أبا الدرداء، كان يشرب ما ذهب ثلثاه وبقي ثلثه .٥٧٢١أنه كان يشرب من الطلاء ما ذهب ثلثاه وبقي ثلثه .٥٧٢٢وسأله، أعرابي عن شراب، يطبخ على النصف فقال لا حتى يذهب ثلثاه ويبقى…٥٧٢٣قال إذا طبخ الطلاء على الثلث فلا بأس به .٥٧٢٤فقال لا تشربه .٥٧٢٥قال سألت الحسن عما يطبخ من العصير قال ما تطبخه حتى يذهب الثلثان ويبقى…٥٧٢٦يقول إن نوحا صلى الله عليه وسلم نازعه الشيطان في عود الكرم فقال هذا…٥٧٢٧تشربوا من الطلاء حتى يذهب ثلثاه ويبقى ثلثه وكل مسكر حرام .٥٧٢٨قال كل مسكر حرام .٥٧٢٩فقال اشربه ما كان طريا . قال إني طبخت شرابا وفي نفسي منه . قال أكنت…٥٧٣٠يقول والله ما تحل النار شيئا ولا تحرمه . قال ثم فسر لي قوله لا تحل…٥٧٣١قال اشرب العصير ما لم يزبد .٥٧٣٢قال اشربه حتى يغلي ما لم يتغير .٥٧٣٣في العصير قال اشربه حتى يغلي .٥٧٣٤قال اشربه ثلاثة أيام إلا أن يغلي .٥٧٣٥أخبرني عمرو بن عثمان بن سعيد بن كثير، قال حدثنا بقية، قال حدثني…٥٧٣٦أخبرنا عيسى بن محمد أبو عمير بن النحاس، عن ضمرة، عن الشيباني، عن ابن…٥٧٣٧قال كان ينبذ لرسول الله صلى الله عليه وسلم فيشربه من الغد ومن بعد…٥٧٣٨أن رسول الله صلى الله عليه وسلم كان ينقع له الزبيب فيشربه يومه والغد…٥٧٣٩قال كان رسول الله صلى الله عليه وسلم ينبذ له نبيذ الزبيب من الليل…٥٧٤٠أخبرنا سويد، قال أنبأنا عبد الله، عن عبيد الله، عن نافع، عن ابن عمر،…٥٧٤١قال كان علي بن حسين رضى الله عنه ينبذ له من الليل فيشربه غدوة وينبذ…٥٧٤٢قال انتبذ عشيا واشربه غدوة .٥٧٤٣وليس، بالنهدي أن أم الفضل، أرسلت إلى أنس بن مالك تسأله عن نبيذ الجر…٥٧٤٤أنه كان يكره أن يجعل، نطل النبيذ في النبيذ ليشتد بالنطل .٥٧٤٥أنه قال في النبيذ خمره درديه .٥٧٤٦قال إنما سميت الخمر لأنها تركت حتى مضى صفوها وبقي كدرها . وكان يكره…٥٧٤٧قال كانوا يرون أن من، شرب شرابا فسكر منه لم يصلح له أن يعود فيه .٥٧٤٨قال لا بأس بنبيذ البختج .٥٧٤٩أخبرنا سويد، قال أنبأنا عبد الله، عن أبي عوانة، عن أبي مسكين، قال…٥٧٥٠قال رحم الله إبراهيم شدد الناس في النبيذ ورخص فيه .٥٧٥١حدثنا عبيد الله بن سعيد، عن أبي أسامة، قال سمعت ابن المبارك، يقول ما…٥٧٥٢يقول ما رأيت رجلا أطلب للعلم من عبد الله بن المبارك الشامات ومصر…٥٧٥٣رضى الله عنه قال كان لأم سليم قدح من عيدان فقالت سقيت فيه رسول الله…٥٧٥٤فقال اشرب الماء واشرب العسل واشرب السويق واشرب اللبن الذي نجعت به .…٥٧٥٥قال أحدث الناس أشربة ما أدري ما هي فما لي شراب منذ عشرين سنة أو قال…٥٧٥٦قال أحدث الناس أشربة ما أدري ما هي وما لي شراب منذ عشرين سنة إلا…٥٧٥٧قال قال طلحة لأهل الكوفة في النبيذ فتنة يربو فيها الصغير ويهرم فيها…٥٧٥٨قال كان ابن شبرمة لا يشرب إلا الماء واللبن .
