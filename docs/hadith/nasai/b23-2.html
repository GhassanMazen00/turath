سنن النسائي

كتاب الزكاة

١٨٤ حديثًا•صفحة ٢ من ٢
٢٥٨٥ما يزال الرجل يسأل حتى يأتي يوم القيامة ليس في وجهه مزعة من لحم٢٥٨٦لو تعلمون ما في المسألة ما مشى أحد إلى أحد يسأله شيئا٢٥٨٧لا وإن كنت سائلا لا بد فاسأل الصالحين٢٥٨٨ما يكون عندي من خير فلن أدخره عنكم ومن يستعفف يعفه الله عز وجل ومن…٢٥٨٩والذي نفسي بيده لأن يأخذ أحدكم حبله فيحتطب على ظهره خير له من أن يأتي…٢٥٩٠من يضمن لي واحدة وله الجنة٢٥٩١لا تصلح المسألة إلا لثلاثة رجل أصابت ماله جائحة فيسأل حتى يصيب سدادا…٢٥٩٢من سأل وله ما يغنيه جاءت خموشا أو كدوحا في وجهه يوم القيامة٢٥٩٣لا تلحفوا في المسألة ولا يسألني أحد منكم شيئا وأنا له كاره فيبارك له…٢٥٩٤من سأل وله أربعون درهما فهو الملحف٢٥٩٥من استغنى أغناه الله عز وجل ومن استعف أعفه الله عز وجل ومن استكفى…٢٥٩٦قال الحارث بن مسكين قراءة عليه وأنا أسمع، عن ابن القاسم، قال أنبأنا…٢٥٩٧لا تحل الصدقة لغني ولا لذي مرة سوي٢٥٩٨إن شئتما ولا حظ فيها لغني ولا لقوي مكتسب٢٥٩٩إن المسائل كدوح يكدح بها الرجل وجهه فمن شاء كدح وجهه ومن شاء ترك إلا…٢٦٠٠المسألة كد يكد بها الرجل وجهه إلا أن يسأل الرجل سلطانا أو في أمر لا…٢٦٠١يا حكيم إن هذا المال خضرة حلوة فمن أخذه بطيب نفس بورك له فيه ومن أخذه…٢٦٠٢يا حكيم إن هذا المال خضرة حلوة من أخذه بسخاوة نفس بورك له فيه ومن…٢٦٠٣يا حكيم إن هذا المال حلوة فمن أخذه بسخاوة نفس بورك له فيه ومن أخذه…٢٦٠٤إذا أعطيت شيئا من غير أن تسأل فكل وتصدق٢٦٠٥ما آتاك الله عز وجل من هذا المال من غير مسألة ولا إشراف فخذه فتموله…٢٦٠٦خذه فتموله أو تصدق به ما جاءك من هذا المال وأنت غير مشرف ولا سائل…٢٦٠٧خذه فتموله وتصدق به فما جاءك من هذا المال وأنت غير مشرف ولا سائل فخذه…٢٦٠٨خذه فتموله وتصدق به وما جاءك من هذا المال وأنت غير مشرف ولا سائل فخذه…٢٦٠٩إن هذه الصدقة إنما هي أوساخ الناس وإنها لا تحل لمحمد ولا لآل محمد صلى…٢٦١٠ابن أخت القوم من أنفسهم٢٦١١ابن أخت القوم منهم٢٦١٢إن الصدقة لا تحل لنا وإن مولى القوم منهم٢٦١٣أهدية أم صدقة٢٦١٤اشتريها واعتقيها فإن الولاء لمن أعتق٢٦١٥لا تشتره وإن أعطاكه بدرهم فإن العائد في صدقته كالكلب يعود في قيئه٢٦١٦لا تعرض في صدقتك٢٦١٧لا تعد في صدقتك٢٦١٨ويزيد، قالا حدثنا عبد الرحمن بن إسحاق، عن الزهري، عن سعيد بن المسيب،…
