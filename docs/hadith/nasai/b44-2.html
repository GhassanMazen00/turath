سنن النسائي

كتاب البيوع

٢٥٧ حديثًا•صفحة ٢ من ٢
٤٥٩٩يقول أما الذي نهى عنه رسول الله صلى الله عليه وسلم أن يباع حتى يستوفى…٤٦٠٠من ابتاع طعاما فلا يبيعه حتى يقبضه٤٦٠١لا تبع طعاما حتى تشتريه وتستوفيه٤٦٠٢أخبرنا إبراهيم بن الحسن، قال حدثنا حجاج، قال قال ابن جريج وأخبرني…٤٦٠٣لا تبعه حتى تقبضه٤٦٠٤والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن وهب، قال أخبرني عمرو…٤٦٠٥والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن القاسم،…٤٦٠٦أنهم كانوا يبتاعون على عهد رسول الله صلى الله عليه وسلم في أعلى السوق…٤٦٠٧أن ابن عمر، حدثهم أنهم، كانوا يبتاعون الطعام على عهد رسول الله صلى…٤٦٠٨قال رأيت الناس يضربون على عهد رسول الله صلى الله عليه وسلم إذا اشتروا…٤٦٠٩قالت اشترى رسول الله صلى الله عليه وسلم من يهودي طعاما إلى أجل ورهنه…٤٦١٠أنه مشى إلى رسول الله صلى الله عليه وسلم بخبز شعير وإهالة سنخة . قال…٤٦١١لا يحل سلف وبيع ولا شرطان في بيع ولا بيع ما ليس عندك٤٦١٢ليس على رجل بيع فيما لا يملك٤٦١٣لا تبع ما ليس عندك٤٦١٤أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن شعبة، عن عبد الله بن أبي…٤٦١٥أخبرنا محمود بن غيلان، قال حدثنا أبو داود، قال أنبأنا شعبة، قال حدثنا…٤٦١٦من أسلف سلفا فليسلف في كيل معلوم ووزن معلوم إلى أجل معلوم٤٦١٧انطلق فابتع له بكرا٤٦١٨أخبرنا عمرو بن منصور، قال حدثنا أبو نعيم، قال حدثنا سفيان، عن سلمة بن…٤٦١٩أجل لا أقضيكها إلا نجيبة٤٦٢٠ويزيد بن زريع، وخالد بن الحارث، قالوا حدثنا شعبة، وأخبرني أحمد بن…٤٦٢١قال جاء عبد فبايع رسول الله صلى الله عليه وسلم على الهجرة ولا يشعر…٤٦٢٢السلف في حبل الحبلة ربا٤٦٢٣أخبرنا محمد بن منصور، قال حدثنا سفيان، عن أيوب، عن سعيد بن جبير، عن…٤٦٢٤أخبرنا قتيبة، قال حدثنا الليث، عن نافع، عن ابن عمر، أن النبي صلى الله…٤٦٢٥والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن القاسم،…٤٦٢٦أخبرنا محمد بن منصور، قال حدثنا سفيان، عن أبي الزبير، عن جابر، قال…٤٦٢٧أخبرنا إسحاق بن منصور، قال حدثنا سفيان، عن حميد الأعرج، عن سليمان، -…٤٦٢٨كذب قد علم أني من أتقاهم لله وآداهم للأمانة٤٦٢٩أخبرنا إسماعيل بن مسعود، عن خالد، عن حسين المعلم، عن عمرو بن شعيب، عن…٤٦٣٠لا يحل سلف وبيع ولا شرطان في بيع ولا ربح ما لم يضمن٤٦٣١أخبرنا محمد بن رافع، قال حدثنا عبد الرزاق، قال حدثنا معمر، عن أيوب،…٤٦٣٢ويعقوب بن إبراهيم، ومحمد بن المثنى، قالوا حدثنا يحيى بن سعيد، قال…٤٦٣٣أخبرنا زياد بن أيوب، قال حدثنا عباد بن العوام، قال حدثنا سفيان بن…٤٦٣٤أخبرنا علي بن حجر، قال حدثنا إسماعيل بن إبراهيم، عن أيوب، وأخبرنا…٤٦٣٥أيما امرئ أبر نخلا ثم باع أصلها فللذي أبر ثمر النخل إلا أن يشترط…٤٦٣٦من ابتاع نخلا بعد أن تؤبر فثمرتها للبائع إلا أن يشترط المبتاع ومن باع…٤٦٣٧أخبرنا علي بن حجر، قال أنبأنا سعدان بن يحيى، عن زكريا، عن عامر، عن…٤٦٣٨يا جابر ما أرى جملك إلا قد انتشط٤٦٣٩ما لك في آخر الناس٤٦٤٠أخبرنا محمد بن منصور، قال حدثنا سفيان، عن أبي الزبير، عن جابر، قال…٤٦٤١أتبيعنيه بكذا وكذا والله يغفر لك٤٦٤٢أعتقيها فإن الولاء لمن أعطى الورق٤٦٤٣اشتريها فأعتقيها فإن الولاء لمن أعتق٤٦٤٤لا يمنعك ذلك فإن الولاء لمن أعتق٤٦٤٥أخبرنا أحمد بن حفص بن عبد الله، قال حدثني أبي قال، حدثني إبراهيم، عن…٤٦٤٦الشفعة في كل شرك ربعة أو حائط لا يصلح له أن يبيع حتى يؤذن شريكه فإن…٤٦٤٧أليس قد ابتعته منك٤٦٤٨إذا اختلف البيعان وليس بينهما بينة فهو ما يقول رب السلعة أو يتركا٤٦٤٩ويوسف بن سعيد، وعبد الرحمن بن خالد، - واللفظ لإبراهيم - قالوا حدثنا…٤٦٥٠أخبرنا أحمد بن حرب، قال حدثنا أبو معاوية، عن الأعمش، عن إبراهيم، عن…٤٦٥١قال توفي رسول الله صلى الله عليه وسلم ودرعه مرهونة عند يهودي بثلاثين…٤٦٥٢أخبرنا قتيبة، قال حدثنا الليث، عن أبي الزبير، عن جابر، قال أعتق رجل…٤٦٥٣أن رجلا، من الأنصار يقال له أبو مذكور أعتق غلاما له عن دبر يقال له…٤٦٥٤وابن أبي خالد، عن سلمة بن كهيل، عن عطاء، عن جابر، أن النبي صلى الله…٤٦٥٥ابتاعي وأعتقي فإن الولاء لمن أعتق٤٦٥٦لا يمنعك ذلك منها ابتاعي وأعتقي فإن الولاء لمن أعتق٤٦٥٧أخبرنا إسماعيل بن مسعود، قال حدثنا خالد، قال حدثنا عبيد الله، عن عبد…٤٦٥٨أخبرنا قتيبة بن سعيد، قال حدثنا مالك، عن عبد الله بن دينار، عن ابن…٤٦٥٩أخبرنا علي بن حجر، قال حدثنا إسماعيل بن إبراهيم، عن شعبة، عن عبد الله…٤٦٦٠أخبرنا الحسين بن حريث، قال حدثنا الفضل بن موسى السيناني، عن حسين بن…٤٦٦١وعبد الله بن محمد بن عبد الرحمن، - واللفظ له - قالا حدثنا سفيان، عن…٤٦٦٢أخبرنا قتيبة بن سعيد، قال حدثنا داود، عن عمرو، عن أبي المنهال، عن…٤٦٦٣أن أبا المنهال، أخبره أن إياس بن عبد صاحب النبي صلى الله عليه وسلم…٤٦٦٤هل علمت أن الله عز وجل حرمها٤٦٦٥قالت لما نزلت آيات الربا قام رسول الله صلى الله عليه وسلم على المنبر…٤٦٦٦أنه سمع أبا مسعود، عقبة بن عمرو قال نهى رسول الله صلى الله عليه وسلم…٤٦٦٧قال قال رسول الله صلى الله عليه وسلم في أشياء حرمها " وثمن الكلب " .٤٦٦٨أخبرني إبراهيم بن الحسن، قال أنبأنا حجاج بن محمد، عن حماد بن سلمة، عن…٤٦٦٩إن الله ورسوله حرم بيع الخمر والميتة والخنزير والأصنام٤٦٧٠أنه سمع جابرا، يقول نهى رسول الله صلى الله عليه وسلم عن بيع ضراب…٤٦٧١أخبرنا إسحاق بن إبراهيم، قال حدثنا إسماعيل بن إبراهيم، عن علي بن…٤٦٧٢أخبرنا عصمة بن الفضل، قال حدثنا يحيى بن آدم، عن إبراهيم بن حميد…٤٦٧٣حدثنا محمد بن بشار، عن محمد، قال حدثنا شعبة، عن المغيرة، قال سمعت ابن…٤٦٧٤أخبرني محمد بن علي بن ميمون، قال حدثنا محمد، قال حدثنا سفيان، عن…٤٦٧٥أخبرنا واصل بن عبد الأعلى، قال حدثنا ابن فضيل، عن الأعمش، عن أبي…٤٦٧٦أيما امرئ أفلس ثم وجد رجل عنده سلعته بعينها فهو أولى به من غيره٤٦٧٧وإبراهيم بن الحسن، - واللفظ له - قال حدثنا حجاج بن محمد، قال قال ابن…٤٦٧٨وعمرو بن الحارث، عن بكير بن الأشج، عن عياض بن عبد الله، عن أبي سعيد…٤٦٧٩أن رسول الله صلى الله عليه وسلم قضى أنه إذا وجدها في يد الرجل غير…٤٦٨٠ولقد، أخبرني عكرمة بن خالد، أن أسيد بن حضير الأنصاري، ثم أحد بني…٤٦٨١الرجل أحق بعين ماله إذا وجده ويتبع البائع من باعه٤٦٨٢أيما امرأة زوجها وليان فهي للأول منهما ومن باع بيعا من رجلين فهو…٤٦٨٣بارك الله لك في أهلك ومالك إنما جزاء السلف الحمد والأداء٤٦٨٤سبحان الله ماذا نزل من التشديد٤٦٨٥أها هنا من بني فلان أحد٤٦٨٦ما من أحد يدان دينا فعلم الله أنه يريد قضاءه إلا أداه الله عنه في…٤٦٨٧من أخذ دينا وهو يريد أن يؤديه أعانه الله عز وجل٤٦٨٨إذا أتبع أحدكم على مليء فليتبع والظلم مطل الغني٤٦٨٩لى الواجد يحل عرضه وعقوبته٤٦٩٠لى الواجد يحل عرضه وعقوبته٤٦٩١مطل الغني ظلم وإذا أتبع أحدكم على مليء فليتبع٤٦٩٢إن على صاحبكم دينا٤٦٩٣خياركم أحسنكم قضاء٤٦٩٤إن رجلا لم يعمل خيرا قط وكان يداين الناس فيقول لرسوله خذ ما تيسر…٤٦٩٥كان رجل يداين الناس وكان إذا رأى إعسار المعسر قال لفتاه تجاوز عنه لعل…٤٦٩٦أدخل الله عز وجل رجلا كان سهلا مشتريا وبائعا وقاضيا ومقتضيا الجنة٤٦٩٧وسعد، يوم بدر فجاء سعد بأسيرين ولم أجئ أنا وعمار بشىء .٤٦٩٨من أعتق شركا له في عبد أتم ما بقي في ماله إن كان له مال يبلغ ثمن العبد٤٦٩٩من أعتق شركا له في مملوك وكان له من المال ما يبلغ ثمنه بقيمة العبد…٤٧٠٠أيكم كانت له أرض أو نخل فلا يبعها حتى يعرضها على شريكه٤٧٠١قال قضى رسول الله صلى الله عليه وسلم بالشفعة في كل شركة لم تقسم ربعة…٤٧٠٢الجار أحق بسقبه٤٧٠٣الجار أحق بسقبه٤٧٠٤الشفعة في كل مال لم يقسم فإذا وقعت الحدود وعرفت الطرق فلا شفعة٤٧٠٥قال قضى رسول الله صلى الله عليه وسلم بالشفعة والجوار .
