٢٢٤٠من استطاع منكم الباءة فليتزوج فإنه أغض للبصر وأحصن للفرج ومن لم يستطع…٢٢٤١من استطاع منكم الباءة فليتزوج ومن لم يجد فعليه بالصوم فإنه له وجاء٢٢٤٢يا معشر الشباب من استطاع منكم الباءة فليتزوج فإنه أغض للبصر وأحصن…٢٢٤٣من كان منكم ذا طول فليتزوج فإنه أغض للبصر وأحصن للفرج ومن لا فالصوم…٢٢٤٤من صام يوما في سبيل الله عز وجل زحزح الله وجهه عن النار بذلك اليوم…٢٢٤٥من صام يوما في سبيل الله باعد الله بينه وبين النار بذلك اليوم سبعين…٢٢٤٦من صام يوما في سبيل الله باعد الله عز وجل وجهه عن النار سبعين خريفا٢٢٤٧من صام يوما في سبيل الله عز وجل باعد الله وجهه من جهنم سبعين عاما٢٢٤٨ما من عبد يصوم يوما في سبيل الله عز وجل إلا بعد الله عز وجل بذلك…٢٢٤٩من صام يوما في سبيل الله عز وجل باعده الله عن النار سبعين خريفا٢٢٥٠من صام يوما في سبيل الله تبارك وتعالى باعد الله وجهه عن النار سبعين…٢٢٥١لا يصوم عبد يوما في سبيل الله إلا باعد الله تعالى بذلك اليوم النار عن…٢٢٥٢من صام يوما في سبيل الله باعد الله بذلك اليوم حر جهنم عن وجهه سبعين…٢٢٥٣من صام يوما في سبيل الله باعد الله بذلك اليوم النار عن وجهه سبعين…٢٢٥٤من صام يوما في سبيل الله عز وجل باعد الله منه جهنم مسيرة مائة عام٢٢٥٥ليس من البر الصيام في السفر٢٢٥٦ليس من البر الصيام في السفر٢٢٥٧ليس من البر الصيام في السفر٢٢٥٨ما بال صاحبكم هذا٢٢٥٩سمع جابرا، نحوه .٢٢٦٠ليس من البر الصيام في السفر عليكم برخصة الله عز وجل فاقبلوها٢٢٦١ليس من البر الصيام في السفر٢٢٦٢ليس من البر الصيام في السفر٢٢٦٣قال خرج رسول الله صلى الله عليه وسلم إلى مكة عام الفتح في رمضان فصام…٢٢٦٤وعبد الرحمن بن محمد بن سلام، قالا حدثنا أبو داود، عن سفيان، عن…٢٢٦٥قال بينما رسول الله صلى الله عليه وسلم يتغدى بمر الظهران ومعه أبو بكر…٢٢٦٦أن رسول الله صلى الله عليه وسلم وأبا بكر وعمر كانوا بمر الظهران مرسل .٢٢٦٧انتظر الغداء يا أبا أمية٢٢٦٨ألا تنتظر الغداء يا أبا أمية٢٢٦٩انتظر الغداء يا أبا أمية٢٢٧٠أنه قدم على النبي صلى الله عليه وسلم فذكر نحوه .٢٢٧١انتظر الغداء يا أبا أمية٢٢٧٢ألا تنتظر الغداء٢٢٧٣أن أبا أمية، أخبره أنه، أتى النبي صلى الله عليه وسلم من سفر نحوه .٢٢٧٤إن الله وضع عن المسافر نصف الصلاة والصوم وعن الحبلى والمرضع٢٢٧٥أخبرنا محمد بن حاتم، قال حدثنا حبان، عوانة، عن أبي بشر، عن هانئ بن الشخير، عن…٢٢٨٠أخبرنا عبد الرحمن بن محمد بن سلام، قال حدثنا أبو داود، قال حدثنا أبو…٢٢٨١أخبرنا عبيد الله بن عبد الكريم، قال حدثنا سهل بن بكار، قال حدثنا أبو…٢٢٨٢أخبرنا
