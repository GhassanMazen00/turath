سنن النسائي

كتاب المزارعة

٩٦ حديثًا
٣٧٦١لا ومقلب القلوب٣٧٦٢لا ومصرف القلوب٣٧٦٣لما خلق الله الجنة والنار أرسل جبريل عليه السلام إلى الجنة فقال انظر…٣٧٦٤من كان حالفا فلا يحلف إلا بالله٣٧٦٥إن الله ينهاكم أن تحلفوا بآبائكم٣٧٦٦إن الله ينهاكم أن تحلفوا بآبائكم٣٧٦٧إن الله ينهاكم أن تحلفوا بآبائكم٣٧٦٨إن الله ينهاكم أن تحلفوا بآبائكم٣٧٦٩لا تحلفوا بآبائكم ولا بأمهاتكم ولا بالأنداد ولا تحلفوا إلا بالله ولا…٣٧٧٠من حلف بملة سوى الإسلام كاذبا فهو كما قال٣٧٧١من حلف بملة سوى الإسلام كاذبا فهو كما قال ومن قتل نفسه بشىء عذب به في…٣٧٧٢من قال إني بريء من الإسلام فإن كان كاذبا فهو كما قال وإن كان صادقا لم…٣٧٧٣- امرأة من جهينة - أن يهوديا، أتى النبي صلى الله عليه وسلم فقال إنكم…٣٧٧٤لا تحلفوا بآبائكم ولا بالطواغيت٣٧٧٥من حلف منكم فقال باللات فليقل لا إله إلا الله ومن قال لصاحبه تعال…٣٧٧٦قل لا إله إلا الله وحده لا شريك له ثلاث مرات وتعوذ بالله من الشيطان…٣٧٧٧قل لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…٣٧٧٨ومحمد بن بشار، عن محمد، قال حدثنا شعبة، عن الأشعث بن سليم، عن معاوية…٣٧٧٩ما على الأرض يمين أحلف عليها فأرى غيرها خيرا منها إلا أتيته٣٧٨٠والله لا أحملكم وما عندي ما أحملكم٣٧٨١من حلف على يمين فرأى غيرها خيرا منها فليكفر عن يمينه وليأت الذي هو خير٣٧٨٢إذا حلف أحدكم على يمين فرأى غيرها خيرا منها فليكفر عن يمينه ولينظر…٣٧٨٣إذا حلفت على يمين فكفر عن يمينك ثم ائت الذي هو خير٣٧٨٤إذا حلفت على يمين فرأيت غيرها خيرا منها فكفر عن يمينك وائت الذي هو خير٣٧٨٥من حلف على يمين فرأى غيرها خيرا منها فليأت الذي هو خير وليكفر عن يمينه٣٧٨٦من حلف على يمين فرأى غيرها خيرا منها فليدع يمينه وليأت الذي هو خير…٣٧٨٧من حلف على يمين فرأى خيرا منها فليأت الذي هو خير وليترك يمينه٣٧٨٨أخبرنا محمد بن منصور، عن سفيان، قال حدثنا أبو الزعراء، عن عمه أبي…٣٧٨٩إذا آليت على يمين فرأيت غيرها خيرا منها فأت الذي هو خير وكفر عن يمينك٣٧٩٠إذا حلفت على يمين فرأيت غيرها خيرا منها فأت الذي هو خير منها وكفر عن…٣٧٩١إذا حلفت على يمين فرأيت غيرها خيرا منها فأت الذي هو خير وكفر عن يمينك٣٧٩٢لا نذر ولا يمين فيما لا تملك ولا في معصية ولا قطيعة رحم٣٧٩٣من حلف فاستثنى فإن شاء مضى وإن شاء ترك غير حنث٣٧٩٤إنما الأعمال بالنية وإنما لامرئ ما نوى فمن كانت هجرته إلى الله ورسوله…٣٧٩٥لا بل شربت عسلا عند زينب بنت جحش ولن أعود له٣٧٩٦كل فنعم الإدام الخل٣٧٩٧يا معشر التجار إن هذا البيع يحضره الحلف والكذب فشوبوا بيعكم بالصدقة٣٧٩٨وعاصم، وجامع، عن أبي وائل، عن قيس بن أبي غرزة، قال كنا نبيع بالبقيع…٣٧٩٩إن هذه السوق يخالطها اللغو والكذب فشوبوها بالصدقة٣٨٠٠يا معشر التجار إنه يشهد بيعكم الحلف والكذب فشوبوه بالصدقة٣٨٠١إنه لا يأتي بخير إنما يستخرج به من البخيل٣٨٠٢إنه لا يرد شيئا إنما يستخرج به من الشحيح٣٨٠٣النذر لا يقدم شيئا ولا يؤخره إنما هو شىء يستخرج به من الشحيح٣٨٠٤لا يأتي النذر على ابن آدم شيئا لم يقدره عليه ولكنه شىء يستخرج به من…٣٨٠٥لا تنذروا فإن النذر لا يغني من القدر شيئا وإنما يستخرج به من البخيل٣٨٠٦من نذر أن يطيع الله فليطعه ومن نذر أن يعصي الله فلا يعصه٣٨٠٧من نذر أن يطيع الله فليطعه ومن نذر أن يعصي الله فلا يعصه٣٨٠٨من نذر أن يطيع الله فليطعه ومن نذر أن يعصي الله فلا يعصه٣٨٠٩خيركم قرني ثم الذين يلونهم ثم الذين يلونهم ثم الذين يلونهم٣٨١٠قال مر رسول الله صلى الله عليه وسلم برجل يقود رجلا في قرن فتناوله…٣٨١١أن طاوسا، أخبره عن ابن عباس، أن النبي صلى الله عليه وسلم مر برجل وهو…٣٨١٢لا نذر في معصية الله ولا فيما لا يملك ابن آدم٣٨١٣من حلف بملة سوى ملة الإسلام كاذبا فهو كما قال ومن قتل نفسه بشىء في…٣٨١٤قال نذرت أختي أن تمشي، إلى بيت الله فأمرتني أن أستفتي لها رسول الله…٣٨١٥مرها فلتختمر ولتركب ولتصم ثلاثة أيام٣٨١٦قال ركبت امرأة البحر فنذرت أن تصوم شهرا فماتت قبل أن تصوم فأتت أختها…٣٨١٧والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن سفيان، عن…٣٨١٨قال استفتى سعد بن عبادة رسول الله صلى الله عليه وسلم في نذر كان على…٣٨١٩وهارون بن إسحاق الهمداني، عن عبدة، عن هشام، - وهو ابن عروة - عن بكر…٣٨٢٠أنه كان عليه ليلة نذر في الجاهلية يعتكفها فسأل رسول الله صلى الله…٣٨٢١أخبرنا محمد بن عبد الله بن يزيد، قال حدثنا سفيان، عن أيوب، عن نافع،…٣٨٢٢أن عمر، كان جعل عليه يوما يعتكفه في الجاهلية فسأل رسول الله صلى الله…٣٨٢٣أمسك عليك بعض مالك فهو خير لك٣٨٢٤أمسك عليك بعض مالك فهو خير لك٣٨٢٥أمسك عليك مالك فهو خير لك٣٨٢٦أمسك عليك بعض مالك فهو خير لك٣٨٢٧كلا والذي نفسي بيده إن الشملة التي أخذها يوم خيبر من المغانم لتشتعل…٣٨٢٨من حلف فقال إن شاء الله فقد استثنى٣٨٢٩من حلف فقال إن شاء الله فقد استثنى٣٨٣٠من حلف على يمين فقال إن شاء الله فهو بالخيار إن شاء أمضى وإن شاء ترك٣٨٣١قال سليمان بن داود لأطوفن الليلة على تسعين امرأة كلهن يأتي بفارس…٣٨٣٢كفارة النذر كفارة اليمين٣٨٣٣لا نذر في معصية٣٨٣٤لا نذر في معصية وكفارته كفارة اليمين٣٨٣٥لا نذر في معصية وكفارته كفارة يمين٣٨٣٦لا نذر في معصية وكفارته كفارة يمين٣٨٣٧لا نذر في معصية وكفارته كفارة اليمين٣٨٣٨لا نذر في معصية وكفارتها كفارة اليمين٣٨٣٩لا نذر في معصية وكفارتها كفارة يمين٣٨٤٠لا نذر في معصية وكفارته كفارة يمين٣٨٤١لا نذر في معصية وكفارتها كفارة يمين٣٨٤٢لا نذر في غضب وكفارته كفارة اليمين٣٨٤٣لا نذر في غضب وكفارته كفارة اليمين٣٨٤٤لا نذر في غضب وكفارته كفارة اليمين٣٨٤٥النذر نذران فما كان من نذر في طاعة الله فذلك لله وفيه الوفاء وما كان…٣٨٤٦لا نذر في غضب وكفارته كفارة يمين٣٨٤٧لا نذر في معصية ولا غضب وكفارته كفارة يمين٣٨٤٨لا نذر في المعصية وكفارته كفارة اليمين٣٨٤٩لا نذر لابن آدم فيما لا يملك ولا في معصية الله عز وجل٣٨٥٠لا نذر في معصية ولا فيما لا يملك ابن آدم٣٨٥١لا نذر في معصية ولا فيما لا يملك ابن آدم٣٨٥٢أخبرنا إسحاق بن إبراهيم، قال أنبأنا حماد بن مسعدة، عن حميد، عن ثابت،…٣٨٥٣أخبرنا محمد بن المثنى، قال حدثنا خالد، قال حدثنا حميد، عن ثابت، عن…٣٨٥٤قال أتى رسول الله صلى الله عليه وسلم على رجل يهادى بين ابنيه فقال "…٣٨٥٥من حلف على يمين فقال إن شاء الله فقد استثنى٣٨٥٦قال سليمان لأطوفن الليلة على تسعين امرأة تلد كل امرأة منهن غلاما…
