كتاب البيوع

٢٥٧ حديثًا•صفحة ١ من ٢
٤٤٤٩إن أطيب ما أكل الرجل من كسبه وإن ولد الرجل من كسبه٤٤٥٠قال إن أولادكم من أطيب كسبكم فكلوا من كسب أولادكم٤٤٥١إن أطيب ما أكل الرجل من كسبه وولده من كسبه٤٤٥٢إن أطيب ما أكل الرجل من كسبه وإن ولده من كسبه٤٤٥٣إن الحلال بين وإن الحرام بين وإن بين ذلك أمورا مشتبهات٤٤٥٤يأتي على الناس زمان ما يبالي الرجل من أين أصاب المال من حلال أو حرام٤٤٥٥يأتي على الناس زمان يأكلون الربا فمن لم يأكله أصابه من غباره٤٤٥٦إن من أشراط الساعة أن يفشو المال ويكثر وتفشو التجارة ويظهر العلم…٤٤٥٧البيعان بالخيار ما لم يفترقا فإن صدقا وبينا بورك في بيعهما وإن كذبا…٤٤٥٨ثلاثة لا يكلمهم الله يوم القيامة ولا ينظر إليهم ولا يزكيهم ولهم عذاب…٤٤٥٩ثلاثة لا ينظر الله إليهم يوم القيامة ولا يزكيهم ولهم عذاب أليم الذي…٤٤٦٠إياكم وكثرة الحلف في البيع فإنه ينفق ثم يمحق٤٤٦١الحلف منفقة للسلعة ممحقة للكسب٤٤٦٢ثلاثة لا يكلمهم الله عز وجل ولا ينظر إليهم يوم القيامة ولا يزكيهم…٤٤٦٣يا معشر التجار إنه يشهد بيعكم الحلف واللغو فشوبوه بالصدقة٤٤٦٤البيعان بالخيار ما لم يفترقا فإن بينا وصدقا بورك لهما في بيعهما وإن…٤٤٦٥المتبايعان كل واحد منهما بالخيار على صاحبه ما لم يفترقا إلا بيع الخيار٤٤٦٦البيعان بالخيار ما لم يفترقا أو يكون خيارا٤٤٦٧المتبايعان بالخيار ما لم يفترقا إلا أن يكون البيع كان عن خيار فإن كان…٤٤٦٨إذا تبايع البيعان فكل واحد منهما بالخيار من بيعه ما لم يفترقا أو يكون…٤٤٦٩البيعان بالخيار ما لم يفترقا أو يقول أحدهما للآخر اختر٤٤٧٠البيعان بالخيار حتى يفترقا أو يكون بيع خيار٤٤٧١البيعان بالخيار حتى يفترقا أو يكون بيع خيار٤٤٧٢إذا تبايع الرجلان فكل واحد منهما بالخيار حتى يفترقا٤٤٧٣إن المتبايعين بالخيار في بيعهما ما لم يفترقا إلا أن يكون البيع خيارا٤٤٧٤المتبايعان لا بيع بينهما حتى يتفرقا إلا بيع الخيار٤٤٧٥كل بيعين لا بيع بينهما حتى يتفرقا إلا بيع الخيار٤٤٧٦كل بيعين فلا بيع بينهما حتى يتفرقا إلا بيع الخيار٤٤٧٧كل بيعين لا بيع بينهما حتى يتفرقا إلا بيع الخيار٤٤٧٨كل بيعين لا بيع بينهما حتى يتفرقا إلا بيع الخيار٤٤٧٩كل بيعين فلا بيع بينهما حتى يتفرقا إلا بيع الخيار٤٤٨٠البيعان بالخيار ما لم يتفرقا أو يكون بيعهما عن خيار٤٤٨١البيعان بالخيار حتى يتفرقا أو يأخذ كل واحد منهما من البيع ما هوي…٤٤٨٢البيعان بالخيار ما لم يتفرقا ويأخذ أحدهما ما رضي من صاحبه أو هوي٤٤٨٣المتبايعان بالخيار ما لم يتفرقا إلا أن يكون صفقة خيار ولا يحل له أن…٤٤٨٤إذا بعت فقل لا خلابة٤٤٨٥إذا بعت فقل لا خلابة٤٤٨٦إذا باع أحدكم الشاة أو اللقحة فلا يحفلها٤٤٨٧لا تلقوا الركبان للبيع ولا تصروا الإبل والغنم من ابتاع من ذلك شيئا…٤٤٨٨من اشترى مصراة فإن رضيها إذا حلبها فليمسكها وإن كرهها فليردها ومعها…٤٤٨٩من ابتاع محفلة أو مصراة فهو بالخيار ثلاثة أيام إن شاء أن يمسكها…٤٤٩٠ووكيع، قالا حدثنا ابن أبي ذئب، عن مخلد بن خفاف، عن عروة، عن عائشة،…٤٤٩١أخبرني عبد الله بن محمد بن تميم، قال حدثنا حجاج، قال حدثني شعبة، عن…٤٤٩٢أن النبي صلى الله عليه وسلم نهى أن يبيع حاضر لباد وإن كان أباه أو…٤٤٩٣حاضر لباد وإن كان أخاه أو أباه .٤٤٩٤حاضر لباد .٤٤٩٥لا يبيع حاضر لباد دعوا الناس يرزق الله بعضهم من بعض٤٤٩٦لا تلقوا الركبان للبيع ولا يبع بعضكم على بيع بعض - ولا تناجشوا - ولا…٤٤٩٧أخبرنا عبد الرحمن بن عبد الله بن عبد الحكم بن أعين، قال حدثنا شعيب بن…٤٤٩٨أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن عبيد الله، عن نافع، عن…٤٤٩٩أخبرنا إسحاق بن إبراهيم، قال قلت لأبي أسامة أحدثكم عبيد الله عن نافع…٤٥٠٠قال نهى رسول الله صلى الله عليه وسلم أن يتلقى الركبان وأن يبيع حاضر…٤٥٠١لا تلقوا الجلب فمن تلقاه فاشترى منه فإذا أتى سيده السوق فهو بالخيار٤٥٠٢لا يبيعن حاضر لباد ولا تناجشوا ولا يساوم الرجل على سوم أخيه ولا يخطب…٤٥٠٣لا يبيع أحدكم على بيع أخيه٤٥٠٤لا يبيع الرجل على بيع أخيه حتى يبتاع أو يذر٤٥٠٥أخبرنا قتيبة، عن مالك، عن نافع، عن ابن عمر، أن النبي صلى الله عليه…٤٥٠٦لا يبيع الرجل على بيع أخيه ولا يبيع حاضر لباد ولا تناجشوا ولا يزيد…٤٥٠٧لا يبيع حاضر لباد ولا تناجشوا ولا يزيد الرجل على بيع أخيه ولا تسأل…٤٥٠٨وعيسى بن يونس، قالا حدثنا الأخضر بن عجلان، عن أبي بكر الحنفي، عن أنس…٤٥٠٩والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن القاسم،…٤٥١٠أخبرنا إبراهيم بن يعقوب بن إسحاق، قال حدثنا عبد الله بن يوسف، قال…٤٥١١والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن وهب، قال أخبرني يونس،…٤٥١٢أخبرنا الحسين بن حريث المروزي، قال حدثنا سفيان، عن الزهري، عن عطاء بن…٤٥١٣أخبرنا محمد بن المصفى بن بهلول، عن محمد بن حرب، عن الزبيدي، عن…٤٥١٤أن عامر بن سعد، أخبره أن أبا سعيد الخدري رضي الله عنه قال نهى رسول…٤٥١٥أخبرنا محمد بن رافع، قال حدثنا عبد الرزاق، قال حدثنا معمر، عن الزهري،…٤٥١٦أخبرنا هارون بن زيد بن أبي الزرقاء، قال حدثنا أبي قال، حدثنا جعفر بن…٤٥١٧أخبرنا محمد بن عبد الأعلى، قال حدثنا المعتمر، قال سمعت عبيد الله، عن…٤٥١٨أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن عبيد الله، قال أخبرني…٤٥١٩لا تبيعوا الثمر حتى يبدو صلاحه٤٥٢٠أخبرنا قتيبة بن سعيد، قال حدثنا سفيان، عن الزهري، عن سالم، عن أبيه،…٤٥٢١لا تبيعوا الثمر حتى يبدو صلاحه ولا تبتاعوا الثمر بالتمر٤٥٢٢لا تبيعوا الثمر حتى يبدو صلاحه٤٥٢٣حدثنا محمد بن منصور، قال حدثنا سفيان، عن ابن جريج، عن عطاء، سمعت جابر…٤٥٢٤وأبي الزبير، عن جابر، أن النبي صلى الله عليه وسلم نهى عن المخابرة…٤٥٢٥أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا هشام، عن أبي…٤٥٢٦والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن القاسم،…٤٥٢٧إن بعت من أخيك ثمرا فأصابته جائحة فلا يحل لك أن تأخذ منه شيئا بم تأخذ…٤٥٢٨من باع ثمرا فأصابته جائحة فلا يأخذ من أخيه - وذكر شيئا - على ما يأكل…٤٥٢٩أن النبي صلى الله عليه وسلم وضع الجوائح .٤٥٣٠قال أصيب رجل في عهد رسول الله صلى الله عليه وسلم في ثمار ابتاعها فكثر…٤٥٣١أخبرنا قتيبة بن سعيد، قال حدثنا سفيان، عن حميد الأعرج، عن سليمان بن…٤٥٣٢أخبرنا قتيبة بن سعيد، قال حدثنا سفيان، عن الزهري، عن سالم، عن أبيه،…٤٥٣٣أخبرني زياد بن أيوب، قال حدثنا ابن علية، قال حدثنا أيوب، عن نافع، عن…٤٥٣٤أخبرنا قتيبة، عن مالك، عن نافع، عن ابن عمر، رضى الله عنهما أن رسول…٤٥٣٥أخبرنا قتيبة بن سعيد، قال حدثنا أبو الأحوص، عن طارق، عن سعيد بن…٤٥٣٦أن رسول الله صلى الله عليه وسلم رخص في العرايا .٤٥٣٧أن رسول الله صلى الله عليه وسلم رخص في العرايا بالتمر والرطب .٤٥٣٨أن رسول الله صلى الله عليه وسلم رخص في بيع العرايا تباع بخرصها .٤٥٣٩أن رسول الله صلى الله عليه وسلم رخص في بيع العرية بخرصها تمرا .٤٥٤٠أن سالما، أخبره أنه، سمع عبد الله بن عمر، يقول إن زيد بن ثابت أخبره…٤٥٤١ويعقوب بن إبراهيم، - واللفظ له - عن عبد الرحمن، عن مالك، عن داود بن…٤٥٤٢أخبرنا عبد الله بن محمد بن عبد الرحمن، قال حدثنا سفيان، عن يحيى، عن…٤٥٤٣أن رافع بن خديج، وسهل بن أبي حثمة، حدثاه أن رسول الله صلى الله عليه…٤٥٤٤رسول الله صلى الله عليه وسلم أنهم قالوا رخص رسول الله صلى الله عليه…٤٥٤٥أينقص الرطب إذا يبس٤٥٤٦أخبرنا محمد بن علي بن ميمون، قال حدثنا محمد بن يوسف الفريابي، قال…٤٥٤٧أنه سمع جابر بن عبد الله، يقول نهى رسول الله صلى الله عليه وسلم عن…٤٥٤٨لا تباع الصبرة من الطعام بالصبرة من الطعام ولا الصبرة من الطعام…٤٥٤٩أخبرنا قتيبة، قال حدثنا الليث، عن نافع، عن ابن عمر، قال نهى رسول الله…٤٥٥٠حدثنا عبد الحميد بن محمد، قال حدثنا مخلد بن يزيد، قال حدثنا ابن جريج،…٤٥٥١أخبرنا علي بن حجر، قال حدثنا إسماعيل، عن أيوب، عن نافع، عن ابن عمر،…٤٥٥٢بعه بالورق ثم اشتر به٤٥٥٣أكل تمر خيبر هكذا٤٥٥٤وإسماعيل بن مسعود، - واللفظ له - عن خالد، قال حدثنا سعيد، عن قتادة،…٤٥٥٥لا صاعى تمر بصاع ولا صاعى حنطة بصاع ولا درهما بدرهمين٤٥٥٦لا صاعى تمر بصاع ولا صاعى حنطة بصاع ولا درهمين بدرهم٤٥٥٧قال أتى بلال رسول الله صلى الله عليه وسلم بتمر برني فقال " ما هذا " .…٤٥٥٨الذهب بالورق ربا إلا هاء وهاء والتمر بالتمر ربا إلا هاء وهاء والبر…٤٥٥٩التمر بالتمر والحنطة بالحنطة والشعير بالشعير والملح بالملح يدا بيد…٤٥٦٠وعبد الله بن عتيك، قالا جمع المنزل بين عبادة بن الصامت ومعاوية حدثهم…٤٥٦١وعبد الله بن عبيد، - وقد كان يدعى ابن هرمز - قال جمع المنزل بين عبادة…٤٥٦٢وعبد الله بن عبيد، قالا جمع المنزل بين عبادة بن الصامت وبين معاوية…٤٥٦٣أخبرني محمد بن آدم، عن عبدة، عن ابن أبي عروبة، عن قتادة، عن مسلم بن…٤٥٦٤الذهب بالذهب تبره وعينه وزنا بوزن والفضة بالفضة تبره وعينه وزنا بوزن…٤٥٦٥أن أبا المتوكل، مر بهم في السوق فقام إليه قوم أنا منهم قال قلنا…٤٥٦٦الذهب الكفة بالكفة٤٥٦٧الدينار بالدينار والدرهم بالدرهم لا فضل بينهما٤٥٦٨أخبرنا قتيبة بن سعيد، عن مالك، عن حميد بن قيس المكي، عن مجاهد، قال…٤٥٦٩الذهب بالذهب وزنا بوزن مثلا بمثل والفضة بالفضة وزنا بوزن مثلا بمثل…٤٥٧٠لا تبيعوا الذهب بالذهب إلا مثلا بمثل ولا تشفوا بعضها على بعض ولا…٤٥٧١ولا تبيعوا غائبا بناجز ولا تشفوا أحدهما على الآخر٤٥٧٢أن معاوية، باع سقاية من ذهب أو ورق بأكثر من وزنها فقال أبو الدرداء…٤٥٧٣لا تباع حتى تفصل٤٥٧٤افصل بعضها من بعض ثم بعها٤٥٧٥ما كان يدا بيد فلا بأس وما كان نسيئة فهو ربا٤٥٧٦إن كان يدا بيد فلا بأس وإن كان نسيئة فلا يصلح٤٥٧٧أخبرنا أحمد بن عبد الله بن الحكم، عن محمد، قال حدثنا شعبة، عن حبيب،…٤٥٧٨وفيما قرأ علينا أحمد بن منيع قال حدثنا عباد بن العوام، قال حدثنا يحيى…٤٥٧٩تبايعوا الذهب بالفضة كيف شئتم والفضة بالذهب كيف شئتم٤٥٨٠لا ربا إلا في النسيئة٤٥٨١إنما الربا في النسيئة٤٥٨٢لا بأس أن تأخذها بسعر يومها ما لم تفترقا وبينكما شىء٤٥٨٣إذا بايعت صاحبك فلا تفارقه وبينك وبينه لبس٤٥٨٤أنه كان يكره أن يأخذ، الدنانير من الدراهم والدراهم من الدنانير .٤٥٨٥أنه كان لا يرى بأسا - يعني - في قبض الدراهم من الدنانير والدنانير من…٤٥٨٦في قبض الدنانير من الدراهم أنه كان يكرهها إذا كان من قرض .٤٥٨٧أنه كان لا يرى بأسا وإن كان من قرض .٤٥٨٨بمثله . قال أبو عبد الرحمن كذا وجدته في هذا الموضع .٤٥٨٩لا بأس أن تأخذ بسعر يومها ما لم تفترقا وبينكما شىء٤٥٩٠قال لما قدم النبي صلى الله عليه وسلم المدينة دعا بميزان فوزن لي…٤٥٩١ومحمد بن عبد الله بن يزيد، عن سفيان، عن مسعر، عن محارب بن دثار، عن…٤٥٩٢أخبرنا يعقوب بن إبراهيم، قال حدثنا عبد الرحمن، عن سفيان، عن سماك، عن…٤٥٩٣ومحمد بن بشار، عن محمد، قال حدثنا شعبة، عن سماك بن حرب، قال سمعت أبا…٤٥٩٤المكيال على مكيال أهل المدينة والوزن على وزن أهل مكة٤٥٩٥من ابتاع طعاما فلا يبعه حتى يستوفيه٤٥٩٦من ابتاع طعاما فلا يبعه حتى يقبضه٤٥٩٧من ابتاع طعاما فلا يبيعه حتى يكتاله٤٥٩٨أخبرنا إسحاق بن منصور، قال أنبأنا عبد الرحمن، عن
