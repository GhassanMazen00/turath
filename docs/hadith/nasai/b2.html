سنن النسائي

كتاب المياه

٢٣ حديثًا
٣٢٥إن الماء لا ينجسه شىء٣٢٦الماء طهور لا ينجسه شىء٣٢٧الماء لا ينجسه شىء٣٢٨إذا كان الماء قلتين لم يحمل الخبث٣٢٩أن أعرابيا، بال في المسجد فقام إليه بعض القوم فقال رسول الله صلى الله…٣٣٠دعوه وأهريقوا على بوله دلوا من ماء فإنما بعثتم ميسرين ولم تبعثوا…٣٣١لا يغتسل أحدكم في الماء الدائم وهو جنب٣٣٢هو الطهور ماؤه الحل ميتته٣٣٣اللهم اغسل خطاياى بماء الثلج والبرد ونق قلبي من الخطايا كما نقيت…٣٣٤اللهم اغسلني من خطاياى بالثلج والماء والبرد٣٣٥إذا ولغ الكلب في إناء أحدكم فليرقه ثم ليغسله سبع مرات٣٣٦إذا ولغ الكلب في الإناء فاغسلوه سبع مرات وعفروه الثامنة بالتراب٣٣٧ما بالهم وبال الكلاب٣٣٨إذا ولغ الكلب في إناء أحدكم فليغسله سبع مرات أولاهن بالتراب٣٣٩إذا ولغ الكلب في إناء أحدكم فليغسله سبع مرات أولاهن بالتراب٣٤٠إنها ليست بنجس إنما هي من الطوافين عليكم والطوافات٣٤١أخبرنا عمرو بن علي، قال حدثنا عبد الرحمن، عن سفيان، عن المقدام بن…٣٤٢قال كان الرجال والنساء يتوضئون في زمان رسول الله صلى الله عليه وسلم…٣٤٣أن رسول الله صلى الله عليه وسلم نهى أن يتوضأ الرجل بفضل وضوء المرأة .٣٤٤أنها كانت تغتسل مع رسول الله صلى الله عليه وسلم في الإناء الواحد .٣٤٥يقول كان رسول الله صلى الله عليه وسلم يتوضأ بمكوك ويغتسل بخمسة مكاكي .٣٤٦أن رسول الله صلى الله عليه وسلم كان يتوضأ بمد ويغتسل بنحو الصاع .٣٤٧قالت كان رسول الله صلى الله عليه وسلم يتوضأ بالمد ويغتسل بالصاع .
