كتاب مناسك الحج

٤٦٧ حديثًا•صفحة ١ من ٤
٢٦١٩إن الله عز وجل قد فرض عليكم الحج٢٦٢٠إن الله تعالى كتب عليكم الحج٢٦٢١فحج عن أبيك واعتمر٢٦٢٢الحجة المبرورة ليس لها جزاء إلا الجنة والعمرة إلى العمرة كفارة لما…٢٦٢٣الحجة المبرورة ليس لها ثواب إلا الجنة٢٦٢٤قال سأل رجل النبي صلى الله عليه وسلم فقال يا رسول الله أى الأعمال…٢٦٢٥وفد الله ثلاثة الغازي والحاج والمعتمر٢٦٢٦جهاد الكبير والصغير والضعيف والمرأة الحج والعمرة٢٦٢٧من حج هذا البيت فلم يرفث ولم يفسق رجع كما ولدته أمه٢٦٢٨لا ولكن أحسن الجهاد وأجمله حج البيت حج مبرور٢٦٢٩العمرة إلى العمرة كفارة لما بينهما والحج المبرور ليس له جزاء إلا الجنة٢٦٣٠تابعوا بين الحج والعمرة فإنهما ينفيان الفقر والذنوب كما ينفي الكير…٢٦٣١تابعوا بين الحج والعمرة فإنهما ينفيان الفقر والذنوب كما ينفي الكير…٢٦٣٢أرأيت لو كان على أختك دين أكنت قاضيه٢٦٣٣نعم لو كان على أمها دين فقضته عنها ألم يكن يجزئ عنها فلتحج عن أمها٢٦٣٤أن امرأة، سألت النبي صلى الله عليه وسلم عن أبيها مات ولم يحج قال "…٢٦٣٥أن امرأة، من خثعم سألت النبي صلى الله عليه وسلم غداة جمع فقالت يا…٢٦٣٦مثله .٢٦٣٧حج عن أبيك واعتمر٢٦٣٨قال جاء رجل من خثعم إلى رسول الله صلى الله عليه وسلم فقال إن أبي شيخ…٢٦٣٩أرأيت لو كان على أبيك دين أكنت قاضيه٢٦٤٠أرأيت لو كان عليه دين فقضيته أكان مجزئا٢٦٤١والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٢٦٤٢أن سليمان بن يسار، أخبره أن ابن عباس أخبره أن امرأة من خثعم استفتت…٢٦٤٣أرأيت لو كان على أمك دين أكنت قاضيه٢٦٤٤أنت أكبر ولد أبيك فحج عنه٢٦٤٥أن امرأة، رفعت صبيا لها إلى رسول الله صلى الله عليه وسلم قالت يا رسول…٢٦٤٦قال رفعت امرأة صبيا لها من هودج فقالت يا رسول الله ألهذا حج قال " نعم…٢٦٤٧قال رفعت امرأة إلى النبي صلى الله عليه وسلم صبيا فقالت ألهذا حج قال "…٢٦٤٨قال صدر رسول الله صلى الله عليه وسلم فلما كان بالروحاء لقي قوما فقال…٢٦٤٩أن رسول الله صلى الله عليه وسلم مر بامرأة وهي في خدرها معها صبي فقالت…٢٦٥٠قال أخبرتني عمرة، أنها سمعت عائشة، تقول خرجنا مع رسول الله صلى الله…٢٦٥١يهل أهل المدينة من ذي الحليفة وأهل الشام من الجحفة وأهل نجد من قرن٢٦٥٢يهل أهل المدينة من ذي الحليفة ويهل أهل الشام من الجحفة ويهل أهل نجد…٢٦٥٣أن رسول الله صلى الله عليه وسلم وقت لأهل المدينة ذا الحليفة ولأهل…٢٦٥٤هن لهن ولكل آت أتى عليهن من غيرهن فمن كان أهله دون الميقات حيث ينشئ…٢٦٥٥يهل أهل المدينة من ذي الحليفة وأهل الشام من الجحفة وأهل نجد من قرن٢٦٥٦أخبرني محمد بن عبد الله بن عمار الموصلي، قال حدثنا أبو هاشم، محمد بن…٢٦٥٧هن لهم ولمن أتى عليهن ممن سواهن لمن أراد الحج والعمرة ومن كان دون ذلك…٢٦٥٨أخبرنا قتيبة، قال حدثنا حماد، عن عمرو، عن طاوس، عن ابن عباس، أن النبي…٢٦٥٩أن أباه، قال بات رسول الله صلى الله عليه وسلم بذي الحليفة ببيداء وصلى…٢٦٦٠أخبرنا عبدة بن عبد الله، عن سويد، عن زهير، عن موسى بن عقبة، عن سالم…٢٦٦١والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٢٦٦٢أن رسول الله صلى الله عليه وسلم صلى الظهر بالبيداء ثم ركب وصعد جبل…٢٦٦٣مرها فلتغتسل ثم لتهل٢٦٦٤أنه خرج حاجا مع رسول الله صلى الله عليه وسلم حجة الوداع ومعه امرأته…٢٦٦٥والمسور بن مخرمة، أنهما اختلفا بالأبواء فقال ابن عباس يغسل المحرم…٢٦٦٦والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٢٦٦٧لا يلبس القميص ولا البرنس ولا السراويل ولا العمامة ولا ثوبا مسه ورس…٢٦٦٨أين الرجل الذي سألني آنفا٢٦٦٩لا تلبسوا القمص ولا العمائم ولا السراويلات ولا البرانس ولا الخفاف إلا…٢٦٧٠لا تلبسوا القميص٢٦٧١السراويل لمن لا يجد الإزار والخفين لمن لا يجد النعلين٢٦٧٢من لم يجد إزارا فليلبس سراويل ومن لم يجد نعلين فليلبس خفين٢٦٧٣لا تلبسوا القميص ولا السراويلات ولا العمائم ولا البرانس ولا الخفاف…٢٦٧٤لا تلبسوا القميص ولا العمائم ولا السراويلات ولا البرانس ولا الخفاف…٢٦٧٥لا تلبسوا القميص ولا السراويلات ولا العمائم ولا البرانس ولا الخفاف…٢٦٧٦لا تلبس القميص ولا العمامة ولا السراويل ولا البرنس ولا الخفين إلا أن…٢٦٧٧لا تلبس القميص ولا العمائم ولا البرانس ولا السراويلات ولا الخفاف إلا…٢٦٧٨لا تلبسوا في الإحرام القميص ولا السراويلات ولا العمائم ولا البرانس…٢٦٧٩إذا لم يجد إزارا فليلبس السراويل وإذا لم يجد النعلين فليلبس الخفين…٢٦٨٠إذا لم يجد المحرم النعلين فليلبس الخفين وليقطعهما أسفل من الكعبين٢٦٨١لا تلبسوا القمص ولا السراويلات ولا الخفاف إلا أن يكون رجل له نعلان…٢٦٨٢إني لبدت رأسي وقلدت هديي فلا أحل حتى أحل من الحج٢٦٨٣والحارث بن مسكين، قراءة عليه وأنا أسمع، واللفظ، له عن ابن وهب، أخبرني…٢٦٨٤قالت طيبت رسول الله صلى الله عليه وسلم عند إحرامه حين أراد أن يحرم…٢٦٨٥قالت طيبت رسول الله صلى الله عليه وسلم لإحرامه قبل أن يحرم ولحله قبل…٢٦٨٦قالت طيبت رسول الله صلى الله عليه وسلم لإحرامه قبل أن يحرم ولحله حين…٢٦٨٧قالت طيبت رسول الله صلى الله عليه وسلم لحرمه حين أحرم ولحله بعد ما…٢٦٨٨قالت طيبت رسول الله صلى الله عليه وسلم لإحلاله وطيبته لإحرامه طيبا لا…٢٦٨٩قال قلت لعائشة بأى شىء طيبت رسول الله صلى الله عليه وسلم قالت بأطيب…٢٦٩٠قالت كنت أطيب رسول الله صلى الله عليه وسلم عند إحرامه بأطيب ما أجد .٢٦٩١قالت كنت أطيب رسول الله صلى الله عليه وسلم بأطيب ما أجد لحرمه ولحله…٢٦٩٢قال قالت عائشة طيبت رسول الله صلى الله عليه وسلم قبل أن يحرم ويوم…٢٦٩٣قالت كأني أنظر إلى وبيص الطيب في رأس رسول الله صلى الله عليه وسلم وهو…٢٦٩٤رضى الله عنها قالت لقد كان يرى وبيص الطيب في مفارق رسول الله صلى الله…٢٦٩٥قالت كأني أنظر إلى وبيص الطيب في رأس رسول الله صلى الله عليه وسلم وهو…٢٦٩٦قالت كنت أنظر إلى وبيص الطيب في أصول شعر رسول الله صلى الله عليه وسلم…٢٦٩٧قالت كأني أنظر إلى وبيص الطيب في مفرق رأس رسول الله صلى الله عليه…٢٦٩٨قالت لقد رأيت وبيص الطيب في رأس رسول الله صلى الله عليه وسلم وهو محرم…٢٦٩٩قالت كأني أنظر إلى وبيص الطيب في مفارق رسول الله صلى الله عليه وسلم…٢٧٠٠وهناد بن السري، عن أبي الأحوص، عن أبي إسحاق، عن الأسود، عن عائشة،…٢٧٠١قالت كنت أطيب رسول الله صلى الله عليه وسلم بأطيب ما كنت أجد من الطيب…٢٧٠٢قالت لقد رأيت وبيص الطيب في مفارق رسول الله صلى الله عليه وسلم بعد…٢٧٠٣قالت كنت أرى وبيص الطيب في مفرق رسول الله صلى الله عليه وسلم بعد ثلاث…٢٧٠٤عند الإحرام فقال لأن أطلي بالقطران أحب إلى من ذلك . فذكرت ذلك لعائشة…٢٧٠٥وسفيان، عن إبراهيم بن محمد بن المنتشر، عن أبيه، قال سمعت ابن عمر،…٢٧٠٦قال نهى النبي صلى الله عليه وسلم أن يتزعفر الرجل .٢٧٠٧أخبرني كثير بن عبيد، عن بقية، عن شعبة، قال حدثني إسماعيل بن إبراهيم،…٢٧٠٨أخبرنا قتيبة، قال حدثنا حماد، عن عبد العزيز، عن أنس، أن رسول الله صلى…٢٧٠٩ما كنت صانعا في حجك٢٧١٠انزع عنك الجبة واغسل عنك الصفرة وما كنت صانعا في حجتك فاصنعه في عمرتك٢٧١١إذا اشتكى رأسه وعينيه أن يضمدهما بصبر٢٧١٢لو استقبلت من أمري ما استدبرت لم أسق الهدى وجعلتها عمرة فمن لم يكن…٢٧١٣اغسلوه بماء وسدر ويكفن في ثوبين خارجا رأسه ووجهه فإنه يبعث يوم…٢٧١٤اغسلوه بماء وسدر وكفنوه في ثيابه ولا تخمروا وجهه ورأسه فإنه يبعث يوم…٢٧١٥وإسحاق بن منصور، عن عبد الرحمن، عن مالك، عن عبد الرحمن بن القاسم، عن…٢٧١٦قالت أهل رسول الله صلى الله عليه وسلم بالحج .٢٧١٧من شاء أن يهل بحج فليهل ومن شاء أن يهل بعمرة فليهل بعمرة٢٧١٨وسليمان، عن إبراهيم، عن الأسود، عن عائشة، قالت خرجنا مع رسول الله صلى…٢٧١٩أخبرنا إسحاق بن إبراهيم، قال أنبأنا جرير، عن منصور، عن أبي وائل، قال…٢٧٢٠فذكر مثله قال فأتيت عمر فقصصت عليه القصة إلا قوله يا هناه .٢٧٢١وغيره، عن رجل، من أهل العراق يقال له شقيق بن سلمة أبو وائل أن رجلا من…٢٧٢٢قال كنت جالسا عند عثمان فسمع عليا، يلبي بعمرة وحجة فقال ألم تكن تنهى…٢٧٢٣أن عثمان، نهى عن المتعة، وأن يجمع الرجل بين الحج والعمرة فقال علي…٢٧٢٤بهذا الإسناد مثله .٢٧٢٥قال كنت مع علي بن أبي طالب حين أمره رسول الله صلى الله عليه وسلم على…٢٧٢٦يقول قال لي عمران بن حصين جمع رسول الله صلى الله عليه وسلم بين حج…٢٧٢٧أن رسول الله صلى الله عليه وسلم جمع بين حج وعمرة ثم لم ينزل فيها كتاب…٢٧٢٨أخبرنا أبو داود، قال حدثنا مسلم بن إبراهيم، قال حدثنا إسماعيل بن…٢٧٢٩لبيك عمرة وحجا لبيك عمرة وحجا٢٧٣٠أخبرنا هناد بن السري، عن أبي الأحوص، عن أبي إسحاق، عن أبي أسماء، عن…٢٧٣١لبيك عمرة وحجا معا٢٧٣٢من كان منكم أهدى فإنه لا يحل من شىء حرم منه حتى يقضي حجه ومن لم يكن…٢٧٣٣أخبرنا عمرو بن علي، قال حدثنا يحيى بن سعيد، قال حدثنا عبد الرحمن بن…٢٧٣٤أنه حدثه أنه، سمع سعد بن أبي وقاص، والضحاك بن قيس، - عام حج معاوية بن…٢٧٣٥ومحمد بن بشار، - واللفظ له - قالا حدثنا محمد، قال حدثنا شعبة، عن…٢٧٣٦وإنها، لفي كتاب الله ولقد فعلها رسول الله صلى الله عليه وسلم يعني…٢٧٣٧أخبرنا عبد الله بن محمد بن عبد الرحمن، قال حدثنا سفيان، عن هشام بن…٢٧٣٨أخبرنا محمد بن المثنى، عن عبد الرحمن، قال حدثنا سفيان، عن قيس، وهو…٢٧٣٩أخبرني إبراهيم بن يعقوب، قال حدثنا عثمان بن عمر، قال حدثنا إسماعيل بن…٢٧٤٠أخبرنا يعقوب بن إبراهيم، قال حدثنا يحيى بن سعيد، قال حدثنا جعفر بن…٢٧٤١والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ لمحمد - قالا حدثنا…٢٧٤٢أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا شعبة، قال أخبرني…٢٧٤٣أخبرنا محمد بن المثنى، قال حدثنا يحيى بن سعيد، عن جعفر بن محمد، قال…٢٧٤٤بما أهللت يا علي٢٧٤٥قال كنت مع علي حين أمره النبي صلى الله عليه وسلم على اليمن فأصبت معه…٢٧٤٦أن ابن عمر، أراد الحج عام نزل الحجاج بابن الزبير فقيل له إنه كائن…٢٧٤٧لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك والملك لا…٢٧٤٨لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك والملك لا…٢٧٤٩لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك والملك لا…٢٧٥٠لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك والملك لا…٢٧٥١لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك٢٧٥٢لبيك إله الحق٢٧٥٣جاءني جبريل فقال لي يا محمد مر أصحابك أن يرفعوا أصواتهم بالتلبية٢٧٥٤أن رسول الله صلى الله عليه وسلم أهل في دبر الصلاة .٢٧٥٥أن رسول الله صلى الله عليه وسلم صلى الظهر بالبيداء ثم ركب وصعد جبل…٢٧٥٦في حجة النبي صلى الله عليه وسلم فلما أتى ذا الحليفة صلى وهو صامت حتى…٢٧٥٧أنه سمع أباه، يقول بيداؤكم هذه التي تكذبون فيها على رسول الله صلى…٢٧٥٨أن سالم بن عبد الله، أخبره أن عبد الله بن عمر قال رأيت رسول الله صلى…٢٧٥٩أنه كان يخبر أن النبي صلى الله عليه وسلم أهل حين استوت به راحلته .٢٧٦٠وابن، جريج وابن إسحاق ومالك بن أنس عن المقبري، عن عبيد بن جريج، قال…٢٧٦١اغتسلي واستثفري بثوب ثم أهلي٢٧٦٢رضى الله عنه قال نفست أسماء بنت عميس محمد بن أبي بكر فأرسلت إلى رسول…٢٧٦٣أخبرنا قتيبة، قال حدثنا الليث، عن أبي الزبير، عن جابر بن عبد الله،…٢٧٦٤من كان معه هدى فليهلل بالحج مع العمرة ثم لا يحل حتى يحل منهما جميعا٢٧٦٥وعكرمة، عن ابن عباس، أن ضباعة، أرادت الحج فأمرها النبي صلى الله عليه…٢٧٦٦قولي لبيك اللهم لبيك ومحلي من الأرض حيث تحبسني فإن لك على ربك ما…٢٧٦٧أهلي واشترطي إن محلي حيث حبستني٢٧٦٨حجي
