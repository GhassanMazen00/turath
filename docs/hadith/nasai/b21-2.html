سنن النسائي

كتاب الجنائز

٢٧٢ حديثًا•صفحة ٢ من ٢
١٩٦٨أن عباد بن عبد الله بن الزبير، أخبره أن عائشة قالت ما صلى رسول الله…١٩٦٩إن ماتت فلا تدفنوها حتى أصلي عليها١٩٧٠إن أخاكم النجاشي قد مات فقوموا فصلوا عليه١٩٧١أن النبي صلى الله عليه وسلم نعى للناس النجاشي اليوم الذي مات فيه ثم…١٩٧٢وأبي، سلمة عن أبي هريرة، قال نعى رسول الله صلى الله عليه وسلم النجاشي…١٩٧٣إن أخاكم قد مات فقوموا فصلوا عليه١٩٧٤قال كنت في الصف الثاني يوم صلى رسول الله صلى الله عليه وسلم على…١٩٧٥إن أخاكم النجاشي قد مات فقوموا فصلوا عليه١٩٧٦قال صليت مع رسول الله صلى الله عليه وسلم على أم كعب ماتت في نفاسها…١٩٧٧أخبرنا محمد بن عبد الله بن يزيد، قال حدثنا أبي قال، حدثنا سعيد، قال…١٩٧٨يزعم أن ابن عمر، صلى على تسع جنائز جميعا فجعل الرجال يلون الإمام…١٩٧٩والفضل بن موسى، ح وأخبرنا سويد، قال أنبأنا عبد الله، عن حسين المكتب،…١٩٨٠أن رسول الله صلى الله عليه وسلم نعى للناس النجاشي وخرج بهم فصف بهم…١٩٨١إذا ماتت فآذنوني١٩٨٢أن زيد بن أرقم، صلى على جنازة فكبر عليها خمسا وقال كبرها رسول الله…١٩٨٣اللهم اغفر له وارحمه واعف عنه وعافه وأكرم نزله ووسع مدخله واغسله بماء…١٩٨٤اللهم اغفر له وارحمه وعافه واعف عنه وأكرم نزله ووسع مدخله واغسله…١٩٨٥وكان، من أصحاب رسول الله صلى الله عليه وسلم عن عبيد بن خالد السلمي أن…١٩٨٦اللهم اغفر لحينا وميتنا وشاهدنا وغائبنا وذكرنا وأنثانا وصغيرنا وكبيرنا١٩٨٧أخبرنا الهيثم بن أيوب، قال حدثنا إبراهيم، - وهو ابن سعد - قال حدثنا…١٩٨٨بفاتحة الكتاب فلما انصرف أخذت بيده فسألته فقلت تقرأ قال نعم إنه حق…١٩٨٩أنه قال السنة في الصلاة على الجنازة أن يقرأ في التكبيرة الأولى بأم…١٩٩٠بنحو ذلك .١٩٩١ما من ميت يصلي عليه أمة من المسلمين يبلغون أن يكونوا مائة يشفعون إلا…١٩٩٢لا يموت أحد من المسلمين فيصلي عليه أمة من الناس فيبلغوا أن يكونوا…١٩٩٣ما من ميت يصلي عليه أمة من الناس إلا شفعوا فيه١٩٩٤من صلى على جنازة فله قيراط ومن انتظرها حتى توضع في اللحد فله قيراطان…١٩٩٥من شهد جنازة حتى يصلى عليها فله قيراط ومن شهد حتى تدفن فله قيراطان١٩٩٦من تبع جنازة رجل مسلم احتسابا فصلى عليها ودفنها فله قيراطان ومن صلى…١٩٩٧من تبع جنازة فصلى عليها ثم انصرف فله قيراط من الأجر ومن تبعها فصلى…١٩٩٨إذا رأيتم الجنازة فقوموا ومن تبعها فلا يقعدن حتى توضع١٩٩٩: أنه ذكر القيام على الجنازة حتى توضع، فقال علي بن أبي طالب : قام…٢٠٠٠قال : رأيت رسول الله صلى الله عليه وسلم قام فقمنا، ورأيناه قعد فقعدنا…٢٠٠١أخبرنا هارون بن إسحاق، قال حدثنا أبو خالد الأحمر، عن عمرو بن قيس، عن…٢٠٠٢زملوهم بدمائهم، فإنه ليس كلم يكلم في الله إلا يأتي يوم القيامة يدمى،…٢٠٠٣يقال له عبيد الله بن معية قال : أصيب رجلان من المسلمين يوم الطائف،…٢٠٠٤: أن النبي صلى الله عليه وسلم أمر بقتلى أحد أن يردوا إلى مصارعهم،…٢٠٠٥ادفنوا القتلى في مصارعهم٢٠٠٦اذهب فوار أباك ولا تحدثن حدثا حتى تأتيني٢٠٠٧قال : الحدوا لي لحدا، وانصبوا على نصبا كما فعل برسول الله صلى الله…٢٠٠٨: أن سعدا، لما حضرته الوفاة قال : الحدوا لي لحدا، وانصبوا على نصبا…٢٠٠٩اللحد لنا والشق لغيرنا٢٠١٠احفروا وأعمقوا وأحسنوا، وادفنوا الاثنين والثلاثة في قبر واحد٢٠١١احفروا وأوسعوا، وادفنوا الاثنين والثلاثة في القبر، وقدموا أكثرهم قرآنا٢٠١٢قال : جعل تحت رسول الله صلى الله عليه وسلم حين دفن قطيفة حمراء .٢٠١٣وحين يقوم قائم الظهيرة حتى تزول الشمس، وحين تضيف الشمس للغروب .٢٠١٤أنه سمع جابرا، يقول : خطب رسول الله صلى الله عليه وسلم فذكر رجلا من…٢٠١٥احفروا وأوسعوا، وادفنوا الاثنين والثلاثة في قبر٢٠١٦احفروا وأوسعوا وأحسنوا، وادفنوا في القبر الاثنين والثلاثة، وقدموا…٢٠١٧احفروا وأحسنوا، وادفنوا الاثنين والثلاثة، وقدموا أكثرهم قرآنا٢٠١٨احفروا وأوسعوا وأحسنوا، وادفنوا الاثنين والثلاثة في القبر، وقدموا…٢٠١٩قال سمع عمرو، جابرا يقول : أتى النبي صلى الله عليه وسلم عبد الله بن…٢٠٢٠يقول : إن النبي صلى الله عليه وسلم أمر بعبد الله بن أبى فأخرجه من…٢٠٢١قال : دفن مع أبي رجل في القبر فلم يطب قلبي حتى أخرجته ودفنته على حدة .٢٠٢٢يزيد بن ثابت : أنهم خرجوا مع رسول الله صلى الله عليه وسلم ذات يوم،…٢٠٢٣مر مع رسول الله صلى الله عليه وسلم على قبر منتبذ، فأمهم وصف خلفه، قلت…٢٠٢٤رأى النبي صلى الله عليه وسلم مر بقبر منتبذ، فصلى عليه وصف أصحابه خلفه…٢٠٢٥: أن النبي صلى الله عليه وسلم صلى على قبر امرأة بعد ما دفنت .٢٠٢٦ويحيى بن آدم، قالا حدثنا مالك بن مغول، عن سماك، عن جابر بن سمرة، قال…٢٠٢٧وأبي الزبير، عن جابر، قال : نهى رسول الله صلى الله عليه وسلم أن يبنى…٢٠٢٨أنه سمع جابرا، يقول : نهى رسول الله صلى الله عليه وسلم عن تقصيص…٢٠٢٩أخبرنا عمران بن موسى، قال حدثنا عبد الوارث، قال حدثنا أيوب، عن أبي…٢٠٣٠أن ثمامة بن شفى، حدثه قال : كنا مع فضالة بن عبيد بأرض الروم فتوفي…٢٠٣١ولا صورة في بيت إلا طمستها .٢٠٣٢نهيتكم عن زيارة القبور فزوروها، ونهيتكم عن لحوم الأضاحي فوق ثلاثة…٢٠٣٣إني كنت نهيتكم أن تأكلوا لحوم الأضاحي إلا ثلاثا، فكلوا وأطعموا…٢٠٣٤استأذنت ربي عز وجل في أن أستغفر لها فلم يؤذن لي، واستأذنت في أن أزور…٢٠٣٥أى عم قل لا إله إلا الله كلمة أحاج لك بها عند الله عز وجل٢٠٣٦أخبرنا إسحاق بن منصور، قال حدثنا عبد الرحمن، عن سفيان، عن أبي إسحاق،…٢٠٣٧ما لك يا عائشة حشيا رابية٢٠٣٨إني بعثت إلى أهل البقيع لأصلي عليهم٢٠٣٩السلام عليكم دار قوم مؤمنين، وإنا وإياكم متواعدون غدا أو مواكلون،…٢٠٤٠السلام عليكم أهل الديار من المؤمنين والمسلمين، وإنا إن شاء الله بكم…٢٠٤١قال : لما مات النجاشي قال النبي صلى الله عليه وسلم : " استغفروا له " .٢٠٤٢وابن المسيب، أن أبا هريرة، أخبرهما : أن رسول الله صلى الله عليه وسلم…٢٠٤٣أخبرنا قتيبة، قال حدثنا عبد الوارث بن سعيد، عن محمد بن جحادة، عن أبي…٢٠٤٤لأن يجلس أحدكم على جمرة حتى تحرق ثيابه خير له من أن يجلس على قبر٢٠٤٥لا تقعدوا على القبور٢٠٤٦لعن الله قوما اتخذوا قبور أنبيائهم مساجد٢٠٤٧لعن الله اليهود والنصارى اتخذوا قبور أنبيائهم مساجد٢٠٤٨لقد سبق هؤلاء شرا كثيرا٢٠٤٩إن العبد إذا وضع في قبره وتولى عنه أصحابه إنه ليسمع قرع نعالهم٢٠٥٠إن العبد إذا وضع في قبره وتولى عنه أصحابه إنه ليسمع قرع نعالهم٢٠٥١إن العبد إذا وضع في قبره وتولى عنه أصحابه إنه ليسمع قرع نعالهم أتاه…٢٠٥٢من يقتله بطنه فلن يعذب في قبره٢٠٥٣كفى ببارقة السيوف على رأسه فتنة٢٠٥٤أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن التيمي، عن أبي عثمان، عن…٢٠٥٥هذا الذي تحرك له العرش وفتحت له أبواب السماء وشهده سبعون ألفا من…٢٠٥٦قال { يثبت الله الذين آمنوا بالقول الثابت في الحياة الدنيا وفي الآخرة…٢٠٥٧أخبرنا محمد بن بشار، قال حدثنا محمد، قال حدثنا شعبة، عن علقمة بن…٢٠٥٨أن النبي صلى الله عليه وسلم سمع صوتا من قبر فقال " متى مات هذا " .…٢٠٥٩يهود تعذب في قبورها٢٠٦٠اللهم إني أعوذ بك من عذاب القبر وأعوذ بك من عذاب النار وأعوذ بك من…٢٠٦١أخبرنا عمرو بن سواد بن الأسود بن عمرو، عن ابن وهب، قال حدثنا يونس بن…٢٠٦٢قد أوحي إلى أنكم تفتنون في القبور قريبا من فتنة الدجال٢٠٦٣قولوا اللهم إنا نعوذ بك من عذاب جهنم وأعوذ بك من عذاب القبر وأعوذ بك…٢٠٦٤إنما تفتن يهود٢٠٦٥إنكم تفتنون في قبوركم٢٠٦٦إنهم ليعذبون في قبورهم عذابا تسمعه البهائم٢٠٦٧صدقتا إنهم يعذبون عذابا تسمعه البهائم كلها٢٠٦٨يعذبان وما يعذبان في كبير٢٠٦٩إنهما ليعذبان وما يعذبان في كبير أما أحدهما فكان لا يستبرئ من بوله…٢٠٧٠ألا إن أحدكم إذا مات عرض عليه مقعده بالغداة والعشي إن كان من أهل…٢٠٧١يعرض على أحدكم إذا مات مقعده من الغداة والعشي فإن كان من أهل النار…٢٠٧٢إذا مات أحدكم عرض على مقعده بالغداة والعشي إن كان من أهل الجنة فمن…٢٠٧٣إنما نسمة المؤمن طائر في شجر الجنة حتى يبعثه الله عز وجل إلى جسده يوم…٢٠٧٤هذا مصرع فلان إن شاء الله غدا٢٠٧٥يا أبا جهل بن هشام ويا شيبة بن ربيعة ويا عتبة بن ربيعة ويا أمية بن…٢٠٧٦هل وجدتم ما وعد ربكم حقا - قال - إنهم ليسمعون الآن ما أقول لهم٢٠٧٧كل بني آدم - وفي حديث مغيرة كل ابن آدم - يأكله التراب إلا عجب الذنب…٢٠٧٨قال الله عز وجل كذبني ابن آدم ولم يكن ينبغي له أن يكذبني وشتمني ابن…٢٠٧٩أسرف عبد على نفسه حتى حضرته الوفاة قال لأهله إذا أنا مت فأحرقوني ثم…٢٠٨٠كان رجل ممن كان قبلكم يسيء الظن بعمله فلما حضرته الوفاة قال لأهله إذا…٢٠٨١إنكم ملاقو الله عز وجل حفاة عراة غرلا٢٠٨٢يحشر الناس يوم القيامة عراة غرلا وأول الخلائق يكسى إبراهيم عليه…٢٠٨٣يبعث الناس يوم القيامة حفاة عراة غرلا٢٠٨٤إنكم تحشرون حفاة عراة٢٠٨٥يحشر الناس يوم القيامة على ثلاث طرائق راغبين راهبين اثنان على بعير…٢٠٨٦أن الناس يحشرون ثلاثة أفواج فوج راكبين طاعمين كاسين وفوج تسحبهم…٢٠٨٧يا أيها الناس إنكم محشورون إلى الله عز وجل عراة٢٠٨٨ما لي لا أرى فلانا٢٠٨٩فلو كنت ثم لأريتكم قبره إلى جانب الطريق تحت الكثيب الأحمر
