١٥٠ حديثًا
١٠٢٩والأسود، أنهما كانا مع عبد الله في بيته فقال أصلى هؤلاء قلنا نعم .…١٠٣٠وعلقمة، قالا صلينا مع عبد الله بن مسعود في بيته فقام بيننا فوضعنا…١٠٣١أخبرنا نوح بن حبيب، قال أنبأنا ابن إدريس، عن عاصم بن كليب، عن عبد…١٠٣٢أخبرنا قتيبة، قال حدثنا أبو عوانة، عن أبي يعفور، عن مصعب بن سعد، قال…١٠٣٣أخبرنا عمرو بن علي، قال حدثنا يحيى بن سعيد، عن إسماعيل بن أبي خالد،…١٠٣٤قال سنت لكم الركب فأمسكوا بالركب .١٠٣٥قال قال عمر إنما السنة الأخذ بالركب .١٠٣٦أخبرنا هناد بن السري، في حديثه عن أبي الأحوص، عن عطاء بن السائب، عن…١٠٣٧أخبرنا أحمد بن سليمان الرهاوي، قال حدثنا حسين، عن زائدة، عن عطاء، عن…١٠٣٨أخبرنا يعقوب بن إبراهيم، عن ابن علية، عن عطاء بن السائب، عن سالم…١٠٣٩قال كان النبي صلى الله عليه وسلم إذا ركع اعتدل فلم ينصب رأسه ولم…١٠٤٠أخبرنا عبيد الله بن سعيد، حدثنا حماد بن مسعدة، عن أشعث، عن محمد، عن…١٠٤١أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى بن سعيد، عن ابن عجلان، عن…١٠٤٢أخبرنا الحسن بن داود المنكدري، قال حدثنا ابن أبي فديك، عن الضحاك بن…١٠٤٣أن إبراهيم بن عبد الله بن حنين، حدثه أن أباه حدثه أنه، سمع عليا، يقول…١٠٤٤أخبرنا قتيبة، عن مالك، عن نافع، عن إبراهيم بن عبد الله بن حنين، عن…١٠٤٥أيها الناس إنه لم يبق من مبشرات النبوة إلا الرؤيا الصالحة يراها…١٠٤٦سبحان ربي العظيم١٠٤٧سبحانك ربنا وبحمدك اللهم اغفر لي١٠٤٨سبوح قدوس رب الملائكة والروح١٠٤٩سبحان ذي الجبروت والملكوت والكبرياء والعظمة١٠٥٠اللهم لك ركعت ولك أسلمت وبك آمنت خشع لك سمعي وبصري وعظامي ومخي وعصبي١٠٥١اللهم لك ركعت وبك آمنت ولك أسلمت وعليك توكلت أنت ربي خشع سمعي وبصري…١٠٥٢اللهم لك ركعت وبك آمنت ولك أسلمت وعليك توكلت أنت ربي خشع سمعي وبصري…١٠٥٣ارجع فصل فإنك لم تصل١٠٥٤أتموا الركوع والسجود إذا ركعتم وسجدتم١٠٥٥سمع الله لمن حمده١٠٥٦أنه رأى رسول الله صلى الله عليه وسلم يرفع يديه إذا ركع وإذا رفع رأسه…١٠٥٧سمع الله لمن حمده١٠٥٨أنه قال ألا أصلي بكم صلاة رسول الله صلى الله عليه وسلم فصلى فلم يرفع…١٠٥٩سمع الله لمن حمده ربنا ولك الحمد١٠٦٠اللهم ربنا ولك الحمد١٠٦١إنما جعل الإمام ليؤتم به فإذا ركع فاركعوا وإذا رفع فارفعوا وإذا قال…١٠٦٢سمع الله لمن حمده١٠٦٣إذا قال الإمام سمع الله لمن حمده فقولوا ربنا ولك الحمد فإن من وافق…١٠٦٤إذا صليتم فأقيموا صفوفكم ثم ليؤمكم أحدكم فإذا كبر الإمام فكبروا وإذا…١٠٦٥أن رسول الله صلى الله عليه وسلم كان ركوعه وإذا رفع رأسه من الركوع…١٠٦٦سمع الله لمن حمده١٠٦٧اللهم ربنا ولك الحمد ملء السموات وملء الأرض وملء ما شئت من شىء بعد١٠٦٨سمع الله لمن حمده ربنا لك الحمد ملء السموات وملء الأرض وملء ما شئت من…١٠٦٩الله أكبر ذا الجبروت والملكوت والكبرياء والعظمة١٠٧٠قال قنت رسول الله صلى الله عليه وسلم شهرا بعد الركوع يدعو على رعل…١٠٧١أن أنس بن مالك، سئل هل قنت رسول الله صلى الله عليه وسلم في صلاة الصبح…١٠٧٢سمع الله لمن حمده١٠٧٣اللهم أنج الوليد بن الوليد وسلمة بن هشام وعياش بن أبي ربيعة…١٠٧٤سمع الله لمن حمده ربنا ولك الحمد١٠٧٥أخبرنا سليمان بن سلم البلخي، قال حدثنا النضر، قال أنبأنا هشام، عن…١٠٧٦وشعبة، عن عمرو بن مرة، ح وأخبرنا عمرو بن علي، قال حدثنا يحيى، عن…١٠٧٧وهشام، عن قتادة، عن أنس، أن رسول الله صلى الله عليه وسلم قنت شهرا -…١٠٧٨اللهم العن فلانا وفلانا١٠٧٩أن رسول الله صلى الله عليه وسلم قنت شهرا يدعو على حى من أحياء العرب…١٠٨٠قال صليت خلف رسول الله صلى الله عليه وسلم فلم يقنت وصليت خلف أبي بكر…١٠٨١أخبرنا قتيبة، قال حدثنا عباد، عن محمد بن عمرو، عن سعيد بن الحارث، عن…١٠٨٢خلف علي بن أبي طالب فكان إذا سجد كبر وإذا رفع رأسه من السجود كبر وإذا…١٠٨٣ويحيى، قالا حدثنا زهير، قال حدثني أبو إسحاق، عن عبد الرحمن بن الأسود،…١٠٨٤قال بايعت رسول الله صلى الله عليه وسلم أن لا أخر إلا قائما .١٠٨٥أنه رأى النبي صلى الله عليه وسلم رفع يديه في صلاته وإذا ركع وإذا رفع…١٠٨٦أنه رأى النبي صلى الله عليه وسلم رفع يديه فذكر مثله .١٠٨٧أن نبي الله صلى الله عليه وسلم كان إذا دخل في الصلاة فذكر نحوه وزاد…١٠٨٨قال كان رسول الله صلى الله عليه وسلم يرفع يديه إذا افتتح الصلاة وإذا…١٠٨٩قال رأيت رسول الله صلى الله عليه وسلم إذا سجد وضع ركبتيه قبل يديه…١٠٩٠يعمد أحدكم في صلاته فيبرك كما يبرك الجمل١٠٩١إذا سجد أحدكم فليضع يديه قبل ركبتيه ولا يبرك بروك البعير١٠٩٢إن اليدين تسجدان كما يسجد الوجه فإذا وضع أحدكم وجهه فليضع يديه وإذا…١٠٩٣قال أمر النبي صلى الله عليه وسلم أن يسجد على سبعة أعضاء ولا يكف شعره…١٠٩٤إذا سجد العبد سجد منه سبعة آراب وجهه وكفاه وركبتاه وقدماه١٠٩٥والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن القاسم،…١٠٩٦أمرت أن أسجد على سبعة - لا أكف الشعر ولا الثياب - الجبهة والأنف…١٠٩٧أمرت أن أسجد على سبعة أعظم على الجبهة وأشار بيده على الأنف واليدين…١٠٩٨وعبد الله بن محمد بن عبد الرحمن الزهري، قالا حدثنا سفيان، عن ابن…١٠٩٩إذا سجد العبد سجد معه سبعة آراب وجهه وكفاه وركبتاه وقدماه١١٠٠اللهم إني أعوذ برضاك من سخطك وبمعافاتك من عقوبتك وبك منك لا أحصي ثناء…١١٠١أخبرنا محمد بن بشار، قال حدثنا يحيى بن سعيد، قال حدثنا عبد الحميد بن…١١٠٢سمع الله لمن حمده١١٠٣لا يفترش أحدكم ذراعيه في السجود افتراش الكلب١١٠٤أخبرنا علي بن حجر المروزي، قال أنبأنا شريك، عن أبي إسحاق، قال وصف لنا…١١٠٥أن رسول الله صلى الله عليه وسلم كان إذا صلى جخى .١١٠٦أن رسول الله صلى الله عليه وسلم كان إذا صلى فرج بين يديه حتى يبدو…١١٠٧قال لو كنت بين يدى رسول الله صلى الله عليه وسلم لأبصرت إبطيه . قال…١١٠٨قال صليت مع رسول الله صلى الله عليه وسلم فكنت أرى عفرة إبطيه إذا سجد .١١٠٩أن النبي صلى الله عليه وسلم كان إذا سجد جافى يديه حتى لو أن بهمة…١١١٠اعتدلوا في السجود ولا يبسط أحدكم ذراعيه انبساط الكلب١١١١لا تجزئ صلاة لا يقيم الرجل فيها صلبه في الركوع والسجود١١١٢أن تميم بن محمود، أخبره أن عبد الرحمن بن شبل أخبره أن رسول الله صلى…١١١٣أمرت أن أسجد على سبعة ولا أكف شعرا ولا ثوبا١١١٤إنما مثل هذا مثل الذي يصلي وهو مكتوف١١١٥قال أمر النبي صلى الله عليه وسلم أن يسجد على سبعة أعظم ونهي أن يكف…١١١٦أخبرنا سويد بن نصر، قال أنبأنا عبد الله بن المبارك، عن خالد بن عبد…١١١٧أتموا الركوع والسجود فوالله إني لأراكم من خلف ظهري في ركوعكم وسجودكم١١١٨وعثمان بن عمر، قال أبو علي حدثنا وقال، عثمان أنبأنا داود بن قيس، عن…١١١٩أن أباه، حدثه أنه، سمع عليا، قال نهاني رسول الله صلى الله عليه وسلم…١١٢٠اللهم قد بلغت - ثلاث مرات - إنه لم يبق من مبشرات النبوة إلا الرؤيا…١١٢١اللهم اجعل في قلبي نورا واجعل في سمعي نورا واجعل في بصري نورا واجعل…١١٢٢سبحانك اللهم ربنا وبحمدك اللهم اغفر لي١١٢٣سبحانك ولك أسلمت وأنت ربي سجد وجهي للذي خلقه وصوره…١١٢٨اللهم لك سجدت وبك آمنت ولك أسلمت اللهم أنت ربي سجد وجهي للذي خلقه…١١٢٩سجد وجهي للذي خلقه وشق سمعه وبصره بحوله وقوته١١٣٠أعوذ برضاك من سخطك وأعوذ بمعافاتك من عقوبتك وأعوذ بك منك لا أحصي ثناء…١١٣١سبحانك اللهم وبحمدك لا إله إلا أنت١١٣٢سبحان ذي الجبروت والملكوت والكبرياء والعظمة١١٣٣سبحان ربي العظيم سبحان ربي العظيم سبحان ربي العظيم١١٣٤سبوح قدوس رب الملائكة والروح١١٣٥أخبرنا محمد بن رافع، قال حدثنا عبد الله بن إبراهيم بن عمر بن كيسان،…١١٣٦وعليك اذهب فصل فإنك لم تصل١١٣٧أقرب ما يكون العبد من ربه عز وجل وهو ساجد فأكثروا الدعاء١١٣٨قال كنت آتي رسول الله صلى الله عليه وسلم بوضوئه وبحاجته فقال " سلني "…١١٣٩ما من عبد يسجد لله سجدة إلا رفعه الله عز وجل بها درجة وحط عنه بها…١١٤٠فأكون أول من يجيز فإذا فرغ الله عز وجل من القضاء بين خلقه وأخرج من…١١٤١كل ذلك لم الله…١١٦٥أخبرنا إسحاق بن إبراهيم، قال حدثنا يحيى، - وهو ابن آدم - قال سمعت…١١٦٦قولوا في كل جلسة التحيات لله والصلوات والطيبات السلام عليك أيها النبي…١١٦٧قولوا التحيات لله والصلوات والطيبات السلام عليك
