سنن النسائي

كتاب الزكاة

١٨٤ حديثًا•صفحة ١ من ٢
٢٤٣٥إنك تأتي قوما أهل كتاب فإذا جئتهم فادعهم إلى أن يشهدوا أن لا إله إلا…٢٤٣٦أخبرنا محمد بن عبد الأعلى، قال حدثنا معتمر، قال سمعت بهز بن حكيم،…٢٤٣٧إسباغ الوضوء شطر الإيمان والحمد لله تملأ الميزان والتسبيح والتكبير…٢٤٣٨والذي نفسي بيده٢٤٣٩من أنفق زوجين من شىء من الأشياء في سبيل الله دعي من أبواب الجنة يا…٢٤٤٠هم الأخسرون ورب الكعبة٢٤٤١ما من رجل له مال لا يؤدي حق ماله إلا جعل له طوقا في عنقه شجاع أقرع…٢٤٤٢أيما رجل كانت له إبل لا يعطي حقها في نجدتها ورسلها٢٤٤٣أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…٢٤٤٤في كل إبل سائمة في كل أربعين ابنة لبون لا يفرق إبل عن حسابها من…٢٤٤٥ليس فيما دون خمسة أوسق صدقة ولا فيما دون خمس ذود صدقة ولا فيما دون…٢٤٤٦ليس فيما دون خمسة ذود صدقة وليس فيما دون خمسة أواق صدقة وليس فيما دون…٢٤٤٧أخبرنا محمد بن عبد الله بن المبارك، قال حدثنا المظفر بن مدرك أبو…٢٤٤٨تأتي الإبل على ربها على خير ما كانت إذا هي لم يعط فيها حقها تطؤه…٢٤٤٩في كل إبل سائمة من كل أربعين ابنة لبون لا تفرق إبل عن حسابها من…٢٤٥٠. أن رسول الله صلى الله عليه وسلم بعثه إلى اليمن وأمره أن يأخذ من كل…٢٤٥١والأعمش، عن إبراهيم، قالا قال معاذ بعثني رسول الله صلى الله عليه وسلم…٢٤٥٢قال لما بعثه رسول الله صلى الله عليه وسلم إلى اليمن أمره أن يأخذ من…٢٤٥٣قال أمرني رسول الله صلى الله عليه وسلم حين بعثني إلى اليمن أن لا آخذ…٢٤٥٤ما من صاحب إبل ولا بقر ولا غنم لا يؤدي حقها إلا وقف لها يوم القيامة…٢٤٥٥أن أبا بكر، رضى الله عنه كتب له أن هذه فرائض الصدقة التي فرض رسول…٢٤٥٦ما من صاحب إبل ولا بقر ولا غنم لا يؤدي زكاتها إلا جاءت يوم القيامة…٢٤٥٧إن في عهدي أن لا نأخذ راضع لبن ولا نجمع بين متفرق ولا نفرق بين مجتمع٢٤٥٨بعثنا مصدق الله ورسوله وإن فلانا أعطاه فصيلا مخلولا اللهم لا تبارك…٢٤٥٩اللهم صل على آل فلان٢٤٦٠ومحمد بن بشار، - واللفظ له - قالا حدثنا يحيى، عن محمد بن أبي إسماعيل،…٢٤٦١إذا أتاكم المصدق فليصدر وهو عنكم راض٢٤٦٢قال استعمل ابن علقمة أبي على عرافة قومه وأمره أن يصدقهم، فبعثني أبي…٢٤٦٣أن ابن علقمة، استعمل أباه على صدقة قومه وساق الحديث .٢٤٦٤ما ينقم ابن جميل إلا أنه كان فقيرا فأغناه الله وأما خالد بن الوليد…٢٤٦٥قال أمر رسول الله صلى الله عليه وسلم بصدقة مثله سواء .٢٤٦٦لولا أنها تعطى فقراء المهاجرين ما أخذتها٢٤٦٧ليس على المسلم في عبده ولا فرسه صدقة٢٤٦٨لا زكاة على الرجل المسلم في عبده ولا فرسه٢٤٦٩ليس على المسلم في عبده ولا في فرسه صدقة٢٤٧٠ليس على المرء في فرسه ولا في مملوكه صدقة٢٤٧١ليس على المسلم في عبده ولا في فرسه صدقة٢٤٧٢ليس على المسلم صدقة في غلامه ولا في فرسه٢٤٧٣ليس فيما دون خمسة أواق صدقة ولا فيما دون خمس ذود صدقة وليس فيما دون…٢٤٧٤ليس فيما دون خمس أوسق من التمر صدقة وليس فيما دون خمس أواق من الورق…٢٤٧٥لا صدقة فيما دون خمس أوساق من التمر ولا فيما دون خمس أواق من الورق…٢٤٧٦ليس فيما دون خمس أواق من الورق صدقة وليس فيما دون خمس من الإبل صدقة…٢٤٧٧قد عفوت عن الخيل والرقيق فأدوا زكاة أموالكم من كل مائتين خمسة٢٤٧٨قد عفوت عن الخيل والرقيق وليس فيما دون مائتين زكاة٢٤٧٩أتؤدين زكاة هذا٢٤٨٠قال جاءت امرأة ومعها بنت لها إلى رسول الله صلى الله عليه وسلم وفي يد…٢٤٨١إن الذي لا يؤدي زكاة ماله يخيل إليه ماله يوم القيامة شجاعا أقرع له…٢٤٨٢من آتاه الله عز وجل مالا فلم يؤد زكاته مثل له ماله يوم القيامة شجاعا…٢٤٨٣ليس فيما دون خمسة أوساق من حب أو تمر صدقة٢٤٨٤لا يحل في البر والتمر زكاة حتى تبلغ خمسة أوسق ولا يحل في الورق زكاة…٢٤٨٥ليس في حب ولا تمر صدقة حتى يبلغ خمسة أوسق ولا فيما دون خمس ذود ولا…٢٤٨٦ليس فيما دون خمس أواق صدقة٢٤٨٧ليس فيما دون خمس أواق صدقة ولا فيما دون خمس ذود صدقة وليس فيما دون…٢٤٨٨فيما سقت السماء والأنهار والعيون أو كان بعلا العشر وما سقي بالسواني…٢٤٨٩فيما سقت السماء والأنهار والعيون العشر وفيما سقي بالسانية نصف العشر٢٤٩٠قال بعثني رسول الله صلى الله عليه وسلم إلى اليمن فأمرني أن آخذ مما…٢٤٩١إذا خرصتم فخذوا ودعوا الثلث فإن لم تأخذوا أو تدعوا الثلث - شك شعبة -…٢٤٩٢والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن وهب، قال حدثني عبد…٢٤٩٣لو شاء رب هذه الصدقة تصدق بأطيب من هذا إن رب هذه الصدقة يأكل حشفا يوم…٢٤٩٤ما كان في طريق مأتي أو في قرية عامرة فعرفها سنة فإن جاء صاحبها وإلا…٢٤٩٥العجماء جرحها جبار والبئر جبار والمعدن جبار وفي الركاز الخمس٢٤٩٦وعبيد الله بن عبد الله، عن أبي هريرة، عن رسول الله صلى الله عليه وسلم…٢٤٩٧جرح العجماء جبار والبئر جبار والمعدن جبار وفي الركاز الخمس٢٤٩٨البئر جبار والعجماء جبار والمعدن جبار وفي الركاز الخمس٢٤٩٩قال جاء هلال إلى رسول الله صلى الله عليه وسلم بعشور نحل له وسأله أن…٢٥٠٠قال فرض رسول الله صلى الله عليه وسلم زكاة رمضان على الحر والعبد…٢٥٠١قال فرض رسول الله صلى الله عليه وسلم صدقة الفطر على الذكر والأنثى…٢٥٠٢قال فرض رسول الله صلى الله عليه وسلم زكاة رمضان على كل صغير وكبير حر…٢٥٠٣والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن القاسم،…٢٥٠٤قال فرض رسول الله صلى الله عليه وسلم زكاة الفطر صاعا من تمر أو صاعا…٢٥٠٥قال فرض رسول الله صلى الله عليه وسلم صدقة الفطر على الصغير والكبير…٢٥٠٦أخبرنا إسماعيل بن مسعود، قال حدثنا يزيد بن زريع، قال أنبأنا شعبة، عن…٢٥٠٧أخبرنا محمد بن عبد الله بن المبارك، قال حدثنا وكيع، عن سفيان، عن سلمة…٢٥٠٨أخبرنا محمد بن المثنى، قال حدثنا خالد، - وهو ابن الحارث - قال حدثنا…٢٥٠٩قال ذكر في صدقة الفطر قال صاعا من بر أو صاعا من تمر أو صاعا من شعير…٢٥١٠يخطب على منبركم - يعني منبر البصرة - يقول صدقة الفطر صاع من طعام .…٢٥١١قال فرض رسول الله صلى الله عليه وسلم صدقة الفطر صاعا من شعير أو صاعا…٢٥١٢أخبرنا محمد بن عبد الله بن المبارك، قال حدثنا وكيع، عن سفيان، عن زيد…٢٥١٣أخبرنا هناد بن السري، عن وكيع، عن داود بن قيس، عن عياض بن عبد الله،…٢٥١٤قال لم نخرج على عهد رسول الله صلى الله عليه وسلم إلا صاعا من تمر أو…٢٥١٥أن ابن عباس، خطب بالبصرة فقال أدوا زكاة صومكم . فجعل الناس ينظر بعضهم…٢٥١٦في عهد النبي صلى الله عليه وسلم صاعا من شعير أو تمر أو سلت أو زبيب .٢٥١٧أخبرنا عمرو بن علي، قال حدثنا يحيى، قال حدثنا داود بن قيس، قال حدثنا…٢٥١٨أن عياض بن عبد الله بن سعد، حدثه أن أبا سعيد الخدري قال كنا نخرج في…٢٥١٩أخبرنا عمرو بن زرارة، قال أنبأنا القاسم، - وهو ابن مالك - عن الجعيد،…٢٥٢٠المكيال مكيال أهل المدينة والوزن وزن أهل مكة٢٥٢١أن رسول الله صلى الله عليه وسلم أمر بصدقة الفطر أن تؤدى قبل خروج…٢٥٢٢إنك تأتي قوما أهل كتاب فادعهم إلى شهادة أن لا إله إلا الله وأني رسول…٢٥٢٣قال رجل لأتصدقن بصدقة فخرج بصدقته فوضعها في يد سارق فأصبحوا يتحدثون…٢٥٢٤إن الله عز وجل لا يقبل صلاة بغير طهور ولا صدقة من غلول٢٥٢٥ما تصدق أحد بصدقة من طيب ولا يقبل الله عز وجل إلا الطيب إلا أخذها…٢٥٢٦إيمان لا شك فيه وجهاد لا غلول فيه وحجة مبرورة٢٥٢٧سبق درهم مائة ألف درهم٢٥٢٨سبق درهم مائة ألف٢٥٢٩أخبرنا الحسين بن حريث، قال أنبأنا الفضل بن موسى، عن الحسين، عن منصور،…٢٥٣٠أخبرنا بشر بن خالد، قال حدثنا غندر، عن شعبة، عن سليمان، عن أبي وائل،…٢٥٣١إن هذا المال خضرة حلوة فمن أخذه بطيب نفس بورك له فيه ومن أخذه بإشراف…٢٥٣٢يد المعطي العليا وابدأ بمن تعول أمك وأباك وأختك وأخاك ثم أدناك أدناك٢٥٣٣اليد العليا خير من اليد السفلى واليد العليا المنفقة واليد السفلى…٢٥٣٤خير الصدقة ما كان عن ظهر غنى واليد العليا خير من اليد السفلى وابدأ…٢٥٣٥ومحمد بن المثنى، قالا حدثنا يحيى، عن ابن عجلان، عن سعيد، عن أبي…٢٥٣٦أن رجلا، دخل المسجد يوم الجمعة ورسول الله صلى الله عليه وسلم يخطب…٢٥٣٧مولى آبي اللحم قال أمرني مولاى أن أقدد، لحما فجاء مسكين فأطعمته منه…٢٥٣٨على كل مسلم صدقة٢٥٣٩إذا تصدقت المرأة من بيت زوجها كان لها أجر وللزوج مثل ذلك وللخازن مثل…٢٥٤٠لا يجوز لامرأة عطية إلا بإذن زوجها٢٥٤١رضى الله عنها أن أزواج، النبي صلى الله عليه وسلم اجتمعن عنده فقلن…٢٥٤٢أن تصدق وأنت صحيح شحيح تأمل العيش وتخشى الفقر٢٥٤٣أفضل الصدقة ما كان عن ظهر غنى واليد العليا خير من اليد السفلى وابدأ…٢٥٤٤خير الصدقة ما كان عن ظهر غنى وابدأ بمن تعول٢٥٤٥إذا أنفق الرجل على أهله وهو يحتسبها كانت له صدقة٢٥٤٦أخبرنا قتيبة، قال حدثنا الليث، عن أبي الزبير، عن جابر، قال أعتق رجل…٢٥٤٧إن مثل المنفق المتصدق والبخيل كمثل رجلين عليهما جبتان أو جنتان من…٢٥٤٨مثل البخيل والمتصدق مثل رجلين عليهما جنتان من حديد قد اضطرت أيديهما…٢٥٤٩أما تريدين أن لا يدخل بيتك شىء ولا يخرج إلا بعلمك٢٥٥٠لا تحصي فيحصي الله عز وجل عليك٢٥٥١ارضخي ما استطعت ولا توكي فيوكي الله عز وجل عليك٢٥٥٢اتقوا النار ولو بشق تمرة٢٥٥٣اتقوا النار ولو بشق التمرة فإن لم تجدوا فبكلمة طيبة٢٥٥٤{ يا أيها الناس اتقوا ربكم الذي خلقكم من نفس واحدة وخلق منها زوجها…٢٥٥٥تصدقوا فإنه سيأتي عليكم زمان يمشي الرجل بصدقته فيقول الذي يعطاها لو…٢٥٥٦اشفعوا تشفعوا ويقضي الله عز وجل على لسان نبيه ما شاء٢٥٥٧إن الرجل ليسألني الشىء فأمنعه حتى تشفعوا فيه فتؤجروا٢٥٥٨إن من الغيرة ما يحب الله عز وجل ومنها ما يبغض الله عز وجل ومن الخيلاء…٢٥٥٩كلوا وتصدقوا والبسوا في غير إسراف ولا مخيلة٢٥٦٠المؤمن للمؤمن كالبنيان يشد بعضه بعضا٢٥٦١الجاهر بالقرآن كالجاهر بالصدقة والمسر بالقرآن كالمسر بالصدقة٢٥٦٢ثلاثة لا ينظر الله عز وجل إليهم يوم القيامة العاق لوالديه والمرأة…٢٥٦٣ثلاثة لا يكلمهم الله عز وجل يوم القيامة ولا ينظر إليهم ولا يزكيهم…٢٥٦٤ثلاثة لا يكلمهم الله عز وجل يوم القيامة ولا ينظر إليهم ولا يزكيهم…٢٥٦٥ردوا السائل ولو بظلف٢٥٦٦لا يأتي رجل مولاه يسأله من فضل عنده فيمنعه إياه إلا دعي له يوم…٢٥٦٧من استعاذ بالله فأعيذوه ومن سألكم بالله فأعطوه ومن استجار بالله…٢٥٦٨أخبرنا محمد بن عبد الأعلى، قال حدثنا المعتمر، قال سمعت بهز بن حكيم،…٢٥٦٩ألا أخبركم بخير الناس منزلا٢٥٧٠ثلاثة يحبهم الله عز وجل وثلاثة يبغضهم الله عز وجل أما الذين يحبهم…٢٥٧١ليس المسكين الذي ترده التمرة والتمرتان واللقمة واللقمتان إن المسكين…٢٥٧٢ليس المسكين بهذا الطواف الذي يطوف على الناس ترده اللقمة واللقمتان…٢٥٧٣ليس المسكين الذي ترده الأكلة والأكلتان والتمرة والتمرتان٢٥٧٤إن لم تجدي شيئا تعطينه إياه إلا ظلفا محرقا فادفعيه إليه٢٥٧٥ثلاثة لا يكلمهم الله عز وجل يوم القيامة الشيخ الزاني والعائل المزهو…٢٥٧٦أربعة يبغضهم الله عز وجل البياع الحلاف والفقير المختال والشيخ الزاني…٢٥٧٧الساعي على الأرملة والمسكين كالمجاهد في سبيل الله عز وجل٢٥٧٨إنما فعلت ذلك لأتألفهم٢٥٧٩إن المسألة لا تحل إلا لثلاثة رجل تحمل بحمالة بين قوم فسأل فيها حتى…٢٥٨٠أقم يا قبيصة حتى تأتينا الصدقة فنأمر لك٢٥٨١إنما أخاف عليكم من بعدي ما يفتح لكم من زهرة٢٥٨٢إن الصدقة على المسكين صدقة وعلى ذي الرحم اثنتان صدقة وصلة٢٥٨٣تصدقن ولو من حليكن٢٥٨٤لأن يحتزم أحدكم حزمة حطب على ظهره فيبيعها خير من أن يسأل رجلا فيعطيه…
