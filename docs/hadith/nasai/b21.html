سنن النسائي

كتاب الجنائز

٢٧٢ حديثًا•صفحة ١ من ٢
١٨١٨لا يتمنين أحد منكم الموت إما محسنا فلعله أن يزداد خيرا وإما مسيئا…١٨١٩لا يتمنين أحدكم الموت إما محسنا فلعله أن يعيش يزداد خيرا وهو خير له…١٨٢٠لا يتمنين أحدكم الموت لضر نزل به في الدنيا ولكن ليقل اللهم أحيني ما…١٨٢١ألا لا يتمنى أحدكم الموت لضر نزل به فإن كان لا بد متمنيا الموت فليقل…١٨٢٢لا تدعوا بالموت ولا تتمنوه فمن كان داعيا لا بد فليقل اللهم أحيني ما…١٨٢٣أخبرنا محمد بن بشار، قال حدثنا يحيى بن سعيد، قال حدثنا إسماعيل، قال…١٨٢٤أكثروا ذكر هاذم اللذات١٨٢٥إذا حضرتم المريض فقولوا خيرا فإن الملائكة يؤمنون على ما تقولون١٨٢٦لقنوا موتاكم لا إله إلا الله١٨٢٧لقنوا هلكاكم قول لا إله إلا الله١٨٢٨موت المؤمن بعرق الجبين١٨٢٩المؤمن يموت بعرق الجبين١٨٣٠قالت مات رسول الله صلى الله عليه وسلم وإنه لبين حاقنتي وذاقنتي فلا…١٨٣١قال آخر نظرة نظرتها إلى رسول الله صلى الله عليه وسلم كشف الستارة…١٨٣٢يا ليته مات بغير مولده١٨٣٣إذا حضر المؤمن أتته ملائكة الرحمة بحريرة بيضاء فيقولون اخرجي راضية…١٨٣٤من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه١٨٣٥قال الله تعالى إذا أحب عبدي لقائي أحببت لقاءه وإذا كره لقائي كرهت…١٨٣٦من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه١٨٣٧من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه١٨٣٨من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه١٨٣٩أن أبا بكر، قبل بين عينى النبي صلى الله عليه وسلم وهو ميت .١٨٤٠ومحمد بن المثنى، قالا حدثنا يحيى، عن سفيان، قال حدثني موسى بن أبي…١٨٤١أخبرنا سويد، قال حدثنا عبد الله، قال قال معمر ويونس قال الزهري…١٨٤٢يقول جيء بأبي يوم أحد وقد مثل به فوضع بين يدى رسول الله صلى الله عليه…١٨٤٣يا أم أيمن أتبكين ورسول الله صلى الله عليه وسلم عندك١٨٤٤أن فاطمة، بكت على رسول الله صلى الله عليه وسلم حين مات فقالت يا أبتاه…١٨٤٥لا تبكيه ما زالت الملائكة تظله بأجنحتها حتى رفعتموه١٨٤٦قد غلبنا عليك أبا الربيع١٨٤٧أخبرنا يونس بن عبد الأعلى، قال حدثنا عبد الله بن وهب، قال قال معاوية…١٨٤٨الميت يعذب ببكاء أهله عليه١٨٤٩يقول ذكر عند عمران بن حصين الميت يعذب ببكاء الحى فقال عمران قاله رسول…١٨٥٠يعذب الميت ببكاء أهله عليه١٨٥١أن قيس بن عاصم، قال لا تنوحوا على فإن رسول الله صلى الله عليه وسلم لم…١٨٥٢لا إسعاد في الإسلام١٨٥٣الميت يعذب في قبره بالنياحة عليه١٨٥٤أخبرنا إبراهيم بن يعقوب، قال حدثنا سعيد بن سليمان، قال أنبأنا هشيم،…١٨٥٥إن الميت ليعذب ببكاء أهله عليه١٨٥٦إنهم ليبكون عليها وإنها لتعذب١٨٥٧إن الله عز وجل يزيد الكافر عذابا ببعض بكاء أهله عليه١٨٥٨إن الميت ليعذب ببعض بكاء أهله عليه١٨٥٩دعهن يا عمر فإن العين دامعة والقلب مصاب والعهد قريب١٨٦٠ليس منا من ضرب الخدود وشق الجيوب ودعا بدعاء الجاهلية١٨٦١أخبرنا عمرو بن علي، قال حدثنا سليمان بن حرب، قال حدثنا شعبة، عن عوف،…١٨٦٢ليس منا من ضرب الخدود وشق الجيوب ودعا بدعوى الجاهلية١٨٦٣أنا بريء ممن حلق وخرق وسلق١٨٦٤ليس منا من ضرب الخدود وشق الجيوب ودعا بدعوى الجاهلية١٨٦٥ليس منا من سلق وحلق وخرق١٨٦٦ليس منا من حلق وسلق وخرق١٨٦٧أخبرنا هناد، عن أبي معاوية، عن الأعمش، عن إبراهيم، عن سهم بن منجاب،…١٨٦٨إن لله ما أخذ وله ما أعطى وكل شىء عند الله بأجل مسمى فلتصبر ولتحتسب١٨٦٩الصبر عند الصدمة الأولى١٨٧٠رضى الله عنه أن رجلا، أتى النبي صلى الله عليه وسلم ومعه ابن له فقال…١٨٧١إن الله لا يرضى لعبده المؤمن إذا ذهب بصفيه من أهل الأرض فصبر واحتسب…١٨٧٢من احتسب ثلاثة من صلبه دخل الجنة١٨٧٣ما من مسلم يتوفى له ثلاثة من الولد لم يبلغوا الحنث إلا أدخله الله…١٨٧٤ما من مسلمين يموت بينهما ثلاثة أولاد لم يبلغوا الحنث إلا غفر الله…١٨٧٥لا يموت لأحد من المسلمين ثلاثة من الولد فتمسه النار إلا تحلة القسم١٨٧٦ما من مسلمين يموت بينهما ثلاثة أولاد لم يبلغوا الحنث إلا أدخلهما الله…١٨٧٧لقد احتظرت بحظار شديد من النار١٨٧٨أن رسول الله صلى الله عليه وسلم نعى زيدا وجعفرا قبل أن يجيء خبرهم…١٨٧٩وابن المسيب، أن أبا هريرة، أخبرهما أن رسول الله صلى الله عليه وسلم…١٨٨٠ما أخرجك من بيتك يا فاطمة١٨٨١اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك بماء وسدر واجعلن في…١٨٨٢ما قالت طال عمرها١٨٨٣أنهن جعلن رأس ابنة النبي صلى الله عليه وسلم ثلاثة قرون . قلت نقضنه…١٨٨٤ابدأن بميامنها ومواضع الوضوء منها١٨٨٥اغسلنها بماء وسدر واغسلنها وترا ثلاثا أو خمسا أو سبعا إن رأيتن ذلك…١٨٨٦اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك بماء وسدر واجعلن في…١٨٨٧اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن بماء وسدر واجعلن في…١٨٨٨ثلاثا أو خمسا أو سبعا أو أكثر من ذلك إن رأيتن ذلك١٨٨٩اغسلنها ثلاثا أو خمسا أو سبعا أو أكثر من ذلك إن رأيتن١٨٩٠اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك بماء وسدر واجعلن في…١٨٩١قال أخبرتني حفصة، عن أم عطية، قالت وجعلنا رأسها ثلاثة قرون .١٨٩٢وقالت، حفصة عن أم عطية، وجعلنا، رأسها ثلاثة قرون .١٨٩٣اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن بماء وسدر واجعلن في…١٨٩٤اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك واغسلنها بالسدر…١٨٩٥إذا ولي أحدكم أخاه فليحسن كفنه١٨٩٦البسوا من ثيابكم البياض فإنها أطهر وأطيب وكفنوا فيها موتاكم١٨٩٧قالت كفن النبي صلى الله عليه وسلم في ثلاثة أثواب سحولية بيض .١٨٩٨أن رسول الله صلى الله عليه وسلم كفن في ثلاثة أثواب بيض سحولية ليس…١٨٩٩قالت كفن رسول الله صلى الله عليه وسلم في ثلاثة أثواب بيض يمانية كرسف…١٩٠٠إذا فرغتم فآذنوني أصلي عليه١٩٠١يقول أتى النبي صلى الله عليه وسلم قبر عبد الله بن أبى وقد وضع في…١٩٠٢سمع جابرا، يقول وكان العباس بالمدينة فطلبت الأنصار ثوبا يكسونه فلم…١٩٠٣أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن الأعمش، ح وأخبرنا…١٩٠٤اغسلوا المحرم في ثوبيه اللذين أحرم فيهما واغسلوه بماء وسدر وكفنوه في…١٩٠٥أطيب الطيب المسك١٩٠٦من خير طيبكم المسك١٩٠٧إذا ماتت فآذنوني١٩٠٨إذا وضع الرجل الصالح على سريره قال قدموني قدموني وإذا وضع الرجل -…١٩٠٩إذا وضعت الجنازة فاحتملها الرجال على أعناقهم فإن كانت صالحة قالت…١٩١٠أسرعوا بالجنازة فإن تك صالحة فخير تقدمونها إليه وإن تك غير ذلك فشر…١٩١١أسرعوا بالجنازة فإن كانت صالحة قدمتموها إلى الخير وإن كانت غير ذلك…١٩١٢أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال أنبأنا عيينة بن عبد…١٩١٣وهشيم، عن عيينة بن عبد الرحمن، عن أبيه، عن أبي بكرة، قال لقد رأيتنا…١٩١٤إذا مرت بكم جنازة فقوموا فمن تبعها فلا يقعد حتى توضع١٩١٥إذا رأى أحدكم الجنازة فلم يكن ماشيا معها فليقم حتى تخلفه أو توضع من…١٩١٦إذا رأيتم الجنازة فقوموا حتى تخلفكم أو توضع١٩١٧إذا رأيتم الجنازة فقوموا فمن تبعها فلا يقعد حتى توضع١٩١٨وأبي، سعيد قالا ما رأينا رسول الله صلى الله عليه وسلم شهد جنازة قط…١٩١٩أن رسول الله صلى الله عليه وسلم مروا عليه بجنازة فقام . وقال عمرو إن…١٩٢٠يزيد بن ثابت أنهم كانوا جلوسا مع النبي صلى الله عليه وسلم فطلعت جنازة…١٩٢١قال كان سهل بن حنيف وقيس بن سعد بن عبادة بالقادسية فمر عليهما بجنازة…١٩٢٢إن للموت فزعا فإذا رأيتم الجنازة فقوموا١٩٢٣أخبرنا محمد بن منصور، قال حدثنا سفيان، عن ابن أبي نجيح، عن مجاهد، عن…١٩٢٤أن جنازة، مرت بالحسن بن علي وابن عباس فقام الحسن ولم يقم ابن عباس…١٩٢٥قال مر بجنازة على الحسن بن علي وابن عباس فقام الحسن ولم يقم ابن عباس…١٩٢٦والحسن بن علي، مرت بهما جنازة فقام أحدهما وقعد الآخر فقال الذي قام…١٩٢٧أن الحسن بن علي، كان جالسا فمر عليه بجنازة فقام الناس حتى جاوزت…١٩٢٨أنه سمع جابرا، يقول قام النبي صلى الله عليه وسلم لجنازة يهودي مرت به…١٩٢٩إنما قمنا للملائكة١٩٣٠مستريح ومستراح منه١٩٣١مستريح ومستراح منه المؤمن يموت فيستريح من أوصاب الدنيا ونصبها وأذاها…١٩٣٢قال مر بجنازة فأثني عليها خيرا فقال النبي صلى الله عليه وسلم " وجبت "…١٩٣٣وجده، أمية بن خلف قال سمعت عامر بن سعد، عن أبي هريرة، قال مروا بجنازة…١٩٣٤أيما مسلم شهد له أربعة قالوا خيرا أدخله الله الجنة١٩٣٥لا تذكروا هلكاكم إلا بخير١٩٣٦لا تسبوا الأموات فإنهم قد أفضوا إلى ما قدموا١٩٣٧يتبع الميت ثلاثة أهله وماله وعمله فيرجع اثنان أهله وماله ويبقى واحد…١٩٣٨للمؤمن على المؤمن ست خصال يعوده إذا مرض ويشهده إذا مات ويجيبه إذا…١٩٣٩أخبرنا سليمان بن منصور البلخي، قال حدثنا أبو الأحوص، ح وأنبأنا هناد…١٩٤٠من تبع جنازة حتى يصلى عليها كان له من الأجر قيراط ومن مشى مع الجنازة…١٩٤١من تبع جنازة حتى يفرغ منها فله قيراطان فإن رجع قبل أن يفرغ منها فله…١٩٤٢الراكب خلف الجنازة والماشي حيث شاء منها والطفل يصلى عليه١٩٤٣الراكب خلف الجنازة والماشي حيث شاء منها والطفل يصلى عليه١٩٤٤وعلي بن حجر، وقتيبة، عن سفيان، عن الزهري، عن سالم، عن أبيه، أنه رأى…١٩٤٥ومنصور، وزياد، وبكر، - هو ابن وائل - كلهم ذكروا أنهم سمعوا من الزهري،…١٩٤٦إن أخاكم قد مات فقوموا فصلوا عليه١٩٤٧أوغير ذلك يا عائشة خلق الله عز وجل الجنة وخلق لها أهلا وخلقهم في…١٩٤٨الراكب خلف الجنازة والماشي حيث شاء منها والطفل يصلى عليه١٩٤٩الله أعلم بما كانوا عاملين١٩٥٠الله أعلم بما كانوا عاملين١٩٥١خلقهم الله حين خلقهم وهو يعلم بما كانوا عاملين١٩٥٢الله أعلم بما كانوا عاملين١٩٥٣أن ابن أبي عمار، أخبره عن شداد بن الهاد، أن رجلا، من الأعراب جاء إلى…١٩٥٤إني فرط لكم وأنا شهيد عليكم١٩٥٥أيهما أكثر أخذا للقرآن١٩٥٦ونوح بن حبيب، قالا حدثنا عبد الرزاق، قال حدثنا معمر، عن الزهري، عن…١٩٥٧أحسن إليها فإذا وضعت فائتني بها١٩٥٨لقد هممت أن لا أصلي عليه١٩٥٩صلوا على صاحبكم إنه غل في سبيل الله١٩٦٠صلوا على صاحبكم فإن عليه دينا١٩٦١هل ترك عليه دينا١٩٦٢أخبرنا نوح بن حبيب القومسي، قال حدثنا عبد الرزاق، قال أنبأنا معمر، عن…١٩٦٣هل ترك لدينه من قضاء١٩٦٤أما أنا فلا أصلي عليه١٩٦٥من تردى من جبل فقتل نفسه فهو في نار جهنم يتردى خالدا مخلدا فيها أبدا…١٩٦٦أخر عني يا عمر١٩٦٧وعلي بن حجر، قالا حدثنا عبد العزيز بن محمد، عن عبد الواحد بن حمزة، عن…
