٢٢٠ حديثًا•صفحة ١ من ٢
٥٥٤٠أخبرنا أبو بكر، أحمد بن محمد بن إسحاق السني قراءة عليه في بيته قال…٥٥٤١أن أنس بن مالك، أخبرهم قال بينا أنا قائم، على الحى وأنا أصغرهم، سنا…٥٥٤٢أخبرنا سويد بن نصر، قال أنبأنا عبد الله، - يعني ابن المبارك - عن سعيد…٥٥٤٣قال حرمت الخمر حين حرمت وإنه لشرابهم البسر والتمر .٥٥٤٤- يعني ابن عبد الله - قال البسر والتمر خمر .٥٥٤٥قال البسر والتمر خمر . رفعه الأعمش .٥٥٤٦الزبيب والتمر هو الخمر٥٥٤٧أخبرنا إسحاق بن منصور، قال أنبأنا عبد الرحمن، عن شعبة، عن الحكم، عن…٥٥٤٨أخبرنا واصل بن عبد الأعلى، قال حدثنا ابن فضيل، عن حبيب بن أبي عمرة،…٥٥٤٩أخبرنا إسحاق بن إبراهيم، قال أنبأنا جرير، عن حبيب بن أبي عمرة، عن…٥٥٥٠أخبرنا الحسين بن منصور بن جعفر، قال حدثنا عبد الله بن نمير، قال حدثنا…٥٥٥١لا تجمعوا بين التمر والزبيب ولا بين الزهو والرطب٥٥٥٢لا تنبذوا الزهو والرطب جميعا ولا تنبذوا الزبيب والرطب جميعا٥٥٥٣قال نهى رسول الله صلى الله عليه وسلم أن يخلط التمر والزبيب وأن يخلط…٥٥٥٤أخبرنا يعقوب بن إبراهيم، عن يحيى، - وهو ابن سعيد - عن ابن جريج، قال…٥٥٥٥لا تخلطوا الزبيب والتمر ولا البسر والتمر٥٥٥٦أخبرنا قتيبة، قال حدثنا الليث، عن عطاء، عن جابر، عن رسول الله صلى…٥٥٥٧أن لا تخلطوا الزبيب والتمر جميعا٥٥٥٨قال البسر وحده حرام ومع التمر حرام .٥٥٥٩وعلي بن سعيد، قالا حدثنا عبد الرحيم، عن حبيب بن أبي عمرة، عن سعيد بن…٥٥٦٠أخبرنا قريش بن عبد الرحمن الباوردي، عن علي بن الحسن، قال أنبأنا…٥٥٦١لا تنبذوا الزهو والرطب ولا تنبذوا الرطب والزبيب جميعا٥٥٦٢أخبرنا قتيبة، قال حدثنا الليث، عن أبي الزبير، عن جابر، عن رسول الله…٥٥٦٣أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن وقاء بن إياس، عن المختار…٥٥٦٤قال شهدت أنس بن مالك أتي ببسر مذنب فجعل يقطعه منه .٥٥٦٤.٢قال قتادة كان أنس يأمر بالتذنوب فيقرض .٥٥٦٥أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن حميد، عن أنس، أنه كان لا…٥٥٦٦لا تنبذوا الزهو والرطب جميعا ولا البسر والزبيب جميعا وانبذوا كل واحد…٥٥٦٧لتنبذوا كل واحد منهما على حدة في الأسقية التي يلاث على أفواهها٥٥٦٨من شربه منكم فليشرب كل واحد منه فردا تمرا فردا أو بسرا فردا أو زبيبا…٥٥٦٩من شرب منكم فليشرب كل واحد منه فردا٥٥٧٠انبذوا كل واحد منهما على حدة٥٥٧١انتبذوا الزبيب فردا والتمر فردا والبسر فردا٥٥٧٢يقول قال رسول الله صلى الله عليه وسلم " الخمر من هاتين " . وقال سويد…٥٥٧٣الخمر من هاتين الشجرتين النخلة والعنبة٥٥٧٤والشعبي، قالا السكر خمر .٥٥٧٥قال السكر خمر .٥٥٧٦قال السكر خمر .٥٥٧٧قال السكر حرام والرزق الحسن حلال .٥٥٧٨رضى الله عنه يخطب على منبر المدينة فقال أيها الناس ألا إنه نزل تحريم…٥٥٧٩وأبي، حيان عن الشعبي، عن ابن عمر، قال سمعت عمر بن الخطاب، رضى الله…٥٥٨٠قال الخمر من خمسة من التمر والحنطة والشعير والعسل والعنب .٥٥٨١أخبرنا سويد بن نصر، قال حدثنا عبد الله، عن ابن عون، عن ابن سيرين، قال…٥٥٨٢كل مسكر حرام وكل مسكر خمر٥٥٨٣كل مسكر حرام وكل مسكر خمر٥٥٨٤قال قال رسول الله صلى الله عليه وسلم " كل مسكر خمر " .٥٥٨٥كل مسكر خمر وكل مسكر حرام٥٥٨٦كل مسكر حرام وكل مسكر خمر٥٥٨٧أخبرنا محمد بن المثنى، قال حدثنا يحيى بن سعيد، عن محمد بن عمرو، عن…٥٥٨٨قال قال رسول الله صلى الله عليه وسلم " كل مسكر حرام " .٥٥٨٩أن رسول الله صلى الله عليه وسلم نهى أن ينبذ في الدباء والمزفت والنقير…٥٥٩٠لا تنبذوا في الدباء ولا المزفت ولا النقير وكل مسكر حرام٥٥٩١كل شراب أسكر فهو حرام٥٥٩٢كل شراب أسكر حرام٥٥٩٣كل شراب أسكر فهو حرام والبتع من العسل٥٥٩٤كل شراب أسكر فهو حرام٥٥٩٥وعبد الله بن الهيثم، عن أبي داود، عن شعبة، عن سعيد بن أبي بردة، عن…٥٥٩٦اشرب ولا تشرب مسكرا٥٥٩٧قال قال رسول الله صلى الله عليه وسلم " كل مسكر حرام " .٥٥٩٨أخبرنا سويد، قال أنبأنا عبد الله، قال أنبأنا الأسود بن شيبان السدوسي،…٥٥٩٩قال كل مسكر حرام .٥٦٠٠أخبرنا سويد، قال أنبأنا عبد الله، عن عبد الملك بن الطفيل الجزري، قال…٥٦٠١قال كتب عمر بن عبد العزيز إلى عدي بن أرطاة كل مسكر حرام .٥٦٠٢أن رسول الله صلى الله عليه وسلم قال " كل مسكر حرام " .٥٦٠٣قال بعثني رسول الله صلى الله عليه وسلم إلى اليمن فقلت يا رسول الله إن…٥٦٠٤وما البتع والمزر٥٦٠٥قال خطب رسول الله صلى الله عليه وسلم فذكر آية الخمر فقال رجل يا رسول…٥٦٠٦وسئل، فقيل له أفتنا في الباذق . فقال سبق محمد الباذق وما أسكر فهو…٥٦٠٧ما أسكر كثيره فقليله حرام٥٦٠٨أنهاكم عن قليل ما أسكر كثيره٥٦٠٩أخبرنا محمد بن عبد الله بن عمار، قال حدثنا الوليد بن كثير، عن الضحاك…٥٦١٠قال علمت أن رسول الله صلى الله عليه وسلم كان يصوم فتحينت فطره بنبيذ…٥٦١١أخبرنا محمد بن عبد الله بن المبارك، قال حدثنا يحيى بن آدم، قال حدثنا…٥٦١٢أخبرنا قتيبة، قال حدثنا عبد الواحد، عن إسماعيل، - وهو ابن سميع - قال…٥٦١٣أخبرنا قتيبة، قال حدثنا أبو عوانة، عن أبي الزبير، عن جابر، أن النبي…٥٦١٤أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن سليمان التيمي، عن طاوس،…٥٦١٥وإبراهيم بن ميسرة، قالا سمعنا طاوسا، يقول جاء رجل إلى ابن عمر قال…٥٦١٦أخبرنا سويد، قال حدثنا عبد الله، عن عيينة بن عبد الرحمن، عن أبيه، قال…٥٦١٧أخبرنا علي بن الحسين، قال حدثنا أمية، عن شعبة، عن جبلة بن سحيم، عن…٥٦١٨أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا شعبة، عن أبي…٥٦١٩أخبرنا أحمد بن عبد الله بن علي بن سويد بن منجوف، قال حدثنا عبد الرحمن…٥٦٢٠أخبرنا عمرو بن زرارة، أنبأنا إسماعيل، عن أيوب، عن رجل، عن سعيد بن…٥٦٢١أخبرنا محمود بن غيلان، قال حدثنا أبو داود، قال أنبأنا شعبة، عن…٥٦٢٢أخبرنا أبو عبد الرحمن، قال أنبأنا محمد بن منصور، قال حدثنا سفيان، قال…٥٦٢٣أخبرنا محمد بن بشار، قال حدثنا محمد، قال حدثنا شعبة، عن أبي رجاء، قال…٥٦٢٤أخبرنا محمود بن غيلان، قال حدثنا أبو داود، قال حدثنا شعبة، عن إبراهيم…٥٦٢٥أخبرنا جعفر بن مسافر، قال حدثنا يحيى بن حسان، قال حدثنا وهيب، قال…٥٦٢٦وحماد، وسليمان، عن إبراهيم، عن الأسود، عن عائشة، قالت نهى رسول الله…٥٦٢٧أخبرنا محمد بن بشار، قال حدثنا يحيى، عن سفيان، عن سليمان، عن إبراهيم…٥٦٢٨والمزفت، .٥٦٢٩أخبرنا قتيبة، قال حدثنا الليث، عن ابن شهاب، عن أنس بن مالك، أنه أخبره…٥٦٣٠أنه سمع أبا هريرة، يقول نهى رسول الله صلى الله عليه وسلم عن الدباء…٥٦٣١أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن عبيد الله، قال أخبرني…٥٦٣٢أخبرنا أحمد بن عبد الله بن الحكم بن فروة، - يقال له ابن كردي بصري -…٥٦٣٣أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن المثنى بن سعيد، عن أبي…٥٦٣٤أخبرنا سويد، قال أنبأنا عبد الله، عن شعبة، عن محارب، قال سمعت ابن…٥٦٣٥أخبرنا سويد، قال أنبأنا عبد الله، عن الأوزاعي، قال حدثني يحيى، حدثني…٥٦٣٦وجميلة بنت عباد، أنهما سمعتا عائشة، قالت سمعت رسول الله صلى الله عليه…٥٦٣٧أخبرنا قريش بن عبد الرحمن، قال أنبأنا علي بن الحسن، قال أنبأنا…٥٦٣٨فقالت قدم وفد عبد القيس على رسول الله صلى الله عليه وسلم فسألوه فيما…٥٦٣٩بذاته .٥٦٤٠- وهو ابن سويد - يقول حدثتني معاذة، عن عائشة، أن رسول الله صلى الله…٥٦٤١أخبرنا سويد، قال أنبأنا عبد الله، عن طود بن عبد الملك القيسي، - بصري…٥٦٤٢أخبرنا زياد بن أيوب، قال حدثنا ابن إدريس، قال سمعت المختار بن فلفل،…٥٦٤٣سمع سعيد بن جبير، يحدث أنه سمع ابن عمر، وابن، عباس أنهما شهدا على…٥٦٤٤أخبرنا سويد، قال أنبأنا عبد الله، عن سليمان التيمي، عن أسماء بنت…٥٦٤٥أخبرنا عمرو بن يزيد، قال حدثنا بهز بن أسد، قال حدثنا شعبة، قال أخبرني…٥٦٤٦انتبذ في سقائك وأوكه واشربه حلوا٥٦٤٧أخبرنا سويد، قال أنبأنا عبد الله، عن ابن جريج، قراءة قال وقال أبو…٥٦٤٨أخبرني أحمد بن خالد، قال حدثنا إسحاق، - يعني الأزرق - قال حدثنا عبد…٥٦٤٩أخبرنا سوار بن عبد الله بن سوار، قال حدثنا خالد بن الحارث، قال حدثنا…٥٦٥٠أن النبي صلى الله عليه وسلم رخص في الجر غير مزفت .٥٦٥١إني كنت نهيتكم عن لحوم الأضاحي فتزودوا وادخروا ومن أراد زيارة القبور…٥٦٥٢إني كنت نهيتكم عن زيارة القبور فزوروها ونهيتكم عن لحوم الأضاحي فوق…٥٦٥٣إني كنت نهيتكم عن ثلاث زيارة القبور فزوروها ولتزدكم زيارتها خيرا…٥٦٥٤كنت نهيتكم عن الأوعية فانتبذوا فيما بدا لكم وإياكم وكل مسكر٥٦٥٥أخبرنا أبو علي، محمد بن يحيى بن أيوب - مروزي - قال حدثنا عبد الله بن…٥٦٥٦أخبرنا محمود بن غيلان، قال حدثنا أبو داود الحفري، وأبو أحمد الزبيري…٥٦٥٧قال أتي رسول الله صلى الله عليه وسلم ليلة أسري به بقدحين من خمر ولبن…٥٦٥٨يشرب ناس من أمتي الخمر يسمونها بغير اسمها٥٦٥٩لا يزني الزاني حين يزني وهو مؤمن ولا يشرب الخمر شاربها حين يشربها وهو…٥٦٦٠لا يزني الزاني حين يزني وهو مؤمن ولا يسرق السارق حين يسرق وهو مؤمن…٥٦٦١من شرب الخمر فاجلدوه ثم إن شرب فاجلدوه ثم إن شرب فاجلدوه ثم إن شرب…٥٦٦٢إذا سكر فاجلدوه ثم إن سكر فاجلدوه ثم إن سكر فاجلدوه٥٦٦٣رضى الله عنه أنه كان يقول ما أبالي شربت الخمر أو عبدت هذه السارية من…٥٦٦٤لا يشرب الخمر رجل من أمتي فيقبل الله منه صلاة أربعين يوما٥٦٦٥وعلي بن حجر، قالا حدثنا خلف، - يعني ابن خليفة - عن منصور بن زاذان، عن…٥٦٦٦أخبرنا سويد، قال أنبأنا عبد الله، عن معمر، عن الزهري، عن أبي بكر بن…٥٦٦٧أن أباه، قال سمعت عثمان، يقول اجتنبوا الخمر فإنها أم الخبائث فإنه كان…٥٦٦٨قال من شرب الخمر فلم ينتش لم تقبل له صلاة مادام في جوفه أو عروقه منها…٥٦٦٩من شرب الخمر فجعلها في بطنه لم يقبل الله منه صلاة سبعا إن مات فيها٥٦٧٠من شرب الخمر شربة لم تقبل له توبة أربعين صباحا فإن تاب تاب الله عليه…٥٦٧١من شرب الخمر في الدنيا ثم لم يتب منها حرمها في الآخرة٥٦٧٢لا يدخل الجنة منان ولا عاق ولا مدمن خمر٥٦٧٣من شرب الخمر في الدنيا فمات وهو يدمنها لم يتب منها لم يشربها في الآخرة٥٦٧٤من شرب الخمر في الدنيا فمات وهو يدمنها لم يشربها في الآخرة٥٦٧٥أخبرنا سويد، قال أنبأنا عبد الله، عن الحسن بن يحيى، عن الضحاك، قال من…٥٦٧٦قال غرب عمر رضى الله حجاج، قال حدثنا أبو…٥٦٨٠أن جسرة بنت دجاجة العامرية، حدثته قالت، سمعت عائشة، سألها أناس كلهم…٥٦٨١. ثم أقبلت على النساء فقالت إياكن والجر الأخضر وإن أسكركن ماء حبكن…٥٦٨٢قال حدثتني والدتي، عن عائشة، أنها
