سنن النسائي

كتاب الوصايا

٦١ حديثًا
٣٦١١أن تصدق وأنت صحيح شحيح تخشى الفقر وتأمل البقاء ولا تمهل حتى إذا بلغت…٣٦١٢أيكم مال وارثه أحب إليه من ماله٣٦١٣{ ألهاكم التكاثر * حتى زرتم المقابر } قال يقول ابن آدم مالي مالي…٣٦١٤مثل الذي يعتق أو يتصدق عند موته مثل الذي يهدي بعد ما يشبع٣٦١٥ما حق امرئ مسلم له شىء يوصى فيه أن يبيت ليلتين إلا ووصيته مكتوبة عنده٣٦١٦ما حق امرئ مسلم له شىء يوصى فيه يبيت ليلتين إلا ووصيته مكتوبة عنده٣٦١٧قوله .٣٦١٨ما حق امرئ مسلم تمر عليه ثلاث ليال إلا وعنده وصيته٣٦١٩ما حق امرئ مسلم له شىء يوصى فيه فيبيت ثلاث ليال إلا ووصيته عنده مكتوبة٣٦٢٠قال سألت ابن أبي أوفى أوصى رسول الله صلى الله عليه وسلم قال لا . قلت…٣٦٢١وأحمد بن حرب، قالا حدثنا أبو معاوية، عن الأعمش، عن شقيق، عن مسروق، عن…٣٦٢٢قالت ما ترك رسول الله صلى الله عليه وسلم درهما ولا دينارا ولا شاة ولا…٣٦٢٣وأحمد بن يوسف، قالا حدثنا عاصم بن يوسف، قال حدثنا حسن بن عياش، عن…٣٦٢٤قالت يقولون إن رسول الله صلى الله عليه وسلم أوصى إلى علي رضى الله عنه…٣٦٢٥قالت توفي رسول الله صلى الله عليه وسلم وليس عنده أحد غيري - قالت -…٣٦٢٦قال مرضت مرضا أشفيت منه فأتاني رسول الله صلى الله عليه وسلم يعودني…٣٦٢٧وأحمد بن سليمان، - واللفظ لأحمد - قالا حدثنا أبو نعيم، قال حدثنا…٣٦٢٨رحم الله سعد ابن عفراء أو يرحم الله سعد ابن عفراء٣٦٢٩آل سعد قال مرض سعد فدخل رسول الله صلى الله عليه وسلم فقال يا رسول…٣٦٣٠أنه اشتكى بمكة فجاءه رسول الله صلى الله عليه وسلم فلما رآه سعد بكى…٣٦٣١قال عادني رسول الله صلى الله عليه وسلم في مرضي فقال " أوصيت " . قلت…٣٦٣٢أن النبي صلى الله عليه وسلم عاده في مرضه فقال يا رسول الله أوصي بمالي…٣٦٣٣أن رسول الله صلى الله عليه وسلم أتى سعدا يعوده فقال له سعد يا رسول…٣٦٣٤الثلث والثلث كثير أو كبير٣٦٣٥سعد بن مالك أن النبي صلى الله عليه وسلم جاءه وهو مريض فقال إنه ليس لي…٣٦٣٦اذهب فبيدر كل تمر على ناحية٣٦٣٧أن أباه، توفي وعليه دين فأتيت النبي صلى الله عليه وسلم فقلت يا رسول…٣٦٣٨اذهب فصنف تمرك أصنافا العجوة على حدة وعذق ابن زيد على حدة وأصنافه ثم…٣٦٣٩هل لك أن تأخذ العام نصفه وتؤخر نصفه٣٦٤٠إذا جددته فوضعته في المربد فآذني٣٦٤١إن الله قد أعطى كل ذي حق حقه ولا وصية لوارث٣٦٤٢إن الله قد قسم لكل إنسان قسمه من الميراث فلا تجوز لوارث وصية٣٦٤٣إن الله عز اسمه قد أعطى كل ذي حق حقه ولا وصية لوارث٣٦٤٤يا بني كعب بن لؤى يا بني مرة بن كعب يا بني عبد شمس ويا بني عبد مناف…٣٦٤٥يا بني عبد مناف اشتروا أنفسكم من ربكم إني لا أملك لكم من الله شيئا يا…٣٦٤٦يا معشر قريش اشتروا أنفسكم من الله لا أغني عنكم من الله شيئا يا بني…٣٦٤٧يا معشر قريش اشتروا أنفسكم من الله لا أغني عنكم من الله شيئا يا بني…٣٦٤٨يا فاطمة ابنة محمد يا صفية بنت عبد المطلب يا بني عبد المطلب لا أغني…٣٦٤٩أن رجلا، قال لرسول الله صلى الله عليه وسلم إن أمي افتلتت نفسها وإنها…٣٦٥٠قال خرج سعد بن عبادة مع النبي صلى الله عليه وسلم في بعض مغازيه وحضرت…٣٦٥١إذا مات الإنسان انقطع عمله إلا من ثلاثة من صدقة جارية وعلم ينتفع به…٣٦٥٢أن رجلا، قال للنبي صلى الله عليه وسلم إن أبي مات وترك مالا ولم يوص…٣٦٥٣أخبرنا موسى بن سعيد، قال حدثنا هشام بن عبد الملك، قال حدثنا حماد بن…٣٦٥٤أن سعدا، سأل النبي صلى الله عليه وسلم إن أمي ماتت ولم توص أفأتصدق…٣٦٥٥أن رجلا، قال يا رسول الله إن أمه توفيت أفينفعها إن تصدقت عنها قال "…٣٦٥٦أخبرني هارون بن عبد الله، قال حدثنا عفان، قال حدثنا سليمان بن كثير،…٣٦٥٧أنه استفتى النبي صلى الله عليه وسلم في نذر كان على أمه فتوفيت قبل أن…٣٦٥٨أنه استفتى النبي صلى الله عليه وسلم في نذر كان على أمه فماتت قبل أن…٣٦٥٩أن عبيد الله بن عبد الله، أخبره عن ابن عباس، قال استفتى سعد رسول الله…٣٦٦٠أن سعد بن عبادة، استفتى النبي صلى الله عليه وسلم في نذر كان على أمه…٣٦٦١أنه قال ماتت أمي وعليها نذر فسألت النبي صلى الله عليه وسلم فأمرني أن…٣٦٦٢قال استفتى سعد بن عبادة الأنصاري رسول الله صلى الله عليه وسلم في نذر…٣٦٦٣قال جاء سعد بن عبادة إلى النبي صلى الله عليه وسلم فقال إن أمي ماتت…٣٦٦٤قال قلت يا رسول الله إن أمي ماتت أفأتصدق عنها قال " نعم " . قلت فأى…٣٦٦٥قال قلت يا رسول الله أى الصدقة أفضل قال " سقى الماء " .٣٦٦٦أن أمه، ماتت فقال يا رسول الله إن أمي ماتت أفأتصدق عنها قال " نعم " .…٣٦٦٧يا أبا ذر إني أراك ضعيفا وإني أحب لك ما أحب لنفسي لا تأمرن على اثنين…٣٦٦٨كل من مال يتيمك غير مسرف ولا مباذر ولا متأثل٣٦٦٩أخبرنا أحمد بن عثمان بن حكيم، قال حدثنا محمد بن الصلت، قال حدثنا أبو…٣٦٧٠في قوله { إن الذين يأكلون أموال اليتامى ظلما } قال كان يكون في حجر…٣٦٧١اجتنبوا
