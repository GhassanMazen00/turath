سنن النسائي

كتاب الطلاق

١٧٤ حديثًا•صفحة ١ من ٢
٣٣٨٩مر عبد الله فليراجعها ثم يدعها حتى تطهر من حيضتها هذه ثم تحيض حيضة…٣٣٩٠مره فليراجعها ثم ليمسكها حتى تطهر ثم تحيض ثم تطهر ثم إن شاء أمسك بعد…٣٣٩١ليراجعها ثم يمسكها حتى تحيض حيضة وتطهر فإن بدا له أن يطلقها طاهرا قبل…٣٣٩٢وعبد الله بن محمد بن تميم، عن حجاج، قال قال ابن جريج أخبرني أبو…٣٣٩٣في قوله عز وجل { يا أيها النبي إذا طلقتم النساء فطلقوهن لعدتهن } قال…٣٣٩٤أنه قال طلاق السنة تطليقة وهي طاهر في غير جماع فإذا حاضت وطهرت طلقها…٣٣٩٥قال طلاق السنة أن يطلقها طاهرا في غير جماع .٣٣٩٦مر عبد الله فليراجعها فإذا اغتسلت فليتركها حتى تحيض فإذا اغتسلت من…٣٣٩٧مره فليراجعها ثم ليطلقها وهي طاهر أو حامل٣٣٩٨أنه طلق امرأته وهي حائض فردها عليه رسول الله صلى الله عليه وسلم حتى…٣٣٩٩طلق امرأته وهي حائض فقال هل تعرف عبد الله بن عمر فإنه طلق امرأته وهي…٣٤٠٠قال قلت لابن عمر رجل طلق امرأته وهي حائض فقال أتعرف عبد الله بن عمر…٣٤٠١أيلعب بكتاب الله وأنا بين أظهركم٣٤٠٢قد نزل فيك وفي صاحبتك فاذهب فائت بها٣٤٠٣إنما النفقة والسكنى للمرأة إذا كان لزوجها عليها الرجعة٣٤٠٤المطلقة ثلاثا ليس لها سكنى ولا نفقة٣٤٠٥ليس لها نفقة ولا سكنى٣٤٠٦أن أبا الصهباء، جاء إلى ابن عباس فقال يا ابن عباس ألم تعلم أن الثلاث،…٣٤٠٧لا حتى يذوق الآخر عسيلتها وتذوق عسيلته٣٤٠٨لعلك تريدين أن ترجعي إلى رفاعة لا حتى يذوق عسيلتك وتذوقي عسيلته٣٤٠٩تريدين أن ترجعي إلى رفاعة لا حتى تذوقي عسيلته ويذوق عسيلتك٣٤١٠أخبرنا علي بن نصر بن علي، قال حدثنا سليمان بن حرب، قال حدثنا حماد بن…٣٤١١لعلك تريدين أن ترجعي إلى رفاعة لا حتى يذوق عسيلتك وتذوقي عسيلته٣٤١٢لا حتى يذوق عسيلتها كما ذاق الأول٣٤١٣ليس ذلك حتى تذوقي عسيلته٣٤١٤لا حتى تذوق العسيلة٣٤١٥لا تحل للأول حتى يجامعها الآخر٣٤١٦أخبرنا عمرو بن منصور، قال حدثنا أبو نعيم، عن سفيان، عن أبي قيس، عن…٣٤١٧لقد عذت بعظيم الحقي بأهلك٣٤١٨تقول أرسل إلى زوجي بطلاقي فشددت على ثيابي ثم أتيت النبي صلى الله عليه…٣٤١٩نحوه .٣٤٢٠قال أتاه رجل فقال إني جعلت امرأتي على حراما . قال كذبت ليست عليك…٣٤٢١بل شربت عسلا عند زينب - وقال - لن أعود له٣٤٢٢أخبرنا محمد بن حاتم بن نعيم، قال حدثنا محمد بن مكي بن عيسى، قال حدثنا…٣٤٢٣ومحمد بن يحيى بن محمد، قالا حدثنا محمد بن موسى بن أعين، قال حدثنا…٣٤٢٤أن عبد الله بن كعب، قال سمعت كعبا، يحدث حديثه حين تخلف عن رسول الله…٣٤٢٥يحدث قال أرسل إلى رسول الله صلى الله عليه وسلم وإلى صاحبى أن رسول…٣٤٢٦قال في حديثه إذا رسول من النبي صلى الله عليه وسلم قد أتاني فقال اعتزل…٣٤٢٧أن أبا حسن، مولى بني نوفل أخبره قال كنت أنا وامرأتي، مملوكين فطلقتها…٣٤٢٨امرأته تطليقتين ثم عتقا أيتزوجها قال نعم . قال عمن قال أفتى بذلك رسول…٣٤٢٩قريظة أنهم عرضوا على رسول الله صلى الله عليه وسلم يوم قريظة فمن كان…٣٤٣٠أخبرنا محمد بن منصور، قال حدثنا سفيان، عن عبد الملك بن عمير، عن عطية…٣٤٣١أن رسول الله صلى الله عليه وسلم عرضه يوم أحد وهو ابن أربع عشرة سنة…٣٤٣٢رفع القلم عن ثلاث عن النائم حتى يستيقظ وعن الصغير حتى يكبر وعن…٣٤٣٣إن الله تعالى تجاوز عن أمتي كل شىء حدثت به أنفسها ما لم تكلم به أو…٣٤٣٤إن الله عز وجل تجاوز لأمتي ما وسوست به وحدثت به أنفسها ما لم تعمل أو…٣٤٣٥إن الله تعالى تجاوز لأمتي عما حدثت به أنفسها ما لم تكلم أو تعمل به٣٤٣٦قال كان لرسول الله صلى الله عليه وسلم جار فارسي طيب المرقة فأتى رسول…٣٤٣٧إنما الأعمال بالنية وإنما لامرئ ما نوى فمن كانت هجرته إلى الله ورسوله…٣٤٣٨انظروا كيف يصرف الله عني شتم قريش ولعنهم إنهم يشتمون مذمما ويلعنون…٣٤٣٩إني ذاكر لك أمرا فلا عليك أن لا تعجلي حتى تستأمري أبويك٣٤٤٠يا عائشة إني ذاكر لك أمرا فلا عليك أن لا تعجلي حتى تستأمري أبويك٣٤٤١أخبرنا عمرو بن علي، قال حدثنا يحيى، - هو ابن سعيد - عن إسماعيل، عن…٣٤٤٢قالت قد خير رسول الله صلى الله عليه وسلم نساءه فلم يكن طلاقا .٣٤٤٣قالت قد خير النبي صلى الله عليه وسلم نساءه فلم يكن طلاقا .٣٤٤٤قالت قد خير رسول الله صلى الله عليه وسلم نساءه أفكان طلاقا .٣٤٤٥أخبرني عبد الله بن محمد الضعيف، قال حدثنا أبو معاوية، قال حدثنا…٣٤٤٦ابدئي بالغلام قبل الجارية٣٤٤٧الولاء لمن أعتق٣٤٤٨اشتريها وأعتقيها فإنما الولاء لمن أعتق٣٤٤٩أعتقيها فإنما الولاء لمن أعطى الورق٣٤٥٠اشتريها وأعتقيها فإن الولاء لمن أعتق٣٤٥١أخبرنا إسحاق بن إبراهيم، قال أنبأنا جرير، عن هشام بن عروة، عن أبيه،…٣٤٥٢رضى الله عنها قالت كان زوج بريرة عبدا .٣٤٥٣الولاء لمن ولي النعمة٣٤٥٤اشتريها فإن الولاء لمن أعتق٣٤٥٥لا ولكني آليت منهن شهرا٣٤٥٦الشهر تسع وعشرون٣٤٥٧وما حملك على ذلك يرحمك الله٣٤٥٨ما حملك على ذلك٣٤٥٩ما حملك على ذلك٣٤٦٠أنها قالت الحمد لله الذي وسع سمعه الأصوات لقد جاءت خولة إلى رسول الله…٣٤٦١المنتزعات والمختلعات هن المنافقات٣٤٦٢أخبرنا محمد بن سلمة، قال أنبأنا ابن القاسم، عن مالك، عن يحيى بن سعيد،…٣٤٦٣أتردين عليه حديقته٣٤٦٤قال جاء رجل إلى النبي صلى الله عليه وسلم فقال إن امرأتي لا تمنع يد…٣٤٦٥أن رجلا، قال يا رسول الله إن تحتي امرأة لا ترد يد لامس قال " طلقها "…٣٤٦٦قد أنزل الله عز وجل فيك وفي صاحبتك فائت بها٣٤٦٧حدثنا أحمد بن علي، قال حدثنا محمد بن أبي بكر، قال حدثنا عمر بن علي،…٣٤٦٨ابصروه فإن جاءت به أبيض سبطا قضيء العينين فهو لهلال بن أمية وإن جاءت…٣٤٦٩أربعة شهداء وإلا فحد في ظهرك٣٤٧٠أخبرنا عيسى بن حماد، قال أنبأنا الليث، عن يحيى بن سعيد، عن عبد الرحمن…٣٤٧١أخبرنا يحيى بن محمد بن السكن، قال حدثنا محمد بن جهضم، عن إسماعيل بن…٣٤٧٢أخبرنا علي بن ميمون، قال حدثنا سفيان، عن عاصم بن كليب، عن أبيه، عن…٣٤٧٣ومحمد بن المثنى، قالا حدثنا يحيى بن سعيد، قال حدثنا عبد الملك بن أبي…٣٤٧٤ومحمد بن المثنى، - واللفظ له - قالا حدثنا معاذ بن هشام، حدثني أبي، عن…٣٤٧٥الله يعلم إن أحدكما كاذب فهل منكما تائب٣٤٧٦حسابكما على الله أحدكما كاذب ولا سبيل لك عليها٣٤٧٧أخبرنا قتيبة، قال حدثنا مالك، عن نافع، عن ابن عمر، قال لاعن رسول الله…٣٤٧٨أن رجلا، من بني فزارة أتى رسول الله صلى الله عليه وسلم فقال إن امرأتي…٣٤٧٩قال جاء رجل من بني فزارة إلى النبي صلى الله عليه وسلم فقال إن امرأتي…٣٤٨٠قال بينما نحن عند رسول الله صلى الله عليه وسلم قام رجل فقال يا رسول…٣٤٨١أيما امرأة أدخلت على قوم رجلا ليس منهم فليست من الله في شىء ولا…٣٤٨٢الولد للفراش وللعاهر الحجر٣٤٨٣الولد للفراش وللعاهر الحجر٣٤٨٤هو لك يا عبد الولد للفراش وللعاهر الحجر واحتجبي منه يا سودة بنت زمعة٣٤٨٥الولد للفراش واحتجبي منه يا سودة فليس لك بأخ٣٤٨٦الولد للفراش وللعاهر الحجر٣٤٨٧الولد للفراش واحتجبي منه يا سودة٣٤٨٨قال أتي علي رضى الله عنه بثلاثة وهو باليمن وقعوا على امرأة في طهر…٣٤٨٩أخبرنا علي بن حجر، قال حدثنا علي بن مسهر، عن الأجلح، عن الشعبي، قال…٣٤٩٠قال كنت عند النبي صلى الله عليه وسلم وعلي رضى الله عنه يومئذ باليمن…٣٤٩١قال بعث رسول الله صلى الله عليه وسلم عليا على اليمن فأتي بغلام تنازع…٣٤٩٢أو ابن أبي الخليل أن ثلاثة، نفر اشتركوا في طهر فذكر نحوه ولم يذكر زيد…٣٤٩٣ألم ترى أن مجززا نظر إلى زيد بن حارثة وأسامة فقال إن بعض هذه الأقدام…٣٤٩٤يا عائشة ألم ترى أن مجززا المدلجي دخل على وعندي أسامة بن زيد فرأى…٣٤٩٥أخبرنا محمود بن غيلان، قال حدثنا عبد الرزاق، قال حدثنا سفيان، عن…٣٤٩٦يا غلام هذا أبوك وهذه أمك فخذ بيد أيهما شئت٣٤٩٧خذ الذي لها عليك وخل سبيلها٣٤٩٨قال قلت لها حدثيني حديثك، . قالت اختلعت من زوجي ثم جئت عثمان فسألته…٣٤٩٩أخبرنا زكريا بن يحيى، قال حدثنا إسحاق بن إبراهيم، قال أنبأنا علي بن…٣٥٠٠لا يحل لامرأة تؤمن بالله واليوم الآخر تحد على ميت فوق ثلاثة أيام إلا…٣٥٠١قد كانت إحداكن تمكث في بيتها في شر أحلاسها حولا ثم خرجت فلا أربعة…٣٥٠٢قد كانت إحداكن تجلس حولا وإنما هي أربعة أشهر وعشرا فإذا كان الحول…٣٥٠٣لا يحل لامرأة تؤمن بالله واليوم الآخر تحد على ميت فوق ثلاث إلا على…٣٥٠٤لا يحل لامرأة تؤمن بالله واليوم الآخر تحد على ميت أكثر من ثلاثة أيام…٣٥٠٥أخبرني محمد بن إسماعيل بن إبراهيم، قال حدثنا السهمي، - يعني عبد الله…٣٥٠٦والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ لمحمد - قالا أنبأنا…٣٥٠٧أن النبي صلى الله عليه وسلم أمر سبيعة أن تنكح إذا تعلت من نفاسها .٣٥٠٨ما يمنعها قد انقضى أجلها٣٥٠٩قد حللت فانكحي من شئت٣٥١٠قد حللت فانكحي من شئت٣٥١١أخبرني محمد بن عبد الله بن بزيع، قال حدثنا يزيد، - وهو ابن زريع - قال…٣٥١٢أن أبا هريرة، وابن، عباس وأبا سلمة بن عبد الرحمن تذاكروا عدة المتوفى…٣٥١٣ومحمد بن عمرو، عن أبي سلمة، عن كريب، عن أم سلمة، قالت وضعت سبيعة بعد…٣٥١٤أن عبد الله بن عباس، وأبا، سلمة بن عبد الرحمن اختلفا في المرأة تنفس…٣٥١٥أخبرنا حسين بن منصور، قال حدثنا جعفر بن عون، قال حدثنا يحيى بن سعيد،…٣٥١٦أن زينب بنت أبي سلمة، أخبرته عن أمها أم سلمة، زوج النبي صلى الله عليه…٣٥١٧أن أبا سلمة بن عبد الرحمن، أخبره قال بينما أنا وأبو هريرة عند ابن…٣٥١٨أن عبيد الله بن عبد الله، حدثه أن أباه كتب إلى عمر بن عبد الله بن…٣٥١٩أخبرنا محمد بن وهب، قال حدثنا محمد بن سلمة، قال حدثني أبو عبد الرحيم،…٣٥٢٠قد حللت حين وضعت حملك٣٥٢١أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا ابن عون، عن…٣٥٢٢أن ابن مسعود، قال من شاء لاعنته ما أنزلت { وأولات الأحمال أجلهن أن…٣٥٢٣ومسروق، وعبيدة، عن عبد الله، أن سورة النساء القصرى، نزلت بعد البقرة .٣٥٢٤أخبرنا محمود بن غيلان، قال حدثنا زيد بن الحباب، قال حدثنا سفيان، عن…٣٥٢٥لا يحل لامرأة تحد على ميت أكثر من ثلاث إلا على زوجها٣٥٢٦لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد فوق ثلاثة أيام إلا على…٣٥٢٧لا يحل لامرأة تؤمن بالله ورسوله أن تحد على ميت فوق ثلاث ليال إلا على…٣٥٢٨اجلسي في بيتك حتى يبلغ الكتاب أجله٣٥٢٩أن زوجها، تكارى علوجا ليعملوا له فقتلوه فذكرت ذلك لرسول الله صلى الله…٣٥٣٠امكثي في أهلك حتى يبلغ الكتاب أجله٣٥٣١نسخت هذه الآية عدتها في أهلها فتعتد حيث شاءت وهو قول الله عز وجل {…٣٥٣٢امكثي في بيتك أربعة أشهر وعشرا حتى يبلغ الكتاب أجله٣٥٣٣لا يحل لامرأة تؤمن بالله واليوم الآخر تحد على ميت فوق ثلاث ليال إلا…٣٥٣٣.٢لا يحل لامرأة تؤمن بالله واليوم الآخر تحد على ميت فوق ثلاث ليال إلا…٣٥٣٣.٣تقول جاءت امرأة إلى رسول الله صلى الله عليه وسلم فقالت يا رسول الله…٣٥٣٤لا تحد امرأة على ميت فوق ثلاث إلا على زوج فإنها تحد عليه أربعة أشهر…٣٥٣٥المتوفى عنها زوجها لا تلبس المعصفر من الثياب ولا الممشقة ولا تختضب…٣٥٣٦لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد على ميت فوق ثلاث إلا على…
