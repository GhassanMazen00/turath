كتاب صلاة العيدين

٤٢ حديثًا
١٥٥٦كان لكم يومان تلعبون فيهما وقد أبدلكم الله بهما خيرا منهما يوم الفطر…١٥٥٧له أن قوما، رأوا الهلال فأتوا النبي صلى الله عليه وسلم فأمرهم أن…١٥٥٨ليخرج العواتق وذوات الخدور والحيض ويشهدن العيد ودعوة المسلمين وليعتزل…١٥٥٩أخرجوا العواتق وذوات الخدور فيشهدن العيد ودعوة المسلمين وليعتزل الحيض…١٥٦٠إنما هذه لباس من لا خلاق له أو إنما يلبس هذه من لا خلاق له١٥٦١أن عليا، استخلف أبا مسعود على الناس فخرج يوم عيد فقال يا أيها الناس…١٥٦٢أخبرنا قتيبة، قال حدثنا أبو عوانة، عن عبد الملك بن أبي سليمان، عن…١٥٦٣إن أول ما نبدأ به في يومنا هذا أن نصلي ثم نذبح فمن فعل ذلك فقد أصاب…١٥٦٤أن رسول الله صلى الله عليه وسلم وأبا بكر وعمر - رضى الله عنهما -…١٥٦٥أن رسول الله صلى الله عليه وسلم كان يخرج العنزة يوم الفطر ويوم الأضحى…١٥٦٦- رضى الله عنه - قال صلاة الأضحى ركعتان وصلاة الفطر ركعتان وصلاة…١٥٦٧قال خرج عمر - رضى الله عنه - يوم عيد فسأل أبا واقد الليثي بأى شىء كان…١٥٦٨أن رسول الله صلى الله عليه وسلم كان يقرأ في العيدين ويوم الجمعة ب {…١٥٦٩يقول أشهد أني شهدت العيد مع رسول الله صلى الله عليه وسلم فبدأ بالصلاة…١٥٧٠أخبرنا قتيبة، قال حدثنا أبو الأحوص، عن منصور، عن الشعبي، عن البراء بن…١٥٧١من أحب أن ينصرف فلينصرف ومن أحب أن يقيم للخطبة فليقم١٥٧٢قال رأيت النبي صلى الله عليه وسلم يخطب وعليه بردان أخضران .١٥٧٣قال رأيت النبي صلى الله عليه وسلم يخطب على ناقة وحبشي آخذ بخطام…١٥٧٤قال سألت جابرا أكان رسول الله صلى الله عليه وسلم يخطب قائما قال كان…١٥٧٥تصدقن فإن أكثركن حطب جهنم١٥٧٦أن رسول الله صلى الله عليه وسلم كان يخرج يوم الفطر ويوم الأضحى إلى…١٥٧٧إذا قلت لصاحبك أنصت والإمام يخطب فقد لغوت١٥٧٨من يهده الله فلا مضل له ومن يضلله فلا هادي له إن أصدق الحديث كتاب…١٥٧٩أن رسول الله صلى الله عليه وسلم كان يخرج يوم العيد فيصلي ركعتين ثم…١٥٨٠أن ابن عباس، خطب بالبصرة فقال أدوا زكاة صومكم فجعل الناس ينظر بعضهم…١٥٨١من صلى صلاتنا ونسك نسكنا فقد أصاب النسك ومن نسك قبل الصلاة فتلك شاة…١٥٨٢قال كنت أصلي مع النبي صلى الله عليه وسلم فكانت صلاته قصدا وخطبته قصدا…١٥٨٣قال رأيت رسول الله صلى الله عليه وسلم يخطب قائما ثم يقعد قعدة لا…١٥٨٤قال كان النبي صلى الله عليه وسلم يخطب قائما ثم يجلس ثم يقوم ويقرأ…١٥٨٥صدق الله { إنما أموالكم وأولادكم فتنة } رأيت هذين يمشيان ويعثران في…١٥٨٦قال له رجل شهدت الخروج مع رسول الله صلى الله عليه وسلم قال نعم ولولا…١٥٨٧أن النبي صلى الله عليه وسلم خرج يوم العيد فصلى ركعتين لم يصل قبلها…١٥٨٨أخبرنا إسماعيل بن مسعود، قال حدثنا حاتم بن وردان، عن أيوب، عن محمد بن…١٥٨٩أن عبد الله بن عمر، أخبره أن رسول الله صلى الله عليه وسلم كان يذبح أو…١٥٩٠قال كان رسول الله صلى الله عليه وسلم يقرأ في الجمعة والعيد ب { سبح…١٥٩١سأل زيد بن أرقم أشهدت مع رسول الله صلى الله عليه وسلم عيدين قال نعم…١٥٩٢قال اجتمع عيدان على عهد ابن الزبير فأخر الخروج حتى تعالى النهار ثم…١٥٩٣دعهن فإن لكل قوم عيدا١٥٩٤أخبرنا محمد بن آدم، عن عبدة، عن هشام، عن أبيه، عن عائشة، قالت جاء…١٥٩٥أخبرنا علي بن خشرم، قال حدثنا الوليد، قال حدثنا الأوزاعي، عن الزهري،…١٥٩٦دعهم يا عمر فإنما هم بنو أرفدة١٥٩٧دعهما يا أبا
