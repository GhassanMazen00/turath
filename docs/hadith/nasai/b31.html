سنن النسائي

كتاب النحل

١٦ حديثًا
٣٦٧٢أكل ولدك نحلت٣٦٧٣أكل ولدك نحلته٣٦٧٤أن أباه، بشير بن سعد جاء بابنه النعمان فقال يا رسول الله إني نحلت…٣٦٧٥أكل بنيك نحلته٣٦٧٦أن أباه، نحله نحلا فقالت له أمه أشهد النبي صلى الله عليه وسلم على ما…٣٦٧٧أكل ولدك نحلته مثل ذا٣٦٧٨أن بشيرا، أتى النبي صلى الله عليه وسلم فقال يا نبي الله نحلت النعمان…٣٦٧٩كل بنيك نحلت مثل الذي نحلت النعمان٣٦٨٠أكل ولدك نحلت مثل ما نحلته٣٦٨١يا بشير ألك ولد سوى هذا٣٦٨٢يا بشير ألك ابن غير هذا٣٦٨٣هل لك بنون سواه٣٦٨٤هل لك ولد غيره٣٦٨٥يقول ذهب بي أبي إلى النبي صلى الله عليه وسلم يشهده على شىء أعطانيه…٣٦٨٦هل لك بنون سواه٣٦٨٧اعدلوا بين أبنائكم اعدلوا بين أبنائكم
