سنن النسائي

كتاب الخيل

٣٣ حديثًا
٣٥٦١كذبوا الآن الآن جاء القتال ولا يزال من أمتي أمة يقاتلون على الحق…٣٥٦٢الخيل معقود في نواصيها الخير إلى يوم القيامة الخيل ثلاثة فهي لرجل أجر…٣٥٦٣الخيل لرجل أجر ولرجل ستر وعلى رجل وزر فأما الذي هي له أجر فرجل ربطها…٣٥٦٤قال لم يكن شىء أحب إلى رسول الله صلى الله عليه وسلم بعد النساء من…٣٥٦٥تسموا بأسماء الأنبياء وأحب الأسماء إلى الله عز وجل عبد الله وعبد…٣٥٦٦قال كان النبي صلى الله عليه وسلم يكره الشكال من الخيل . واللفظ…٣٥٦٧أخبرنا محمد بن بشار، قال حدثنا يحيى، قال حدثنا سفيان، قال حدثني سلم…٣٥٦٨الشؤم في ثلاثة المرأة والفرس والدار٣٥٦٩الشؤم في الدار والمرأة والفرس٣٥٧٠إن يك في شىء ففي الربعة والمرأة والفرس٣٥٧١البركة في نواصي الخيل٣٥٧٢الخيل معقود في نواصيها الخير إلى يوم القيامة الأجر والغنيمة٣٥٧٣الخيل في نواصيها الخير إلى يوم القيامة٣٥٧٤الخيل معقود في نواصيها الخير إلى يوم القيامة٣٥٧٥الخيل معقود في نواصيها الخير إلى يوم القيامة الأجر والمغنم٣٥٧٦الخيل معقود في نواصيها الخير إلى يوم القيامة الأجر والمغنم٣٥٧٧الخيل معقود في نواصيها الخير إلى يوم القيامة الأجر والمغنم٣٥٧٨إن الله يدخل بالسهم الواحد ثلاثة نفر الجنة صانعه يحتسب في صنعه الخير…٣٥٧٩ما من فرس عربي إلا يؤذن له عند كل سحر بدعوتين اللهم خولتني من خولتني…٣٥٨٠إنما يفعل ذلك الذين لا يعلمون٣٥٨١أخبرنا حميد بن مسعدة، قال حدثنا حماد، عن أبي جهضم، عن عبد الله بن…٣٥٨٢من احتبس فرسا في سبيل الله إيمانا بالله وتصديقا لوعد الله كان شبعه…٣٥٨٣أن رسول الله صلى الله عليه وسلم سابق بين الخيل يرسلها من الحفياء وكان…٣٥٨٤والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٣٥٨٥لا سبق إلا في نصل أو حافر أو خف٣٥٨٦لا سبق إلا في نصل أو خف أو حافر٣٥٨٧رضى الله عنه قال لا يحل سبق إلا على خف أو حافر .٣٥٨٨إن حقا على الله أن لا يرتفع من الدنيا شىء إلا وضعه٣٥٨٩لا سبق إلا في خف أو حافر٣٥٩٠لا جلب ولا جنب ولا شغار في الإسلام ومن انتهب نهبة فليس منا٣٥٩١لا جلب ولا جنب ولا شغار في الإسلام٣٥٩٢حق على الله أن لا يرفع شىء نفسه في الدنيا إلا وضعه الله٣٥٩٣أنه كان يقول ضرب رسول الله صلى الله عليه وسلم عام خيبر للزبير بن…
