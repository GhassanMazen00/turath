سنن النسائي

كتاب الصيد والذبائح

٩٨ حديثًا
٤٢٦٣إذا أرسلت كلبك فاذكر اسم الله عليه فإن أدركته لم يقتل فاذبح واذكر اسم…٤٢٦٤ما أصبت بحده فكل وما أصبت بعرضه فهو وقيذ٤٢٦٥إذا أرسلت الكلب المعلم وذكرت اسم الله عليه فأخذ فكل٤٢٦٦ما أصبت بقوسك فاذكر اسم الله عليه وكل وما أصبت بكلبك المعلم فاذكر اسم…٤٢٦٧إذا أرسلت كلابك المعلمة فأمسكن عليك فكل٤٢٦٨إذا أرسلت كلبك فخالطته أكلب لم تسم عليها فلا تأكل فإنك لا تدري أيها…٤٢٦٩إذا أرسلت كلبك فسميت فكل وإن وجدت كلبا آخر مع كلبك فلا تأكل فإنما…٤٢٧٠لا تأكل فإنما سميت على كلبك ولم تسم على غيره٤٢٧١أخبرنا أحمد بن عبد الله بن الحكم، قال حدثنا محمد، قال حدثنا شعبة، عن…٤٢٧٢إذا أرسلت كلبك فسميت فكل وإن أكل منه فلا تأكل فإنما أمسك على نفسه…٤٢٧٣لا تأكل فإنما سميت على كلبك ولم تسم على غيره٤٢٧٤ما أصاب بحده فكل وما أصاب بعرضه فهو وقيذ٤٢٧٥إذا أرسلت كلبك فذكرت اسم الله عليه فقتل ولم يأكل فكل وإن أكل منه فلا…٤٢٧٦قال أخبرتني ميمونة، أن رسول الله صلى الله عليه وسلم قال له جبريل عليه…٤٢٧٧أن رسول الله صلى الله عليه وسلم أمر بقتل الكلاب غير ما استثنى منها .٤٢٧٨أخبرنا وهب بن بيان، قال حدثنا ابن وهب، قال أخبرني يونس، قال قال ابن…٤٢٧٩أن رسول الله صلى الله عليه وسلم أمر بقتل الكلاب إلا كلب صيد أو كلب…٤٢٨٠لولا أن الكلاب أمة من الأمم لأمرت بقتلها فاقتلوا منها الأسود البهيم…٤٢٨١الملائكة لا تدخل بيتا فيه صورة ولا كلب ولا جنب٤٢٨٢لا تدخل الملائكة بيتا فيه كلب ولا صورة٤٢٨٣إن جبريل عليه السلام كان وعدني أن يلقاني الليلة فلم يلقني أما والله…٤٢٨٤من اقتنى كلبا نقص من أجره كل يوم قيراطان إلا ضاريا أو صاحب ماشية٤٢٨٥من اقتنى كلبا لا يغني عنه زرعا ولا ضرعا نقص من عمله كل يوم قيراط٤٢٨٦من أمسك كلبا إلا كلبا ضاريا أو كلب ماشية نقص من أجره كل يوم قيراطان٤٢٨٧من اقتنى كلبا إلا كلب صيد أو ماشية نقص من أجره كل يوم قيراطان٤٢٨٨من اتخذ كلبا إلا كلب صيد أو ماشية أو زرع نقص من أجره كل يوم قيراط٤٢٨٩من اتخذ كلبا إلا كلب صيد أو زرع أو ماشية نقص من عمله كل يوم قيراط٤٢٩٠من اقتنى كلبا ليس بكلب صيد ولا ماشية ولا أرض فإنه ينقص من أجره…٤٢٩١من اقتنى كلبا إلا كلب ماشية أو كلب صيد نقص من عمله كل يوم قيراط٤٢٩٢أنه سمع أبا مسعود، عقبة قال نهى رسول الله صلى الله عليه وسلم عن ثمن…٤٢٩٣لا يحل ثمن الكلب ولا حلوان الكاهن ولا مهر البغي٤٢٩٤شر الكسب مهر البغي وثمن الكلب وكسب الحجام٤٢٩٥أخبرني إبراهيم بن الحسن المقسمي، قال حدثنا حجاج بن محمد، عن حماد بن…٤٢٩٦ما أمسك عليك كلابك فكل٤٢٩٧إن لهذه البهائم أوابد كأوابد الوحش فما غلبكم منها فاصنعوا به هكذا٤٢٩٨إذا رميت سهمك فاذكر اسم الله عز وجل فإن وجدته قد قتل فكل إلا أن تجده…٤٢٩٩إذا أرسلت سهمك وكلبك وذكرت اسم الله فقتل سهمك فكل٤٣٠٠إذا وجدت السهم فيه ولم تجد فيه أثر سبع وعلمت أن سهمك قتله فكل٤٣٠١إذا رأيت سهمك فيه ولم تر فيه أثرا غيره وعلمت أنه قتله فكل٤٣٠٢إذا وجدت فيه سهمك ولم يأكل منه سبع فكل٤٣٠٣أخبرني أحمد بن خالد الخلال، قال حدثنا معن، قال أنبأنا معاوية، - وهو…٤٣٠٤أهرق الدم بما شئت واذكر اسم الله عز وجل٤٣٠٥إذا أرسلت الكلاب - يعني المعلمة - وذكرت اسم الله فأمسكن عليك فكل٤٣٠٦إذا أصاب بحده فكل وإذا أصاب بعرضه فقتل فإنه وقيذ فلا تأكل٤٣٠٧إذا أصاب بحده فكل وإذا أصاب بعرضه فلا تأكل٤٣٠٨ما أصبت بحده فكل وما أصاب بعرضه فهو وقيذ٤٣٠٩من سكن البادية جفا ومن اتبع الصيد غفل ومن اتبع السلطان افتتن٤٣١٠ما يمنعك أن تأكل٤٣١١وعمرو بن عثمان، ومحمد بن عبد الرحمن، عن موسى بن طلحة، عن ابن…٤٣١٢أخبرنا إسماعيل بن مسعود، قال حدثنا خالد، عن شعبة، عن هشام، - وهو ابن…٤٣١٣وداود، عن الشعبي، عن ابن صفوان، قال أصبت أرنبين فلم أجد ما أذكيهما به…٤٣١٤لا آكله ولا أحرمه٤٣١٥لست بآكله ولا محرمه٤٣١٦لا ولكن لم يكن بأرض قومي فأجدني أعافه٤٣١٧لا ولكنه طعام ليس في أرض قومي فأجدني أعافه٤٣١٨أخبرنا إسماعيل بن مسعود، قال حدثنا خالد، قال حدثنا شعبة، عن أبي بشر،…٤٣١٩أخبرنا زياد بن أيوب، قال حدثنا هشيم، قال أنبأنا أبو بشر، عن سعيد بن…٤٣٢٠إن أمة من بني إسرائيل مسخت دواب في الأرض وإني لا أدري أى الدواب هي٤٣٢١إن أمة مسخت لا يدرى ما فعلت وإني لا أدري لعل هذا منها٤٣٢٢إن أمة مسخت والله أعلم٤٣٢٣أخبرنا محمد بن منصور، قال حدثنا سفيان، قال حدثني ابن جريج، عن عبد…٤٣٢٤كل ذي ناب من السباع فأكله حرام٤٣٢٥ومحمد بن المثنى، عن سفيان، عن الزهري، عن أبي إدريس، عن أبي ثعلبة…٤٣٢٦لا تحل النهبى ولا يحل من السباع كل ذي ناب ولا تحل المجثمة٤٣٢٧وأحمد بن عبدة، قالا حدثنا حماد، عن عمرو، - وهو ابن دينار - عن محمد بن…٤٣٢٨أخبرنا قتيبة، قال حدثنا سفيان، عن عمرو، عن جابر، قال أطعمنا رسول الله…٤٣٢٩وعمرو بن دينار، عن جابر، وعن ابن أبي نجيح، عن عطاء، عن جابر، قال…٤٣٣٠أخبرنا علي بن حجر، قال حدثنا عبيد الله، - وهو ابن عمرو - قال حدثنا…٤٣٣١لا يحل أكل لحوم الخيل والبغال والحمير٤٣٣٢أخبرنا كثير بن عبيد، قال حدثنا بقية، عن ثور بن يزيد، عن صالح بن يحيى…٤٣٣٣أخبرنا محمد بن المثنى، عن عبد الرحمن، عن سفيان، عن عبد الكريم، عن…٤٣٣٤والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن سفيان، عن…٤٣٣٥ومالك، وأسامة، عن ابن شهاب، عن الحسن، وعبد الله، ابنى محمد عن أبيهما،…٤٣٣٦أخبرنا إسحاق بن إبراهيم، قال أنبأنا محمد بن بشر، قال أنبأنا عبيد…٤٣٣٧أن النبي صلى الله عليه وسلم مثله ولم يقل خيبر .٤٣٣٨أخبرنا محمد بن عبد الأعلى، قال حدثنا عبد الرزاق، قال حدثنا معمر، عن…٤٣٣٩أخبرنا محمد بن عبد الله بن يزيد المقرئ، قال حدثنا سفيان، عن أبي إسحاق…٤٣٤٠الله أكبر الله أكبر خربت خيبر إنا إذا نزلنا بساحة قوم فساء صباح…٤٣٤١ألا إن لحوم الحمر الإنس لا تحل لمن يشهد أني رسول الله٤٣٤٢أخبرنا عمرو بن عثمان، عن بقية، قال حدثني الزبيدي، عن الزهري، عن أبي…٤٣٤٣أخبرنا قتيبة، قال حدثنا المفضل، - هو ابن فضالة - عن ابن جريج، عن أبي…٤٣٤٤دعوه فيوشك صاحبه أن يأتيه٤٣٤٥أخبرنا محمد بن وهب، قال حدثنا محمد بن سلمة، قال حدثني أبو عبد الرحيم،…٤٣٤٦أن أبا موسى، أتي بدجاجة فتنحى رجل من القوم فقال ما شأنك قال إني…٤٣٤٧أخبرنا علي بن حجر، قال حدثنا إسماعيل، عن أيوب، عن القاسم التميمي، عن…٤٣٤٨أخبرنا إسماعيل بن مسعود، عن بشر، - هو ابن المفضل - قال حدثنا سعيد، عن…٤٣٤٩ما من إنسان قتل عصفورا فما فوقها بغير حقها إلا سأله الله عز وجل عنها٤٣٥٠هو الطهور ماؤه الحلال ميتته٤٣٥١أخبرنا محمد بن آدم، قال حدثنا عبدة، عن هشام، عن وهب بن كيسان، عن جابر…٤٣٥٢هل معكم منه شىء٤٣٥٣إن كان بقي معكم شىء فابعثوا به إلينا٤٣٥٤أخبرنا محمد بن عمر بن علي بن مقدم المقدمي، قال حدثنا معاذ بن هشام،…٤٣٥٥أن طبيبا، ذكر ضفدعا في دواء عند رسول الله صلى الله عليه وسلم فنهى…٤٣٥٦سمع عبد الله بن أبي أوفى، قال غزونا مع رسول الله صلى الله عليه وسلم…٤٣٥٧فقال غزوت مع رسول الله صلى الله عليه وسلم ست غزوات نأكل الجراد .٤٣٥٨أن نملة قرصت نبيا من الأنبياء فأمر بقرية النمل فأحرقت فأوحى الله عز…٤٣٥٩أخبرنا إسحاق بن إبراهيم، قال أنبأنا النضر، - وهو ابن شميل - قال…٤٣٦٠نحوه ولم يرفعه .
