سنن النسائي

كتاب الصلاة

٤٦ حديثًا
٤٤٨بينا أنا عند البيت بين النائم واليقظان إذ أقبل أحد الثلاثة بين…٤٤٩فرض الله عز وجل على أمتي خمسين صلاة فرجعت بذلك حتى أمر بموسى عليه…٤٥٠أتيت بدابة فوق الحمار ودون البغل خطوها عند منتهى طرفها فركبت ومعي…٤٥١قال لما أسري برسول الله صلى الله عليه وسلم انتهي به إلى سدرة المنتهى…٤٥٢أن عبد ربه بن سعيد، حدثه أن البناني حدثه عن أنس بن مالك، أن الصلوات،…٤٥٣قالت أول ما فرضت الصلاة ركعتين فأقرت صلاة السفر وأتمت صلاة الحضر .٤٥٤أخبرنا محمد بن هاشم البعلبكي، قال أنبأنا الوليد، قال أخبرني أبو عمرو…٤٥٥قالت فرضت الصلاة ركعتين ركعتين فأقرت صلاة السفر وزيد في صلاة الحضر .٤٥٦وعبد الرحمن، قالا حدثنا أبو عوانة، عن بكير بن الأخنس، عن مجاهد، عن…٤٥٧أخبرنا يوسف بن سعيد، قال حدثنا حجاج بن محمد، قال حدثنا محمد بن عبد…٤٥٨خمس صلوات في اليوم والليلة٤٥٩افترض الله على عباده صلوات خمسا٤٦٠ألا تبايعون رسول الله صلى الله عليه وسلم٤٦١خمس صلوات كتبهن الله على العباد من جاء بهن لم يضيع منهن شيئا استخفافا…٤٦٢أرأيتم لو أن نهرا بباب أحدكم يغتسل منه كل يوم خمس مرات هل يبقى من…٤٦٣إن العهد الذي بيننا وبينهم الصلاة فمن تركها فقد كفر٤٦٤ليس بين العبد وبين الكفر إلا ترك الصلاة٤٦٥إن أول ما يحاسب به العبد بصلاته فإن صلحت فقد أفلح وأنجح وإن فسدت فقد…٤٦٦إن أول ما يحاسب به العبد يوم القيامة صلاته فإن وجدت تامة كتبت تامة…٤٦٧أول ما يحاسب به العبد صلاته فإن كان أكملها وإلا قال الله عز وجل…٤٦٨تعبد الله ولا تشرك به شيئا وتقيم الصلاة وتؤتي الزكاة وتصل الرحم ذرها٤٦٩وإبراهيم بن ميسرة، سمعا أنسا، قال صليت مع النبي صلى الله عليه وسلم…٤٧٠ومحمد بن بشار، قالا حدثنا محمد بن جعفر، قال حدثنا شعبة، عن الحكم بن…٤٧١لن يلج النار من صلى قبل طلوع الشمس وقبل غروبها٤٧٢أخبرنا قتيبة، عن مالك، عن زيد بن أسلم، عن القعقاع بن حكيم، عن أبي…٤٧٣شغلونا عن الصلاة الوسطى حتى غربت الشمس٤٧٤من ترك صلاة العصر فقد حبط عمله٤٧٥أخبرنا يعقوب بن إبراهيم، قال حدثنا هشيم، قال أنبأنا منصور بن زاذان،…٤٧٦قال كان رسول الله صلى الله عليه وسلم يقوم في الظهر فيقرأ قدر ثلاثين…٤٧٧أن النبي صلى الله عليه وسلم صلى الظهر بالمدينة أربعا وصلى العصر بذي…٤٧٨من فاتته صلاة العصر فكأنما وتر أهله وماله٤٧٩من الصلاة صلاة من فاتته فكأنما وتر أهله وماله٤٨٠يقول صلاة من فاتته فكأنما وتر أهله وماله . قال ابن عمر قال رسول الله…٤٨١قال رأيت سعيد بن جبير بجمع أقام فصلى المغرب ثلاث ركعات ثم أقام فصلى -…٤٨٢إنه ليس أحد يصلي هذه الصلاة غيركم٤٨٣أخبرنا عمرو بن يزيد، قال حدثنا بهز بن أسد، قال حدثنا شعبة، قال أخبرني…٤٨٤قال رأيت عبد الله بن عمر صلى بجمع فأقام فصلى المغرب ثلاثا ثم صلى…٤٨٥يتعاقبون فيكم ملائكة بالليل وملائكة بالنهار ويجتمعون في صلاة الفجر…٤٨٦تفضل صلاة الجمع على صلاة أحدكم وحده بخمسة وعشرين جزءا ويجتمع ملائكة…٤٨٧لا يلج النار أحد صلى قبل طلوع الشمس وقبل أن تغرب٤٨٨أخبرنا محمد بن بشار، قال حدثنا يحيى بن سعيد، حدثنا سفيان، قال حدثنا…٤٨٩قال قدم رسول الله صلى الله عليه وسلم المدينة فصلى نحو بيت المقدس ستة…٤٩٠قال كان رسول الله صلى الله عليه وسلم يسبح على الراحلة قبل أى وجه…٤٩١ومحمد بن المثنى، عن يحيى، عن عبد الملك، قال حدثنا سعيد بن جبير، عن…٤٩٢قال كان رسول الله صلى الله عليه وسلم يصلي على راحلته في السفر حيثما…٤٩٣قال بينما الناس بقباء في صلاة الصبح جاءهم آت فقال إن رسول الله صلى…
