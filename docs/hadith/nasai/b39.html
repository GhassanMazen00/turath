سنن النسائي

كتاب البيعة

٦٣ حديثًا
٤١٤٩أخبرنا الإمام أبو عبد الرحمن النسائي، من لفظه قال أنبأنا قتيبة بن…٤١٥٠أن عبادة بن الصامت، قال بايعنا رسول الله صلى الله عليه وسلم على السمع…٤١٥١والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٤١٥٢ويحيى بن سعيد، عن عبادة بن الوليد بن عبادة بن الصامت، عن أبيه، عن…٤١٥٣أن أباه الوليد، حدثه عن جده، عبادة بن الصامت قال بايعنا رسول الله صلى…٤١٥٤ويحيى بن سعيد، أنهما سمعا عبادة بن الوليد، يحدث عن أبيه، - أما سيار…٤١٥٥عليك بالطاعة في منشطك ومكرهك وعسرك ويسرك وأثرة عليك٤١٥٦قال بايعت رسول الله صلى الله عليه وسلم على النصح لكل مسلم .٤١٥٧قال جرير بايعت النبي صلى الله عليه وسلم على السمع والطاعة وأن أنصح…٤١٥٨سمع جابرا، يقول لم نبايع رسول الله صلى الله عليه وسلم على الموت إنما…٤١٥٩قال قلت لسلمة بن الأكوع على أى شىء بايعتم النبي صلى الله عليه وسلم…٤١٦٠أبايعه على الجهاد وقد انقطعت الهجرة٤١٦١تبايعوني على أن لا تشركوا بالله شيئا ولا تسرقوا ولا تزنوا ولا تقتلوا…٤١٦٢ألا تبايعوني على ما بايع عليه النساء أن لا تشركوا بالله شيئا ولا…٤١٦٣ارجع إليهما فأضحكهما كما أبكيتهما٤١٦٤ويحك إن شأن الهجرة شديد فهل لك من إبل٤١٦٥أن تهجر ما كره ربك عز وجل٤١٦٦قال قال ابن عباس إن رسول الله صلى الله عليه وسلم وأبا بكر وعمر كانوا…٤١٦٧عليك بالهجرة فإنه لا مثل لها٤١٦٨أبايعه على الجهاد وقد انقطعت الهجرة٤١٦٩لا هجرة بعد فتح مكة ولكن جهاد ونية فإذا استنفرتم فانفروا٤١٧٠لا هجرة ولكن جهاد ونية فإذا استنفرتم فانفروا٤١٧١يقول لا هجرة بعد وفاة رسول الله صلى الله عليه وسلم .٤١٧٢لا تنقطع الهجرة ما قوتل الكفار٤١٧٣أخبرنا محمود بن خالد، قال حدثنا مروان بن محمد، قال حدثنا عبد الله بن…٤١٧٤أوتستطيع ذلك يا جرير أوتطيق ذلك٤١٧٥قال بايعت رسول الله صلى الله عليه وسلم على إقام الصلاة وإيتاء الزكاة…٤١٧٦قال أتيت رسول الله صلى الله عليه وسلم فذكر نحوه .٤١٧٧أبايعك على أن تعبد الله وتقيم الصلاة وتؤتي الزكاة وتناصح المسلمين…٤١٧٨أبايعكم على أن لا تشركوا بالله شيئا ولا تسرقوا ولا تزنوا ولا تقتلوا…٤١٧٩قالت لما أردت أن أبايع، رسول الله صلى الله عليه وسلم قلت يا رسول الله…٤١٨٠أخبرنا الحسن بن أحمد، قال حدثنا أبو الربيع، قال أنبأنا حماد، قال…٤١٨١فيما استطعتن وأطقتن٤١٨٢ارجع فقد بايعتك٤١٨٣أخبرنا عبد الرحمن بن محمد بن سلام، قال حدثنا عمر بن يونس، عن عكرمة بن…٤١٨٤قال جاء عبد فبايع النبي صلى الله عليه وسلم على الهجرة ولا يشعر النبي…٤١٨٥إنما المدينة كالكير تنفي خبثها وتنصع طيبها٤١٨٦أنه دخل على الحجاج فقال يا ابن الأكوع ارتددت على عقبيك وذكر كلمة…٤١٨٧أخبرنا قتيبة، قال حدثنا سفيان، عن عبد الله بن دينار، ح وأخبرني علي بن…٤١٨٨أخبرنا الحسن بن محمد، قال حدثنا حجاج، عن ابن جريج، قال أخبرني موسى بن…٤١٨٩فيما استطعت والنصح لكل مسلم٤١٩٠فيما استطعتن وأطقتن٤١٩١إنه لم يكن نبي قبلي إلا كان حقا عليه أن يدل أمته على ما يعلمه خيرا…٤١٩٢ولو استعمل عليكم عبد حبشي يقودكم بكتاب الله فاسمعوا له وأطيعوا٤١٩٣من أطاعني فقد أطاع الله ومن عصاني فقد عصى الله ومن أطاع أميري فقد…٤١٩٤{ يا أيها الذين آمنوا أطيعوا الله وأطيعوا الرسول } قال نزلت في عبد…٤١٩٥الغزو غزوان فأما من ابتغى وجه الله وأطاع الإمام وأنفق الكريمة واجتنب…٤١٩٦إنما الإمام جنة يقاتل من ورائه ويتقى به فإن أمر بتقوى الله وعدل فإن…٤١٩٧إنما الدين النصيحة٤١٩٨إنما الدين النصيحة٤١٩٩إن الدين النصيحة إن الدين النصيحة إن الدين النصيحة٤٢٠٠أن رسول الله صلى الله عليه وسلم قال " الدين النصيحة " . قالوا لمن يا…٤٢٠١ما من وال إلا وله بطانتان بطانة تأمره بالمعروف وتنهاه عن المنكر…٤٢٠٢ما بعث الله من نبي ولا استخلف من خليفة إلا كانت له بطانتان بطانة…٤٢٠٣ما بعث من نبي ولا كان بعده من خليفة إلا وله بطانتان بطانة تأمره…٤٢٠٤من ولي منكم عملا فأراد الله به خيرا جعل له وزيرا صالحا إن نسي ذكره…٤٢٠٥لو دخلتموها لم تزالوا فيها إلى يوم القيامة٤٢٠٦على المرء المسلم السمع والطاعة فيما أحب وكره إلا أن يؤمر بمعصية فإذا…٤٢٠٧إنه ستكون بعدي أمراء من صدقهم بكذبهم وأعانهم على ظلمهم فليس مني ولست…٤٢٠٨اسمعوا هل سمعتم أنه ستكون بعدي أمراء من دخل عليهم فصدقهم بكذبهم…٤٢٠٩كلمة حق عند سلطان جائر٤٢١٠بايعوني على أن لا تشركوا بالله شيئا ولا تسرقوا ولا تزنوا٤٢١١إنكم ستحرصون على الإمارة وإنها ستكون ندامة وحسرة فنعمت المرضعة وبئست…
