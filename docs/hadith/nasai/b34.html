سنن النسائي

كتاب العمرى

٤٢ حديثًا
٣٧٢٠العمرى هي للوارث٣٧٢١أن رسول الله صلى الله عليه وسلم قال " العمرى للوارث " .٣٧٢٢أن النبي صلى الله عليه وسلم قضى بالعمرى للوارث .٣٧٢٢.٢أن النبي صلى الله عليه وسلم قضى بالعمرى للوارث .٣٧٢٣من أعمر شيئا فهو لمعمره محياه ومماته ولا ترقبوا فمن أرقب شيئا فهو…٣٧٢٤أخبرني زكريا بن يحيى، قال حدثنا زيد بن أخزم، قال أنبأنا معاذ بن هشام،…٣٧٢٥إن العمرى جائزة٣٧٢٦بتل رسول الله صلى الله عليه وسلم العمرى والرقبى .٣٧٢٧أن رسول الله صلى الله عليه وسلم خطبهم فقال " العمرى جائزة " .٣٧٢٨أخبرنا أحمد بن سليمان، قال أنبأنا عبيد الله، عن إسرائيل، عن عبد…٣٧٢٩أخبرنا محمد بن المثنى، قال حدثنا محمد، قال حدثنا شعبة، قال سمعت…٣٧٣٠من أعطي شيئا حياته فهو له حياته وموته٣٧٣١لا ترقبوا ولا تعمروا فمن أرقب أو أعمر شيئا فهو لورثته٣٧٣٢لا عمرى ولا رقبى فمن أعمر شيئا أو أرقبه فهو له حياته ومماته٣٧٣٣لا عمرى ولا رقبى فمن أعمر شيئا أو أرقبه فهو له حياته ومماته٣٧٣٤من أرقب رقبى فهو له٣٧٣٥من أعمر شيئا فهو له حياته ومماته٣٧٣٦يا معشر الأنصار امسكوا عليكم - يعني أموالكم - لا تعمروها فإنه من أعمر…٣٧٣٧امسكوا عليكم أموالكم ولا تعمروها فمن أعمر شيئا حياته فهو له حياته…٣٧٣٨الرقبى لمن أرقبها٣٧٣٩العمرى جائزة لأهلها والرقبى جائزة لأهلها٣٧٤٠من أعمر عمرى فهي له ولعقبه يرثها من يرثه من عقبه٣٧٤١العمرى لمن أعمرها هي له ولعقبه يرثها من يرثه من عقبه٣٧٤٢العمرى لمن أعمرها هي له ولعقبه يرثها من يرثه من عقبه٣٧٤٣أيما رجل أعمر رجلا عمرى له ولعقبه فهي له ولمن يرثه من عقبه موروثة٣٧٤٤من أعمر رجلا عمرى له ولعقبه فقد قطع قوله حقه وهي لمن أعمر ولعقبه٣٧٤٥أيما رجل أعمر عمرى له ولعقبه فإنها للذي يعطاها لا ترجع إلى الذي…٣٧٤٦أن جابرا، أخبره أن رسول الله صلى الله عليه وسلم قضى أنه من أعمر رجلا…٣٧٤٧أن رسول الله صلى الله عليه وسلم قضى فيمن أعمر عمرى له ولعقبه فهي له…٣٧٤٨أيما رجل أعمر رجلا عمرى له ولعقبه قال قد أعطيتكها وعقبك ما بقي منكم…٣٧٤٩أن رسول الله صلى الله عليه وسلم قضى بالعمرى أن يهب الرجل للرجل ولعقبه…٣٧٥٠العمرى لمن وهبت له٣٧٥١العمرى لمن وهبت له٣٧٥٢لا عمرى فمن أعمر شيئا فهو له٣٧٥٣من أعمر شيئا فهو له٣٧٥٤أخبرنا محمد بن المثنى، قال حدثنا محمد، قال حدثنا شعبة، عن قتادة، عن…٣٧٥٥فقلت حدث محمد بن سيرين، عن شريح، قال قضى نبي الله صلى الله عليه وسلم…٣٧٥٦لا يجوز لامرأة هبة في مالها إذا ملك زوجها عصمتها٣٧٥٧لا يجوز لامرأة عطية إلا بإذن زوجها٣٧٥٨أهدية أم صدقة فإن كانت هدية فإنما يبتغى بها وجه رسول الله صلى الله…٣٧٥٩لقد هممت أن لا أقبل هدية إلا من قرشي أو أنصاري أو ثقفي أو دوسي٣٧٦٠أخبرنا إسحاق بن إبراهيم، قال حدثنا وكيع، قال حدثنا شعبة، عن قتادة، عن…
