كتاب الزينة من السنن

٣٣٩ حديثًا•صفحة ١ من ٣
٥٠٤٠عشرة من الفطرة قص الشارب وقص الأظفار وغسل البراجم وإعفاء اللحية…٥٠٤١أخبرنا محمد بن عبد الأعلى، قال حدثنا المعتمر، عن أبيه، قال سمعت طلقا،…٥٠٤٢قال عشرة من السنة السواك وقص الشارب والمضمضة والاستنشاق وتوفير اللحية…٥٠٤٣خمس من الفطرة الختان وحلق العانة ونتف الضبع وتقليم الظفر وتقصير الشارب٥٠٤٤قال خمس من الفطرة تقليم الأظفار وقص الشارب ونتف الإبط وحلق العانة…٥٠٤٥أحفوا الشوارب وأعفوا اللحى٥٠٤٦أعفوا اللحى وأحفوا الشوارب٥٠٤٧من لم يأخذ شاربه فليس منا٥٠٤٨احلقوه كله أو اتركوه كله٥٠٤٩نهى رسول الله صلى الله عليه وسلم أن تحلق المرأة رأسها .٥٠٥٠نهاني الله عز وجل عن القزع٥٠٥١أخبرنا أحمد بن سليمان، قال حدثنا أبو داود، عن سفيان، عن عبيد الله بن…٥٠٥٢قال أتيت النبي صلى الله عليه وسلم ولي شعر فقال " ذباب " . فظننت أنه…٥٠٥٣قال كان شعر النبي صلى الله عليه وسلم شعرا رجلا ليس بالجعد ولا بالسبط…٥٠٥٤أخبرنا قتيبة، قال حدثنا أبو عوانة، عن داود الأودي، عن حميد بن عبد…٥٠٥٥أخبرنا علي بن حجر، قال حدثنا عيسى بن يونس، عن هشام بن حسان، عن الحسن،…٥٠٥٦أخبرنا محمد بن بشار، قال حدثنا أبو داود، قال حدثنا حماد بن سلمة، عن…٥٠٥٧ومحمد، قالا الترجل غب .٥٠٥٨أخبرنا إسماعيل بن مسعود، قال حدثنا خالد بن الحارث، عن كهمس، عن عبد…٥٠٥٩قالت كان رسول الله صلى الله عليه وسلم يحب التيامن يأخذ بيمينه ويعطي…٥٠٦٠قال ما رأيت أحدا أحسن في حلة حمراء من رسول الله صلى الله عليه وسلم…٥٠٦١قال كان شعر رسول الله صلى الله عليه وسلم إلى أنصاف أذنيه .٥٠٦٢قال ما رأيت رجلا أحسن في حلة من رسول الله صلى الله عليه وسلم . قال…٥٠٦٣أخبرنا الحسن بن إسماعيل بن سليمان، قال حدثنا عبدة بن سليمان، عن…٥٠٦٤أخبرني إبراهيم بن يعقوب، قال حدثنا سعيد بن سليمان، قال حدثنا أبو…٥٠٦٥أخبرنا إبراهيم بن المستمر العروقي، قال حدثنا الصلت بن محمد، قال حدثنا…٥٠٦٦قال أتيت النبي صلى الله عليه وسلم ولي جمة قال " ذباب " . وظننت أنه…٥٠٦٧يا رويفع لعل الحياة ستطول بك بعدي فأخبر الناس أنه من عقد لحيته أو…٥٠٦٨أخبرنا قتيبة، عن عبد العزيز، عن عمارة بن غزية، عن عمرو بن شعيب، عن…٥٠٦٩اليهود والنصارى لا تصبغ فخالفوهم٥٠٧٠أخبرنا إسحاق بن إبراهيم، قال أنبأنا عبد الرزاق، قال حدثنا معمر، عن…٥٠٧١إن اليهود والنصارى لا تصبغ فخالفوا عليهم فاصبغوا٥٠٧٢إن اليهود والنصارى لا تصبغ فخالفوهم٥٠٧٣غيروا الشيب ولا تشبهوا باليهود٥٠٧٤غيروا الشيب ولا تشبهوا باليهود٥٠٧٥قوم يخضبون بهذا السواد آخر الزمان كحواصل الحمام لا يريحون رائحة الجنة٥٠٧٦غيروا هذا بشىء واجتنبوا السواد٥٠٧٧أفضل ما غيرتم به الشمط الحناء والكتم٥٠٧٨إن أحسن ما غيرتم به الشيب الحناء والكتم٥٠٧٩إن من أحسن ما غيرتم به الشيب الحناء والكتم٥٠٨٠إن أحسن ما غيرتم به الشيب الحناء والكتم٥٠٨١إن أحسن ما غيرتم به الشيب الحناء والكتم٥٠٨٢إن أحسن ما غيرتم به الشيب الحناء والكتم٥٠٨٣صلى الله عليه وسلم وكان قد لطخ لحيته بالحناء .٥٠٨٤رضي الله عنه قال أتيت النبي صلى الله عليه وسلم ورأيته قد لطخ لحيته…٥٠٨٥قال رأيت ابن عمر يصفر لحيته بالخلوق فقلت يا أبا عبد الرحمن إنك تصفر…٥٠٨٦أنه سأله هل خضب رسول الله صلى الله عليه وسلم قال لم يبلغ ذلك إنما كان…٥٠٨٧أن رسول الله صلى الله عليه وسلم لم يكن يخضب إنما كان الشمط عند…٥٠٨٨أن نبي الله صلى الله عليه وسلم كان يكره عشر خصال الصفرة يعني الخلوق…٥٠٨٩إني لم أدر أيد امرأة هي أو رجل٥٠٩٠بالحناء قالت لا بأس به ولكن أكره هذا لأن حبي صلى الله عليه وسلم كان…٥٠٩١أخبرنا عبد الرحمن بن عبد الله بن عبد الحكم، قال حدثنا أبي وأبو الأسود…٥٠٩٢أن معاوية، قال إن رسول الله صلى الله عليه وسلم نهى عن الزور .٥٠٩٣أيما امرأة زادت في رأسها شعرا ليس منه فإنه زور تزيد فيه٥٠٩٤أخبرني محمد بن إسماعيل بن إبراهيم، قال حدثنا أبو النضر، قال حدثنا…٥٠٩٥أخبرنا إسحاق بن إبراهيم، قال حدثنا محمد بن بشر، قال حدثنا عبيد الله،…٥٠٩٦أخبرنا العباس بن عبد العظيم، قال حدثنا عبد الله بن محمد بن أسماء، قال…٥٠٩٧لعن الله الواصلة والمستوصلة٥٠٩٨أن امرأة، أتت عبد الله بن مسعود فقالت إني امرأة زعراء أيصلح أن أصل في…٥٠٩٩أخبرنا عبد الرحمن بن محمد بن سلام، قال حدثنا أبو داود الحفري، عن…٥١٠٠قال قال عبد الله المتفلجات . وساق الحديث .٥١٠١أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا أبان بن صمعة، عن…٥١٠٢قال آكل الربا وموكله وكاتبه إذا علموا ذلك والواشمة والموشومة للحسن…٥١٠٣ومغيرة، وابن، عون عن الشعبي، عن الحارث، عن علي، أن رسول الله صلى الله…٥١٠٤أخبرنا حميد بن مسعدة، قال حدثنا يزيد بن زريع، قال حدثنا ابن عون، عن…٥١٠٥حدثنا قتيبة، قال حدثنا خلف، - يعني ابن خليفة - عن عطاء بن السائب، عن…٥١٠٦لا تشمن ولا تستوشمن٥١٠٧أخبرنا أبو علي، محمد بن يحيى المروزي قال حدثنا عبد الله بن عثمان، عن…٥١٠٨أخبرنا محمد بن معمر، قال حدثنا يحيى بن حماد، قال حدثنا أبو عوانة، عن…٥١٠٩لعن الله المتنمصات والموتشمات والمتفلجات اللاتي يغيرن خلق الله عز وجل٥١١٠أخبرنا محمد بن حاتم، قال حدثنا حبان، قال حدثنا عبد الله، عن حيوة بن…٥١١١أخبرنا أحمد بن عمرو بن السرح، قال حدثنا ابن وهب، قال أخبرني الليث، عن…٥١١٢حدثنا قتيبة، قال حدثنا الليث، عن يزيد بن أبي حبيب، عن أبي الحصين…٥١١٣إن من خير أكحالكم الإثمد إنه يجلو البصر وينبت الشعر٥١١٤رسول الله صلى الله عليه وسلم قال كان إذا ادهن رأسه لم ير منه وإذا لم…٥١١٥أن ابن عمر، كان يصبغ ثيابه بالزعفران فقيل له فقال كان رسول الله صلى…٥١١٦قال سألت عائشة أكان رسول الله صلى الله عليه وسلم يتطيب قالت نعم…٥١١٧طيب الرجال ما ظهر ريحه وخفي لونه وطيب النساء ما ظهر لونه وخفي ريحه٥١١٨طيب الرجال ما ظهر ريحه وخفي لونه وطيب النساء ما ظهر لونه وخفي ريحه٥١١٩إن امرأة من بني إسرائيل اتخذت خاتما من ذهب وحشته مسكا٥١٢٠قال جاء رجل إلى النبي صلى الله عليه وسلم به ردع من خلوق فقال له النبي…٥١٢١وقال، على إثره يحدث عن يعلى بن مرة، أنه مر على النبي صلى الله عليه…٥١٢٢اذهب فاغسله ثم اغسله ولا تعد٥١٢٣.٥١٢٤يا يعلى لك امرأة٥١٢٥أى يعلى هل لك امرأة٥١٢٦أيما امرأة استعطرت فمرت على قوم ليجدوا من ريحها فهي زانية٥١٢٧إذا خرجت المرأة إلى المسجد فلتغتسل من الطيب كما تغتسل من الجنابة٥١٢٨أيما امرأة أصابت بخورا فلا تشهد معنا العشاء الآخرة٥١٢٩إذا شهدت إحداكن صلاة العشاء فلا تمس طيبا٥١٣٠إذا شهدت إحداكن العشاء فلا تمس طيبا٥١٣١أيتكن خرجت إلى المسجد فلا تقربن طيبا٥١٣٢امرأة عبد الله أن رسول الله صلى الله عليه وسلم أمرها أن لا تمس الطيب…٥١٣٣إذا خرجت المرأة إلى العشاء الآخرة فلا تمس طيبا٥١٣٤إذا شهدت إحداكن الصلاة فلا تمس طيبا٥١٣٥قال كان ابن عمر إذا استجمر استجمر بالألوة غير مطراة وبكافور يطرحه مع…٥١٣٦إن كنتم تحبون حلية الجنة وحريرها فلا تلبسوها في الدنيا٥١٣٧يا معشر النساء أما لكن في الفضة ما تحلين أما إنه ليس من امرأة تحلت…٥١٣٨يا معشر النساء أما لكن في الفضة ما تحلين أما إنه ليس منكن امرأة تحلى…٥١٣٩أيما امرأة تحلت - يعني - بقلادة من ذهب جعل في عنقها مثلها من النار…٥١٤٠يا فاطمة أيغرك أن يقول الناس ابنة رسول الله وفي يدها سلسلة من نار٥١٤١قال جاءت بنت هبيرة إلى رسول الله صلى الله عليه وسلم وفي يدها فتخ من…٥١٤٢قال كنت قاعدا عند النبي صلى الله عليه وسلم فأتته امرأة فقالت يا رسول…٥١٤٣ألا أخبرك بما هو أحسن من هذا لو نزعت هذا وجعلت مسكتين من ورق ثم…٥١٤٤إن هذين حرام على ذكور أمتي٥١٤٥إن هذين حرام على ذكور أمتي٥١٤٦إن هذين حرام على ذكور أمتي٥١٤٧هذا حرام على ذكور أمتي٥١٤٨أحل الذهب والحرير لإناث أمتي وحرم على ذكورها٥١٤٩أخبرنا الحسن بن قزعة، عن سفيان بن حبيب، عن خالد، عن أبي قلابة، عن…٥١٥٠أخبرنا محمد بن بشار، قال حدثنا عبد الوهاب، قال حدثنا خالد، عن ميمون،…٥١٥١أنه سمع معاوية، وعنده، جمع من أصحاب محمد صلى الله عليه وسلم قال…٥١٥٢أخبرنا أحمد بن حرب، قال أنبأنا أسباط، عن مغيرة، عن مطر، عن أبي شيخ،…٥١٥٣أن معاوية، عام حج جمع نفرا من أصحاب رسول الله صلى الله عليه وسلم في…٥١٥٤حمان أن معاوية، عام حج جمع نفرا من أصحاب رسول الله صلى الله عليه وسلم…٥١٥٥أخبرني شعيب بن شعيب بن إسحاق، قال حدثنا عبد الوهاب بن سعيد، قال حدثنا…٥١٥٦أخبرنا نصير بن الفرج، قال حدثنا عمارة بن بشر، عن الأوزاعي، عن يحيى بن…٥١٥٧وأخبرنا العباس بن الوليد بن مزيد، عن عقبة، عن الأوزاعي، حدثني يحيى،…٥١٥٨أخبرنا محمد بن عبد الله بن عبد الرحيم البرقي، قال حدثنا عبد الله بن…٥١٥٩وحوله، ناس من المهاجرين والأنصار فقال لهم أتعلمون أن رسول الله صلى…٥١٦٠أخبرني زياد بن أيوب، قال حدثنا علي بن غراب، قال حدثنا بيهس بن فهدان،…٥١٦١عرفجة بن أسعد أنه أصيب أنفه يوم الكلاب في الجاهلية فاتخذ أنفا من ورق…٥١٦٢رأى جده قال أصيب أنفه يوم الكلاب في الجاهلية قال فاتخذ أنفا من فضة…٥١٦٣قال قال عمر لصهيب ما لي أرى عليك خاتم الذهب قال قد رآه من هو خير منك…٥١٦٤إني كنت ألبس هذا الخاتم وإني لن ألبسه أبدا٥١٦٥أخبرنا قتيبة، قال حدثنا أبو الأحوص، عن أبي
