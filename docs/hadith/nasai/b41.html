سنن النسائي

كتاب الفرع والعتيرة

٤١ حديثًا
٤٢٢٢لا فرع ولا عتيرة٤٢٢٣لا فرع ولا عتيرة٤٢٢٤يا أيها الناس إن على أهل بيت في كل عام أضحاة وعتيرة٤٢٢٥حق فإن تركته حتى يكون بكرا فتحمل عليه في سبيل الله أو تعطيه أرملة خير…٤٢٢٦أنه سمع جده الحارث بن عمرو، يحدث أنه لقي رسول الله صلى الله عليه وسلم…٤٢٢٧أنه لقي رسول الله صلى الله عليه وسلم في حجة الوداع فقلت بأبي أنت يا…٤٢٢٨اذبحوا لله عز وجل في أى شهر ما كان وبروا الله عز وجل وأطعموا٤٢٢٩اذبحوا في أى شهر ما كان وبروا الله عز وجل وأطعموا٤٢٣٠إني كنت نهيتكم عن لحوم الأضاحي فوق ثلاث كيما تسعكم فقد جاء الله عز…٤٢٣١اذبحوها في أى شهر كان وبروا الله عز وجل وأطعموا٤٢٣٢اذبحوا لله عز وجل في أى شهر ما كان وبروا الله عز وجل وأطعموا٤٢٣٣أخبرنا عمرو بن علي، قال حدثنا عبد الرحمن، قال حدثنا أبو عوانة، عن…٤٢٣٤أن النبي صلى الله عليه وسلم مر على شاة ميتة ملقاة فقال " لمن هذه " .…٤٢٣٥هلا انتفعتم بجلدها٤٢٣٦لو نزعوا جلدها فانتفعوا به٤٢٣٧ألا دفعتم إهابها فاستمتعتم به٤٢٣٨ألا أخذتم إهابها فدبغتم فانتفعتم٤٢٣٩ألا انتفعتم بإهابها٤٢٤٠أخبرنا محمد بن عبد العزيز بن أبي رزمة، قال أنبأنا الفضل بن موسى، عن…٤٢٤١أيما إهاب دبغ فقد طهر٤٢٤٢أنه سمع أبا الخير، عن ابن وعلة، أنه سأل ابن عباس فقال إنا نغزو هذا…٤٢٤٣أليس قد دبغتها٤٢٤٤أخبرنا الحسين بن منصور بن جعفر النيسابوري، قال حدثنا الحسين بن محمد،…٤٢٤٥أخبرنا عبيد الله بن سعد بن إبراهيم بن سعد، قال حدثنا عمي، قال حدثنا…٤٢٤٦ذكاة الميتة دباغها٤٢٤٧ذكاة الميتة دباغها٤٢٤٨لو أخذتم إهابها٤٢٤٩أن لا تنتفعوا من الميتة بإهاب ولا عصب٤٢٥٠أن لا تستمتعوا من الميتة بإهاب ولا عصب٤٢٥١أن لا تنتفعوا من الميتة بإهاب ولا عصب٤٢٥٢أن رسول الله صلى الله عليه وسلم أمر أن يستمتع بجلود الميتة إذا دبغت .٤٢٥٣أخبرنا عبيد الله بن سعيد، عن يحيى، عن ابن أبي عروبة، عن قتادة، عن أبي…٤٢٥٤أخبرني عمرو بن عثمان، قال حدثنا بقية، عن بحير، عن خالد بن معدان، عن…٤٢٥٥أخبرنا عمرو بن عثمان، قال حدثنا بقية، عن بحير، عن خالد، قال وفد…٤٢٥٦إن الله عز وجل ورسوله حرم بيع الخمر والميتة والخنزير والأصنام٤٢٥٧قاتل الله اليهود حرمت عليهم الشحوم فجملوها٤٢٥٨ألقوها وما حولها وكلوه٤٢٥٩خذوها وما حولها فألقوه٤٢٦٠إن كان جامدا فألقوها وما حولها وإن كان مائعا فلا تقربوه٤٢٦١ما كان على أهل هذه الشاة لو انتفعوا بإهابها٤٢٦٢إذا وقع الذباب في إناء أحدكم فليمقله
