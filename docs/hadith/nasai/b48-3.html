سنن النسائي

كتاب الزينة من السنن

٣٣٩ حديثًا•صفحة ٣ من ٣
٥٣٤٠أخبرنا قتيبة، قال حدثنا الليث، عن ابن شهاب، عن عبيد الله بن عبد الله،…٥٣٤١أخبرنا الحسين بن حريث، قال أنبأنا سفيان، عن الزهري، عن عطاء بن يزيد،…٥٣٤٢حدثنا قتيبة، قال حدثنا الليث، عن أبي الزبير، عن جابر، أن رسول الله…٥٣٤٣قال رأيت على النبي صلى الله عليه وسلم عمامة حرقانية .٥٣٤٤أن رسول الله صلى الله عليه وسلم دخل يوم فتح مكة وعليه عمامة سوداء…٥٣٤٥قال دخل النبي صلى الله عليه وسلم يوم الفتح وعليه عمامة سوداء .٥٣٤٦قال كأني أنظر الساعة إلى رسول الله صلى الله عليه وسلم على المنبر…٥٣٤٧لا تدخل الملائكة بيتا فيه كلب ولا صورة٥٣٤٨لا تدخل الملائكة بيتا فيه كلب ولا صورة تماثيل٥٣٤٩إلا ما كان رقما في ثوب٥٣٥٠لا تدخل الملائكة بيتا فيه صورة٥٣٥١إن الملائكة لا تدخل بيتا فيه تصاوير٥٣٥٢قالت خرج رسول الله صلى الله عليه وسلم خرجة ثم دخل وقد علقت قراما فيه…٥٣٥٣يا عائشة حوليه فإني كلما دخلت فرأيته ذكرت الدنيا٥٣٥٤يا عائشة أخريه عني٥٣٥٥أن أباه، حدثه عن عائشة، أنها نصبت سترا فيه تصاوير فدخل رسول الله صلى…٥٣٥٦أشد الناس عذابا يوم القيامة الذين يضاهون بخلق الله٥٣٥٧إن أشد الناس عذابا يوم القيامة الذين يشبهون بخلق الله٥٣٥٨من صور صورة في الدنيا كلف يوم القيامة أن ينفخ فيها الروح وليس بنافخه٥٣٥٩من صور صورة عذب حتى ينفخ فيها الروح وليس بنافخ فيها٥٣٦٠من صور صورة كلف يوم القيامة أن ينفخ فيها الروح وليس بنافخ٥٣٦١إن أصحاب هذه الصور الذين يصنعونها يعذبون يوم القيامة يقال لهم أحيوا…٥٣٦٢إن أصحاب هذه الصور يعذبون يوم القيامة ويقال لهم أحيوا ما خلقتم٥٣٦٣زوج النبي صلى الله عليه وسلم أنها قالت إن أشد الناس عذابا يوم القيامة…٥٣٦٤إن من أشد الناس عذابا يوم القيامة المصورون٥٣٦٥أخبرنا هناد بن السري، عن أبي بكر، عن أبي إسحاق، عن مجاهد، عن أبي…٥٣٦٦ومعتمر بن سليمان، عن أشعث، عن محمد بن سيرين، عن عبد الله بن شقيق، عن…٥٣٦٧أن نعل، رسول الله صلى الله عليه وسلم كان لها قبالان .٥٣٦٨قال كان لنعل رسول الله صلى الله عليه وسلم قبالان .٥٣٦٩إذا انقطع شسع نعل أحدكم فلا يمش في نعل واحدة حتى يصلحها٥٣٧٠إذا انقطع شسع نعل أحدكم فلا يمش في الأخرى حتى يصلحها٥٣٧١ما هذا الذي تصنعين يا أم سليم٥٣٧٢إنه لعلك تدرك أموالا تقسم بين أقوام وإنما يكفيك من ذلك خادم ومركب في…٥٣٧٣قال كانت قبيعة سيف رسول الله صلى الله عليه وسلم من فضة .٥٣٧٤وجرير، قالا حدثنا قتادة، عن أنس، قال كان نعل سيف رسول الله صلى الله…٥٣٧٥قال كانت قبيعة سيف رسول الله صلى الله عليه وسلم من فضة .٥٣٧٦قل اللهم سددني واهدني٥٣٧٧أخبرنا يعقوب بن إبراهيم، عن عبد الرحمن، عن سليمان بن المغيرة، عن حميد…٥٣٧٨أخبرنا عبد الرحمن بن محمد بن سلام، قال حدثنا إسحاق الأزرق، قال حدثنا…
