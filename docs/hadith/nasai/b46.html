سنن النسائي

كتاب قطع السارق

١١٥ حديثًا
٤٨٧٠لا يزني الزاني حين يزني وهو مؤمن ولا يسرق السارق حين يسرق وهو مؤمن…٤٨٧١لا يزني الزاني حين يزني وهو مؤمن ولا يسرق حين يسرق وهو مؤمن ولا يشرب…٤٨٧٢قال لا يزني الزاني حين يزني وهو مؤمن ولا يسرق وهو مؤمن ولا يشرب الخمر…٤٨٧٣لعن الله السارق يسرق البيضة فتقطع يده ويسرق الحبل فتقطع يده٤٨٧٤أنه رفع إليه نفر من الكلاعيين أن حاكة سرقوا متاعا فحبسهم أياما ثم خلى…٤٨٧٥أن رسول الله صلى الله عليه وسلم حبس ناسا في تهمة .٤٨٧٦أن رسول الله صلى الله عليه وسلم حبس رجلا في تهمة ثم خلى سبيله .٤٨٧٧أن رسول الله صلى الله عليه وسلم أتي بلص اعترف اعترافا ولم يوجد معه…٤٨٧٨أبا وهب أفلا كان قبل أن تأتينا به٤٨٧٩فلولا كان هذا قبل أن تأتيني به يا أبا وهب٤٨٨٠أن رجلا، سرق ثوبا فأتي به رسول الله صلى الله عليه وسلم فأمر بقطعه…٤٨٨١أنه طاف بالبيت وصلى ثم لف رداء له من برد فوضعه تحت رأسه فنام فأتاه لص…٤٨٨٢هلا كان هذا قبل أن تأتينا به٤٨٨٣فهلا كان هذا قبل أن تأتيني به٤٨٨٤فهلا قبل أن تأتيني به تركته٤٨٨٥تعافوا الحدود قبل أن تأتوني به فما أتاني من حد فقد وجب٤٨٨٦تعافوا الحدود فيما بينكم فما بلغني من حد فقد وجب٤٨٨٧رضى الله عنهما أن امرأة، مخزومية كانت تستعير المتاع فتجحده فأمر النبي…٤٨٨٨رضى الله عنهما قال كانت امرأة مخزومية تستعير متاعا على ألسنة جاراتها…٤٨٨٩لتتب هذه المرأة إلى الله ورسوله وترد ما تأخذ على القوم٤٨٩٠لتتب هذه المرأة وتؤدي ما عندها٤٨٩١لو كانت فاطمة بنت محمد لقطعت يدها٤٨٩٢أن امرأة، من بني مخزوم استعارت حليا على لسان أناس فجحدتها فأمر بها…٤٨٩٣أن سعيد بن المسيب، حدثه نحوه، .٤٨٩٤لو كانت فاطمة لقطعت يدها٤٨٩٥يا أسامة إنما هلكت بنو إسرائيل حين كانوا إذا أصاب الشريف فيهم الحد…٤٨٩٦لو كانت فاطمة لقطعتها٤٨٩٧يا أسامة إن بني إسرائيل هلكوا بمثل هذا كان إذا سرق فيهم الشريف تركوه…٤٨٩٨أتشفع إلى في حد من حدود الله٤٨٩٩أتشفع في حد من حدود الله٤٩٠٠إن بني إسرائيل كانوا إذا سرق فيهم الشريف تركوه وإذا سرق الوضيع قطعوه…٤٩٠١إنما هلك الذين من قبلكم أنهم كانوا إذا سرق فيهم الشريف تركوه وإذا سرق…٤٩٠٢أتشفع في حد من حدود الله٤٩٠٣أتكلمني في حد من حدود الله٤٩٠٤حد يعمل في الأرض خير لأهل الأرض من أن يمطروا ثلاثين صباحا٤٩٠٥قال قال أبو هريرة إقامة حد بأرض خير لأهلها من مطر أربعين ليلة .٤٩٠٦يقول قطع رسول الله صلى الله عليه وسلم في مجن قيمته خمسة دراهم . كذا…٤٩٠٧أن نافعا، حدثهم أن عبد الله بن عمر قال قطع رسول الله صلى الله عليه…٤٩٠٨أن رسول الله صلى الله عليه وسلم قطع في مجن ثمنه ثلاثة دراهم .٤٩٠٩أن نافعا، حدثه أن عبد الله بن عمر حدثه أن النبي صلى الله عليه وسلم…٤٩١٠وإسماعيل بن أمية، وعبيد الله، وموسى بن عقبة، عن نافع، عن ابن عمر، أن…٤٩١١أن رسول الله صلى الله عليه وسلم قطع في مجن . قال أبو عبد الرحمن هذا…٤٩١٢قال قطع أبو بكر رضى الله عنه في مجن قيمته خمسة دراهم . هذا الصواب .٤٩١٣أخبرنا محمد بن المثنى، عن أبي داود، قال حدثنا شعبة، عن قتادة، قال…٤٩١٤رضي الله عنها قطع رسول الله صلى الله عليه وسلم في ربع دينار .٤٩١٥لا تقطع اليد إلا في ثمن المجن ثلث دينار أو نصف دينار فصاعدا٤٩١٦تقطع يد السارق في ربع دينار٤٩١٧تقطع يد السارق في ربع دينار فصاعدا٤٩١٨تقطع يد السارق في ربع دينار فصاعدا٤٩١٩تقطع يد السارق في ربع دينار فصاعدا٤٩٢٠قالت تقطع اليد في ربع دينار فصاعدا .٤٩٢١وقتيبة بن سعيد، عن سفيان، عن الزهري، عن عمرة، عن عائشة، قالت كان رسول…٤٩٢٢تقطع يد السارق في ربع دينار فصاعدا٤٩٢٣تقطع يد السارق في ربع دينار فصاعدا٤٩٢٤تقول يقطع في ربع دينار فصاعدا . قال أبو عبد الرحمن هذا الصواب من حديث…٤٩٢٥قالت القطع في ربع دينار فصاعدا .٤٩٢٦وعبد، ربه ورزيق صاحب أيلة أنهم سمعوا عمرة عن عائشة قالت القطع في ربع…٤٩٢٧قالت ما طال على ولا نسيت القطع في ربع دينار فصاعدا .٤٩٢٨لا يقطع السارق إلا في ربع دينار فصاعدا٤٩٢٩أخبرنا أحمد بن عمرو بن السرح، قال حدثنا ابن وهب، قال أخبرني عبد…٤٩٣٠قالت قالت عائشة القطع في ربع دينار فصاعدا .٤٩٣١يقطع يد السارق في ثمن المجن وثمن المجن ربع دينار٤٩٣٢أن محمد بن عبد الرحمن، حدثه عن عمرة، عن عائشة، قالت كان رسول الله صلى…٤٩٣٣لا تقطع اليد إلا في ربع دينار٤٩٣٤تقطع اليد في المجن٤٩٣٥لا تقطع يد السارق فيما دون المجن٤٩٣٦لا تقطع يد السارق إلا في ربع دينار فصاعدا٤٩٣٧لا تقطع اليد إلا في المجن أو ثمنه٤٩٣٨لا تقطع اليد إلا في المجن أو ثمنه٤٩٣٩لا تقطع اليد إلا في المجن أو ثمنه٤٩٤٠أخبرنا عمرو بن علي، قال حدثنا عبد الرحمن بن مهدي، قال حدثنا همام، عن…٤٩٤١قالت لم تقطع يد سارق في أدنى من حجفة أو ترس وكل واحد منهما ذو ثمن .٤٩٤٢أن النبي صلى الله عليه وسلم قطع في قيمة خمسة دراهم .٤٩٤٣قال لم يقطع النبي صلى الله عليه وسلم السارق إلا في ثمن المجن وثمن…٤٩٤٤قال لم تكن تقطع اليد على عهد رسول الله صلى الله عليه وسلم إلا في ثمن…٤٩٤٥قال لم تقطع اليد في زمن رسول الله صلى الله عليه وسلم إلا في ثمن المجن…٤٩٤٦وعطاء، عن أيمن، قال لم تقطع اليد في عهد رسول الله صلى الله عليه وسلم…٤٩٤٧ومجاهد، عن أيمن، قال يقطع السارق في ثمن المجن وكان ثمن المجن على عهد…٤٩٤٨لا تقطع اليد إلا في ثمن المجن٤٩٤٩ومجاهد، عن أيمن، قال لا يقطع السارق في أقل من ثمن المجن .٤٩٥٠أن عطاء بن أبي رباح، حدثه أن عبد الله بن عباس كان يقول ثمنه يومئذ…٤٩٥١مثله كان ثمن المجن على عهد رسول الله صلى الله عليه وسلم يقوم عشرة…٤٩٥٢مرسل .٤٩٥٣أخبرني حميد بن مسعدة، عن سفيان، - وهو ابن حبيب - عن العرزمي، - وهو…٤٩٥٤حدثنا سوار بن عبد الله بن سوار، قال حدثنا خالد بن الحارث، قال حدثنا…٤٩٥٥قال من توضأ فأحسن وضوءه ثم شهد صلاة العتمة في جماعة ثم صلى إليها…٤٩٥٦قال كان ثمن المجن على عهد رسول الله صلى الله عليه وسلم عشرة دراهم .٤٩٥٧لا تقطع اليد في ثمر معلق فإذا ضمه الجرين قطعت في ثمن المجن ولا تقطع…٤٩٥٨ما أصاب من ذي حاجة غير متخذ خبنة فلا شىء عليه ومن خرج بشىء منه فعليه…٤٩٥٩هي ومثلها والنكال وليس في شىء من الماشية قطع إلا فيما آواه المراح…٤٩٦٠لا قطع في ثمر ولا كثر٤٩٦١لا قطع في ثمر ولا كثر٤٩٦٢لا قطع في ثمر ولا كثر٤٩٦٣لا قطع في ثمر ولا كثر٤٩٦٤لا قطع في ثمر ولا كثر٤٩٦٥لا قطع في ثمر ولا كثر٤٩٦٦لا قطع في ثمر ولا كثر٤٩٦٧لا قطع في ثمر ولا كثر٤٩٦٨لا قطع في ثمر ولا كثر٤٩٦٩لا قطع في ثمر ولا كثر٤٩٧٠لا قطع في ثمر ولا كثر٤٩٧١ليس على خائن ولا منتهب ولا مختلس قطع٤٩٧٢ليس على خائن ولا منتهب ولا مختلس قطع٤٩٧٣ليس على المختلس قطع٤٩٧٤أخبرني إبراهيم بن الحسن، عن حجاج، قال قال ابن جريج قال أبو الزبير قال…٤٩٧٥ليس على مختلس ولا منتهب ولا خائن قطع٤٩٧٦قال ليس على خائن قطع . قال أبو عبد الرحمن أشعث بن سوار ضعيف .٤٩٧٧أن رسول الله صلى الله عليه وسلم أتي بلص فقال " اقتلوه " . فقالوا يا…٤٩٧٨أخبرنا محمد بن عبد الله بن عبيد بن عقيل، قال حدثنا جدي، قال حدثنا…٤٩٧٩لا تقطع الأيدي في السفر٤٩٨٠إذا سرق العبد فبعه ولو بنش٤٩٨١أنه أخبره قال كنت في سبى قريظة وكان ينظر فمن خرج شعرته قتل ومن لم…٤٩٨٢يد السارق في عنقه قال سنة قطع رسول الله صلى الله عليه وسلم يد سارق…٤٩٨٣قال قلت لفضالة بن عبيد أرأيت تعليق اليد في عنق السارق من السنة هو قال…٤٩٨٤لا يغرم صاحب سرقة إذا أقيم عليه الحد
