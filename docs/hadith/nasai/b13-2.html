سنن النسائي

كتاب السهو

١٨٨ حديثًا•صفحة ٢ من ٢
١٣٢٩أن النبي صلى الله عليه وسلم سلم ثم تكلم ثم سجد سجدتى السهو .١٣٣٠أن رسول الله صلى الله عليه وسلم سلم ثم سجد سجدتى السهو وهو جالس ثم…١٣٣١أن النبي صلى الله عليه وسلم صلى ثلاثا ثم سلم فقال الخرباق إنك صليت…١٣٣٢قال رمقت رسول الله صلى الله عليه وسلم في صلاته فوجدت قيامه وركعته…١٣٣٣قال ابن شهاب أخبرتني هند بنت الحارث الفراسية، أن أم سلمة، أخبرتها أن…١٣٣٤أنه صلى مع رسول الله صلى الله عليه وسلم صلاة الصبح فلما صلى انحرف .١٣٣٥قال إنما كنت أعلم انقضاء صلاة رسول الله صلى الله عليه وسلم بالتكبير .١٣٣٦قال أمرني رسول الله صلى الله عليه وسلم أن أقرأ المعوذات دبر كل صلاة .١٣٣٧اللهم أنت السلام ومنك السلام تباركت يا ذا الجلال والإكرام١٣٣٨اللهم أنت السلام ومنك السلام تباركت يا ذا الجلال والإكرام١٣٣٩لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…١٣٤٠قال كان عبد الله بن الزبير يهلل في دبر الصلاة يقول لا إله إلا الله…١٣٤١لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…١٣٤٢لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…١٣٤٣لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء قدير١٣٤٤إن تكلم بخير كان طابعا عليهن إلى يوم القيامة وإن تكلم بغير ذلك كان…١٣٤٥قالت حدثتني عائشة، - رضى الله عنها - قالت دخلت على امرأة من اليهود…١٣٤٦اللهم أصلح لي ديني الذي جعلته لي عصمة وأصلح لي دنياى التي جعلت فيها…١٣٤٧قال كان أبي يقول في دبر الصلاة اللهم إني أعوذ بك من الكفر والفقر…١٣٤٨خلتان لا يحصيهما رجل مسلم إلا دخل الجنة وهما يسير ومن يعمل بهما قليل١٣٤٩معقبات لا يخيب قائلهن يسبح الله في دبر كل صلاة ثلاثا وثلاثين ويحمده…١٣٥٠قال أمروا أن يسبحوا، دبر كل صلاة ثلاثا وثلاثين ويحمدوا ثلاثا وثلاثين…١٣٥١افعلوا كما قال الأنصاري١٣٥٢أن النبي صلى الله عليه وسلم مر عليها وهي في المسجد تدعو ثم مر بها…١٣٥٣إذا صليتم فقولوا سبحان الله ثلاثا وثلاثين والحمد لله ثلاثا وثلاثين…١٣٥٤من سبح في دبر صلاة الغداة مائة تسبيحة وهلل مائة تهليلة غفرت له ذنوبه…١٣٥٥والحسين بن محمد الذارع، - واللفظ له - قالا حدثنا عثام بن علي، قال…١٣٥٦إني كنت أجاور هذه العشر ثم بدا لي أن أجاور هذه العشر الأواخر فمن كان…١٣٥٧قال كان رسول الله صلى الله عليه وسلم إذا صلى الفجر قعد في مصلاه حتى…١٣٥٨وذكر، آخر عن سماك بن حرب، قال قلت لجابر بن سمرة كنت تجالس رسول الله…١٣٥٩أخبرنا قتيبة بن سعيد، قال حدثنا أبو عوانة، عن السدي، قال سألت أنس بن…١٣٦٠أخبرنا أبو حفص، عمرو بن علي قال حدثنا يحيى، قال حدثنا الأعمش، عن…١٣٦١أن مكحولا، حدثه أن مسروق بن الأجدع حدثه عن عائشة، قالت رأيت رسول الله…١٣٦٢قالت كان النساء يصلين مع رسول الله صلى الله عليه وسلم الفجر فكان إذا…١٣٦٣إني إمامكم فلا تبادروني بالركوع ولا بالسجود ولا بالقيام ولا بالانصراف…١٣٦٤إن الرجل إذا صلى مع الإمام حتى ينصرف حسب له قيام ليلة١٣٦٥إني ذكرت وأنا في العصر شيئا من تبر كان عندنا فكرهت أن يبيت عندنا…١٣٦٦فوالله ما صليتها
