سنن النسائي

كتاب النكاح

١٩٣ حديثًا•صفحة ١ من ٢
٣١٩٦أخبرنا أبو داود، سليمان بن سيف قال حدثنا جعفر بن عون، قال أنبأنا ابن…٣١٩٧قال توفي رسول الله صلى الله عليه وسلم وعنده تسع نسوة يصيبهن إلا سودة…٣١٩٨أن أنسا، حدثهم أن النبي صلى الله عليه وسلم كان يطوف على نسائه في…٣١٩٩قالت كنت أغار على اللاتي وهبن أنفسهن للنبي صلى الله عليه وسلم فأقول…٣٢٠٠اذهب فاطلب ولو خاتما من حديد٣٢٠١إني ذاكر لك أمرا فلا عليك أن لا تعجلي حتى تستأمري أبويك٣٢٠٢رضى الله عنها قالت قد خير رسول الله صلى الله عليه وسلم نساءه أوكان…٣٢٠٣أخبرنا عمرو بن علي، قال حدثنا عبد الرحمن، عن سفيان، عن إسماعيل، عن…٣٢٠٤قال قالت عائشة ما مات رسول الله صلى الله عليه وسلم حتى أحل له النساء .٣٢٠٥قالت ما توفي رسول الله صلى الله عليه وسلم حتى أحل الله له أن يتزوج من…٣٢٠٦من كان منكم ذا طول فليتزوج فإنه أغض للبصر وأحصن للفرج ومن لا فالصوم…٣٢٠٧من استطاع الباءة فليتزوج فإنه أغض للبصر وأحصن للفرج ومن لم يستطع…٣٢٠٨من استطاع منكم الباءة فليتزوج ومن لم يستطع فعليه بالصوم فإنه له وجاء٣٢٠٩يا معشر الشباب من استطاع منكم الباءة فلينكح فإنه أغض للبصر وأحصن…٣٢١٠يا معشر الشباب من استطاع منكم الباءة فليتزوج٣٢١١يا معشر الشباب من استطاع منكم الباءة فليتزوج٣٢١٢أخبرنا محمد بن عبيد، قال حدثنا عبد الله بن المبارك، عن معمر، عن…٣٢١٣أخبرنا إسماعيل بن مسعود، قال حدثنا خالد، عن أشعث، عن الحسن، عن سعد بن…٣٢١٤أخبرنا إسحاق بن إبراهيم، قال أنبأنا معاذ بن هشام، قال حدثني أبي، عن…٣٢١٥يا أبا هريرة جف القلم بما أنت لاق فاختص على ذلك أو دع٣٢١٦أخبرنا محمد بن عبد الله الخلنجي، قال حدثنا أبو سعيد، مولى بني هاشم…٣٢١٧ما بال أقوام يقولون كذا وكذا لكني أصلي وأنام وأصوم وأفطر وأتزوج…٣٢١٨ثلاثة حق على الله عز وجل عونهم المكاتب الذي يريد الأداء والناكح الذي…٣٢١٩أتزوجت يا جابر٣٢٢٠يا جابر هل أصبت امرأة بعدي٣٢٢١قال خطب أبو بكر وعمر رضى الله عنهما فاطمة فقال رسول الله صلى الله…٣٢٢٢انتقلي عند ابن أم مكتوم الأعمى الذي سماه الله عز وجل في كتابه٣٢٢٣أن أبا حذيفة بن عتبة بن ربيعة بن عبد شمس، - وكان ممن شهد بدرا مع رسول…٣٢٢٤وابن عبد الله بن ربيعة، عن عائشة، زوج النبي صلى الله عليه وسلم وأم…٣٢٢٥إن أحساب أهل الدنيا الذي يذهبون إليه المال٣٢٢٦أتزوجت يا جابر٣٢٢٧تزوجوا الولود الودود فإني مكاثر بكم٣٢٢٨أن مرثد بن أبي مرثد الغنوي، - وكان رجلا شديدا - وكان يحمل الأسارى من…٣٢٢٩وغيره، عن هارون بن رئاب، عن عبد الله بن عبيد بن عمير، وعبد الكريم، عن…٣٢٣٠تنكح النساء لأربعة لمالها ولحسبها ولجمالها ولدينها فاظفر بذات الدين…٣٢٣١التي تسره إذا نظر وتطيعه إذا أمر ولا تخالفه في نفسها ومالها بما يكره٣٢٣٢إن الدنيا كلها متاع وخير متاع الدنيا المرأة الصالحة٣٢٣٣إن فيهم لغيرة شديدة٣٢٣٤قال خطب رجل امرأة من الأنصار فقال له رسول الله صلى الله عليه وسلم "…٣٢٣٥قال خطبت امرأة على عهد رسول الله صلى الله عليه وسلم فقال النبي صلى…٣٢٣٦قالت تزوجني رسول الله صلى الله عليه وسلم في شوال وأدخلت عليه في شوال…٣٢٣٧من أحبني فليحب أسامة٣٢٣٨لا يخطب أحدكم على خطبة بعض٣٢٣٩لا تناجشوا ولا يبع حاضر لباد ولا يبع الرجل على بيع أخيه ولا يخطب على…٣٢٤٠لا يخطب أحدكم على خطبة أخيه٣٢٤١لا يخطب أحدكم على خطبة أخيه حتى ينكح أو يترك٣٢٤٢لا يخطب أحدكم على خطبة أخيه٣٢٤٣يحدث أن عبد الله بن عمر، كان يقول نهى رسول الله صلى الله عليه وسلم أن…٣٢٤٤ليس لك سكنى ولا نفقة فاعتدي عند فلانة٣٢٤٥والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ لمحمد - عن ابن…٣٢٤٦ألا نظرت إليها فإن في أعين الأنصار شيئا٣٢٤٧انظر إليها فإن في أعين الأنصار شيئا٣٢٤٨أخبرنا إسحاق بن إبراهيم، قال أنبأنا عبد الرزاق، قال أنبأنا معمر، عن…٣٢٤٩يقول كنت عند أنس بن مالك وعنده ابنة له فقال جاءت امرأة إلى رسول الله…٣٢٥٠أن امرأة، عرضت نفسها على النبي صلى الله عليه وسلم فضحكت ابنة أنس…٣٢٥١أخبرنا سويد بن نصر، قال أنبأنا عبد الله، قال حدثنا سليمان بن المغيرة،…٣٢٥٢يقول كانت زينب بنت جحش تفخر على نساء النبي صلى الله عليه وسلم تقول إن…٣٢٥٣إذا هم أحدكم بالأمر فليركع ركعتين من غير الفريضة ثم يقول اللهم إني…٣٢٥٤ارجع إليها فقل لها أما قولك إني امرأة غيرى فسأدعو الله لك فيذهب غيرتك…٣٢٥٥أن رسول الله صلى الله عليه وسلم تزوجها وهي بنت ست وبنى بها وهي بنت…٣٢٥٦قالت تزوجني رسول الله صلى الله عليه وسلم لسبع سنين ودخل على لتسع سنين…٣٢٥٧قال قالت عائشة تزوجني رسول الله صلى الله عليه وسلم لتسع سنين وصحبته…٣٢٥٨وأحمد بن حرب، قالا حدثنا أبو معاوية، عن الأعمش، عن إبراهيم، عن…٣٢٥٩أنه سمع عبد الله بن عمر، يحدث أن عمر بن الخطاب، رضى الله عنه حدثنا…٣٢٦٠الأيم أحق بنفسها من وليها والبكر تستأذن في نفسها وإذنها صماتها٣٢٦١الأيم أحق بنفسها من وليها واليتيمة تستأمر وإذنها صماتها٣٢٦٢الأيم أولى بأمرها واليتيمة تستأمر في نفسها وإذنها صماتها٣٢٦٣ليس للولي مع الثيب أمر واليتيمة تستأمر فصمتها إقرارها٣٢٦٤الثيب أحق بنفسها والبكر يستأمرها أبوها وإذنها صماتها٣٢٦٥لا تنكح الثيب حتى تستأذن ولا تنكح البكر حتى تستأمر٣٢٦٦استأمروا النساء في أبضاعهن٣٢٦٧لا تنكح الأيم حتى تستأمر ولا تنكح البكر حتى تستأذن٣٢٦٨ومجمع، ابنى يزيد بن جارية الأنصاري عن خنساء بنت خذام، أن أباها، زوجها…٣٢٦٩أن فتاة، دخلت عليها فقالت إن أبي زوجني ابن أخيه ليرفع بي خسيسته وأنا…٣٢٧٠تستأمر اليتيمة في نفسها فإن سكتت فهو إذنها وإن أبت فلا جواز عليها٣٢٧١ويعلى بن حكيم، عن عكرمة، عن ابن عباس، قال تزوج رسول الله صلى الله…٣٢٧٢أن ابن عباس، أخبره أن النبي صلى الله عليه وسلم تزوج ميمونة وهو محرم .٣٢٧٣أن النبي صلى الله عليه وسلم نكح ميمونة وهو محرم جعلت أمرها إلى العباس…٣٢٧٤أن رسول الله صلى الله عليه وسلم تزوج ميمونة وهو محرم .٣٢٧٥لا ينكح المحرم ولا ينكح ولا يخطب٣٢٧٦لا ينكح المحرم ولا ينكح ولا يخطب٣٢٧٧أن الحمد لله نستعينه ونستغفره ونعوذ بالله من شرور أنفسنا من يهده الله…٣٢٧٨إن الحمد لله نحمده ونستعينه من يهده الله فلا مضل له ومن يضلل الله فلا…٣٢٧٩قال تشهد رجلان عند النبي صلى الله عليه وسلم فقال أحدهما من يطع الله…٣٢٨٠يقول إني لفي القوم عند النبي صلى الله عليه وسلم فقامت امرأة فقالت يا…٣٢٨١إن أحق الشروط أن يوفى به ما استحللتم به الفروج٣٢٨٢إن أحق الشروط أن يوفى به ما استحللتم به الفروج٣٢٨٣لعلك تريدين أن ترجعي إلى رفاعة لا حتى يذوق عسيلتك وتذوقي عسيلته٣٢٨٤أن زينب بنت أبي سلمة، - وأمها أم سلمة زوج النبي صلى الله عليه وسلم -…٣٢٨٥أن عروة بن الزبير، حدثه عن زينب بنت أبي سلمة، أن أم حبيبة، زوج النبي…٣٢٨٦أعلى أم سلمة لو أني لم أنكح أم سلمة ما حلت لي إن أباها أخي من الرضاعة٣٢٨٧أنها قالت يا رسول الله هل لك في أختي قال " فأصنع ماذا " . قالت تزوجها…٣٢٨٨لا يجمع بين المرأة وعمتها ولا بين المرأة وخالتها٣٢٨٩أنه سمع أبا هريرة، يقول نهى رسول الله صلى الله عليه وسلم أن يجمع بين…٣٢٩٠أن جعفر بن ربيعة، حدثه عن عراك بن مالك، وعبد الرحمن الأعرج، عن أبي…٣٢٩١أخبرنا قتيبة، قال حدثنا الليث، عن يزيد بن أبي حبيب، عن عراك بن مالك،…٣٢٩٢لا تنكح المرأة على عمتها ولا على خالتها٣٢٩٣قال نهى رسول الله صلى الله عليه وسلم أن تنكح المرأة على عمتها أو على…٣٢٩٤لا تنكح المرأة على عمتها ولا على خالتها٣٢٩٥لا تنكح المرأة على عمتها ولا على خالتها٣٢٩٦قال نهى رسول الله صلى الله عليه وسلم أن تنكح المرأة على عمتها والعمة…٣٢٩٧لا تنكح المرأة على عمتها ولا على خالتها٣٢٩٨يقول نهى رسول الله صلى الله عليه وسلم أن تنكح المرأة على عمتها…٣٢٩٩قال نهى رسول الله صلى الله عليه وسلم أن تنكح المرأة على عمتها أو على…٣٣٠٠ما حرمته الولادة حرمه الرضاع٣٣٠١لا تحتجبي منه فإنه يحرم من الرضاع ما يحرم من النسب٣٣٠٢يحرم من الرضاع ما يحرم من النسب٣٣٠٣يحرم من الرضاع ما يحرم من الولادة٣٣٠٤أخبرنا هناد بن السري، عن أبي معاوية، عن الأعمش، عن سعد بن عبيدة، عن…٣٣٠٥إنها ابنة أخي من الرضاعة٣٣٠٦إنها ابنة أخي من الرضاعة وإنه يحرم من الرضاع ما يحرم من النسب٣٣٠٧والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٣٣٠٨لا تحرم الإملاجة ولا الإملاجتان٣٣٠٩لا تحرم المصة والمصتان٣٣١٠لا تحرم المصة والمصتان٣٣١١لا تحرم الخطفة والخطفتان٣٣١٢انظرن ما إخوانكن - ومرة أخرى - انظرن من إخوانكن من الرضاعة فإن…٣٣١٣أن عائشة، أخبرتها أن رسول الله صلى الله عليه وسلم كان عندها وأنها…٣٣١٤أن عائشة، قالت جاء عمي أبو الجعد من الرضاعة فرددته - قال وقال هشام هو…٣٣١٥ائذني له فإنه عمك٣٣١٦ائذني له فإنه عمك٣٣١٧ائذني له فإنه عمك٣٣١٨ائذني له فإنه عمك٣٣١٩زوج النبي صلى الله عليه وسلم تقول جاءت سهلة بنت سهيل إلى رسول الله…٣٣٢٠قال سمعناه من عبد الرحمن، - وهو ابن القاسم - عن أبيه، عن عائشة، قالت…٣٣٢١وربيعة، عن القاسم، عن عائشة، قالت أمر النبي صلى الله عليه وسلم امرأة…٣٣٢٢أرضعيه تحرمي عليه بذلك٣٣٢٣أرضعيه تحرمي عليه٣٣٢٤ومالك، عن ابن شهاب، عن عروة، قال أبى سائر أزواج النبي صلى الله عليه…٣٣٢٥أن أمه، زينب بنت أبي سلمة أخبرته أن أمها أم سلمة زوج النبي صلى الله…٣٣٢٦لقد هممت أن أنهى عن الغيلة حتى ذكرت أن فارس والروم يصنعه٣٣٢٧وحميد بن مسعدة، قالا حدثنا يزيد بن زريع، قال حدثنا ابن عون، عن محمد…٣٣٢٨إن ما قد قدر في الرحم سيكون٣٣٢٩غرة عبد أو أمة٣٣٣٠وكيف بها وقد زعمت أنها قد أرضعتكما دعها عنك٣٣٣١قال لقيت خالي ومعه الراية فقلت أين تريد قال أرسلني رسول الله صلى الله…٣٣٣٢قال أصبت عمي ومعه راية فقلت أين تريد فقال بعثني رسول الله صلى الله…٣٣٣٣أن نبي الله صلى الله عليه وسلم بعث جيشا إلى أوطاس فلقوا عدوا فقاتلوهم…٣٣٣٤أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن عبيد الله، قال أخبرني…٣٣٣٥لا جلب ولا جنب ولا شغار في الإسلام ومن انتهب نهبة فليس منا٣٣٣٦لا جلب ولا جنب ولا شغار في الإسلام٣٣٣٧أخبرنا هارون بن عبد الله، قال حدثنا معن، قال حدثنا مالك، عن نافع، ح…٣٣٣٨وعبد الرحمن بن محمد بن سلام، قالا حدثنا إسحاق الأزرق، عن عبيد الله،…٣٣٣٩هل عندك من شىء٣٣٤٠قال تزوج أبو طلحة أم سليم فكان صداق ما بينهما الإسلام أسلمت أم سليم…٣٣٤١أخبرنا محمد بن النضر بن مساور، قال أنبأنا جعفر بن سليمان، عن ثابت، عن…٣٣٤٢وعبد العزيز، - يعني ابن صهيب - عن أنس بن مالك، ح وأنبأنا قتيبة، قال…٣٣٤٣أعتق رسول الله صلى الله عليه وسلم صفية وجعل عتقها مهرها . واللفظ…٣٣٤٤ثلاثة يؤتون أجرهم مرتين رجل كانت له أمة فأدبها فأحسن أدبها وعلمها…٣٣٤٥من أعتق جاريته ثم تزوجها فله أجران
