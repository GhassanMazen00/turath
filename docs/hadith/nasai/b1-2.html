سنن النسائي

كتاب الطهارة

٣٢٤ حديثًا•صفحة ٢ من ٣
١٥١من توضأ فأحسن الوضوء ثم صلى ركعتين يقبل عليهما بقلبه ووجهه وجبت له…١٥٢قال قال علي كنت رجلا مذاء وكانت ابنة النبي صلى الله عليه وسلم تحتي…١٥٣يغسل مذاكيره ويتوضأ وضوءه للصلاة١٥٤يكفي من ذلك الوضوء١٥٥يغسل مذاكيره ويتوضأ١٥٦إذا وجد أحدكم ذلك فلينضح فرجه ويتوضأ وضوءه للصلاة١٥٧قال استحييت أن أسأل النبي، صلى الله عليه وسلم عن المذى من أجل فاطمة…١٥٨أنه سمع زر بن حبيش، يحدث قال أتيت رجلا يدعى صفوان بن عسال فقعدت على…١٥٩وإسماعيل بن مسعود، قالا حدثنا يزيد بن زريع، قال حدثنا شعبة، عن عاصم،…١٦٠لا ينصرف حتى يجد ريحا أو يسمع صوتا١٦١إذا استيقظ أحدكم من منامه فلا يدخل يده في الإناء حتى يفرغ عليها ثلاث…١٦٢إذا نعس الرجل وهو في الصلاة فلينصرف لعله يدعو على نفسه وهو لا يدري١٦٣إذا مس أحدكم ذكره فليتوضأ١٦٤ويتوضأ من مس الذكر١٦٥وهل هو إلا مضغة منك أو بضعة منك١٦٦قالت إن كان رسول الله صلى الله عليه وسلم ليصلي وإني لمعترضة بين يديه…١٦٧قالت لقد رأيتموني معترضة بين يدى رسول الله صلى الله عليه وسلم ورسول…١٦٨قالت كنت أنام بين يدى رسول الله صلى الله عليه وسلم ورجلاى في قبلته…١٦٩أعوذ برضاك من سخطك وبمعافاتك من عقوبتك وأعوذ بك منك لا أحصي ثناء عليك…١٧٠أخبرنا محمد بن المثنى، عن يحيى بن سعيد، عن سفيان، قال أخبرني أبو روق،…١٧١توضئوا مما مست النار١٧٢توضئوا مما مست النار١٧٣أخبرنا الربيع بن سليمان، قال حدثنا إسحاق بن بكر، - وهو ابن مضر - قال…١٧٤توضئوا مما مست النار١٧٥توضئوا مما مست النار١٧٦توضئوا مما غيرت النار١٧٧توضئوا مما غيرت النار١٧٨توضئوا مما أنضجت النار١٧٩توضئوا مما مست النار١٨٠توضئوا مما مست النار١٨١توضئوا مما مست النار١٨٢أن رسول الله صلى الله عليه وسلم أكل كتفا فجاءه بلال فخرج إلى الصلاة…١٨٣أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا ابن جريج، عن…١٨٤قال شهدت رسول الله صلى الله عليه وسلم أكل خبزا ولحما ثم قام إلى…١٨٥قال كان آخر الأمرين من رسول الله صلى الله عليه وسلم ترك الوضوء مما…١٨٦والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن القاسم،…١٨٧أخبرنا قتيبة، قال حدثنا الليث، عن عقيل، عن الزهري، عن عبيد الله بن…١٨٨أنه أسلم فأمره النبي صلى الله عليه وسلم أن يغتسل بماء وسدر .١٨٩أنه سمع أبا هريرة، يقول إن ثمامة بن أثال الحنفي انطلق إلى نجل قريب من…١٩٠رضى الله عنه أنه أتى النبي صلى الله عليه وسلم فقال إن أبا طالب مات .…١٩١إذا جلس بين شعبها الأربع ثم اجتهد فقد وجب الغسل١٩٢إذا قعد بين شعبها الأربع ثم اجتهد فقد وجب الغسل١٩٣إذا رأيت المذى فاغسل ذكرك وتوضأ وضوءك للصلاة وإذا فضخت الماء فاغتسل١٩٤إذا رأيت المذى فتوضأ واغسل ذكرك وإذا رأيت فضخ الماء فاغتسل١٩٥إذا أنزلت الماء فلتغتسل١٩٦أن عائشة، أخبرته أن أم سليم كلمت رسول الله صلى الله عليه وسلم وعائشة…١٩٧نعم إذا رأت الماء١٩٨إذا رأت الماء فلتغتسل١٩٩أخبرنا عبد الجبار بن العلاء، عن سفيان، عن عمرو، عن عبد الرحمن بن…٢٠٠ماء الرجل غليظ أبيض وماء المرأة رقيق أصفر فأيهما سبق كان الشبه٢٠١إنما ذلك عرق فإذا أقبلت الحيضة فدعي الصلاة وإذا أدبرت فاغسلي عنك الدم…٢٠٢إذا أقبلت الحيضة فاتركي الصلاة وإذا أدبرت فاغتسلي٢٠٣إن هذه ليست بالحيضة ولكن هذا عرق فاغتسلي ثم صلي٢٠٤إن هذه ليست بالحيضة ولكن هذا عرق فإذا أدبرت الحيضة فاغتسلي وصلي وإذا…٢٠٥إن هذه ليست بالحيضة ولكن هذا عرق فاغتسلي وصلي٢٠٦إنما ذلك عرق فاغتسلي وصلي٢٠٧امكثي قدر ما كانت تحبسك حيضتك ثم اغتسلي٢٠٨لتنظر عدد الليالي والأيام التي كانت تحيض من الشهر قبل أن يصيبها الذي…٢٠٩إنها ليست بالحيضة ولكنها ركضة من الرحم فلتنظر قدر قرئها التي كانت…٢١٠ليست بالحيضة إنما هو عرق٢١١إنما ذلك عرق فانظري إذا أتاك قرؤك فلا تصلي فإذا مر قرؤك فتطهري ثم صلي…٢١٢لا إنما ذلك عرق وليس بالحيضة فإذا أقبلت الحيضة فدعي الصلاة وإذا أدبرت…٢١٣رضى الله عنها أن امرأة، مستحاضة على عهد رسول الله صلى الله عليه وسلم…٢١٤مرها أن تغتسل وتهل٢١٥إذا كان دم الحيض - فإنه دم أسود يعرف - فأمسكي عن الصلاة فإذا كان…٢١٦إن دم الحيض دم أسود يعرف فإذا كان ذلك فأمسكي عن الصلاة وإذا كان الآخر…٢١٧إنما ذلك عرق وليست بالحيضة فإذا أقبلت الحيضة فدعي الصلاة وإذا أدبرت…٢١٨إنما ذلك عرق وليست بالحيضة فإذا أقبلت الحيضة فدعي الصلاة فإذا ذهب…٢١٩أن بنت أبي حبيش، قالت يا رسول الله إني لا أطهر أفأترك الصلاة قال " لا…٢٢٠لا يغتسل أحدكم في الماء الدائم وهو جنب٢٢١لا يبولن أحدكم في الماء الراكد ثم يغتسل منه٢٢٢أنه سأل عائشة - رضى الله عنها - أى الليل كان يغتسل رسول الله صلى الله…٢٢٣قال دخلت على عائشة رضى الله عنها فسألتها قلت أكان رسول الله صلى الله…٢٢٤قال كنت أخدم رسول الله صلى الله عليه وسلم فكان إذا أراد أن يغتسل قال…٢٢٥رضى الله عنها أنها ذهبت إلى النبي صلى الله عليه وسلم يوم الفتح فوجدته…٢٢٦قال أتي مجاهد بقدح حزرته ثمانية أرطال فقال حدثتني عائشة رضى الله عنها…٢٢٧أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا شعبة، عن أبي بكر…٢٢٨أخبرنا قتيبة بن سعيد، قال حدثنا الليث، عن ابن شهاب، عن عروة، عن…٢٢٩يقول كان رسول الله صلى الله عليه وسلم يتوضأ بمكوك ويغتسل بخمسة مكاكي .٢٣٠أخبرنا قتيبة بن سعيد، قال حدثنا أبو الأحوص، عن أبي إسحاق، عن أبي…٢٣١وابن، جريج عن الزهري، عن عروة، عن عائشة، رضى الله عنها قالت كنت أغتسل…٢٣٢أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن هشام بن عروة، ح وأنبأنا…٢٣٣صلى الله عليه وسلم من إناء واحد من الجنابة .٢٣٤أخبرنا قتيبة بن سعيد، حدثنا عبيدة بن حميد، عن منصور، عن إبراهيم، عن…٢٣٥أخبرنا عمرو بن علي، قال حدثنا يحيى، قال حدثنا سفيان، قال حدثني منصور،…٢٣٦قال أخبرتني خالتي، ميمونة أنها كانت تغتسل ورسول الله صلى الله عليه…٢٣٧أخبرنا سويد بن نصر، قال حدثنا عبد الله، عن سعيد بن يزيد، قال سمعت عبد…٢٣٨أخبرنا قتيبة، قال حدثنا أبو عوانة، عن داود الأودي، عن حميد بن عبد…٢٣٩أخبرنا محمد بن بشار، عن محمد، قال حدثنا شعبة، عن عاصم، ح وأخبرنا سويد…٢٤٠أن رسول الله صلى الله عليه وسلم اغتسل هو وميمونة من إناء واحد في قصعة…٢٤١إنما يكفيك أن تحثي على رأسك ثلاث حثيات من ماء ثم تفيضين على جسدك٢٤٢انقضي رأسك وامتشطي وأهلي بالحج ودعي العمرة٢٤٣قال حدثتني عائشة، رضى الله عنها أن رسول الله صلى الله عليه وسلم كان…٢٤٤أخبرنا أحمد بن سليمان، قال حدثنا يزيد، قال حدثنا شعبة، عن عطاء بن…٢٤٥أخبرنا محمود بن غيلان، أنبأنا النضر، قال أنبأنا شعبة، قال أنبأنا عطاء…٢٤٦قال وصفت عائشة غسل النبي صلى الله عليه وسلم من الجنابة قالت كان يغسل…٢٤٧رضى الله عنها أن النبي صلى الله عليه وسلم كان إذا اغتسل من الجنابة…٢٤٨حدثتني عائشة، - رضى الله عنها - عن غسل النبي، صلى الله عليه وسلم من…٢٤٩رضى الله عنها أن رسول الله صلى الله عليه وسلم كان يشرب رأسه ثم يحثي…٢٥٠أما أنا فأفيض على رأسي ثلاث أكف٢٥١خذي فرصة من مسك فتطهري بها٢٥٢- رضى الله عنها - قالت كان رسول الله صلى الله عليه وسلم لا يتوضأ بعد…٢٥٣قال حدثتني خالتي، ميمونة قالت أدنيت لرسول الله صلى الله عليه وسلم…٢٥٤أن النبي صلى الله عليه وسلم اغتسل فأتي بمنديل فلم يمسه وجعل يقول…٢٥٥وعبد الرحمن، عن شعبة، عن الحكم، عن إبراهيم، عن الأسود، عن عائشة، -…٢٥٦رضى الله عنها أن رسول الله صلى الله عليه وسلم كان إذا أراد أن ينام…٢٥٧أن عائشة، - رضى الله عنها - قالت كان رسول الله صلى الله عليه وسلم إذا…٢٥٨- رضى الله عنها - قالت إن رسول الله صلى الله عليه وسلم كان إذا أراد…٢٥٩أن عمر، قال يا رسول الله أينام أحدنا وهو جنب قال " إذا توضأ " .٢٦٠توضأ واغسل ذكرك ثم نم٢٦١لا تدخل الملائكة بيتا فيه صورة ولا كلب ولا جنب٢٦٢إذا أراد أحدكم أن يعود توضأ٢٦٣ويعقوب بن إبراهيم، - واللفظ لإسحاق - قالا حدثنا إسماعيل بن إبراهيم،…٢٦٤أن رسول الله صلى الله عليه وسلم كان يطوف على نسائه في غسل واحد .٢٦٥أخبرنا علي بن حجر، قال أنبأنا إسماعيل بن إبراهيم، عن شعبة، عن عمرو بن…٢٦٦قال كان رسول الله صلى الله عليه وسلم يقرأ القرآن على كل حال ليس…٢٦٧إني رأيتك فحدت عني٢٦٨إن المسلم لا ينجس٢٦٩أين كنت يا أبا هريرة٢٧٠يا عائشة ناوليني الثوب٢٧١ناوليني الخمرة من المسجد٢٧٢بهذا الإسناد مثله .٢٧٣أن ميمونة، قالت كان رسول الله صلى الله عليه وسلم يضع رأسه في حجر…٢٧٤وعلي بن حجر، - واللفظ له - أنبأنا سفيان، عن منصور، عن أمه، عن عائشة،…٢٧٥أخبرنا عمرو بن علي، قال حدثنا يحيى، قال حدثنا سفيان، قال حدثني منصور،…٢٧٦وذكر، آخر عن أبي الأسود، عن عروة، عن عائشة، - رضى الله عنها - قالت…٢٧٧أخبرنا قتيبة بن سعيد، عن مالك، عن هشام بن عروة، عن أبيه، عن عائشة، -…٢٧٨- رضى الله عنها - مثل ذلك .٢٧٩أخبرنا قتيبة، قال حدثنا يزيد، - وهو ابن المقدام بن شريح بن هانئ - عن…٢٨٠أخبرنا أيوب بن محمد الوزان، قال حدثنا عبد الله بن جعفر، قال حدثنا…٢٨١أخبرنا محمد بن منصور، قال حدثنا سفيان، عن مسعر، عن المقدام بن شريح،…٢٨٢وسفيان، عن المقدام بن شريح، عن أبيه، عن عائشة، - رضى الله عنها - قالت…٢٨٣وإسحاق بن إبراهيم، قالا حدثنا معاذ بن هشام، - واللفظ له - قال حدثني…٢٨٤أخبرنا محمد بن المثنى، قال حدثنا يحيى بن سعيد، عن جابر بن صبح، قال…٢٨٥أخبرنا قتيبة، قال حدثنا أبو الأحوص، عن أبي إسحاق، عن عمرو بن شرحبيل،…٢٨٦أخبرنا إسحاق بن إبراهيم، قال أنبأنا جرير، عن منصور، عن إبراهيم، عن…٢٨٧والليث، عن ابن شهاب، عن حبيب، مولى عروة عن بدية، - وكان الليث يقول…٢٨٨أخبرنا إسحاق بن إبراهيم، قال حدثنا سليمان بن حرب، قال حدثنا حماد بن…٢٨٩أخبرنا عمرو بن علي، قال حدثنا يحيى، عن شعبة، عن الحكم، عن عبد الحميد،…٢٩٠أخبرنا إسحاق بن إبراهيم، قال أنبأنا سفيان، عن عبد الرحمن بن القاسم،…٢٩١اغتسلي واستثفري ثم أهلي٢٩٢حكيه بضلع واغسليه بماء وسدر٢٩٣حتيه ثم اقرصيه بالماء ثم انضحيه وصلي فيه٢٩٤أنه سأل أم حبيبة زوج النبي صلى الله عليه وسلم هل كان رسول الله صلى…٢٩٥قالت كنت أغسل الجنابة من ثوب رسول الله صلى الله عليه وسلم فيخرج إلى…٢٩٦قالت كنت أفرك الجنابة - وقالت مرة أخرى المني - من ثوب رسول الله صلى…٢٩٧أن عائشة، قالت لقد رأيتني وما أزيد على أن أفركه من ثوب رسول الله صلى…٢٩٨قالت كنت أفركه من ثوب النبي صلى الله عليه وسلم .٢٩٩قالت كنت أراه في ثوب رسول الله صلى الله عليه وسلم فأحكه .٣٠٠قالت لقد رأيتني أفرك الجنابة من ثوب رسول الله صلى الله عليه وسلم .
