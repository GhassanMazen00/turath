كتاب الزينة من السنن

٣٣٩ حديثًا•صفحة ٢ من ٣
٥١٩٠ما أرانا إلا قد أوجعناك وأغرمناك٥١٩١أن رجلا، ممن أدرك النبي صلى الله عليه وسلم لبس خاتما من ذهب نحوه .…٥١٩٢أن رسول الله صلى الله عليه وسلم رأى على رجل خاتما من ذهب نحوه .٥١٩٣أن النبي صلى الله عليه وسلم رأى في يد رجل خاتم ذهب فضرب إصبعه بقضيب…٥١٩٤أن رسول الله صلى الله عليه وسلم مرسل . قال أبو عبد الرحمن والمراسيل…٥١٩٥ما لي أرى عليك حلية أهل النار٥١٩٦أن النبي صلى الله عليه وسلم اتخذ خاتما من ورق فصه حبشي ونقش فيه محمد…٥١٩٧قال كان لرسول الله صلى الله عليه وسلم خاتم فضة يتختم به في يمينه فصه…٥١٩٨قال كان خاتم رسول الله صلى الله عليه وسلم من فضة وكان فصه منه .٥١٩٩أن النبي صلى الله عليه وسلم كان خاتمه من ورق فصه منه .٥٢٠٠قال كان خاتم النبي صلى الله عليه وسلم من فضة فصه منه .٥٢٠١قال أراد رسول الله صلى الله عليه وسلم أن يكتب إلى الروم فقالوا إنهم…٥٢٠٢أخبرنا أحمد بن عثمان أبو الجوزاء، قال حدثنا أبو داود، قال حدثنا قرة…٥٢٠٣أن النبي صلى الله عليه وسلم كان يلبس خاتمه في يمينه .٥٢٠٤أن النبي صلى الله عليه وسلم كان يتختم بيمينه .٥٢٠٥معيقيب أنه قال كان خاتم النبي صلى الله عليه وسلم حديدا ملويا عليه فضة…٥٢٠٦إنه كان في يدك جمرة من نار٥٢٠٧من أراد أن يصوغ عليه فليفعل ولا تنقشوا على نقشه٥٢٠٨إنا قد اتخذنا خاتما ونقشنا فيه نقشا فلا ينقش أحد على نقشه٥٢٠٩لا تستضيئوا بنار المشركين ولا تنقشوا على خواتيمكم عربيا٥٢١٠يا علي سل الله الهدى والسداد٥٢١١ومحمد بن بشار، قالا حدثنا عبد الرحمن، عن سفيان، عن عاصم بن كليب، عن…٥٢١٢قل اللهم اهدني وسددني٥٢١٣أن رسول الله صلى الله عليه وسلم كان إذا دخل الخلاء نزع خاتمه .٥٢١٤قال اتخذ رسول الله صلى الله عليه وسلم خاتما من ذهب وجعل فصه من قبل…٥٢١٥أن رسول الله صلى الله عليه وسلم اتخذ خاتما من ذهب وجعل فصه مما يلي…٥٢١٦لا ينبغي لأحد أن ينقش على نقش خاتمي هذا٥٢١٧أن رسول الله صلى الله عليه وسلم لبس خاتما من ذهب ثلاثة أيام فلما رآه…٥٢١٨أن رسول الله صلى الله عليه وسلم اتخذ خاتما من ذهب وكان فصه في باطن…٥٢١٩لا تصحب الملائكة ركبا معهم جلجل٥٢٢٠لا تصحب الملائكة رفقة فيها جلجل٥٢٢١لا تصحب الملائكة رفقة فيها جلجل٥٢٢٢لا تدخل الملائكة بيتا فيه جلجل ولا جرس ولا تصحب الملائكة رفقة فيها جرس٥٢٢٣قال كنت جالسا عند رسول الله صلى الله عليه وسلم فرآني رث الثياب فقال "…٥٢٢٤أنه أتى النبي صلى الله عليه وسلم في ثوب دون فقال له النبي صلى الله…٥٢٢٥خمس من الفطرة قص الشارب ونتف الإبط وتقليم الأظفار والاستحداد والختان٥٢٢٦أحفوا الشوارب وأعفوا اللحى٥٢٢٧لا تبكوا على أخي بعد اليوم٥٢٢٨أخبرنا أحمد بن عبدة، قال أنبأنا حماد، قال حدثنا عبيد الله، عن نافع،…٥٢٢٩أنه أخبره أنه، سمع ابن عمر، يقول سمعت رسول الله صلى الله عليه وسلم…٥٢٣٠أخبرنا إسحاق بن إبراهيم، قال أنبأنا محمد بن بشر، قال حدثنا عبيد الله،…٥٢٣١أخبرنا محمد بن بشار، قال حدثنا يحيى، قال حدثنا عبيد الله، قال أخبرني…٥٢٣٢قال كان رسول الله صلى الله عليه وسلم رجلا مربوعا عريض ما بين المنكبين…٥٢٣٣قال ما رأيت من ذي لمة أحسن في حلة من رسول الله صلى الله عليه وسلم وله…٥٢٣٤قال كان شعر النبي صلى الله عليه وسلم إلى نصف أذنيه .٥٢٣٥أن النبي صلى الله عليه وسلم كان يضرب شعره إلى منكبيه .٥٢٣٦أما يجد هذا ما يسكن به شعره٥٢٣٧قال كانت له جمة ضخمة فسأل النبي صلى الله عليه وسلم فأمره أن يحسن…٥٢٣٨أن رسول الله صلى الله عليه وسلم كان يسدل شعره وكان المشركون يفرقون…٥٢٣٩أن رجلا، من أصحاب النبي صلى الله عليه وسلم يقال له عبيد قال إن رسول…٥٢٤٠وذكر، أن رسول الله صلى الله عليه وسلم كان يحب التيامن ما استطاع في…٥٢٤١إن اليهود والنصارى لا يصبغون فخالفوهم٥٢٤٢قال أتي النبي صلى الله عليه وسلم بأبي قحافة ورأسه ولحيته كأنه ثغامة…٥٢٤٣قال رأيت ابن عمر يصفر لحيته فقلت له في ذلك فقال رأيت النبي صلى الله…٥٢٤٤قال كان النبي صلى الله عليه وسلم يلبس النعال السبتية ويصفر لحيته…٥٢٤٥إنما هلكت بنو إسرائيل حين اتخذ نساؤهم مثل هذا٥٢٤٦ومحمد بن بشار، عن محمد بن جعفر، قال حدثنا شعبة، عن عمرو بن مرة، عن…٥٢٤٧أخبرنا عمرو بن يحيى بن الحارث، قال حدثنا محبوب بن موسى، قال أنبأنا…٥٢٤٨أخبرنا محمد بن عبد الله بن عبد الرحيم، قال حدثنا أسد بن موسى، قال…٥٢٤٩أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن عبيد الله، عن نافع، عن…٥٢٥٠لعن الله الواصلة والمستوصلة٥٢٥١أخبرنا إسحاق بن إبراهيم، قال أنبأنا محمد بن بشر، قال حدثنا عبيد الله،…٥٢٥٢أخبرنا محمد بن بشار، قال حدثنا محمد، قال حدثنا شعبة، عن منصور، عن…٥٢٥٣أخبرنا أحمد بن سعيد، قال حدثنا وهب بن جرير، حدثنا أبي قال، سمعت…٥٢٥٤أخبرنا محمد بن يحيى بن محمد، قال حدثنا عمر بن حفص، قال حدثنا أبي، عن…٥٢٥٥أخبرنا محمد بن المثنى، قال حدثنا محمد بن جعفر، قال حدثنا شعبة، عن…٥٢٥٦قال نهى رسول الله صلى الله عليه وسلم أن يتزعفر الرجل .٥٢٥٧قال نهى رسول الله صلى الله عليه وسلم أن يزعفر الرجل جلده .٥٢٥٨قال كان النبي صلى الله عليه وسلم إذا أتي بطيب لم يرده .٥٢٥٩من عرض عليه طيب فلا يرده فإنه خفيف المحمل طيب الرائحة٥٢٦٠إذا شهدت إحداكن العشاء فلا تمس طيبا٥٢٦١إذا خرجت إلى العشاء فلا تمس طيبا٥٢٦٢أيتكن خرجت إلى المسجد فلا تقربن طيبا٥٢٦٣أيما امرأة أصابت بخورا فلا تشهد معنا العشاء الآخرة٥٢٦٤والمستمر، عن أبي نضرة، عن أبي سعيد، قال ذكر النبي صلى الله عليه وسلم…٥٢٦٥إن الله عز وجل أحل لإناث أمتي الحرير والذهب وحرمه على ذكورها٥٢٦٦وخاتم الذهب، وأن أقرأ وأنا راكع، .٥٢٦٧أخبرنا يعقوب بن إبراهيم، قال حدثنا يحيى، عن ابن عجلان، قال أخبرني…٥٢٦٨أن أباه، حدثه أنه، سمع عليا، يقول نهاني رسول الله صلى الله عليه وسلم…٥٢٦٩قال الحارث بن مسكين قراءة عليه وأنا أسمع، عن ابن القاسم، حدثني مالك،…٥٢٧٠أن نافعا، أخبره حدثني ابن حنين، أن عليا، حدثه قال نهاني رسول الله صلى…٥٢٧١أن محمد بن إبراهيم، حدثه عن ابن حنين، عن علي، قال نهاني رسول الله صلى…٥٢٧٢أن ابن حنين، حدثه أن عليا قال إن رسول الله صلى الله عليه وسلم نهى عن…٥٢٧٣أخبرنا محمد بن المثنى، قال حدثنا محمد، قال حدثنا شعبة، عن قتادة، قال…٥٢٧٤أخبرنا أحمد بن حفص بن عبد الله، حدثني أبي قال، حدثني إبراهيم بن…٥٢٧٥إني كنت ألبس هذا الخاتم وإني لن ألبسه أبدا٥٢٧٦قال كان نقش خاتم رسول الله صلى الله عليه وسلم محمد رسول الله .٥٢٧٧أن النبي صلى الله عليه وسلم اتخذ خاتما من ورق وفصه حبشي ونقشه محمد…٥٢٧٨قال أراد رسول الله صلى الله عليه وسلم أن يكتب إلى الروم فقالوا إنهم…٥٢٧٩أن رسول الله صلى الله عليه وسلم اتخذ خاتما من ورق وفصه حبشي .٥٢٨٠قال كان خاتم النبي صلى الله عليه وسلم من فضة وفصه منه .٥٢٨١قد اصطنعنا خاتما ونقشنا عليه نقشا فلا ينقش عليه أحد٥٢٨٢إنا قد اتخذنا خاتما ونقشنا عليه نقشا فلا ينقش عليه أحد٥٢٨٣أن النبي صلى الله عليه وسلم كان يتختم في يمينه .٥٢٨٤قال كأني أنظر إلى بياض خاتم النبي صلى الله عليه وسلم في إصبعه اليسرى .٥٢٨٥رسول الله صلى الله عليه وسلم قال كأني أنظر إلى وبيص خاتمه من فضة .…٥٢٨٦أخبرنا محمد بن بشار، قال حدثنا محمد، قال حدثنا شعبة، عن عاصم بن كليب،…٥٢٨٧قال نهاني رسول الله صلى الله عليه وسلم أن ألبس في إصبعي هذه وفي…٥٢٨٨لا ينبغي لأحد أن ينقش على نقش خاتمي هذا٥٢٨٩شغلني هذا عنكم منذ اليوم إليه نظرة وإليكم نظرة٥٢٩٠إني كنت ألبس هذا الخاتم وأجعل فصه من داخل٥٢٩١أنه رأى في يد رسول الله صلى الله عليه وسلم خاتما من ورق يوما واحدا…٥٢٩٢أن رسول الله صلى الله عليه وسلم اتخذ خاتما من ذهب وكان جعل فصه في…٥٢٩٣قال اتخذ رسول الله صلى الله عليه وسلم خاتما من ذهب وجعل فصه مما يلي…٥٢٩٤قال دخلت على رسول الله صلى الله عليه وسلم فرآني سيئ الهيئة فقال النبي…٥٢٩٥إنما يلبس هذه من لا خلاق له في الآخرة٥٢٩٦قال رأيت على زينب بنت النبي صلى الله عليه وسلم قميص حرير يحيى، - وهو ابن…٥٣٠١لا تشربوا في إناء الذهب والفضة ولا تلبسوا الديباج ولا الحرير فإنها…٥٣٠٢أتعجبون من هذه لمناديل سعد في الجنة أحسن مما ترون٥٣٠٣نهاني عنه جبريل عليه السلام٥٣٠٤من لبس الحرير في الدنيا فاتبعته تسأله…٥٣٠٩أخبرنا سليمان بن منصور، قال حدثنا أبو الأحوص، عن أشعث بن أبي الشعثاء،…٥٣١٠أن رسول الله صلى الله عليه وسلم أرخص لعبد الرحمن بن عوف والزبير بن…٥٣١١أن النبي صلى الله عليه وسلم رخص لعبد الرحمن والزبير في قمص حرير كانت…٥٣١٢لا يلبس الحرير إلا من ليس له منه شىء في الآخرة إلا هكذا٥٣١٣أنه لم يرخص في الديباج إلا موضع أربع أصابع .٥٣١٤قال رأيت النبي صلى الله عليه وسلم وعليه حلة حمراء مترجلا لم أر قبله…٥٣١٥قال كان أحب الثياب إلى نبي الله صلى الله عليه وسلم الحبرة .٥٣١٦هذه ثياب الكفار فلا تلبسها٥٣١٧اذهب فاطرحهما عنك٥٣١٨أن إبراهيم بن عبد الله بن حنين، أخبره أن أباه حدثه أنه، سمع
