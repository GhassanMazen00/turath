سنن النسائي

كتاب مناسك الحج

٤٦٧ حديثًا•صفحة ٢ من ٤
٢٧٦٩والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن وهب، قال أخبرني يونس،…٢٧٧٠أنه كان ينكر الاشتراط في الحج ويقول ما حسبكم سنة نبيكم صلى الله عليه…٢٧٧١أخبرنا محمد بن عبد الأعلى، قال حدثنا محمد بن ثور، عن معمر، عن الزهري،…٢٧٧٢أن رسول الله صلى الله عليه وسلم أشعر بدنه .٢٧٧٣أن رسول الله صلى الله عليه وسلم أشعر بدنه من الجانب الأيمن وسلت الدم…٢٧٧٤أن النبي صلى الله عليه وسلم لما كان بذي الحليفة أمر ببدنته فأشعر في…٢٧٧٥وعمرة بنت عبد الرحمن، عن عائشة، أنها قالت كان رسول الله صلى الله عليه…٢٧٧٦قالت كنت أفتل قلائد هدى رسول الله صلى الله عليه وسلم فيبعث بها ثم…٢٧٧٧قالت إن كنت لأفتل قلائد هدى رسول الله صلى الله عليه وسلم ثم يقيم ولا…٢٧٧٨قالت كنت أفتل القلائد لهدى رسول الله صلى الله عليه وسلم فيقلد هديه ثم…٢٧٧٩قالت لقد رأيتني أفتل قلائد الغنم لهدى رسول الله صلى الله عليه وسلم ثم…٢٧٨٠أخبرنا الحسن بن محمد الزعفراني، قال حدثنا حسين، - يعني ابن حسن - عن…٢٧٨١إني لبدت رأسي وقلدت هديي فلا أحل حتى أنحر٢٧٨٢أن نبي الله صلى الله عليه وسلم لما أتى ذا الحليفة أشعر الهدى في جانب…٢٧٨٣قالت فتلت قلائد بدن رسول الله صلى الله عليه وسلم بيدى ثم قلدها…٢٧٨٤قالت فتلت قلائد بدن رسول الله صلى الله عليه وسلم ثم لم يحرم ولم يترك…٢٧٨٥قالت كنت أفتل قلائد هدى رسول الله صلى الله عليه وسلم غنما .٢٧٨٦أن رسول الله صلى الله عليه وسلم كان يهدي الغنم .٢٧٨٧أن رسول الله صلى الله عليه وسلم أهدى مرة غنما وقلدها .٢٧٨٨قالت كنت أفتل قلائد هدى رسول الله صلى الله عليه وسلم غنما ثم لا يحرم .٢٧٨٩قالت كنت أفتل قلائد هدى رسول الله صلى الله عليه وسلم غنما ثم لا يحرم .٢٧٩٠أخبرنا الحسين بن عيسى، - ثقة - قال حدثنا عبد الصمد بن عبد الوارث، قال…٢٧٩١أن رسول الله صلى الله عليه وسلم لما أتى ذا الحليفة أشعر الهدى من جانب…٢٧٩٢أنهم كانوا إذا كانوا حاضرين مع رسول الله صلى الله عليه وسلم بالمدينة…٢٧٩٣قالت كنت أفتل قلائد هدى رسول الله صلى الله عليه وسلم بيدى ثم يقلدها…٢٧٩٤وقتيبة، عن سفيان، عن الزهري، عن عروة، عن عائشة، قالت كنت أفتل قلائد…٢٧٩٥قال قالت عائشة كنت أفتل قلائد هدى رسول الله صلى الله عليه وسلم فلا…٢٧٩٦قالت إن كنت لأفتل قلائد هدى رسول الله صلى الله عليه وسلم ويخرج بالهدى…٢٧٩٧أخبرنا محمد بن قدامة، قال حدثنا جرير، عن منصور، عن إبراهيم، عن…٢٧٩٨سمعه يحدث، عن جابر، أنه سمعه يحدث، أن النبي صلى الله عليه وسلم ساق…٢٧٩٩أن رسول الله صلى الله عليه وسلم رأى رجلا يسوق بدنة قال " اركبها " .…٢٨٠٠أن رسول الله صلى الله عليه وسلم رأى رجلا يسوق بدنة . فقال " اركبها "…٢٨٠١أن النبي صلى الله عليه وسلم رأى رجلا يسوق بدنة وقد جهده المشى قال "…٢٨٠٢اركبها بالمعروف إذا ألجئت إليها حتى تجد ظهرا٢٨٠٣أوما كنت طفت ليالي قدمنا مكة٢٨٠٤أخبرنا عمرو بن علي، قال حدثنا يحيى، عن يحيى، عن عمرة، عن عائشة، قالت…٢٨٠٥أحلوا واجعلوها عمرة٢٨٠٦أخبرنا محمد بن بشار، قال حدثنا محمد، قال حدثنا شعبة، عن عبد الملك، عن…٢٨٠٧أخبرنا هناد بن السري، عن عبدة، عن ابن أبي عروبة، عن مالك بن دينار، عن…٢٨٠٨أخبرنا إسحاق بن إبراهيم، قال أنبأنا عبد العزيز، - وهو الدراوردي - عن…٢٨٠٩وعياش العامري، عن إبراهيم التيمي، عن أبيه، عن أبي ذر، في متعة الحج…٢٨١٠ومحمد بن بشار، قالا حدثنا محمد، قال حدثنا شعبة، قال سمعت عبد الوارث…٢٨١١أخبرنا بشر بن خالد، قال أنبأنا غندر، عن شعبة، عن سليمان، عن إبراهيم…٢٨١٢أخبرنا محمد بن عبد الله بن المبارك، قال حدثنا يحيى بن آدم، قال حدثنا…٢٨١٣قال كانوا يرون أن العمرة، في أشهر الحج من أفجر الفجور في الأرض…٢٨١٤يقول أهل رسول الله صلى الله عليه وسلم بالعمرة وأهل أصحابه بالحج وأمر…٢٨١٥هذه عمرة استمتعناها فمن لم يكن عنده هدى فليحل الحل كله فقد دخلت…٢٨١٦إنما هي طعمة أطعمكموها الله عز وجل٢٨١٧أخبرنا عمرو بن علي، قال حدثنا يحيى بن سعيد، قال حدثنا ابن جريج، قال…٢٨١٨دعوه فإنه يوشك أن يأتي صاحبه٢٨١٩أما إنه لم نرده عليك إلا أنا حرم٢٨٢٠إنا حرم لا نأكل الصيد٢٨٢١أن ابن عباس، قال لزيد بن أرقم ما علمت أن النبي صلى الله عليه وسلم…٢٨٢١.٢إنا لا نأكل إنا حرم٢٨٢٢قال أهدى الصعب بن جثامة إلى رسول الله صلى الله عليه وسلم رجل حمار وحش…٢٨٢٣وحبيب، - وهو ابن أبي ثابت - عن سعيد بن جبير، عن ابن عباس، أن الصعب بن…٢٨٢٤أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا هشام، عن يحيى بن…٢٨٢٥أن أباه، أخبره أنه، غزا مع رسول الله صلى الله عليه وسلم غزوة الحديبية…٢٨٢٦هل أشرتم أو أعنتم٢٨٢٧صيد البر لكم حلال ما لم تصيدوه أو يصاد لكم٢٨٢٨خمس ليس على المحرم في قتلهن جناح الغراب والحدأة والعقرب والفأرة…٢٨٢٩خمس يقتلهن المحرم الحية والفأرة والحدأة والغراب الأبقع والكلب العقور٢٨٣٠أن رسول الله صلى الله عليه وسلم أذن في قتل خمس من الدواب للمحرم…٢٨٣١أنه لم يكن شىء إلا يطفئ على إبراهيم عليه السلام إلا هذه الدابة٢٨٣٢خمس من الدواب لا جناح على من قتلهن - أو في قتلهن - وهو حرام الحدأة…٢٨٣٣خمس لا جناح على من قتلهن الحدأة والغراب والفأرة والعقرب والكلب العقور٢٨٣٤يقتل العقرب والفويسقة والحدأة والغراب والكلب العقور٢٨٣٥خمس من الدواب لا جناح في قتلهن على من قتلهن في الحرم والإحرام الفأرة…٢٨٣٦أخبرنا محمد بن منصور، قال حدثنا سفيان، قال حدثنا ابن جريج، عن عبد…٢٨٣٧قال تزوج النبي صلى الله عليه وسلم ميمونة وهو محرم .٢٨٣٨أن أبا الشعثاء، حدثه عن ابن عباس، أن رسول الله صلى الله عليه وسلم نكح…٢٨٣٩أن رسول الله صلى الله عليه وسلم تزوج ميمونة وهما محرمان .٢٨٤٠أن رسول الله صلى الله عليه وسلم تزوج ميمونة وهو محرم .٢٨٤١وصفوان بن عمرو الحمصي، قالا حدثنا أبو المغيرة، قال حدثنا الأوزاعي، عن…٢٨٤٢لا ينكح المحرم ولا يخطب ولا ينكح٢٨٤٣أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن مالك، أخبرني نافع، عن…٢٨٤٤لا ينكح المحرم ولا يخطب٢٨٤٥أن رسول الله صلى الله عليه وسلم احتجم وهو محرم .٢٨٤٦وعطاء، عن ابن عباس، أن النبي صلى الله عليه وسلم احتجم وهو محرم .٢٨٤٧أخبرنا محمد بن منصور، عن سفيان، قال أنبأنا عمرو بن دينار، قال سمعت…٢٨٤٨أن النبي صلى الله عليه وسلم احتجم وهو محرم من وثء كان به .٢٨٤٩أن رسول الله صلى الله عليه وسلم احتجم وهو محرم على ظهر القدم من وثء…٢٨٥٠قال قال علقمة بن أبي علقمة أنه سمع الأعرج، قال سمعت عبد الله ابن…٢٨٥١صم ثلاثة أيام أو أطعم ستة مساكين مدين مدين أو انسك شاة أى ذلك فعلت…٢٨٥٢انطلق فاحلقه وتصدق على ستة مساكين٢٨٥٣اغسلوه بماء وسدر وكفنوه في ثوبيه ولا تمسوه بطيب ولا تخمروا رأسه فإنه…٢٨٥٤اغسلوه بماء وسدر وكفنوه في ثوبين٢٨٥٥اغسلوه بماء وسدر وكفنوه في ثوبين ولا تحنطوه ولا تخمروا رأسه فإن الله…٢٨٥٦اغسلوه وكفنوه ولا تغطوا رأسه ولا تقربوه طيبا فإنه يبعث يهل٢٨٥٧يغسل ويكفن في ثوبين ولا يغطى رأسه ووجهه فإنه يقوم يوم القيامة ملبيا٢٨٥٨اغسلوه بماء وسدر وألبسوه ثوبيه ولا تخمروا رأسه فإنه يأتي يوم القيامة…٢٨٥٩أن عبد الله بن عبد الله، وسالم بن عبد الله، أخبراه أنهما، كلما عبد…٢٨٦٠من عرج أو كسر فقد حل وعليه حجة أخرى٢٨٦١من كسر أو عرج فقد حل وعليه حجة أخرى٢٨٦٢أن عبد الله بن عمر، حدثه أن رسول الله صلى الله عليه وسلم كان ينزل بذي…٢٨٦٣أخبرني عمران بن يزيد، عن شعيب، قال حدثنا ابن جريج، قال أخبرني مزاحم…٢٨٦٤أن النبي صلى الله عليه وسلم خرج من الجعرانة ليلا كأنه سبيكة فضة…٢٨٦٥أن رسول الله صلى الله عليه وسلم دخل مكة من الثنية العليا التي…٢٨٦٦رضى الله عنه أن النبي صلى الله عليه وسلم دخل مكة ولواؤه أبيض .٢٨٦٧أن النبي صلى الله عليه وسلم دخل مكة وعليه المغفر فقيل ابن خطل متعلق…٢٨٦٨أن النبي صلى الله عليه وسلم دخل مكة عام الفتح وعلى رأسه المغفر .٢٨٦٩أن النبي صلى الله عليه وسلم دخل يوم فتح مكة وعليه عمامة سوداء بغير…٢٨٧٠قال قدم رسول الله صلى الله عليه وسلم وأصحابه لصبح رابعة وهم يلبون…٢٨٧١من شاء أن يجعلها عمرة فليفعل٢٨٧٢قال عطاء قال جابر قدم النبي صلى الله عليه وسلم مكة صبيحة رابعة مضت من…٢٨٧٣خل عنه فلهو أسرع فيهم من نضح النبل٢٨٧٤هذا البلد حرمه الله يوم خلق السموات والأرض فهو حرام بحرمة الله إلى…٢٨٧٥إن هذا البلد حرام حرمه الله عز وجل لم يحل فيه القتال لأحد قبلي وأحل…٢٨٧٦إن مكة حرمها الله ولم يحرمها الناس ولا يحل لامرئ يؤمن بالله واليوم…٢٨٧٧يغزو هذا البيت جيش فيخسف بهم بالبيداء٢٨٧٨لا تنتهي البعوث عن غزو هذا البيت حتى يخسف بجيش منهم٢٨٧٩يبعث جند إلى هذا الحرم فإذا كانوا ببيداء من الأرض خسف بأولهم وآخرهم…٢٨٨٠ليؤمن هذا البيت جيش يغزونه حتى إذا كانوا ببيداء من الأرض خسف بأوسطهم…٢٨٨١خمس فواسق يقتلن في الحل والحرم الغراب والحدأة والكلب العقور والعقرب…٢٨٨٢خمس فواسق يقتلن في الحل والحرم الحية والكلب العقور والغراب الأبقع…٢٨٨٣أخبرنا أحمد بن سليمان، قال حدثنا يحيى بن آدم، عن حفص بن غياث، عن…٢٨٨٤أخبرنا عمرو بن علي، قال حدثنا يحيى، قال حدثنا ابن جريج، أخبرني أبو…٢٨٨٥قالت أمرني رسول الله صلى الله عليه وسلم بقتل الأوزاغ .٢٨٨٦ويونس، عن ابن شهاب، عن عروة، عن عائشة، أن رسول الله صلى الله عليه…٢٨٨٧خمس من الدواب كلهن فاسق يقتلن في الحل والحرم الكلب العقور والغراب…٢٨٨٨خمس من الدواب كلها فاسق يقتلن في الحرم الغراب والحدأة والكلب العقور…٢٨٨٩خمس من الدواب لا حرج على من قتلهن العقرب والغراب والحدأة والفأرة…٢٨٩٠خمس فواسق يقتلن في الحل والحرم الحدأة والغراب والفأرة والعقرب والكلب…٢٨٩١خمس فواسق يقتلن في الحرم العقرب والفأرة والغراب والكلب العقور والحدأة٢٨٩٢هذه مكة حرمها الله عز وجل يوم خلق السموات والأرض لم تحل لأحد قبلي ولا…٢٨٩٣خل عنه فوالذي نفسي بيده لكلامه أشد عليهم من وقع النبل٢٨٩٤أن النبي صلى الله عليه وسلم لما قدم مكة استقبله أغيلمة بني هاشم - قال…٢٨٩٥أخبرنا محمد بن بشار، قال حدثنا محمد، قال حدثنا شعبة، قال سمعت أبا…٢٨٩٦أن عبد الرحمن بن طارق بن علقمة، أخبره عن أمه، أن النبي صلى الله عليه…٢٨٩٧صلاة في مسجدي أفضل من ألف صلاة فيما سواه من المساجد إلا المسجد الحرام٢٨٩٨صلاة في مسجدي هذا أفضل من ألف صلاة فيما سواه من المساجد إلا المسجد…٢٨٩٩صلاة في مسجدي هذا أفضل من ألف صلاة فيما سواه من المساجد إلا الكعبة٢٩٠٠ألم ترى أن قومك حين بنوا الكعبة اقتصروا عن قواعد إبراهيم عليه السلام٢٩٠١لولا حداثة عهد قومك بالكفر لنقضت البيت فبنيته على أساس إبراهيم عليه…٢٩٠٢لولا أن قومي - وفي حديث محمد قومك - حديث عهد بجاهلية لهدمت الكعبة…٢٩٠٣يا عائشة لولا أن قومك حديث عهد بجاهلية لأمرت بالبيت فهدم فأدخلت فيه…٢٩٠٤يخرب الكعبة ذو السويقتين من الحبشة٢٩٠٥أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا ابن عون، عن…٢٩٠٦قال دخل رسول الله صلى الله عليه وسلم البيت ومعه الفضل بن عباس وأسامة…٢٩٠٧أن ابن عمر، قال دخل رسول الله صلى الله عليه وسلم الكعبة ودنا خروجه…٢٩٠٨يقول أتي ابن عمر في منزله فقيل هذا رسول الله صلى الله عليه وسلم قد…٢٩٠٩قال دخل رسول الله صلى الله عليه وسلم الكعبة فسبح في نواحيها وكبر ولم…٢٩١٠لولا أن الناس حديث عهدهم بكفر وليس عندي من النفقة ما يقوي على بنائه…٢٩١١ادخلي الحجر فإنه من البيت٢٩١٢إذا أردت دخول البيت فصلي ها هنا فإنما هو قطعة من البيت ولكن قومك…٢٩١٣أن ابن عباس، قال لم يصل النبي صلى الله عليه وسلم في الكعبة ولكنه كبر…٢٩١٤هذه القبلة هذه القبلة٢٩١٥هذه القبلة هذه القبلة٢٩١٦قال خرج رسول الله صلى الله عليه وسلم من البيت صلى ركعتين في قبل…٢٩١٧أن النبي صلى الله عليه وسلم دخل البيت فدعا في نواحيه كلها ولم يصل فيه…
