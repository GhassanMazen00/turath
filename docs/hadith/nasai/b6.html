سنن النسائي

كتاب المواقيت

١٣٢ حديثًا
٤٩٤نزل جبريل فأمني فصليت معه ثم صليت معه ثم صليت معه ثم صليت معه ثم صليت…٤٩٥أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا شعبة، قال حدثنا…٤٩٦أن رسول الله صلى الله عليه وسلم خرج حين زاغت الشمس فصلى بهم صلاة…٤٩٧أخبرنا يعقوب بن إبراهيم، قال حدثنا حميد بن عبد الرحمن، قال حدثنا…٤٩٨يقول كان النبي صلى الله عليه وسلم إذا نزل منزلا لم يرتحل منه حتى يصلي…٤٩٩قال كان رسول الله صلى الله عليه وسلم إذا كان الحر أبرد بالصلاة وإذا…٥٠٠إذا اشتد الحر فأبردوا عن الصلاة فإن شدة الحر من فيح جهنم٥٠١أبردوا بالظهر فإن الذي تجدون من الحر من فيح جهنم٥٠٢هذا جبريل عليه السلام جاءكم يعلمكم دينكم٥٠٣قال كان قدر صلاة رسول الله صلى الله عليه وسلم الظهر في الصيف ثلاثة…٥٠٤أخبرنا عبيد الله بن سعيد، قال حدثنا عبد الله بن الحارث، قال حدثنا…٥٠٥أن رسول الله صلى الله عليه وسلم صلى صلاة العصر والشمس في حجرتها لم…٥٠٦وإسحاق بن عبد الله، عن أنس، أن رسول الله صلى الله عليه وسلم كان يصلي…٥٠٧أنه أخبره أن رسول الله صلى الله عليه وسلم كان يصلي العصر والشمس…٥٠٨أخبرنا إسحاق بن إبراهيم، قال حدثنا جرير، عن منصور، عن ربعي بن حراش،…٥٠٩أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن أبي بكر بن عثمان بن سهل…٥١٠أخبرنا إسحاق بن إبراهيم، قال حدثنا أبو علقمة المدني، قال حدثنا محمد…٥١١تلك صلاة المنافق جلس يرقب صلاة العصر حتى إذا كانت بين قرنى الشيطان…٥١٢الذي تفوته صلاة العصر فكأنما وتر أهله وماله٥١٣ما بين هاتين الصلاتين وقت٥١٤من أدرك ركعتين من صلاة العصر قبل أن تغرب الشمس أو ركعة من صلاة الصبح…٥١٥من أدرك ركعة من صلاة العصر قبل أن تغيب الشمس أو أدرك ركعة من الفجر…٥١٦إذا أدرك أحدكم أول سجدة من صلاة العصر قبل أن تغرب الشمس فليتم صلاته…٥١٧من أدرك ركعة من صلاة الصبح قبل أن تطلع الشمس فقد أدرك الصبح ومن أدرك…٥١٨لا صلاة بعد العصر حتى تغيب الشمس ولا بعد الصبح حتى تطلع الشمس٥١٩أقم معنا هذين اليومين٥٢٠من أسلم من أصحاب النبي صلى الله عليه وسلم أنهم كانوا يصلون مع نبي…٥٢١إن هذه الصلاة عرضت على من كان قبلكم فضيعوها ومن حافظ عليها كان له…٥٢٢وقت صلاة الظهر ما لم تحضر العصر ووقت صلاة العصر ما لم تصفر الشمس ووقت…٥٢٣الوقت فيما بين هذين٥٢٤رسول الله صلى الله عليه وسلم وذاك زمن الحجاج بن يوسف . قال خرج رسول…٥٢٥أخبرنا محمد بن بشار، قال حدثنا يحيى، قال حدثنا عوف، قال حدثني سيار بن…٥٢٦ما بين هذين وقت كله٥٢٧ومحمد بن بشار، قالا حدثنا محمد، قال حدثنا شعبة، عن سعد بن إبراهيم، عن…٥٢٨بميقات هذه الصلاة عشاء الآخرة كان رسول الله صلى الله عليه وسلم يصليها…٥٢٩قال والله إني لأعلم الناس بوقت هذه الصلاة صلاة العشاء الآخرة كان رسول…٥٣٠أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن عوف، عن سيار بن سلامة،…٥٣١لولا أن أشق على أمتي لأمرتهم أن لا يصلوها إلا هكذا٥٣٢إنه الوقت لولا أن أشق على أمتي٥٣٣قال كان رسول الله صلى الله عليه وسلم يؤخر العشاء الآخرة .٥٣٤لولا أن أشق على أمتي لأمرتهم بتأخير العشاء وبالسواك عند كل صلاة٥٣٥ما ينتظرها غيركم٥٣٦إنه لوقتها لولا أن أشق على أمتي٥٣٧إنكم تنتظرون صلاة ما ينتظرها أهل دين غيركم ولولا أن يثقل على أمتي…٥٣٨إن الناس قد صلوا وناموا وأنتم لم تزالوا في صلاة ما انتظرتم الصلاة…٥٣٩إنكم لن تزالوا في صلاة ما انتظرتموها٥٤٠لو يعلم الناس ما في النداء والصف الأول ثم لم يجدوا إلا أن يستهموا…٥٤١لا تغلبنكم الأعراب على اسم صلاتكم هذه فإنهم يعتمون على الإبل وإنها…٥٤٢لا تغلبنكم الأعراب على اسم صلاتكم ألا إنها العشاء٥٤٣أن جابر بن عبد الله، قال صلى رسول الله صلى الله عليه وسلم الصبح حين…٥٤٤أين السائل عن وقت الصلاة ما بين هذين وقت٥٤٥قالت إن كان رسول الله صلى الله عليه وسلم ليصلي الصبح فينصرف النساء…٥٤٦أخبرنا إسحاق بن إبراهيم، قال حدثنا سفيان، عن الزهري، عن عروة، عن…٥٤٧الله أكبر خربت خيبر - مرتين - إنا إذا نزلنا بساحة قوم فساء صباح…٥٤٨أخبرنا عبيد الله بن سعيد، قال حدثنا يحيى، عن ابن عجلان، قال حدثني…٥٤٩ما أسفرتم بالفجر فإنه أعظم بالأجر٥٥٠من أدرك سجدة من الصبح قبل أن تطلع الشمس فقد أدركها ومن أدرك سجدة من…٥٥١من أدرك ركعة من الفجر قبل أن تطلع الشمس فقد أدركها ومن أدرك ركعة من…٥٥٢ومحمد بن عبد الأعلى، قالا حدثنا خالد، عن شعبة، عن أبي صدقة، عن أنس بن…٥٥٣من أدرك من الصلاة ركعة فقد أدرك الصلاة٥٥٤من أدرك من الصلاة ركعة فقد أدركها٥٥٥من أدرك من الصلاة ركعة فقد أدرك الصلاة٥٥٦من أدرك من الصلاة ركعة فقد أدركها٥٥٧من أدرك ركعة من الجمعة أو غيرها فقد تمت صلاته٥٥٨من أدرك ركعة من صلاة من الصلوات فقد أدركها إلا أنه يقضي ما فاته٥٥٩الشمس تطلع ومعها قرن الشيطان فإذا ارتفعت فارقها فإذا استوت قارنها…٥٦٠أخبرنا سويد بن نصر، قال حدثنا عبد الله، عن موسى بن علي بن رباح، قال…٥٦١أخبرنا قتيبة، عن مالك، عن محمد بن يحيى بن حبان، عن الأعرج، عن أبي…٥٦٢واحد، من أصحاب النبي صلى الله عليه وسلم منهم عمر - وكان من أحبهم إلى…٥٦٣لا يتحر أحدكم فيصلي عند طلوع الشمس وعند غروبها٥٦٤أن رسول الله صلى الله عليه وسلم نهى أن يصلى مع طلوع الشمس أو غروبها .٥٦٥أخبرنا حميد بن مسعدة، قال حدثنا سفيان، - وهو ابن حبيب - عن موسى بن…٥٦٦سمع أبا سعيد الخدري، يقول نهى رسول الله صلى الله عليه وسلم عن الصلاة…٥٦٧لا صلاة بعد الفجر حتى تبزغ الشمس ولا صلاة بعد العصر حتى تغرب الشمس٥٦٨أخبرني محمود بن غيلان، حدثنا الوليد، قال أخبرني عبد الرحمن بن نمر، عن…٥٦٩أخبرنا أحمد بن حرب، قال حدثنا سفيان، عن هشام بن حجير، عن طاوس، عن ابن…٥٧٠لا تتحروا بصلاتكم طلوع الشمس ولا غروبها فإنها تطلع بين قرنى شيطان٥٧١إذا طلع حاجب الشمس فأخروا الصلاة حتى تشرق وإذا غاب حاجب الشمس فأخروا…٥٧٢نعم إن أقرب ما يكون الرب عز وجل من العبد جوف الليل الآخر فإن استطعت…٥٧٣أخبرنا إسحاق بن إبراهيم، قال حدثنا جرير، عن منصور، عن هلال بن يساف،…٥٧٤قالت عائشة ما ترك رسول الله صلى الله عليه وسلم السجدتين بعد العصر…٥٧٥قال قالت عائشة رضى الله تعالى عنها ما دخل على رسول الله صلى الله عليه…٥٧٦والأسود، قالا نشهد على عائشة أنها قالت كان رسول الله صلى الله عليه…٥٧٧قالت صلاتان ما تركهما رسول الله صلى الله عليه وسلم في بيتي سرا ولا…٥٧٨كان رسول الله صلى الله عليه وسلم يصليهما بعد العصر فقالت إنه كان…٥٧٩هما ركعتان كنت أصليهما بعد الظهر فشغلت عنهما حتى صليت العصر٥٨٠أخبرنا إسحاق بن إبراهيم، قال أنبأنا وكيع، قال حدثنا طلحة بن يحيى، عن…٥٨١قبل غروب الشمس فقال كان عبد الله بن الزبير يصليهما فأرسل إليه معاوية…٥٨٢أن أبا الخير، حدثه أن أبا تميم الجيشاني قام ليركع ركعتين قبل المغرب…٥٨٣أنها قالت كان رسول الله صلى الله عليه وسلم إذا طلع الفجر لا يصلي إلا…٥٨٤وأيوب بن محمد، قالا حدثنا حجاج بن محمد، قال أيوب حدثنا وقال، حسن…٥٨٥يا بني عبد مناف لا تمنعوا أحدا طاف بهذا البيت وصلى أية ساعة شاء من…٥٨٦قال كان رسول الله صلى الله عليه وسلم إذا ارتحل قبل أن تزيغ الشمس أخر…٥٨٧والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن القاسم،…٥٨٨إذا حضر أحدكم الأمر الذي يخاف فوته فليصل هذه الصلاة٥٨٩قال صليت مع النبي صلى الله عليه وسلم بالمدينة ثمانيا جميعا وسبعا…٥٩٠أنه صلى بالبصرة الأولى والعصر ليس بينهما شىء والمغرب والعشاء ليس…٥٩١- شيخ من قريش - قال صحبت ابن عمر إلى الحمى فلما غربت الشمس هبت أن…٥٩٢قال رأيت رسول الله صلى الله عليه وسلم إذا أعجله السير في السفر يؤخر…٥٩٣قال غابت الشمس ورسول الله صلى الله عليه وسلم بمكة فجمع بين الصلاتين…٥٩٤أخبرني عمرو بن سواد بن الأسود بن عمرو، قال أنبأنا ابن وهب، قال حدثنا…٥٩٥أخبرنا محمود بن خالد، قال حدثنا الوليد، قال حدثنا ابن جابر، قال حدثني…٥٩٦أخبرنا قتيبة بن سعيد، حدثنا العطاف، عن نافع، قال أقبلنا مع ابن عمر من…٥٩٧إذا حضر أحدكم أمر يخشى فوته فليصل هذه الصلاة٥٩٨أن رسول الله صلى الله عليه وسلم كان إذا جد به السير جمع بين المغرب…٥٩٩قال كان رسول الله صلى الله عليه وسلم إذا جد به السير أو حزبه أمر جمع…٦٠٠قال رأيت النبي صلى الله عليه وسلم إذا جد به السير جمع بين المغرب…٦٠١قال صلى رسول الله صلى الله عليه وسلم الظهر والعصر جميعا والمغرب…٦٠٢أن النبي صلى الله عليه وسلم كان يصلي بالمدينة يجمع بين الصلاتين بين…٦٠٣قال صليت وراء رسول الله صلى الله عليه وسلم ثمانيا جميعا وسبعا جميعا .٦٠٤أن جابر بن عبد الله، قال سار رسول الله صلى الله عليه وسلم حتى أتى…٦٠٥أن أبا أيوب الأنصاري، أخبره أنه، صلى مع رسول الله صلى الله عليه وسلم…٦٠٦قال كنت مع ابن عمر حيث أفاض من عرفات فلما أتى جمعا جمع بين المغرب…٦٠٧أن النبي صلى الله عليه وسلم صلى المغرب والعشاء بالمزدلفة .٦٠٨قال ما رأيت النبي صلى الله عليه وسلم جمع بين صلاتين إلا بجمع وصلى…٦٠٩ومحمد بن أبي حرملة، عن كريب، عن ابن عباس، عن أسامة بن زيد، وكان…٦١٠الصلاة على وقتها وبر الوالدين والجهاد في سبيل الله عز وجل٦١١إقام الصلاة لوقتها وبر الوالدين والجهاد في سبيل الله عز وجل٦١٢وعمرو بن يزيد، قالا حدثنا ابن أبي عدي، عن شعبة، عن إبراهيم بن محمد بن…٦١٣من نسي صلاة فليصلها إذا ذكرها٦١٤كفارتها أن يصليها إذا ذكرها٦١٥إنه ليس في النوم تفريط إنما التفريط في اليقظة فإذا نسي أحدكم صلاة أو…٦١٦ليس في النوم تفريط إنما التفريط فيمن لم يصل الصلاة حتى يجيء وقت…٦١٧فليصلها أحدكم من الغد لوقتها٦١٨إذا نسيت الصلاة فصل إذا ذكرت فإن الله تعالى يقول { أقم الصلاة لذكري }٦١٩من نسي صلاة فليصلها إذا ذكرها فإن الله تعالى قال { أقم الصلاة لذكري }٦٢٠من نسي صلاة فليصلها إذا ذكرها فإن الله تعالى يقول { أقم الصلاة للذكرى…٦٢١أخبرنا هناد بن السري، عن أبي الأحوص، عن عطاء بن السائب، عن بريد بن…٦٢٢ما على الأرض عصابة يذكرون الله عز وجل غيركم٦٢٣ليأخذ كل رجل برأس راحلته فإن هذا منزل حضرنا فيه الشيطان٦٢٤من يكلؤنا الليلة لا نرقد عن صلاة الصبح٦٢٥قال أدلج رسول الله صلى الله عليه وسلم ثم عرس فلم يستيقظ حتى طلعت…
