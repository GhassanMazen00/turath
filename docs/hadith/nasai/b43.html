سنن النسائي

كتاب الضحايا

٨٨ حديثًا
٤٣٦١من رأى هلال ذي الحجة فأراد أن يضحي فلا يأخذ من شعره ولا من أظفاره حتى…٤٣٦٢من أراد أن يضحي فلا يقلم من أظفاره ولا يحلق شيئا من شعره في عشر الأول…٤٣٦٣قال من أراد أن يضحي، فدخلت أيام العشر فلا يأخذ من شعره ولا أظفاره .…٤٣٦٤إذا دخلت العشر فأراد أحدكم أن يضحي فلا يمس من شعره ولا من بشره شيئا٤٣٦٥أمرت بيوم الأضحى عيدا جعله الله عز وجل لهذه الأمة٤٣٦٦أن عبد الله، أخبره أن رسول الله صلى الله عليه وسلم كان يذبح أو ينحر…٤٣٦٧أن رسول الله صلى الله عليه وسلم نحر يوم الأضحى بالمدينة - قال - وقد…٤٣٦٨من ذبح قبل الصلاة فليذبح شاة مكانها ومن لم يكن ذبح فليذبح على اسم…٤٣٦٩أربع لا يجزن العوراء البين عورها والمريضة البين مرضها والعرجاء البين…٤٣٧٠أربعة لا يجزين في الأضاحي العوراء البين عورها والمريضة البين مرضها…٤٣٧١لا يجوز من الضحايا العوراء البين عورها والعرجاء البين عرجها والمريضة…٤٣٧٢أخبرني محمد بن آدم، عن عبد الرحيم، - وهو ابن سليمان - عن زكريا بن أبي…٤٣٧٣أخبرنا أبو داود، قال حدثنا الحسن بن محمد بن أعين، قال حدثنا زهير، قال…٤٣٧٤رضى الله عنه قال نهى رسول الله صلى الله عليه وسلم أن نضحي بمقابلة أو…٤٣٧٥لا يضحى بمقابلة ولا مدابرة ولا شرقاء ولا خرقاء ولا عوراء٤٣٧٦أن سلمة، - وهو ابن كهيل - أخبره قال سمعت حجية بن عدي، يقول سمعت عليا،…٤٣٧٧يقول نهى رسول الله صلى الله عليه وسلم أن يضحى بأعضب القرن . فذكرت ذلك…٤٣٧٨لا تذبحوا إلا مسنة إلا أن يعسر عليكم فتذبحوا جذعة من الضأن٤٣٧٩أن رسول الله صلى الله عليه وسلم أعطاه غنما يقسمها على صحابته فبقي…٤٣٨٠أن رسول الله صلى الله عليه وسلم قسم بين أصحابه ضحايا فصارت لي جذعة…٤٣٨١قال قسم رسول الله صلى الله عليه وسلم بين أصحابه أضاحي فأصابني جذعة…٤٣٨٢أخبرنا سليمان بن داود، عن ابن وهب، قال أخبرني عمرو، عن بكير بن الأشج،…٤٣٨٣إن الجذع يوفي مما يوفي منه الثني٤٣٨٤إن الجذعة تجزئ ما تجزئ منه الثنية٤٣٨٥أخبرنا إسحاق بن إبراهيم، قال حدثنا إسماعيل، عن عبد العزيز، - وهو ابن…٤٣٨٦قال ضحى رسول الله صلى الله عليه وسلم بكبشين أملحين .٤٣٨٧قال ضحى النبي صلى الله عليه وسلم بكبشين أملحين أقرنين ذبحهما بيده…٤٣٨٨أخبرنا إسماعيل بن مسعود، قال حدثنا حاتم بن وردان، عن أيوب، عن محمد بن…٤٣٨٩أخبرنا حميد بن مسعدة، في حديثه عن يزيد بن زريع، عن ابن عون، عن محمد،…٤٣٩٠قال ضحى رسول الله صلى الله عليه وسلم بكبش أقرن فحيل يمشي في سواد…٤٣٩١أخبرنا أحمد بن عبد الله بن الحكم، قال حدثنا محمد بن جعفر، قال حدثنا…٤٣٩٢أخبرنا محمد بن عبد العزيز بن غزوان، قال حدثنا الفضل بن موسى، عن حسين،…٤٣٩٣أخبرنا محمد بن المثنى، عن يحيى، عن عبد الملك، عن عطاء، عن جابر، قال…٤٣٩٤من وجه قبلتنا وصلى صلاتنا ونسك نسكنا فلا يذبح حتى يصلي٤٣٩٥من صلى صلاتنا ونسك نسكنا فقد أصاب النسك ومن نسك قبل الصلاة فتلك شاة…٤٣٩٦من كان ذبح قبل الصلاة فليعد٤٣٩٧أنه ذبح قبل النبي صلى الله عليه وسلم فأمره النبي صلى الله عليه وسلم…٤٣٩٨من ذبح قبل الصلاة فليذبح مكانها أخرى ومن كان لم يذبح حتى صلينا فليذبح…٤٣٩٩أنه أصاب أرنبين ولم يجد حديدة يذبحهما به فذكاهما بمروة فأتى النبي صلى…٤٤٠٠أن ذئبا، نيب في شاة فذبحوها بالمروة فرخص النبي صلى الله عليه وسلم في…٤٤٠١أنهر الدم بما شئت واذكر اسم الله عز وجل٤٤٠٢قال كانت لرجل من الأنصار ناقة ترعى في قبل أحد فعرض لها فنحرها بوتد .…٤٤٠٣ما أنهر الدم وذكر اسم الله فكل إلا بسن أو ظفر٤٤٠٤ما أنهر الدم وذكر اسم الله عز وجل فكلوا ما لم يكن سنا أو ظفرا…٤٤٠٥إن الله كتب الإحسان على كل شىء فإذا قتلتم فأحسنوا القتلة وإذا ذبحتم…٤٤٠٦أخبرنا عيسى بن أحمد العسقلاني، - عسقلان بلخ - قال حدثنا ابن وهب، قال…٤٤٠٧أن ذئبا، نيب في شاة فذبحوها بمروة فرخص النبي صلى الله عليه وسلم في…٤٤٠٨لو طعنت في فخذها لأجزأك٤٤٠٩ما أنهر الدم وذكر اسم الله عز وجل فكل ما خلا السن والظفر٤٤١٠ما أنهر الدم وذكر اسم الله عز وجل فكل ليس السن والظفر وسأحدثكم أما…٤٤١١إن الله عز وجل كتب الإحسان على كل شىء فإذا قتلتم فأحسنوا القتلة وإذا…٤٤١٢إن الله كتب الإحسان على كل شىء فإذا قتلتم فأحسنوا القتلة وإذا ذبحتم…٤٤١٣إن الله عز وجل كتب الإحسان على كل شىء فإذا قتلتم فأحسنوا القتلة وإذا…٤٤١٤إن الله عز وجل كتب الإحسان على كل شىء فإذا قتلتم فأحسنوا القتلة وإذا…٤٤١٥أخبرنا إسماعيل بن مسعود، قال حدثنا خالد، عن شعبة، أخبرني قتادة، قال…٤٤١٦قال كان رسول الله صلى الله عليه وسلم يضحي بكبشين أملحين أقرنين وكان…٤٤١٧قال لقد رأيته - يعني النبي صلى الله عليه وسلم - يذبحهما بيده واضعا…٤٤١٨أن أنس بن مالك، حدثهم أن نبي الله صلى الله عليه وسلم ضحى بكبشين…٤٤١٩والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٤٤٢٠ومحمد بن عبد الله بن يزيد، قالا حدثنا سفيان، عن هشام بن عروة، عن…٤٤٢١أخبرني محمد بن آدم، قال حدثنا عبدة، عن هشام بن عروة، عن فاطمة، عن…٤٤٢٢لعن الله من لعن والده ولعن الله من ذبح لغير الله ولعن الله من آوى…٤٤٢٣أن رسول الله صلى الله عليه وسلم نهى أن تؤكل لحوم الأضاحي بعد ثلاث .٤٤٢٤أخبرنا يعقوب بن إبراهيم، عن غندر، قال حدثنا معمر، قال حدثنا الزهري،…٤٤٢٥أن أبا عبيد، أخبره أن علي بن أبي طالب قال إن رسول الله صلى الله عليه…٤٤٢٦كلوا وتزودوا وادخروا٤٤٢٧- هو عبد الله بن خباب - أن أبا سعيد الخدري، قدم من سفر فقدم إليه أهله…٤٤٢٨قال حدثتني زينب، عن أبي سعيد الخدري، أن رسول الله صلى الله عليه وسلم…٤٤٢٩إني كنت نهيتكم عن ثلاث عن زيارة القبور فزوروها ولتزدكم زيارتها خيرا…٤٤٣٠إني كنت نهيتكم عن لحوم الأضاحي بعد ثلاث وعن النبيذ إلا في سقاء وعن…٤٤٣١كلوا وادخروا ثلاثا٤٤٣٢أخبرنا يعقوب بن إبراهيم، عن عبد الرحمن، عن سفيان، عن عبد الرحمن بن…٤٤٣٣أخبرنا يوسف بن عيسى، قال حدثنا الفضل بن موسى، قال حدثنا يزيد، - وهو…٤٤٣٤أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن ابن عون، عن ابن سيرين، عن…٤٤٣٥قال دلي جراب من شحم يوم خيبر فالتزمته قلت لا أعطي أحدا منه شيئا…٤٤٣٦اذكروا اسم الله عز وجل عليه وكلوا٤٤٣٧في قوله عز وجل { ولا تأكلوا مما لم يذكر اسم الله عليه } قال خاصمهم…٤٤٣٨لا تحل المجثمة٤٤٣٩قال دخلت مع أنس على الحكم - يعني ابن أيوب - فإذا أناس يرمون دجاجة في…٤٤٤٠لا تمثلوا بالبهائم٤٤٤١أخبرنا قتيبة بن سعيد، قال حدثنا هشيم، عن أبي بشر، عن سعيد بن جبير، عن…٤٤٤٢لعن الله من مثل بالحيوان٤٤٤٣لا تتخذوا شيئا فيه الروح غرضا٤٤٤٤لا تتخذوا شيئا فيه الروح غرضا٤٤٤٥من قتل عصفورا فما فوقها بغير حقها سأل الله عز جل عنها يوم القيامة٤٤٤٦من قتل عصفورا عبثا عج إلى الله عز وجل يوم القيامة يقول يا رب إن فلانا…٤٤٤٧وقال، مرة عن جده، أن رسول الله صلى الله عليه وسلم نهى يوم خيبر عن…٤٤٤٨أخبرنا إسماعيل بن مسعود، قال حدثنا خالد، قال حدثنا هشام، قال حدثنا…
