سنن النسائي

كتاب السهو

١٨٨ حديثًا•صفحة ١ من ٢
١١٧٩أخبرنا قتيبة بن سعيد، قال حدثنا أبو عوانة، عن عبد الرحمن بن الأصم،…١١٨٠قال صلى علي بن أبي طالب فكان يكبر في كل خفض ورفع يتم التكبير . فقال…١١٨١ومحمد بن بشار، - واللفظ له - قالا حدثنا يحيى بن سعيد، قال حدثنا عبد…١١٨٢أخبرنا محمد بن عبد الأعلى الصنعاني، قال حدثنا المعتمر، قال سمعت عبيد…١١٨٣ما منعك إذ أومأت إليك أن تصلي١١٨٤ما بالهم رافعين أيديهم في الصلاة كأنها أذناب الخيل الشمس اسكنوا في…١١٨٥ما بال هؤلاء يسلمون بأيديهم كأنها أذناب خيل شمس أما يكفي أحدهم أن يضع…١١٨٦صاحب رسول الله صلى الله عليه وسلم قال مررت على رسول الله صلى الله…١١٨٧قال قال ابن عمر دخل النبي صلى الله عليه وسلم مسجد قباء ليصلي فيه فدخل…١١٨٨أنه سلم على رسول الله صلى الله عليه وسلم وهو يصلي فرد عليه .١١٨٩إنك سلمت على آنفا وأنا أصلي١١٩٠قال بعثني النبي صلى الله عليه وسلم فأتيته وهو يسير مشرقا أو مغربا…١١٩١إذا قام أحدكم في الصلاة فلا يمسح الحصى فإن الرحمة تواجهه١١٩٢إن كنت لا بد فاعلا فمرة١١٩٣ما بال أقوام يرفعون أبصارهم إلى السماء في صلاتهم١١٩٤إذا كان أحدكم في الصلاة فلا يرفع بصره إلى السماء أن يلتمع بصره١١٩٥لا يزال الله عز وجل مقبلا على العبد في صلاته ما لم يلتفت فإذا صرف…١١٩٦اختلاس يختلسه الشيطان من الصلاة١١٩٧أخبرنا عمرو بن علي، قال حدثنا عبد الرحمن، قال حدثنا أبو الأحوص، عن…١١٩٨أخبرنا عمرو بن علي، قال حدثنا عبد الرحمن، قال حدثنا إسرائيل، عن أشعث…١١٩٩قال قالت عائشة إن الالتفات في الصلاة اختلاس يختلسه الشيطان من الصلاة .١٢٠٠إن كنتم آنفا تفعلون فعل فارس والروم يقومون على ملوكهم وهم قعود فلا…١٢٠١أخبرنا أبو عمار الحسين بن حريث، قال حدثنا الفضل بن موسى، عن عبد الله…١٢٠٢ويزيد، - وهو ابن زريع - عن معمر، عن يحيى بن أبي كثير، عن ضمضم بن جوس،…١٢٠٣أن رسول الله صلى الله عليه وسلم أمر بقتل الأسودين في الصلاة .١٢٠٤أن رسول الله صلى الله عليه وسلم كان يصلي وهو حامل أمامة فإذا سجد…١٢٠٥قال رأيت النبي صلى الله عليه وسلم يؤم الناس وهو حامل أمامة بنت أبي…١٢٠٦أخبرنا إسحاق بن إبراهيم، قال حدثنا حاتم بن وردان، قال حدثنا برد بن…١٢٠٧التسبيح للرجال والتصفيق للنساء١٢٠٨التسبيح للرجال والتصفيق للنساء١٢٠٩التسبيح للرجال والتصفيق للنساء١٢١٠التسبيح للرجال والتصفيق للنساء١٢١١قال كان لي من رسول الله صلى الله عليه وسلم ساعة آتيه فيها فإذا أتيته…١٢١٢قال قال علي كان لي من رسول الله صلى الله عليه وسلم مدخلان مدخل بالليل…١٢١٣قال قال لي علي كانت لي منزلة من رسول الله صلى الله عليه وسلم لم تكن…١٢١٤قال أتيت النبي صلى الله عليه وسلم وهو يصلي ولجوفه أزيز كأزيز المرجل…١٢١٥أخبرنا محمد بن سلمة، عن ابن وهب، عن معاوية بن صالح، قال حدثني ربيعة…١٢١٦لقد تحجرت واسعا١٢١٧لقد تحجرت واسعا١٢١٨ذاك شىء يجدونه في صدورهم فلا يصدنهم١٢١٩أخبرنا إسماعيل بن مسعود، قال حدثنا يحيى بن سعيد، قال حدثنا إسماعيل بن…١٢٢٠إن الله عز وجل - يعني - أحدث في الصلاة أن لا تكلموا إلا بذكر الله وما…١٢٢١إن الله عز وجل يحدث من أمره ما يشاء وإنه قد أحدث من أمره أن لا يتكلم…١٢٢٢أخبرنا قتيبة بن سعيد، عن مالك، عن ابن شهاب، عن عبد الرحمن الأعرج، عن…١٢٢٣أخبرنا قتيبة، قال حدثنا الليث، عن يحيى بن سعيد، عن عبد الرحمن بن…١٢٢٤لم أنس ولم تقصر الصلاة١٢٢٥أن رسول الله صلى الله عليه وسلم انصرف من اثنتين فقال له ذو اليدين…١٢٢٦أخبرنا قتيبة، عن مالك، عن داود بن الحصين، عن أبي سفيان، مولى ابن أبي…١٢٢٧أنه سمع أبا سلمة، يحدث عن أبي هريرة، أن رسول الله صلى الله عليه وسلم…١٢٢٨لم تنقص الصلاة ولم أنس١٢٢٩قال نسي رسول الله صلى الله عليه وسلم فسلم في سجدتين . فقال له ذو…١٢٣٠ما يقول ذو اليدين١٢٣١أن أبا بكر بن سليمان بن أبي حثمة، أخبره أنه، بلغه أن رسول الله صلى…١٢٣٢وأبي، سلمة وأبي بكر بن عبد الرحمن وابن أبي حثمة عن أبي هريرة، أنه قال…١٢٣٣أن رسول الله صلى الله عليه وسلم سجد يوم ذي اليدين سجدتين بعد السلام .١٢٣٤أخبرنا عمرو بن سواد بن الأسود، قال أخبرنا ابن وهب، قال أنبأنا عمرو بن…١٢٣٥وخالد الحذاء، عن ابن سيرين، عن أبي هريرة، أن النبي صلى الله عليه وسلم…١٢٣٦أن النبي صلى الله عليه وسلم صلى بهم فسها فسجد سجدتين ثم سلم .١٢٣٧قال سلم رسول الله صلى الله عليه وسلم في ثلاث ركعات من العصر فدخل…١٢٣٨إذا شك أحدكم في صلاته فليلغ الشك وليبن على اليقين فإذا استيقن بالتمام…١٢٣٩إذا لم يدر أحدكم صلى ثلاثا أم أربعا فليصل ركعة ثم يسجد بعد ذلك سجدتين…١٢٤٠إذا شك أحدكم في صلاته فليتحر الذي يرى أنه الصواب فيتمه ثم - يعني -…١٢٤١إذا شك أحدكم في صلاته فليتحر ويسجد سجدتين بعد ما يفرغ١٢٤٢لو حدث في الصلاة شىء أنبأتكموه ولكني إنما أنا بشر أنسى كما تنسون…١٢٤٣أخبرنا الحسن بن إسماعيل بن سليمان المجالدي، قال حدثنا الفضيل، - يعني…١٢٤٤أخبرنا إسماعيل بن مسعود، قال حدثنا خالد بن الحارث، عن شعبة، قال كتب…١٢٤٥يقول قال عبد الله من أوهم في صلاته فليتحر الصواب ثم يسجد سجدتين بعد…١٢٤٦قال من شك أو أوهم فليتحر الصواب ثم ليسجد سجدتين .١٢٤٧قال كانوا يقولون إذا أوهم يتحرى الصواب ثم يسجد سجدتين .١٢٤٨من شك في صلاته فليسجد سجدتين بعد ما يسلم١٢٤٩من شك في صلاته فليسجد سجدتين بعد التسليم١٢٥٠من شك في صلاته فليسجد سجدتين بعد ما يسلم١٢٥١من شك في صلاته فليسجد سجدتين١٢٥٢إن أحدكم إذا قام يصلي جاءه الشيطان فلبس عليه صلاته حتى لا يدري كم صلى…١٢٥٣إذا نودي للصلاة أدبر الشيطان له ضراط فإذا قضي التثويب أقبل حتى يخطر…١٢٥٤ومحمد بن بشار، - واللفظ لابن المثنى - قالا حدثنا يحيى، عن شعبة، عن…١٢٥٥ومغيرة، عن إبراهيم، عن علقمة، عن عبد الله، عن النبي صلى الله عليه…١٢٥٦أخبرنا محمد بن رافع، قال حدثني يحيى بن آدم، قال حدثنا مفضل بن مهلهل،…١٢٥٧أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن مالك بن مغول، قال سمعت…١٢٥٨أن علقمة، صلى خمسا فلما سلم قال إبراهيم بن سويد يا أبا شبل صليت خمسا…١٢٥٩أخبرنا سويد بن نصر، قال أنبأنا عبد الله، عن أبي بكر النهشلي، عن عبد…١٢٦٠من نسي شيئا من صلاته فليسجد مثل هاتين السجدتين١٢٦١ويونس، والليث، أن ابن شهاب، أخبرهم عن عبد الرحمن الأعرج، أن عبد الله…١٢٦٢ومحمد بن بشار، بندار - واللفظ له - قالا حدثنا يحيى بن سعيد، قال حدثنا…١٢٦٣قال رأيت رسول الله صلى الله عليه وسلم يرفع يديه إذا افتتح الصلاة وإذا…١٢٦٤أنه رأى النبي صلى الله عليه وسلم جلس في الصلاة فافترش رجله اليسرى…١٢٦٥قال قلت لأنظرن إلى صلاة رسول الله صلى الله عليه وسلم كيف يصلي فقام…١٢٦٦أخبرنا محمد بن منصور، قال حدثنا سفيان، قال حدثنا يحيى بن سعيد، عن…١٢٦٧بالحصى في الصلاة فلما انصرف نهاني وقال اصنع كما كان رسول الله صلى…١٢٦٨قال قلت لأنظرن إلى صلاة رسول الله صلى الله عليه وسلم كيف يصلي فنظرت…١٢٦٩أن رسول الله صلى الله عليه وسلم كان إذا جلس في الصلاة وضع يديه على…١٢٧٠أخبرنا أيوب بن محمد الوزان، قال حدثنا حجاج، قال ابن جريج أخبرني زياد،…١٢٧١قال رأيت رسول الله صلى الله عليه وسلم واضعا يده اليمنى على فخذه…١٢٧٢أن رجلا، كان يدعو بأصبعيه فقال رسول الله صلى الله عليه وسلم " أحد أحد…١٢٧٣أخبرنا محمد بن عبد الله بن المبارك المخرمي، قال حدثنا أبو معاوية، قال…١٢٧٤من أهل البصرة أن أباه، حدثه أنه، رأى رسول الله صلى الله عليه وسلم…١٢٧٥أن رسول الله صلى الله عليه وسلم كان إذا قعد في التشهد وضع كفه اليسرى…١٢٧٦لينتهين أقوام عن رفع أبصارهم عند الدعاء في الصلاة إلى السماء أو…١٢٧٧لا تقولوا هكذا فإن الله عز وجل هو السلام ولكن قولوا التحيات لله…١٢٧٨أخبرنا أحمد بن سليمان، قال حدثنا يحيى بن آدم، قال حدثنا عبد الرحمن بن…١٢٧٩إن الله عز وجل هو السلام فإذا قعد أحدكم فليقل التحيات لله والصلوات…١٢٨٠إذا قمتم إلى الصلاة فأقيموا صفوفكم ثم ليؤمكم أحدكم فإذا كبر فكبروا…١٢٨١بسم الله وبالله التحيات لله والصلوات والطيبات السلام عليك أيها النبي…١٢٨٢إن لله ملائكة سياحين في الأرض يبلغوني من أمتي السلام١٢٨٣إنه أتاني الملك فقال يا محمد إن ربك يقول أما يرضيك أنه لا يصلي عليك…١٢٨٤عجلت أيها المصلي١٢٨٥قولوا اللهم صل على محمد وعلى آل محمد كما صليت على آل إبراهيم وبارك…١٢٨٦قولوا اللهم صل على محمد كما صليت على آل إبراهيم اللهم بارك على محمد…١٢٨٧قولوا اللهم صل على محمد وعلى آل محمد كما صليت على آل إبراهيم إنك حميد…١٢٨٨قولوا اللهم صل على محمد وعلى آل محمد كما صليت على إبراهيم وآل إبراهيم…١٢٨٩قولوا اللهم صل على محمد وآل محمد كما صليت على آل إبراهيم إنك حميد…١٢٩٠قولوا اللهم صل على محمد وعلى آل محمد كما صليت على إبراهيم وآل إبراهيم…١٢٩١قولوا اللهم صل على محمد وعلى آل محمد كما صليت على إبراهيم إنك حميد…١٢٩٢صلوا على واجتهدوا في الدعاء وقولوا اللهم صل على محمد وعلى آل محمد١٢٩٣قولوا اللهم صل على محمد عبدك ورسولك كما صليت على إبراهيم وبارك على…١٢٩٤قولوا اللهم صل على محمد وأزواجه وذريته١٢٩٥إنه جاءني جبريل صلى الله عليه وسلم فقال أما يرضيك يا محمد أن لا يصلي…١٢٩٦من صلى على واحدة صلى الله عليه عشرا١٢٩٧من صلى على صلاة واحدة صلى الله عليه عشر صلوات وحطت عنه عشر خطيئات…١٢٩٨لا تقولوا السلام على الله فإن الله هو السلام ولكن إذا جلس أحدكم فليقل…١٢٩٩سبحي الله عشرا واحمديه عشرا وكبريه عشرا ثم سليه حاجتك يقل نعم نعم١٣٠٠قال كنت مع رسول الله صلى الله عليه وسلم جالسا - يعني - ورجل قائم يصلي…١٣٠١أن محجن بن الأدرع، حدثه أن رسول الله صلى الله عليه وسلم دخل المسجد…١٣٠٢قل اللهم إني ظلمت نفسي ظلما كثيرا ولا يغفر الذنوب إلا أنت فاغفر لي…١٣٠٣إني لأحبك يا معاذ١٣٠٤اللهم إني أسألك الثبات في الأمر والعزيمة على الرشد وأسألك شكر نعمتك…١٣٠٥اللهم بعلمك الغيب وقدرتك على الخلق أحيني ما علمت الحياة خيرا لي…١٣٠٦اللهم بعلمك الغيب وقدرتك على الخلق أحيني ما علمت الحياة خيرا لي…١٣٠٧اللهم إني أعوذ بك من شر ما عملت ومن شر ما لم أعمل١٣٠٨نعم عذاب القبر حق١٣٠٩اللهم إني أعوذ بك من عذاب القبر وأعوذ بك من فتنة المسيح الدجال وأعوذ…١٣١٠إذا تشهد أحدكم فليتعوذ بالله من أربع من عذاب جهنم وعذاب القبر وفتنة…١٣١١أحسن الكلام كلام الله وأحسن الهدى هدى محمد صلى الله عليه وسلم١٣١٢أنه رأى رجلا يصلي فطفف فقال له حذيفة منذ كم تصلي هذه الصلاة قال منذ…١٣١٣ارجع فصل فإنك لم تصل١٣١٤ارجع فصل فإنك لم تصل١٣١٥أخبرنا محمد بن بشار، قال حدثنا يحيى، عن سعيد، عن قتادة، عن زرارة بن…١٣١٦أخبرنا محمد بن إسماعيل بن إبراهيم، قال حدثنا سليمان، - يعني ابن داود…١٣١٧أخبرنا إسحاق بن إبراهيم، قال أنبأنا أبو عامر العقدي، قال حدثنا عبد…١٣١٨ما بال هؤلاء الذين يرمون بأيديهم كأنها أذناب الخيل الشمس أما يكفي أن…١٣١٩السلام عليكم ورحمة الله السلام عليكم ورحمة الله١٣٢٠السلام عليكم ورحمة الله عن يمينه السلام عليكم ورحمة الله عن يساره١٣٢١السلام عليكم ورحمة الله١٣٢٢السلام عليكم ورحمة الله١٣٢٣أخبرنا محمد بن آدم، عن عمر بن عبيد، عن أبي إسحاق، عن أبي الأحوص، عن…١٣٢٤السلام عليكم ورحمة الله السلام عليكم ورحمة الله١٣٢٥السلام عليكم ورحمة الله١٣٢٦ما شأنكم تشيرون بأيديكم كأنها أذناب خيل شمس إذا سلم أحدكم فليلتفت إلى…١٣٢٧سأفعل إن شاء الله١٣٢٨وعمرو بن الحارث، ويونس بن يزيد، أن ابن شهاب، أخبرهم عن
