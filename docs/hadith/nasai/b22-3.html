سنن النسائي

كتاب الصيام

٣٤٥ حديثًا•صفحة ٣ من ٣
٢٣٩٠لكني أنا أقوم وأنام وأصوم وأفطر فقم ونم وصم وأفطر٢٣٩١ألم أخبر أنك تقوم الليل وتصوم النهار٢٣٩٢أنت الذي تقول ذلك٢٣٩٣بلغني أنك قلت لأصومن الدهر ولأقرأن القرآن٢٣٩٤صم يوما ولك أجر ما بقي٢٣٩٥صم من كل عشرة أيام يوما ولك أجر تلك التسعة٢٣٩٦صم يوما ولك أجر عشرة٢٣٩٧إنه بلغني أنك تقوم الليل وتصوم النهار٢٣٩٨أخبرنا علي بن الحسين، قال حدثنا أمية، عن شعبة، عن حبيب، قال حدثني أبو…٢٣٩٩يا عبد الله بن عمرو إنك تصوم الدهر وتقوم الليل وإنك إذا فعلت ذلك هجمت…٢٤٠٠اقرإ القرآن في شهر٢٤٠١ألم أخبر أنك تصوم ولا تفطر وتصلي الليل فلا تفعل فإن لعينك حظا ولنفسك…٢٤٠٢أما يكفيك من كل شهر ثلاثة أيام٢٤٠٣صم من الشهر يوما ولك أجر ما بقي٢٤٠٤قال أوصاني حبيبي صلى الله عليه وسلم بثلاثة لا أدعهن إن شاء الله تعالى…٢٤٠٥قال أمرني رسول الله صلى الله عليه وسلم بثلاث بنوم على وتر والغسل يوم…٢٤٠٦قال أمرني رسول الله صلى الله عليه وسلم بركعتى الضحى وأن لا أنام إلا…٢٤٠٧رضى الله عنه قال أمرني رسول الله صلى الله عليه وسلم بنوم على وتر…٢٤٠٨شهر الصبر وثلاثة أيام من كل شهر صوم الدهر٢٤٠٩من صام ثلاثة أيام من الشهر فقد صام الدهر كله٢٤١٠من صام ثلاثة أيام من كل شهر فقد تم صوم الشهر أو فله صوم الشهر٢٤١١صيام حسن ثلاثة أيام من الشهر٢٤١٢قال عثمان بن أبي العاص نحوه مرسل .٢٤١٣يقول كان النبي صلى الله عليه وسلم يصوم ثلاثة أيام من كل شهر .٢٤١٤أن رسول الله صلى الله عليه وسلم كان يصوم ثلاثة أيام من كل شهر يوم…٢٤١٥أخبرنا علي بن محمد بن علي، قال حدثنا خلف بن تميم، عن زهير، عن الحر بن…٢٤١٦قالت أربع لم يكن يدعهن النبي صلى الله عليه وسلم صيام عاشوراء والعشر…٢٤١٧أزواج النبي صلى الله عليه وسلم أن رسول الله صلى الله عليه وسلم كان…٢٤١٨أزواج النبي صلى الله عليه وسلم قالت كان النبي صلى الله عليه وسلم يصوم…٢٤١٩قالت كان رسول الله صلى الله عليه وسلم يأمر بصيام ثلاثة أيام أول خميس…٢٤٢٠صيام ثلاثة أيام من كل شهر صيام الدهر وأيام البيض صبيحة ثلاث عشرة…٢٤٢١ما يمنعك أن تأكل٢٤٢٢أخبرنا محمد بن عبد العزيز، قال أنبأنا الفضل بن موسى، عن فطر، عن يحيى…٢٤٢٣أخبرنا عمرو بن يزيد، قال حدثنا عبد الرحمن، قال حدثنا شعبة، عن الأعمش،…٢٤٢٤إذا صمت شيئا من الشهر فصم ثلاث عشرة وأربع عشرة وخمس عشرة٢٤٢٥عليك بصيام ثلاث عشرة وأربع عشرة وخمس عشرة٢٤٢٦أن النبي صلى الله عليه وسلم أمر رجلا بصيام ثلاث عشرة وأربع عشرة وخمس…٢٤٢٧أخبرنا أحمد بن عثمان بن حكيم، عن بكر، عن عيسى، عن محمد، عن الحكم، عن…٢٤٢٨أن رجلا، أتى النبي صلى الله عليه وسلم بأرنب وكان النبي صلى الله عليه…٢٤٢٩كلوا فإني لو اشتهيتها أكلتها٢٤٣٠أن رسول الله صلى الله عليه وسلم كان يأمر بهذه الأيام الثلاث البيض…٢٤٣١أن النبي صلى الله عليه وسلم أمرهم بصيام ثلاثة أيام البيض قال " هي صوم…٢٤٣٢أخبرنا محمد بن معمر، قال حدثنا حبان، قال حدثنا همام، قال حدثنا أنس بن…٢٤٣٣صم يوما من الشهر٢٤٣٤صم يوما من كل شهر
