كتاب مناسك الحج

٤٦٧ حديثًا•صفحة ٣ من ٤
٢٩١٨أخبرنا عمرو بن علي، قال حدثنا يحيى، قال حدثنا السائب بن عمر، قال…٢٩١٩إن مسحهما يحطان الخطيئة٢٩٢٠أن طاوسا، أخبره عن ابن عباس، أن النبي صلى الله عليه وسلم مر وهو يطوف…٢٩٢١قال مر رسول الله صلى الله عليه وسلم برجل يقوده رجل بشىء ذكره في نذر…٢٩٢٢الطواف بالبيت صلاة فأقلوا من الكلام٢٩٢٣قال قال عبد الله بن عمر أقلوا الكلام في الطواف فإنما أنتم في الصلاة .٢٩٢٤يا بني عبد مناف لا تمنعن أحدا طاف بهذا البيت وصلى أى ساعة شاء من ليل…٢٩٢٥طوفي من وراء الناس وأنت راكبة٢٩٢٦إذا أقيمت الصلاة فطوفي على بعيرك من وراء الناس٢٩٢٧طوفي من وراء المصلين وأنت راكبة٢٩٢٨قالت طاف رسول الله صلى الله عليه وسلم في حجة الوداع حول الكعبة على…٢٩٢٩أن وبرة، حدثه قال سمعت عبد الله بن عمر، وسأله، رجل أطوف بالبيت وقد…٢٩٣٠وسألناه، عن رجل، قدم معتمرا فطاف بالبيت ولم يطف بين الصفا والمروة…٢٩٣١لولا أن معي الهدى لأحللت٢٩٣٢قرن الحج والعمرة فطاف طوافا واحدا وقال هكذا رأيت رسول الله صلى الله…٢٩٣٣وأيوب بن موسى، وإسماعيل بن أمية، وعبيد الله بن عمر، عن نافع، قال خرج…٢٩٣٤أن النبي صلى الله عليه وسلم طاف طوافا واحدا .٢٩٣٥الحجر الأسود من الجنة٢٩٣٦أن عمر، قبل الحجر والتزمه وقال رأيت أبا القاسم صلى الله عليه وسلم بك…٢٩٣٧وجرير، عن الأعمش، عن إبراهيم، عن عابس بن ربيعة، قال رأيت عمر جاء إلى…٢٩٣٨قال رأيت طاوسا يمر بالركن فإن وجد عليه زحاما مر ولم يزاحم وإن رآه…٢٩٣٩{ واتخذوا من مقام إبراهيم مصلى }٢٩٤٠أن عبد الله بن عمر، كان يرمل الثلاث ويمشي الأربع ويزعم أن رسول الله…٢٩٤١أن رسول الله صلى الله عليه وسلم كان إذا طاف في الحج والعمرة أول ما…٢٩٤٢وسليمان بن داود، عن ابن وهب، قال أخبرني يونس، عن ابن شهاب، عن سالم،…٢٩٤٣وعبد الرحمن، ابنا عبد الله بن عبد الحكم قالا حدثنا شعيب بن الليث، عن…٢٩٤٤والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٢٩٤٥قال لما قدم النبي صلى الله عليه وسلم وأصحابه مكة قال المشركون وهنتهم…٢٩٤٦فقال رأيت رسول الله صلى الله عليه وسلم يستلمه ويقبله . فقال الرجل…٢٩٤٧أن النبي صلى الله عليه وسلم كان يستلم الركن اليماني والحجر في كل طواف…٢٩٤٨ومحمد بن المثنى، قالا حدثنا خالد، قال حدثنا عبيد الله، عن نافع، عن…٢٩٤٩قال لم أر رسول الله صلى الله عليه وسلم يمسح من البيت إلا الركنين…٢٩٥٠وابن، جريج ومالك عن المقبري، عن عبيد بن جريج، قال قلت لابن عمر رأيتك…٢٩٥١والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن وهب، قال أخبرني يونس،…٢٩٥٢قال قال عبد الله رضي الله عنه ما تركت استلام هذين الركنين منذ رأيت…٢٩٥٣قال ما تركت استلام الحجر في رخاء ولا شدة منذ رأيت رسول الله صلى الله…٢٩٥٤وسليمان بن داود، عن ابن وهب، قال أخبرني يونس، عن ابن شهاب، عن عبيد…٢٩٥٥أن رسول الله صلى الله عليه وسلم كان يطوف بالبيت على راحلته فإذا انتهى…٢٩٥٦قال كانت المرأة تطوف بالبيت وهي عريانة تقول اليوم يبدو بعضه أو كله…٢٩٥٧ألا لا يحجن بعد العام مشرك ولا يطوف بالبيت عريان٢٩٥٨وعثمان بن عمر، قالا حدثنا شعبة، عن المغيرة، عن الشعبي، عن المحرر بن…٢٩٥٩قال رأيت النبي صلى الله عليه وسلم حين فرغ من سبعه جاء حاشية المطاف…٢٩٦٠قال يعني ابن عمر قدم رسول الله صلى الله عليه وسلم فطاف بالبيت سبعا…٢٩٦١نبدأ بما بدأ الله٢٩٦٢إن الصفا والمروة من شعائر الله فابدءوا بما بدأ الله به٢٩٦٣أن رسول الله صلى الله عليه وسلم لما انتهى إلى مقام إبراهيم قرأ {…٢٩٦٤ومغيرة، ح وأنبأنا يعقوب بن إبراهيم، قال حدثنا هشيم، قال أنبأنا عاصم،…٢٩٦٥قال سقيت رسول الله صلى الله عليه وسلم من زمزم فشربه وهو قائم .٢٩٦٦أخبرنا محمد بن بشار، قال حدثنا محمد، قال حدثنا شعبة، عن عمرو بن…٢٩٦٧أخبرنا محمد بن منصور، قال حدثنا سفيان، عن الزهري، عن عروة، قال قرأت…٢٩٦٨أخبرني عمرو بن عثمان، قال حدثنا أبي، عن شعيب، عن الزهري، عن عروة، قال…٢٩٦٩نبدأ بما بدأ الله به٢٩٧٠نبدأ بما بدأ الله به٢٩٧١أن رسول الله صلى الله عليه وسلم رقي على الصفا حتى إذا نظر إلى البيت…٢٩٧٢لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء قدير٢٩٧٣أنه سمع أباه، يحدث أنه سمع جابرا، عن حجة النبي، صلى الله عليه وسلم ثم…٢٩٧٤نبدأ بما بدأ الله به٢٩٧٥أنه سمع جابر بن عبد الله، يقول طاف النبي صلى الله عليه وسلم في حجة…٢٩٧٦قال رأيت ابن عمر يمشي بين الصفا والمروة فقال إن أمشي فقد رأيت رسول…٢٩٧٧كبير .٢٩٧٨قال سألوا ابن عمر هل رأيت رسول الله صلى الله عليه وسلم رمل بين الصفا…٢٩٧٩قال إنما سعى النبي صلى الله عليه وسلم بين الصفا والمروة ليري المشركين…٢٩٨٠لا يقطع الوادي إلا شدا٢٩٨١والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٢٩٨٢قال لما تصوبت قدما رسول الله صلى الله عليه وسلم في بطن الوادي رمل حتى…٢٩٨٣أخبرنا يعقوب بن إبراهيم، قال حدثنا يحيى بن سعيد، قال حدثنا جعفر بن…٢٩٨٤لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء قدير٢٩٨٥لا إله إلا الله وحده لا شريك له له الملك وله الحمد يحيي ويميت وهو على…٢٩٨٦أنه سمع جابرا، يقول لم يطف النبي صلى الله عليه وسلم وأصحابه بين الصفا…٢٩٨٧أن طاوسا، أخبره أن ابن عباس أخبره عن معاوية، أنه قصر عن النبي صلى…٢٩٨٨أخبرنا محمد بن يحيى بن عبد الله، قال حدثنا عبد الرزاق، قال أنبأنا…٢٩٨٩قال أخذت من أطراف شعر رسول الله صلى الله عليه وسلم بمشقص كان معي بعد…٢٩٩٠من كان معه هدى فليقم على إحرامه ومن لم يكن معه هدى فليحلل٢٩٩١من أهل بعمرة ولم يهد فليحلل ومن أهل بعمرة فأهدى فلا يحل ومن أهل بحجة…٢٩٩٢من لم يكن معه هدى فليحلل ومن كان معه هدى فليقم على إحرامه٢٩٩٣أخبرنا إسحاق بن إبراهيم، قال قرأت على أبي قرة موسى بن طارق عن ابن…٢٩٩٤أحلوا واجعلوها عمرة٢٩٩٥إذا كنت بين الأخشبين من منى ونفخ - بيده نحو المشرق - فإن هناك واديا…٢٩٩٦أخبرنا محمد بن حاتم بن نعيم، قال أنبأنا سويد، قال أنبأنا عبد الله، عن…٢٩٩٧وعبد الرحمن بن محمد بن سلام، قالا حدثنا إسحاق الأزرق، عن سفيان…٢٩٩٨أخبرنا يحيى بن حبيب بن عربي، قال حدثنا حماد، عن يحيى بن سعيد…٢٩٩٩أخبرنا يعقوب بن إبراهيم الدورقي، قال حدثنا هشيم، قال حدثنا يحيى، عن…٣٠٠٠قال قلت لأنس ونحن غاديان من منى إلى عرفات ما كنتم تصنعون في التلبية…٣٠٠١- وهو الثقفي - قال قلت لأنس غداة عرفة ما تقول في التلبية في هذا اليوم…٣٠٠٢أخبرنا إسحاق بن إبراهيم، قال أنبأنا عبد الله بن إدريس، عن أبيه، عن…٣٠٠٣ما من يوم أكثر من أن يعتق الله عز وجل فيه عبدا أو أمة من النار من يوم…٣٠٠٤إن يوم عرفة ويوم النحر وأيام التشريق عيدنا أهل الإسلام وهي أيام أكل…٣٠٠٥أن ابن شهاب، حدثه عن سالم بن عبد الله، قال كتب عبد الملك بن مروان إلى…٣٠٠٦قال كنت مع ابن عباس بعرفات فقال ما لي لا أسمع الناس يلبون قلت يخافون…٣٠٠٧قال رأيت رسول الله صلى الله عليه وسلم يخطب على جمل أحمر بعرفة قبل…٣٠٠٨قال رأيت رسول الله صلى الله عليه وسلم يخطب يوم عرفة على جمل أحمر .٣٠٠٩أن عبد الله بن عمر، جاء إلى الحجاج بن يوسف يوم عرفة حين زالت الشمس…٣٠١٠قال كان رسول الله صلى الله عليه وسلم يصلي الصلاة لوقتها إلا بجمع…٣٠١١قال قال أسامة بن زيد كنت رديف النبي صلى الله عليه وسلم بعرفات فرفع…٣٠١٢قالت كانت قريش تقف بالمزدلفة ويسمون الحمس وسائر العرب تقف بعرفة فأمر…٣٠١٣قال أضللت بعيرا لي فذهبت أطلبه بعرفة يوم عرفة فرأيت النبي صلى الله…٣٠١٤كونوا على مشاعركم فإنكم على إرث من إرث أبيكم إبراهيم عليه السلام٣٠١٥عرفة كلها موقف٣٠١٦الحج عرفة فمن أدرك ليلة عرفة قبل طلوع الفجر من ليلة جمع فقد تم حجه٣٠١٧قال أفاض رسول الله صلى الله عليه وسلم من عرفات وردفه أسامة بن زيد…٣٠١٨يا أيها الناس عليكم بالسكينة والوقار فإن البر ليس في إيضاع الإبل٣٠١٩السكينة السكينة٣٠٢٠وكان، رديف رسول الله صلى الله عليه وسلم أن رسول الله صلى الله عليه…٣٠٢١قال أفاض رسول الله صلى الله عليه وسلم وعليه السكينة وأمرهم بالسكينة…٣٠٢٢السكينة عباد الله٣٠٢٣صلى الله عليه وسلم في حجة الوداع قال كان يسير العنق فإذا وجد فجوة نص…٣٠٢٤أن النبي صلى الله عليه وسلم حيث أفاض من عرفة مال إلى الشعب قال فقلت…٣٠٢٥أخبرنا محمود بن غيلان، قال حدثنا وكيع، قال حدثنا سفيان، عن إبراهيم بن…٣٠٢٦أن رسول الله صلى الله عليه وسلم جمع بين المغرب والعشاء بجمع .٣٠٢٧أن النبي صلى الله عليه وسلم جمع بين المغرب والعشاء بجمع .٣٠٢٨أن رسول الله صلى الله عليه وسلم جمع بين المغرب والعشاء بجمع بإقامة…٣٠٢٩أن عبيد الله بن عبد الله، أخبره أن أباه قال جمع رسول الله صلى الله…٣٠٣٠قال صلى رسول الله صلى الله عليه وسلم المغرب والعشاء بجمع بإقامة واحدة…٣٠٣١أن كريبا، قال سألت أسامة بن زيد - وكان ردف رسول الله صلى الله عليه…٣٠٣٢قدم النبي صلى الله عليه وسلم ليلة المزدلفة في ضعفة أهله .٣٠٣٣قال كنت فيمن قدم النبي صلى الله عليه وسلم ليلة المزدلفة في ضعفة أهله .٣٠٣٤وعفان، وسليمان، عن شعبة، عن مشاش، عن عطاء، عن ابن عباس، عن الفضل، أن…٣٠٣٥أن أم حبيبة، أخبرته أن النبي صلى الله عليه وسلم أمرها أن تغلس من جمع…٣٠٣٦أخبرنا عبد الجبار بن العلاء، عن سفيان، عن عمرو، عن سالم بن شوال، عن…٣٠٣٧قالت إنما أذن النبي صلى الله عليه وسلم لسودة في الإفاضة قبل الصبح من…٣٠٣٨قال ما رأيت رسول الله صلى الله عليه وسلم صلى صلاة قط إلا لميقاتها إلا…٣٠٣٩من صلى معنا صلاتنا هذه ها هنا ثم أقام معنا وقد وقف قبل ذلك بعرفة ليلا…٣٠٤٠من أدرك جمعا مع الإمام والناس حتى يفيض منها فقد أدرك الحج ومن لم يدرك…٣٠٤١من صلى هذه الصلاة معنا وقد وقف قبل ذلك بعرفة ليلا أو نهارا فقد تم حجه…٣٠٤٢من صلى هذه الصلاة معنا ووقف هذا الموقف حتى يفيض وأفاض قبل ذلك من…٣٠٤٣من صلى صلاة الغداة ها هنا معنا وقد أتى عرفة قبل ذلك فقد قضى تفثه وتم…٣٠٤٤الحج عرفة من جاء ليلة جمع قبل صلاة الصبح فقد أدرك حجه أيام منى ثلاثة…٣٠٤٥المزدلفة كلها موقف٣٠٤٦لبيك اللهم لبيك٣٠٤٧أخبرنا إسماعيل بن مسعود، قال حدثنا خالد، قال حدثنا شعبة، عن أبي…٣٠٤٨أن داود بن عبد الرحمن، حدثهم أن عمرو بن دينار حدثه أن عطاء بن أبي…٣٠٤٩عائشة قالت وددت أني استأذنت رسول الله صلى الله عليه وسلم كما استأذنته…٣٠٥٠أن مولى، لأسماء بنت أبي بكر أخبره قال جئت مع أسماء بنت أبي بكر منى…٣٠٥١معه كيف كان رسول الله صلى الله عليه وسلم يسير في حجة الوداع حين دفع…٣٠٥٢عليكم بالسكينة٣٠٥٣أن النبي صلى الله عليه وسلم أوضع في وادي محسر .٣٠٥٤صلى الله عليه وسلم . فقال إن رسول الله صلى الله عليه وسلم دفع من…٣٠٥٥وعبد الملك بن أبي سليمان، عن عطاء، عن ابن عباس، عن الفضل بن عباس، أنه…٣٠٥٦أن رسول الله صلى الله عليه وسلم لبى حتى رمى الجمرة .٣٠٥٧قال قال ابن عباس قال لي رسول الله صلى الله عليه وسلم غداة العقبة وهو…٣٠٥٨عليكم بالسكينة٣٠٥٩قال قال رسول الله صلى الله عليه وسلم غداة العقبة وهو واقف على راحلته…٣٠٦٠قالت حججت في حجة النبي صلى الله عليه وسلم فرأيت بلالا يقود بخطام…٣٠٦١قال رأيت رسول الله صلى الله عليه وسلم يرمي جمرة العقبة يوم النحر على…٣٠٦٢يا أيها الناس خذوا مناسككم فإني لا أدري لعلي لا أحج بعد عامي هذا٣٠٦٣قال رمى رسول الله صلى الله عليه وسلم الجمرة يوم النحر ضحى ورمى بعد…٣٠٦٤أبيني لا ترموا جمرة العقبة حتى تطلع الشمس٣٠٦٥أن النبي صلى الله عليه وسلم قدم أهله وأمرهم أن لا يرموا الجمرة حتى…٣٠٦٦قال حدثتني عائشة بنت طلحة، عن خالتها، عائشة أم المؤمنين أن رسول الله…٣٠٦٧قال كان رسول الله صلى الله عليه وسلم يسأل أيام منى فيقول " لا حرج " .…
