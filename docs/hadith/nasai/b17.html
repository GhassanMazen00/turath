سنن النسائي

كتاب الاستسقاء

٢٥ حديثًا
١٥٠٤اللهم على رءوس الجبال والآكام وبطون الأودية ومنابت الشجر١٥٠٥عباد بن تميم يحدث أبي - أن عبد الله بن زيد الذي، أري النداء قال إن…١٥٠٦ومحمد بن المثنى، عن عبد الرحمن، عن سفيان، عن هشام بن إسحاق بن عبد…١٥٠٧أن رسول الله صلى الله عليه وسلم استسقى وعليه خميصة سوداء .١٥٠٨رسول الله صلى الله عليه وسلم في الاستسقاء فقال خرج رسول الله صلى الله…١٥٠٩أن عمه، حدثه أنه، خرج مع رسول الله صلى الله عليه وسلم يستسقي فحول…١٥١٠أن النبي صلى الله عليه وسلم استسقى وصلى ركعتين وقلب رداءه .١٥١١أنه سمع عباد بن تميم، يقول سمعت عبد الله بن زيد، يقول خرج رسول الله…١٥١٢أنه رأى رسول الله صلى الله عليه وسلم في الاستسقاء استقبل القبلة وقلب…١٥١٣قال كان رسول الله صلى الله عليه وسلم لا يرفع يديه في شىء من الدعاء…١٥١٤أنه رأى رسول الله صلى الله عليه وسلم عند أحجار الزيت يستسقي وهو مقنع…١٥١٥أخبرنا عيسى بن حماد، قال حدثنا الليث، عن سعيد، - وهو المقبري - عن…١٥١٦أخبرنا محمد بن بشار، قال حدثني أبو هشام المغيرة بن سلمة، قال حدثني…١٥١٧اللهم اسقنا اللهم اسقنا١٥١٨اللهم أغثنا اللهم أغثنا١٥١٩ويونس، عن ابن شهاب، قال أخبرني عباد بن تميم، أنه سمع عمه، وكان، من…١٥٢٠أن النبي صلى الله عليه وسلم خرج يستسقي فصلى ركعتين واستقبل القبلة .١٥٢١فقال ابن عباس ما منعه أن يسألني، خرج رسول الله صلى الله عليه وسلم…١٥٢٢أن النبي صلى الله عليه وسلم خرج فاستسقى فصلى ركعتين جهر فيهما…١٥٢٣اللهم اجعله صيبا نافعا١٥٢٤قال الله عز وجل ما أنعمت على عبادي من نعمة إلا أصبح فريق منهم بها…١٥٢٥ألم تسمعوا ماذا قال ربكم الليلة قال ما أنعمت على عبادي من نعمة إلا…١٥٢٦لو أمسك الله عز وجل المطر عن عباده خمس سنين ثم أرسله لأصبحت طائفة من…١٥٢٧اللهم حوالينا ولا علينا١٥٢٨اللهم حوالينا ولا علينا
