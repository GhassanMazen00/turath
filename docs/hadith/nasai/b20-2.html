كتاب قيام الليل وتطوع النهار

٢٢٠ حديثًا•صفحة ٢ من ٢
١٧٤٨أخبرنا محمد بن بشار، قال حدثنا عبد الرحمن، عن شعبة، عن ثابت البناني،…١٧٤٩قالت كان رسول الله صلى الله عليه وسلم يصلي إحدى عشرة ركعة فيما بين أن…١٧٥٠سبحان الملك القدوس١٧٥١سبحان الملك القدوس١٧٥٢سبحان الملك القدوس١٧٥٣سبحان الملك القدوس١٧٥٤سبحان الملك القدوس١٧٥٥أن النبي صلى الله عليه وسلم كان يوتر . وساق الحديث .١٧٥٦رسول الله صلى الله عليه وسلم من الليل فقالت كان يصلي ثلاث عشرة ركعة…١٧٥٧أن النبي صلى الله عليه وسلم كان لا يدع أربع ركعات قبل الظهر وركعتين…١٧٥٨أنه سمع أباه، يحدث أنه سمع عائشة، قالت كان رسول الله صلى الله عليه…١٧٥٩ركعتا الفجر خير من الدنيا وما فيها١٧٦٠أخبرنا قتيبة بن سعيد، قال حدثنا الليث، عن نافع، عن ابن عمر، عن حفصة،…١٧٦١قال أخبرتني حفصة، أن النبي صلى الله عليه وسلم كان إذا أضاء له الفجر…١٧٦٢قالت كان رسول الله صلى الله عليه وسلم إذا سكت المؤذن بالأولى من صلاة…١٧٦٣لا تكن مثل فلان كان يقوم الليل فترك قيام الليل١٧٦٤لا تكن يا عبد الله مثل فلان كان يقوم الليل فترك قيام الليل١٧٦٥أخبرنا محمد بن إبراهيم البصري، قال حدثنا خالد بن الحارث، قال قرأت على…١٧٦٦قال حدثتني حفصة، أن رسول الله صلى الله عليه وسلم كان يركع ركعتين…١٧٦٧قالت كان رسول الله صلى الله عليه وسلم يركع بين النداء والصلاة ركعتين…١٧٦٨أن النبي صلى الله عليه وسلم كان يصلي بين النداء والإقامة ركعتين…١٧٦٩أن ابن عمر، حدثه أن حفصة حدثته أن رسول الله صلى الله عليه وسلم كان…١٧٧٠قال أخبرتني حفصة، أن رسول الله صلى الله عليه وسلم كان يصلي قبل الصبح…١٧٧١أنها أخبرته أن رسول الله صلى الله عليه وسلم كان إذا نودي لصلاة الصبح…١٧٧٢أنها أخبرته أن رسول الله صلى الله عليه وسلم كان إذا سكت المؤذن صلى…١٧٧٣أن حفصة أم المؤمنين، أخبرته أن رسول الله صلى الله عليه وسلم كان إذا…١٧٧٤قال حدثتني أختي، حفصة أنه كان يصلي قبل الفجر ركعتين خفيفتين .١٧٧٥أن رسول الله صلى الله عليه وسلم كان يصلي ركعتين إذا طلع الفجر .١٧٧٦أنها قالت كان رسول الله صلى الله عليه وسلم إذا طلع الفجر لا يصلي إلا…١٧٧٧أخبرنا قتيبة بن سعيد، قال حدثنا الليث، عن نافع، عن ابن عمر، عن حفصة،…١٧٧٨قال ابن عمر أخبرتني حفصة، أن رسول الله صلى الله عليه وسلم كان يركع…١٧٧٩قال أخبرتني حفصة، أن رسول الله صلى الله عليه وسلم كان إذا أضاء له…١٧٨٠أن رسول الله صلى الله عليه وسلم كان يصلي ركعتين خفيفتين بين النداء…١٧٨١رسول الله صلى الله عليه وسلم بالليل قالت كان يصلي ثلاث عشرة ركعة يصلي…١٧٨٢قال كان النبي صلى الله عليه وسلم يصلي ركعتى الفجر إذا سمع الأذان…١٧٨٣ذاك رجل لا يتوسد القرآن١٧٨٤ما من امرئ تكون له صلاة بليل فغلبه عليها نوم إلا كتب الله له أجر…١٧٨٥من كانت له صلاة صلاها من الليل فنام عنها كان ذلك صدقة تصدق الله عز…١٧٨٦أن رسول الله صلى الله عليه وسلم قال فذكر نحوه . قال أبو عبد الرحمن…١٧٨٧من أتى فراشه وهو ينوي أن يقوم يصلي من الليل فغلبته عيناه حتى أصبح كتب…١٧٨٨وأبي الدرداء، موقوفا .١٧٨٩أن رسول الله صلى الله عليه وسلم كان إذا لم يصل من الليل منعه من ذلك…١٧٩٠من نام عن حزبه أو عن شىء منه فقرأه فيما بين صلاة الفجر وصلاة الظهر…١٧٩١من نام عن حزبه - أو قال جزئه - من الليل فقرأه فيما بين صلاة الصبح إلى…١٧٩٢أن عمر بن الخطاب، قال من فاته حزبه من الليل فقرأه حين تزول الشمس إلى…١٧٩٣قال من فاته ورده من الليل فليقرأه في صلاة قبل الظهر فإنها تعدل صلاة…١٧٩٤من ثابر على اثنتى عشرة ركعة في اليوم والليلة دخل الجنة أربعا قبل…١٧٩٥من ثابر على اثنتى عشرة ركعة بنى الله عز وجل له بيتا في الجنة أربعا…١٧٩٦من ركع ثنتى عشرة ركعة في يومه وليلته سوى المكتوبة بنى الله له بها…١٧٩٧من ركع اثنتى عشرة ركعة في اليوم والليلة سوى المكتوبة بنى الله عز وجل…١٧٩٨من صلى في يوم ثنتى عشرة ركعة بنى الله عز وجل له بيتا في الجنة١٧٩٩من صلى ثنتى عشرة ركعة بالنهار أو بالليل بنى الله عز وجل له بيتا في…١٨٠٠ومحمد بن مكي، قالا أنبأنا عبد الله، عن أبي يونس القشيري، عن ابن أبي…١٨٠١اثنتا عشرة ركعة من صلاهن بنى الله له بيتا في الجنة أربع ركعات قبل…١٨٠٢من صلى اثنتى عشرة ركعة بنى الله له بيتا في الجنة أربعا قبل الظهر…١٨٠٣قالت من صلى في اليوم والليلة ثنتى عشرة ركعة سوى المكتوبة بني له بيت…١٨٠٤من صلى في اليوم والليلة ثنتى عشرة ركعة بني له بيت في الجنة١٨٠٥قالت من صلى في الليل والنهار ثنتى عشرة ركعة سوى المكتوبة بني له بيت…١٨٠٦وحبان، قالا حدثنا عبد الله، عن إسماعيل، عن المسيب بن رافع، عن أم…١٨٠٧أن أم حبيبة، حدثته أنه، من صلى في يوم ثنتى عشرة ركعة بني له بيت في…١٨٠٨من صلى في يوم ثنتى عشرة ركعة سوى الفريضة بنى الله له أو بني له بيت في…١٨٠٩من صلى ثنتى عشرة ركعة في يوم وليلة بنى الله له بيتا في الجنة١٨١٠قالت من صلى في يوم اثنتى عشرة ركعة بني له بيت في الجنة .١٨١١من صلى في يوم ثنتى عشرة ركعة سوى الفريضة بنى الله له بيتا في الجنة١٨١٢من ركع أربع ركعات قبل الظهر وأربعا بعدها حرم الله عز وجل لحمه على…١٨١٣ما من عبد مؤمن يصلي أربع ركعات بعد الظهر فتمس وجهه النار أبدا إن شاء…١٨١٤من صلى أربع ركعات قبل الظهر وأربعا بعدها حرمه الله عز وجل على النار١٨١٥أخبرنا محمود بن خالد، عن مروان بن محمد، قال حدثنا سعيد بن عبد العزيز،…١٨١٦من حافظ على أربع ركعات قبل الظهر وأربع بعدها حرمه الله تعالى على النار١٨١٧من صلى أربعا قبل الظهر وأربعا بعدها لم تمسه النار
