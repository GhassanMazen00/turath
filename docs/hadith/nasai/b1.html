سنن النسائي

كتاب الطهارة

٣٢٤ حديثًا•صفحة ١ من ٣
١إذا استيقظ أحدكم من نومه فلا يغمس يده في وضوئه حتى يغسلها ثلاثا فإن…٢وقتيبة بن سعيد، عن جرير، عن منصور، عن أبي وائل، عن حذيفة، قال كان…٣قال دخلت على رسول الله صلى الله عليه وسلم وهو يستن وطرف السواك على…٤إنا لا - أو لن - نستعين على العمل من أراده ولكن اذهب أنت٥السواك مطهرة للفم مرضاة للرب٦قد أكثرت عليكم في السواك٧لولا أن أشق على أمتي لأمرتهم بالسواك عند كل صلاة٨قال قلت لعائشة بأى شىء كان يبدأ النبي صلى الله عليه وسلم إذا دخل بيته…٩الفطرة خمس الاختتان والاستحداد وقص الشارب وتقليم الأظفار ونتف الإبط١٠خمس من الفطرة قص الشارب ونتف الإبط وتقليم الأظفار والاستحداد والختان١١خمس من الفطرة الختان وحلق العانة ونتف الإبط وتقليم الأظفار وأخذ الشارب١٢الفطرة قص الأظفار وأخذ الشارب وحلق العانة١٣من لم يأخذ شاربه فليس منا١٤أخبرنا قتيبة، قال حدثنا جعفر، - هو ابن سليمان - عن أبي عمران الجوني،…١٥أحفوا الشوارب وأعفوا اللحى١٦وعمارة بن خزيمة بن ثابت، عن عبد الرحمن بن أبي قراد، قال خرجت مع رسول…١٧أن النبي صلى الله عليه وسلم كان إذا ذهب المذهب أبعد قال فذهب لحاجته -…١٨قال كنت أمشي مع رسول الله صلى الله عليه وسلم فانتهى إلى سباطة قوم…١٩اللهم إني أعوذ بك من الخبث والخبائث٢٠إذا ذهب أحدكم إلى الغائط أو البول فلا يستقبل القبلة ولا يستدبرها٢١لا تستقبلوا القبلة ولا تستدبروها لغائط أو بول ولكن شرقوا أو غربوا٢٢إذا أتى أحدكم الغائط فلا يستقبل القبلة ولكن ليشرق أو ليغرب٢٣واسع بن حبان، عن عبد الله بن عمر، قال لقد ارتقيت على ظهر بيتنا فرأيت…٢٤إذا بال أحدكم فلا يأخذ ذكره بيمينه٢٥إذا دخل أحدكم الخلاء فلا يمس ذكره بيمينه٢٦أن رسول الله صلى الله عليه وسلم أتى سباطة قوم فبال قائما .٢٧أن حذيفة، قال إن رسول الله صلى الله عليه وسلم أتى سباطة قوم فبال…٢٨ومنصور، عن أبي وائل، عن حذيفة، أن النبي صلى الله عليه وسلم مشى إلى…٢٩قالت من حدثكم أن رسول الله صلى الله عليه وسلم بال قائما فلا تصدقوه ما…٣٠أوما علمت ما أصاب صاحب بني إسرائيل كانوا إذا أصابهم شىء من البول…٣١إنهما يعذبان وما يعذبان في كبير أما هذا فكان لا يستنزه من بوله وأما…٣٢قال قال ابن جريج أخبرتني حكيمة بنت أميمة، عن أمها، أميمة بنت رقيقة…٣٣قالت يقولون إن النبي صلى الله عليه وسلم أوصى إلى علي لقد دعا بالطست…٣٤لا يبولن أحدكم في جحر٣٥أخبرنا قتيبة، قال حدثنا الليث، عن أبي الزبير، عن جابر، عن رسول الله…٣٦لا يبولن أحدكم في مستحمه فإن عامة الوسواس منه٣٧وقبيصة، قالا أنبأنا سفيان، عن الضحاك بن عثمان، عن نافع، عن ابن عمر،…٣٨أنه سلم على النبي صلى الله عليه وسلم وهو يبول فلم يرد عليه السلام حتى…٣٩أن رسول الله صلى الله عليه وسلم نهى أن يستطيب أحدكم بعظم أو روث .٤٠إنما أنا لكم مثل الوالد أعلمكم إذا ذهب أحدكم إلى الخلاء فلا يستقبل…٤١أخبرنا إسحاق بن إبراهيم، قال أنبأنا أبو معاوية، قال حدثنا الأعمش، عن…٤٢أنه سمع عبد الله، يقول أتى النبي صلى الله عليه وسلم الغائط وأمرني أن…٤٣إذا استجمرت فأوتر٤٤إذا ذهب أحدكم إلى الغائط فليذهب معه بثلاثة أحجار فليستطب بها فإنها…٤٥أخبرنا إسحاق بن إبراهيم، قال أنبأنا النضر، قال أنبأنا شعبة، عن عطاء…٤٦أنها قالت مرن أزواجكن أن يستطيبوا بالماء فإني أستحييهم منه إن رسول…٤٧إذا شرب أحدكم فلا يتنفس في إنائه وإذا أتى الخلاء فلا يمس ذكره بيمينه…٤٨أن النبي صلى الله عليه وسلم نهى أن يتنفس في الإناء وأن يمس ذكره…٤٩لا يستنجي أحدكم بدون ثلاثة أحجار٥٠أن النبي صلى الله عليه وسلم توضأ فلما استنجى دلك يده بالأرض .٥١يا جرير هات طهورا٥٢إذا كان الماء قلتين لم يحمل الخبث٥٣أن أعرابيا، بال في المسجد فقام إليه بعض القوم فقال رسول الله صلى الله…٥٤قال بال أعرابي في المسجد فأمر النبي صلى الله عليه وسلم بدلو من ماء…٥٥يقول جاء أعرابي إلى المسجد فبال فصاح به الناس فقال رسول الله صلى الله…٥٦دعوه وأهريقوا على بوله دلوا من ماء فإنما بعثتم ميسرين ولم تبعثوا…٥٧لا يبولن أحدكم في الماء الدائم ثم يتوضأ منه٥٨لا يبولن أحدكم في الماء الدائم ثم يغتسل منه٥٩هو الطهور ماؤه الحل ميتته٦٠أقول اللهم باعد بيني وبين خطاياى كما باعدت بين المشرق والمغرب اللهم…٦١اللهم اغسل خطاياى بماء الثلج والبرد ونق قلبي من الخطايا كما نقيت…٦٢اللهم اغفر له وارحمه وعافه واعف عنه وأكرم نزله وأوسع مدخله واغسله…٦٣إذا شرب الكلب في إناء أحدكم فليغسله سبع مرات٦٤إذا ولغ الكلب في إناء أحدكم فليغسله سبع مرات٦٥أنه أخبره هلال بن أسامة، أنه سمع أبا سلمة، يخبر عن أبي هريرة، عن…٦٦إذا ولغ الكلب في إناء أحدكم فليرقه ثم ليغسله سبع مرات٦٧إذا ولغ الكلب في الإناء فاغسلوه سبع مرات وعفروه الثامنة بالتراب٦٨إنها ليست بنجس إنما هي من الطوافين عليكم والطوافات٦٩أخبرنا محمد بن عبد الله بن يزيد، قال حدثنا سفيان، عن أيوب، عن محمد،…٧٠أخبرنا عمرو بن علي، قال حدثنا عبد الرحمن، عن سفيان، عن المقدام بن…٧١قال كان الرجال والنساء يتوضئون في زمان رسول الله صلى الله عليه وسلم…٧٢أنها أخبرته أنها، كانت تغتسل مع رسول الله صلى الله عليه وسلم في…٧٣يقول كان رسول الله صلى الله عليه وسلم يتوضأ بمكوك ويغتسل بخمسة مكاكي .٧٤وهي أم عمارة بنت كعب أن النبي صلى الله عليه وسلم توضأ فأتي بماء في…٧٥إنما الأعمال بالنية وإنما لامرئ ما نوى فمن كانت هجرته إلى الله وإلى…٧٦قال رأيت رسول الله صلى الله عليه وسلم وحانت صلاة العصر فالتمس الناس…٧٧حى على الطهور والبركة من الله عز وجل٧٨هل مع أحد منكم ماء٧٩والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن وهب، عن…٨٠قال ألا أخبركم بوضوء، رسول الله صلى الله عليه وسلم فتوضأ مرة مرة .٨١أن عبد الله بن عمر، توضأ ثلاثا ثلاثا يسند ذلك إلى النبي صلى الله عليه…٨٢أخبرنا محمد بن إبراهيم البصري، عن بشر بن المفضل، عن ابن عون، عن عامر…٨٣قال رأيت رسول الله صلى الله عليه وسلم استوكف ثلاثا .٨٤من توضأ نحو وضوئي هذا ثم صلى ركعتين لا يحدث نفسه فيهما بشىء غفر له ما…٨٥من توضأ مثل وضوئي هذا ثم قام فصلى ركعتين لا يحدث فيهما نفسه بشىء غفر…٨٦إذا توضأ أحدكم فليجعل في أنفه ماء ثم ليستنثر٨٧أسبغ الوضوء وبالغ في الاستنشاق إلا أن تكون صائما٨٨من توضأ فليستنثر ومن استجمر فليوتر٨٩إذا توضأت فاستنثر وإذا استجمرت فأوتر٩٠إذا استيقظ أحدكم من منامه فتوضأ فليستنثر ثلاث مرات فإن الشيطان يبيت…٩١أنه دعا بوضوء فتمضمض واستنشق ونثر بيده اليسرى ففعل هذا ثلاثا ثم قال…٩٢أخبرنا قتيبة، قال حدثنا أبو عوانة، عن خالد بن علقمة، عن عبد خير، قال…٩٣رضى الله عنه أنه أتي بكرسي فقعد عليه ثم دعا بتور فيه ماء فكفأ على…٩٤وحميد بن مسعدة، عن يزيد، - وهو ابن زريع - قال حدثني شعبة، عن مالك بن…٩٥أن محمد بن علي، أخبره قال أخبرني أبي علي، أن الحسين بن علي، قال دعاني…٩٦- وهو ابن قيس - قال رأيت عليا - رضى الله عنه - توضأ فغسل كفيه حتى…٩٧والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن القاسم،…٩٨أنه قال لعبد الله بن زيد بن عاصم - وهو جد عمرو بن يحيى - هل تستطيع أن…٩٩- الذي أري النداء - قال رأيت رسول الله صلى الله عليه وسلم توضأ فغسل…١٠٠سالم سبلان قال وكانت عائشة تستعجب بأمانته وتستأجره فأرتني كيف كان…١٠١أخبرنا الهيثم بن أيوب الطالقاني، قال حدثنا عبد العزيز بن محمد، قال…١٠٢قال توضأ رسول الله صلى الله عليه وسلم فغرف غرفة فمضمض واستنشق ثم غرف…١٠٣إذا توضأ العبد المؤمن فتمضمض خرجت الخطايا من فيه فإذا استنثر خرجت…١٠٤قال رأيت النبي صلى الله عليه وسلم يمسح على الخفين والخمار .١٠٥وحفص بن غياث، عن الأعمش، عن الحكم، عن عبد الرحمن بن أبي ليلى، عن…١٠٦قال رأيت رسول الله صلى الله عليه وسلم يمسح على الخمار والخفين .١٠٧أخبرنا عمرو بن علي، قال حدثني يحيى بن سعيد، قال حدثنا سليمان التيمي،…١٠٨وحميد بن مسعدة، عن يزيد، - وهو ابن زريع - قال حدثنا حميد، قال حدثنا…١٠٩أخبرنا يعقوب بن إبراهيم، قال حدثنا هشيم، قال أخبرنا يونس بن عبيد، عن…١١٠ويل للعقب من النار١١١ويل للأعقاب من النار أسبغوا الوضوء١١٢أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا شعبة، قال أخبرني…١١٣أنه كان مع رسول الله صلى الله عليه وسلم في سفر فأتي بماء فقال على…١١٤إذا توضأت فأسبغ الوضوء وخلل بين الأصابع١١٥قال رأيت عليا توضأ فغسل كفيه ثلاثا وتمضمض واستنشق ثلاثا وغسل وجهه…١١٦من توضأ نحو وضوئي هذا ثم قام فركع ركعتين لا يحدث فيهما نفسه غفر له ما…١١٧ومالك، وابن، جريج عن المقبري، عن عبيد بن جريج، قال قلت لابن عمر رأيتك…١١٨أنه توضأ ومسح على خفيه فقيل له أتمسح فقال قد رأيت رسول الله صلى الله…١١٩أنه رأى رسول الله صلى الله عليه وسلم توضأ ومسح على الخفين .١٢٠قال دخل رسول الله صلى الله عليه وسلم وبلال الأسواق فذهب لحاجته ثم خرج…١٢١والحارث بن مسكين، قراءة عليه وأنا أسمع، - واللفظ له - عن ابن وهب، عن…١٢٢أخبرنا قتيبة، قال حدثنا إسماعيل، - وهو ابن جعفر - عن موسى بن عقبة، عن…١٢٣أخبرنا علي بن خشرم، قال حدثنا عيسى، عن الأعمش، عن مسلم، عن مسروق، عن…١٢٤أخبرنا قتيبة بن سعيد، قال حدثنا الليث بن سعد، عن يحيى، عن سعد بن…١٢٥حديث ١٢٥١٢٦أخبرنا قتيبة، قال حدثنا سفيان، عن عاصم، عن زر، عن صفوان بن عسال، قال…١٢٧ومالك بن مغول، وزهير، وأبو بكر بن عياش وسفيان بن عيينة عن عاصم، عن…١٢٨- رضى الله عنه - قال جعل رسول الله صلى الله عليه وسلم للمسافر ثلاثة…١٢٩أخبرنا هناد بن السري، عن أبي معاوية، عن الأعمش، عن الحكم، عن القاسم…١٣٠قال رأيت عليا - رضى الله عنه - صلى الظهر ثم قعد لحوائج الناس فلما…١٣١أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا شعبة، عن عمرو بن…١٣٢إنما أمرت بالوضوء إذا قمت إلى الصلاة١٣٣عمدا فعلته يا عمر١٣٤أن رسول الله صلى الله عليه وسلم كان إذا توضأ أخذ حفنة من ماء فقال بها…١٣٥��ن منصور، ح وأنبأنا أحمد بن حرب، قال حدثنا قاسم، - وهو ابن يزيد…١٣٦قال رأيت عليا - رضى الله عنه - توضأ ثلاثا ثلاثا ثم قام فشرب فضل وضوئه…١٣٧قال شهدت النبي صلى الله عليه وسلم بالبطحاء وأخرج بلال فضل وضوئه…١٣٨يقول مرضت فأتاني رسول الله صلى الله عليه وسلم وأبو بكر يعوداني…١٣٩لا يقبل الله صلاة بغير طهور ولا صدقة من غلول١٤٠هكذا الوضوء فمن زاد على هذا فقد أساء وتعدى وظلم١٤١أخبرنا يحيى بن حبيب بن عربي، قال حدثنا حماد، قال حدثنا أبو جهضم، قال…١٤٢قال قال رسول الله صلى الله عليه وسلم " أسبغوا الوضوء " .١٤٣ألا أخبركم بما يمحو الله به الخطايا ويرفع به الدرجات إسباغ الوضوء على…١٤٤من توضأ كما أمر وصلى كما أمر غفر له ما قدم من عمل١٤٥من أتم الوضوء كما أمره الله عز وجل فالصلوات الخمس كفارات لما بينهن١٤٦ما من امرئ يتوضأ فيحسن وضوءه ثم يصلي الصلاة إلا غفر له ما بينه وبين…١٤٧أما الوضوء فإنك إذا توضأت فغسلت كفيك فأنقيتهما خرجت خطاياك من بين…١٤٨من توضأ فأحسن الوضوء ثم قال أشهد أن لا إله إلا الله وأشهد أن محمدا…١٤٩تبلغ حلية المؤمن حيث يبلغ الوضوء١٥٠السلام عليكم دار قوم مؤمنين وإنا إن شاء الله بكم لاحقون
