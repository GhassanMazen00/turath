سنن النسائي

كتاب القسامة

١٦٤ حديثًا•صفحة ١ من ٢
٤٧٠٦أخبرنا محمد بن يحيى، قال حدثنا أبو معمر، قال حدثنا عبد الوارث، قال…٤٧٠٧ويونس بن عبد الأعلى، قال أنبأنا ابن وهب، قال أخبرني يونس، عن ابن…٤٧٠٨وسليمان بن يسار، عن أناس، من أصحاب رسول الله صلى الله عليه وسلم أن…٤٧٠٩أخبرنا محمد بن رافع، قال حدثنا عبد الرزاق، قال أنبأنا معمر، عن…٤٧١٠أن سهل بن أبي حثمة، أخبره أن عبد الله بن سهل ومحيصة خرجا إلى خيبر من…٤٧١١أنه أخبره ورجال، من كبراء قومه أن عبد، الله بن سهل ومحيصة خرجا إلى…٤٧١٢كبر الكبر في السن٤٧١٣الكبر ليبدأ الأكبر٤٧١٤أن عبد الله بن سهل، ومحيصة بن مسعود بن زيد، أنهما أتيا خيبر وهو يومئذ…٤٧١٥أخبرنا إسماعيل بن مسعود، قال حدثنا بشر بن المفضل، قال حدثنا يحيى بن…٤٧١٦أن عبد الله بن سهل الأنصاري، ومحيصة بن مسعود، خرجا إلى خيبر فتفرقا في…٤٧١٧أخبرنا محمد بن منصور، قال حدثنا سفيان، قال حدثنا يحيى بن سعيد، عن…٤٧١٨قال الحارث بن مسكين قراءة عليه وأنا أسمع، عن ابن القاسم، حدثني مالك،…٤٧١٩زعم أن رجلا، من الأنصار يقال له سهل بن أبي حثمة أخبره أن نفرا من قومه…٤٧٢٠أقم شاهدين على من قتله أدفعه إليكم برمته٤٧٢١لا يحل دم امرئ مسلم إلا بإحدى ثلاث النفس بالنفس والثيب الزاني والتارك…٤٧٢٢أما إنه إن كان صادقا ثم قتلته دخلت النار٤٧٢٣قال جيء بالقاتل الذي قتل إلى رسول الله صلى الله عليه وسلم جاء به ولي…٤٧٢٤أخبرنا محمد بن بشار، قال حدثنا يحيى بن سعيد، عن عوف بن أبي جميلة، قال…٤٧٢٥أخبرنا محمد بن بشار، قال حدثنا يحيى، قال حدثنا جامع بن مطر الحبطي، عن…٤٧٢٦أخبرنا عمرو بن منصور، قال حدثنا حفص بن عمر، - وهو الحوضي - قال حدثنا…٤٧٢٧ذكر أن علقمة بن وائل، أخبره عن أبيه، أنه كان قاعدا عند رسول الله صلى…٤٧٢٨أن علقمة بن وائل، حدثه أن أباه حدثه قال إني لقاعد مع رسول الله صلى…٤٧٢٩القاتل والمقتول في النار٤٧٣٠أن رجلا، أتى بقاتل وليه رسول الله صلى الله عليه وسلم فقال النبي صلى…٤٧٣١اذهب فاقتله كما قتل أخاك٤٧٣٢أخبرنا القاسم بن زكريا بن دينار، قال حدثنا عبيد الله بن موسى، قال…٤٧٣٣أن الآيات التي، في المائدة التي قالها الله عز وجل { فاحكم بينهم أو…٤٧٣٤المؤمنون تكافأ دماؤهم وهم يد على من سواهم ويسعى بذمتهم أدناهم ألا لا…٤٧٣٥المؤمنون تكافأ دماؤهم وهم يد على من سواهم يسعى بذمتهم أدناهم لا يقتل…٤٧٣٦من قتل عبده قتلناه ومن جدعه جدعناه ومن أخصاه أخصيناه٤٧٣٧من قتل عبده قتلناه ومن جدع عبده جدعناه٤٧٣٨من قتل عبده قتلناه ومن جدع عبده جدعناه٤٧٣٩أنه سمع طاوسا، يحدث عن ابن عباس، عن عمر، رضى الله عنه أنه نشد قضاء…٤٧٤٠رضى الله عنه أن يهوديا، قتل جارية على أوضاح لها فأقاده رسول الله صلى…٤٧٤١أن يهوديا، أخذ أوضاحا من جارية ثم رضخ رأسها بين حجرين فأدركوها وبها…٤٧٤٢قال خرجت جارية عليها أوضاح فأخذها يهودي فرضخ رأسها وأخذ ما عليها من…٤٧٤٣لا يحل قتل مسلم إلا في إحدى ثلاث خصال زان محصن فيرجم ورجل يقتل مسلما…٤٧٤٤العقل وفكاك الأسير وأن لا يقتل مسلم بكافر٤٧٤٥المؤمنون تكافأ دماؤهم يسعى بذمتهم أدناهم وهم يد على من سواهم لا يقتل…٤٧٤٦المؤمنون تتكافأ دماؤهم يسعى بذمتهم أدناهم لا يقتل مؤمن بكافر ولا ذو…٤٧٤٧من قتل معاهدا في غير كنهه حرم الله عليه الجنة٤٧٤٨من قتل نفسا معاهدة بغير حلها حرم الله عليه الجنة أن يشم ريحها٤٧٤٩من قتل رجلا من أهل الذمة لم يجد ريح الجنة وإن ريحها ليوجد من مسيرة…٤٧٥٠من قتل قتيلا من أهل الذمة لم يجد ريح الجنة وإن ريحها ليوجد من مسيرة…٤٧٥١أن غلاما، لأناس فقراء قطع أذن غلام لأناس أغنياء فأتوا النبي صلى الله…٤٧٥٢كتاب الله القصاص٤٧٥٣من قتل عبده قتلناه ومن جدع عبده جدعناه٤٧٥٤من خصى عبده خصيناه ومن جدع عبده جدعناه٤٧٥٥أن أخت الربيع أم حارثة، جرحت إنسانا فاختصموا إلى النبي صلى الله عليه…٤٧٥٦إن من عباد الله من لو أقسم على الله لأبره٤٧٥٧يا أنس كتاب الله القصاص٤٧٥٨ما تأمرني تأمرني أن آمره أن يدع يده في فيك تقضمها كما يقضم الفحل إن…٤٧٥٩أردت أن تقضم لحم أخيك كما يقضم الفحل٤٧٦٠يعض أحدكم أخاه كما يعض الفحل لا دية له٤٧٦١أن يعلى، قال في الذي عض فندرت ثنيته أن النبي صلى الله عليه وسلم قال "…٤٧٦٢أردت أن تقضم ذراع أخيك كما يقضم الفحل٤٧٦٣يعض أحدكم أخاه كما يعض البكر٤٧٦٤يعض أحدكم أخاه كما يعض البكر٤٧٦٥ينطلق أحدكم إلى أخيه فيعضه كعضيض الفحل ثم يأتي يطلب العقل لا عقل لها٤٧٦٦أن رجلا، عض يد رجل فانتزعت ثنيته فأتى النبي صلى الله عليه وسلم…٤٧٦٧أيدعها يقضمها كقضم الفحل٤٧٦٨قال غزوت مع رسول الله صلى الله عليه وسلم في غزوة تبوك فاستأجرت أجيرا…٤٧٦٩أفيدع يده في فيك تقضمها٤٧٧٠بمثل الذي عض فندرت ثنيته أن النبي صلى الله عليه وسلم قال " لا دية لك…٤٧٧١أيدعها في فيك تقضمها كقضم الفحل٤٧٧٢يعمد أحدكم فيعض أخاه كما يعض الفحل٤٧٧٣أخبرنا وهب بن بيان، قال حدثنا ابن وهب، قال أخبرني عمرو بن الحارث، عن…٤٧٧٤أخبرنا أحمد بن سعيد الرباطي، قال حدثنا وهب بن جرير، أنبأنا أبي قال،…٤٧٧٥أيها الناس أى أهل الأرض تعلمون أكرم على الله عز وجل٤٧٧٦لا وأستغفر الله لا أحمل لك حتى تقيدني مما جبذت برقبتي٤٧٧٧أن عمر، قال رأيت رسول الله صلى الله عليه وسلم يقص من نفسه .٤٧٧٨أن النبي صلى الله عليه وسلم بعث أبا جهم بن حذيفة مصدقا فلاحه رجل في…٤٧٧٩أن يهوديا، رأى على جارية أوضاحا فقتلها بحجر فأتي بها النبي صلى الله…٤٧٨٠إني بريء من كل مسلم مع مشرك٤٧٨١قال كان في بني إسرائيل القصاص ولم تكن فيهم الدية فأنزل الله عز وجل {…٤٧٨٢قال { كتب عليكم القصاص في القتلى الحر بالحر } قال كان بنو إسرائيل…٤٧٨٣قال أتي رسول الله صلى الله عليه وسلم في قصاص فأمر فيه بالعفو .٤٧٨٤وبهز بن أسد، وعفان بن مسلم، قالوا حدثنا عبد الله بن بكر المزني، قال…٤٧٨٥من قتل له قتيل فهو بخير النظرين إما أن يقاد وإما أن يفدى٤٧٨٦من قتل له قتيل فهو بخير النظرين إما أن يقاد وإما أن يفدى٤٧٨٧أن رسول الله صلى الله عليه وسلم قال " من قتل له قتيل " . مرسل .٤٧٨٨وعلى المقتتلين أن ينحجزوا الأول فالأول وإن كانت امرأة٤٧٨٩من قتل في عميا أو رميا تكون بينهم بحجر أو سوط أو بعصا فعقله عقل خطإ…٤٧٩٠من قتل في عمية أو رمية بحجر أو سوط أو عصا فعقله عقل الخطإ ومن قتل…٤٧٩١قتيل الخطإ شبه العمد بالسوط أو العصا مائة من الإبل أربعون منها في…٤٧٩٢أن رسول الله صلى الله عليه وسلم خطب يوم الفتح . مرسل .٤٧٩٣ألا وإن قتيل الخطإ شبه العمد ما كان بالسوط والعصا مائة من الإبل…٤٧٩٤ألا وإن قتيل الخطإ شبه العمد بالسوط والعصا والحجر مائة من الإبل فيها…٤٧٩٥ألا إن قتيل الخطإ قتيل السوط والعصا فيه مائة من الإبل مغلظة أربعون…٤٧٩٦ألا وإن كل قتيل خطإ العمد أو شبه العمد قتيل السوط والعصا منها أربعون…٤٧٩٧ألا وإن قتيل الخطإ العمد قتيل السوط والعصا منها أربعون في بطونها…٤٧٩٨ألا وإن قتيل الخطإ العمد قتيل السوط والعصا منها أربعون في بطونها…٤٧٩٩الحمد لله الذي صدق وعده ونصر عبده وهزم الأحزاب وحده ألا إن قتيل العمد…٤٨٠٠الخطأ شبه العمد - يعني بالعصا والسوط - مائة من الإبل منها أربعون في…٤٨٠١من قتل خطأ فديته مائة من الإبل ثلاثون بنت مخاض وثلاثون بنت لبون…٤٨٠٢يقول قضى رسول الله صلى الله عليه وسلم دية الخطإ عشرين بنت مخاض وعشرين…٤٨٠٣قال قتل رجل رجلا على عهد رسول الله صلى الله عليه وسلم فجعل النبي صلى…٤٨٠٤سمعناه مرة، يقول عن ابن عباس، أن النبي صلى الله عليه وسلم قضى باثنى…٤٨٠٥عقل المرأة مثل عقل الرجل حتى يبلغ الثلث من ديتها٤٨٠٦عقل أهل الذمة نصف عقل المسلمين٤٨٠٧عقل الكافر نصف عقل المؤمن٤٨٠٨قال قضى رسول الله صلى الله عليه وسلم في المكاتب يقتل بدية الحر على…٤٨٠٩أن نبي الله صلى الله عليه وسلم قضى في المكاتب أن يودى بقدر ما عتق منه…٤٨١٠قال قضى رسول الله صلى الله عليه وسلم في المكاتب يودى بقدر ما أدى من…٤٨١١المكاتب يعتق بقدر ما أدى ويقام عليه الحد بقدر ما عتق منه ويرث بقدر ما…٤٨١٢أن مكاتبا، قتل على عهد رسول الله صلى الله عليه وسلم فأمر أن يودى ما…٤٨١٣وإبراهيم بن يونس بن محمد، قالا حدثنا عبيد الله بن موسى، قال حدثنا…٤٨١٤أن امرأة، خذفت امرأة فأسقطت المخذوفة فرفع ذلك إلى النبي صلى الله عليه…٤٨١٥أخبرنا أحمد بن سليمان، قال حدثنا يزيد، قال أنبأنا كهمس، عن عبد الله…٤٨١٦أن عمر، استشار الناس في الجنين فقال حمل بن مالك قضى رسول الله صلى…٤٨١٧قال قضى رسول الله صلى الله عليه وسلم في جنين امرأة من بني لحيان سقط…٤٨١٨إنما هذا من إخوان الكهان٤٨١٩أن امرأتين، من هذيل في زمان رسول الله صلى الله عليه وسلم رمت إحداهما…٤٨٢٠إنما هذا من الكهان٤٨٢١أسجع كسجع الأعراب٤٨٢٢أسجع كسجع الأعراب٤٨٢٣سجع كسجع الجاهلية٤٨٢٤قال ضربت امرأة من بني لحيان ضرتها بعمود الفسطاط فقتلتها وكان…٤٨٢٥أسجع كسجع الأعراب٤٨٢٦أسجع كسجع الأعراب٤٨٢٧أسجع كسجع الأعراب هو ما أقول لكم٤٨٢٨أسجع كسجع الجاهلية وكهانتها إن في الصبي غرة٤٨٢٩أنه سمع جابرا، يقول كتب رسول الله صلى الله عليه وسلم على كل بطن عقولة…٤٨٣٠من تطبب ولم يعلم منه طب قبل ذلك فهو ضامن٤٨٣١مثله سواء .٤٨٣٢قال أتيت النبي صلى الله عليه وسلم مع أبي فقال " من هذا معك " . قال…٤٨٣٣ألا لا تجني نفس على الأخرى٤٨٣٤لا تجني نفس على أخرى٤٨٣٥لا تجني نفس على أخرى٤٨٣٦لا تجني نفس على أخرى٤٨٣٧لا - يعني - لا تجني نفس على نفس٤٨٣٨لا تجني نفس على أخرى٤٨٣٩لا تجني أم على ولد٤٨٤٠أن رسول الله صلى الله عليه وسلم قضى في العين العوراء السادة لمكانها…٤٨٤١في الأسنان خمس من الإبل٤٨٤٢الأسنان سواء خمسا خمسا٤٨٤٣في الأصابع عشر عشر٤٨٤٤الأصابع سواء عشرا٤٨٤٥قال قضى رسول الله صلى الله عليه وسلم أن الأصابع سواء عشرا عشرا من…٤٨٤٦وفيما هنالك من الأصابع عشرا عشرا٤٨٤٧أخبرنا عمرو بن علي، قال حدثنا يحيى بن سعيد، قال حدثنا شعبة، قال حدثني…٤٨٤٨فهذه وهذه سواء الإبهام والخنصر .٤٨٤٩قال الأصابع عشر عشر .٤٨٥٠وفي الأصابع عشر عشر٤٨٥١وابن، جريج عن عمرو بن شعيب، عن أبيه، عن جده، أن النبي صلى الله عليه…٤٨٥٢وفي المواضح خمس خمس٤٨٥٣من محمد النبي صلى الله عليه وسلم إلى شرحبيل بن عبد كلال ونعيم بن عبد…٤٨٥٤وفي العين الواحدة نصف الدية وفي اليد الواحدة نصف الدية وفي الرجل…٤٨٥٥هذا بيان من الله ورسوله { يا أيها الذين آمنوا أوفوا بالعقود }
