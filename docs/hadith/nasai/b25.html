سنن النسائي

كتاب الجهاد

١١١ حديثًا
٣٠٨٥أخبرنا عبد الرحمن بن محمد بن سلام، قال حدثنا إسحاق الأزرق، قال حدثنا…٣٠٨٦إني أمرت بالعفو فلا تقاتلوا٣٠٨٧بعثت بجوامع الكلم ونصرت بالرعب وبينا أنا نائم أتيت بمفاتيح خزائن…٣٠٨٨أخبرنا هارون بن سعيد، عن خالد بن نزار، قال أخبرني القاسم بن مبرور، عن…٣٠٨٩بعثت بجوامع الكلم ونصرت بالرعب وبينا أنا نائم أتيت بمفاتيح خزائن…٣٠٩٠أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…٣٠٩١أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…٣٠٩٢أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…٣٠٩٣أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوها عصموا مني…٣٠٩٤أمرت أن أقاتل الناس حتى يشهدوا أن لا إله إلا الله وأني رسول الله…٣٠٩٥أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قالها فقد عصم مني…٣٠٩٦جاهدوا المشركين بأموالكم وأيديكم وألسنتكم٣٠٩٧من مات ولم يغز ولم يحدث نفسه بغزو مات على شعبة نفاق٣٠٩٨والذي نفسي بيده لولا أن رجالا من المؤمنين لا تطيب أنفسهم أن يتخلفوا…٣٠٩٩أخبرنا محمد بن عبد الله بن بزيع، قال حدثنا بشر، - يعني ابن المفضل -…٣١٠٠أخبرنا محمد بن يحيى بن عبد الله، قال حدثنا يعقوب بن إبراهيم بن سعد،…٣١٠١ائتوني بالكتف واللوح٣١٠٢أخبرنا محمد بن عبيد، قال حدثنا أبو بكر بن عياش، عن أبي إسحاق، عن…٣١٠٣وشعبة، قالا حدثنا حبيب بن أبي ثابت، عن أبي العباس، عن عبد الله بن…٣١٠٤أن جاهمة، جاء إلى النبي صلى الله عليه وسلم فقال يا رسول الله أردت أن…٣١٠٥من جاهد بنفسه وماله في سبيل الله٣١٠٦ألا أخبركم بخير الناس وشر الناس إن من خير الناس رجلا عمل في سبيل الله…٣١٠٧قال لا يبكي أحد من خشية الله فتطعمه النار حتى يرد اللبن في الضرع ولا…٣١٠٨لا يلج النار رجل بكى من خشية الله تعالى حتى يعود اللبن في الضرع ولا…٣١٠٩لا يجتمعان في النار مسلم قتل كافرا ثم سدد وقارب ولا يجتمعان في جوف…٣١١٠لا يجتمع غبار في سبيل الله ودخان جهنم في جوف عبد أبدا ولا يجتمع الشح…٣١١١لا يجتمع غبار في سبيل الله ودخان جهنم في وجه رجل أبدا ولا يجتمع الشح…٣١١٢لا يجتمع غبار في سبيل الله ودخان جهنم في جوف عبد ولا يجتمع الشح…٣١١٣لا يجتمع غبار في سبيل الله عز وجل ودخان جهنم في منخرى مسلم أبدا٣١١٤لا يجتمع غبار في سبيل الله ودخان جهنم في منخرى مسلم ولا يجتمع شح…٣١١٥أنه سمع أبا هريرة، يقول لا يجمع الله عز وجل غبارا في سبيل الله ودخان…٣١١٦من اغبرت قدماه في سبيل الله فهو حرام على النار٣١١٧حرمت عين على النار سهرت في سبيل الله٣١١٨الغدوة والروحة في سبيل الله عز وجل أفضل من الدنيا وما فيها٣١١٩غدوة في سبيل الله أو روحة خير مما طلعت عليه الشمس وغربت٣١٢٠ثلاثة كلهم حق على الله عز وجل عونه المجاهد في سبيل الله والناكح الذي…٣١٢١وفد الله عز وجل ثلاثة الغازي والحاج والمعتمر٣١٢٢تكفل الله عز وجل لمن جاهد في سبيله لا يخرجه إلا الجهاد في سبيله…٣١٢٣انتدب الله عز وجل لمن يخرج في سبيله لا يخرجه إلا الإيمان بي والجهاد…٣١٢٤مثل المجاهد في سبيل الله - والله أعلم بمن يجاهد في سبيل الله - كمثل…٣١٢٥ما من غازية تغزو في سبيل الله فيصيبون غنيمة إلا تعجلوا ثلثى أجرهم من…٣١٢٦أيما عبد من عبادي خرج مجاهدا في سبيل الله ابتغاء مرضاتي ضمنت له أن…٣١٢٧مثل المجاهد في سبيل الله - والله أعلم بمن يجاهد في سبيله - كمثل…٣١٢٨لا أجده هل تستطيع إذا خرج المجاهد تدخل مسجدا فتقوم لا تفتر وتصوم لا…٣١٢٩إيمان بالله وجهاد في سبيل الله عز وجل٣١٣٠قال سأل رجل رسول الله صلى الله عليه وسلم أى الأعمال أفضل قال " إيمان…٣١٣١يا أبا سعيد من رضي بالله ربا وبالإسلام دينا وبمحمد نبيا وجبت له الجنة٣١٣٢من أقام الصلاة وآتى الزكاة ومات لا يشرك بالله شيئا كان حقا على الله…٣١٣٣أنا زعيم - والزعيم الحميل - لمن آمن بي وأسلم وهاجر ببيت في ربض الجنة…٣١٣٤إن الشيطان قعد لابن آدم بأطرقه فقعد له بطريق الإسلام فقال تسلم وتذر…٣١٣٥من أنفق زوجين في سبيل الله نودي في الجنة يا عبد الله هذا خير فمن كان…٣١٣٦من قاتل لتكون كلمة الله هي العليا فهو في سبيل الله عز وجل٣١٣٧أول الناس يقضى لهم يوم القيامة ثلاثة رجل استشهد فأتي به فعرفه نعمه…٣١٣٨من غزا في سبيل الله ولم ينو إلا عقالا فله ما نوى٣١٣٩من غزا وهو لا يريد إلا عقالا فله ما نوى٣١٤٠قال جاء رجل إلى النبي صلى الله عليه وسلم فقال أرأيت رجلا غزا يلتمس…٣١٤١من قاتل في سبيل الله عز وجل من رجل مسلم فواق ناقة وجبت له الجنة ومن…٣١٤٢من شاب شيبة في سبيل الله تعالى كانت له نورا يوم القيامة ومن رمى بسهم…٣١٤٣من بلغ بسهم في سبيل الله فهو له درجة في الجنة٣١٤٤من شاب شيبة في الإسلام في سبيل الله كانت له نورا يوم القيامة٣١٤٥من رمى بسهم في سبيل الله فبلغ العدو أخطأ أو أصاب كان له كعدل رقبة ومن…٣١٤٦إن الله عز وجل يدخل ثلاثة نفر الجنة بالسهم الواحد صانعه يحتسب في صنعه…٣١٤٧لا يكلم أحد في سبيل الله - والله أعلم بمن يكلم في سبيله - إلا جاء يوم…٣١٤٨زملوهم بدمائهم فإنه ليس كلم يكلم في الله إلا أتى يوم القيامة جرحه…٣١٤٩وذكر، آخر قبله عن عمارة بن غزية، عن أبي الزبير، عن جابر بن عبد الله،…٣١٥٠وعبد الله، ابنا كعب بن مالك أن سلمة بن الأكوع، قال لما كان يوم خيبر…٣١٥١لولا أن أشق على أمتي لم أتخلف عن سرية ولكن لا يجدون حمولة ولا أجد ما…٣١٥٢والذي نفسي بيده لولا أن رجالا من المؤمنين لا تطيب أنفسهم بأن يتخلفوا…٣١٥٣ما من الناس من نفس مسلمة يقبضها ربها تحب أن ترجع إليكم وأن لها الدنيا…٣١٥٤أخبرنا محمد بن منصور، قال حدثنا سفيان، عن عمرو، قال سمعت جابرا، يقول…٣١٥٥أخبرنا محمد بن بشار، قال حدثنا أبو عاصم، قال حدثنا محمد بن عجلان، عن…٣١٥٦والحارث بن مسكين، قراءة عليه وأنا أسمع، عن ابن القاسم، قال حدثني…٣١٥٧أن الجهاد في سبيل الله والإيمان بالله أفضل الأعمال٣١٥٨سمع محمد بن قيس، عن عبد الله بن أبي قتادة، عن أبيه، قال جاء رجل إلى…٣١٥٩ما على الأرض من نفس تموت ولها عند الله خير تحب أن ترجع إليكم ولها…٣١٦٠يؤتى بالرجل من أهل الجنة فيقول الله عز وجل يا ابن آدم كيف وجدت منزلك…٣١٦١الشهيد لا يجد مس القتل إلا كما يجد أحدكم القرصة يقرصها٣١٦٢من سأل الله عز وجل الشهادة بصدق بلغه الله منازل الشهداء وإن مات على…٣١٦٣خمس من قبض في شىء منهن فهو شهيد المقتول في سبيل الله شهيد والغرق في…٣١٦٤يختصم الشهداء والمتوفون على فرشهم إلى ربنا في الذين يتوفون من الطاعون…٣١٦٥إن الله عز وجل يعجب من رجلين يقتل أحدهما صاحبه - وقال مرة أخرى ليضحك…٣١٦٦يضحك الله إلى رجلين يقتل أحدهما الآخر كلاهما يدخل الجنة يقاتل هذا في…٣١٦٧من رابط يوما وليلة في سبيل الله كان له كأجر صيام شهر وقيامه ومن مات…٣١٦٨من رابط في سبيل الله يوما وليلة كانت له كصيام شهر وقيامه فإن مات جرى…٣١٦٩رباط يوم في سبيل الله خير من ألف يوم فيما سواه من المنازل٣١٧٠يوم في سبيل الله خير من ألف يوم فيما سواه٣١٧١ناس من أمتي عرضوا على غزاة في سبيل الله يركبون ثبج هذا البحر ملوك على…٣١٧٢رأيت قوما من أمتي يركبون هذا البحر كالملوك على الأسرة٣١٧٣أخبرني أحمد بن عثمان بن حكيم، قال حدثنا زكريا بن عدي، قال حدثنا عبيد…٣١٧٤حدثني محمد بن إسماعيل بن إبراهيم، قال حدثنا يزيد، قال أنبأنا هشيم،…٣١٧٥عصابتان من أمتي أحرزهما الله من النار عصابة تغزو الهند وعصابة تكون مع…٣١٧٦{ تمت كلمة ربك صدقا وعدلا لا مبدل لكلماته وهو السميع العليم }٣١٧٧لا تقوم الساعة حتى يقاتل المسلمون الترك قوما وجوههم كالمجان المطرقة…٣١٧٨إنما ينصر الله هذه الأمة بضعيفها بدعوتهم وصلاتهم وإخلاصهم٣١٧٩ابغوني الضعيف فإنكم إنما ترزقون وتنصرون بضعفائكم٣١٨٠من جهز غازيا في سبيل الله فقد غزا ومن خلفه في أهله بخير فقد غزا٣١٨١من جهز غازيا فقد غزا ومن خلف غازيا في أهله بخير فقد غزا٣١٨٢من يبتاع مربد بني فلان غفر الله له٣١٨٣من أنفق زوجين في سبيل الله عز وجل نودي في الجنة يا عبد الله هذا خير…٣١٨٤من أنفق زوجين في سبيل الله دعته خزنة الجنة من أبواب الجنة يا فلان هلم…٣١٨٥ما من عبد مسلم ينفق من كل مال له زوجين في سبيل الله إلا استقبلته حجبة…٣١٨٦من أنفق نفقة في سبيل الله كتبت له بسبعمائة ضعف٣١٨٧ليأتين يوم القيامة بسبعمائة ناقة مخطومة٣١٨٨الغزو غزوان فأما من ابتغى وجه الله وأطاع الإمام وأنفق الكريمة وياسر…٣١٨٩حرمة نساء المجاهدين على القاعدين كحرمة أمهاتهم وما من رجل يخلف في…٣١٩٠حرمة نساء المجاهدين على القاعدين كحرمة أمهاتهم وإذا خلفه في أهله…٣١٩١حرمة نساء المجاهدين على القاعدين في الحرمة كأمهاتهم وما من رجل من…٣١٩٢جاهدوا بأيديكم وألسنتكم وأموالكم٣١٩٣من خاف ثأرهن فليس منا٣١٩٤وما تعدون الشهادة إلا من قتل في سبيل الله إن شهداءكم إذا لقليل القتل…٣١٩٥دعهن يبكين ما دام بينهن فإذا وجب فلا تبكين باكية
