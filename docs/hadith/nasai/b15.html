سنن النسائي

كتاب تقصير الصلاة فى السفر

٢٦ حديثًا
١٤٣٣صدقة تصدق الله بها عليكم فاقبلوا صدقته١٤٣٤أخبرنا قتيبة، قال حدثنا الليث، عن ابن شهاب، عن عبد الله بن أبي بكر بن…١٤٣٥أن رسول الله صلى الله عليه وسلم خرج من مكة إلى المدينة لا يخاف إلا رب…١٤٣٦أخبرنا محمد بن عبد الأعلى، قال حدثنا خالد، قال حدثنا ابن عون، عن…١٤٣٧فقال إنما أفعل كما رأيت رسول الله صلى الله عليه وسلم يفعل .١٤٣٨قال خرجت مع رسول الله صلى الله عليه وسلم من المدينة إلى مكة فلم يزل…١٤٣٩قال صليت مع رسول الله صلى الله عليه وسلم في السفر ركعتين ومع أبي بكر…١٤٤٠قال صلاة الجمعة ركعتان والفطر ركعتان والنحر ركعتان والسفر ركعتان تمام…١٤٤١قال فرضت صلاة الحضر على لسان نبيكم صلى الله عليه وسلم أربعا وصلاة…١٤٤٢قال إن الله عز وجل فرض الصلاة على لسان نبيكم صلى الله عليه وسلم في…١٤٤٣- وهو ابن سلمة - قال قلت لابن عباس كيف أصلي بمكة إذا لم أصل في جماعة…١٤٤٤أن موسى بن سلمة، حدثهم أنه، سأل ابن عباس قلت تفوتني الصلاة في جماعة…١٤٤٥قال صليت مع النبي صلى الله عليه وسلم بمنى آمن ما كان الناس وأكثره…١٤٤٦أخبرنا عمرو بن علي، قال حدثنا يحيى بن سعيد، قال حدثنا شعبة، قال حدثنا…١٤٤٧أنه قال صليت مع رسول الله صلى الله عليه وسلم بمنى ومع أبي بكر وعمر…١٤٤٨- رضى الله عنه - قال صليت بمنى مع رسول الله صلى الله عليه وسلم ركعتين…١٤٤٩قال صلى عثمان بمنى أربعا حتى بلغ ذلك عبد الله فقال لقد صليت مع رسول…١٤٥٠قال صليت مع النبي صلى الله عليه وسلم بمنى ركعتين ومع أبي بكر - رضى…١٤٥١قال صلى رسول الله صلى الله عليه وسلم بمنى ركعتين وصلاها أبو بكر…١٤٥٢أخبرنا حميد بن مسعدة، قال حدثنا يزيد، قال أنبأنا يحيى بن أبي إسحاق،…١٤٥٣أن رسول الله صلى الله عليه وسلم أقام بمكة خمسة عشر يصلي ركعتين ركعتين…١٤٥٤يمكث المهاجر بعد قضاء نسكه ثلاثا١٤٥٥يمكث المهاجر بمكة بعد نسكه ثلاثا١٤٥٦أنها اعتمرت مع رسول الله صلى الله عليه وسلم من المدينة إلى مكة حتى…١٤٥٧قال كان ابن عمر لا يزيد في السفر على ركعتين لا يصلي قبلها ولا بعدها .…١٤٥٨كنت مع ابن عمر في سفر فصلى الظهر والعصر ركعتين ثم انصرف إلى طنفسة له…
