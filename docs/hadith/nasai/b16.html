سنن النسائي

كتاب الكسوف

٤٥ حديثًا
١٤٥٩إن الشمس والقمر آيتان من آيات الله تعالى لا ينكسفان لموت أحد ولا…١٤٦٠بأسهم لي بالمدينة إذ انكسفت الشمس فجمعت أسهمي وقلت لأنظرن ما أحدثه…١٤٦١إن الشمس والقمر لا يخسفان لموت أحد ولا لحياته ولكنهما آيتان من آيات…١٤٦٢إن الشمس والقمر لا ينكسفان لموت أحد ولكنهما آيتان من آيات الله عز وجل…١٤٦٣إن الشمس والقمر آيتان من آيات الله عز وجل وإنهما لا ينكسفان لموت أحد…١٤٦٤ومحمد بن عبد الأعلى، قالا حدثنا خالد، قال حدثنا أشعث، عن الحسن، عن…١٤٦٥قالت خسفت الشمس على عهد رسول الله صلى الله عليه وسلم فأمر النبي صلى…١٤٦٦أن عائشة، زوج النبي صلى الله عليه وسلم قالت كسفت الشمس في حياة رسول…١٤٦٧أخبرنا يعقوب بن إبراهيم، عن إسماعيل ابن علية، قال حدثنا سفيان الثوري،…١٤٦٨أخبرنا محمد بن المثنى، عن يحيى، عن سفيان، قال حدثنا حبيب بن أبي ثابت،…١٤٦٩أن رسول الله صلى الله عليه وسلم صلى يوم كسفت الشمس أربع ركعات في…١٤٧٠أصدق فظننت أنه يريد عائشة أنها قالت كسفت الشمس على عهد رسول الله صلى…١٤٧١أخبرنا إسحاق بن إبراهيم، قال حدثنا معاذ بن هشام، قال حدثني أبي، عن…١٤٧٢سمع الله لمن حمده ربنا ولك الحمد١٤٧٣قالت خسفت الشمس على عهد رسول الله صلى الله عليه وسلم فنودي الصلاة…١٤٧٤إن الشمس والقمر آيتان من آيات الله لا يخسفان لموت أحد ولا لحياته فإذا…١٤٧٥إن الناس يفتنون في قبورهم كفتنة الدجال١٤٧٦إنكم تفتنون في القبور كفتنة الدجال١٤٧٧أن رسول الله صلى الله عليه وسلم صلى في كسوف في صفة زمزم أربع ركعات في…١٤٧٨قال كسفت الشمس على عهد رسول الله صلى الله عليه وسلم في يوم شديد الحر…١٤٧٩قال خسفت الشمس على عهد رسول الله صلى الله عليه وسلم فأمر فنودي الصلاة…١٤٨٠أخبرنا يحيى بن عثمان، قال حدثنا ابن حمير، عن معاوية بن سلام، عن يحيى…١٤٨١سمع الله لمن حمده١٤٨٢لم تعدني هذا وأنا فيهم لم تعدني هذا ونحن نستغفرك١٤٨٣إن الشمس والقمر آيتان من آيات الله وإنهما لا ينكسفان لموت أحد ولا…١٤٨٤أخبرنا هلال بن العلاء بن هلال، قال حدثنا الحسين بن عياش، قال حدثنا…١٤٨٥إن ناسا يزعمون أن الشمس والقمر لا ينكسفان إلا لموت عظيم من العظماء…١٤٨٦إن الشمس والقمر آيتان من آيات الله وإنهما لا ينكسفان لموت أحد ولا…١٤٨٧إن الشمس والقمر لا ينخسفان لموت أحد ولكنهما خلقان من خلقه وإن الله عز…١٤٨٨إذا خسفت الشمس والقمر فصلوا كأحدث صلاة صليتموها١٤٨٩أخبرنا أحمد بن عثمان بن حكيم، قال حدثنا أبو نعيم، عن الحسن بن صالح،…١٤٩٠إن أهل الجاهلية كانوا يقولون إن الشمس والقمر لا ينخسفان إلا لموت عظيم…١٤٩١إن الشمس والقمر آيتان من آيات الله يخوف الله عز وجل بهما عباده وإنهما…١٤٩٢أن رسول الله صلى الله عليه وسلم صلى ركعتين مثل صلاتكم هذه وذكر كسوف…١٤٩٣إن الشمس والقمر آيتان من آيات الله لا يخسفان لموت أحد ولا لحياته فإذا…١٤٩٤سمع الله لمن حمده ربنا ولك الحمد١٤٩٥. أن النبي صلى الله عليه وسلم صلى بهم في كسوف الشمس لا نسمع له صوتا .١٤٩٦رب لم تعدني هذا وأنا أستغفرك لم تعدني هذا وأنا فيهم١٤٩٧سمع الله لمن حمده١٤٩٨قالت صلى رسول الله صلى الله عليه وسلم في الكسوف فقام فأطال القيام ثم…١٤٩٩إن الناس يفتنون في قبورهم كفتنة الدجال١٥٠٠إن الشمس والقمر لا ينكسفان لموت أحد ولا لحياته فإذا رأيتم ذلك فصلوا…١٥٠١أن النبي صلى الله عليه وسلم خطب حين انكسفت الشمس فقال " أما بعد " .١٥٠٢إن الشمس والقمر آيتان من آيات الله يخوف بهما عباده وإنهما لا ينكسفان…١٥٠٣إن هذه الآيات التي يرسل الله لا تكون لموت أحد ولا لحياته ولكن الله…
