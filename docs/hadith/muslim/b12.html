٢٢٩ حديثًا•صفحة ١ من ٢
٢٢٦٣ليس فيما دون خمسة أوسق صدقة ولا فيما دون خمس ذود صدقة ولا فيما دون…٢٢٦٤بهذا الإسناد . مثله .٢٢٦٥وحدثنا محمد بن رافع، حدثنا عبد الرزاق، أخبرنا ابن جريج، أخبرني عمرو،…٢٢٦٦ليس فيما دون خمسة أوسق صدقة وليس فيما دون خمس ذود صدقة وليس فيما دون…٢٢٦٧ليس فيما دون خمسة أوساق من تمر ولا حب صدقة٢٢٦٨ليس في حب ولا تمر صدقة حتى يبلغ خمسة أوسق ولا فيما دون خمس ذود صدقة…٢٢٦٩بن أمية بهذا الإسناد . مثل حديث ابن مهدي .٢٢٧٠ومعمر، عن إسماعيل، بن أمية بهذا الإسناد . مثل حديث ابن مهدي ويحيى بن…٢٢٧١ليس فيما دون خمس أواق من الورق صدقة وليس فيما دون خمس ذود من الإبل…٢٢٧٢فيما سقت الأنهار والغيم العشور وفيما سقي بالسانية نصف العشر٢٢٧٣ليس على المسلم في عبده ولا فرسه صدقة٢٢٧٤ليس على المسلم في عبده ولا فرسه صدقة٢٢٧٥حدثنا يحيى بن يحيى، أخبرنا سليمان بن بلال، ح وحدثنا قتيبة، حدثنا…٢٢٧٦ليس في العبد صدقة إلا صدقة الفطر٢٢٧٧ما ينقم ابن جميل إلا أنه كان فقيرا فأغناه الله وأما خالد فإنكم تظلمون…٢٢٧٨وقتيبة بن سعيد، قالا حدثنا مالك، ح وحدثنا يحيى بن يحيى، - واللفظ له -…٢٢٧٩قال فرض رسول الله صلى الله عليه وسلم زكاة الفطر صاعا من تمر أو صاعا…٢٢٨٠عمر قال فرض النبي صلى الله عليه وسلم صدقة رمضان على الحر والعبد…٢٢٨١أن عبد الله بن عمر، قال إن رسول الله صلى الله عليه وسلم أمر بزكاة…٢٢٨٢أن رسول الله صلى الله عليه وسلم فرض زكاة الفطر من رمضان على كل نفس من…٢٢٨٣عبد الله بن سعد بن أبي سرح أنه سمع أبا سعيد الخدري، يقول كنا نخرج…٢٢٨٤حدثنا عبد الله بن مسلمة بن قعنب، حدثنا داود، - يعني ابن قيس - عن…٢٢٨٥أنه سمع أبا سعيد الخدري، يقول كنا نخرج زكاة الفطر ورسول الله صلى الله…٢٢٨٦وحدثني محمد بن رافع، حدثنا عبد الرزاق، أخبرنا ابن جريج، عن الحارث بن،…٢٢٨٧أن معاوية، لما جعل نصف الصاع من الحنطة عدل صاع من تمر أنكر ذلك أبو…٢٢٨٨أن رسول الله صلى الله عليه وسلم أمر بزكاة الفطر أن تؤدى قبل خروج…٢٢٨٩الله بن عمر أن رسول الله صلى الله عليه وسلم أمر بإخراج زكاة الفطر أن…٢٢٩٠ما من صاحب ذهب ولا فضة لا يؤدي منها حقها إلا إذا كان يوم القيامة صفحت…٢٢٩١ما من صاحب إبل لا يؤدي حقها٢٢٩٢ما من صاحب كنز لا يؤدي زكاته إلا أحمي عليه في نار جهنم فيجعل صفائح…٢٢٩٣فيكوى بها جنبه وظهره٢٢٩٤إذا لم يؤد المرء حق الله أو الصدقة في إبله٢٢٩٦ما من صاحب إبل لا يفعل فيها حقها إلا جاءت يوم القيامة أكثر ما كانت قط…٢٢٩٧ما من صاحب إبل ولا بقر ولا غنم لا يؤدي حقها إلا أقعد لها يوم القيامة…٢٢٩٨حدثنا أبو كامل، فضيل بن حسين الجحدري حدثنا عبد الواحد بن زياد، حدثنا…٢٢٩٩بن أبي إسماعيل بهذا الإسناد نحوه .٢٣٠٠هم الأخسرون ورب الكعبة٢٣٠١والذي نفسي بيده ما على الأرض رجل يموت فيدع إبلا أو بقرا أو غنما لم…٢٣٠٢ما يسرني أن لي أحدا ذهبا تأتي على ثالثة وعندي منه دينار إلا دينار…٢٣٠٣وحدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، عن محمد بن زياد،…٢٣٠٤حدثنا يحيى بن يحيى، وأبو بكر بن أبي شيبة وابن نمير وأبو كريب كلهم عن…٢٣٠٥وحدثنا قتيبة بن سعيد، حدثنا جرير، عن عبد العزيز، - وهو ابن رفيع - عن…٢٣٠٦وحدثني زهير بن حرب، حدثنا إسماعيل بن إبراهيم، عن الجريري، عن أبي،…٢٣٠٧وحدثنا شيبان بن فروخ، حدثنا أبو الأشهب، حدثنا خليد العصري، عن الأحنف،…٢٣٠٨قال الله تبارك وتعالى يا ابن آدم أنفق أنفق عليك٢٣٠٩إن الله قال لي أنفق أنفق عليك٢٣١٠أفضل دينار ينفقه الرجل دينار ينفقه على عياله ودينار ينفقه الرجل على…٢٣١١دينار أنفقته في سبيل الله ودينار أنفقته في رقبة ودينار تصدقت به على…٢٣١٢كفى بالمرء إثما أن يحبس عمن يملك قوته٢٣١٣حدثنا قتيبة بن سعيد، حدثنا ليث، ح وحدثنا محمد بن رمح، أخبرنا الليث،…٢٣١٤أن رجلا، من الأنصار - يقال له أبو مذكور - أعتق غلاما له عن دبر يقال…٢٣١٥بخ ذلك مال رابح ذلك مال رابح قد سمعت ما قلت فيها وإني أرى أن تجعلها…٢٣١٦اجعلها في قرابتك٢٣١٧لو أعطيتها أخوالك كان أعظم لأجرك٢٣١٨تصدقن يا معشر النساء ولو من حليكن٢٣١٩تصدقن ولو من حليكن٢٣٢٠نعم لك فيهم أجر ما أنفقت عليهم٢٣٢١في هذا الإسناد بمثله .٢٣٢٢إن المسلم إذا أنفق على أهله نفقة وهو يحتسبها كانت له صدقة٢٣٢٣في هذا الإسناد .٢٣٢٤قالت قلت يا رسول الله إن أمي قدمت على وهى راغبة - أو راهبة - أفأصلها…٢٣٢٥قالت قدمت على أمي وهي مشركة في عهد قريش إذ عاهدهم فاستفتيت رسول الله…٢٣٢٦أن رجلا، أتى النبي صلى الله عليه وسلم فقال يا رسول الله إن أمي افتلتت…٢٣٢٧بهذا الإسناد . وفي حديث أبي أسامة ولم توص . كما قال ابن بشر ولم يقل…٢٣٢٨حدثنا قتيبة بن سعيد، حدثنا أبو عوانة، ح وحدثنا أبو بكر بن أبي شيبة،…٢٣٢٩أوليس قد جعل الله لكم ما تصدقون إن بكل تسبيحة صدقة وكل تكبيرة صدقة…٢٣٣٠إنه خلق كل إنسان من بني آدم على ستين وثلاثمائة مفصل فمن كبر الله وحمد…٢٣٣١زيد بهذا الإسناد . مثله غير أنه قال " أو أمر بمعروف " . وقال " فإنه…٢٣٣٢بن فروخ أنه سمع عائشة، تقول قال رسول الله صلى الله عليه وسلم " خلق كل…٢٣٣٣على كل مسلم صدقة٢٣٣٤بهذا الإسناد.٢٣٣٥كل سلامى من الناس عليه صدقة كل يوم تطلع فيه الشمس - قال - تعدل بين…٢٣٣٦ما من يوم يصبح العباد فيه إلا ملكان ينزلان فيقول أحدهما اللهم أعط…٢٣٣٧تصدقوا فيوشك الرجل يمشي بصدقته فيقول الذي أعطيها لو جئتنا بها بالأمس…٢٣٣٨ليأتين على الناس زمان يطوف الرجل فيه بالصدقة من الذهب ثم لا يجد أحدا…٢٣٤٠لا تقوم الساعة حتى يكثر فيكم المال فيفيض حتى يهم رب المال من يقبله…٢٣٤١تقيء الأرض أفلاذ كبدها أمثال الأسطوان من الذهب والفضة فيجيء القاتل…٢٣٤٢ما تصدق أحد بصدقة من طيب - ولا يقبل الله إلا الطيب - إلا أخذها الرحمن…٢٣٤٣لا يتصدق أحد بتمرة من كسب طيب إلا أخذها الله بيمينه فيربيها كما يربي…٢٣٤٤من الكسب الطيب فيضعها في حقها٢٣٤٥وحدثنيه أبو الطاهر، أخبرنا عبد الله بن وهب، أخبرني هشام بن سعد، عن…٢٣٤٦أيها الناس إن الله طيب لا يقبل إلا طيبا وإن الله أمر المؤمنين بما أمر…٢٣٤٧من استطاع منكم أن يستتر من النار ولو بشق تمرة فليفعل٢٣٤٨ما منكم من أحد إلا سيكلمه الله ليس بينه وبينه ترجمان فينظر أيمن منه…٢٣٤٩حدثنا أبو بكر بن أبي شيبة، وأبو كريب قالا حدثنا أبو معاوية، عن…٢٣٥٠اتقوا النار ولو بشق تمرة فإن لم تجدوا فبكلمة طيبة٢٣٥١{ يا أيها الناس اتقوا ربكم الذي خلقكم من نفس واحدة } إلى آخر الآية {…٢٣٥٢وحدثنا أبو بكر بن أبي شيبة، حدثنا أبو أسامة، ح وحدثنا عبيد الله بن…٢٣٥٣أما بعد فإن الله أنزل في كتابه { يا أيها الناس اتقوا ربكم } الآية٢٣٥٤قال جاء ناس من الأعراب إلى رسول الله صلى الله عليه وسلم عليهم الصوف…٢٣٥٥حدثني يحيى بن معين، حدثنا غندر، حدثنا شعبة، ح وحدثنيه بشر بن خالد، -…٢٣٥٦وحدثنا محمد بن بشار، حدثني سعيد بن الربيع، ح وحدثنيه إسحاق بن منصور،…٢٣٥٧ألا رجل يمنح أهل بيت ناقة تغدو بعس وتروح بعس إن أجرها لعظيم٢٣٥٨من منح منيحة غدت بصدقة وراحت بصدقة صبوحها وغبوقها٢٣٥٩مثل المنفق والمتصدق كمثل رجل عليه جبتان أو جنتان من لدن ثديهما إلى…٢٣٦٠مثل البخيل والمتصدق كمثل رجلين عليهما جنتان من حديد قد اضطرت أيديهما…٢٣٦١مثل البخيل والمتصدق مثل رجلين عليهما جنتان من حديد إذا هم المتصدق…٢٣٦٢قال رجل لأتصدقن الليلة بصدقة فخرج بصدقته فوضعها في يد زانية فأصبحوا…٢٣٦٣إن الخازن المسلم الأمين الذي ينفذ - وربما قال يعطي - ما أمر به فيعطيه…٢٣٦٤إذا أنفقت المرأة من طعام بيتها غير مفسدة كان لها أجرها بما أنفقت…٢٣٦٥بهذا الإسناد وقال " من طعام زوجها " .٢٣٦٦إذا أنفقت المرأة من بيت زوجها غير مفسدة كان لها أجرها وله مثله بما…٢٣٦٧بهذا الإسناد نحوه٢٣٦٨نعم والأجر بينكما نصفان٢٣٦٩مولى آبي اللحم قال أمرني مولاى أن أقدد، لحما فجاءني مسكين فأطعمته منه…٢٣٧٠لا تصم المرأة وبعلها شاهد إلا بإذنه ولا تأذن في بيته وهو شاهد إلا…٢٣٧١من أنفق زوجين في سبيل الله نودي في الجنة يا عبد الله هذا خير . فمن…٢٣٧٢والحسن الحلواني، وعبد بن حميد، قالوا حدثنا يعقوب، - وهو ابن إبراهيم…٢٣٧٣من أنفق زوجين في سبيل الله دعاه خزنة الجنة كل خزنة باب أى فل هلم٢٣٧٤من أصبح منكم اليوم صائما٢٣٧٥أنفقي - أو انضحي أو انفحي - ولا تحصي فيحصي الله عليك٢٣٧٦انفحي - أو انضحي أو أنفقي - ولا تحصي فيحصي الله عليك ولا توعي فيوعي…٢٣٧٧أن النبي صلى الله عليه وسلم قال لها نحو حديثهم .٢٣٧٨ارضخي ما استطعت ولا توعي فيوعي الله عليك٢٣٧٩سبعة يظلهم الله في ظله يوم لا ظل إلا ظله الإمام العادل وشاب نشأ…٢٣٨٠ورجل معلق بالمسجد إذا خرج منه حتى يعود إليه٢٣٨٢أن تصدق وأنت صحيح شحيح تخشى الفقر وتأمل الغنى ولا تمهل حتى إذا بلغت…٢٣٨٣أما وأبيك لتنبأنه أن تصدق وأنت صحيح شحيح تخشى الفقر وتأمل البقاء ولا…٢٣٨٤بهذا الإسناد نحو حديث جرير غير أنه قال أى الصدقة أفضل .٢٣٨٥اليد العليا خير من اليد السفلى واليد العليا المنفقة والسفلى السائلة٢٣٨٦أفضل الصدقة - أو خير الصدقة - عن ظهر غنى واليد العليا خير من اليد…٢٣٨٧إن هذا المال خضرة حلوة فمن أخذه بطيب نفس بورك له فيه ومن أخذه بإشراف…٢٣٨٨يا ابن آدم إنك أن تبذل الفضل خير لك وأن تمسكه شر لك ولا تلام على كفاف…٢٣٨٩من يرد الله به خيرا يفقهه في الدين٢٣٩٠لا تلحفوا في المسألة فوالله لا يسألني أحد منكم شيئا فتخرج له مسألته…٢٣٩١حدثنا ابن أبي عمر المكي، حدثنا سفيان، عن عمرو بن دينار، حدثني وهب بن،…٢٣٩٢من يرد الله به خيرا يفقهه في الدين وإنما أنا قاسم ويعطي الله٢٣٩٣ليس المسكين بهذا الطواف الذي يطوف على الناس فترده اللقمة واللقمتان…٢٣٩٤ليس المسكين بالذي ترده التمرة والتمرتان ولا اللقمة واللقمتان إنما…٢٣٩٥وعبد الرحمن بن أبي عمرة، أنهما سمعا أبا هريرة، يقول قال رسول الله صلى…٢٣٩٦لا تزال المسألة بأحدكم حتى يلقى الله وليس في وجهه مزعة لحم٢٣٩٧بهذا الإسناد . مثله ولم يذكر " مزعة " .٢٣٩٨ما يزال الرجل يسأل الناس حتى يأتي يوم القيامة وليس في وجهه مزعة لحم٢٣٩٩من سأل الناس أموالهم تكثرا فإنما يسأل جمرا فليستقل أو ليستكثر٢٤٠٠لأن يغدو أحدكم فيحطب على ظهره فيتصدق به ويستغني به من الناس خير له من…٢٤٠١والله لأن يغدو أحدكم فيحطب على ظهره فيبيعه٢٤٠٢لأن يحتزم أحدكم حزمة من حطب فيحملها على ظهره فيبيعها خير له من أن…٢٤٠٣ألا تبايعون رسول الله٢٤٠٤أقم حتى تأتينا الصدقة فنأمر لك بها٢٤٠٥خذه وما جاءك من هذا المال وأنت غير مشرف ولا سائل فخذه وما لا فلا…٢٤٠٦خذه فتموله أو تصدق به وما جاءك من هذا المال وأنت غير مشرف ولا سائل…٢٤٠٧وحدثني أبو الطاهر، أخبرنا ابن وهب، قال عمرو وحدثني ابن شهاب، بمثل ذلك…٢٤٠٨إذا أعطيت شيئا من غير أن تسأل فكل وتصدق٢٤٠٩أنه قال استعملني عمر بن الخطاب - رضى الله عنه - على الصدقة . بمثل…٢٤١٠قلب الشيخ شاب على حب اثنتين حب العيش والمال٢٤١١قلب الشيخ شاب على حب اثنتين طول الحياة وحب المال٢٤١٢يهرم ابن آدم وتشب منه اثنتان الحرص على المال والحرص على العمر٢٤١٣ومحمد بن المثنى، قالا حدثنا معاذ بن هشام، حدثني أبي، عن قتادة، عن…٢٤١٤وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت قتادة، يحدث…٢٤١٥لو كان لابن آدم واديان من مال لابتغى واديا ثالثا ولا يملأ جوف ابن آدم…
