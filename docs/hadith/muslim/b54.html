كتاب الفتن وأشراط الساعة

١٥٨ حديثًا•صفحة ١ من ٢
٧٢٣٥لا إله إلا الله ويل للعرب من شر قد اقترب فتح اليوم من ردم يأجوج…٧٢٣٦وسعيد بن عمرو الأشعثي، وزهير بن حرب، وابن، أبي عمر قالوا حدثنا سفيان،…٧٢٣٧لا إله إلا الله ويل للعرب من شر قد اقترب فتح اليوم من ردم يأجوج…٧٢٣٨بإسناده .٧٢٣٩فتح اليوم من ردم يأجوج ومأجوج مثل هذه٧٢٤٠يعوذ عائذ بالبيت فيبعث إليه بعث فإذا كانوا ببيداء من الأرض خسف بهم٧٢٤١بهذا الإسناد وفي حديثه قال فلقيت أبا جعفر فقلت إنها إنما قالت ببيداء…٧٢٤٢ليؤمن هذا البيت جيش يغزونه حتى إذا كانوا ببيداء من الأرض يخسف بأوسطهم…٧٢٤٣سيعوذ بهذا البيت - يعني الكعبة - قوم ليست لهم منعة ولا عدد ولا عدة…٧٢٤٤العجب إن ناسا من أمتي يؤمون بالبيت برجل من قريش قد لجأ بالبيت حتى إذا…٧٢٤٥هل ترون ما أرى إني لأرى مواقع الفتن خلال بيوتكم كمواقع القطر٧٢٤٦بهذا الإسناد نحوه .٧٢٤٧ستكون فتن القاعد فيها خير من القائم والقائم فيها خير من الماشي…٧٢٤٨من الصلاة صلاة من فاتته فكأنما وتر أهله وماله٧٢٤٩تكون فتنة النائم فيها خير من اليقظان واليقظان فيها خير من القائم…٧٢٥٠إنها ستكون فتن ألا ثم تكون فتنة القاعد فيها خير من الماشي فيها…٧٢٥١إن استطاع النجاء٧٢٥٢إذا تواجه المسلمان بسيفيهما فالقاتل والمقتول في النار٧٢٥٣إذا التقى المسلمان بسيفيهما فالقاتل والمقتول في النار٧٢٥٤إلى آخره .٧٢٥٥إذا المسلمان حمل أحدهما على أخيه السلاح فهما في جرف جهنم فإذا قتل…٧٢٥٧لا تقوم الساعة حتى يكثر الهرج٧٢٥٨إن الله زوى لي الأرض فرأيت مشارقها ومغاربها وإن أمتي سيبلغ ملكها ما…٧٢٥٩إن الله تعالى زوى لي الأرض حتى رأيت مشارقها ومغاربها وأعطاني الكنزين…٧٢٦٠سألت ربي ثلاثا فأعطاني ثنتين ومنعني واحدة سألت ربي أن لا يهلك أمتي…٧٢٦١أنه أقبل مع رسول الله صلى الله عليه وسلم في طائفة من أصحابه فمر بمسجد…٧٢٦٢منهن ثلاث لا يكدن يذرن شيئا ومنهن فتن كرياح الصيف منها صغار ومنها كبار٧٢٦٣وإسحاق بن إبراهيم، قال عثمان حدثنا وقال، إسحاق أخبرنا جرير، عن…٧٢٦٤بهذا الإسناد إلى قوله ونسيه من نسيه . ولم يذكر ما بعده .٧٢٦٥وحدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، ح وحدثني أبو…٧٢٦٦بهذا الإسناد نحوه .٧٢٦٧وحجاج بن الشاعر، جميعا عن أبي عاصم، - قال حجاج حدثنا أبو عاصم، -…٧٢٦٨فتنة الرجل في أهله وماله ونفسه وولده وجاره يكفرها الصيام والصلاة…٧٢٦٩يقول .٧٢٧٠والأعمش، عن أبي وائل، عن حذيفة، قال قال عمر من يحدثنا عن الفتنة،…٧٢٧١ومحمد بن حاتم، قالا حدثنا معاذ بن معاذ، حدثنا ابن، عون عن محمد، قال…٧٢٧٢لا تقوم الساعة حتى يحسر الفرات عن جبل من ذهب يقتتل الناس عليه فيقتل…٧٢٧٣بهذا الإسناد نحوه وزاد فقال أبي إن رأيته فلا تقربنه .٧٢٧٤يوشك الفرات أن يحسر عن كنز من ذهب فمن حضره فلا يأخذ منه شيئا٧٢٧٥يوشك الفرات أن يحسر عن جبل من ذهب فمن حضره فلا يأخذ منه شيئا٧٢٧٦يوشك الفرات أن يحسر عن جبل من ذهب فإذا سمع به الناس ساروا إليه فيقول…٧٢٧٧منعت العراق درهمها وقفيزها ومنعت الشأم مديها ودينارها ومنعت مصر…٧٢٧٨لا تقوم الساعة حتى ينزل الروم بالأعماق أو بدابق فيخرج إليهم جيش من…٧٢٧٩تقوم الساعة والروم أكثر الناس٧٢٨٠تقوم الساعة والروم أكثر الناس٧٢٨١إني لأعرف أسماءهم وأسماء آبائهم وألوان خيولهم هم خير فوارس على ظهر…٧٢٨٢قال كنت عند ابن مسعود فهبت ريح حمراء . وساق الحديث بنحوه . وحديث ابن…٧٢٨٣قال كنت في بيت عبد الله بن مسعود والبيت ملآن - قال - فهاجت ريح حمراء…٧٢٨٤تغزون جزيرة العرب فيفتحها الله ثم فارس فيفتحها الله ثم تغزون الروم…٧٢٨٥حدثنا أبو خيثمة، زهير بن حرب وإسحاق بن إبراهيم وابن أبي عمر المكي -…٧٢٨٦حدثنا عبيد الله بن معاذ العنبري، حدثنا أبي، حدثنا شعبة، عن فرات…٧٢٨٧وحدثناه محمد بن بشار، حدثنا محمد، - يعني ابن جعفر - حدثنا شعبة، عن…٧٢٨٨بنحوه قال والعاشرة نزول عيسى ابن مريم . قال شعبة ولم يرفعه عبد العزيز…٧٢٨٩لا تقوم الساعة حتى تخرج نار من أرض الحجاز تضيء أعناق الإبل ببصرى٧٢٩٠تبلغ المساكن إهاب أو يهاب٧٢٩١ليست السنة بأن لا تمطروا ولكن السنة أن تمطروا وتمطروا ولا تنبت الأرض…٧٢٩٢ألا إن الفتنة ها هنا ألا إن الفتنة ها هنا من حيث يطلع قرن الشيطان٧٢٩٣الفتنة ها هنا من حيث يطلع قرن الشيطان٧٢٩٤ها إن الفتنة ها هنا ها إن الفتنة ها هنا ها إن الفتنة ها هنا من حيث…٧٢٩٥رأس الكفر من ها هنا من حيث يطلع قرن الشيطان٧٢٩٦ها إن الفتنة ها هنا ها إن الفتنة ها هنا٧٢٩٧إن الفتنة تجيء من ها هنا٧٢٩٨لا تقوم الساعة حتى تضطرب أليات نساء دوس حول ذي الخلصة٧٢٩٩لا يذهب الليل والنهار حتى تعبد اللات والعزى٧٣٠٠بن جعفر بهذا الإسناد نحوه .٧٣٠٣والذي نفسي بيده ليأتين على الناس زمان لا يدري القاتل في أى شىء قتل…٧٣٠٤والذي نفسي بيده لا تذهب الدنيا حتى يأتي على الناس يوم لا يدري القاتل…٧٣٠٥يخرب الكعبة ذو السويقتين من الحبشة٧٣٠٦يخرب الكعبة ذو السويقتين من الحبشة٧٣٠٧ذو السويقتين من الحبشة يخرب بيت الله عز وجل٧٣٠٨لا تقوم الساعة حتى يخرج رجل من قحطان يسوق الناس بعصاه٧٣٠٩لا تذهب الأيام والليالي حتى يملك رجل يقال له الجهجاه٧٣١٠لا تقوم الساعة حتى تقاتلوا قوما كأن وجوههم المجان المطرقة ولا تقوم…٧٣١١لا تقوم الساعة حتى تقاتلكم أمة ينتعلون الشعر وجوههم مثل المجان المطرقة٧٣١٢لا تقوم الساعة حتى تقاتلوا قوما نعالهم الشعر ولا تقوم الساعة حتى…٧٣١٣لا تقوم الساعة حتى يقاتل المسلمون الترك قوما وجوههم كالمجان المطرقة…٧٣١٤تقاتلون بين يدى الساعة قوما نعالهم الشعر كأن وجوههم المجان المطرقة…٧٣١٥يكون في آخر أمتي خليفة يحثي المال حثيا لا يعده عددا٧٣١٦- يعني الجريري - بهذا الإسناد نحوه .٧٣١٧حديث ٧٣١٧٧٣١٨من خلفائكم خليفة يحثو المال حثيا لا يعده عددا٧٣٢٠بؤس ابن سمية تقتلك فئة باغية٧٣٢١وهريم بن عبد الأعلى، قالا حدثنا خالد بن الحارث، ح وحدثنا إسحاق بن…٧٣٢٢تقتلك الفئة الباغية٧٣٢٣والحسن، عن أمهما، عن أم سلمة، عن النبي صلى الله عليه وسلم بمثله .٧٣٢٤تقتل عمارا الفئة الباغية٧٣٢٦يهلك أمتي هذا الحى من قريش٧٣٢٧قد مات كسرى فلا كسرى بعده وإذا هلك قيصر فلا قيصر بعده والذي نفسي بيده…٧٣٢٨هلك كسرى ثم لا يكون كسرى بعده وقيصر ليهلكن ثم لا يكون قيصر بعده…٧٣٤٤تربت يداك أتشهد أني رسول الله٧٣٤٥قد خبأت لك خبيئا٧٣٤٦أتشهد أني رسول الله٧٣٤٧ومحمد بن عبد الأعلى، قالا حدثنا معتمر، قال سمعت أبي قال، حدثنا أبو…٧٣٤٨ومحمد بن المثنى، قالا حدثنا عبد الأعلى، حدثنا داود، عن أبي نضرة، عن…٧٣٤٩ومحمد بن عبد الأعلى، قالا حدثنا معتمر، قال سمعت أبي يحدث، عن أبي…٧٣٥٠حدثنا محمد بن المثنى، حدثنا سالم بن نوح، أخبرني الجريري، عن أبي نضرة،…٧٣٥١قال قال رسول الله صلى الله عليه وسلم لابن صائد " ما تربة الجنة " .…٧٣٥٢درمكة بيضاء مسك خالص٧٣٥٣قال رأيت جابر بن عبد الله يحلف بالله أن ابن صائد الدجال، فقلت أتحلف…٧٣٥٦أتشهد أني رسول الله٧٣٥٧وعبد بن حميد، قالا حدثنا يعقوب، - وهو ابن إبراهيم بن سعد - حدثنا أبي،…٧٣٥٨وسلمة بن شبيب، جميعا عن عبد الرزاق، أخبرنا معمر، عن الزهري، عن سالم،…٧٣٥٩إنما يخرج من غضبة يغضبها٧٣٦٠إن أول ما يبعثه على الناس غضب يغضبه٧٣٦١إن الله تعالى ليس بأعور . ألا وإن المسيح الدجال أعور العين اليمنى كأن…٧٣٦٢حدثني أبو الربيع، وأبو كامل قالا حدثنا حماد، - وهو ابن زيد - عن أيوب،…٧٣٦٣ما من نبي إلا وقد أنذر أمته الأعور الكذاب ألا إنه أعور وإن ربكم ليس…٧٣٦٤الدجال مكتوب بين عينيه ك ف ر أى كافر٧٣٦٥الدجال ممسوح العين مكتوب بين عينيه كافر٧٣٦٦الدجال أعور العين اليسرى جفال الشعر معه جنة ونار فناره جنة وجنته نار٧٣٦٧لأنا أعلم بما مع الدجال منه معه نهران يجريان أحدهما رأى العين ماء…٧٣٦٩إن الدجال يخرج وإن معه ماء ونارا فأما الذي يراه الناس ماء فنار تحرق…٧٣٧٠لأنا بما مع الدجال أعلم منه إن معه نهرا من ماء ونهرا من نار فأما الذي…٧٣٧٢ألا أخبركم عن الدجال حديثا ما حدثه نبي قومه إنه أعور وإنه يجيء معه…٧٣٧٣جبير بن نفير الحضرمي أنه سمع النواس بن سمعان الكلابي، ح وحدثني محمد…٧٣٧٤لقد كان بهذه مرة ماء ثم يسيرون حتى ينتهوا إلى جبل الخمر وهو جبل بيت…٧٣٧٥يأتي وهو محرم عليه أن يدخل نقاب المدينة فينتهي إلى بعض السباخ التي…٧٣٧٦في هذا الإسناد بمثله .٧٣٧٧يخرج الدجال فيتوجه قبله رجل من المؤمنين فتلقاه المسالح مسالح الدجال…٧٣٧٨وما ينصبك منه إنه لا يضرك٧٣٧٩حدثنا سريج بن يونس، حدثنا هشيم، عن إسماعيل، عن قيس، عن المغيرة، بن…٧٣٨٠وابن، نمير قالا حدثنا وكيع، ح وحدثنا إسحاق، بن إبراهيم أخبرنا جرير، ح…٧٣٨١يخرج الدجال في أمتي فيمكث أربعين - لا أدري أربعين يوما أو أربعين شهرا…٧٣٨٢يخرج الدجال في أمتي٧٣٨٣إن أول الآيات خروجا طلوع الشمس من مغربها وخروج الدابة على الناس ضحى…٧٣٨٤قال جلس إلى مروان بن الحكم بالمدينة ثلاثة نفر من المسلمين فسمعوه وهو،…٧٣٨٥وحدثنا نصر بن علي الجهضمي، حدثنا أبو أحمد، حدثنا سفيان، عن أبي حيان،…٧٣٨٦من أحبني فليحب أسامة٧٣٨٧إن بني عم لتميم الداري ركبوا في البحر٧٣٨٨هذه طيبة وذاك الدجال٧٣٨٩أيها الناس حدثني تميم الداري أن أناسا من قومه كانوا في البحر في سفينة…٧٣٩٠ليس من بلد إلا سيطؤه الدجال إلا مكة والمدينة وليس نقب من أنقابها إلا…٧٣٩١أن رسول الله صلى الله عليه وسلم قال . فذكر نحوه غير أنه قال فيأتي…٧٣٩٢يتبع الدجال من يهود أصبهان سبعون ألفا عليهم الطيالسة٧٣٩٣ليفرن الناس من الدجال في الجبال٧٣٩٤وعبد بن حميد، قالا حدثنا أبو عاصم، عن ابن جريج، بهذا الإسناد .٧٣٩٥ما بين خلق آدم إلى قيام الساعة خلق أكبر من الدجال٧٣٩٦أمر أكبر من الدجال٧٣٩٧بادروا بالأعمال ستا طلوع الشمس من مغربها أو الدخان أو الدجال أو…٧٣٩٨بادروا بالأعمال ستا الدجال والدخان ودابة الأرض وطلوع الشمس من مغربها…٧٣٩٩ومحمد بن المثنى، قالا حدثنا عبد الصمد بن عبد الوارث، حدثنا همام، عن…٧٤٠٠العبادة في الهرج كهجرة إلى٧٤٠١بهذا الإسناد نحوه .٧٤٠٢لا تقوم الساعة إلا على شرار الناس٧٤٠٣بعثت أنا والساعة هكذا٧٤٠٤بعثت أنا والساعة كهاتين٧٤٠٥بعثت أنا والساعة هكذا٧٤٠٦وحدثنا عبيد الله بن معاذ، حدثنا أبي ح، وحدثنا محمد بن الوليد، حدثنا…٧٤٠٧وحدثناه محمد بن بشار، حدثنا ابن أبي عدي، عن شعبة، عن حمزة، - يعني…٧٤٠٨بعثت أنا والساعة كهاتين
