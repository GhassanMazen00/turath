بقي من الناس اثنان٤٧٠٥إن هذا الأمر لا ينقضي حتى يمضي فيهم اثنا عشر خليفة٤٧٠٦لا يزال أمر الناس ماضيا ما وليهم اثنا عشر رجلا٤٧٠٧لا يزال أمر الناس ماضيا٤٧٠٨لا يزال الإسلام عزيزا إلى اثنى عشر خليفة٤٧٠٩لا يزال هذا الأمر عزيزا إلى اثنى عشر خليفة٤٧١٠لا يزال هذا الدين عزيزا منيعا إلى اثنى عشر خليفة٤٧١١لا يزال الدين قائما حتى تقوم الساعة أو يكون عليكم اثنا عشر خليفة كلهم…٤٧١٢حدثنا محمد بن رافع، حدثنا ابن أبي فديك، حدثنا ابن أبي ذئب، عن مهاجر،…٤٧١٣قال حضرت أبي حين أصيب فأثنوا عليه وقالوا جزاك الله خيرا . فقال راغب…٤٧١٤وابن أبي عمر، ومحمد بن رافع، وعبد بن حميد، وألفاظهم، متقاربة قال…٤٧١٥يا عبد الرحمن لا تسأل الإمارة فإنك إن أعطيتها عن مسألة أكلت إليها وإن…٤٧١٦ومنصور، وحميد، ح وحدثنا أبو كامل الجحدري، حدثنا حماد بن زيد، عن سماك…٤٧١٧إنا والله لا نولي على هذا العمل أحدا سأله ولا أحدا حرص عليه٤٧١٨ما تقول يا أبا موسى أو يا عبد الله بن قيس٤٧١٩يا أبا ذر إنك ضعيف وإنها أمانة وإنها يوم القيامة خزى وندامة إلا من…٤٧٢٠يا أبا ذر إني أراك ضعيفا وإني أحب لك ما أحب لنفسي لا تأمرن على اثنين…٤٧٢١إن المقسطين عند الله على منابر من نور عن يمين الرحمن عز وجل وكلتا…٤٧٢٢اللهم من ولي من أمر أمتي شيئا فشق عليهم فاشقق عليه ومن ولي من أمر…٤٧٢٣وحدثني محمد بن حاتم، حدثنا ابن مهدي، حدثنا جرير بن حازم، عن حرملة،…٤٧٢٤ألا كلكم راع وكلكم مسئول عن رعيته فالأمير الذي على الناس راع وهو…٤٧٢٧الرجل راع في مال أبيه ومسئول عن رعيته٤٧٢٨وحدثني أحمد بن عبد الرحمن بن وهب، أخبرني عمي عبد الله بن وهب، أخبرني…٤٧٢٩ما من عبد يسترعيه الله رعية يموت يوم يموت وهو غاش لرعيته إلا حرم الله…٤٧٣٠قال دخل ابن زياد على معقل بن يسار وهو وجع . بمثل حديث أبي الأشهب وزاد…٤٧٣١ما من أمير يلي أمر المسلمين ثم لا يجهد لهم وينصح إلا لم يدخل معهم…٤٧٣٢.٤٧٣٣إن شر الرعاء الحطمة فإياك أن تكون منهم٤٧٣٤لا ألفين أحدكم يجيء يوم القيامة على رقبته بعير له رغاء يقول يا رسول…٤٧٣٥وعمارة بن القعقاع، جميعا عن أبي زرعة، عن أبي هريرة، بمثل حديث إسماعيل…٤٧٣٦وحدثني أحمد بن سعيد بن صخر الدارمي، حدثنا سليمان بن حرب، حدثنا حماد،…٤٧٣٧وحدثني أحمد بن الحسن بن خراش، حدثنا أبو معمر، حدثنا عبد الوارث، حدثنا…٤٧٣٨ما بال عامل أبعثه فيقول هذا لكم وهذا أهدي لي . أفلا قعد في بيت أبيه…٤٧٣٩أفلا قعدت في بيت أبيك وأمك فتنظر أيهدى إليك أم لا٤٧٤٠فهلا جلست في بيت أبيك وأمك حتى تأتيك هديتك إن كنت صادقا٤٧٤١تعلمن والله والذي نفسي بيده لا يأخذ أحدكم منها شيئا٤٧٤٢وحدثناه إسحاق بن إبراهيم، أخبرنا جرير، عن الشيباني، عن عبد الله بن…٤٧٤٣من استعملناه منكم على عمل فكتمنا مخيطا فما فوقه كان غلولا يأتي به يوم…٤٧٤٤بهذا الإسناد بمثله .٤٧٤٥وحدثناه إسحاق بن إبراهيم الحنظلي، أخبرنا الفضل بن موسى، حدثنا…٤٧٤٦وهارون بن عبد الله، قالا حدثنا حجاج بن محمد، قال قال ابن جريج نزل {…٤٧٤٩من أطاعني فقد أطاع الله ومن عصاني فقد عصى الله ومن أطاع أميري فقد…٤٧٥٠أن أبا سلمة بن عبد الرحمن، أخبره أنه، سمع أبا هريرة، يقول قال رسول…٤٧٥١سمع أبا علقمة، سمع أبا هريرة، عن النبي صلى الله عليه وسلم نحو حديثهم٤٧٥٢وحدثنا محمد بن رافع، حدثنا عبد الرزاق، حدثنا معمر، عن همام بن منبه،…٤٧٥٣أن أبا يونس، مولى أبي هريرة حدثه قال سمعت أبا هريرة، يقول عن رسول…٤٧٥٤عليك السمع والطاعة في عسرك ويسرك ومنشطك ومكرهك وأثرة عليك٤٧٥٥وعبد الله بن براد الأشعري، وأبو كريب قالوا حدثنا ابن إدريس، عن شعبة،…٤٧٥٦بهذا الإسناد وقالا في الحديث عبدا حبشيا مجدع الأطراف .٤٧٥٧بهذا الإسناد كما قال ابن إدريس عبدا مجدع الأطراف .٤٧٥٨ولو استعمل عليكم عبد يقودكم بكتاب الله فاسمعوا له وأطيعوا٤٧٥٩وعبد الرحمن بن مهدي، عن شعبة، بهذا الإسناد وقال عبدا حبشيا .٤٧٦٠بهذا الإسناد وقال عبدا حبشيا مجدعا .٤٧٦١وحدثنا عبد الرحمن بن بشر، حدثنا بهز، حدثنا شعبة، بهذا الإسناد ولم…٤٧٦٢إن أمر عليكم عبد مجدع - حسبتها قالت - أسود يقودكم بكتاب الله فاسمعوا…٤٧٦٣على المرء المسلم السمع والطاعة فيما أحب وكره إلا أن يؤمر بمعصية فإن…٤٧٦٤ومحمد بن المثنى، قالا حدثنا يحيى، وهو القطان ح وحدثنا ابن نمير، حدثنا…٤٧٦٥لو دخلتموها لم تزالوا فيها إلى يوم القيامة٤٧٦٦لو دخلوها ما خرجوا منها إنما الطاعة في المعروف٤٧٦٧بهذا الإسناد نحوه .٤٧٦٨حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن إدريس، عن يحيى بن سعيد،…٤٧٦٩في هذا الإسناد مثله .٤٧٧٠وحدثنا ابن أبي عمر، حدثنا عبد العزيز، - يعني الدراوردي - عن يزيد، -…٤٧٧١إلا أن تروا كفرا بواحا عندكم من الله فيه برهان٤٧٧٢إنما الإمام جنة يقاتل من ورائه ويتقى به فإن أمر بتقوى الله عز وجل…٤٧٧٣كانت بنو إسرائيل تسوسهم الأنبياء كلما هلك نبي خلفه نبي وإنه لا نبي…٤٧٧٤وعبد الله بن براد الأشعري، قالا حدثنا عبد الله، بن إدريس عن الحسن بن…٤٧٧٥إنها ستكون بعدي أثرة وأمور تنكرونها٤٧٧٦إنه لم يكن نبي قبلي إلا كان حقا عليه أن يدل أمته على خير ما يعلمه لهم…٤٧٧٧وابن، نمير وأبو سعيد الأشج قالوا حدثنا وكيع، ح وحدثنا أبو كريب، حدثنا…٤٧٧٨رب الكعبة الصائدي قال رأيت جماعة عند الكعبة . فذكر نحو حديث الأعمش .٤٧٧٩إنكم ستلقون بعدي أثرة فاصبروا حتى تلقوني على الحوض٤٧٨٢اسمعوا وأطيعوا فإنما عليهم ما حملوا وعليكم ما حملتم٤٧٨٣اسمعوا وأطيعوا فإنما عليهم ما حملوا وعليكم ما حملتم٤٧٨٤أنه سمع أبا إدريس الخولاني، يقول سمعت حذيفة بن اليمان، يقول كان الناس…٤٧٨٥وحدثني محمد بن سهل بن عسكر التميمي، حدثنا يحيى بن حسان، ح وحدثنا عبد…٤٧٨٦من خرج من الطاعة وفارق الجماعة فمات مات ميتة جاهلية ومن قاتل تحت راية…٤٧٨٧لا يتحاشى من مؤمنها٤٧٨٨من خرج من الطاعة وفارق الجماعة ثم مات مات ميتة جاهلية ومن قتل تحت…٤٧٨٩وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن غيلان بن جرير،…٤٧٩٠من رأى من أميره شيئا يكرهه فليصبر فإنه من فارق الجماعة شبرا فمات…٤٧٩١من كره من أميره شيئا فليصبر عليه فإنه ليس أحد من الناس خرج من السلطان…٤٧٩٢من قتل تحت راية عمية يدعو عصبية أو ينصر عصبية فقتلة جاهلية٤٧٩٣من خلع يدا من طاعة لقي الله يوم القيامة لا حجة له ومن مات وليس في…٤٧٩٤وحدثنا ابن نمير، حدثنا يحيى بن عبد الله بن بكير، حدثنا ليث، عن عبيد…٤٧٩٥حدثنا عمرو بن علي، حدثنا ابن مهدي، ح وحدثنا محمد بن عمرو بن جبلة،…٤٧٩٦إنه ستكون هنات وهنات ، فمن أراد أن يفرق أمر هذه الأمة وهي جميع ،…٤٧٩٧ورجل، سماه كلهم عن زياد بن علاقة، عن عرفجة، عن النبي صلى الله عليه…٤٧٩٨من أتاكم وأمركم جميع على رجل واحد يريد أن يشق عصاكم أو يفرق جماعتكم…٤٧٩٩إذا بويع لخليفتين فاقتلوا الآخر منهما٤٨٠٠ستكون أمراء فتعرفون وتنكرون فمن عرف برئ ومن أنكر سلم ولكن من رضي وتابع٤٨٠١إنه يستعمل عليكم أمراء فتعرفون وتنكرون فمن كره فقد برئ ومن أنكر فقد…٤٨٠٢فمن أنكر فقد برئ ومن كره فقد سلم٤٨٠٣ولكن من رضي وتابع٤٨٠٤خيار أئمتكم الذين تحبونهم ويحبونكم ويصلون عليكم وتصلون عليهم وشرار…٤٨٠٥خيار أئمتكم الذين تحبونهم ويحبونكم وتصلون عليهم ويصلون عليكم وشرار…٤٨٠٦وحدثنا إسحاق بن موسى الأنصاري، حدثنا الوليد بن مسلم، حدثنا ابن جابر،…٤٨٠٧حدثنا قتيبة بن سعيد، حدثنا ليث بن سعد، ح وحدثنا محمد بن رمح، أخبرنا…٤٨٠٨قال لم نبايع رسول الله صلى الله عليه وسلم على
