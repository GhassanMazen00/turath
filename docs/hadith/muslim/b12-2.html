٢٢٩ حديثًا•صفحة ٢ من ٢
٢٤١٦وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، أخبرنا شعبة، قال سمعت…٢٤١٧لو كان لابن آدم واد من ذهب أحب أن له واديا آخر ولن يملأ فاه إلا…٢٤١٨لو أن لابن آدم ملء واد مالا لأحب أن يكون إليه مثله ولا يملأ نفس ابن…٢٤١٩حدثني سويد بن سعيد، حدثنا علي بن مسهر، عن داود، عن أبي حرب بن أبي،…٢٤٢٠ليس الغنى عن كثرة العرض ولكن الغنى غنى النفس٢٤٢١لا والله ما أخشى عليكم أيها الناس إلا ما يخرج الله لكم من زهرة الدنيا٢٤٢٢أخوف ما أخاف عليكم ما يخرج الله لكم من زهرة الدنيا٢٤٢٣إن مما أخاف عليكم بعدي ما يفتح عليكم من زهرة الدنيا وزينتها٢٤٢٤ما يكن عندي من خير فلن أدخره عنكم ومن يستعفف يعفه الله ومن يستغن يغنه…٢٤٢٥بهذا الإسناد . نحوه .٢٤٢٦قد أفلح من أسلم ورزق كفافا وقنعه الله بما آتاه٢٤٢٧اللهم اجعل رزق آل محمد قوتا٢٤٢٨إنهم خيروني أن يسألوني بالفحش أو يبخلوني فلست بباخل٢٤٢٩قال كنت أمشي مع رسول الله صلى الله عليه وسلم وعليه رداء نجراني غليظ…٢٤٣٠حدثنا زهير بن حرب، حدثنا عبد الصمد بن عبد الوارث، حدثنا همام، ح…٢٤٣١حدثنا قتيبة بن سعيد، حدثنا ليث، عن ابن أبي مليكة، عن المسور بن مخرمة،…٢٤٣٢خبأت هذا لك خبأت هذا لك٢٤٣٣وعبد بن حميد، قالا حدثنا يعقوب، - وهو ابن إبراهيم بن سعد - حدثنا أبي،…٢٤٣٤وعبد، بن حميد قالا أخبرنا عبد الرزاق، أخبرنا معمر، كلهم عن الزهري،…٢٤٣٥أقتالا أى سعد إني لأعطي الرجل٢٤٣٦ما حديث بلغني عنكم٢٤٣٧وعبد بن حميد، قالا حدثنا يعقوب، - وهو ابن إبراهيم بن سعد - حدثنا أبي،…٢٤٣٨وحدثني زهير بن حرب، حدثنا يعقوب بن إبراهيم، حدثنا ابن أخي ابن شهاب،…٢٤٣٩أفيكم أحد من غيركم٢٤٤٠ما الذي بلغني عنكم٢٤٤١وإبراهيم بن محمد بن عرعرة، - يزيد أحدهما على الآخر الحرف بعد الحرف -…٢٤٤٢يا للمهاجرين يا للمهاجرين٢٤٤٣قال أعطى رسول الله صلى الله عليه وسلم أبا سفيان بن حرب وصفوان بن أمية…٢٤٤٤بهذا الإسناد أن النبي صلى الله عليه وسلم قسم غنائم حنين فأعطى أبا…٢٤٤٥بهذا الإسناد ولم يذكر في الحديث علقمة بن علاثة ولا صفوان بن أمية ولم…٢٤٤٦يا معشر الأنصار ألم أجدكم ضلالا فهداكم الله بي وعالة فأغناكم الله بي…٢٤٤٧فمن يعدل إن لم يعدل الله ورسوله٢٤٤٨قد أوذي موسى بأكثر من هذا فصبر٢٤٤٩ويلك ومن يعدل إذا لم أكن أعدل لقد خبت وخسرت إن لم أكن أعدل٢٤٥٠أنه سمع جابر بن عبد الله، ح. وحدثنا أبو بكر بن أبي شيبة، حدثنا زيد بن…٢٤٥١إني إنما فعلت ذلك لأتألفهم٢٤٥٢ألا تأمنوني وأنا أمين من في السماء يأتيني خبر السماء صباحا ومساء٢٤٥٣حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن عمارة بن القعقاع، بهذا…٢٤٥٤لئن أدركتهم لأقتلنهم قتل ثمود٢٤٥٥يخرج في هذه الأمة - ولم يقل منها - قوم تحقرون صلاتكم مع صلاتهم…٢٤٥٦ويلك ومن يعدل إن لم أعدل قد خبت وخسرت إن لم أعدل٢٤٥٧هم شر الخلق - أو من أشر الخلق - يقتلهم أدنى الطائفتين إلى الحق٢٤٥٨تمرق مارقة عند فرقة من المسلمين يقتلها أولى الطائفتين بالحق٢٤٥٩تكون في أمتي فرقتان فتخرج من بينهما مارقة يلي قتلهم أولاهم بالحق٢٤٦٠تمرق مارقة في فرقة من الناس فيلي قتلهم أولى الطائفتين بالحق٢٤٦١حدثني عبيد الله القواريري، حدثنا محمد بن عبد الله بن الزبير، حدثنا…٢٤٦٢سيخرج في آخر الزمان قوم أحداث الأسنان سفهاء الأحلام يقولون من خير قول…٢٤٦٣بهذا الإسناد . مثله .٢٤٦٤يمرقون من الدين كما يمرق السهم من الرمية٢٤٦٥وحماد بن زيد، ح وحدثنا قتيبة بن سعيد، حدثنا حماد بن زيد، ح وحدثنا أبو…٢٤٦٦نحو حديث أيوب مرفوعا .٢٤٦٧يخرج قوم من أمتي يقرءون القرآن ليس قراءتكم إلى قراءتهم بشىء ولا…٢٤٦٨يقولون الحق بألسنتهم لا يجوز هذا منهم - وأشار إلى حلقه - من أبغض خلق…٢٤٦٩إن بعدي من أمتي - أو سيكون بعدي من أمتي - قوم يقرءون القرآن لا يجاوز…٢٤٧٠قوم يقرءون القرآن بألسنتهم لا يعدو تراقيهم يمرقون من الدين كما يمرق…٢٤٧١بهذا الإسناد وقال يخرج منه أقوام .٢٤٧٢يتيه قوم قبل المشرق محلقة رءوسهم٢٤٧٣كخ كخ ارم بها أما علمت أنا لا نأكل الصدقة٢٤٧٤أنا لا، تحل لنا الصدقة٢٤٧٥أنا لا، نأكل الصدقة٢٤٧٦إني لأنقلب إلى أهلي فأجد التمرة ساقطة على فراشي ثم أرفعها لآكلها ثم…٢٤٧٧والله إني لأنقلب إلى أهلي فأجد التمرة ساقطة على فراشي - أو في بيتي -…٢٤٧٨لولا أن تكون من الصدقة لأكلتها٢٤٧٩لولا أن تكون من الصدقة لأكلتها٢٤٨٠لولا أن تكون صدقة لأكلتها٢٤٨١أخرجا ما تصرران٢٤٨٢إن هذه الصدقات إنما هي أوساخ الناس وإنها لا تحل لمحمد ولا لآل محمد٢٤٨٣أن عبيد بن السباق، قال إن جويرية زوج النبي صلى الله عليه وسلم أخبرته…٢٤٨٤وعمرو الناقد، وإسحاق بن إبراهيم، جميعا عن ابن عيينة، عن الزهري، بهذا…٢٤٨٥هو لها صدقة ولنا هدية٢٤٨٦هو لها صدقة ولنا هدية٢٤٨٧هو عليها صدقة ولكم هدية فكلوه٢٤٨٨وحدثنا أبو بكر بن أبي شيبة، حدثنا حسين بن علي، عن زائدة، عن سماك، عن…٢٤٨٩وهو لنا منها هدية٢٤٩٠حدثني زهير بن حرب، حدثنا إسماعيل بن إبراهيم، عن خالد، عن حفصة، عن أم…٢٤٩١. أن النبي صلى الله عليه وسلم كان إذا أتي بطعام سأل عنه فإن قيل هدية…٢٤٩٢اللهم صل عليهم٢٤٩٣بهذا الإسناد غير أنه قال " صل عليهم " .٢٤٩٤إذا أتاكم المصدق فليصدر عنكم وهو عنكم راض
