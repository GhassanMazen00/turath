٥٩ حديثًا
٥٠٦٤من كان ذبح أضحيته قبل أن يصلي - أو نصلي - فليذبح مكانها أخرى ومن كان…٥٠٦٥من ذبح قبل الصلاة فليذبح شاة مكانها ومن لم يكن ذبح فليذبح على اسم الله٥٠٦٦وابن، أبي عمر عن ابن عيينة، كلاهما عن الأسود بن قيس، بهذا الإسناد…٥٠٦٧من كان ذبح قبل أن يصلي فليعد مكانها ومن لم يكن ذبح فليذبح باسم الله٥٠٦٨وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، بهذا الإسناد مثله .٥٠٦٩قال ضحى خالي أبو بردة قبل الصلاة فقال رسول الله صلى الله عليه وسلم "…٥٠٧٠أن خاله أبا بردة بن نيار، ذبح قبل أن يذبح النبي، صلى الله عليه وسلم…٥٠٧١لا يذبحن أحد حتى يصلي٥٠٧٢من صلى صلاتنا ووجه قبلتنا ونسك نسكنا فلا يذبح حتى يصلي٥٠٧٣إن أول ما نبدأ به في يومنا هذا نصلي ثم نرجع فننحر فمن فعل ذلك فقد…٥٠٧٤سمع الشعبي، عن البراء بن عازب، عن النبي صلى الله عليه وسلم مثله .٥٠٧٥وهناد بن السري، قالا حدثنا أبو الأحوص، ح وحدثنا عثمان بن أبي شيبة،…٥٠٧٦لا يضحين أحد حتى يصلي٥٠٧٧حدثنا محمد بن بشار، حدثنا محمد، - يعني ابن جعفر - حدثنا شعبة، عن…٥٠٧٨بهذا الإسناد . ولم يذكر الشك في قوله هي خير من مسنة .٥٠٧٩من كان ذبح قبل الصلاة فليعد٥٠٨٠وهشام، عن محمد، عن أنس بن مالك، أن رسول الله صلى الله عليه وسلم صلى…٥٠٨١من كان ضحى فليعد٥٠٨٢لا تذبحوا إلا مسنة إلا أن يعسر عليكم فتذبحوا جذعة من الضأن٥٠٨٣وحدثني محمد بن حاتم، حدثنا محمد بن بكر، أخبرنا ابن جريج، أخبرني أبو…٥٠٨٤أن رسول الله صلى الله عليه وسلم أعطاه غنما يقسمها على أصحابه ضحايا…٥٠٨٥حدثنا أبو بكر بن أبي شيبة، حدثنا يزيد بن هارون، عن هشام الدستوائي، عن…٥٠٨٦الله أن عقبة بن عامر الجهني، أخبره أن رسول الله صلى الله عليه وسلم…٥٠٨٧قال ضحى النبي صلى الله عليه وسلم بكبشين أملحين أقرنين ذبحهما بيده…٥٠٨٨قال ضحى رسول الله صلى الله عليه وسلم بكبشين أملحين أقرنين قال ورأيته…٥٠٨٩وحدثنا يحيى بن حبيب، حدثنا خالد، - يعني ابن الحارث - حدثنا شعبة،…٥٠٩٠باسم الله والله أكبر٥٠٩١يا عائشة هلمي المدية٥٠٩٢أعجل أو أرني ما أنهر الدم وذكر اسم الله فكل ليس السن والظفر وسأحدثك…٥٠٩٣وحدثنا إسحاق بن إبراهيم، أخبرنا وكيع، حدثنا سفيان بن سعيد بن مسروق،…٥٠٩٤وحدثنا ابن أبي عمر، حدثنا سفيان، عن إسماعيل بن مسلم، عن سعيد بن…٥٠٩٥وحدثنيه القاسم بن زكرياء، حدثنا حسين بن علي، عن زائدة، عن سعيد بن…٥٠٩٦وحدثنا محمد بن الوليد بن عبد الحميد، حدثنا محمد بن جعفر، حدثنا شعبة،…٥٠٩٧حدثني عبد الجبار بن العلاء، حدثنا سفيان، حدثنا الزهري، عن أبي عبيد،…٥٠٩٨حدثني حرملة بن يحيى، أخبرنا ابن وهب، حدثني يونس، عن ابن شهاب، حدثني…٥٠٩٩بهذا الإسناد مثله .٥١٠٠لا يأكل أحد من لحم أضحيته فوق ثلاثة أيام٥١٠١وحدثني محمد بن حاتم، حدثنا يحيى بن سعيد، عن ابن جريج، ح وحدثني محمد،…٥١٠٢وعبد بن حميد، قال ابن أبي عمر حدثنا وقال عبد، أخبرنا عبد الرزاق،…٥١٠٣ادخروا ثلاثا ثم تصدقوا بما بقي٥١٠٤كلوا وتزودوا وادخروا٥١٠٥حدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، ح وحدثنا يحيى بن أيوب،…٥١٠٦حدثنا إسحاق بن إبراهيم، أخبرنا زكرياء بن عدي، عن عبيد الله بن عمرو،…٥١٠٧وحدثنا أبو بكر بن أبي شيبة، حدثنا سفيان بن عيينة، عن عمرو، عن عطاء،…٥١٠٨يا أهل المدينة لا تأكلوا لحوم الأضاحي فوق ثلاث٥١٠٩من ضحى منكم فلا يصبحن في بيته بعد ثالثة شيئا٥١١٠يا ثوبان أصلح لحم هذه٥١١١وابن، رافع قالا حدثنا زيد بن حباب، ح وحدثنا إسحاق بن إبراهيم الحنظلي،…٥١١٦لا فرع ولا عتيرة٥١١٧نهيتكم عن زيارة القبور فزوروها ونهيتكم عن لحوم الأضاحي فوق ثلاث…٥١١٨أن رسول الله صلى الله عليه وسلم قال " كنت نهيتكم " . فذكر بمعنى حديث…٥١١٩إذا دخلت العشر وأراد أحدكم أن يضحي فلا يمس من شعره وبشره شيئا٥١٢٠إذا دخل العشر وعنده أضحية يريد أن يضحي فلا يأخذن شعرا ولا يقلمن ظفرا٥١٢١إذا رأيتم هلال ذي الحجة وأراد أحدكم أن يضحي فليمسك عن شعره وأظفاره٥١٢٢حدثني الحسن بن علي الحلواني، حدثنا أبو أسامة، حدثني محمد بن عمرو،…٥١٢٣وأحمد بن عبد الرحمن بن أخي ابن وهب، قالا حدثنا عبد الله بن وهب،…٥١٢٤لعن الله من لعن والده ولعن الله من ذبح لغير الله ولعن الله من آوى…٥١٢٥لعن الله من ذبح لغير الله ولعن الله من آوى محدثا ولعن الله من لعن…٥١٢٦لعن الله من ذبح لغير الله ولعن الله من سرق منار الأرض ولعن الله من…
