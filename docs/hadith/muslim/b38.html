٥٩ حديثًا
٥٥٨٦تسموا باسمي ولا تكنوا بكنيتي٥٥٨٧إن أحب أسمائكم إلى الله عبد الله وعبد الرحمن٥٥٨٨تسموا باسمي ولا تكتنوا بكنيتي فإنما أنا قاسم أقسم بينكم٥٥٨٩سموا باسمي ولا تكنوا بكنيتي فإنما بعثت قاسما أقسم بينكم٥٥٩٠فإنما بعثت قاسما أقسم بينكم٥٥٩١تسموا باسمي ولا تكنوا بكنيتي فإني أنا أبو القاسم أقسم بينكم٥٥٩٢إنما جعلت قاسما أقسم بينكم٥٥٩٣أحسنت الأنصار سموا باسمي ولا تكتنوا بكنيتي٥٥٩٤إنما بعثت قاسما أقسم بينكم٥٥٩٥أسم ابنك عبد الرحمن٥٥٩٦وحدثني أمية بن بسطام، حدثنا يزيد يعني ابن زريع، ح وحدثني علي بن حجر،…٥٥٩٧تسموا باسمي ولا تكنوا بكنيتي٥٥٩٨إنهم كانوا يسمون بأنبيائهم والصالحين قبلهم٥٥٩٩وقال، يحيى أخبرنا المعتمر بن سليمان، قال سمعت الركين، يحدث عن أبيه،…٥٦٠٠لا تسم غلامك رباحا ولا يسارا ولا أفلح ولا نافعا٥٦٠١أحب الكلام إلى الله أربع سبحان الله والحمد لله ولا إله إلا الله والله…٥٦٠٢وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، كلهم عن منصور، بإسناد…٥٦٠٣حدثنا محمد بن أحمد بن أبي خلف، حدثنا روح، حدثنا ابن جريج، أخبرني أبو…٥٦٠٤وزهير بن حرب، ومحمد بن المثنى، وعبيد الله بن سعيد، ومحمد بن بشار…٥٦٠٥أن ابنة لعمر، كانت يقال لها عاصية فسماها رسول الله صلى الله عليه وسلم…٥٦٠٦وابن أبي عمر، - واللفظ لعمرو - قالا حدثنا سفيان، عن محمد بن عبد…٥٦٠٧ومحمد بن المثنى، ومحمد بن بشار، قالوا حدثنا محمد بن جعفر، حدثنا شعبة،…٥٦٠٨حدثتني زينب، بنت أم سلمة قالت كان اسمي برة فسماني رسول الله صلى الله…٥٦٠٩لا تزكوا أنفسكم الله أعلم بأهل البر منكم٥٦١٠إن أخنع اسم عند الله رجل تسمى ملك الأملاك٥٦١١أغيظ رجل على الله يوم القيامة وأخبثه وأغيظه عليه رجل كان يسمى ملك…٥٦١٢بن مالك قال ذهبت بعبد الله بن أبي طلحة الأنصاري إلى رسول الله صلى…٥٦١٣قال كان ابن لأبي طلحة يشتكي فخرج أبو طلحة فقبض الصبي فلما رجع أبو…٥٦١٤بهذه القصة نحو حديث يزيد .٥٦١٥وعبد الله بن براد الأشعري، وأبو كريب قالوا حدثنا أبو أسامة، عن بريد،…٥٦١٦وفاطمة بنت المنذر بن الزبير، أنهما قالا خرجت أسماء بنت أبي بكر حين…٥٦١٧فأتيت المدينة فنزلت بقباء فولدته بقباء ثم أتيت رسول الله صلى الله…٥٦١٨أنها هاجرت إلى رسول الله صلى الله عليه وسلم وهي حبلى بعبد الله بن…٥٦١٩أن رسول الله صلى الله عليه وسلم كان يؤتى بالصبيان فيبرك عليهم ويحنكهم…٥٦٢٠حدثنا أبو بكر بن أبي شيبة، حدثنا أبو خالد الأحمر، عن هشام، عن أبيه،…٥٦٢١قال أتي بالمنذر بن أبي أسيد إلى رسول الله صلى الله عليه وسلم حين ولد…٥٦٢٢أبا عمير ما فعل النغير٥٦٢٣مالك قال قال لي رسول الله صلى الله عليه وسلم " يا بنى " .٥٦٢٤أى بنى وما ينصبك منه إنه لن يضرك٥٦٢٥وابن، نمير قالا حدثنا وكيع، ح وحدثنا سريج بن، يونس حدثنا هشيم، ح…٥٦٢٦إذا استأذن أحدكم ثلاثا فلم يؤذن له فليرجع٥٦٢٧وابن أبي عمر، قالا حدثنا سفيان، عن يزيد بن خصيفة، بهذا الإسناد . وزاد…٥٦٢٨الاستئذان ثلاث فإن أذن لك وإلا فارجع٥٦٢٩أن أبا موسى، أتى باب عمر فاستأذن فقال عمر واحدة . ثم استأذن الثانية…٥٦٣٠وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي مسلمة، عن أبي…٥٦٣١أن أبا موسى، استأذن على عمر ثلاثا فكأنه وجده مشغولا فرجع فقال عمر ألم…٥٦٣٢بهذا الإسناد نحوه ولم يذكر في حديث النضر ألهاني عنه الصفق بالأسواق .٥٦٣٣الاستئذان ثلاث فإن أذن لك وإلا فارجع٥٦٣٤حدثنا محمد بن عبد الله بن نمير، حدثنا عبد الله بن إدريس، عن شعبة، عن…٥٦٣٥حدثنا يحيى بن يحيى، وأبو بكر بن أبي شيبة - واللفظ لأبي بكر - قال يحيى…٥٦٣٦بهذا الإسناد . وفي حديثهم كأنه كره ذلك .٥٦٣٨لو أعلم أنك تنظرني لطعنت به في عينك٥٦٣٩لو أعلم أنك تنظر طعنت به في عينك إنما جعل الله الإذن من أجل البصر٥٦٤٠وعمرو الناقد، وزهير بن حرب، وابن أبي عمر، قالوا حدثنا سفيان بن عيينة،…٥٦٤١أن رجلا، اطلع من بعض حجر النبي صلى الله عليه وسلم فقام إليه بمشقص أو…٥٦٤٢من اطلع في بيت قوم بغير إذنهم فقد حل لهم أن يفقئوا عينه٥٦٤٣لو أن رجلا اطلع عليك بغير إذن فخذفته بحصاة ففقأت عينه ما كان عليك من…٥٦٤٤حدثني قتيبة بن سعيد، حدثنا يزيد بن زريع، ح وحدثنا أبو بكر بن أبي…٥٦٤٥وقال، إسحاق أخبرنا وكيع، حدثنا سفيان، كلاهما عن يونس، بهذا الإسناد…
