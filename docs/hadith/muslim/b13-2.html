كتاب الصيام

٢٨٥ حديثًا•صفحة ٢ من ٢
٢٦٤٥إن هذا يوم كان يصومه أهل الجاهلية فمن أحب أن يصومه فليصمه ومن أحب أن…٢٦٤٦- رضى الله عنهما - قال ذكر عند النبي صلى الله عليه وسلم صوم يوم…٢٦٤٧ذاك يوم كان يصومه أهل الجاهلية فمن شاء صامه ومن شاء تركه٢٦٤٨قال دخل الأشعث بن قيس على عبد الله وهو يتغدى فقال يا أبا محمد ادن إلى…٢٦٤٩وعثمان بن أبي شيبة، قالا حدثنا جرير، عن الأعمش، بهذا الإسناد وقالا…٢٦٥٠ويحيى بن سعيد القطان، عن سفيان، ح وحدثني محمد بن حاتم، - واللفظ له -…٢٦٥١قال دخل الأشعث بن قيس على ابن مسعود وهو يأكل يوم عاشوراء فقال يا أبا…٢٦٥٢حدثنا أبو بكر بن أبي شيبة، حدثنا عبيد الله بن موسى، أخبرنا شيبان، عن…٢٦٥٣هذا يوم عاشوراء ولم يكتب الله عليكم صيامه وأنا صائم فمن أحب منكم أن…٢٦٥٤شهاب في هذا الإسناد بمثله .٢٦٥٥إني صائم فمن شاء أن يصوم فليصم٢٦٥٦نحن أولى بموسى منكم٢٦٥٧.٢٦٥٨ما هذا اليوم الذي تصومونه٢٦٥٩لم يسمه .٢٦٦٠وابن، نمير قالا حدثنا أبو أسامة، عن أبي عميس، عن قيس بن مسلم، عن طارق…٢٦٦١وحدثناه أحمد بن المنذر، حدثنا حماد بن أسامة، حدثنا أبو العميس، أخبرني…٢٦٦٢وعمرو الناقد، جميعا عن سفيان، - قال أبو بكر حدثنا ابن عيينة، - عن…٢٦٦٣الله بن أبي يزيد في هذا الإسناد بمثله .٢٦٦٤وحدثنا أبو بكر بن أبي شيبة، حدثنا وكيع بن الجراح، عن حاجب بن عمر، عن…٢٦٦٥وحدثني محمد بن حاتم، حدثنا يحيى بن سعيد القطان، عن معاوية بن عمرو،…٢٦٦٦فإذا كان العام المقبل - إن شاء الله - صمنا اليوم التاسع٢٦٦٧لئن بقيت إلى قابل لأصومن التاسع٢٦٦٨من كان لم يصم فليصم ومن كان أكل فليتم صيامه إلى الليل٢٦٦٩من كان أصبح صائما فليتم صومه ومن كان أصبح مفطرا فليتم بقية يومه٢٦٧٠وحدثناه يحيى بن يحيى، حدثنا أبو معشر العطار، عن خالد بن ذكوان، قال…٢٦٧١وحدثنا يحيى بن يحيى قال قرأت على مالك عن ابن شهاب عن أبي عبيد مولى…٢٦٧٢وحدثنا يحيى بن يحيى، قال قرأت على مالك عن محمد بن يحيى بن حبان، عن…٢٦٧٣لا يصلح الصيام في يومين يوم الأضحى ويوم الفطر من رمضان٢٦٧٤وحدثنا أبو كامل الجحدري، حدثنا عبد العزيز بن المختار، حدثنا عمرو بن…٢٦٧٥وحدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن ابن عون، عن زياد بن جبير،…٢٦٧٦أخبرتني عمرة، عن عائشة، - رضى الله عنها - قالت نهى رسول الله صلى الله…٢٦٧٧أيام التشريق أيام أكل وشرب٢٦٧٨حدثنا محمد بن عبد الله بن نمير، حدثنا إسماعيل، - يعني ابن علية - عن…٢٦٧٩أنه لا يدخل الجنة إلا مؤمن . وأيام منى أيام أكل وشرب٢٦٨٠بن طهمان بهذا الإسناد غير أنه قال فناديا .٢٦٨١حدثنا عمرو الناقد، حدثنا سفيان بن عيينة، عن عبد الحميد بن جبير، عن…٢٦٨٢بن جبير بن شيبة أنه أخبره محمد بن عباد بن جعفر، أنه سأل جابر بن عبد…٢٦٨٣لا يصم أحدكم يوم الجمعة إلا أن يصوم قبله أو يصوم بعده٢٦٨٤لا تختصوا ليلة الجمعة بقيام من بين الليالي ولا تخصوا يوم الجمعة بصيام…٢٦٨٥- رضى الله عنه - قال لما نزلت هذه الآية { وعلى الذين يطيقونه فدية…٢٦٨٦حدثني عمرو بن سواد العامري، أخبرنا عبد الله بن وهب، أخبرنا عمرو بن،…٢٦٨٧- رضى الله عنها - تقول كان يكون على الصوم من رمضان فما أستطيع أن…٢٦٨٨بهذا الإسناد غير أنه قال وذلك لمكان رسول الله صلى الله عليه وسلم .٢٦٨٩بن سعيد بهذا الإسناد وقال فظننت أن ذلك لمكانها من النبي صلى الله عليه…٢٦٩٠بهذا الإسناد ولم يذكرا في الحديث الشغل برسول الله صلى الله عليه وسلم .٢٦٩١وحدثني محمد بن أبي عمر المكي، حدثنا عبد العزيز بن محمد الدراوردي، عن…٢٦٩٢من مات وعليه صيام صام عنه وليه٢٦٩٣أرأيت لو كان عليها دين أكنت تقضينه٢٦٩٤لو كان على أمك دين أكنت قاضيه عنها٢٦٩٥ومجاهد، وعطاء، عن ابن، عباس - رضى الله عنهما - عن النبي صلى الله عليه…٢٦٩٦أرأيت لو كان على أمك دين فقضيتيه أكان يؤدي ذلك عنها٢٦٩٧وجب أجرك وردها عليك الميراث٢٦٩٨- رضى الله عنه - قال كنت جالسا عند النبي صلى الله عليه وسلم . بمثل…٢٦٩٩- رضى الله عنه - قال جاءت امرأة إلى النبي صلى الله عليه وسلم . فذكر…٢٧٠٠بهذا الإسناد وقال صوم شهرين .٢٧٠١- رضى الله عنه - قال أتت امرأة إلى النبي صلى الله عليه وسلم . بمثل…٢٧٠٢إذا دعي أحدكم إلى طعام وهو صائم فليقل إني صائم٢٧٠٣إذا أصبح أحدكم يوما صائما فلا يرفث ولا يجهل فإن امرؤ شاتمه أو قاتله…٢٧٠٤قال الله عز وجل كل عمل ابن آدم له إلا الصيام هو لي وأنا أجزي به…٢٧٠٥وقتيبة بن سعيد، قالا حدثنا المغيرة، - وهو الحزامي - عن أبي الزناد، عن…٢٧٠٦قال الله عز وجل كل عمل ابن آدم له إلا الصيام فإنه لي وأنا أجزي به…٢٧٠٧كل عمل ابن آدم يضاعف الحسنة عشر أمثالها إلى سبعمائة ضعف قال الله عز…٢٧٠٨إن الله عز وجل يقول إن الصوم لي وأنا أجزي به إن للصائم فرحتين إذا…٢٧٠٩إذا لقي الله فجزاه فرح٢٧١٠إن في الجنة بابا يقال له الريان يدخل منه الصائمون يوم القيامة لا يدخل…٢٧١١ما من عبد يصوم يوما في سبيل الله إلا باعد الله بذلك اليوم وجهه عن…٢٧١٢بهذا الإسناد .٢٧١٣من صام يوما في سبيل الله باعد الله وجهه عن النار سبعين خريفا٢٧١٤يا عائشة هل عندكم شىء٢٧١٥وحدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن طلحة بن يحيى، عن عمته،…٢٧١٦من نسي وهو صائم فأكل أو شرب فليتم صومه فإنما أطعمه الله وسقاه٢٧١٧بن شقيق قال قلت لعائشة رضى الله عنها هل كان النبي صلى الله عليه وسلم…٢٧١٨قال قلت لعائشة - رضى الله عنها - أكان رسول الله صلى الله عليه وسلم…٢٧١٩وهشام، عن محمد، عن عبد الله بن شقيق، - قال حماد وأظن أيوب قد سمعه من…٢٧٢٠قال سألت عائشة - رضى الله عنها - بمثله ولم يذكر في الإسناد هشاما ولا…٢٧٢١- رضى الله عنها - أنها قالت كان رسول الله صلى الله عليه وسلم يصوم حتى…٢٧٢٢وعمرو الناقد، جميعا عن ابن عيينة، - قال أبو بكر حدثنا سفيان بن عيينة،…٢٧٢٣خذوا من الأعمال ما تطيقون فإن الله لن يمل حتى تملوا٢٧٢٤- رضى الله عنهما - قال ما صام رسول الله صلى الله عليه وسلم شهرا كاملا…٢٧٢٥وحدثنا محمد بن بشار وأبو بكر بن نافع عن غندر عن شعبة عن أبي بشر بهذا…٢٧٢٦حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، ح وحدثنا ابن نمير،…٢٧٢٧في هذا الإسناد . بمثله .٢٧٢٨وابن أبي خلف، قالا حدثنا روح بن عبادة، حدثنا حماد، عن ثابت، عن أنس،…٢٧٢٩آنت الذي تقول ذلك٢٧٣٠ألم أخبر أنك تصوم الدهر وتقرأ القرآن كل ليلة٢٧٣١من كل شهر ثلاثة أيام٢٧٣٢اقرإ القرآن في كل شهر٢٧٣٣يا عبد الله لا تكن بمثل فلان كان يقوم الليل فترك قيام الليل٢٧٣٤ألم أخبر أنك تصوم ولا تفطر وتصلي الليل فلا تفعل فإن لعينك حظا ولنفسك…٢٧٣٥بهذا الإسناد وقال إن أبا العباس الشاعر أخبره . قال مسلم أبو العباس…٢٧٣٦يا عبد الله بن عمرو إنك لتصوم الدهر وتقوم الليل وإنك إذا فعلت ذلك…٢٧٣٧بهذا الإسناد وقال " ونفهت النفس " .٢٧٣٨ألم أخبر أنك تقوم الليل وتصوم النهار٢٧٣٩إن أحب الصيام إلى الله صيام داود وأحب الصلاة إلى الله صلاة داود عليه…٢٧٤٠أحب الصيام إلى الله صيام داود كان يصوم نصف الدهر وأحب الصلاة إلى الله…٢٧٤١أما يكفيك من كل شهر ثلاثة أيام٢٧٤٢صم يوما ولك أجر ما بقي٢٧٤٣يا عبد الله بن عمرو بلغني أنك تصوم النهار وتقوم الليل فلا تفعل فإن…٢٧٤٤قال حدثتني معاذة، العدوية أنها سألت عائشة زوج النبي صلى الله عليه…٢٧٤٥يا فلان أصمت من سرة هذا الشهر٢٧٤٦لا صام ولا أفطر - أو قال - لم يصم ولم يفطر٢٧٤٧ومحمد بن بشار، - واللفظ لابن المثنى - قالا حدثنا محمد بن جعفر، حدثنا…٢٧٤٨بهذا الإسناد .٢٧٤٩في هذا الإسناد بمثل حديث شعبة غير أنه ذكر فيه الاثنين ولم يذكر الخميس…٢٧٥٠فيه ولدت وفيه أنزل على٢٧٥١أصمت من سرر شعبان٢٧٥٢هل صمت من سرر هذا الشهر شيئا٢٧٥٣هل صمت من سرر هذا الشهر شيئا٢٧٥٤ويحيى اللؤلؤي، قالا أخبرنا النضر، أخبرنا شعبة، حدثنا عبد الله بن هانئ…٢٧٥٥أفضل الصيام بعد رمضان شهر الله المحرم وأفضل الصلاة بعد الفريضة صلاة…٢٧٥٦أفضل الصلاة بعد الصلاة المكتوبة الصلاة في جوف الليل وأفضل الصيام بعد…٢٧٥٧وحدثنا أبو بكر بن أبي شيبة، حدثنا حسين بن علي، عن زائدة، عن عبد…٢٧٥٨من صام رمضان ثم أتبعه ستا من شوال كان كصيام الدهر٢٧٥٩وحدثنا ابن نمير، حدثنا أبي، حدثنا سعد بن سعيد، أخو يحيى بن سعيد…٢٧٦٠- رضى الله عنه - يقول قال رسول الله صلى الله عليه وسلم بمثله .٢٧٦١أرى رؤياكم قد تواطأت في السبع الأواخر فمن كان متحريها فليتحرها في…٢٧٦٢تحروا ليلة القدر في السبع الأواخر٢٧٦٣أرى رؤياكم في العشر الأواخر فاطلبوها في الوتر منها٢٧٦٤إن ناسا منكم قد أروا أنها في السبع الأول وأري ناس منكم أنها في السبع…٢٧٦٥التمسوها في العشر الأواخر - يعني ليلة القدر - فإن ضعف أحدكم أو عجز…٢٧٦٦من كان ملتمسها فليلتمسها في العشر الأواخر٢٧٦٧تحينوا ليلة القدر في العشر الأواخر٢٧٦٨أريت ليلة القدر ثم أيقظني بعض أهلي فنسيتها فالتمسوها في العشر الغوابر٢٧٦٩إني كنت أجاور هذه العشر ثم بدا لي أن أجاور هذه العشر الأواخر فمن كان…٢٧٧٠فليثبت في معتكفه٢٧٧١إني اعتكفت العشر الأول ألتمس هذه الليلة ثم اعتكفت العشر الأوسط ثم…٢٧٧٢إني أريت ليلة القدر وإني نسيتها - أو أنسيتها - فالتمسوها في العشر…٢٧٧٣كثير بهذا الإسناد . نحوه وفي حديثهما رأيت رسول الله صلى الله عليه…٢٧٧٤يا أيها الناس إنها كانت أبينت لي ليلة القدر وإني خرجت لأخبركم بها…٢٧٧٥أريت ليلة القدر ثم أنسيتها وأراني صبحها أسجد في ماء وطين٢٧٧٦التمسوا - وقال وكيع - تحروا ليلة القدر في العشر الأواخر من رمضان٢٧٧٧وابن أبي عمر، كلاهما عن ابن عيينة، - قال ابن حاتم حدثنا سفيان بن…٢٧٧٨وحدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت عبدة،…٢٧٧٩أيكم يذكر حين طلع القمر وهو مثل شق جفنة
