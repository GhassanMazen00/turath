كتاب الجنائز

١٣٥ حديثًا
٢١٢٣لقنوا موتاكم لا إله إلا الله٢١٢٤جميعا بهذا الإسناد .٢١٢٥لقنوا موتاكم لا إله إلا الله٢١٢٦ما من مسلم تصيبه مصيبة فيقول ما أمره الله إنا لله وإنا إليه راجعون…٢١٢٧ما من عبد تصيبه مصيبة فيقول إنا لله وإنا إليه راجعون اللهم أجرني في…٢١٢٨وحدثنا محمد بن عبد الله بن نمير، حدثنا أبي، حدثنا سعد بن سعيد، أخبرني…٢١٢٩إذا حضرتم المريض أو الميت فقولوا خيرا فإن الملائكة يؤمنون على ما…٢١٣٠إن الروح إذا قبض تبعه البصر٢١٣١واخلفه في تركته٢١٣٢ألم تروا الإنسان إذا مات شخص بصره٢١٣٣بهذا الإسناد .٢١٣٤أتريدين أن تدخلي الشيطان بيتا أخرجه الله منه٢١٣٥ارجع إليها فأخبرها إن لله ما أخذ وله ما أعطى وكل شىء عنده بأجل مسمى…٢١٣٦بهذا الإسناد . غير أن حديث حماد أتم وأطول .٢١٣٧وعمرو بن سواد العامري، قالا أخبرنا عبد الله بن وهب، أخبرني عمرو بن…٢١٣٨يا أخا الأنصار كيف أخي سعد بن عبادة٢١٣٩الصبر عند الصدمة الأولى٢١٤٠اتقي الله واصبري٢١٤١بهذا الإسناد نحو حديث عثمان بن عمر بقصته . وفي حديث عبد الصمد مر…٢١٤٢إن الميت يعذب ببكاء أهله عليه٢١٤٣الميت يعذب في قبره بما نيح عليه٢١٤٤الميت يعذب في قبره بما نيح عليه٢١٤٥إن الميت ليعذب ببكاء الحى٢١٤٦إن الميت ليعذب ببكاء الحى٢١٤٧من يبكى عليه يعذب٢١٤٨المعول عليه يعذب٢١٥١وحدثنا عبد الرحمن بن بشر، حدثنا سفيان، قال عمرو عن ابن أبي مليكة، كنا…٢١٥٢إن الميت يعذب ببكاء الحى٢١٥٣أنتم تبكون وإنه ليعذب٢١٥٤إن الميت يعذب في قبره ببكاء أهله عليه٢١٥٥بهذا الإسناد بمعنى حديث أبي أسامة وحديث أبي أسامة أتم .٢١٥٦إنهم ليبكون عليها وإنها لتعذب في قبرها٢١٥٧من نيح عليه فإنه يعذب بما نيح عليه يوم القيامة٢١٥٨وحدثني علي بن حجر السعدي، حدثنا علي بن مسهر، أخبرنا محمد بن قيس،…٢١٥٩وحدثناه ابن أبي عمر، حدثنا مروان، - يعني الفزاري - حدثنا سعيد بن…٢١٦٠أربع في أمتي من أمر الجاهلية لا يتركونهن الفخر في الأحساب والطعن في…٢١٦١اذهب فاحث في أفواههن من التراب٢١٦٢بهذا الإسناد . نحوه . وفي حديث عبد العزيز وما تركت رسول الله صلى الله…٢١٦٣حدثني أبو الربيع الزهراني، حدثنا حماد، حدثنا أيوب، عن محمد، عن أم…٢١٦٤حدثنا إسحاق بن إبراهيم، أخبرنا أسباط، حدثنا هشام، عن حفصة، عن أم…٢١٦٥وزهير بن حرب، وإسحاق بن إبراهيم، جميعا عن أبي معاوية، - قال زهير…٢١٦٦حدثنا يحيى بن أيوب، حدثنا ابن علية، أخبرنا أيوب، عن محمد بن سيرين،…٢١٦٧وحدثنا أبو بكر بن أبي شيبة، حدثنا أبو أسامة، ح وحدثنا إسحاق بن…٢١٦٨اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك بماء وسدر واجعلن في…٢١٦٩قالت مشطناها ثلاثة قرون .٢١٧٠وقتيبة، بن سعيد قالا حدثنا حماد، ح وحدثنا يحيى بن أيوب، حدثنا ابن…٢١٧١ثلاثا أو خمسا أو سبعا أو أكثر من ذلك إن رأيتن ذلك٢١٧٢قالت اغسلنها وترا ثلاثا أو خمسا أو سبعا قال وقالت أم عطية مشطناها…٢١٧٣اغسلنها وترا ثلاثا أو خمسا واجعلن في الخامسة كافورا أو شيئا من كافور…٢١٧٤اغسلنها وترا خمسا أو أكثر من ذلك٢١٧٥ابدأن بميامنها ومواضع الوضوء منها٢١٧٦ابدأن بميامنها ومواضع الوضوء منها٢١٧٧ضعوها مما يلي رأسه واجعلوا على رجليه الإذخر٢١٧٨وابن أبي عمر، جميعا عن ابن عيينة، عن الأعمش، بهذا الإسناد . نحوه .٢١٧٩قالت كفن رسول الله صلى الله عليه وسلم في ثلاثة أثواب بيض سحولية من…٢١٨٠قالت أدرج رسول الله صلى الله عليه وسلم في حلة يمنية كانت لعبد الله بن…٢١٨١وابن، عيينة وابن إدريس وعبدة ووكيع ح وحدثناه يحيى بن يحيى، أخبرنا عبد…٢١٨٢أنه قال سألت عائشة زوج النبي صلى الله عليه وسلم فقلت لها في كم كفن…٢١٨٣وحسن الحلواني، وعبد بن حميد، قال عبد أخبرني وقال، الآخران حدثنا…٢١٨٤وعبد بن حميد، قالا أخبرنا عبد الرزاق، قال أخبرنا معمر، ح وحدثنا عبد…٢١٨٥إذا كفن أحدكم أخاه فليحسن كفنه٢١٨٦أسرعوا بالجنازة فإن تك صالحة فخير - لعله قال - تقدمونها عليه وإن تكن…٢١٨٧وعبد بن حميد، جميعا عن عبد الرزاق، أخبرنا معمر، ح وحدثنا يحيى بن…٢١٨٨أسرعوا بالجنازة فإن كانت صالحة قربتموها إلى الخير وإن كانت غير ذلك…٢١٨٩من شهد الجنازة حتى يصلى عليها فله قيراط ومن شهدها حتى تدفن فله قيراطان٢١٩٠وعبد، بن حميد عن عبد الرزاق، كلاهما عن معمر، عن الزهري، عن سعيد بن…٢١٩١ومن اتبعها حتى تدفن٢١٩٢من صلى على جنازة ولم يتبعها فله قيراط فإن تبعها فله قيراطان٢١٩٣من صلى على جنازة فله قيراط ومن اتبعها حتى توضع في القبر فقيراطان٢١٩٤من تبع جنازة فله قيراط من الأجر٢١٩٥من خرج مع جنازة من بيتها وصلى عليها ثم تبعها حتى تدفن كان له قيراطان…٢١٩٦من صلى على جنازة فله قيراط فإن شهد دفنها فله قيراطان القيراط مثل أحد٢١٩٧وحدثني ابن بشار، حدثنا معاذ بن هشام، حدثني أبي قال، وحدثنا ابن…٢١٩٨ما من ميت يصلي عليه أمة من المسلمين يبلغون مائة كلهم يشفعون له إلا…٢١٩٩ما من رجل مسلم يموت فيقوم على جنازته أربعون رجلا لا يشركون بالله شيئا…٢٢٠٠وجبت وجبت وجبت٢٢٠١وحدثني أبو الربيع الزهراني، حدثنا حماد يعني ابن زيد، ح وحدثني يحيى…٢٢٠٢مستريح ومستراح منه٢٢٠٣يستريح من أذى الدنيا ونصبها إلى رحمة الله٢٢٠٤أن رسول الله صلى الله عليه وسلم نعى للناس النجاشي في اليوم الذي مات…٢٢٠٥وأبي، سلمة بن عبد الرحمن أنهما حدثاه عن أبي هريرة، أنه قال نعى لنا…٢٢٠٦وحسن الحلواني، وعبد بن حميد، قالوا حدثنا يعقوب، - وهو ابن إبراهيم بن…٢٢٠٧أن رسول الله صلى الله عليه وسلم صلى٢٢٠٨مات اليوم عبد لله صالح أصحمة٢٢٠٩إن أخا لكم قد مات فقوموا فصلوا عليه٢٢١٠إن أخا لكم قد مات فقوموا فصلوا عليه٢٢١١ومحمد بن عبد الله بن نمير، قالا حدثنا عبد الله بن، إدريس عن الشيباني،…٢٢١٢وحدثنا يحيى بن يحيى، أخبرنا هشيم، ح وحدثنا حسن بن الربيع، وأبو كامل…٢٢١٣وهارون بن عبد الله، جميعا عن وهب بن جرير، عن شعبة، عن إسماعيل بن أبي…٢٢١٤أن النبي صلى الله عليه وسلم صلى على قبر .٢٢١٥أفلا كنتم آذنتموني٢٢١٦ومحمد بن المثنى، وابن، بشار قالوا حدثنا محمد، بن جعفر حدثنا شعبة، -…٢٢١٧إذا رأيتم الجنازة فقوموا لها حتى تخلفكم أو توضع٢٢١٨إذا رأى أحدكم الجنازة فإن لم يكن ماشيا معها فليقم حتى تخلفه أو توضع…٢٢١٩إذا رأى أحدكم الجنازة فليقم حين يراها حتى تخلفه إذا كان غير متبعها٢٢٢٠إذا اتبعتم جنازة فلا تجلسوا حتى توضع٢٢٢١إذا رأيتم الجنازة فقوموا فمن تبعها فلا يجلس حتى توضع٢٢٢٢إن الموت فزع فإذا رأيتم الجنازة فقوموا٢٢٢٣أنه سمع جابرا، يقول قام النبي صلى الله عليه وسلم لجنازة مرت به حتى…٢٢٢٤يقول قام النبي صلى الله عليه وسلم وأصحابه لجنازة يهودي حتى توارت .٢٢٢٦أن قيس بن سعد، وسهل بن حنيف، كانا بالقادسية فمرت بهما جنازة فقاما…٢٢٢٧وحدثنا قتيبة بن سعيد، حدثنا ليث، ح وحدثنا محمد بن رمح بن المهاجر، -…٢٢٢٨وإسحاق بن إبراهيم، وابن أبي عمر، جميعا عن الثقفي، - قال ابن المثنى…٢٢٢٩بهذا الإسناد .٢٢٣٠وحدثني زهير بن حرب، حدثنا عبد الرحمن بن مهدي، حدثنا شعبة، عن محمد، بن…٢٢٣١وعبيد الله بن سعيد، قالا حدثنا يحيى، - وهو القطان - عن شعبة، بهذا…٢٢٣٣اللهم اغفر له وارحمه وعافه واعف عنه وأكرم نزله ووسع مدخله واغسله…٢٢٣٤صالح بالإسنادين جميعا . نحو حديث ابن وهب .٢٢٣٥قال صليت خلف النبي صلى الله عليه وسلم وصلى على أم كعب ماتت وهي نفساء…٢٢٣٦ويزيد بن هارون، ح وحدثني علي بن حجر، أخبرنا ابن المبارك، والفضل بن…٢٢٣٧وعقبة بن مكرم العمي، قالا حدثنا ابن أبي عدي، عن حسين، عن عبد الله بن…٢٢٣٨سمرة قال أتي النبي صلى الله عليه وسلم بفرس معرورى فركبه حين انصرف من…٢٢٣٩كم من عذق معلق - أو مدلى - في الجنة لابن الدحداح٢٢٤٠أن سعد بن أبي وقاص، قال في مرضه الذي هلك فيه الحدوا لي لحدا وانصبوا…٢٢٤١ووكيع، جميعا عن شعبة، ح وحدثنا محمد بن المثنى، - واللفظ له - قال…٢٢٤٢- في رواية أبي الطاهر - أن أبا علي الهمداني، حدثه - وفي، رواية هارون…٢٢٤٤بهذا الإسناد وقال ولا صورة إلا طمستها .٢٢٤٥قال نهى رسول الله صلى الله عليه وسلم أن يجصص القبر وأن يقعد عليه وأن…٢٢٤٦أنه سمع جابر بن عبد، الله يقول سمعت النبي صلى الله عليه وسلم بمثله .٢٢٤٧.٢٢٤٨لأن يجلس أحدكم على جمرة فتحرق ثيابه فتخلص إلى جلده خير له من أن يجلس…٢٢٤٩بهذا الإسناد . نحوه.٢٢٥٠لا تجلسوا على القبور ولا تصلوا إليها٢٢٥١لا تصلوا إلى القبور ولا تجلسوا عليها٢٢٥٢وإسحاق بن إبراهيم الحنظلي، - واللفظ لإسحاق - قال علي حدثنا وقال،…٢٢٥٣وحدثني محمد بن حاتم، حدثنا بهز، حدثنا وهيب، حدثنا موسى بن عقبة، عن…٢٢٥٤ومحمد بن رافع، - واللفظ لابن رافع - قالا حدثنا ابن أبي فديك، أخبرنا…٢٢٥٥السلام عليكم دار قوم مؤمنين وأتاكم ما توعدون غدا مؤجلون وإنا إن شاء…٢٢٥٦ما لك يا عائش حشيا رابية٢٢٥٧وزهير بن حرب، قالا حدثنا محمد بن عبد الله، الأسدي عن سفيان، عن علقمة…٢٢٥٨استأذنت ربي أن أستغفر لأمي فلم يأذن لي واستأذنته أن أزور قبرها فأذن لي٢٢٥٩استأذنت ربي في أن أستغفر لها فلم يؤذن لي واستأذنته في أن أزور قبرها…٢٢٦٠نهيتكم عن زيارة القبور فزوروها ونهيتكم عن لحوم الأضاحي فوق ثلاث…٢٢٦١ومحمد بن رافع، وعبد بن حميد، جميعا عن عبد الرزاق، عن معمر، عن عطاء…٢٢٦٢قال أتي النبي صلى الله عليه وسلم برجل قتل نفسه بمشاقص فلم يصل عليه .
