٢١٠ حديثًا•صفحة ٢ من ٢
٥٨٠٠لا عدوى ولا طيرة ويعجبني الفأل الكلمة الحسنة الكلمة الطيبة٥٨٠١لا عدوى ولا طيرة ويعجبني الفأل٥٨٠٢لا عدوى ولا طيرة وأحب الفأل الصالح٥٨٠٣لا عدوى ولا هامة ولا طيرة وأحب الفأل الصالح٥٨٠٤الشؤم في الدار والمرأة والفرس٥٨٠٥لا عدوى ولا طيرة وإنما الشؤم في ثلاثة المرأة والفرس والدار٥٨٠٦وحمزة، ابنى عبد الله عن أبيهما، عن النبي صلى الله عليه وسلم ح وحدثنا…٥٨٠٧إن يكن من الشؤم شىء حق ففي الفرس والمرأة والدار٥٨٠٨بهذا الإسناد مثله ولم يقل " حق " .٥٨٠٩إن كان الشؤم في شىء ففي الفرس والمسكن والمرأة٥٨١٠إن كان ففي المرأة والفرس والمسكن٥٨١١وحدثنا أبو بكر بن أبي شيبة، حدثنا الفضل بن دكين، حدثنا هشام بن سعد،…٥٨١٢إن كان في شىء ففي الربع والخادم والفرس٥٨١٣فلا تأتوا الكهان٥٨١٤وعبد بن حميد، قالا أخبرنا عبد الرزاق، أخبرنا معمر، ح وحدثنا أبو بكر…٥٨١٥كان نبي من الأنبياء يخط فمن وافق خطه فذاك٥٨١٦تلك الكلمة الحق يخطفها الجني فيقذفها في أذن وليه ويزيد فيها مائة كذبة٥٨١٧أنه سمع عروة، يقول قالت عائشة سأل أناس رسول الله صلى الله عليه وسلم…٥٨١٨.٥٨١٩ماذا كنتم تقولون في الجاهلية إذا رمي بمثل هذا٥٨٢٠ولكن يقرفون فيه ويزيدون٥٨٢١من أتى عرافا فسأله عن شىء لم تقبل له صلاة أربعين ليلة٥٨٢٢إنا قد بايعناك فارجع٥٨٢٣وابن، نمير عن هشام، ح وحدثنا أبو كريب، حدثنا عبدة، حدثنا هشام، عن…٥٨٢٤بهذا الإسناد وقال الأبتر وذو الطفيتين .٥٨٢٥اقتلوا الحيات وذا الطفيتين والأبتر فإنهما يستسقطان الحبل ويلتمسان…٥٨٢٨واللفظ، له حدثنا ليث، عن نافع، أن أبا لبابة، كلم ابن عمر ليفتح له…٥٨٢٩وحدثنا شيبان بن فروخ، حدثنا جرير بن حازم، حدثنا نافع، قال كان ابن عمر…٥٨٣٠أنه سمع أبا لبابة، يخبر ابن عمر أن رسول الله صلى الله عليه وسلم نهى…٥٨٣١أن أبا لبابة، أخبره أن رسول الله صلى الله عليه وسلم نهى عن قتل الجنان…٥٨٣٢أن أبا لبابة بن عبد المنذر الأنصاري، - وكان مسكنه بقباء فانتقل إلى…٥٨٣٣وحدثني إسحاق بن منصور، أخبرنا محمد بن جهضم، حدثنا إسماعيل، - وهو…٥٨٣٤أن نافعا، حدثه أن أبا لبابة مر بابن عمر وهو عند الأطم الذي عند دار…٥٨٣٥حدثنا يحيى بن يحيى، وأبو بكر بن أبي شيبة وأبو كريب وإسحاق بن إبراهيم…٥٨٣٦وعثمان بن أبي شيبة، قالا حدثنا جرير، عن الأعمش، في هذا الإسناد بمثله .٥٨٣٧أن رسول الله صلى الله عليه وسلم أمر محرما بقتل حية بمنى .٥٨٣٨قال بينما نحن مع رسول الله صلى الله عليه وسلم في غار . بمثل حديث جرير…٥٨٣٩خذ عليك سلاحك فإني أخشى عليك قريظة٥٨٤٠إن لهذه البيوت عوامر فإذا رأيتم شيئا منها فحرجوا عليها ثلاثا فإن ذهب…٥٨٤١إن بالمدينة نفرا من الجن قد أسلموا فمن رأى شيئا من هذه العوامر…٥٨٤٢وعمرو الناقد، وإسحاق بن إبراهيم، وابن أبي عمر، قال إسحاق أخبرنا وقال…٥٨٤٣أن سعيد بن المسيب، أخبره أن أم شريك أخبرته أنها، استأمرت النبي صلى…٥٨٤٤وعبد بن حميد، قالا أخبرنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن عامر…٥٨٤٥وحرملة، قالا أخبرنا ابن وهب، أخبرني يونس، عن الزهري، عن عروة، عن…٥٨٤٦من قتل وزغة في أول ضربة فله كذا وكذا حسنة ومن قتلها في الضربة الثانية…٥٨٤٧من قتل وزغا في أول ضربة كتبت له مائة حسنة وفي الثانية دون ذلك وفي…٥٨٤٨في أول ضربة سبعين حسنة٥٨٤٩أن نملة قرصت نبيا من الأنبياء فأمر بقرية النمل فأحرقت فأوحى الله إليه…٥٨٥٠نزل نبي من الأنبياء تحت شجرة فلدغته نملة فأمر بجهازه فأخرج من تحتها…٥٨٥١نزل نبي من الأنبياء تحت شجرة فلدغته نملة فأمر بجهازه فأخرج من تحتها…٥٨٥٢عذبت امرأة في هرة سجنتها حتى ماتت فدخلت فيها النار لا هي أطعمتها…٥٨٥٣وحدثني نصر بن علي الجهضمي، حدثنا عبد الأعلى، عن عبيد الله بن عمر، عن…٥٨٥٤وعبد الله بن جعفر، عن معن بن عيسى، عن مالك، عن نافع، عن ابن عمر، عن…٥٨٥٥عذبت امرأة في هرة لم تطعمها ولم تسقها ولم تتركها تأكل من خشاش الأرض٥٨٥٦بهذا الإسناد وفي حديثهما " ربطتها " . وفي حديث أبي معاوية " حشرات…٥٨٥٧وعبد بن حميد، قال عبد أخبرنا وقال ابن رافع، حدثنا عبد الرزاق، أخبرنا…٥٨٥٨وحدثنا محمد بن رافع، حدثنا عبد الرزاق، أخبرنا معمر، عن همام بن منبه،…٥٨٥٩بينما رجل يمشي بطريق اشتد عليه العطش فوجد بئرا فنزل فيها فشرب ثم خرج…٥٨٦٠أن امرأة بغيا رأت كلبا في يوم حار يطيف ببئر قد أدلع لسانه من العطش…٥٨٦١بينما كلب يطيف بركية قد كاد يقتله العطش إذ رأته بغي من بغايا بني…
