٨٨ حديثًا
٤٢٥٤إن الله عز وجل ينهاكم أن تحلفوا بآبائكم٤٢٥٥وعبد بن حميد، قالا حدثنا عبد الرزاق، أخبرنا معمر، كلاهما عن الزهري،…٤٢٥٦وعمرو الناقد، وزهير بن حرب، قالوا حدثنا سفيان، بن عيينة عن الزهري، عن…٤٢٥٧ألا إن الله عز وجل ينهاكم أن تحلفوا بآبائكم فمن كان حالفا فليحلف…٤٢٥٨وابن أبي ذئب، ح وحدثنا إسحاق بن إبراهيم، وابن، رافع عن عبد الرزاق، عن…٤٢٥٩من كان حالفا فلا يحلف إلا بالله٤٢٦٠من حلف منكم فقال في حلفه باللات . فليقل لا إله إلا الله . ومن قال…٤٢٦١وحدثني سويد بن سعيد، حدثنا الوليد بن مسلم، عن الأوزاعي، ح وحدثنا…٤٢٦٢لا تحلفوا بالطواغي ولا بآبائكم٤٢٦٣والله لا أحملكم وما عندي ما أحملكم عليه٤٢٦٤والله لا أحملكم على شىء٤٢٦٥والله لا أحملكم وما عندي ما أحملكم عليه٤٢٦٦والقاسم، التميمي عن زهدم الجرمي، قال كان بين هذا الحى من جرم وبين…٤٢٦٧وإسحاق بن إبراهيم، وابن، نمير عن إسماعيل، ابن علية عن أيوب، عن القاسم…٤٢٦٨إني والله ما نسيتها٤٢٦٩ما عندي ما أحملكم والله ما أحملكم٤٢٧٠حدثنا محمد بن عبد الأعلى التيمي، حدثنا المعتمر، عن أبيه، حدثنا أبو…٤٢٧١من حلف على يمين فرأى غيرها خيرا منها فليأتها وليكفر عن يمينه٤٢٧٢من حلف على يمين فرأى غيرها خيرا منها فليكفر عن يمينه وليفعل٤٢٧٣من حلف على يمين فرأى غيرها خيرا منها فليأت الذي هو خير وليكفر عن يمينه٤٢٧٤فليكفر يمينه وليفعل الذي هو خير٤٢٧٥من حلف على يمين ثم رأى أتقى لله منها فليأت التقوى٤٢٧٦من حلف على يمين فرأى غيرها خيرا منها فليأت الذي هو خير وليترك يمينه٤٢٧٧إذا حلف أحدكم على اليمين فرأى خيرا منها فليكفرها وليأت الذي هو خير٤٢٧٨أنه سمع النبي صلى الله عليه وسلم يقول ذلك .٤٢٧٩من حلف على يمين ثم رأى خيرا منها فليأت الذي هو خير٤٢٨٠أن رجلا، سأله فذكر مثله وزاد ولك أربعمائة في عطائي .٤٢٨١يا عبد الرحمن بن سمرة لا تسأل الإمارة فإنك إن أعطيتها عن مسألة وكلت…٤٢٨٢ومنصور، وحميد، ح وحدثنا أبو كامل الجحدري، حدثنا حماد بن زيد، عن سماك…٤٢٨٣يمينك على ما يصدقك عليه صاحبك٤٢٨٤اليمين على نية المستحلف٤٢٨٥لو كان استثنى لولدت كل واحدة منهن غلاما فارسا يقاتل في سبيل الله٤٢٨٦قال سليمان بن داود نبي الله لأطوفن الليلة على سبعين امرأة كلهن تأتي…٤٢٨٧وحدثنا ابن أبي عمر، حدثنا سفيان، عن أبي الزناد، عن الأعرج، عن أبي…٤٢٨٨لو قال إن شاء الله . لم يحنث وكان دركا لحاجته٤٢٨٩قال سليمان بن داود لأطوفن الليلة على تسعين امرأة كلها تأتي بفارس…٤٢٩٠كلها تحمل غلاما يجاهد في سبيل الله٤٢٩١والله لأن يلج أحدكم بيمينه في أهله آثم له عند الله من أن يعطي كفارته…٤٢٩٢ومحمد بن المثنى، وزهير بن حرب، - واللفظ لزهير - قالوا حدثنا يحيى، -…٤٢٩٣ومحمد بن العلاء، وإسحاق، بن إبراهيم جميعا عن حفص بن غياث، ح وحدثنا…٤٢٩٤اذهب فاعتكف يوما٤٢٩٥وحدثنا عبد بن حميد، أخبرنا عبد الرزاق، أخبرنا معمر، عن أيوب، عن نافع،…٤٢٩٦وحدثنا أحمد بن عبدة الضبي، حدثنا حماد بن زيد، حدثنا أيوب، عن نافع،…٤٢٩٧. بهذا الحديث في النذر وفي حديثهما جميعا اعتكاف يوم .٤٢٩٨من لطم مملوكه أو ضربه فكفارته أن يعتقه٤٢٩٩من ضرب غلاما له حدا لم يأته أو لطمه فإن كفارته أن يعتقه٤٣٠٠بإسناد شعبة وأبي عوانة أما حديث ابن مهدي فذكر فيه " حدا لم يأته " .…٤٣٠١حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، ح وحدثنا ابن نمير،…٤٣٠٢ومحمد بن عبد الله بن نمير، - واللفظ لأبي بكر - قالا حدثنا ابن إدريس،…٤٣٠٣وابن، بشار قالا حدثنا ابن أبي عدي، عن شعبة، عن حصين، عن هلال بن يساف،…٤٣٠٤وحدثنا عبد الوارث بن عبد الصمد، حدثني أبي، حدثنا شعبة، قال قال لي…٤٣٠٥ومحمد بن المثنى، عن وهب بن جرير، أخبرنا شعبة، قال قال لي محمد بن…٤٣٠٦حدثنا أبو كامل الجحدري، حدثنا عبد الواحد، - يعني ابن زياد - حدثنا…٤٣٠٧بإسناد عبد الواحد . نحو حديثه . غير أن في حديث جرير فسقط من يدي السوط…٤٣٠٨اعلم أبا مسعود لله أقدر عليك منك عليه٤٣٠٩والله لله أقدر عليك منك عليه٤٣١٠بهذا الإسناد ولم يذكر قوله أعوذ بالله أعوذ برسول الله صلى الله عليه…٤٣١١من قذف مملوكه بالزنا يقام عليه الحد يوم القيامة إلا أن يكون كما قال٤٣١٢القاسم صلى الله عليه وسلم نبي التوبة .٤٣١٣يا أبا ذر إنك امرؤ فيك جاهلية٤٣١٤إنك امرؤ فيك جاهلية٤٣١٥إنك امرؤ فيك جاهلية إخوانكم وخولكم جعلهم الله تحت أيديكم فمن كان أخوه…٤٣١٦للمملوك طعامه وكسوته ولا يكلف من العمل إلا ما يطيق٤٣١٧إذا صنع لأحدكم خادمه طعامه ثم جاءه به وقد ولي حره ودخانه فليقعده معه…٤٣١٨إن العبد إذا نصح لسيده وأحسن عبادة الله فله أجره مرتين٤٣١٩ومحمد بن المثنى، قالا حدثنا يحيى، وهو القطان ح وحدثنا ابن نمير، حدثنا…٤٣٢٠للعبد المملوك المصلح أجران٤٣٢١وحدثنيه زهير بن حرب، حدثنا أبو صفوان الأموي، أخبرني يونس، عن ابن…٤٣٢٢إذا أدى العبد حق الله وحق مواليه كان له أجران٤٣٢٣بهذا الإسناد .٤٣٢٤نعما للمملوك أن يتوفى يحسن عبادة الله وصحابة سيده نعما له٤٣٢٥من أعتق شركا له في عبد فكان له مال يبلغ ثمن العبد قوم عليه قيمة العدل…٤٣٢٦من أعتق شركا له من مملوك فعليه عتقه كله إن كان له مال يبلغ ثمنه فإن…٤٣٢٧من أعتق نصيبا له في عبد فكان له من المال قدر ما يبلغ قيمته قوم عليه…٤٣٢٨وإن لم يكن له مال فقد عتق منه ما عتق٤٣٢٩من أعتق عبدا بينه وبين آخر قوم عليه في ماله قيمة عدل لا وكس ولا شطط…٤٣٣٠من أعتق شركا له في عبد عتق ما بقي في ماله إذا كان له مال يبلغ ثمن…٤٣٣١ومحمد بن بشار، - واللفظ لابن المثنى - قالا حدثنا محمد بن جعفر، حدثنا…٤٣٣٢من أعتق شقيصا من مملوك فهو حر من ماله٤٣٣٣من أعتق شقيصا له في عبد فخلاصه في ماله إن كان له مال فإن لم يكن له…٤٣٣٤ثم يستسعى في نصيب الذي لم يعتق غير مشقوق عليه٤٣٣٥بن حصين . أن رجلا، أعتق ستة مملوكين له عند موته لم يكن له مال غيرهم…٤٣٣٦وابن أبي، عمر عن الثقفي، كلاهما عن أيوب، بهذا الإسناد . أما حماد…٤٣٣٧وأحمد بن عبدة، قالا حدثنا يزيد بن زريع، حدثنا هشام بن حسان، عن محمد…٤٣٣٨أن رجلا، من الأنصار أعتق غلاما له عن دبر لم يكن له مال غيره فبلغ ذلك…٤٣٣٩وإسحاق بن إبراهيم، عن ابن عيينة، قال أبو بكر حدثنا سفيان بن عيينة،…٤٣٤٠وابن، رمح عن الليث بن سعد، عن أبي الزبير، عن جابر، عن النبي صلى الله…٤٣٤١أبي رباح وأبي الزبير وعمرو بن دينار أن جابر بن عبد الله، حدثهم في،…
