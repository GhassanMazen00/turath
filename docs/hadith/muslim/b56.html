٣٩ حديثًا
٧٥٢٣قيل لبني إسرائيل { ادخلوا الباب سجدا وقولوا حطة يغفر لكم خطاياكم }…٧٥٢٤والحسن بن علي الحلواني، وعبد بن، حميد - قال عبد حدثني وقال الآخران،…٧٥٢٥بن شهاب أن اليهود، قالوا لعمر إنكم تقرءون آية لو أنزلت فينا لاتخذنا…٧٥٢٦حدثنا أبو بكر بن أبي شيبة، وأبو كريب - واللفظ لأبي بكر - قال حدثنا…٧٥٢٧وحدثني عبد بن حميد، أخبرنا جعفر بن عون، أخبرنا أبو عميس، عن قيس، بن…٧٥٢٨{ وإن خفتم أن لا، تقسطوا في اليتامى فانكحوا ما طاب لكم من النساء مثنى…٧٥٢٩وعبد بن حميد، جميعا عن يعقوب بن إبراهيم بن سعد، حدثنا أبي، عن صالح،…٧٥٣٠في قوله { وإن خفتم أن لا تقسطوا في اليتامى } قالت أنزلت في الرجل تكون…٧٥٣١في قوله { وما يتلى عليكم في الكتاب في يتامى النساء اللاتي لا تؤتونهن…٧٥٣٢في قوله { ويستفتونك في النساء قل الله يفتيكم فيهن } الآية قالت هي…٧٥٣٣في قوله { ومن كان فقيرا فليأكل بالمعروف } قالت أنزلت في والي مال…٧٥٣٤في قوله تعالى { ومن كان غنيا فليستعفف ومن كان فقيرا فليأكل بالمعروف }…٧٥٣٥بهذا الإسناد .٧٥٣٦في قوله عز وجل { إذ جاءوكم من فوقكم ومن أسفل منكم وإذ زاغت الأبصار…٧٥٣٧{ وإن امرأة خافت من بعلها نشوزا أو إعراضا } الآية قالت أنزلت في…٧٥٣٨في قوله عز وجل { وإن امرأة خافت من بعلها نشوزا أو إعراضا } قالت نزلت…٧٥٣٩قال قالت لي عائشة يا ابن أختي أمروا أن يستغفروا، لأصحاب النبي صلى…٧٥٤٠بهذا الإسناد مثله .٧٥٤١حدثنا عبيد الله بن معاذ العنبري، حدثنا أبي، حدثنا شعبة، عن المغيرة بن…٧٥٤٢وابن، بشار قالا حدثنا محمد بن جعفر، ح وحدثنا إسحاق، بن إبراهيم أخبرنا…٧٥٤٣ومحمد بن بشار، قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن منصور، عن سعيد…٧٥٤٤حدثني هارون بن عبد الله، حدثنا أبو النضر، هاشم بن القاسم الليثي حدثنا…٧٥٤٥وعبد الرحمن بن بشر العبدي، قالا حدثنا يحيى، - وهو ابن سعيد القطان -…٧٥٤٦وهارون بن عبد الله، وعبد بن حميد، قال عبد أخبرنا وقال الآخران، حدثنا…٧٥٤٧بهذا الإسناد مثله وقال آخر سورة وقال عبد المجيد ولم يقل ابن سهيل .٧٥٤٨وإسحاق بن إبراهيم، وأحمد بن عبدة الضبي، - واللفظ لابن أبي شيبة - قال…٧٥٤٩يقول كانت الأنصار إذا حجوا فرجعوا لم يدخلوا البيوت إلا من ظهورها -…٧٥٥٠أن ابن مسعود، قال ما كان بين إسلامنا وبين أن عاتبنا الله بهذه الآية {…٧٥٥١قال كانت المرأة تطوف بالبيت وهي عريانة فتقول من يعيرني تطوافا تجعله…٧٥٥٢حدثنا أبو بكر بن أبي شيبة، وأبو كريب جميعا عن أبي معاوية، - واللفظ…٧٥٥٣أن جارية، لعبد الله بن أبى ابن سلول يقال لها مسيكة وأخرى يقال لها…٧٥٥٤في قوله عز وجل { أولئك الذين يدعون يبتغون إلى ربهم الوسيلة أيهم أقرب…٧٥٥٦مسعود { أولئك الذين يدعون يبتغون إلى ربهم الوسيلة } قال نزلت في نفر…٧٥٥٨حدثني عبد الله بن مطيع، حدثنا هشيم، عن أبي بشر، عن سعيد بن جبير، قال…٧٥٥٩حدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن أبي حيان، عن الشعبي،…٧٥٦٠وحدثنا أبو كريب، أخبرنا ابن إدريس، حدثنا أبو حيان، عن الشعبي، عن ابن،…٧٥٦١بهذا الإسناد . بمثل حديثهما غير أن ابن علية في حديثه العنب . كما قال…٧٥٦٢حدثنا عمرو بن زرارة، حدثنا هشيم، عن أبي هاشم، عن أبي مجلز، عن قيس، بن…٧٥٦٣يقسم لنزلت { هذان خصمان } بمثل حديث هشيم .
