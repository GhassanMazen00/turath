٧٩ حديثًا
٧٠٤٥إنه ليأتي الرجل العظيم السمين يوم القيامة لا يزن عند الله جناح بعوضة…٧٠٤٦حدثنا أحمد بن عبد الله بن يونس، حدثنا فضيل، - يعني ابن عياض - عن…٧٠٤٧{ وما قدروا الله حق قدره }٧٠٤٨حدثنا عمر بن حفص بن غياث، حدثنا أبي، حدثنا الأعمش، قال سمعت إبراهيم،…٧٠٤٩بهذا الإسناد غير أن في، حديثهم جميعا والشجر على إصبع والثرى على إصبع…٧٠٥٠يقبض الله تبارك وتعالى الأرض يوم القيامة ويطوي السماء بيمينه ثم يقول…٧٠٥١يطوي الله عز وجل السموات يوم القيامة ثم يأخذهن بيده اليمنى ثم يقول…٧٠٥٢يأخذ الله عز وجل سمواته وأرضيه بيديه فيقول أنا الله - ويقبض أصابعه…٧٠٥٣يأخذ الجبار عز وجل سمواته وأرضيه بيديه٧٠٥٤خلق الله عز وجل التربة يوم السبت وخلق فيها الجبال يوم الأحد وخلق…٧٠٥٥يحشر الناس يوم القيامة على أرض بيضاء عفراء كقرصة النقي ليس فيها علم…٧٠٥٦حدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن داود، عن الشعبي، عن…٧٠٥٧تكون الأرض يوم القيامة خبزة واحدة يكفؤها الجبار بيده كما يكفأ أحدكم…٧٠٥٨لو تابعني عشرة من اليهود لم يبق على ظهرها يهودي إلا أسلم٧٠٥٩حدثنا عمر بن حفص بن غياث، حدثنا أبي، حدثنا الأعمش، حدثني إبراهيم، عن…٧٠٦٠وعلي بن خشرم، قالا أخبرنا عيسى بن يونس، كلاهما عن الأعمش، عن إبراهيم،…٧٠٦١حدثنا أبو سعيد الأشج، قال سمعت عبد الله بن إدريس، يقول سمعت الأعمش،…٧٠٦٢وعبد الله بن سعيد الأشج، - واللفظ لعبد الله - قالا حدثنا وكيع، حدثنا…٧٠٦٣حدثنا أبو كريب، حدثنا أبو معاوية، ح وحدثنا ابن نمير، حدثنا أبي ح،…٧٠٦٤أنه سمع أنس بن مالك، يقول قال أبو جهل اللهم إن كان هذا هو الحق من…٧٠٦٥لو دنا مني لاختطفته الملائكة عضوا عضوا٧٠٦٦اللهم سبع كسبع يوسف٧٠٦٧لمضر إنك لجريء٧٠٦٨قال خمس قد مضين الدخان واللزام والروم والبطشة والقمر .٧٠٦٩بهذا الإسناد مثله .٧٠٧٠ومحمد بن بشار، قالا حدثنا محمد بن جعفر، حدثنا شعبة، ح وحدثنا أبو بكر…٧٠٧١وزهير بن حرب، قالا حدثنا سفيان بن عيينة، عن ابن أبي، نجيح عن مجاهد،…٧٠٧٢حدثنا أبو بكر بن أبي شيبة، وأبو كريب وإسحاق بن إبراهيم جميعا عن أبي،…٧٠٧٣قال انشق القمر على عهد رسول الله صلى الله عليه وسلم فلقتين فستر الجبل…٧٠٧٤حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا شعبة، عن الأعمش، عن مجاهد،…٧٠٧٥نحو حديثه غير أن في، حديث ابن أبي عدي فقال " اشهدوا اشهدوا " .٧٠٧٨ومحمد بن جعفر، وأبو داود كلهم عن شعبة، عن قتادة، عن أنس، قال انشق…٧٠٧٩قال إن القمر انشق على زمان رسول الله صلى الله عليه وسلم .٧٠٨٢ما أحد أصبر على أذى يسمعه من الله تعالى إنهم يجعلون له ندا ويجعلون له…٧٠٨٣يقول الله تبارك وتعالى لأهون أهل النار عذابا لو كانت لك الدنيا وما…٧٠٨٤ولا أدخلك النار٧٠٨٥يقال للكافر يوم القيامة أرأيت لو كان لك ملء الأرض ذهبا أكنت تفتدي به…٧٠٨٦فيقال له كذبت قد سئلت ما هو أيسر من ذلك٧٠٨٧أليس الذي أمشاه على رجليه في الدنيا قادرا على أن يمشيه على وجهه يوم…٧٠٨٨يؤتى بأنعم أهل الدنيا من أهل النار يوم القيامة فيصبغ في النار صبغة ثم…٧٠٨٩إن الله لا يظلم مؤمنا حسنة يعطى بها في الدنيا ويجزى بها في الآخرة…٧٠٩٠إن الكافر إذا عمل حسنة أطعم بها طعمة من الدنيا وأما المؤمن فإن الله…٧٠٩١حدثنا محمد بن عبد الله الرزي، أخبرنا عبد الوهاب بن عطاء، عن سعيد، عن…٧٠٩٢مثل المؤمن كمثل الزرع لا تزال الريح تميله ولا يزال المؤمن يصيبه…٧٠٩٣وعبد بن حميد، عن عبد الرزاق، حدثنا معمر، عن الزهري، بهذا الإسناد غير…٧٠٩٤مثل المؤمن كمثل الخامة من الزرع تفيئها الريح وتصرعها مرة وتعدلها أخرى…٧٠٩٥مثل المؤمن كمثل الخامة من الزرع تفيئها الرياح تصرعها مرة وتعدلها حتى…٧٠٩٦ومثل الكافر كمثل الأرزة٧٠٩٧ومثل الكافر مثل الأرزة٧٠٩٨إن من الشجر شجرة لا يسقط ورقها وإنها مثل المسلم فحدثوني ما هي٧٠٩٩أخبروني عن شجرة مثلها مثل المؤمن٧١٠٠وابن أبي عمر، قالا حدثنا سفيان بن عيينة، عن ابن أبي نجيح، عن مجاهد،…٧١٠١عمر يقول أتي رسول الله صلى الله عليه وسلم بجمار . فذكر نحو حديثهم .٧١٠٢أخبروني بشجرة شبه أو كالرجل المسلم لا يتحات ورقها٧١٠٣إن الشيطان قد أيس أن يعبده المصلون في جزيرة العرب ولكن في التحريش…٧١٠٤بهذا الإسناد .٧١٠٥إن عرش إبليس على البحر فيبعث سراياه فيفتنون الناس فأعظمهم عنده أعظمهم…٧١٠٦إن إبليس يضع عرشه على الماء ثم يبعث سراياه فأدناهم منه منزلة أعظمهم…٧١٠٧يبعث الشيطان سراياه فيفتنون الناس فأعظمهم عنده منزلة أعظمهم فتنة٧١٠٨ما منكم من أحد إلا وقد وكل به قرينه من الجن٧١٠٩وقد وكل به قرينه من الجن وقرينه من الملائكة٧١١٠ما لك يا عائشة أغرت٧١١٣ما من أحد يدخله عمله الجنة٧١١٤ليس أحد منكم ينجيه عمله٧١١٥ليس أحد ينجيه عمله٧١١٦لن يدخل أحدا منكم عمله الجنة٧١١٧قاربوا وسددوا واعلموا أنه لن ينجو أحد منكم بعمله٧١١٨وحدثنا ابن نمير، حدثنا أبي، حدثنا الأعمش، عن أبي سفيان، عن جابر، عن…٧١١٩بالإسنادين جميعا كرواية ابن نمير .٧١٢٠حدثنا أبو بكر بن أبي شيبة، وأبو كريب قالا حدثنا أبو معاوية، عن…٧١٢١لا يدخل أحدا منكم عمله الجنة ولا يجيره من النار ولا أنا إلا برحمة من…٧١٢٢سددوا وقاربوا وأبشروا فإنه لن يدخل الجنة أحدا عمله٧١٢٣بهذا الإسناد ولم يذكر " وأبشروا " .٧١٢٤أفلا أكون عبدا شكورا٧١٢٥أفلا أكون عبدا شكورا٧١٢٦يا عائشة أفلا أكون عبدا شكورا٧١٢٧حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، وأبو معاوية ح وحدثنا ابن نمير،…٧١٢٨وعلي بن خشرم، قالا أخبرنا عيسى، بن يونس ح وحدثنا ابن أبي عمر، حدثنا…٧١٢٩وحدثنا إسحاق بن إبراهيم، أخبرنا جرير، عن منصور، ح وحدثنا ابن أبي عمر،…
