٣٩٦ حديثًا•صفحة ١ من ٣
١١٦١قال قلت يا رسول الله أى مسجد وضع في الأرض أول قال " المسجد الحرام " .…١١٦٢حدثني علي بن حجر السعدي، أخبرنا علي بن مسهر، حدثنا الأعمش، عن إبراهيم…١١٦٣أعطيت خمسا لم يعطهن أحد قبلي كان كل نبي يبعث إلى قومه خاصة وبعثت إلى…١١٦٤أن رسول الله صلى الله عليه وسلم قال . فذكر نحوه .١١٦٥فضلنا على الناس بثلاث جعلت صفوفنا كصفوف الملائكة وجعلت لنا الأرض كلها…١١٦٦قال قال رسول الله صلى الله عليه وسلم . بمثله .١١٦٧فضلت على الأنبياء بست أعطيت جوامع الكلم ونصرت بالرعب وأحلت لي الغنائم…١١٦٨بعثت بجوامع الكلم ونصرت بالرعب وبينا أنا نائم أتيت بمفاتيح خزائن…١١٦٩وأبو سلمة بن عبد الرحمن أن أبا هريرة، قال سمعت رسول الله صلى الله…١١٧٠وعبد بن حميد، قالا حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن ابن…١١٧١نصرت بالرعب على العدو وأوتيت جوامع الكلم وبينما أنا نائم أتيت بمفاتيح…١١٧٢نصرت بالرعب وأوتيت جوامع الكلم١١٧٣يا بني النجار ثامنوني بحائطكم هذا١١٧٤أن رسول الله صلى الله عليه وسلم كان يصلي في مرابض الغنم قبل أن يبنى…١١٧٥يقول كان رسول الله صلى الله عليه وسلم بمثله .١١٧٦قال صليت مع النبي صلى الله عليه وسلم إلى بيت المقدس ستة عشر شهرا حتى…١١٧٧حدثنا محمد بن المثنى، وأبو بكر بن خلاد جميعا عن يحيى، - قال ابن…١١٧٨قال بينما الناس في صلاة الصبح بقباء إذ جاءهم آت فقال إن رسول الله صلى…١١٧٩قال بينما الناس في صلاة الغداة إذ جاءهم رجل . بمثل حديث مالك .١١٨٠أن رسول الله صلى الله عليه وسلم كان يصلي نحو بيت المقدس فنزلت { قد…١١٨١إن أولئك إذا كان فيهم الرجل الصالح فمات بنوا على قبره مسجدا وصوروا…١١٨٢وعمرو الناقد، قالا حدثنا وكيع، حدثنا هشام بن عروة، عن أبيه، عن عائشة،…١١٨٣قالت ذكرن أزواج النبي صلى الله عليه وسلم كنيسة رأينها بأرض الحبشة…١١٨٤لعن الله اليهود والنصارى اتخذوا قبور أنبيائهم مساجد١١٨٥قاتل الله اليهود اتخذوا قبور أنبيائهم مساجد١١٨٦لعن الله اليهود والنصارى اتخذوا قبور أنبيائهم مساجد١١٨٧لعنة الله على اليهود والنصارى اتخذوا قبور أنبيائهم مساجد١١٨٨إني أبرأ إلى الله أن يكون لي منكم خليل فإن الله تعالى قد اتخذني خليلا…١١٨٩من بنى مسجدا لله تعالى - قال بكير حسبت أنه قال - يبتغي به وجه الله -…١١٩٠من بنى مسجدا لله بنى الله له في الجنة مثله١١٩١وعلقمة، قالا أتينا عبد الله بن مسعود في داره فقال أصلى هؤلاء خلفكم…١١٩٢والأسود، أنهما دخلا على عبد الله . بمعنى حديث أبي معاوية . وفي حديث…١١٩٣والأسود، أنهما دخلا على عبد الله فقال أصلى من خلفكم قالا نعم . فقام…١١٩٤حدثنا قتيبة بن سعيد، وأبو كامل الجحدري - واللفظ لقتيبة - قالا حدثنا…١١٩٥حدثنا خلف بن هشام، حدثنا أبو الأحوص، ح قال وحدثنا ابن أبي عمر، حدثنا…١١٩٦حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن إسماعيل بن أبي خالد، عن…١١٩٧حدثني الحكم بن موسى، حدثنا عيسى بن يونس، حدثنا إسماعيل بن أبي خالد،…١١٩٨حدثنا إسحاق بن إبراهيم، أخبرنا محمد بن بكر، ح قال وحدثنا حسن…١١٩٩إن هذه الصلاة لا يصلح فيها شىء من كلام الناس إنما هو التسبيح والتكبير…١٢٠٠بهذا الإسناد نحوه .١٢٠١إن في الصلاة شغلا١٢٠٢بهذا الإسناد . نحوه .١٢٠٣حدثنا يحيى بن يحيى، أخبرنا هشيم، عن إسماعيل بن أبي خالد، عن الحارث بن…١٢٠٤ووكيع، ح قال وحدثنا إسحاق بن إبراهيم، أخبرنا عيسى بن يونس، كلهم عن…١٢٠٥إنك سلمت آنفا وأنا أصلي١٢٠٦ما فعلت في الذي أرسلتك له فإنه لم يمنعني أن أكلمك إلا أني كنت أصلي١٢٠٧إنه لم يمنعني أن أرد عليك إلا أني كنت أصلي١٢٠٨قال بعثني رسول الله صلى الله عليه وسلم في حاجة . بمعنى حديث حماد .١٢٠٩إن عفريتا من الجن جعل يفتك على البارحة ليقطع على الصلاة وإن الله…١٢١٠في هذا الإسناد وليس في حديث ابن جعفر قوله فذعته . وأما ابن أبي شيبة…١٢١١حدثنا محمد بن سلمة المرادي، حدثنا عبد الله بن وهب، عن معاوية بن صالح،…١٢١٢وقتيبة بن سعيد، قالا حدثنا مالك، عن عامر بن عبد الله بن الزبير، ح…١٢١٣وابن، عجلان سمعا عامر بن عبد الله بن الزبير، يحدث عن عمرو بن سليم…١٢١٤يقول رأيت رسول الله صلى الله عليه وسلم يصلي للناس وأمامة بنت أبي…١٢١٥سمع أبا قتادة، يقول بينا نحن في المسجد جلوس خرج علينا رسول الله صلى…١٢١٦انظري غلامك النجار يعمل لي أعوادا أكلم الناس عليها١٢١٧أن رجالا، أتوا سهل بن سعد ح قال وحدثنا أبو بكر بن أبي شيبة، وزهير بن…١٢١٨وحدثني الحكم بن موسى القنطري، حدثنا عبد الله بن المبارك، ح قال وحدثنا…١٢١٩إن كنت لا بد فاعلا فواحدة١٢٢٣إذا كان أحدكم يصلي فلا يبصق قبل وجهه فإن الله قبل وجهه إذا صلى١٢٢٤ومحمد بن رمح، عن الليث بن سعد، ح وحدثني زهير بن حرب، حدثنا إسماعيل، -…١٢٢٥حدثنا يحيى بن يحيى، وأبو بكر بن أبي شيبة وعمرو الناقد جميعا عن سفيان،…١٢٢٦وحرملة، قالا حدثنا ابن وهب، عن يونس، ح قال وحدثني زهير بن حرب، حدثنا…١٢٢٧أن النبي صلى الله عليه وسلم رأى بصاقا في جدار القبلة أو مخاطا أو…١٢٢٨ما بال أحدكم يقوم مستقبل ربه فيتنخع أمامه أيحب أحدكم أن يستقبل فيتنخع…١٢٢٩وحدثنا شيبان بن فروخ، حدثنا عبد الوارث، ح قال وحدثنا يحيى بن يحيى،…١٢٣٠إذا كان أحدكم في الصلاة فإنه يناجي ربه فلا يبزقن بين يديه ولا عن…١٢٣١البزاق في المسجد خطيئة وكفارتها دفنها١٢٣٢التفل في المسجد خطيئة وكفارتها دفنها١٢٣٣عرضت على أعمال أمتي حسنها وسيئها فوجدت في محاسن أعمالها الأذى يماط عن…١٢٣٤قال صليت مع رسول الله صلى الله عليه وسلم فرأيته تنخع فدلكها بنعله .١٢٣٥أنه صلى مع النبي صلى الله عليه وسلم قال فتنخع فدلكها بنعله اليسرى .١٢٣٦سعيد بن يزيد قال قلت لأنس بن مالك أكان رسول الله صلى الله عليه وسلم…١٢٣٧قال سألت أنسا . بمثله .١٢٣٨شغلتني أعلام هذه فاذهبوا بها إلى أبي جهم وائتوني بأنبجانيه١٢٣٩اذهبوا بهذه الخميصة إلى أبي جهم بن حذيفة وائتوني بأنبجانيه فإنها…١٢٤٠أن النبي صلى الله عليه وسلم كانت له خميصة لها علم فكان يتشاغل بها في…١٢٤١إذا حضر العشاء وأقيمت الصلاة فابدءوا بالعشاء١٢٤٢إذا قرب العشاء وحضرت الصلاة فابدءوا به قبل أن تصلوا صلاة المغرب ولا…١٢٤٣وحفص، ووكيع، عن هشام، عن أبيه، عن عائشة، عن النبي صلى الله عليه وسلم…١٢٤٤إذا وضع عشاء أحدكم وأقيمت الصلاة فابدءوا بالعشاء ولا يعجلن حتى يفرغ…١٢٤٥وحدثنا محمد بن إسحاق المسيبي، حدثني أنس، - يعني ابن عياض - عن موسى بن…١٢٤٦لا صلاة بحضرة الطعام ولا وهو يدافعه الأخبثان١٢٤٧وقتيبة بن سعيد، وابن، حجر قالوا حدثنا إسماعيل، - وهو ابن جعفر -…١٢٤٨من أكل من هذه الشجرة - يعني الثوم - فلا يأتين المساجد١٢٤٩من أكل من هذه البقلة فلا يقربن مساجدنا حتى يذهب ريحها١٢٥٠من أكل من هذه الشجرة فلا يقربنا ولا يصلي معنا١٢٥٢من أكل من هذه الشجرة فلا يقربن مسجدنا ولا يؤذينا بريح الثوم١٢٥٣من أكل من هذه الشجرة المنتنة فلا يقربن مسجدنا فإن الملائكة تأذى مما…١٢٥٤من أكل ثوما أو بصلا فليعتزلنا أو ليعتزل مسجدنا وليقعد في بيته١٢٥٥من أكل من هذه الشجرة - يريد الثوم - فلا يغشنا في مسجدنا١٢٥٦من أكل من هذه الشجرة الخبيثة شيئا فلا يقربنا في المسجد١٢٥٧وأحمد بن عيسى، قالا حدثنا ابن وهب، أخبرني عمرو، عن بكير بن الأشج، عن…١٢٥٨يا عمر ألا تكفيك آية الصيف التي في آخر سورة النساء١٢٥٩وإسحاق بن إبراهيم، كلاهما عن شبابة بن سوار، قال حدثنا شعبة، جميعا عن…١٢٦٠من سمع رجلا ينشد ضالة في المسجد فليقل لا ردها الله عليك فإن المساجد…١٢٦١مولى شداد أنه سمع أبا هريرة، يقول سمعت رسول الله صلى الله عليه وسلم…١٢٦٢لا وجدت . إنما بنيت المساجد لما بنيت له١٢٦٣لا وجدت إنما بنيت المساجد لما بنيت له١٢٦٤قال جاء أعرابي بعد ما صلى النبي صلى الله عليه وسلم صلاة الفجر . فأدخل…١٢٦٥إن أحدكم إذا قام يصلي جاءه الشيطان فلبس عليه حتى لا يدري كم صلى فإذا…١٢٦٦وزهير بن حرب، قالا حدثنا سفيان، - وهو ابن عيينة - ح قال وحدثنا قتيبة…١٢٦٧إذا نودي بالأذان أدبر الشيطان له ضراط حتى لا يسمع الأذان فإذا قضي…١٢٦٨إن الشيطان إذا ثوب بالصلاة ولى وله ضراط١٢٦٩حدثنا يحيى بن يحيى، قال قرأت على مالك عن ابن شهاب، عن عبد الرحمن…١٢٧٠حليف بني عبد المطلب أن رسول الله صلى الله عليه وسلم قام في صلاة الظهر…١٢٧١أن رسول الله صلى الله عليه وسلم قام في الشفع الذي يريد أن يجلس في…١٢٧٢إذا شك أحدكم في صلاته فلم يدر كم صلى ثلاثا أم أربعا فليطرح الشك وليبن…١٢٧٣يسجد سجدتين قبل السلام١٢٧٤وحدثنا عثمان، وأبو بكر ابنا أبي شيبة وإسحاق بن إبراهيم جميعا عن جرير،…١٢٧٥فلينظر أحرى ذلك للصواب١٢٧٦فلينظر أحرى ذلك للصواب١٢٧٧بهذا الإسناد وقال " فليتحر الصواب " .١٢٧٨فليتحر أقرب ذلك إلى الصواب١٢٧٩فليتحر الذي يرى أنه الصواب١٢٨٠بإسناد هؤلاء وقال " فليتحر الصواب " .١٢٨١أن النبي صلى الله عليه وسلم صلى الظهر خمسا فلما سلم قيل له أزيد في…١٢٨٢أنه صلى بهم خمسا .١٢٨٣حدثنا عثمان بن أبي شيبة، - واللفظ له - حدثنا جرير، عن الحسن بن عبيد…١٢٨٤وحدثناه عون بن سلام الكوفي، أخبرنا أبو بكر النهشلي، عن عبد الرحمن بن…١٢٨٥إنما أنا بشر مثلكم أنسى كما تنسون فإذا نسي أحدكم فليسجد سجدتين وهو…١٢٨٦أن النبي صلى الله عليه وسلم سجد سجدتى السهو بعد السلام والكلام .١٢٨٧وحدثني القاسم بن زكرياء، حدثنا حسين بن علي الجعفي، عن زائدة، عن…١٢٨٨ما يقول ذو اليدين١٢٨٩حدثنا أبو الربيع الزهراني، حدثنا حماد، حدثنا أيوب، عن محمد، عن أبي…١٢٩٠حدثنا قتيبة بن سعيد، عن مالك بن أنس، عن داود بن الحصين، عن أبي سفيان،…١٢٩١أن رسول الله صلى الله عليه وسلم صلى ركعتين من صلاة الظهر ثم سلم فأتاه…١٢٩٢مع النبي صلى الله عليه وسلم صلاة الظهر سلم رسول الله صلى الله عليه…١٢٩٣وزهير بن حرب، جميعا عن ابن علية، - قال زهير حدثنا إسماعيل بن إبراهيم،…١٢٩٤قال سلم رسول الله صلى الله عليه وسلم في ثلاث ركعات من العصر ثم قام…١٢٩٥وعبيد الله بن سعيد، ومحمد بن المثنى، كلهم عن يحيى القطان، - قال زهير…١٢٩٦حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن بشر، حدثنا عبيد الله بن عمر،…١٢٩٧ومحمد بن بشار، قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي إسحاق، قال…١٢٩٨ويحيى بن أيوب، وقتيبة بن سعيد، وابن، حجر قال يحيى بن يحيى أخبرنا وقال…١٢٩٩أن أبا هريرة، قرأ لهم { إذا السماء انشقت } فسجد فيها فلما انصرف…١٣٠٠وحدثني إبراهيم بن موسى، أخبرنا عيسى، عن الأوزاعي، ح قال وحدثنا محمد…١٣٠١وعمرو الناقد، قالا حدثنا سفيان بن عيينة، عن أيوب بن موسى، عن عطاء بن…١٣٠٢أنه قال سجد رسول الله صلى الله عليه وسلم في { إذا السماء انشقت } و {…١٣٠٣وحدثني حرملة بن يحيى، حدثنا ابن وهب، أخبرني عمرو بن الحارث، عن عبيد…١٣٠٤ومحمد بن عبد الأعلى، قالا حدثنا المعتمر، عن أبيه، عن بكر، عن أبي…١٣٠٥بهذا الإسناد . غير أنهم لم يقولوا خلف أبي القاسم صلى الله عليه وسلم .١٣٠٦وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا عمر، حدثنا سفيان، عن مسلم بن أبي مريم، عن علي بن عبد…١٣١٣ومنصور، عن مجاهد، عن أبي معمر، أن أميرا، كان بمكة يسلم تسليمتين فقال…١٣١٤- قال شعبة - رفعه مرة - أن أميرا أو رجلا سلم تسليمتين فقال عبد الله…
