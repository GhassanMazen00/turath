تعلمني الأعراب…١٠٢٠قال لقد كانت صلاة الظهر تقام فيذهب الذاهب إلى البقيع فيقضي حاجته ثم…١٠٢١وحدثني محمد بن حاتم، حدثنا عبد الرحمن بن مهدي، عن معاوية بن صالح، عن…١٠٢٢وعبد الله بن عمرو بن العاص، وعبد عبد الله بن نمير، حدثنا أبي، حدثنا مسعر، عن عدي بن…١٠٤٠يا معاذ أفتان أنت اقرأ بكذا واقرأ بكذا١٠٤١أتريد أن تكون فتانا يا معاذ إذا أممت الناس فاقرأ بالشمس وضحاها . وسبح…١٠٤٢أن معاذ بن جبل، كان يصلي مع رسول الله صلى الله عليه وسلم العشاء…١٠٤٣وأبو الربيع الزهراني، قال أبو الربيع حدثنا حماد، حدثنا أيوب، عن عمرو…١٠٤٤يا أيها الناس إن منكم منفرين فأيكم أم الناس فليوجز فإن من ورائه…١٠٤٥ووكيع، ح قال وحدثنا ابن نمير، حدثنا أبي ح، وحدثنا ابن أبي عمر، حدثنا…١٠٤٦إذا أم أحدكم الناس فليخفف فإن فيهم الصغير والكبير والضعيف والمريض…١٠٤٧إذا ما قام أحدكم للناس فليخفف الصلاة فإن فيهم الكبير وفيهم الضعيف…١٠٤٨إذا صلى أحدكم للناس فليخفف فإن في الناس الضعيف والسقيم وذا الحاجة١٠٤٩أنه سمع أبا هريرة، يقول قال رسول الله صلى الله عليه وسلم . بمثله غير…١٠٥٠أن النبي صلى الله عليه وسلم قال له " أم قومك " . قال قلت يا رسول الله…١٠٥١إذا أممت قوما فأخف بهم الصلاة١٠٥٢وأبو الربيع الزهراني، قالا حدثنا حماد بن زيد، عن عبد العزيز بن صهيب،…١٠٥٣وقتيبة بن سعيد، قال يحيى أخبرنا وقال، قتيبة حدثنا أبو عوانة، عن…١٠٥٤ويحيى بن أيوب، وقتيبة بن سعيد، وعلي بن حجر، قال يحيى بن يحيى أخبرنا…١٠٥٥قال أنس كان رسول الله صلى الله عليه وسلم يسمع بكاء الصبي مع أمه وهو…١٠٥٦إني لأدخل الصلاة أريد إطالتها فأسمع بكاء الصبي فأخفف من شدة وجد أمه به١٠٥٧قال رمقت الصلاة مع محمد صلى الله عليه وسلم فوجدت قيامه فركعته…١٠٥٨وحدثنا عبيد الله بن معاذ العنبري، حدثنا أبي، حدثنا شعبة، عن الحكم،…١٠٥٩وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن الحكم، أن مطر بن…١٠٦٠حدثنا خلف بن هشام، حدثنا حماد بن زيد، عن ثابت، عن أنس، قال إني لا آلو…١٠٦١سمع الله لمن حمده١٠٦٢وهو غير كذوب أنهم كانوا يصلون خلف رسول الله صلى الله عليه وسلم فإذا…١٠٦٣سمع الله لمن حمده١٠٦٤سمع الله لمن حمده١٠٦٥وابن، نمير قالا حدثنا سفيان بن عيينة، حدثنا أبان، وغيره، عن الحكم، عن…١٠٦٦حدثنا محرز بن عون بن أبي عون، حدثنا خلف بن خليفة الأشجعي أبو أحمد، عن…١٠٦٧سمع الله لمن حمده اللهم ربنا لك الحمد ملء السموات وملء الأرض وملء ما…١٠٦٨اللهم ربنا لك الحمد ملء السموات وملء الأرض وملء ما شئت من شىء بعد١٠٦٩اللهم لك الحمد ملء السماء وملء الأرض وملء ما شئت من شىء بعد اللهم…١٠٧٠كما ينقى الثوب الأبيض من الدرن١٠٧١ربنا لك الحمد ملء السموات والأرض وملء ما شئت من شىء بعد أهل الثناء…١٠٧٢اللهم ربنا لك الحمد ملء السموات وملء الأرض وما بينهما وملء ما شئت من…١٠٧٣وملء ما شئت من شىء بعد١٠٧٤أيها الناس إنه لم يبق من مبشرات النبوة إلا الرؤيا الصالحة يراها…١٠٧٥اللهم هل بلغت١٠٧٦وحرملة، قالا أخبرنا ابن وهب، عن يونس، عن ابن شهاب، قال حدثني إبراهيم…١٠٧٧أنه سمع علي بن أبي طالب، يقول نهاني رسول الله صلى الله عليه وسلم عن…١٠٧٨وحدثني أبو بكر بن إسحاق، أخبرنا ابن أبي مريم، أخبرنا محمد بن جعفر،…١٠٧٩وإسحاق، قالا أخبرنا أبو عامر العقدي، حدثنا داود بن قيس، حدثني إبراهيم…١٠٨٠وقتيبة، وابن، حجر قالوا حدثنا إسماعيل، - يعنون ابن جعفر - أخبرني…١٠٨١ولم يذكر في السجود .١٠٨٢أنه قال نهيت أن أقرأ، وأنا راكع، . لا يذكر في الإسناد عليا .١٠٨٣أقرب ما يكون العبد من ربه وهو ساجد فأكثروا الدعاء١٠٨٤اللهم اغفر لي ذنبي
