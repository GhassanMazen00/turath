٣٢١ حديثًا•صفحة ٣ من ٣
١١٤٠وعمرو الناقد، وزهير بن حرب، قالوا حدثنا سفيان بن عيينة، عن الزهري، عن…١١٤١حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن هشام، عن أبيه، عن عائشة،…١١٤٢وحدثني عمرو بن علي، حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي بكر بن…١١٤٣وذكر، عندها ما يقطع الصلاة الكلب والحمار والمرأة . فقالت عائشة قد…١١٤٤حدثنا إسحاق بن إبراهيم، أخبرنا جرير، عن منصور، عن إبراهيم، عن الأسود،…١١٤٥قالت كنت أنام بين يدى رسول الله صلى الله عليه وسلم ورجلاى في قبلته…١١٤٦قال حدثتني ميمونة، زوج النبي صلى الله عليه وسلم قالت كان رسول الله…١١٤٧وزهير بن حرب، قال زهير حدثنا وكيع، حدثنا طلحة بن يحيى، عن عبيد الله…١١٤٨أولكلكم ثوبان١١٤٩وأبي، سلمة عن أبي هريرة، عن النبي صلى الله عليه وسلم بمثله .١١٥٠أوكلكم يجد ثوبين١١٥١لا يصلي أحدكم في الثوب الواحد ليس على عاتقيه منه شىء١١٥٢أن عمر بن أبي سلمة، أخبره قال رأيت رسول الله صلى الله عليه وسلم يصلي…١١٥٣وإسحاق بن إبراهيم، عن وكيع، قال حدثنا هشام بن عروة، بهذا الإسناد غير…١١٥٤قال رأيت رسول الله صلى الله عليه وسلم يصلي في بيت أم سلمة في ثوب قد…١١٥٥وعيسى بن حماد، قالا حدثنا الليث، عن يحيى بن سعيد، عن أبي أمامة بن سهل…١١٥٦قال رأيت النبي صلى الله عليه وسلم يصلي في ثوب واحد متوشحا به .١١٥٧جميعا بهذا الإسناد وفي حديث ابن نمير قال دخلت على رسول الله صلى الله…١١٥٨أن أبا الزبير المكي، حدثه أنه، رأى جابر بن عبد الله يصلي في ثوب…١١٥٩وإسحاق بن إبراهيم، - واللفظ لعمرو - قال حدثني عيسى بن يونس، حدثنا…١١٦٠بهذا الإسناد . وفي رواية أبي كريب واضعا طرفيه على عاتقيه . ورواية أبي…
