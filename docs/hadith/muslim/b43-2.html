كتاب الفضائل

٢١٧ حديثًا•صفحة ٢ من ٢
٦١٠١بهذا الإسناد . نحو حديث يزيد بن زريع .٦١٠٢أن رسول الله صلى الله عليه وسلم توفي وهو ابن خمس وستين .٦١٠٣بهذا الإسناد .٦١٠٤قال أقام رسول الله صلى الله عليه وسلم بمكة خمس عشرة سنة يسمع الصوت…٦١٠٥أنا محمد وأنا أحمد وأنا الماحي الذي يمحى بي الكفر وأنا الحاشر الذي…٦١٠٦إن لي أسماء أنا محمد وأنا أحمد وأنا الماحي الذي يمحو الله بي الكفر…٦١٠٧وحدثني عبد الملك بن شعيب بن الليث، قال حدثني أبي، عن جدي، حدثني عقيل،…٦١٠٨أنا محمد وأحمد والمقفي والحاشر ونبي التوبة ونبي الرحمة٦١٠٩ما بال رجال بلغهم عني أمر ترخصت فيه فكرهوه وتنزهوا عنه فوالله لأنا…٦١١٠بإسناد جرير نحو حديثه .٦١١١ما بال أقوام يرغبون عما رخص لي فيه فوالله لأنا أعلمهم بالله وأشدهم له…٦١١٢اسق يا زبير ثم أرسل الماء إلى جارك٦١١٣ما نهيتكم عنه فاجتنبوه وما أمرتكم به فافعلوا منه ما استطعتم فإنما…٦١١٤بهذا الإسناد مثله سواء .٦١١٥ذروني ما تركتكم٦١١٦إن أعظم المسلمين في المسلمين جرما من سأل عن شىء لم يحرم على المسلمين…٦١١٧أعظم المسلمين في المسلمين جرما من سأل عن أمر لم يحرم فحرم على الناس…٦١١٨رجل سأل عن شىء ونقر عنه٦١١٩عرضت على الجنة والنار فلم أر كاليوم في الخير والشر ولو تعلمون ما أعلم…٦١٢٠وحدثنا محمد بن معمر بن ربعي القيسي، حدثنا روح بن عبادة، حدثنا شعبة،…٦١٢١من أحب أن يسألني عن شىء فليسألني عنه فوالله لا تسألونني عن شىء إلا…٦١٢٢حدثنا عبد بن حميد، أخبرنا عبد الرزاق، أخبرنا معمر، ح وحدثنا عبد الله…٦١٢٣سلوني لا تسألوني عن شىء إلا بينته لكم٦١٢٤بهذه القصة .٦١٢٥ومحمد بن العلاء الهمداني، قالا حدثنا أبو أسامة عن بريد، عن أبي بردة،…٦١٢٦حدثنا قتيبة بن سعيد الثقفي، وأبو كامل الجحدري - وتقاربا في اللفظ وهذا…٦١٢٧وعباس بن عبد العظيم العنبري، وأحمد بن، جعفر المعقري قالوا حدثنا النضر…٦١٢٨لو لم تفعلوا لصلح٦١٢٩والذي نفس محمد في يده ليأتين على أحدكم يوم ولا يراني ثم لأن يراني أحب…٦١٣٠أنا أولى الناس بابن مريم الأنبياء أولاد علات وليس بيني وبينه نبي٦١٣١أنا أولى الناس بعيسى الأنبياء أبناء علات وليس بيني وبين عيسى نبي٦١٣٢أنا أولى الناس بعيسى ابن مريم في الأولى والآخرة٦١٣٣ما من مولود يولد إلا نخسه الشيطان فيستهل صارخا من نخسة الشيطان إلا…٦١٣٤يمسه حين يولد فيستهل صارخا من مسة الشيطان إياه٦١٣٥كل بني آدم يمسه الشيطان يوم ولدته أمه إلا مريم وابنها٦١٣٦صياح المولود حين يقع نزغة من الشيطان٦١٣٧رأى عيسى ابن مريم رجلا يسرق فقال له عيسى سرقت قال كلا والذي لا إله…٦١٣٨ذاك إبراهيم عليه السلام٦١٣٩يقول قال رجل يا رسول الله . بمثله .٦١٤٠وحدثني محمد بن المثنى، حدثنا عبد الرحمن، عن سفيان، عن المختار، قال…٦١٤١اختتن إبراهيم النبي عليه السلام وهو ابن ثمانين سنة بالقدوم٦١٤٢أن سعيد بن المسيب، وأبا، عبيد أخبراه عن أبي هريرة، عن رسول الله صلى…٦١٤٣يغفر الله للوط إنه أوى إلى ركن شديد٦١٤٥لم يكذب إبراهيم النبي عليه السلام قط إلا ثلاث كذبات ثنتين في ذات الله…٦١٤٦كانت بنو إسرائيل يغتسلون عراة ينظر بعضهم إلى سوأة بعض وكان موسى عليه…٦١٤٧قال كان موسى عليه السلام رجلا حييا - قال - فكان لا يرى متجردا - قال -…٦١٤٨فلو كنت ثم لأريتكم قبره إلى جانب الطريق تحت الكثيب الأحمر٦١٤٩جاء ملك الموت إلى موسى عليه السلام فقال له أجب ربك - قال - فلطم موسى…٦١٥٠بمثل هذا الحديث .٦١٥١حدثني زهير بن حرب، حدثنا حجين بن المثنى، حدثنا عبد العزيز بن عبد…٦١٥٢بهذا الإسناد سواء .٦١٥٣لا تخيروني على موسى فإن الناس يصعقون فأكون أول من يفيق فإذا موسى باطش…٦١٥٤وسعيد بن المسيب، عن أبي هريرة، قال استب رجل من المسلمين ورجل من…٦١٥٥فلا أدري أكان ممن صعق فأفاق قبلي أو اكتفى بصعقة الطور٦١٥٦لا تخيروا بين الأنبياء٦١٥٧أتيت - وفي رواية هداب مررت - على موسى ليلة أسري بي عند الكثيب الأحمر…٦١٥٨مررت على موسى وهو يصلي في قبره٦١٥٩قال - يعني الله تبارك وتعالى - لا ينبغي لعبد لي - وقال ابن المثنى…٦١٦٠ما ينبغي لعبد أن يقول أنا خير من يونس بن متى٦١٦١ومحمد بن المثنى، وعبيد الله بن سعيد، قالوا حدثنا يحيى، بن سعيد عن…٦١٦٢كان زكرياء نجارا٦١٦٣قام موسى عليه السلام خطيبا في بني إسرائيل فسئل أى الناس أعلم فقال أنا…٦١٦٤حدثني محمد بن عبد الأعلى القيسي، حدثنا المعتمر بن سليمان التيمي، عن…٦١٦٥إنه بينما موسى عليه السلام في قومه يذكرهم بأيام الله وأيام الله…٦١٦٦نحو حديثه .٦١٦٧أن النبي صلى الله عليه وسلم قرأ { لتخذت عليه أجرا } .٦١٦٨بينما موسى في ملإ من بني إسرائيل إذ جاءه رجل فقال له هل تعلم أحدا…
