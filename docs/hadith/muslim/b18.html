٨٦ حديثًا
٣٦٥٢مره فليراجعها ثم ليتركها حتى تطهر ثم تحيض ثم تطهر ثم إن شاء أمسك بعد…٣٦٥٣وقتيبة، وابن، رمح - واللفظ ليحيى - قال قتيبة حدثنا ليث، وقال الآخران،…٣٦٥٤مره فليراجعها ثم ليدعها حتى تطهر ثم تحيض حيضة أخرى فإذا طهرت فليطلقها…٣٦٥٥وابن المثنى، قالا حدثنا عبد الله بن إدريس، عن عبيد الله، بهذا الإسناد…٣٦٥٦أن ابن عمر، طلق امرأته وهى حائض فسأل عمر النبي صلى الله عليه وسلم…٣٦٥٧مره فليراجعها حتى تحيض حيضة أخرى مستقبلة سوى حيضتها التي طلقها فيها…٣٦٥٨بهذا الإسناد غير أنه قال قال ابن عمر فراجعتها وحسبت لها التطليقة التي…٣٦٥٩مره فليراجعها ثم ليطلقها طاهرا أو حاملا٣٦٦٠مره فليراجعها حتى تطهر ثم تحيض حيضة أخرى ثم تطهر ثم يطلق بعد أو يمسك٣٦٦١وحدثني علي بن حجر السعدي، حدثنا إسماعيل بن إبراهيم، عن أيوب، عن ابن،…٣٦٦٢وقتيبة، قالا حدثنا حماد، عن أيوب، بهذا الإسناد . نحوه غير أنه قال…٣٦٦٣يطلقها في قبل عدتها٣٦٦٤قال قلت لابن عمر رجل طلق امرأته وهى حائض فقال أتعرف عبد الله بن عمر…٣٦٦٥ليراجعها . فإذا طهرت فإن شاء فليطلقها٣٦٦٦مره فليراجعها فإذا طهرت فليطلقها لطهرها٣٦٦٧مره فليراجعها ثم إذا طهرت فليطلقها٣٦٦٨بهذا الإسناد غير أن في، حديثهما " ليرجعها " . وفي حديثهما قال قلت له…٣٦٦٩أنه سمع ابن عمر، يسأل عن رجل، طلق امرأته حائضا فقال أتعرف عبد الله بن…٣٦٧٠أنه سمع عبد الرحمن بن أيمن، مولى عزة يسأل ابن عمر وأبو الزبير يسمع…٣٦٧١. نحو هذه القصة .٣٦٧٢أنه سمع عبد الرحمن بن أيمن، مولى عروة يسأل ابن عمر وأبو الزبير يسمع…٣٦٧٣ومحمد بن رافع، - واللفظ لابن رافع - قال إسحاق أخبرنا وقال ابن رافع،…٣٦٧٤أن أبا الصهباء، قال لابن عباس أتعلم أنما كانت الثلاث تجعل واحدة على…٣٦٧٥أن أبا الصهباء، قال لابن عباس هات من هناتك ألم يكن الطلاق الثلاث على…٣٦٧٦وحدثنا زهير بن حرب، حدثنا إسماعيل بن إبراهيم، عن هشام، - يعني…٣٦٧٧بن أبي كثير أن يعلى بن حكيم، أخبره أن سعيد بن جبير أخبره أنه، سمع ابن…٣٦٧٨بل شربت عسلا عند زينب بنت جحش ولن أعود له٣٦٧٩حدثنا أبو كريب، محمد بن العلاء وهارون بن عبد الله قالا حدثنا أبو…٣٦٨٠بهذا الإسناد نحوه .٣٦٨١إني ذاكر لك أمرا فلا عليك أن لا تعجلي حتى تستأمري أبويك٣٦٨٢حدثنا سريج بن يونس، حدثنا عباد بن عباد، عن عاصم، عن معاذة العدوية، عن…٣٦٨٣بهذا الإسناد . نحوه .٣٦٨٤حدثنا يحيى بن يحيى التميمي، أخبرنا عبثر، عن إسماعيل بن أبي خالد، عن…٣٦٨٥وحدثناه أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن إسماعيل بن أبي،…٣٦٨٦أن رسول الله صلى الله عليه وسلم خير نساءه فلم يكن طلاقا .٣٦٨٧وحدثني إسحاق بن منصور، أخبرنا عبد الرحمن، عن سفيان، عن عاصم الأحول،…٣٦٨٨حدثنا يحيى بن يحيى، وأبو بكر بن أبي شيبة وأبو كريب قال يحيى أخبرنا…٣٦٨٩بمثله .٣٦٩٠هن حولي كما ترى يسألنني النفقة٣٦٩١ما يبكيك يا ابن الخطاب٣٦٩٢أنه سمع عبد الله بن عباس، يحدث قال مكثت سنة وأنا أريد، أن أسأل، عمر…٣٦٩٣وحدثنا محمد بن المثنى، حدثنا عفان، حدثنا حماد بن سلمة، أخبرني يحيى…٣٦٩٤وزهير بن حرب، - واللفظ لأبي بكر - قالا حدثنا سفيان بن عيينة، عن يحيى…٣٦٩٦ومحمد بن أبي عمر، - وتقاربا في لفظ الحديث - قال ابن أبي عمر حدثنا…٣٦٩٧ليس لك عليه نفقة٣٦٩٨لا نفقة لك ولا سكنى٣٦٩٩لا نفقة لك فانتقلي فاذهبي إلى ابن أم مكتوم فكوني عنده فإنه رجل أعمى…٣٧٠٠ليست لها نفقة وعليها العدة٣٧٠١لا تفوتينا بنفسك٣٧٠٢وعبد بن حميد، جميعا عن يعقوب بن إبراهيم، بن سعد حدثنا أبي، عن صالح،…٣٧٠٣بهذا الإسناد مثله مع قول عروة إن عائشة أنكرت ذلك على فاطمة .٣٧٠٤وعبد بن حميد، - واللفظ لعبد - قالا أخبرنا عبد، الرزاق أخبرنا معمر، عن…٣٧٠٥وحصين، ومغيرة، وأشعث، ومجالد وإسماعيل بن أبي خالد وداود كلهم عن…٣٧٠٦وداود، ومغيرة، وإسماعيل، وأشعث عن الشعبي، أنه قال دخلت على فاطمة بنت…٣٧٠٧ثلاثا أين تعتد قالت طلقني بعلي ثلاثا فأذن لي النبي صلى الله عليه وسلم…٣٧٠٨ليس لها سكنى ولا نفقة٣٧٠٩انتقلي إلى بيت ابن عمك عمرو بن أم مكتوم فاعتدي عنده٣٧١٠وحدثناه محمد بن عمرو بن جبلة، حدثنا أبو أحمد، حدثنا عمار بن رزيق، عن…٣٧١١بقصته .٣٧١٢إذا حللت فآذنيني٣٧١٣تقول أرسل إلى زوجي أبو عمرو بن حفص بن المغيرة عياش بن أبي ربيعة…٣٧١٤وحدثني إسحاق بن منصور، أخبرنا أبو عاصم، حدثنا سفيان الثوري، حدثني أبو…٣٧١٥وحدثنا عبيد الله بن معاذ العنبري، حدثنا أبي، حدثنا شعبة، حدثني أبو…٣٧١٦قالت طلقني زوجي ثلاثا فلم يجعل لي رسول الله صلى الله عليه وسلم سكنى…٣٧١٧تزوج يحيى بن سعيد بن العاص بنت عبد الرحمن بن الحكم فطلقها فأخرجها من…٣٧١٨بنت قيس قالت قلت يا رسول الله زوجي طلقني ثلاثا وأخاف أن يقتحم على .…٣٧١٩أنها قالت ما لفاطمة خير أن تذكر هذا . قال تعني قولها لا سكنى ولا نفقة…٣٧٢٠قال قال عروة بن الزبير لعائشة ألم ترى إلى فلانة بنت الحكم طلقها زوجها…٣٧٢١بلى فجدي نخلك فإنك عسى أن تصدقي أو تفعلي معروفا٣٧٢٢وحرملة بن يحيى، - وتقاربا في اللفظ - قال حرملة حدثنا وقال أبو الطاهر،…٣٧٢٣أن أبا سلمة بن عبد الرحمن، وابن، عباس اجتمعا عند أبي هريرة وهما…٣٧٢٤وعمرو، الناقد قالا حدثنا يزيد بن هارون، كلاهما عن يحيى بن سعيد، بهذا…٣٧٢٨لا يحل لامرأة تؤمن بالله واليوم الآخر تحد على ميت فوق ثلاث إلا على…٣٧٢٩لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد فوق ثلاث إلا على زوج…٣٧٣١وحدثته زينب، عن أمها، وعن زينب، زوج النبي صلى الله عليه وسلم أو عن…٣٧٣٢قد كانت إحداكن تكون في شر بيتها في أحلاسها - أو في شر أحلاسها في…٣٧٣٣بالحديثين جميعا حديث أم سلمة في الكحل وحديث أم سلمة وأخرى من أزواج…٣٧٣٤لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد فوق ثلاث إلا على زوج…٣٧٣٥لا يحل لامرأة تؤمن بالله واليوم الآخر -
