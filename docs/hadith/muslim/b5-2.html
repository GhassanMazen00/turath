كتاب الْمَسَاجِدِ وَمَوَاضِعِ الصَّلاَةِ

٣٩٦ حديثًا•صفحة ٢ من ٣
١٣١٥وحدثنا إسحاق بن إبراهيم، أخبرنا أبو عامر العقدي، حدثنا عبد الله بن…١٣١٦حدثنا زهير بن حرب، حدثنا سفيان بن عيينة، عن عمرو، قال أخبرني بذا أبو…١٣١٧مولى ابن عباس أنه سمعه يخبر، عن ابن عباس، قال ما كنا نعرف انقضاء صلاة…١٣١٨أن أبا معبد، مولى ابن عباس أخبره أن ابن عباس أخبره أن رفع الصوت…١٣١٩إنما تفتن يهود١٣٢٠وحرملة بن يحيى، وعمرو بن سواد، قال حرملة أخبرنا وقال الآخران، حدثنا…١٣٢١صدقتا إنهم يعذبون عذابا تسمعه البهائم١٣٢٢حدثنا هناد بن السري، حدثنا أبو الأحوص، عن أشعث، عن أبيه، عن مسروق، عن…١٣٢٣وزهير بن حرب، قالا حدثنا يعقوب بن إبراهيم بن سعد، قال حدثنا أبي، عن…١٣٢٤إذا تشهد أحدكم فليستعذ بالله من أربع يقول اللهم إني أعوذ بك من عذاب…١٣٢٥اللهم إني أعوذ بك من عذاب القبر وأعوذ بك من فتنة المسيح الدجال وأعوذ…١٣٢٦إذا فرغ أحدكم من التشهد الآخر فليتعوذ بالله من أربع من عذاب جهنم ومن…١٣٢٧اللهم إني أعوذ بك من عذاب القبر وعذاب النار وفتنة المحيا والممات وشر…١٣٢٨عوذوا بالله من عذاب الله عوذوا بالله من عذاب القبر عوذوا بالله من…١٣٢٩حدثنا محمد بن عباد، حدثنا سفيان، عن ابن طاوس، عن أبيه، عن أبي هريرة،…١٣٣٠وحدثنا محمد بن عباد، وأبو بكر بن أبي شيبة وزهير بن حرب قالوا حدثنا…١٣٣١حدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، عن بديل، عن عبد…١٣٣٣قولوا اللهم إنا نعوذ بك من عذاب جهنم وأعوذ بك من عذاب القبر وأعوذ بك…١٣٣٤اللهم أنت السلام ومنك السلام تباركت ذا الجلال والإكرام١٣٣٥اللهم أنت السلام ومنك السلام تباركت ذا الجلال والإكرام١٣٣٦يا ذا الجلال والإكرام١٣٣٧يا ذا الجلال والإكرام١٣٣٨لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…١٣٣٩وحدثناه أبو بكر بن أبي شيبة، وأبو كريب وأحمد بن سنان قالوا حدثنا أبو…١٣٤٠وهو على كل شىء قدير١٣٤١كاتب المغيرة بن شعبة قال كتب معاوية إلى المغيرة . بمثل حديث منصور…١٣٤٢لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…١٣٤٣لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…١٣٤٤مولى لهم أن عبد الله بن الزبير، كان يهلل دبر كل صلاة . بمثل حديث ابن…١٣٤٥يخطب على هذا المنبر وهو يقول كان رسول الله صلى الله عليه وسلم يقول…١٣٤٦أن أبا الزبير المكي، حدثه أنه، سمع عبد الله بن الزبير، وهو يقول في…١٣٤٧- وهذا حديث قتيبة أن فقراء، المهاجرين أتوا رسول الله صلى الله عليه…١٣٤٨وحدثني أمية بن بسطام العيشي، حدثنا يزيد بن زريع، حدثنا روح، عن سهيل،…١٣٤٩معقبات لا يخيب قائلهن - أو فاعلهن - دبر كل صلاة مكتوبة ثلاث وثلاثون…١٣٥٢من سبح الله في دبر كل صلاة ثلاثا وثلاثين وحمد الله ثلاثا وثلاثين وكبر…١٣٥٣قال قال رسول الله صلى الله عليه وسلم بمثله .١٣٥٦ويونس المؤدب، و غيرهما قالوا حدثنا عبد الواحد بن زياد، قال حدثني…١٣٥٧أيكم المتكلم بالكلمات١٣٥٨من القائل كلمة كذا وكذا١٣٥٩إذا أقيمت الصلاة فلا تأتوها تسعون وأتوها تمشون وعليكم السكينة فما…١٣٦٠إذا ثوب للصلاة فلا تأتوها وأنتم تسعون وأتوها وعليكم السكينة فما…١٣٦١إذا نودي بالصلاة فأتوها وأنتم تمشون وعليكم السكينة فما أدركتم فصلوا…١٣٦٢إذا ثوب بالصلاة فلا يسع إليها أحدكم ولكن ليمش وعليه السكينة والوقار…١٣٦٣أن أباه، أخبره قال بينما نحن نصلي مع رسول الله صلى الله عليه وسلم…١٣٦٤بهذا الإسناد .١٣٦٧وحرملة بن يحيى، قالا حدثنا ابن وهب، أخبرني يونس، عن ابن شهاب، قال…١٣٦٨قال أقيمت الصلاة وصف الناس صفوفهم وخرج رسول الله صلى الله عليه وسلم…١٣٦٩أن الصلاة، كانت تقام لرسول الله صلى الله عليه وسلم فيأخذ الناس مصافهم…١٣٧٠قال كان بلال يؤذن إذا دحضت فلا يقيم حتى يخرج النبي صلى الله عليه وسلم…١٣٧١من أدرك ركعة من الصلاة فقد أدرك الصلاة١٣٧٢من أدرك ركعة من الصلاة مع الإمام فقد أدرك الصلاة١٣٧٣وعمرو الناقد، وزهير بن حرب، قالوا حدثنا ابن عيينة، ح قال وحدثنا أبو…١٣٧٤من أدرك ركعة من الصبح قبل أن تطلع الشمس فقد أدرك الصبح ومن أدرك ركعة…١٣٧٥وحدثنا عبد بن حميد، أخبرنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن أبي…١٣٧٦من أدرك من العصر سجدة قبل أن تغرب الشمس أو من الصبح قبل أن تطلع فقد…١٣٧٧من أدرك من العصر ركعة قبل أن تغرب الشمس فقد أدرك ومن أدرك من الفجر…١٣٧٨بهذا الإسناد .١٣٧٩نزل جبريل فأمني فصليت معه ثم صليت معه ثم صليت معه ثم صليت معه ثم صليت…١٣٨٠أن عمر بن عبد العزيز، أخر الصلاة يوما فدخل عليه عروة بن الزبير فأخبره…١٣٨١قال عروة ولقد حدثتني عائشة، زوج النبي صلى الله عليه وسلم أن رسول الله…١٣٨٢وعمرو الناقد، قال عمرو حدثنا سفيان، عن الزهري، عن عروة، عن عائشة، كان…١٣٨٣أن عائشة، زوج النبي صلى الله عليه وسلم أخبرته أن رسول الله صلى الله…١٣٨٤وابن، نمير قالا حدثنا وكيع، عن هشام، عن أبيه، عن عائشة، قالت كان رسول…١٣٨٥إذا صليتم الفجر فإنه وقت إلى أن يطلع قرن الشمس الأول ثم إذا صليتم…١٣٨٦وقت الظهر ما لم يحضر العصر ووقت العصر ما لم تصفر الشمس ووقت المغرب ما…١٣٨٧بهذا الإسناد . وفي حديثهما قال شعبة رفعه مرة ولم يرفعه مرتين .١٣٨٨وقت الظهر إذا زالت الشمس وكان ظل الرجل كطوله ما لم يحضر العصر ووقت…١٣٨٩وقت صلاة الفجر ما لم يطلع قرن الشمس الأول ووقت صلاة الظهر إذا زالت…١٣٩٠لا يستطاع العلم براحة الجسم .١٣٩١وعبيد الله بن سعيد، كلاهما عن الأزرق، - قال زهير حدثنا إسحاق بن يوسف…١٣٩٢اشهد معنا الصلاة١٣٩٣الوقت بين هذين١٣٩٤سمعه منه، عن أبيه، أن سائلا، أتى النبي صلى الله عليه وسلم فسأله عن…١٣٩٥إذا اشتد الحر فأبردوا بالصلاة فإن شدة الحر من فيح جهنم١٣٩٦أن ابن شهاب، أخبره قال أخبرني أبو سلمة، وسعيد بن المسيب، أنهما سمعا…١٣٩٧إذا كان اليوم الحار فأبردوا بالصلاة فإن شدة الحر من فيح جهنم١٣٩٨إن هذا الحر من فيح جهنم فأبردوا بالصلاة١٣٩٩أبردوا عن الحر في الصلاة فإن شدة الحر من فيح جهنم١٤٠٠يحدث أنه سمع زيد بن وهب، يحدث عن أبي ذر، قال أذن مؤذن رسول الله صلى…١٤٠١اشتكت النار إلى ربها فقالت يا رب أكل بعضي بعضا . فأذن لها بنفسين نفس…١٤٠٢إذا كان الحر فأبردوا عن الصلاة فإن شدة الحر من فيح جهنم١٤٠٣قالت النار رب أكل بعضي بعضا فأذن لي أتنفس . فأذن لها بنفسين نفس في…١٤٠٤ومحمد بن بشار، كلاهما عن يحيى القطان، وابن، مهدي - قال ابن المثنى…١٤٠٥وحدثنا أبو بكر بن أبي شيبة، حدثنا أبو الأحوص، سلام بن سليم عن أبي…١٤٠٦وعون بن سلام، - قال عون أخبرنا وقال ابن يونس، واللفظ، له حدثنا زهير،…١٤٠٧حدثنا يحيى بن يحيى، حدثنا بشر بن المفضل، عن غالب القطان، عن بكر بن…١٤٠٨أنه أخبره أن رسول الله صلى الله عليه وسلم كان يصلي العصر والشمس…١٤٠٩. أن رسول الله صلى الله عليه وسلم كان يصلي العصر بمثله سواء .١٤١٠وحدثنا يحيى بن يحيى، قال قرأت على مالك عن ابن شهاب، عن أنس بن مالك،…١٤١١وحدثنا يحيى بن يحيى، قال قرأت على مالك عن إسحاق بن عبد الله بن أبي…١٤١٢تلك صلاة المنافق يجلس يرقب الشمس حتى إذا كانت بين قرنى الشيطان قام…١٤١٣وحدثنا منصور بن أبي مزاحم، حدثنا عبد الله بن المبارك، عن أبي بكر بن…١٤١٤ومحمد بن سلمة المرادي، وأحمد بن عيسى، - وألفاظهم متقاربة - قال عمرو…١٤١٥حدثنا محمد بن مهران الرازي، حدثنا الوليد بن مسلم، حدثنا الأوزاعي، عن…١٤١٦وشعيب بن إسحاق الدمشقي، قالا حدثنا الأوزاعي، بهذا الإسناد غير أنه قال…١٤١٧الذي تفوته صلاة العصر كأنما وتر أهله وماله١٤١٨وعمرو الناقد، قالا حدثنا سفيان، عن الزهري، عن سالم، عن أبيه، . قال…١٤١٩من فاتته العصر فكأنما وتر أهله وماله١٤٢٠ملأ الله قبورهم وبيوتهم نارا كما حبسونا وشغلونا عن الصلاة الوسطى حتى…١٤٢١بهذا الإسناد .١٤٢٢شغلونا عن صلاة الوسطى حتى آبت الشمس ملأ الله قبورهم نارا أو بيوتهم أو…١٤٢٣بيوتهم وقبورهم١٤٢٤شغلونا عن الصلاة الوسطى حتى غربت الشمس ملأ الله قبورهم وبيوتهم - أو…١٤٢٥شغلونا عن الصلاة الوسطى صلاة العصر ملأ الله بيوتهم وقبورهم نارا١٤٢٦شغلونا عن الصلاة الوسطى صلاة العصر ملأ الله أجوافهم وقبورهم نارا١٤٢٧وحدثنا يحيى بن يحيى التميمي، قال قرأت على مالك عن زيد بن أسلم، عن…١٤٣٠فوالله إن صليتها١٤٣١وإسحاق بن إبراهيم، قال أبو بكر حدثنا وقال، إسحاق أخبرنا وكيع، عن علي…١٤٣٢يتعاقبون فيكم ملائكة بالليل وملائكة بالنهار ويجتمعون في صلاة الفجر…١٤٣٣والملائكة يتعاقبون فيكم١٤٣٤أما إنكم سترون ربكم كما ترون هذا القمر لا تضامون في رؤيته فإن استطعتم…١٤٣٥أما إنكم ستعرضون على ربكم فترونه كما ترون هذا القمر١٤٣٦لن يلج النار أحد صلى قبل طلوع الشمس وقبل غروبها١٤٣٧لا يلج النار من صلى قبل طلوع الشمس وقبل غروبها١٤٣٨من صلى البردين دخل الجنة١٤٣٩بهذا الإسناد ونسبا أبا بكر فقالا ابن أبي موسى .١٤٤٠أن رسول الله صلى الله عليه وسلم كان يصلي المغرب إذا غربت الشمس وتوارت…١٤٤١وحدثنا محمد بن مهران الرازي، حدثنا الوليد بن مسلم، حدثنا الأوزاعي،…١٤٤٢وحدثنا إسحاق بن إبراهيم الحنظلي، أخبرنا شعيب بن إسحاق الدمشقي، حدثنا…١٤٤٣ما ينتظرها أحد من أهل الأرض غيركم١٤٤٤بهذا الإسناد . مثله . ولم يذكر قول الزهري وذكر لي . وما بعده .١٤٤٥إنه لوقتها لولا أن أشق على أمتي١٤٤٦إنكم لتنتظرون صلاة ما ينتظرها أهل دين غيركم ولولا أن يثقل على أمتي…١٤٤٧ليس أحد من أهل الأرض الليلة ينتظر الصلاة غيركم١٤٤٨إن الناس قد صلوا وناموا وإنكم لم تزالوا في صلاة ما انتظرتم الصلاة١٤٤٩وحدثني حجاج بن الشاعر، حدثنا أبو زيد، سعيد بن الربيع حدثنا قرة بن…١٤٥٠وحدثني عبد الله بن الصباح العطار، حدثنا عبيد الله بن عبد المجيد…١٤٥١على رسلكم أعلمكم وأبشروا أن من نعمة الله عليكم أنه ليس من الناس أحد…١٤٥٢لولا أن يشق على أمتي لأمرتهم أن يصلوها كذلك١٤٥٣وقتيبة بن سعيد، وأبو بكر بن أبي شيبة قال يحيى أخبرنا وقال الآخران،…١٤٥٤قال كان رسول الله صلى الله عليه وسلم يصلي الصلوات نحوا من صلاتكم وكان…١٤٥٥لا تغلبنكم الأعراب على اسم صلاتكم ألا إنها العشاء وهم يعتمون بالإبل١٤٥٦لا تغلبنكم الأعراب على اسم صلاتكم العشاء فإنها في كتاب الله العشاء…١٤٥٧وعمرو الناقد، وزهير بن حرب، كلهم عن سفيان بن عيينة، - قال عمرو حدثنا…١٤٥٨أن ابن شهاب، أخبره قال أخبرني عروة بن الزبير، أن عائشة، زوج النبي صلى…١٤٥٩وإسحاق بن موسى الأنصاري، قالا حدثنا معن، عن مالك، عن يحيى بن سعيد، عن…١٤٦٠وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن سعد بن إبراهيم، عن…١٤٦١سمع محمد بن عمرو بن الحسن بن علي، قال كان الحجاج يؤخر الصلوات فسألنا…١٤٦٢وحدثنا يحيى بن حبيب الحارثي، حدثنا خالد بن الحارث، حدثنا شعبة، أخبرني…١٤٦٣يقول كان رسول الله صلى الله عليه وسلم لا يبالي بعض تأخير صلاة العشاء…١٤٦٤وحدثناه أبو كريب، حدثنا سويد بن عمرو الكلبي، عن حماد بن سلمة، عن سيار…١٤٦٥كيف أنت إذا كانت عليك أمراء يؤخرون الصلاة عن وقتها أو يميتون الصلاة…١٤٦٦يا أبا ذر إنه سيكون بعدي أمراء يميتون الصلاة فصل الصلاة لوقتها فإن…١٤٦٧فإن أدركت القوم وقد صلوا كنت قد أحرزت صلاتك وإلا كانت لك نافلة١٤٦٨كيف أنت إذا بقيت في قوم يؤخرون الصلاة عن وقتها١٤٦٩صل الصلاة لوقتها فإن أدركتك الصلاة معهم فصل ولا تقل إني قد صليت فلا…١٤٧٠كيف أنتم - أو قال كيف أنت - إذا بقيت في قوم يؤخرون الصلاة عن وقتها…١٤٧١صلوا الصلاة لوقتها واجعلوا صلاتكم معهم نافلة١٤٧٢صلاة الجماعة أفضل من صلاة أحدكم وحده بخمسة وعشرين جزءا١٤٧٣تفضل صلاة في الجميع على صلاة الرجل وحده خمسا وعشرين درجة
