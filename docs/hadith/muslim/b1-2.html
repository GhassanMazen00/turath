كتاب الإيمان

٤٣٦ حديثًا•صفحة ٢ من ٣
٢٤٥فعصيت فلي النار٢٤٦إن بين الرجل وبين الشرك والكفر ترك الصلاة٢٤٧بين الرجل وبين الشرك والكفر ترك الصلاة٢٤٩وعبد بن حميد، عن عبد الرزاق، أخبرنا معمر، عن الزهري، بهذا الإسناد…٢٥٠الإيمان بالله والجهاد في سبيله٢٥١فتعين الصانع أو تصنع لأخرق٢٥٢قال سألت رسول الله صلى الله عليه وسلم أى العمل أفضل قال " الصلاة…٢٥٣الصلاة على مواقيتها٢٥٤الصلاة على وقتها٢٥٥حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، بهذا الإسناد مثله…٢٥٦أفضل الأعمال - أو العمل - الصلاة لوقتها وبر الوالدين٢٥٧أن تجعل لله ندا وهو خلقك٢٥٨أن تدعو لله ندا وهو خلقك٢٥٩ألا أنبئكم بأكبر الكبائر - ثلاثا - الإشراك بالله وعقوق الوالدين…٢٦٠الشرك بالله وعقوق الوالدين وقتل النفس وقول الزور٢٦١الشرك بالله وقتل النفس وعقوق الوالدين٢٦٢اجتنبوا السبع الموبقات٢٦٣من الكبائر شتم الرجل والديه٢٦٤ومحمد بن المثنى، وابن، بشار جميعا عن محمد بن جعفر، عن شعبة، ح وحدثني…٢٦٥لا يدخل الجنة من كان في قلبه مثقال ذرة من كبر٢٦٦لا يدخل النار أحد في قلبه مثقال حبة خردل من إيمان ولا يدخل الجنة أحد…٢٦٧لا يدخل الجنة من كان في قلبه مثقال ذرة من كبر٢٦٨من مات يشرك بالله شيئا دخل النار٢٦٩من مات لا يشرك بالله شيئا دخل الجنة ومن مات يشرك بالله شيئا دخل النار٢٧٠من لقي الله لا يشرك به شيئا دخل الجنة ومن لقيه يشرك به دخل النار٢٧١أن نبي الله صلى الله عليه وسلم قال بمثله .٢٧٢أتاني جبريل - عليه السلام - فبشرني أنه من مات من أمتك لا يشرك بالله…٢٧٣ما من عبد قال لا إله إلا الله ثم مات على ذلك إلا دخل الجنة٢٧٤أنه أخبره أنه، قال يا رسول الله أرأيت إن لقيت رجلا من الكفار فقاتلني…٢٧٥وعبد بن حميد، قالا أخبرنا عبد الرزاق، قال أخبرنا معمر، ح وحدثنا إسحاق…٢٧٦ثم الجندعي أن عبيد، الله بن عدي بن الخيار أخبره أن المقداد بن عمرو بن…٢٧٧أقال لا إله إلا الله وقتلته٢٧٨يا أسامة أقتلته بعد ما قال لا إله إلا الله٢٧٩أن خالدا الأثبج ابن أخي، صفوان بن محرز حدث عن صفوان بن محرز، أنه حدث…٢٨٠من حمل علينا السلاح فليس منا٢٨١من سل علينا السيف فليس منا٢٨٢من حمل علينا السلاح فليس منا٢٨٣من حمل علينا السلاح فليس منا ومن غشنا فليس منا٢٨٤ما هذا يا صاحب الطعام٢٨٥ليس منا من ضرب الخدود أو شق الجيوب أو دعا بدعوى الجاهلية٢٨٦وعلي بن خشرم، قالا حدثنا عيسى بن يونس، جميعا عن الأعمش، بهذا الإسناد…٢٨٧أن القاسم بن مخيمرة، حدثه قال حدثني أبو بردة بن أبي موسى، قال وجع أبو…٢٨٨أنا بريء ممن حلق وسلق وخرق٢٨٩حدثنا عبد الله بن مطيع، حدثنا هشيم، عن حصين، عن عياض الأشعري، عن…٢٩٠لا يدخل الجنة نمام٢٩١لا يدخل الجنة قتات٢٩٢لا يدخل الجنة قتات٢٩٣ثلاثة لا يكلمهم الله يوم القيامة ولا ينظر إليهم ولا يزكيهم ولهم عذاب…٢٩٤ثلاثة لا يكلمهم الله يوم القيامة المنان الذي لا يعطي شيئا إلا منه…٢٩٥ثلاثة لا يكلمهم الله ولا ينظر إليهم ولا يزكيهم ولهم عذاب أليم٢٩٦ثلاثة لا يكلمهم الله يوم القيامة ولا يزكيهم - قال أبو معاوية ولا ينظر…٢٩٧ثلاث لا يكلمهم الله يوم القيامة ولا ينظر إليهم ولا يزكيهم ولهم عذاب…٢٩٨ورجل ساوم رجلا بسلعة٢٩٩ثلاثة لا يكلمهم الله ولا ينظر إليهم ولهم عذاب أليم رجل حلف على يمين…٣٠٠من قتل نفسه بحديدة فحديدته في يده يتوجأ بها في بطنه في نار جهنم خالدا…٣٠١وحدثني زهير بن حرب، حدثنا جرير، ح وحدثنا سعيد بن عمرو الأشعثي، حدثنا…٣٠٢من حلف على يمين بملة غير الإسلام كاذبا فهو كما قال ومن قتل نفسه بشىء…٣٠٣ليس على رجل نذر فيما لا يملك ولعن المؤمن كقتله ومن قتل نفسه بشىء في…٣٠٤وإسحاق بن منصور، وعبد الوارث بن عبد الصمد، كلهم عن عبد الصمد بن عبد…٣٠٥هذا من أهل النار٣٠٦أما إنه من أهل النار٣٠٧إن رجلا ممن كان قبلكم خرجت به قرحة فلما آذته انتزع سهما من كنانته…٣٠٨خرج برجل فيمن كان قبلكم خراج٣٠٩كلا إني رأيته في النار في بردة غلها أو عباءة٣١٠كلا والذي نفس محمد بيده إن الشملة لتلتهب عليه نارا أخذها من الغنائم…٣١١اللهم وليديه فاغفر٣١٢إن الله يبعث ريحا من اليمن ألين من الحرير فلا تدع أحدا في قلبه - قال…٣١٣بادروا بالأعمال فتنا كقطع الليل المظلم يصبح الرجل مؤمنا ويمسي كافرا…٣١٤يا أبا عمرو ما شأن ثابت أشتكى٣١٥قال كان ثابت بن قيس بن شماس خطيب الأنصار فلما نزلت هذه الآية . بنحو…٣١٦قال لما نزلت { لا ترفعوا أصواتكم فوق صوت النبي } ولم يذكر سعد بن معاذ…٣١٧وحدثنا هريم بن عبد الأعلى الأسدي، حدثنا المعتمر بن سليمان، قال سمعت…٣١٨أما من أحسن منكم في الإسلام فلا يؤاخذ بها ومن أساء أخذ بعمله في…٣١٩من أحسن في الإسلام لم يؤاخذ بما عمل في الجاهلية . ومن أساء في الإسلام…٣٢٠بهذا الإسناد مثله .٣٢١حدثنا محمد بن المثنى العنزي، وأبو معن الرقاشي وإسحاق بن منصور كلهم عن…٣٢٢وإبراهيم بن دينار، - واللفظ لإبراهيم - قالا حدثنا حجاج، - وهو ابن…٣٢٣أسلمت على ما أسلفت من خير٣٢٤أسلمت على ما أسلفت من خير٣٢٥أسلمت على ما أسلفت لك من الخير٣٢٦أن حكيم بن حزام، أعتق في الجاهلية مائة رقبة وحمل على مائة بعير ثم…٣٢٧ليس هو كما تظنون إنما هو كما قال لقمان لابنه { يا بنى لا تشرك بالله…٣٢٨وعلي بن خشرم، قالا أخبرنا عيسى، - وهو ابن يونس ح وحدثنا منجاب بن…٣٢٩أتريدون أن تقولوا كما قال أهل الكتابين من قبلكم سمعنا وعصينا بل قولوا…٣٣٠قولوا سمعنا وأطعنا وسلمنا٣٣١إن الله تجاوز لأمتي ما حدثت به أنفسها ما لم يتكلموا أو يعملوا به٣٣٢إن الله عز وجل تجاوز لأمتي عما حدثت به أنفسها ما لم تعمل أو تكلم به٣٣٣وهشام، ح وحدثني إسحاق بن منصور، أخبرنا الحسين بن علي، عن زائدة، عن…٣٣٤قال الله عز وجل إذا هم عبدي بسيئة فلا تكتبوها عليه فإن عملها فاكتبوها…٣٣٥قال الله عز وجل إذا هم عبدي بحسنة ولم يعملها كتبتها له حسنة فإن عملها…٣٣٦قال الله عز وجل إذا تحدث عبدي بأن يعمل حسنة فأنا أكتبها له حسنة ما لم…٣٣٧من هم بحسنة فلم يعملها كتبت له حسنة ومن هم بحسنة فعملها كتبت له عشرا…٣٣٨إن الله كتب الحسنات والسيئات ثم بين ذلك فمن هم بحسنة فلم يعملها كتبها…٣٣٩ومحاها الله ولا يهلك على الله إلا هالك٣٤٠حدثني زهير بن حرب، حدثنا جرير، عن سهيل، عن أبيه، عن أبي هريرة، قال…٣٤١وحدثنا محمد بن بشار، حدثنا ابن أبي عدي، عن شعبة، ح وحدثني محمد بن…٣٤٢حدثنا يوسف بن يعقوب الصفار، حدثني علي بن عثام، عن سعير بن الخمس، عن…٣٤٣لا يزال الناس يتساءلون حتى يقال هذا خلق الله الخلق فمن خلق الله فمن…٣٤٤يأتي الشيطان أحدكم فيقول من خلق السماء من خلق الأرض فيقول الله٣٤٥يأتي الشيطان أحدكم فيقول من خلق كذا وكذا حتى يقول له من خلق ربك فإذا…٣٤٦يأتي العبد الشيطان فيقول من خلق كذا وكذا٣٤٧لا يزال الناس يسألونكم عن العلم حتى يقولوا هذا الله خلقنا فمن خلق الله٣٤٨ويعقوب الدورقي، قالا حدثنا إسماعيل، وهو ابن علية عن أيوب، عن محمد،…٣٤٩لا يزالون يسألونك يا أبا هريرة حتى يقولوا هذا الله فمن خلق الله٣٥٠ليسألنكم الناس عن كل شىء حتى يقولوا الله خلق كل شىء فمن خلقه٣٥١قال الله عز وجل إن أمتك لا يزالون يقولون ما كذا ما كذا حتى يقولوا هذا…٣٥٢قال الله إن أمتك٣٥٣من اقتطع حق امرئ مسلم بيمينه فقد أوجب الله له النار وحرم عليه الجنة٣٥٤وإسحاق بن إبراهيم، وهارون بن عبد الله، جميعا عن أبي أسامة، عن الوليد…٣٥٥من حلف على يمين صبر يقتطع بها مال امرئ مسلم هو فيها فاجر لقي الله وهو…٣٥٦شاهداك أو يمينه٣٥٧من حلف على مال امرئ مسلم بغير حقه لقي الله وهو عليه غضبان٣٥٨قال جاء رجل من حضرموت ورجل من كندة إلى النبي صلى الله عليه وسلم فقال…٣٥٩وإسحاق بن إبراهيم، جميعا عن أبي الوليد، قال زهير حدثنا هشام بن عبد…٣٦٠قال جاء رجل إلى رسول الله صلى الله عليه وسلم فقال يا رسول الله أرأيت…٣٦١من قتل دون ماله فهو شهيد٣٦٢بهذا الإسناد مثله .٣٦٣ما من عبد يسترعيه الله رعية يموت يوم يموت وهو غاش لرعيته إلا حرم الله…٣٦٤لا يسترعي الله عبدا رعية يموت حين يموت وهو غاش لها إلا حرم الله عليه…٣٦٥وحدثني القاسم بن زكرياء، حدثنا حسين، - يعني الجعفي - عن زائدة، عن…٣٦٦ما من أمير يلي أمر المسلمين ثم لا يجهد لهم وينصح إلا لم يدخل معهم…٣٦٧أن الأمانة نزلت في جذر قلوب الرجال ثم نزل القرآن فعلموا من القرآن…٣٦٨بهذا الإسناد مثله .٣٦٩تعرض الفتن على القلوب كالحصير عودا عودا فأى قلب أشربها نكت فيه نكتة…٣٧٠وحدثني ابن أبي عمر، حدثنا مروان الفزاري، حدثنا أبو مالك الأشجعي، عن…٣٧١وعمرو بن علي، وعقبة بن مكرم العمي، قالوا حدثنا محمد بن أبي عدي، عن…٣٧٢بدأ الإسلام غريبا وسيعود كما بدأ غريبا فطوبى للغرباء٣٧٣إن الإسلام بدأ غريبا وسيعود غريبا كما بدأ وهو يأرز بين المسجدين كما…٣٧٤إن الإيمان ليأرز إلى المدينة كما تأرز الحية إلى جحرها٣٧٥لا تقوم الساعة حتى لا يقال في الأرض الله الله٣٧٦لا تقوم الساعة على أحد يقول الله الله٣٧٧أحصوا لي كم يلفظ الإسلام٣٧٨حدثنا ابن أبي عمر، حدثنا سفيان، عن الزهري، عن عامر بن سعد، عن أبيه،…٣٧٩حدثني زهير بن حرب، حدثنا يعقوب بن إبراهيم، حدثنا ابن أخي ابن شهاب، عن…٣٨٠وعبد بن حميد، قالا حدثنا يعقوب، - وهو ابن إبراهيم بن سعد - حدثنا أبي،…٣٨١أقتالا أى سعد إني لأعطي الرجل٣٨٢نحن أحق بالشك من إبراهيم صلى الله عليه وسلم إذ قال { رب أرني كيف تحيي…٣٨٣ولكن ليطمئن قلبي٣٨٥ما من الأنبياء من نبي إلا قد أعطي من الآيات ما مثله آمن عليه البشر…٣٨٦والذي نفس محمد بيده لا يسمع بي أحد من هذه الأمة يهودي ولا نصراني ثم…٣٨٧ثلاثة يؤتون أجرهم مرتين رجل من أهل الكتاب آمن بنبيه وأدرك النبي صلى…٣٨٩والذي نفسي بيده ليوشكن أن ينزل فيكم ابن مريم صلى الله عليه وسلم حكما…٣٩٠إماما مقسطا وحكما عدلا٣٩١والله لينزلن ابن مريم حكما عادلا فليكسرن الصليب وليقتلن الخنزير…٣٩٢كيف أنتم إذا نزل ابن مريم فيكم وإمامكم منكم٣٩٣كيف أنتم إذا نزل ابن مريم فيكم وأمكم٣٩٤كيف أنتم إذا نزل فيكم ابن مريم فأمكم منكم٣٩٥لا تزال طائفة من أمتي يقاتلون على الحق ظاهرين إلى يوم القيامة - قال -…٣٩٦لا تقوم الساعة حتى تطلع الشمس من مغربها فإذا طلعت من مغربها آمن الناس…٣٩٧وابن، نمير وأبو كريب قالوا حدثنا ابن فضيل، ح وحدثني زهير بن حرب،…
