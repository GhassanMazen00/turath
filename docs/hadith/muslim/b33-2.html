٢٥٧ حديثًا•صفحة ٢ من ٢
٤٨٥٩تضمن الله لمن خرج في سبيله لا يخرجه إلا جهادا في سبيلي وإيمانا بي…٤٨٦٠بهذا الإسناد .٤٨٦١تكفل الله لمن جاهد في سبيله لا يخرجه من بيته إلا جهاد في سبيله وتصديق…٤٨٦٢لا يكلم أحد في سبيل الله - والله أعلم بمن يكلم في سبيله - إلا جاء يوم…٤٨٦٣كل كلم يكلمه المسلم في سبيل الله ثم تكون يوم القيامة كهيئتها إذا طعنت…٤٨٦٤لولا أن أشق على المؤمنين ما قعدت خلاف سرية٤٨٦٥لولا أن أشق على أمتي لأحببت أن لا أتخلف خلف سرية٤٨٦٦تضمن الله لمن خرج في سبيله - إلى قوله - ما تخلفت خلاف سرية تغزو في…٤٨٦٧ما من نفس تموت لها عند الله خير يسرها أنها ترجع إلى الدنيا ولا أن لها…٤٨٦٨ما من أحد يدخل الجنة يحب أن يرجع إلى الدنيا وأن له ما على الأرض من…٤٨٦٩قال قيل للنبي صلى الله عليه وسلم ما يعدل الجهاد في سبيل الله عز وجل…٤٨٧٠بهذا الإسناد نحوه .٤٨٧١أنه سمع أبا سلام، قال حدثني النعمان بن بشير، قال كنت عند منبر رسول…٤٨٧٢أنه سمع أبا سلام، قال حدثني النعمان بن بشير، قال كنت عند منبر رسول…٤٨٧٣لغدوة في سبيل الله أو روحة خير من الدنيا وما فيها٤٨٧٤والغدوة يغدوها العبد في سبيل الله خير من الدنيا وما فيها٤٨٧٥غدوة أو روحة في سبيل الله خير من الدنيا وما فيها٤٨٧٦لولا أن رجالا من أمتي٤٨٧٧غدوة في سبيل الله أو روحة خير مما طلعت عليه الشمس وغربت٤٨٧٨وحيوة بن شريح، قال كل واحد منهما حدثني شرحبيل بن، شريك عن أبي عبد…٤٨٧٩يا أبا سعيد من رضي بالله ربا وبالإسلام دينا وبمحمد نبيا وجبت له الجنة٤٨٨٠أن الجهاد في سبيل الله والإيمان بالله أفضل الأعمال٤٨٨١ومحمد بن المثنى، قالا حدثنا يزيد بن هارون، أخبرنا يحيى، - يعني ابن…٤٨٨٢وحدثنا سعيد بن منصور، حدثنا سفيان، عن عمرو بن دينار، عن محمد بن قيس،…٤٨٨٣يغفر للشهيد كل ذنب إلا الدين٤٨٨٤القتل في سبيل الله يكفر كل شىء إلا الدين٤٨٨٥أرواحهم في جوف طير خضر لها قناديل معلقة بالعرش تسرح من الجنة حيث شاءت…٤٨٨٦رجل يجاهد في سبيل الله بماله ونفسه٤٨٨٧مؤمن يجاهد بنفسه وماله في سبيل الله٤٨٨٨بهذا الإسناد فقال " ورجل في شعب " . ولم يقل " ثم رجل " .٤٨٨٩من خير معاش الناس لهم رجل ممسك عنان فرسه في سبيل الله يطير على متنه…٤٨٩٠في شعبة من هذه الشعاب٤٨٩١في شعب من الشعاب٤٨٩٢يضحك الله إلى رجلين يقتل أحدهما الآخر كلاهما يدخل الجنة٤٨٩٣وزهير بن حرب، وأبو كريب قالوا حدثنا وكيع، عن سفيان، عن أبي الزناد،…٤٨٩٤يضحك الله لرجلين يقتل أحدهما الآخر كلاهما يدخل الجنة٤٨٩٥لا يجتمع كافر وقاتله في النار أبدا٤٨٩٦لا يجتمعان في النار اجتماعا يضر أحدهما الآخر٤٨٩٧لك بها يوم القيامة سبعمائة ناقة كلها مخطومة٤٨٩٨بهذا الإسناد .٤٨٩٩وحدثنا أبو بكر بن أبي شيبة، وأبو كريب وابن أبي عمر - واللفظ لأبي كريب…٤٩٠٠بهذا الإسناد .٤٩٠١ائت فلانا فإنه قد كان تجهز فمرض٤٩٠٢من جهز غازيا في سبيل الله فقد غزا ومن خلفه في أهله بخير فقد غزا٤٩٠٣من جهز غازيا فقد غزا ومن خلف غازيا في أهله فقد غزا٤٩٠٤لينبعث من كل رجلين أحدهما والأجر بينهما٤٩٠٧ليخرج من كل رجلين رجل٤٩٠٨حرمة نساء المجاهدين على القاعدين كحرمة أمهاتهم وما من رجل من القاعدين…٤٩٠٩قال قال - يعني النبي صلى الله عليه وسلم . بمعنى حديث الثوري .٤٩١٠فقال فخذ من حسناته ما شئت٤٩١١ومحمد بن بشار، - واللفظ لابن المثنى - قالا حدثنا محمد بن جعفر، حدثنا…٤٩١٢قال لما نزلت { لا يستوي القاعدون من المؤمنين } كلمه ابن أم مكتوم…٤٩١٣وسويد بن سعيد، - واللفظ لسعيد - أخبرنا سفيان، عن عمرو، سمع جابرا،…٤٩١٤عمل هذا يسيرا وأجر كثيرا٤٩١٥إن لنا طلبة فمن كان ظهره حاضرا فليركب معنا٤٩١٦إن أبواب الجنة تحت ظلال السيوف٤٩١٧إن إخوانكم قد قتلوا وإنهم قالوا اللهم بلغ عنا نبينا أنا قد لقيناك…٤٩١٨قال قال أنس عمي الذي سميت به لم يشهد مع رسول الله صلى الله عليه وسلم…٤٩١٩من قاتل لتكون كلمة الله أعلى فهو في سبيل الله٤٩٢٠من قاتل لتكون كلمة الله هي العليا فهو في سبيل الله٤٩٢١وحدثناه إسحاق بن إبراهيم، أخبرنا عيسى بن يونس، حدثنا الأعمش، عن شقيق،…٤٩٢٢من قاتل لتكون كلمة الله هي العليا فهو في سبيل الله٤٩٢٣إن أول الناس يقضى يوم القيامة عليه رجل استشهد فأتي به فعرفه نعمه…٤٩٢٤فقال له ناتل الشام واقتص الحديث بمثل حديث خالد بن الحارث .٤٩٢٥ما من غازية تغزو في سبيل الله فيصيبون الغنيمة إلا تعجلوا ثلثى أجرهم…٤٩٢٦ما من غازية أو سرية تغزو فتغنم وتسلم إلا كانوا قد تعجلوا ثلثى أجورهم…٤٩٢٧إنما الأعمال بالنية وإنما لامرئ ما نوى فمن كانت هجرته إلى الله ورسوله…٤٩٢٨حدثنا محمد بن رمح بن المهاجر، أخبرنا الليث، ح وحدثنا أبو الربيع…٤٩٢٩من طلب الشهادة صادقا أعطيها ولو لم تصبه٤٩٣٠من سأل الله الشهادة بصدق بلغه الله منازل الشهداء وإن مات على فراشه٤٩٣١من مات ولم يغز ولم يحدث به نفسه مات على شعبة من نفاق٤٩٣٢إن بالمدينة لرجالا ما سرتم مسيرا ولا قطعتم واديا إلا كانوا معكم حبسهم…٤٩٣٣إلا شركوكم في الأجر٤٩٣٤ناس من أمتي عرضوا على غزاة في سبيل الله يركبون ثبج هذا البحر ملوكا…٤٩٣٥أريت قوما من أمتي يركبون ظهر البحر كالملوك على الأسرة٤٩٣٦ناس من أمتي عرضوا على يركبون ظهر هذا البحر الأخضر٤٩٣٧وقتيبة، وابن، حجر قالوا حدثنا إسماعيل، - وهو ابن جعفر - عن عبد الله…٤٩٣٨رباط يوم وليلة خير من صيام شهر وقيامه وإن مات جرى عليه عمله الذي كان…٤٩٣٩حدثني أبو الطاهر، أخبرنا ابن وهب، عن عبد الرحمن بن شريح، عن عبد…٤٩٤٠بينما رجل يمشي بطريق وجد غصن شوك على الطريق فأخره فشكر الله له فغفر له٤٩٤١ما تعدون الشهيد فيكم٤٩٤٢ومن غرق فهو شهيد٤٩٤٣وزاد، فيه " والغرق شهيد " .٤٩٤٤الطاعون شهادة لكل مسلم٤٩٤٥في هذا الإسناد بمثله .٤٩٤٦وأعدوا لهم ما استطعتم من قوة ألا إن القوة الرمى ألا إن القوة الرمى…٤٩٤٧ستفتح عليكم أرضون ويكفيكم الله فلا يعجز أحدكم أن يلهو بأسهمه٤٩٤٨وحدثناه داود بن رشيد، حدثنا الوليد، عن بكر بن مضر، عن عمرو بن الحارث،…٤٩٤٩من علم الرمى ثم تركه فليس منا أو قد عصى٤٩٥٠لا تزال طائفة من أمتي ظاهرين على الحق لا يضرهم من خذلهم حتى يأتي أمر…٤٩٥١لن يزال قوم من أمتي ظاهرين على الناس حتى يأتيهم أمر الله وهم ظاهرون٤٩٥٢وحدثنيه محمد بن رافع، حدثنا أبو أسامة، حدثني إسماعيل، عن قيس، قال…٤٩٥٣لن يبرح هذا الدين قائما يقاتل عليه عصابة من المسلمين حتى تقوم الساعة٤٩٥٤لا تزال طائفة من أمتي يقاتلون على الحق ظاهرين إلى يوم القيامة٤٩٥٥لا تزال طائفة من أمتي قائمة بأمر الله لا يضرهم من خذلهم أو خالفهم حتى…٤٩٥٦من يرد الله به خيرا يفقهه في الدين ولا تزال عصابة من المسلمين يقاتلون…٤٩٥٧لا تزال عصابة من أمتي يقاتلون على أمر الله قاهرين لعدوهم لا يضرهم من…٤٩٥٨لا يزال أهل الغرب ظاهرين على الحق حتى تقوم الساعة٤٩٥٩إذا سافرتم في الخصب فأعطوا الإبل حظها من الأرض وإذا سافرتم في السنة…٤٩٦٠إذا سافرتم في الخصب فأعطوا الإبل حظها من الأرض وإذا سافرتم في السنة…٤٩٦١السفر قطعة من العذاب يمنع أحدكم نومه وطعامه وشرابه فإذا قضى أحدكم…٤٩٦٢أن رسول الله صلى الله عليه وسلم كان لا يطرق أهله ليلا وكان يأتيهم…٤٩٦٣وحدثنيه زهير بن حرب، حدثنا عبد الصمد بن عبد الوارث، حدثنا همام، حدثنا…٤٩٦٤أمهلوا حتى ندخل ليلا - أى عشاء - كى تمتشط الشعثة وتستحد المغيبة٤٩٦٥إذا قدم أحدكم ليلا فلا يأتين أهله طروقا حتى تستحد المغيبة وتمتشط…٤٩٦٦بهذا الإسناد مثله .٤٩٦٧قال نهى رسول الله صلى الله عليه وسلم إذا أطال الرجل الغيبة أن يأتي…
