٥١١٤حديث ٥١١٤٥١١٥حديث ٥١١٥٥١٢٧حدثنا يحيى بن يحيى التميمي، أخبرنا حجاج بن محمد، عن ابن جريج، حدثني…٥١٢٨بهذا الإسناد مثله .٥١٢٩بن علي أن حسين بن علي، أخبره أن عليا قال كانت لي شارف من نصيبي من…٥١٣٠بهذا الإسناد مثله .٥١٣١قال كنت ساقي القوم يوم حرمت الخمر في بيت أبي طلحة وما شرابهم إلا…٥١٣٢وحدثنا يحيى بن أيوب، حدثنا ابن علية، أخبرنا عبد العزيز بن صهيب، قال…٥١٣٣وحدثنا يحيى بن أيوب، حدثنا ابن علية، قال وأخبرنا سليمان التيمي، حدثنا…٥١٣٤حدثنا محمد بن عبد الأعلى، حدثنا المعتمر، عن أبيه، قال قال أنس كنت…٥١٣٥وحدثنا يحيى بن أيوب، حدثنا ابن علية، قال وأخبرنا سعيد بن أبي عروبة،…٥١٣٦ومحمد بن المثنى، وابن، بشار قالوا أخبرنا معاذ، بن هشام حدثني أبي، عن…٥١٣٧أن قتادة بن دعامة، حدثه أنه، سمع أنس بن مالك، يقول إن رسول الله صلى…٥١٣٨وحدثني أبو الطاهر، أخبرنا ابن وهب، أخبرني مالك بن أنس، عن إسحاق بن،…٥١٣٩سمع أنس بن مالك، يقول لقد أنزل الله الآية التي حرم الله فيها الخمر…٥١٤٠حدثنا يحيى بن يحيى، أخبرنا عبد الرحمن بن مهدي، ح وحدثنا زهير بن حرب،…٥١٤١إنه ليس بدواء ولكنه داء٥١٤٢الخمر من هاتين الشجرتين النخلة والعنبة٥١٤٣الخمر من هاتين الشجرتين النخلة والعنبة٥١٤٤الخمر من هاتين الشجرتين الكرمة والنخلة٥١٤٥أن النبي صلى الله عليه وسلم نهى أن يخلط الزبيب والتمر والبسر والتمر .٥١٤٦حدثنا قتيبة بن سعيد، حدثنا ليث، عن عطاء بن أبي رباح، عن جابر بن عبد،…٥١٤٧لا تجمعوا بين الرطب والبسر وبين الزبيب والتمر نبيذا٥١٤٨وحدثنا قتيبة بن سعيد، حدثنا ليث، ح وحدثنا محمد بن رمح، أخبرنا الليث،…٥١٤٩حدثنا يحيى بن يحيى، أخبرنا يزيد بن زريع، عن التيمي، عن أبي نضرة، عن…٥١٥٠حدثنا يحيى بن أيوب، حدثنا ابن علية، حدثنا سعيد بن يزيد أبو مسلمة، عن…٥١٥١مسلمة بهذا الإسناد مثله .٥١٥٢من شرب النبيذ منكم فليشربه زبيبا فردا أو تمرا فردا أو بسرا فردا٥١٥٣وحدثنيه أبو بكر بن إسحاق، حدثنا روح بن عبادة، حدثنا إسماعيل بن مسلم،…٥١٥٤لا تنتبذوا الزهو والرطب جميعا ولا تنتبذوا الزبيب والتمر جميعا…٥١٥٥بهذا الإسناد مثله .٥١٥٦لا تنتبذوا الزهو والرطب جميعا ولا تنتبذوا الرطب والزبيب جميعا ولكن…٥١٥٧الرطب والزهو والتمر والزبيب٥١٥٨انتبذوا كل واحد على حدته٥١٥٩وحدثني أبو سلمة بن عبد الرحمن، عن أبي قتادة، عن النبي صلى الله عليه…٥١٦٠ينبذ كل واحد منهما على حدته٥١٦١قال قال رسول الله صلى الله عليه وسلم بمثله .٥١٦٢وحدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن الشيباني، عن حبيب،…٥١٦٣بهذا الإسناد في التمر والزبيب ولم يذكر البسر والتمر .٥١٦٤أنه كان يقول قد نهي أن ينبذ البسر والرطب جميعا والتمر والزبيب جميعا .٥١٦٥أنه قال قد نهي أن ينبذ البسر والرطب جميعا والتمر والزبيب جميعا .٥١٦٦حدثنا قتيبة بن سعيد، حدثنا ليث، عن ابن شهاب، عن أنس بن مالك، أنه…٥١٦٧وحدثني عمرو الناقد، حدثنا سفيان بن عيينة، عن الزهري، عن أنس بن مالك،…٥١٦٨لا تنتبذوا في الدباء ولا في المزفت٥١٦٩حدثني محمد بن حاتم، حدثنا بهز، حدثنا وهيب، عن سهيل، عن أبيه، عن أبي،…٥١٧٠أنهاكم عن الدباء والحنتم والنقير والمقير - والحنتم المزادة المجبوبة -…٥١٧١حدثنا سعيد بن عمرو الأشعثي، أخبرنا عبثر، ح وحدثني زهير بن حرب، حدثنا…٥١٧٢وإسحاق بن إبراهيم، كلاهما عن جرير، قال زهير حدثنا جرير، عن منصور، عن…٥١٧٣وحدثنا سعيد بن عمرو الأشعثي، أخبرنا عبثر، عن الأعمش، عن إبراهيم، عن…٥١٧٤وشعبة، قالا حدثنا منصور، وسليمان، وحماد، عن إبراهيم، عن الأسود، عن…٥١٧٥حدثنا شيبان بن فروخ، حدثنا القاسم، - يعني ابن الفضل - حدثنا ثمامة بن،…٥١٧٦وحدثنا يعقوب بن إبراهيم، حدثنا ابن علية، حدثنا إسحاق بن سويد، عن…٥١٧٧سويد بهذا الإسناد إلا أنه جعل مكان المزفت المقير .٥١٧٨أنهاكم عن الدباء والحنتم والنقير والمقير٥١٧٩حدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن الشيباني، عن حبيب،…٥١٨٠حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن فضيل، عن حبيب بن أبي عمرة،…٥١٨١حدثنا محمد بن المثنى، حدثنا عبد الرحمن بن مهدي، عن شعبة، عن يحيى…٥١٨٢حدثنا يحيى بن يحيى، أخبرنا يزيد بن زريع، عن التيمي، ح وحدثنا يحيى بن،…٥١٨٣حدثنا يحيى بن أيوب، حدثنا ابن علية، أخبرنا سعيد بن أبي عروبة، عن…٥١٨٤بهذا الإسناد أن نبي الله صلى الله عليه وسلم نهى أن ينتبذ . فذكر مثله .٥١٨٥وحدثنا نصر بن علي الجهضمي، حدثني أبي، حدثنا المثنى، - يعني ابن سعيد -…٥١٨٦وسريج بن يونس، - واللفظ لأبي بكر - قالا حدثنا مروان بن معاوية، عن…٥١٨٧فقال حرم رسول الله صلى الله عليه وسلم نبيذ الجر . فأتيت ابن عباس فقلت…٥١٨٨أن رسول الله صلى الله عليه وسلم خطب الناس في بعض مغازيه قال ابن عمر…٥١٨٩وابن، رمح عن الليث بن سعد، ح وحدثنا أبو الربيع، وأبو كامل قالا حدثنا…٥١٩٠وحدثنا يحيى بن يحيى، أخبرنا حماد بن زيد، عن ثابت، قال قلت لابن عمر…٥١٩١حدثنا يحيى بن أيوب، حدثنا ابن علية، حدثنا سليمان التيمي، عن طاوس، قال…٥١٩٢أن رجلا، جاءه فقال أنهى النبي صلى الله عليه وسلم أن ينبذ في الجر…٥١٩٣وحدثني محمد بن حاتم، حدثنا بهز، حدثنا وهيب، حدثنا عبد الله بن طاوس،…٥١٩٤أنه سمع طاوسا، يقول كنت جالسا عند ابن عمر فجاءه رجل فقال أنهى رسول…٥١٩٥وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن محارب بن دثار، قال…٥١٩٦وحدثنا سعيد بن عمرو الأشعثي، أخبرنا عبثر، عن الشيباني، عن محارب بن،…٥١٩٧انتبذوا في الأسقية٥١٩٨حدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، عن جبلة، قال…٥١٩٩حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا شعبة، عن عمرو بن مرة، حدثني…٥٢٠٠وابن، بشار قالا حدثنا أبو داود، حدثنا شعبة، في هذا الإسناد .٥٢٠١وحدثنا أبو بكر بن أبي شيبة، حدثنا يزيد بن هارون، أخبرنا عبد الخالق…٥٢٠٢وابن، عمر أن رسول الله صلى الله عليه وسلم نهى عن النقير والمزفت…٥٢٠٣أنه سمع ابن عمر، يقول سمعت رسول الله صلى الله عليه وسلم ينهى عن الجر…٥٢٠٥أن النبي صلى الله عليه وسلم كان ينبذ له في تور من حجارة .٥٢٠٦وحدثنا أحمد بن يونس، حدثنا زهير، حدثنا أبو الزبير، ح وحدثنا يحيى بن،…٥٢٠٩كنت نهيتكم عن الأشربة في ظروف الأدم فاشربوا في كل وعاء غير أن لا…٥٢١٠وابن أبي عمر، - واللفظ لابن أبي عمر - قالا حدثنا سفيان، عن سليمان…٥٢١١كل شراب أسكر فهو حرام٥٢١٢كل شراب أسكر فهو حرام٥٢١٣كل شراب مسكر حرام٥٢١٥بشرا ويسرا وعلما ولا تنفرا٥٢١٦ادعوا الناس وبشرا ولا تنفرا ويسرا ولا تعسرا٥٢١٧أن رجلا، قدم من جيشان - وجيشان من اليمن - فسأل النبي صلى الله عليه…٥٢١٨كل مسكر خمر وكل مسكر حرام ومن شرب الخمر في الدنيا فمات وهو يدمنها لم…٥٢١٩كل مسكر خمر وكل مسكر حرام٥٢٢٠بهذا الإسناد مثله .٥٢٢١كل مسكر خمر وكل خمر حرام٥٢٢٢من شرب الخمر في الدنيا حرمها في الآخرة٥٢٢٣من شرب الخمر في الدنيا فلم يتب منها حرمها في الآخرة فلم يسقها٥٢٢٤من شرب الخمر في الدنيا لم يشربها في الآخرة إلا أن يتوب٥٢٢٥وحدثنا ابن أبي عمر، حدثنا هشام، - يعني ابن سليمان المخزومي - عن ابن،…٥٢٢٦يقول كان رسول الله صلى الله عليه وسلم ينتبذ له أول الليل فيشربه إذا…٥٢٢٧قال ذكروا النبيذ عند ابن عباس فقال كان رسول الله صلى الله عليه وسلم…٥٢٢٨قال كان رسول الله صلى الله عليه وسلم ينقع له الزبيب فيشربه اليوم…٥٢٢٩قال كان رسول الله صلى الله عليه وسلم ينبذ له الزبيب في السقاء فيشربه…٥٢٣٠وشرائها، والتجارة فيها فقال أمسلمون أنتم قالوا نعم . قال فإنه لا يصلح…٥٢٣١فدعت عائشة جارية حبشية فقالت سل هذه فإنها كانت تنبذ لرسول الله صلى…٥٢٣٢حدثنا محمد بن المثنى العنزي، حدثنا عبد الوهاب الثقفي، عن يونس، عن…٥٢٣٣قال دعا أبو أسيد الساعدي رسول الله صلى الله عليه وسلم في عرسه فكانت…٥٢٣٤يقول أتى أبو أسيد الساعدي رسول الله صلى الله عليه وسلم فدعا رسول الله…٥٢٣٥بهذا الحديث وقال في تور من حجارة فلما فرغ رسول الله صلى الله عليه…٥٢٣٦حدثني محمد بن سهل التميمي، وأبو بكر بن إسحاق - قال أبو بكر أخبرنا…٥٢٣٧وزهير بن حرب، قالا حدثنا عفان، حدثنا حماد، بن سلمة عن ثابت، عن أنس،…٥٢٣٨حدثنا عبيد الله بن معاذ العنبري، حدثنا أبي، حدثنا شعبة، عن أبي إسحاق،…٥٢٣٩وابن، بشار - واللفظ لابن المثنى - قالا حدثنا محمد، بن جعفر حدثنا…٥٢٤٠وزهير بن حرب، - واللفظ لابن عباد - قالا حدثنا أبو صفوان أخبرنا يونس،…٥٢٤١أنه سمع أبا هريرة، يقول أتي رسول الله صلى الله عليه وسلم . بمثله ولم…٥٢٤٢ألا خمرته ولو تعرض عليه عودا٥٢٤٣وزكرياء، بن إسحاق قالا أخبرنا أبو الزبير، أنه سمع جابر بن عبد الله،…٥٢٤٤حدثنا أبو بكر بن أبي شيبة، وأبو كريب - واللفظ لأبي كريب - قالا حدثنا…٥٢٤٥ألا خمرته ولو تعرض عليه عودا٥٢٤٦غطوا الإناء وأوكوا السقاء وأغلقوا الباب وأطفئوا السراج فإن الشيطان لا…٥٢٤٧واكفئوا الإناء أو خمروا الإناء٥٢٤٨قال قال رسول الله صلى الله عليه وسلم " أغلقوا الباب " . فذكر بمثل…٥٢٤٩والفويسقة تضرم البيت على أهله٥٢٥٠إذا كان جنح الليل - أو أمسيتم - فكفوا صبيانكم فإن الشيطان ينتشر حينئذ…٥٢٥١اذكروا اسم الله عز وجل٥٢٥٢وعمرو بن دينار، كرواية روح .٥٢٥٣لا ترسلوا فواشيكم وصبيانكم إذا غابت الشمس حتى تذهب فحمة العشاء فإن…٥٢٥٤وحدثني محمد بن المثنى، حدثنا عبد الرحمن، حدثنا سفيان، عن أبي الزبير،…٥٢٥٥غطوا الإناء وأوكوا السقاء فإن في السنة ليلة ينزل فيها وباء لا يمر…٥٢٥٦فإن في السنة يوما ينزل فيه وباء٥٢٥٧لا تتركوا النار في بيوتكم حين تنامون٥٢٥٨إن هذه النار إنما هي عدو لكم فإذا نمتم فأطفئوها عنكم٥٢٥٩إن الشيطان يستحل الطعام أن لا يذكر اسم الله عليه وإنه جاء بهذه…٥٢٦٠وحدثناه إسحاق بن إبراهيم الحنظلي، أخبرنا عيسى بن يونس، أخبرنا الأعمش،…٥٢٦١بهذا الإسناد وقدم مجيء الجارية قبل مجيء الأعرابي .٥٢٦٢إذا دخل الرجل بيته فذكر الله عند دخوله وعند طعامه قال الشيطان لا مبيت…٥٢٦٣وإن لم يذكر اسم الله عند طعامه وإن لم يذكر اسم الله عند دخوله٥٢٦٤لا تأكلوا بالشمال فإن الشيطان يأكل بالشمال٥٢٦٥إذا أكل أحدكم فليأكل بيمينه وإذا شرب فليشرب بيمينه فإن الشيطان يأكل…٥٢٦٦بإسناد سفيان .٥٢٦٧لا يأكلن أحد منكم بشماله ولا يشربن بها فإن الشيطان يأكل بشماله ويشرب…٥٢٦٨أن أباه، حدثه أن رجلا أكل عند رسول الله صلى الله عليه وسلم بشماله…٥٢٦٩يا غلام سم الله وكل بيمينك وكل مما يليك٥٢٧٠أنه قال أكلت يوما مع رسول الله صلى الله عليه وسلم فجعلت آخذ من لحم…٥٢٧١وحدثنا عمرو الناقد، حدثنا سفيان بن عيينة، عن الزهري، عن عبيد الله، عن…٥٢٧٢وحدثني حرملة بن يحيى، أخبرنا ابن وهب، أخبرني يونس، عن ابن شهاب، عن…٥٢٧٣بهذا الإسناد . مثله غير أنه قال واختناثها أن يقلب رأسها ثم يشرب منه .٥٢٧٤حدثنا هداب بن خالد، حدثنا همام، حدثنا قتادة، عن أنس، أن النبي صلى…٥٢٧٥حدثنا محمد بن المثنى، حدثنا عبد الأعلى، حدثنا سعيد، عن قتادة، عن أنس،…٥٢٧٦وحدثناه قتيبة بن سعيد، وأبو بكر بن أبي شيبة قالا حدثنا وكيع، عن هشام،…٥٢٧٧حدثنا هداب بن خالد، حدثنا همام، حدثنا قتادة، عن أبي عيسى الأسواري، عن…٥٢٧٨ومحمد بن المثنى، وابن، بشار - واللفظ لزهير وابن المثنى - قالوا حدثنا…
