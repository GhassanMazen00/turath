١٤٣ حديثًا
٥٣٤الطهور شطر الإيمان والحمد لله تملأ الميزان . وسبحان الله والحمد لله…٥٣٥لا تقبل صلاة بغير طهور ولا صدقة من غلول٥٣٦وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، ح وحدثنا أبو بكر بن…٥٣٧لا تقبل صلاة أحدكم إذا أحدث حتى يتوضأ٥٣٨من توضأ نحو وضوئي هذا ثم قام فركع ركعتين لا يحدث فيهما نفسه غفر له ما…٥٣٩من توضأ نحو وضوئي هذا ثم صلى ركعتين لا يحدث فيهما نفسه غفر له ما تقدم…٥٤٠لا يتوضأ رجل مسلم فيحسن الوضوء فيصلي صلاة إلا غفر الله له ما بينه…٥٤١فيحسن وضوءه ثم يصلي المكتوبة٥٤٢لا يتوضأ رجل فيحسن وضوءه ثم يصلي الصلاة إلا غفر له ما بينه وبين…٥٤٣ما من امرئ مسلم تحضره صلاة مكتوبة فيحسن وضوءها وخشوعها وركوعها إلا…٥٤٤من توضأ هكذا غفر له ما تقدم من ذنبه وكانت صلاته ومشيه إلى المسجد نافلة٥٤٥أن عثمان، توضأ بالمقاعد فقال ألا أريكم وضوء رسول الله صلى الله عليه…٥٤٦ما أدري أحدثكم بشىء أو أسكت٥٤٧من أتم الوضوء كما أمره الله تعالى فالصلوات المكتوبات كفارات لما بينهن٥٤٨من توضأ هكذا ثم خرج إلى المسجد لا ينهزه إلا الصلاة غفر له ما خلا من…٥٤٩من توضأ للصلاة فأسبغ الوضوء ثم مشى إلى الصلاة المكتوبة فصلاها مع…٥٥٠الصلاة الخمس والجمعة إلى الجمعة كفارة لما بينهن ما لم تغش الكبائر٥٥١الصلوات الخمس والجمعة إلى الجمعة كفارات لما بينهن٥٥٢الصلوات الخمس والجمعة إلى الجمعة ورمضان إلى رمضان مكفرات ما بينهن إذا…٥٥٣ما من مسلم يتوضأ فيحسن وضوءه ثم يقوم فيصلي ركعتين مقبل عليهما بقلبه…٥٥٤من توضأ فقال أشهد أن لا إله إلا الله وحده لا شريك له وأشهد أن محمدا…٥٥٥حدثني محمد بن الصباح، حدثنا خالد بن عبد الله، عن عمرو بن يحيى بن…٥٥٦بهذا الإسناد نحوه ولم يذكر الكعبين .٥٥٧بهذا الإسناد وقال مضمض واستنثر ثلاثا . ولم يقل من كف واحدة . وزاد بعد…٥٥٨بمثل إسنادهم واقتص الحديث وقال فيه فمضمض واستنشق واستنثر من ثلاث…٥٥٩وأبو الطاهر، قالوا حدثنا ابن وهب، أخبرني عمرو بن الحارث، أن حبان بن…٥٦٠إذا استجمر أحدكم فليستجمر وترا وإذا توضأ أحدكم فليجعل في أنفه ماء ثم…٥٦١إذا توضأ أحدكم فليستنشق بمنخريه من الماء ثم لينتثر٥٦٢من توضأ فليستنثر ومن استجمر فليوتر٥٦٣أنه سمع أبا هريرة، وأبا، سعيد الخدري يقولان قال رسول الله صلى الله…٥٦٤إذا استيقظ أحدكم من منامه فليستنثر ثلاث مرات فإن الشيطان يبيت على…٥٦٥إذا استجمر أحدكم فليوتر٥٦٦ويل للأعقاب من النار٥٦٧أن أبا عبد الله، مولى شداد بن الهاد حدثه أنه، دخل على عائشة فذكر عنها…٥٦٨وحدثني محمد بن حاتم، وأبو معن الرقاشي قالا حدثنا عمر بن يونس، حدثنا…٥٦٩حدثني سلمة بن شبيب، حدثنا الحسن بن أعين، حدثنا فليح، حدثني نعيم بن…٥٧٠ويل للأعقاب من النار أسبغوا الوضوء٥٧١وابن، بشار قالا حدثنا محمد بن جعفر، قال حدثنا شعبة، كلاهما عن منصور،…٥٧٢ويل للأعقاب من النار٥٧٣ويل للأعقاب من النار٥٧٤ويل للعراقيب من النار٥٧٥ويل للأعقاب من النار٥٧٦ارجع فأحسن وضوءك٥٧٧إذا توضأ العبد المسلم - أو المؤمن - فغسل وجهه خرج من وجهه كل خطيئة…٥٧٨من توضأ فأحسن الوضوء خرجت خطاياه من جسده حتى تخرج من تحت أظفاره٥٧٩أنتم الغر المحجلون يوم القيامة من إسباغ الوضوء فمن استطاع منكم فليطل…٥٨٠إن أمتي يأتون يوم القيامة غرا محجلين من أثر الوضوء فمن استطاع منكم أن…٥٨١إن حوضي أبعد من أيلة من عدن لهو أشد بياضا من الثلج وأحلى من العسل…٥٨٢ترد على أمتي الحوض وأنا أذود الناس عنه كما يذود الرجل إبل الرجل عن…٥٨٣إن حوضي لأبعد من أيلة من عدن والذي نفسي بيده إني لأذود عنه الرجال كما…٥٨٤السلام عليكم دار قوم مؤمنين وإنا إن شاء الله بكم لاحقون وددت أنا قد…٥٨٥السلام عليكم دار قوم مؤمنين وإنا إن شاء الله بكم لاحقون٥٨٦تبلغ الحلية من المؤمن حيث يبلغ الوضوء٥٨٧ألا أدلكم على ما يمحو الله به الخطايا ويرفع به الدرجات٥٨٨فذلكم الرباط فذلكم الرباط٥٨٩لولا أن أشق على المؤمنين - وفي حديث زهير على أمتي - لأمرتهم بالسواك…٥٩٠قال سألت عائشة قلت بأى شىء كان يبدأ النبي صلى الله عليه وسلم إذا دخل…٥٩١أن النبي صلى الله عليه وسلم كان إذا دخل بيته بدأ بالسواك .٥٩٢قال دخلت على النبي صلى الله عليه وسلم وطرف السواك على لسانه .٥٩٣قال كان رسول الله صلى الله عليه وسلم إذا قام ليتهجد يشوص فاه بالسواك .٥٩٤قال كان رسول الله صلى الله عليه وسلم إذا قام من الليل . بمثله ولم…٥٩٥وابن، بشار قالا حدثنا عبد الرحمن، حدثنا سفيان، عن منصور، وحصين،…٥٩٦أن ابن عباس، حدثه أنه، بات عند النبي صلى الله عليه وسلم ذات ليلة فقام…٥٩٧الفطرة خمس - أو خمس من الفطرة - الختان والاستحداد وتقليم الأظفار ونتف…٥٩٨الفطرة خمس الاختتان والاستحداد وقص الشارب وتقليم الأظفار ونتف الإبط٥٩٩وقتيبة بن سعيد، كلاهما عن جعفر، - قال يحيى أخبرنا جعفر بن سليمان، -…٦٠٠أحفوا الشوارب وأعفوا اللحى٦٠١وحدثناه قتيبة بن سعيد، عن مالك بن أنس، عن أبي بكر بن نافع، عن أبيه،…٦٠٢خالفوا المشركين أحفوا الشوارب وأوفوا اللحى٦٠٣جزوا الشوارب وأرخوا اللحى خالفوا المجوس٦٠٤عشر من الفطرة قص الشارب وإعفاء اللحية والسواك واستنشاق الماء وقص…٦٠٥في هذا الإسناد مثله غير أنه قال قال أبوه ونسيت العاشرة .٦٠٦ووكيع، عن الأعمش، ح وحدثنا يحيى بن يحيى، - واللفظ له - أخبرنا أبو…٦٠٧لا يستنجي أحدكم بدون ثلاثة أحجار٦٠٨أنه سمع جابرا، يقول نهى رسول الله صلى الله عليه وسلم أن يتمسح بعظم أو…٦٠٩إذا أتيتم الغائط فلا تستقبلوا القبلة ولا تستدبروها ببول ولا غائط ولكن…٦١٠إذا جلس أحدكم على حاجته فلا يستقبل القبلة ولا يستدبرها٦١١واسع بن حبان، قال كنت أصلي في المسجد وعبد الله بن عمر مسند ظهره إلى…٦١٢واسع بن حبان، عن ابن عمر، قال رقيت على بيت أختي حفصة فرأيت رسول الله…٦١٣لا يمسكن أحدكم ذكره بيمينه وهو يبول ولا يتمسح من الخلاء بيمينه ولا…٦١٤إذا دخل أحدكم الخلاء فلا يمس ذكره بيمينه٦١٥أن النبي صلى الله عليه وسلم نهى أن يتنفس في الإناء وأن يمس ذكره…٦١٦قالت إن كان رسول الله صلى الله عليه وسلم ليحب التيمن في طهوره إذا…٦١٧قالت كان رسول الله صلى الله عليه وسلم يحب التيمن في شأنه كله في نعليه…٦١٨وقتيبة، وابن، حجر جميعا عن إسماعيل بن جعفر، - قال ابن أيوب حدثنا…٦١٩حدثنا يحيى بن يحيى، أخبرنا خالد بن عبد الله، عن خالد، عن عطاء بن أبي…٦٢٠وغندر، عن شعبة، ح وحدثنا محمد بن المثنى، - واللفظ له - حدثنا محمد بن…٦٢١قال كان رسول الله صلى الله عليه وسلم يتبرز لحاجته فآتيه بالماء فيتغسل…٦٢٢وإسحاق بن إبراهيم، وأبو كريب جميعا عن أبي معاوية، ح وحدثنا أبو بكر بن…٦٢٣وعلي بن خشرم، قالا أخبرنا عيسى بن يونس، ح وحدثناه محمد بن أبي عمر،…٦٢٤قال كنت مع النبي صلى الله عليه وسلم فانتهى إلى سباطة قوم فبال قائما…٦٢٥حدثنا يحيى بن يحيى، أخبرنا جرير، عن منصور، عن أبي وائل، قال كان أبو…٦٢٦حدثنا قتيبة بن سعيد، حدثنا ليث، ح وحدثنا محمد بن رمح بن المهاجر،…٦٢٧بهذا الإسناد وقال فغسل وجهه ويديه ومسح برأسه ثم مسح على الخفين .٦٢٨رسول الله صلى الله عليه وسلم ذات ليلة إذ نزل فقضى حاجته ثم جاء فصببت…٦٢٩يا مغيرة خذ الإداوة٦٣٠وعلي بن خشرم، جميعا عن عيسى بن يونس، - قال إسحاق أخبرنا عيسى، - حدثنا…٦٣١حدثنا محمد بن عبد الله بن نمير، حدثنا أبي، حدثنا زكرياء، عن عامر، قال…٦٣٢إني أدخلتهما طاهرتين٦٣٣وحدثني محمد بن عبد الله بن بزيع، حدثنا يزيد، - يعني ابن زريع - حدثنا…٦٣٤ومحمد بن عبد الأعلى، قالا حدثنا المعتمر، عن أبيه، قال حدثني بكر بن…٦٣٥وحدثنا محمد بن عبد الأعلى، حدثنا المعتمر، عن أبيه، عن بكر، عن الحسن،…٦٣٦ومحمد بن حاتم، جميعا عن يحيى القطان، قال ابن حاتم حدثنا يحيى بن سعيد،…٦٣٩على الخفين فقالت عليك بابن أبي طالب فسله فإنه كان يسافر مع رسول الله…٦٤٠بهذا الإسناد مثله .٦٤١وحدثني زهير بن حرب، حدثنا أبو معاوية، عن الأعمش، عن الحكم، عن القاسم…٦٤٢عمدا صنعته يا عمر٦٤٣إذا استيقظ أحدكم من نومه فلا يغمس يده في الإناء حتى يغسلها ثلاثا فإنه…٦٤٤وأبي، صالح عن أبي هريرة، في حديث أبي معاوية قال قال رسول الله صلى…٦٤٥وعمرو الناقد، وزهير بن حرب، قالوا حدثنا سفيان بن عيينة، عن الزهري، عن…٦٤٦إذا استيقظ أحدكم فليفرغ على يده ثلاث مرات قبل أن يدخل يده في إنائه…٦٤٧وابن، رافع قالا حدثنا عبد الرزاق، قالا جميعا أخبرنا ابن جريج، أخبرني…٦٤٨إذا ولغ الكلب في إناء أحدكم فليرقه ثم ليغسله سبع مرار٦٤٩بهذا الإسناد مثله ولم يقل فليرقه .٦٥٠إذا شرب الكلب في إناء أحدكم فليغسله سبع مرات٦٥١طهور إناء أحدكم إذا ولغ فيه الكلب أن يغسله سبع مرات أولاهن بالتراب٦٥٢طهور إناء أحدكم إذا ولغ الكلب فيه أن يغسله سبع مرات٦٥٣ما بالهم وبال الكلاب٦٥٤في هذا الإسناد بمثله غير أن في رواية يحيى بن سعيد من الزيادة ورخص في…٦٥٥ومحمد بن رمح، قالا أخبرنا الليث، ح وحدثنا قتيبة، حدثنا الليث، عن أبي…٦٥٦لا يبولن أحدكم في الماء الدائم ثم يغتسل منه٦٥٧لا تبل في الماء الدائم الذي لا يجري ثم تغتسل منه٦٥٨لا يغتسل أحدكم في الماء الدائم وهو جنب٦٥٩دعوه ولا تزرموه٦٦٠وقتيبة بن سعيد، جميعا عن الدراوردي، - قال يحيى بن يحيى أخبرنا عبد…٦٦١- وهو عم إسحاق - قال بينما نحن في المسجد مع رسول الله صلى الله عليه…٦٦٢زوج النبي صلى الله عليه وسلم أن رسول الله صلى الله عليه وسلم كان يؤتى…٦٦٣قالت أتي رسول الله صلى الله عليه وسلم بصبي يرضع فبال في حجره فدعا…٦٦٤بهذا الإسناد مثل حديث ابن نمير .٦٦٥أنها أتت رسول الله صلى الله عليه وسلم بابن لها لم يأكل الطعام فوضعته…٦٦٦بهذا الإسناد وقال فدعا بماء فرشه .٦٦٧أن ابن شهاب، أخبره قال أخبرني عبيد الله بن عبد الله بن عتبة بن مسعود،…٦٦٨والأسود، أن رجلا، نزل بعائشة فأصبح يغسل ثوبه فقالت عائشة إنما كان…٦٦٩وهمام، عن عائشة، في المني قالت كنت أفركه من ثوب رسول الله صلى الله…٦٧٠ومغيرة، كل هؤلاء عن إبراهيم، عن الأسود، عن عائشة، في حت المني من ثوب…٦٧١بنحو حديثهم .٦٧٢وحدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن بشر، عن عمرو بن ميمون، قال…٦٧٣وابن أبي زائدة، كلهم عن عمرو بن ميمون، بهذا الإسناد أما ابن أبي زائدة…٦٧٤قال كنت نازلا على عائشة فاحتلمت في ثوبى فغمستهما في الماء فرأتني…٦٧٥تحته ثم تقرصه بالماء ثم تنضحه ثم تصلي فيه٦٧٦ومالك بن أنس، وعمرو بن الحارث، كلهم عن هشام بن عروة، بهذا الإسناد مثل…٦٧٧أما إنهما ليعذبان وما يعذبان في كبير أما أحدهما فكان يمشي بالنميمة…٦٧٨وكان الآخر لا يستنزه عن البول أو من البول
