٥٦٤٦يسلم الراكب على الماشي والماشي على القاعد والقليل على الكثير٥٦٤٧ما لكم ولمجالس الصعدات اجتنبوا مجالس الصعدات٥٦٤٨إياكم والجلوس بالطرقات٥٦٤٩بهذا الإسناد .٥٦٥٠حق المسلم على المسلم خمس٥٦٥١حق المسلم على المسلم ست٥٦٥٢إذا سلم عليكم أهل الكتاب فقولوا وعليكم٥٦٥٣وابن، بشار - واللفظ لهما - قالا حدثنا محمد بن جعفر، حدثنا شعبة، قال…٥٦٥٤إن اليهود إذا سلموا عليكم يقول أحدهم السام عليكم فقل عليك٥٦٥٥وحدثني زهير بن حرب، حدثنا عبد الرحمن، عن سفيان، عن عبد الله بن دينار،…٥٦٥٦يا عائشة إن الله يحب الرفق في الأمر كله٥٦٥٧وعبد بن حميد، جميعا عن يعقوب بن إبراهيم، بن سعد حدثنا أبي، عن صالح، ح…٥٦٥٨حدثنا أبو كريب، حدثنا أبو معاوية، عن الأعمش، عن مسلم، عن مسروق، عن…٥٦٥٩مه يا عائشة فإن الله لا يحب الفحش والتفحش٥٦٦٠وحجاج بن الشاعر، قالا حدثنا حجاج بن محمد، قال قال ابن جريج أخبرني أبو…٥٦٦١لا تبدءوا اليهود ولا النصارى بالسلام فإذا لقيتم أحدهم في طريق فاضطروه…٥٦٦٢إذا لقيتم اليهود٥٦٦٣بن مالك أن رسول الله صلى الله عليه وسلم مر على غلمان فسلم عليهم .٥٦٦٤بهذا الإسناد .٥٦٦٥ومحمد بن الوليد، قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن سيار، قال…٥٦٦٦إذنك على أن يرفع الحجاب وأن تستمع سوادي حتى أنهاك٥٦٦٧ومحمد بن عبد الله بن نمير، وإسحاق بن إبراهيم، قال إسحاق أخبرنا وقال…٥٦٦٨إنه قد أذن لكن أن تخرجن لحاجتكن٥٦٧١أن أزواج، رسول الله صلى الله عليه وسلم كن يخرجن بالليل إذا تبرزن إلى…٥٦٧٢بهذا الإسناد نحوه .٥٦٧٣ألا لا يبيتن رجل عند امرأة ثيب إلا أن يكون ناكحا أو ذا محرم٥٦٧٤إياكم والدخول على النساء٥٦٧٥والليث، بن سعد وحيوة بن شريح وغيرهم أن يزيد بن أبي حبيب، حدثهم بهذا…٥٦٧٦يقول الحمو أخ الزوج وما أشبهه من أقارب الزوج ابن العم ونحوه .٥٦٧٧إن الله قد برأها من ذلك٥٦٧٨يا فلان هذه زوجتي فلانة٥٦٧٩على رسلكما إنها صفية بنت حيى٥٦٨٠إن الشيطان يبلغ من الإنسان مبلغ الدم٥٦٨١ألا أخبركم عن النفر الثلاثة أما أحدهم فأوى إلى الله فآواه الله وأما…٥٦٨٢أبي كثير أن إسحاق بن عبد الله بن أبي طلحة، حدثه في، هذا الإسناد بمثله…٥٦٨٣لا يقيمن أحدكم الرجل من مجلسه ثم يجلس فيه٥٦٨٤لا يقيم الرجل الرجل من مقعده ثم يجلس فيه ولكن تفسحوا وتوسعوا٥٦٨٥ولكن تفسحوا وتوسعوا٥٦٨٦لا يقيمن أحدكم أخاه ثم يجلس في مجلسه٥٦٨٧بهذا الإسناد مثله .٥٦٨٨لا يقيمن أحدكم أخاه يوم الجمعة ثم ليخالف إلى مقعده فيقعد فيه ولكن…٥٦٨٩وقال، قتيبة أيضا حدثنا عبد العزيز، - يعني ابن محمد - كلاهما عن سهيل،…٥٦٩٠لا يدخل هؤلاء عليكم٥٦٩١ألا أرى هذا يعرف ما ها هنا لا يدخلن عليكن٥٦٩٢قالت تزوجني الزبير وما له في الأرض من مال ولا مملوك ولا شىء غير فرسه…٥٦٩٣أن أسماء، قالت كنت أخدم الزبير خدمة البيت وكان له فرس وكنت أسوسه فلم…٥٦٩٤إذا كان ثلاثة فلا يتناجى اثنان دون واحد٥٦٩٥وابن، نمير ح وحدثنا ابن، نمير حدثنا أبي ح، وحدثنا محمد بن المثنى،…٥٦٩٦إذا كنتم ثلاثة فلا يتناجى اثنان دون الآخر حتى تختلطوا بالناس من أجل…٥٦٩٧إذا كنتم ثلاثة فلا يتناجى اثنان دون صاحبهما فإن ذلك يحزنه٥٦٩٨بهذا الإسناد .٥٦٩٩زوج النبي صلى الله عليه وسلم أنها قالت كان إذا اشتكى رسول الله صلى…٥٧٠٠أن جبريل، أتى النبي صلى الله عليه وسلم فقال يا محمد اشتكيت فقال " نعم…٥٧٠١حدثنا محمد بن رافع، حدثنا عبد الرزاق، حدثنا معمر، عن همام بن منبه،…٥٧٠٢العين حق ولو كان شىء سابق القدر سبقته العين وإذا استغسلتم فاغسلوا٥٧٠٣يا عائشة أشعرت أن الله أفتاني فيما استفتيته فيه جاءني رجلان فقعد…٥٧٠٤فأمرت بها فدفنت٥٧٠٥ما كان الله ليسلطك على ذاك٥٧٠٦يحدث أن يهودية، جعلت سما في لحم ثم أتت به رسول الله صلى الله عليه…٥٧٠٧أذهب الباس رب الناس واشف أنت الشافي لا شفاء إلا شفاؤك شفاء لا يغادر…٥٧٠٨حدثنا يحيى بن يحيى، أخبرنا هشيم، ح حدثنا أبو بكر بن أبي شيبة، وأبو…٥٧٠٩أذهب الباس رب الناس اشفه أنت الشافي لا شفاء إلا شفاؤك شفاء لا يغادر…٥٧١٠أذهب الباس رب الناس واشف أنت الشافي لا شفاء إلا شفاؤك شفاء لا يغادر…٥٧١١ومسلم بن صبيح، عن مسروق، عن عائشة، قالت كان رسول الله صلى الله عليه…٥٧١٢أذهب الباس رب الناس بيدك الشفاء لا كاشف له إلا أنت٥٧١٣بهذا الإسناد مثله .٥٧١٤ويحيى بن أيوب، قالا حدثنا عباد بن عباد، عن هشام، بن عروة عن أبيه، عن…٥٧١٥أن النبي صلى الله عليه وسلم كان إذا اشتكى يقرأ على نفسه بالمعوذات…٥٧١٦وحرملة، قالا أخبرنا ابن وهب، أخبرني يونس، ح وحدثنا عبد بن حميد،…٥٧١٧فقالت رخص رسول الله صلى الله عليه وسلم لأهل بيت من الأنصار في الرقية…٥٧١٨قالت رخص رسول الله صلى الله عليه وسلم لأهل بيت من الأنصار في الرقية…٥٧١٩باسم الله تربة أرضنا بريقة بعضنا ليشفى به سقيمنا بإذن ربنا٥٧٢٠أن رسول الله صلى الله عليه وسلم كان يأمرها أن تسترقي من العين .٥٧٢١بهذا الإسناد مثله .٥٧٢٢قالت كان رسول الله صلى الله عليه وسلم يأمرني أن أسترقي من العين .٥٧٢٣في الرقى قال رخص في الحمة والنملة والعين .٥٧٢٤قال رخص رسول الله صلى الله عليه وسلم في الرقية من العين والحمة…٥٧٢٥بها نظرة فاسترقوا لها٥٧٢٦ما لي أرى أجسام بني أخي ضارعة تصيبهم الحاجة٥٧٢٧من استطاع منكم أن ينفع أخاه فليفعل٥٧٢٨بهذا الإسناد . مثله غير أنه قال فقال رجل من القوم أرقيه يا رسول الله…٥٧٢٩من استطاع منكم أن ينفع أخاه فليفعل٥٧٣٠بهذا الإسناد مثله .٥٧٣١ما أرى بأسا من استطاع منكم أن ينفع أخاه فلينفعه٥٧٣٢اعرضوا على رقاكم لا بأس بالرقى ما لم يكن فيه شرك٥٧٣٣وما أدراك أنها رقية٥٧٣٤بهذا الإسناد وقال في الحديث فجعل يقرأ أم القرآن ويجمع بزاقه ويتفل…٥٧٣٥ما كان يدريه أنها رقية اقسموا واضربوا لي بسهم معكم٥٧٣٦وحدثني محمد بن المثنى، حدثنا وهب بن جرير، حدثنا هشام، بهذا الإسناد .…٥٧٣٧ضع يدك على الذي تألم من جسدك وقل باسم الله . ثلاثا . وقل سبع مرات…٥٧٣٨ذاك شيطان يقال له خنزب فإذا أحسسته فتعوذ بالله منه واتفل على يسارك…٥٧٣٩أنه أتى النبي صلى الله عليه وسلم . فذكر بمثله ولم يذكر في حديث سالم…٥٧٤٠قال قلت يا رسول الله . ثم ذكر بمثل حديثهم .٥٧٤١لكل داء دواء فإذا أصيب دواء الداء برأ بإذن الله عز وجل٥٧٤٢وأبو الطاهر، قالا حدثنا ابن وهب، أخبرني عمرو، أن بكيرا، حدثه أن عاصم…٥٧٤٣إن كان في شىء من أدويتكم خير ففي شرطة محجم أو شربة من عسل أو لذعة بنار٥٧٤٤أن أم سلمة، استأذنت رسول الله صلى الله عليه وسلم في الحجامة فأمر…٥٧٤٥قال بعث رسول الله صلى الله عليه وسلم إلى أبى بن كعب طبيبا فقطع منه…٥٧٤٦بهذا الإسناد ولم يذكرا فقطع منه عرقا .٥٧٤٧قال رمي أبى يوم الأحزاب على أكحله فكواه رسول الله صلى الله عليه وسلم .٥٧٤٨قال رمي سعد بن معاذ في أكحله - قال - فحسمه النبي صلى الله عليه وسلم…٥٧٤٩أن النبي صلى الله عليه وسلم احتجم وأعطى الحجام أجره واستعط .٥٧٥٠وقال، أبو كريب - واللفظ له - أخبرنا وكيع، عن مسعر، عن عمرو بن عامر…٥٧٥١الحمى من فيح جهنم فابردوها بالماء٥٧٥٢إن شدة الحمى من فيح جهنم فابردوها بالماء٥٧٥٣الحمى من فيح جهنم فأطفئوها بالماء٥٧٥٤الحمى من فيح جهنم فأطفئوها بالماء٥٧٥٥الحمى من فيح جهنم فابردوها بالماء٥٧٥٦وعبدة بن سليمان، جميعا عن هشام، بهذا الإسناد مثله .٥٧٥٧أنها كانت تؤتى بالمرأة الموعوكة فتدعو بالماء فتصبه في جيبها وتقول إن…٥٧٥٨أنها من فيح جهنم٥٧٥٩إن الحمى فور من جهنم فابردوها بالماء٥٧٦٠الحمى من فور جهنم فابردوها عنكم بالماء٥٧٦١لا يبقى أحد منكم إلا لد غير العباس فإنه لم يشهدكم٥٧٦٣علامه تدغرن أولادكن بهذا العلاق عليكن بهذا العود الهندي فإن فيه سبعة…٥٧٦٥علامه تدغرن أولادكن بهذا الإعلاق عليكم بهذا العود الهندي - يعني به…٥٧٦٦إن في الحبة السوداء شفاء من كل داء إلا السام٥٧٦٧وحرملة، قالا أخبرنا ابن وهب، أخبرني يونس، عن ابن، شهاب عن سعيد بن…٥٧٦٨ما من داء إلا في الحبة السوداء منه شفاء إلا السام٥٧٦٩التلبينة مجمة لفؤاد المريض تذهب بعض الحزن٥٧٧٠ومحمد بن بشار، - واللفظ لابن المثنى - قالا حدثنا محمد بن جعفر، حدثنا…٥٧٧١أن رجلا، أتى النبي صلى الله عليه وسلم فقال إن أخي عرب بطنه . فقال له…٥٧٧٢الطاعون رجز أو عذاب أرسل على بني إسرائيل أو على من كان قبلكم فإذا…٥٧٧٣الطاعون آية الرجز ابتلى الله عز وجل به ناسا من عباده فإذا سمعتم به…٥٧٧٤إن هذا الطاعون رجز سلط على من كان قبلكم أو على بني إسرائيل فإذا كان…٥٧٧٥هو عذاب أو رجز أرسله الله على طائفة من بني إسرائيل أو ناس كانوا قبلكم…٥٧٧٦دينار بإسناد ابن جريج نحو حديثه .٥٧٧٧إن هذا الوجع أو السقم رجز عذب به بعض الأمم قبلكم ثم بقي بعد بالأرض…٥٧٧٨بإسناد يونس نحو حديثه .٥٧٧٩إذا كنت بأرض فوقع بها فلا تخرج منها وإذا بلغك أنه بأرض فلا تدخلها٥٧٨٠بهذا الإسناد غير أنه لم يذكر قصة عطاء بن يسار في أول الحديث .٥٧٨١وخزيمة بن ثابت، وأسامة بن زيد، قالوا قال رسول الله صلى الله عليه وسلم…٥٧٨٢وإسحاق بن إبراهيم، كلاهما عن جرير، عن الأعمش، عن حبيب، عن إبراهيم بن…٥٧٨٣وحدثنيه وهب بن بقية، أخبرنا خالد، - يعني الطحان - عن الشيباني، عن…٥٧٨٤إذا سمعتم به بأرض فلا تقدموا عليه وإذا وقع بأرض وأنتم بها فلا تخرجوا…٥٧٨٥ومحمد بن رافع، وعبد بن حميد، قال ابن رافع حدثنا وقال الآخران، أخبرنا…٥٧٨٦وحرملة بن يحيى، قالا أخبرنا ابن وهب، أخبرني يونس، عن ابن شهاب، بهذا…٥٧٨٧إذا سمعتم به بأرض فلا تقدموا عليه . وإذا وقع بأرض وأنتم بها فلا…٥٧٨٨لا عدوى ولا صفر ولا هامة٥٧٨٩لا عدوى ولا طيرة ولا صفر ولا هامة٥٧٩٠أن أبا هريرة، قال قال النبي صلى الله عليه وسلم " لا عدوى " . فقام…٥٧٩١وحرملة، - وتقاربا في اللفظ - قالا أخبرنا ابن وهب، أخبرني يونس، عن ابن…٥٧٩٢وحسن الحلواني، وعبد بن حميد، قال عبد حدثني وقال، الآخران حدثنا يعقوب،…٥٧٩٣بهذا الإسناد نحوه .٥٧٩٤لا عدوى ولا هامة ولا نوء ولا صفر٥٧٩٥لا عدوى ولا طيرة ولا غول٥٧٩٦لا
