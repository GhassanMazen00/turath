كتاب الجنة وصفة نعيمها وأهلها

١٠٢ حديثًا
٧١٣٠حفت الجنة بالمكاره وحفت النار بالشهوات٧١٣١وحدثني زهير بن حرب، حدثنا شبابة، حدثني ورقاء، عن أبي الزناد، عن…٧١٣٢قال الله عز وجل أعددت لعبادي الصالحين ما لا عين رأت ولا أذن سمعت ولا…٧١٣٣قال الله عز وجل أعددت لعبادي الصالحين ما لا عين رأت ولا أذن سمعت ولا…٧١٣٤يقول الله عز وجل أعددت لعبادي الصالحين ما لا عين رأت ولا أذن سمعت ولا…٧١٣٥فيها ما لا عين رأت ولا أذن سمعت ولا خطر على قلب بشر٧١٣٦إن في الجنة لشجرة يسير الراكب في ظلها مائة سنة٧١٣٧حدثنا قتيبة بن سعيد، حدثنا المغيرة، - يعني ابن عبد الرحمن الحزامي -…٧١٣٩إن في الجنة لشجرة يسير الراكب في ظلها مائة عام لا يقطعها٧١٤٠إن الله يقول لأهل الجنة يا أهل الجنة . فيقولون لبيك ربنا وسعديك…٧١٤٢إن أهل الجنة ليتراءون الغرفة في الجنة كما تراءون الكوكب في السماء٧١٤٣إن أهل الجنة ليتراءون أهل الغرف من فوقهم كما تتراءون الكوكب الدري…٧١٤٥من أشد أمتي لي حبا ناس يكونون بعدي يود أحدهم لو رآني بأهله وماله٧١٤٦إن في الجنة لسوقا يأتونها كل جمعة فتهب ريح الشمال فتحثو في وجوههم…٧١٤٧إن أول زمرة تدخل الجنة على صورة القمر ليلة البدر والتي تليها على أضوإ…٧١٤٨قال اختصم الرجال والنساء أيهم في الجنة أكثر فسألوا أبا هريرة فقال قال…٧١٤٩أول من يدخل الجنة٧١٥٠أول زمرة تدخل الجنة من أمتي على صورة القمر ليلة البدر ثم الذين يلونهم…٧١٥١أول زمرة تلج الجنة صورهم على صورة القمر ليلة البدر لا يبصقون فيها ولا…٧١٥٢إن أهل الجنة يأكلون فيها ويشربون ولا يتفلون ولا يبولون ولا يتغوطون…٧١٥٣بهذا الإسناد إلى قوله " كرشح المسك " .٧١٥٤يأكل أهل الجنة فيها ويشربون ولا يتغوطون ولا يمتخطون ولا يبولون ولكن…٧١٥٥ويلهمون التسبيح والتكبير كما يلهمون النفس٧١٥٦من يدخل الجنة ينعم لا يبأس لا تبلى ثيابه ولا يفنى شبابه٧١٥٧ينادي مناد إن لكم أن تصحوا فلا تسقموا أبدا وإن لكم أن تحيوا فلا…٧١٥٨إن للمؤمن في الجنة لخيمة من لؤلؤة واحدة مجوفة طولها ستون ميلا للمؤمن…٧١٥٩في الجنة خيمة من لؤلؤة مجوفة عرضها ستون ميلا في كل زاوية منها أهل ما…٧١٦٠الخيمة درة طولها في السماء ستون ميلا في كل زاوية منها أهل للمؤمن لا…٧١٦١سيحان وجيحان والفرات والنيل كل من أنهار الجنة٧١٦٢يدخل الجنة أقوام أفئدتهم مثل أفئدة الطير٧١٦٣خلق الله عز وجل آدم على صورته طوله ستون ذراعا فلما خلقه قال اذهب فسلم…٧١٦٤يؤتى بجهنم يومئذ لها سبعون ألف زمام مع كل زمام سبعون ألف ملك يجرونها٧١٦٥ناركم هذه التي يوقد ابن آدم جزء من سبعين جزءا من حر جهنم٧١٦٦كلهن مثل حرها٧١٦٧حدثنا يحيى بن أيوب، حدثنا خلف بن خليفة، حدثنا يزيد بن كيسان، عن أبي،…٧١٦٨هذا وقع في أسفلها فسمعتم وجبتها٧١٦٩إن منهم من تأخذه النار إلى كعبيه ومنهم من تأخذه إلى حجزته ومنهم من…٧١٧٠منهم من تأخذه النار إلى كعبيه ومنهم من تأخذه النار إلى ركبتيه ومنهم…٧١٧١ومحمد بن بشار، قالا حدثنا روح، حدثنا سعيد، بهذا الإسناد وجعل مكان…٧١٧٢احتجت النار والجنة فقالت هذه يدخلني الجبارون والمتكبرون . وقالت هذه…٧١٧٣تحاجت النار والجنة فقالت النار أوثرت بالمتكبرين والمتجبرين . وقالت…٧١٧٤احتجت الجنة والنار٧١٧٥تحاجت الجنة والنار فقالت النار أوثرت بالمتكبرين والمتجبرين . وقالت…٧١٧٦احتجت الجنة والنار٧١٧٧لا تزال جهنم تقول هل من مزيد . حتى يضع فيها رب العزة تبارك وتعالى…٧١٧٨وحدثني زهير بن حرب، حدثنا عبد الصمد بن عبد الوارث، حدثنا أبان بن…٧١٧٩لا تزال جهنم يلقى فيها وتقول هل من مزيد حتى يضع رب العزة فيها قدمه…٧١٨٠يبقى من الجنة ما شاء الله أن يبقى ثم ينشئ الله تعالى لها خلقا مما يشاء٧١٨١يجاء بالموت يوم القيامة كأنه كبش أملح - زاد أبو كريب - فيوقف بين…٧١٨٢إذا أدخل أهل الجنة الجنة وأهل النار النار قيل يا أهل الجنة٧١٨٣يدخل الله أهل الجنة الجنة ويدخل أهل النار النار ثم يقوم مؤذن بينهم…٧١٨٤إذا صار أهل الجنة إلى الجنة وصار أهل النار إلى النار أتي بالموت حتى…٧١٨٥ضرس الكافر أو ناب الكافر مثل أحد وغلظ جلده مسيرة ثلاث٧١٨٦ما بين منكبى الكافر في النار مسيرة ثلاثة أيام للراكب المسرع٧١٨٧ألا أخبركم بأهل الجنة٧١٨٨بهذا الإسناد بمثله غير أنه قال " ألا أدلكم " .٧١٨٩ألا أخبركم بأهل الجنة كل ضعيف متضعف لو أقسم على الله لأبره ألا أخبركم…٧١٩٠رب أشعث مدفوع بالأبواب لو أقسم على الله لأبره٧١٩١إذ انبعث أشقاها انبعث بها رجل عزيز عارم منيع في رهطه مثل أبي زمعة٧١٩٢رأيت عمرو بن لحى بن قمعة بن خندف أبا بني كعب هؤلاء يجر قصبه في النار٧١٩٣رأيت عمرو بن عامر الخزاعي يجر قصبه في النار وكان أول من سيب السيوب٧١٩٤صنفان من أهل النار لم أرهما قوم معهم سياط كأذناب البقر يضربون بها…٧١٩٥يوشك إن طالت بك مدة أن ترى قوما في أيديهم مثل أذناب البقر يغدون في…٧١٩٦إن طالت بك مدة أوشكت أن ترى قوما يغدون في سخط الله ويروحون في لعنته…٧١٩٧والله ما الدنيا في الآخرة إلا مثل ما يجعل أحدكم إصبعه هذه - وأشار…٧١٩٨يحشر الناس يوم القيامة حفاة عراة غرلا٧١٩٩وابن، نمير قالا حدثنا أبو خالد الأحمر، عن حاتم، بن أبي صغيرة بهذا…٧٢٠٠إنكم ملاقو الله مشاة حفاة عراة غرلا٧٢٠١يا أيها الناس إنكم تحشرون إلى الله حفاة عراة غرلا { كما بدأنا أول خلق…٧٢٠٢يحشر الناس على ثلاث طرائق راغبين راهبين واثنان على بعير وثلاثة على…٧٢٠٣يقوم أحدهم في رشحه إلى أنصاف أذنيه٧٢٠٤حتى يغيب أحدهم في رشحه إلى أنصاف أذنيه٧٢٠٥إن العرق يوم القيامة ليذهب في الأرض سبعين باعا وإنه ليبلغ إلى أفواه…٧٢٠٦تدنى الشمس يوم القيامة من الخلق حتى تكون منهم كمقدار ميل٧٢٠٧ألا إن ربي أمرني أن أعلمكم ما جهلتم مما علمني يومي هذا كل مال نحلته…٧٢٠٨كل مال نحلته عبدا حلال٧٢٠٩حدثني عبد الرحمن بن بشر العبدي، حدثنا يحيى بن سعيد، عن هشام، - صاحب…٧٢١٠وحدثني أبو عمار، حسين بن حريث حدثنا الفضل بن موسى، عن الحسين، عن مطر،…٧٢١١إن أحدكم إذا مات عرض عليه مقعده بالغداة والعشي إن كان من أهل الجنة…٧٢١٢إذا مات الرجل عرض عليه مقعده بالغداة والعشي إن كان من أهل الجنة…٧٢١٣من يعرف أصحاب هذه الأقبر٧٢١٤لولا أن لا تدافنوا لدعوت الله أن يسمعكم من عذاب القبر٧٢١٥يهود تعذب في قبورها٧٢١٦إن العبد إذا وضع في قبره وتولى عنه أصحابه إنه ليسمع قرع نعالهم٧٢١٧إن الميت إذا وضع في قبره إنه ليسمع خفق نعالهم إذا انصرفوا٧٢١٨إن العبد إذا وضع في قبره وتولى عنه أصحابه٧٢١٩{ يثبت الله الذين آمنوا بالقول الثابت } قال٧٢٢٠ومحمد بن المثنى، وأبو بكر بن نافع قالوا حدثنا عبد الرحمن، - يعنون ابن…٧٢٢١إذا خرجت روح المؤمن تلقاها ملكان يصعدانها٧٢٢٢هذا مصرع فلان غدا إن شاء الله٧٢٢٣يا أبا جهل بن هشام يا أمية بن خلف يا عتبة بن ربيعة يا شيبة بن ربيعة…٧٢٢٤حدثني يوسف بن حماد المعني، حدثنا عبد الأعلى، عن سعيد، عن قتادة، عن…٧٢٢٥من حوسب يوم القيامة عذب٧٢٢٦بهذا الإسناد نحوه .٧٢٢٧ليس أحد يحاسب إلا هلك٧٢٢٨من نوقش الحساب هلك٧٢٢٩لا يموتن أحدكم إلا وهو يحسن بالله الظن٧٢٣٠بهذا الإسناد مثله .٧٢٣١لا يموتن أحدكم إلا وهو يحسن الظن بالله عز وجل٧٢٣٢يبعث كل عبد على ما مات عليه٧٢٣٣حدثنا أبو بكر بن نافع، حدثنا عبد الرحمن بن مهدي، عن سفيان، عن الأعمش،…٧٢٣٤إذا أراد الله بقوم عذابا أصاب العذاب من كان فيهم ثم بعثوا على أعمالهم
