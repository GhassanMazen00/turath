٤١ حديثًا
٤١٦٣لا تبتعه ولا تعد في صدقتك فإن العائد في صدقته كالكلب يعود في قيئه٤١٦٤لا تبتعه وإن أعطاكه بدرهم٤١٦٥لا تشتره وإن أعطيته بدرهم فإن مثل العائد في صدقته كمثل الكلب يعود في…٤١٦٦بهذا الإسناد غير أن مالك وروح أتم وأكثر .٤١٦٧لا تبتعه ولا تعد في صدقتك٤١٦٨وابن، رمح جميعا عن الليث بن سعد، ح وحدثنا المقدمي، ومحمد بن المثنى…٤١٦٩لا تعد في صدقتك يا عمر٤١٧٠مثل الذي يرجع في صدقته كمثل الكلب يقيء ثم يعود في قيئه فيأكله٤١٧١يذكر بهذا الإسناد نحوه .٤١٧٢أن محمد ابن فاطمة بنت رسول الله، صلى الله عليه وسلم حدثه بهذا…٤١٧٣إنما مثل الذي يتصدق بصدقة ثم يعود في صدقته كمثل الكلب يقيء ثم يأكل…٤١٧٤العائد في هبته كالعائد في قيئه٤١٧٥بهذا الإسناد مثله .٤١٧٦العائد في هبته كالكلب يقيء ثم يعود في قيئه٤١٧٧أكل ولدك نحلته مثل هذا٤١٧٨قال أتى بي أبي إلى رسول الله صلى الله عليه وسلم فقال إني نحلت ابني…٤١٧٩وإسحاق بن إبراهيم، وابن أبي عمر، عن ابن عيينة، ح وحدثنا قتيبة، وابن،…٤١٨٠قال وقد أعطاه أبوه غلاما فقال له النبي صلى الله عليه وسلم " ما هذا…٤١٨١أفعلت هذا بولدك كلهم٤١٨٢يا بشير ألك ولد سوى هذا٤١٨٣أن رسول الله صلى الله عليه وسلم قال " ألك بنون سواه " . قال نعم . قال…٤١٨٤لا تشهدني على جور٤١٨٥أكل بنيك قد نحلت مثل ما نحلت النعمان٤١٨٦أكل ولدك أعطيته هذا٤١٨٧قال قالت امرأة بشير انحل ابني غلامك وأشهد لي رسول الله صلى الله عليه…٤١٨٨أيما رجل أعمر عمرى له ولعقبه فإنها للذي أعطيها لا ترجع إلى الذي…٤١٨٩من أعمر رجلا عمرى له ولعقبه فقد قطع قوله حقه فيها وهي لمن أعمر ولعقبه٤١٩٠أيما رجل أعمر رجلا عمرى له ولعقبه فقال قد أعطيتكها وعقبك ما بقي منكم…٤١٩١وعبد بن حميد، - واللفظ لعبد - قالا أخبرنا عبد، الرزاق أخبرنا معمر، عن…٤١٩٢- وهو ابن عبد الله - أن رسول الله صلى الله عليه وسلم قضى فيمن أعمر…٤١٩٣العمرى لمن وهبت له٤١٩٤أن نبي الله صلى الله عليه وسلم قال بمثله .٤١٩٥يرفعه إلى النبي صلى الله عليه وسلم .٤١٩٦أمسكوا عليكم أموالكم ولا تفسدوها فإنه من أعمر عمرى فهي للذي أعمرها…٤١٩٧أمسكوا عليكم أموالكم٤١٩٨وإسحاق بن منصور، - واللفظ لابن رافع - قالا حدثنا عبد الرزاق، أخبرنا…٤١٩٩وإسحاق بن إبراهيم، - واللفظ لأبي بكر قال إسحاق أخبرنا وقال أبو بكر،…٤٢٠٠ومحمد بن بشار، قالا حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت قتادة،…٤٢٠١العمرى ميراث لأهلها٤٢٠٢وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن قتادة، عن النضر بن…٤٢٠٣بهذا الإسناد غير أنه قال " ميراث لأهلها " . أو قال " جائزة " .
