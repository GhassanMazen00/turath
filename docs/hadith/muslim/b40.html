كتاب الألفاظ من الأدب وغيرها

٢٣ حديثًا
٥٨٦٢قال الله عز وجل يسب ابن آدم الدهر وأنا الدهر بيدي الليل والنهار٥٨٦٣قال الله عز وجل يؤذيني ابن آدم يسب الدهر وأنا الدهر أقلب الليل والنهار٥٨٦٤قال الله عز وجل يؤذيني ابن آدم يقول يا خيبة الدهر . فلا يقولن أحدكم…٥٨٦٥لا يقولن أحدكم يا خيبة الدهر . فإن الله هو الدهر٥٨٦٦لا تسبوا الدهر فإن الله هو الدهر٥٨٦٧لا يسب أحدكم الدهر فإن الله هو الدهر ولا يقولن أحدكم للعنب الكرم .…٥٨٦٨لا تقولوا كرم . فإن الكرم قلب المؤمن٥٨٦٩لا تسموا العنب الكرم فإن الكرم الرجل المسلم٥٨٧٠لا يقولن أحدكم الكرم . فإنما الكرم قلب المؤمن٥٨٧١لا يقولن أحدكم للعنب الكرم . إنما الكرم الرجل المسلم٥٨٧٢لا تقولوا الكرم . ولكن قولوا الحبلة٥٨٧٣لا تقولوا الكرم . ولكن قولوا العنب والحبلة٥٨٧٤لا يقولن أحدكم عبدي وأمتي . كلكم عبيد الله وكل نسائكم إماء الله ولكن…٥٨٧٥لا يقولن أحدكم عبدي . فكلكم عبيد الله ولكن ليقل فتاى . ولا يقل العبد…٥٨٧٦ولا يقل العبد لسيده مولاى٥٨٧٧لا يقل أحدكم اسق ربك أطعم ربك وضئ ربك . ولا يقل أحدكم ربي . وليقل…٥٨٧٨لا يقولن أحدكم خبثت نفسي . ولكن ليقل لقست نفسي٥٨٧٩بهذا الإسناد .٥٨٨٠لا يقل أحدكم خبثت نفسي . وليقل لقست نفسي٥٨٨١كانت امرأة من بني إسرائيل قصيرة تمشي مع امرأتين طويلتين فاتخذت رجلين…٥٨٨٢والمستمر، قالا سمعنا أبا نضرة، يحدث عن أبي سعيد الخدري، أن رسول الله…٥٨٨٣من عرض عليه ريحان فلا يرده فإنه خفيف المحمل طيب الريح٥٨٨٤قال كان ابن عمر إذا استجمر استجمر بالألوة غير مطراة وبكافور يطرحه مع…
