كتاب صلاة المسافرين وقصرها

٣٧٧ حديثًا•صفحة ٢ من ٣
١٧٢٣يا عائشة إن عينى تنامان ولا ينام قلبي١٧٢٤رسول الله صلى الله عليه وسلم فقالت كان يصلي ثلاث عشرة ركعة يصلي ثمان…١٧٢٥رسول الله صلى الله عليه وسلم . بمثله غير أن في حديثهما تسع ركعات…١٧٢٦سمع أبا سلمة، قال أتيت عائشة فقلت أى أمه أخبريني عن صلاة رسول الله…١٧٢٧تقول كانت صلاة رسول الله صلى الله عليه وسلم من الليل عشر ركعات ويوتر…١٧٢٨قال سألت الأسود بن يزيد عما حدثته عائشة، عن صلاة، رسول الله صلى الله…١٧٢٩قالت كان رسول الله صلى الله عليه وسلم يصلي من الليل حتى يكون آخر…١٧٣٠رسول الله صلى الله عليه وسلم فقالت كان يحب الدائم . قال قلت أى حين…١٧٣١قالت ما ألفى رسول الله صلى الله عليه وسلم السحر الأعلى في بيتي - أو…١٧٣٢ونصر بن علي، وابن أبي عمر، قال أبو بكر حدثنا سفيان بن عيينة، عن أبي…١٧٣٣وحدثنا ابن أبي عمر، حدثنا سفيان، عن زياد بن سعد، عن ابن أبي عتاب، عن…١٧٣٤قومي فأوتري يا عائشة١٧٣٥أن رسول الله صلى الله عليه وسلم كان يصلي صلاته بالليل وهي معترضة بين…١٧٣٦قالت من كل الليل قد أوتر رسول الله صلى الله عليه وسلم فانتهى وتره إلى…١٧٣٧وزهير بن حرب، قالا حدثنا وكيع، عن سفيان، عن أبي حصين، عن يحيى بن…١٧٣٨قالت كل الليل قد أوتر رسول الله صلى الله عليه وسلم فانتهى وتره إلى…١٧٣٩أليس لكم في أسوة١٧٤٠أنه طلق امرأته ثم انطلق إلى المدينة ليبيع عقاره . فذكر نحوه .١٧٤١وحدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن بشر، حدثنا سعيد بن أبي…١٧٤٢ومحمد بن رافع، كلاهما عن عبد الرزاق، أخبرنا معمر، عن قتادة، عن زرارة…١٧٤٣وقتيبة بن سعيد، جميعا عن أبي عوانة، قال سعيد حدثنا أبو عوانة، عن…١٧٤٤قالت كان رسول الله صلى الله عليه وسلم إذا عمل عملا أثبته وكان إذا نام…١٧٤٥من نام عن حزبه أو عن شىء منه فقرأه فيما بين صلاة الفجر وصلاة الظهر…١٧٤٦صلاة الأوابين حين ترمض الفصال١٧٤٧صلاة الأوابين إذا رمضت الفصال١٧٤٨صلاة الليل مثنى مثنى فإذا خشي أحدكم الصبح صلى ركعة واحدة توتر له ما…١٧٤٩مثنى مثنى فإذا خشيت الصبح فأوتر بركعة١٧٥٠صلاة الليل مثنى مثنى فإذا خفت الصبح فأوتر بواحدة١٧٥١مثنى مثنى فإذا خشيت الصبح فصل ركعة واجعل آخر صلاتك وترا١٧٥٢وبديل، وعمران بن حدير، عن عبد الله بن شقيق، عن ابن عمر، ح وحدثنا محمد…١٧٥٣بادروا الصبح بالوتر١٧٥٤أن ابن عمر، قال من صلى من الليل فليجعل آخر صلاته وترا فإن رسول الله…١٧٥٥اجعلوا آخر صلاتكم بالليل وترا١٧٥٦أن ابن عمر، كان يقول من صلى من الليل فليجعل آخر صلاته وترا قبل الصبح…١٧٥٧الوتر ركعة من آخر الليل١٧٥٨الوتر ركعة من آخر الليل١٧٥٩ركعة من آخر الليل١٧٦٠من صلى فليصل مثنى مثنى فإن أحس أن يصبح سجد سجدة فأوترت له ما صلى١٧٦١حدثنا خلف بن هشام، وأبو كامل قالا حدثنا حماد بن زيد، عن أنس بن سيرين،…١٧٦٢وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن أنس بن سيرين، قال…١٧٦٣صلاة الليل مثنى مثنى فإذا رأيت أن الصبح يدركك فأوتر بواحدة١٧٦٤أوتروا قبل أن تصبحوا١٧٦٥أوتروا قبل الصبح١٧٦٦من خاف أن لا يقوم من آخر الليل فليوتر أوله ومن طمع أن يقوم آخره…١٧٦٧أيكم خاف أن لا يقوم من آخر الليل فليوتر ثم ليرقد ومن وثق بقيام من…١٧٦٨أفضل الصلاة طول القنوت١٧٦٩وحدثنا أبو بكر بن أبي شيبة، وأبو كريب قالا حدثنا أبو معاوية، حدثنا…١٧٧٠إن في الليل لساعة لا يوافقها رجل مسلم يسأل الله خيرا من أمر الدنيا…١٧٧١إن من الليل ساعة لا يوافقها عبد مسلم يسأل الله خيرا إلا أعطاه إياه١٧٧٢ينزل ربنا تبارك وتعالى كل ليلة إلى السماء الدنيا حين يبقى ثلث الليل…١٧٧٣ينزل الله إلى السماء الدنيا كل ليلة حين يمضي ثلث الليل الأول فيقول…١٧٧٤إذا مضى شطر الليل أو ثلثاه ينزل الله تبارك وتعالى إلى السماء الدنيا…١٧٧٥ينزل الله في السماء الدنيا لشطر الليل أو لثلث الليل الآخر فيقول من…١٧٧٦إن الله يمهل حتى إذا ذهب ثلث الليل الأول نزل إلى السماء الدنيا فيقول…١٧٧٧وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي إسحاق، بهذا…١٧٧٩من قام رمضان إيمانا واحتسابا غفر له ما تقدم من ذنبه١٧٨٠من قام رمضان إيمانا واحتسابا غفر له ما تقدم من ذنبه١٧٨١من صام رمضان إيمانا واحتسابا غفر له ما تقدم من ذنبه ومن قام ليلة…١٧٨٢من يقم ليلة القدر فيوافقها - أراه قال - إيمانا واحتسابا غفر له١٧٨٣قد رأيت الذي صنعتم فلم يمنعني من الخروج إليكم إلا أني خشيت أن تفرض…١٧٨٤أما بعد فإنه لم يخف على شأنكم الليلة ولكني خشيت أن تفرض عليكم صلاة…١٧٨٥حدثنا محمد بن مهران الرازي، حدثنا الوليد بن مسلم، حدثنا الأوزاعي،…١٧٨٦حدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت عبدة…١٧٨٧بهذا الإسناد . نحوه ولم يذكر إنما شك شعبة . وما بعده .١٧٨٨اللهم اجعل في قلبي نورا وفي بصري نورا وفي سمعي نورا وعن يميني نورا…١٧٨٩مولى ابن عباس أن ابن عباس، أخبره أنه، بات ليلة عند ميمونة أم المؤمنين…١٧٩٠بهذا الإسناد وزاد ثم عمد إلى شجب من ماء فتسوك وتوضأ وأسبغ الوضوء ولم…١٧٩١حدثني هارون بن سعيد الأيلي، حدثنا ابن وهب، حدثنا عمرو، عن عبد ربه بن…١٧٩٢قال بت ليلة عند خالتي ميمونة بنت الحارث فقلت لها إذا قام رسول الله…١٧٩٣ومحمد بن حاتم، عن ابن عيينة، - قال ابن أبي عمر حدثنا سفيان، - عن عمرو…١٧٩٤اللهم اجعل في قلبي نورا وفي سمعي نورا وفي بصري نورا وعن يميني نورا…١٧٩٥. قال سلمة فلقيت كريبا فقال قال ابن عباس كنت عند خالتي ميمونة فجاء…١٧٩٦وهناد بن السري، قالا حدثنا أبو الأحوص، عن سعيد بن مسروق، عن سلمة بن…١٧٩٧اللهم اجعل لي في قلبي نورا وفي لساني نورا وفي سمعي نورا وفي بصري نورا…١٧٩٨أنه قال رقدت في بيت ميمونة ليلة كان النبي صلى الله عليه وسلم عندها…١٧٩٩اللهم اجعل في قلبي نورا وفي لساني نورا واجعل في سمعي نورا واجعل في…١٨٠٠قال بت ذات ليلة عند خالتي ميمونة فقام النبي صلى الله عليه وسلم يصلي…١٨٠١ومحمد بن رافع، قالا حدثنا وهب بن جرير، أخبرني أبي قال، سمعت قيس بن…١٨٠٢قال بت عند خالتي ميمونة . نحو حديث ابن جريج وقيس بن سعد .١٨٠٣وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي جمرة، قال سمعت…١٨٠٤أن عبد الله بن قيس بن مخرمة، أخبره عن زيد بن خالد الجهني، أنه قال…١٨٠٥ألا تشرع يا جابر١٨٠٦قالت كان رسول الله صلى الله عليه وسلم إذا قام من الليل ليصلي افتتح…١٨٠٧إذا قام أحدكم من الليل فليفتتح صلاته بركعتين خفيفتين١٨٠٨اللهم لك الحمد أنت نور السموات والأرض ولك الحمد أنت قيام السموات…١٨٠٩وابن، نمير وابن أبي عمر قالوا حدثنا سفيان، ح وحدثنا محمد بن رافع، قال…١٨١٠وحدثنا شيبان بن فروخ، حدثنا مهدي، - وهو ابن ميمون - حدثنا عمران،…١٨١١اللهم رب جبرائيل وميكائيل وإسرافيل فاطر السموات والأرض عالم الغيب…١٨١٢وجهت وجهي للذي فطر السموات والأرض حنيفا وما أنا من المشركين إن صلاتي…١٨١٣وحدثناه زهير بن حرب، حدثنا عبد الرحمن بن مهدي، ح وحدثنا إسحاق بن،…١٨١٤سبحان ربي العظيم١٨١٥وإسحاق بن إبراهيم، كلاهما عن جرير، - قال عثمان حدثنا جرير، - عن…١٨١٦وسويد بن سعيد، عن علي بن مسهر، عن الأعمش، بهذا الإسناد . مثله .١٨١٧ذاك رجل بال الشيطان في أذنيه١٨١٨أن الحسين بن علي، حدثه عن علي بن أبي طالب، أن النبي صلى الله عليه…١٨١٩يعقد الشيطان على قافية رأس أحدكم ثلاث عقد إذا نام بكل عقدة يضرب عليك…١٨٢٠اجعلوا من صلاتكم في بيوتكم ولا تتخذوها قبورا١٨٢١صلوا في بيوتكم ولا تتخذوها قبورا١٨٢٢إذا قضى أحدكم الصلاة في مسجده فليجعل لبيته نصيبا من صلاته فإن الله…١٨٢٣مثل البيت الذي يذكر الله فيه والبيت الذي لا يذكر الله فيه مثل الحى…١٨٢٤لا تجعلوا بيوتكم مقابر إن الشيطان ينفر من البيت الذي تقرأ فيه سورة…١٨٢٥ما زال بكم صنيعكم حتى ظننت أنه سيكتب عليكم فعليكم بالصلاة في بيوتكم…١٨٢٦ولو كتب عليكم ما قمتم به١٨٢٧يا أيها الناس عليكم من الأعمال ما تطيقون فإن الله لا يمل حتى تملوا…١٨٢٨أنه سمع أبا سلمة، يحدث عن عائشة، أن رسول الله صلى الله عليه وسلم سئل…١٨٢٩وإسحاق بن إبراهيم، قال زهير حدثنا جرير، عن منصور، عن إبراهيم، عن…١٨٣٠أحب الأعمال إلى الله تعالى أدومها وإن قل١٨٣١قال دخل رسول الله صلى الله عليه وسلم المسجد وحبل ممدود بين ساريتين…١٨٣٢وحدثناه شيبان بن فروخ، حدثنا عبد الوارث، عن عبد العزيز، عن أنس، عن…١٨٣٣لا تنام الليل خذوا من العمل ما تطيقون فوالله لا يسأم الله حتى تسأموا١٨٣٤قالت دخل على رسول الله صلى الله عليه وسلم وعندي امرأة فقال " من هذه "…١٨٣٥إذا نعس أحدكم في الصلاة فليرقد حتى يذهب عنه النوم فإن أحدكم إذا صلى…١٨٣٦إذا قام أحدكم من الليل فاستعجم القرآن على لسانه فلم يدر ما يقول…١٨٣٧يرحمه الله لقد أذكرني كذا وكذا آية كنت أسقطتها من سورة كذا وكذا١٨٣٨رحمه الله لقد أذكرني آية كنت أنسيتها١٨٣٩إنما مثل صاحب القرآن كمثل الإبل المعقلة إن عاهد عليها أمسكها وإن…١٨٤٠وإذا قام صاحب القرآن فقرأه بالليل والنهار ذكره وإذا لم يقم به نسيه١٨٤١بئسما لأحدهم يقول نسيت آية كيت وكيت بل هو نسي استذكروا القرآن فلهو…١٨٤٢لا يقل أحدكم نسيت آية كيت وكيت بل هو نسي١٨٤٣بئسما للرجل أن يقول نسيت سورة كيت وكيت أو نسيت آية كيت وكيت بل هو نسي١٨٤٤تعاهدوا هذا القرآن فوالذي نفس محمد بيده لهو أشد تفلتا من الإبل في…١٨٤٥ما أذن الله لشىء ما أذن لنبي يتغنى بالقرآن١٨٤٦كما يأذن لنبي يتغنى بالقرآن١٨٤٧ما أذن الله لشىء ما أذن لنبي حسن الصوت يتغنى بالقرآن يجهر به١٨٤٨بهذا الإسناد . مثله سواء وقال إن رسول الله صلى الله عليه وسلم . ولم…١٨٤٩ما أذن الله لشىء كأذنه لنبي يتغنى بالقرآن يجهر به١٨٥٠وقتيبة بن سعيد، وابن، حجر قالوا حدثنا إسماعيل، - وهو ابن جعفر - عن…١٨٥١إن عبد الله بن قيس أو الأشعري أعطي مزمارا من مزامير آل داود١٨٥٢لو رأيتني وأنا أستمع لقراءتك البارحة لقد أوتيت مزمارا من مزامير آل…١٨٥٣ووكيع، عن شعبة، عن معاوية بن قرة، قال سمعت عبد الله بن مغفل المزني،…١٨٥٤ومحمد بن بشار، قال ابن المثنى حدثنا محمد بن جعفر، حدثنا شعبة، عن…١٨٥٥بهذا الإسناد نحوه وفي حديث خالد بن الحارث قال على راحلة يسير وهو يقرأ…١٨٥٦تلك السكينة تنزلت للقرآن١٨٥٧اقرأ فلان فإنها السكينة تنزلت عند القرآن أو تنزلت للقرآن١٨٥٨يقول . فذكرا نحوه غير أنهما قالا تنقز .١٨٥٩وحجاج بن الشاعر، - وتقاربا في اللفظ - قالا حدثنا يعقوب بن إبراهيم،…١٨٦٠مثل المؤمن الذي يقرأ القرآن مثل الأترجة ريحها طيب وطعمها طيب ومثل…١٨٦١بهذا الإسناد . مثله غير أن في حديث همام بدل المنافق الفاجر .١٨٦٢الماهر بالقرآن مع السفرة الكرام البررة والذي يقرأ القرآن ويتتعتع فيه…١٨٦٣والذي يقرأ وهو يشتد عليه له أجران١٨٦٤إن الله أمرني أن أقرأ عليك١٨٦٥إن الله أمرني أن أقرأ عليك { لم يكن الذين كفروا }١٨٦٦يقول قال رسول الله صلى الله عليه وسلم لأبى بمثله .١٨٦٧اقرأ على القرآن١٨٦٨ومنجاب بن الحارث التميمي، جميعا عن علي بن مسهر، عن الأعمش، بهذا…١٨٦٩وحدثنا أبو بكر بن أبي شيبة، وأبو كريب قالا حدثنا أبو أسامة، حدثني…١٨٧٠حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن الأعمش، عن إبراهيم، عن علقمة،…١٨٧١وعلي بن خشرم، قالا أخبرنا عيسى بن يونس، ح وحدثنا أبو بكر بن أبي شيبة،…١٨٧٢أيحب أحدكم إذا رجع إلى أهله أن يجد فيه ثلاث خلفات عظام سمان١٨٧٣أيكم يحب أن يغدو كل يوم إلى بطحان أو إلى العقيق فيأتي منه بناقتين…
