كتاب صلاة المسافرين وقصرها

٣٧٧ حديثًا•صفحة ١ من ٣
١٥٧٠زوج النبي صلى الله عليه وسلم أنها قالت فرضت الصلاة ركعتين ركعتين في…١٥٧١وحرملة بن يحيى، قالا حدثنا ابن وهب، عن يونس، عن ابن شهاب، قال حدثني…١٥٧٢أن الصلاة، أول ما فرضت ركعتين فأقرت صلاة السفر وأتمت صلاة الحضر . قال…١٥٧٣صدقة تصدق الله بها عليكم فاقبلوا صدقته١٥٧٤قال قلت لعمر بن الخطاب . بمثل حديث ابن إدريس .١٥٧٥وسعيد بن منصور، وأبو الربيع، وقتيبة بن سعيد، قال يحيى أخبرنا وقال…١٥٧٦وعمرو الناقد، جميعا عن القاسم بن مالك، - قال عمرو حدثنا قاسم بن مالك…١٥٧٧وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت قتادة، يحدث…١٥٧٨بهذا الإسناد نحوه .١٥٧٩وحدثنا عبد الله بن مسلمة بن قعنب، حدثنا عيسى بن حفص بن عاصم بن عمر بن…١٥٨٠في السفر فقال صحبت رسول الله صلى الله عليه وسلم في السفر فما رأيته…١٥٨١وأبو الربيع الزهراني، وقتيبة بن سعيد، قالوا حدثنا حماد، وهو ابن زيد ح…١٥٨٢وإبراهيم بن ميسرة، سمعا أنس بن مالك، يقول صليت مع رسول الله صلى الله…١٥٨٣ومحمد بن بشار، كلاهما عن غندر، - قال أبو بكر حدثنا محمد بن جعفر، غندر…١٥٨٤ومحمد بن بشار، جميعا عن ابن مهدي، - قال زهير حدثنا عبد الرحمن بن…١٥٨٥ولم يسم شرحبيل وقال إنه أتى أرضا يقال لها دومين من حمص على رأس ثمانية…١٥٨٦حدثنا يحيى بن يحيى التميمي، أخبرنا هشيم، عن يحيى بن أبي إسحاق، عن أنس…١٥٨٧وحدثناه قتيبة، حدثنا أبو عوانة، ح وحدثناه أبو كريب، حدثنا ابن علية،…١٥٨٨وحدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا شعبة، قال حدثني يحيى بن…١٥٨٩وحدثنا ابن نمير، حدثنا أبي ح، وحدثنا أبو كريب، حدثنا أبو أسامة، جميعا…١٥٩٠وحدثني حرملة بن يحيى، حدثنا ابن وهب، أخبرني عمرو، وهو ابن الحارث عن…١٥٩١وعبد بن حميد، قالا أخبرنا عبد الرزاق، أخبرنا معمر، جميعا عن الزهري،…١٥٩٢قال صلى رسول الله صلى الله عليه وسلم بمنى ركعتين وأبو بكر بعده وعمر…١٥٩٣وعبيد الله بن سعيد، قالا حدثنا يحيى، وهو القطان ح وحدثناه أبو كريب،…١٥٩٤سمع حفص بن عاصم، عن ابن عمر، قال صلى النبي صلى الله عليه وسلم بمنى…١٥٩٥بهذا الإسناد ولم يقولا في الحديث بمنى . ولكن قالا صلى في السفر .١٥٩٦حدثنا قتيبة بن سعيد، حدثنا عبد الواحد، عن الأعمش، حدثنا إبراهيم، قال…١٥٩٧وابن، خشرم قالا أخبرنا عيسى، كلهم عن الأعمش، بهذا الإسناد نحوه .١٥٩٨وقتيبة، قال يحيى أخبرنا وقال، قتيبة حدثنا أبو الأحوص، عن أبي إسحاق،…١٥٩٩قال صليت خلف رسول الله صلى الله عليه وسلم بمنى والناس أكثر ما كانوا…١٦٠٠ألا صلوا في الرحال١٦٠١أنه نادى بالصلاة في ليلة ذات برد وريح ومطر فقال في آخر ندائه ألا صلوا…١٦٠٢أنه نادى بالصلاة بضجنان ثم ذكر بمثله وقال ألا صلوا في رحالكم . ولم…١٦٠٣ليصل من شاء منكم في رحله١٦٠٤أنه قال لمؤذنه في يوم مطير إذا قلت أشهد أن لا إله إلا الله أشهد أن…١٦٠٥بنحوه .١٦٠٦وعاصم الأحول، بهذا الإسناد ولم يذكر في حديثه يعني النبي صلى الله عليه…١٦٠٧قال أذن مؤذن ابن عباس يوم جمعة في يوم مطير . فذكر نحو حديث ابن علية…١٦٠٨أن ابن عباس، أمر مؤذنه - في حديث معمر - في يوم جمعة في يوم مطير .…١٦٠٩- قال وهيب لم يسمعه منه - قال أمر ابن عباس مؤذنه في يوم جمعة في يوم…١٦١٠أن رسول الله صلى الله عليه وسلم كان يصلي سبحته حيثما توجهت به ناقته .١٦١١أن النبي صلى الله عليه وسلم كان يصلي على راحلته حيث توجهت به .١٦١٢قال كان رسول الله صلى الله عليه وسلم يصلي وهو مقبل من مكة إلى المدينة…١٦١٣وابن أبي زائدة، ح وحدثنا ابن نمير، حدثنا أبي كلهم، عن عبد الملك، بهذا…١٦١٤قال رأيت رسول الله صلى الله عليه وسلم يصلي على حمار وهو موجه إلى خيبر…١٦١٥أنه قال كنت أسير مع ابن عمر بطريق مكة - قال سعيد - فلما خشيت الصبح…١٦١٦أنه قال كان رسول الله صلى الله عليه وسلم يصلي على راحلته حيثما توجهت…١٦١٧أنه قال كان رسول الله صلى الله عليه وسلم يوتر على راحلته .١٦١٨قال كان رسول الله صلى الله عليه وسلم يسبح على الراحلة قبل أى وجه توجه…١٦١٩وحرملة، قالا أخبرنا ابن وهب، أخبرني يونس، عن ابن شهاب، عن عبد الله بن…١٦٢٠وحدثني محمد بن حاتم، حدثنا عفان بن مسلم، حدثنا همام، حدثنا أنس بن…١٦٢١قال كان رسول الله صلى الله عليه وسلم إذا عجل به السير جمع بين المغرب…١٦٢٢أن ابن عمر، كان إذا جد به السير جمع بين المغرب والعشاء بعد أن يغيب…١٦٢٣وقتيبة بن سعيد، وأبو بكر بن أبي شيبة وعمرو الناقد كلهم عن ابن عيينة،…١٦٢٤أن أباه، قال رأيت رسول الله صلى الله عليه وسلم إذا أعجله السير في…١٦٢٥قال كان رسول الله صلى الله عليه وسلم إذا ارتحل قبل أن تزيغ الشمس أخر…١٦٢٦قال كان النبي صلى الله عليه وسلم إذا أراد أن يجمع بين الصلاتين في…١٦٢٧وعمرو بن سواد، قالا أخبرنا ابن وهب، حدثني جابر بن إسماعيل، عن عقيل،…١٦٢٨قال صلى رسول الله صلى الله عليه وسلم الظهر والعصر جميعا والمغرب…١٦٢٩وعون بن سلام، جميعا عن زهير، - قال ابن يونس حدثنا زهير، - حدثنا أبو…١٦٣٠أن رسول الله صلى الله عليه وسلم جمع بين الصلاة في سفرة سافرها في غزوة…١٦٣١حدثنا أحمد بن عبد الله بن يونس، حدثنا زهير، حدثنا أبو الزبير، عن أبي…١٦٣٢قال جمع رسول الله صلى الله عليه وسلم في غزوة تبوك بين الظهر والعصر…١٦٣٣قال جمع رسول الله صلى الله عليه وسلم بين الظهر والعصر والمغرب والعشاء…١٦٣٤وحدثنا أبو بكر بن أبي شيبة، حدثنا سفيان بن عيينة، عن عمرو، عن جابر بن…١٦٣٥أن رسول الله صلى الله عليه وسلم صلى بالمدينة سبعا وثمانيا الظهر…١٦٣٦وحدثني أبو الربيع الزهراني، حدثنا حماد، عن الزبير بن الخريت، عن عبد…١٦٣٧وحدثنا ابن أبي عمر، حدثنا وكيع، حدثنا عمران بن حدير، عن عبد الله بن…١٦٣٨ووكيع، عن الأعمش، عن عمارة، عن الأسود، عن عبد الله، قال لا يجعلن…١٦٣٩وعيسى بن يونس، ح وحدثناه علي بن خشرم، أخبرنا عيسى، جميعا عن الأعمش،…١٦٤٠وحدثنا قتيبة بن سعيد، حدثنا أبو عوانة، عن السدي، قال سألت أنسا كيف…١٦٤١وزهير بن حرب، قالا حدثنا وكيع، عن سفيان، عن السدي، عن أنس، أن النبي…١٦٤٢رب قني عذابك يوم تبعث - أو تجمع - عبادك١٦٤٣وزهير بن حرب، قالا حدثنا وكيع، عن مسعر، بهذا الإسناد ولم يذكر يقبل…١٦٤٤إذا أقيمت الصلاة فلا صلاة إلا المكتوبة١٦٤٥إذا أقيمت الصلاة فلا صلاة إلا المكتوبة١٦٤٦بهذا الإسناد مثله .١٦٤٧وحدثنا حسن الحلواني، حدثنا يزيد بن هارون، أخبرنا حماد بن زيد، عن…١٦٤٩يوشك أن يصلي أحدكم الصبح أربعا١٦٥٠أتصلي الصبح أربعا١٦٥١يا فلان بأى الصلاتين اعتددت أبصلاتك وحدك أم بصلاتك معنا١٦٥٢إذا دخل أحدكم المسجد فليقل اللهم افتح لي أبواب رحمتك . وإذا خرج فليقل…١٦٥٣وحدثنا حامد بن عمر البكراوي، حدثنا بشر بن المفضل، حدثنا عمارة بن…١٦٥٤إذا دخل أحدكم المسجد فليركع ركعتين قبل أن يجلس١٦٥٥ما منعك أن تركع ركعتين قبل أن تجلس١٦٥٦قال كان لي على النبي صلى الله عليه وسلم دين فقضاني وزادني ودخلت عليه…١٦٥٧سمع جابر بن عبد الله، يقول اشترى مني رسول الله صلى الله عليه وسلم…١٦٥٨قال خرجت مع رسول الله صلى الله عليه وسلم في غزاة فأبطأ بي جملي وأعيى…١٦٥٩أن عبد الرحمن بن عبد الله بن كعب، أخبره عن أبيه عبد الله بن كعب، وعن…١٦٦٠قال قلت لعائشة هل كان النبي صلى الله عليه وسلم يصلي الضحى قالت لا إلا…١٦٦١قال قلت لعائشة أكان النبي صلى الله عليه وسلم يصلي الضحى قالت لا إلا…١٦٦٢أنها قالت ما رأيت رسول الله صلى الله عليه وسلم يصلي سبحة الضحى قط .…١٦٦٣- يعني الرشك - حدثتني معاذة، أنها سألت عائشة - رضى الله عنها - كم كان…١٦٦٤وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن يزيد، بهذا الإسناد…١٦٦٥أن معاذة العدوية، حدثتهم عن عائشة، قالت كان رسول الله صلى الله عليه…١٦٦٦وابن، بشار جميعا عن معاذ بن هشام، قال حدثني أبي، عن قتادة، بهذا…١٦٦٧وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن عمرو بن مرة، عن…١٦٦٨ومحمد بن سلمة المرادي، قالا أخبرنا عبد الله بن وهب، أخبرني يونس، عن…١٦٦٩أن أبا مرة، مولى أم هانئ بنت أبي طالب أخبره أنه، سمع أم هانئ بنت أبي…١٦٧٠أن رسول الله صلى الله عليه وسلم صلى في بيتها عام الفتح ثماني ركعات في…١٦٧١يصبح على كل سلامى من أحدكم صدقة فكل تسبيحة صدقة وكل تحميدة صدقة وكل…١٦٧٢قال أوصاني خليلي صلى الله عليه وسلم بثلاث بصيام ثلاثة أيام من كل شهر…١٦٧٣وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن عباس الجريري،…١٦٧٤وحدثني سليمان بن معبد، حدثنا معلى بن أسد، حدثنا عبد العزيز بن مختار،…١٦٧٥ومحمد بن رافع، قالا حدثنا ابن أبي فديك، عن الضحاك بن عثمان، عن…١٦٧٦أن حفصة أم المؤمنين، أخبرته أن رسول الله صلى الله عليه وسلم كان إذا…١٦٧٧وقتيبة، وابن، رمح عن الليث بن سعد، ح وحدثني زهير بن حرب، وعبيد الله…١٦٧٨قالت كان رسول الله صلى الله عليه وسلم إذا طلع الفجر لا يصلي إلا…١٦٧٩بهذا الإسناد مثله .١٦٨٠أخبرتني حفصة، أن النبي صلى الله عليه وسلم كان إذا أضاء له الفجر صلى…١٦٨١قالت كان رسول الله صلى الله عليه وسلم يصلي ركعتى الفجر إذا سمع الأذان…١٦٨٢بهذا الإسناد وفي حديث أبي أسامة إذا طلع الفجر .١٦٨٣أن نبي الله صلى الله عليه وسلم كان يصلي ركعتين بين النداء والإقامة من…١٦٨٤أنه سمع عمرة، تحدث عن عائشة، أنها كانت تقول كان رسول الله صلى الله…١٦٨٥سمع عمرة بنت عبد الرحمن، عن عائشة، قالت كان رسول الله صلى الله عليه…١٦٨٦أن النبي صلى الله عليه وسلم لم يكن على شىء من النوافل أشد معاهدة منه…١٦٨٧وابن، نمير جميعا عن حفص بن غياث، - قال ابن نمير حدثنا حفص، - عن ابن…١٦٨٨ركعتا الفجر خير من الدنيا وما فيها١٦٨٩لهما أحب إلى من الدنيا جميعا١٦٩٠وابن أبي عمر، قالا حدثنا مروان بن معاوية، عن يزيد، - هو ابن كيسان -…١٦٩١أن ابن عباس، أخبره أن رسول الله صلى الله عليه وسلم كان يقرأ في ركعتى…١٦٩٢وحدثنا أبو بكر بن أبي شيبة، حدثنا أبو خالد الأحمر، عن عثمان بن حكيم،…١٦٩٣في هذا الإسناد . بمثل حديث مروان الفزاري .١٦٩٤من صلى اثنتى عشرة ركعة في يوم وليلة بني له بهن بيت في الجنة١٦٩٥من صلى في يوم ثنتى عشرة سجدة تطوعا بني له بيت في الجنة١٦٩٦ما من عبد مسلم يصلي لله كل يوم ثنتى عشرة ركعة تطوعا غير فريضة إلا بنى…١٦٩٧ما من عبد مسلم توضأ فأسبغ الوضوء ثم صلى لله كل يوم١٦٩٨وعبيد الله بن سعيد، قالا حدثنا يحيى، - وهو ابن سعيد - عن عبيد الله،…١٦٩٩حدثنا يحيى بن يحيى، أخبرنا هشيم، عن خالد، عن عبد الله بن شقيق، قال…١٧٠٠وأيوب، عن عبد الله بن شقيق، عن عائشة، قالت كان رسول الله صلى الله…١٧٠١وحدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، عن بديل، عن…١٧٠٢رسول الله صلى الله عليه وسلم بالليل فقالت كان يصلي ليلا طويلا قائما…١٧٠٣رسول الله صلى الله عليه وسلم فقالت كان رسول الله صلى الله عليه وسلم…١٧٠٤قالت ما رأيت رسول الله صلى الله عليه وسلم يقرأ في شىء من صلاة الليل…١٧٠٥وأبي النضر، عن أبي سلمة بن عبد الرحمن، عن عائشة، أن رسول الله صلى…١٧٠٦وإسحاق بن إبراهيم، قال أبو بكر حدثنا إسماعيل ابن علية، عن الوليد بن…١٧٠٧قال قلت لعائشة كيف كان يصنع رسول الله صلى الله عليه وسلم في الركعتين…١٧٠٨قال قلت لعائشة هل كان النبي صلى الله عليه وسلم يصلي وهو قاعد قالت نعم…١٧٠٩وحدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا كهمس، عن عبد الله بن شقيق،…١٧١٠وهارون بن عبد الله، قالا حدثنا حجاج بن محمد، قال قال ابن جريج أخبرني…١٧١١وحسن الحلواني، كلاهما عن زيد، قال حسن حدثنا زيد بن الحباب، حدثني…١٧١٢أنها قالت ما رأيت رسول الله صلى الله عليه وسلم صلى في سبحته قاعدا حتى…١٧١٣وحرملة، قالا أخبرنا ابن وهب، أخبرني يونس، ح وحدثنا إسحاق بن إبراهيم،…١٧١٤أن النبي صلى الله عليه وسلم لم يمت حتى صلى قاعدا .١٧١٥صلاة الرجل قاعدا نصف الصلاة١٧١٦ومحمد بن المثنى، وابن، بشار جميعا عن محمد بن جعفر، عن شعبة، ح وحدثنا…١٧١٧أن رسول الله صلى الله عليه وسلم كان يصلي بالليل إحدى عشرة ركعة يوتر…١٧٢٠قالت كان رسول الله صلى الله عليه وسلم يصلي من الليل ثلاث عشرة ركعة…١٧٢١بهذا الإسناد .١٧٢٢أن عائشة، أخبرته أن رسول الله صلى الله عليه وسلم كان يصلي ثلاث عشرة…
