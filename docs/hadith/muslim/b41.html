كتاب الشعر

١٠ حديثًا
٥٨٨٧قال استنشدني رسول الله صلى الله عليه وسلم . بمثل حديث إبراهيم بن…٥٨٨٨أشعر كلمة تكلمت بها العرب كلمة لبيد ألا كل شىء ما خلا الله باطل٥٨٨٩أصدق كلمة قالها شاعر كلمة لبيد ألا كل شىء ما خلا الله باطل وكاد أمية…٥٨٩٠أصدق بيت قاله الشاعر ألا كل شىء ما خلا الله باطل وكاد ابن أبي الصلت…٥٨٩١أصدق بيت قالته الشعراء ألا كل شىء ما خلا الله باطل٥٨٩٢إن أصدق كلمة قالها شاعر كلمة لبيد ألا كل شىء ما خلا الله باطل٥٨٩٣لأن يمتلئ جوف الرجل قيحا يريه خير من أن يمتلئ شعرا٥٨٩٤لأن يمتلئ جوف أحدكم قيحا يريه خير من أن يمتلئ شعرا٥٨٩٥خذوا الشيطان أو أمسكوا الشيطان لأن يمتلئ جوف رجل قيحا خير له من أن…٥٨٩٦من لعب بالنردشير فكأنما صبغ يده في لحم خنزير ودمه
