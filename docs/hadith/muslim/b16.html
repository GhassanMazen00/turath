١٦٨ حديثًا•صفحة ١ من ٢
٣٨٨حديث ٣٨٨٣٣٩٨يا معشر الشباب من استطاع منكم الباءة فليتزوج فإنه أغض للبصر وأحصن…٣٣٩٩قال إني لأمشي مع عبد الله بن مسعود بمنى إذ لقيه عثمان بن عفان فقال…٣٤٠٠يا معشر الشباب من استطاع منكم الباءة فليتزوج فإنه أغض للبصر وأحصن…٣٤٠١يومئذ فذكر حديثا رئيت أنه حدث به، من أجلي قال قال رسول الله صلى الله…٣٤٠٢بمثل حديثهم ولم يذكر فلم ألبث حتى تزوجت .٣٤٠٣ما بال أقوام قالوا كذا وكذا لكني أصلي وأنام وأصوم وأفطر وأتزوج النساء…٣٤٠٤وحدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن المبارك، ح وحدثنا أبو…٣٤٠٥وحدثني أبو عمران، محمد بن جعفر بن زياد حدثنا إبراهيم بن سعد، عن ابن،…٣٤٠٦أنه سمع سعد بن أبي وقاص، يقول أراد عثمان بن مظعون أن يتبتل، فنهاه…٣٤٠٧إن المرأة تقبل في صورة شيطان وتدبر في صورة شيطان فإذا أبصر أحدكم…٣٤٠٨أن النبي صلى الله عليه وسلم رأى امرأة . فذكر بمثله غير أنه قال فأتى…٣٤٠٩إذا أحدكم أعجبته المرأة فوقعت في قلبه فليعمد إلى امرأته فليواقعها فإن…٣٤١٠وابن، بشر عن إسماعيل، عن قيس، قال سمعت عبد الله، يقول كنا نغزو مع…٣٤١١وحدثنا عثمان بن أبي شيبة، حدثنا جرير، عن إسماعيل بن أبي خالد، بهذا…٣٤١٢وحدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن إسماعيل، بهذا الإسناد قال…٣٤١٣وسلمة بن الأكوع، قالا خرج علينا منادي رسول الله صلى الله عليه وسلم…٣٤١٤وحدثني أمية بن بسطام العيشي، حدثنا يزيد، - يعني ابن زريع - حدثنا روح،…٣٤١٥وحدثنا الحسن الحلواني، حدثنا عبد الرزاق، أخبرنا ابن جريج، قال قال…٣٤١٦حدثني محمد بن رافع، حدثنا عبد الرزاق، أخبرنا ابن جريج، أخبرني أبو…٣٤١٧حدثنا حامد بن عمر البكراوي، حدثنا عبد الواحد، - يعني ابن زياد - عن…٣٤١٨قال رخص رسول الله صلى الله عليه وسلم عام أوطاس في المتعة ثلاثا ثم نهى…٣٤١٩من كان عنده شىء من هذه النساء التي يتمتع فليخل سبيلها٣٤٢٠أن أباه، غزا مع رسول الله صلى الله عليه وسلم فتح مكة قال فأقمنا بها…٣٤٢١وحدثني أحمد بن سعيد بن صخر الدارمي، حدثنا أبو النعمان، حدثنا وهيب،…٣٤٢٢يا أيها الناس إني قد كنت أذنت لكم في الاستمتاع من النساء وإن الله قد…٣٤٢٣بهذا الإسناد قال رأيت رسول الله صلى الله عليه وسلم قائما بين الركن…٣٤٢٤حدثنا إسحاق بن إبراهيم، أخبرنا يحيى بن آدم، حدثنا إبراهيم بن سعد، عن…٣٤٢٥وحدثنا يحيى بن يحيى، أخبرنا عبد العزيز بن الربيع بن سبرة بن معبد، قال…٣٤٢٦وابن، نمير قالا حدثنا سفيان بن عيينة، عن الزهري، عن الربيع بن سبرة،…٣٤٢٧وحدثنا أبو بكر بن أبي شيبة، حدثنا ابن علية، عن معمر، عن الزهري، عن…٣٤٢٨وعبد بن حميد، عن يعقوب بن إبراهيم بن سعد، حدثنا أبي، عن صالح، أخبرنا…٣٤٢٩أن عبد الله بن الزبير، قام بمكة فقال إن ناسا - أعمى الله قلوبهم كما…٣٤٣٠ألا إنها حرام من يومكم هذا إلى يوم القيامة ومن كان أعطى شيئا فلا يأخذه٣٤٣١والحسن، ابنى محمد بن علي عن أبيهما، عن علي بن أبي طالب، أن رسول الله…٣٤٣٢بهذا الإسناد وقال سمع علي بن أبي طالب، يقول لفلان إنك رجل تائه نهانا…٣٤٣٣وابن، نمير وزهير بن حرب جميعا عن ابن عيينة، - قال زهير حدثنا سفيان بن…٣٤٣٤وعبد الله، ابنى محمد بن علي عن أبيهما، عن علي، أنه سمع ابن عباس، يلين…٣٤٣٥وحرملة بن يحيى، قالا أخبرنا ابن وهب، أخبرني يونس، عن ابن شهاب، عن…٣٤٣٦لا يجمع بين المرأة وعمتها ولا بين المرأة وخالتها٣٤٣٧وحدثنا محمد بن رمح بن المهاجر، أخبرنا الليث، عن يزيد بن أبي حبيب، عن…٣٤٣٨لا تنكح العمة على بنت الأخ ولا ابنة الأخت على الخالة٣٤٣٩أنه سمع أبا هريرة، يقول نهى رسول الله صلى الله عليه وسلم أن يجمع…٣٤٤٠لا تنكح المرأة على عمتها ولا على خالتها٣٤٤١أنه سمع أبا هريرة، يقول قال رسول الله صلى الله عليه وسلم بمثله .٣٤٤٢لا يخطب الرجل على خطبة أخيه ولا يسوم على سوم أخيه ولا تنكح المرأة على…٣٤٤٣قال نهى رسول الله صلى الله عليه وسلم أن تنكح المرأة على عمتها أو…٣٤٤٤وابن، بشار وأبو بكر بن نافع - واللفظ لابن المثنى وابن نافع - قالوا…٣٤٤٥بهذا الإسناد مثله .٣٤٤٦لا ينكح المحرم ولا ينكح ولا يخطب٣٤٤٧إن المحرم لا ينكح ولا ينكح٣٤٤٨لا ينكح المحرم ولا ينكح ولا يخطب٣٤٤٩المحرم لا ينكح ولا يخطب٣٤٥٠أن عمر بن عبيد الله بن معمر، أراد أن ينكح، ابنه طلحة بنت شيبة بن جبير…٣٤٥١وابن، نمير وإسحاق الحنظلي جميعا عن ابن عيينة، - قال ابن نمير حدثنا…٣٤٥٢أنه قال تزوج رسول الله صلى الله عليه وسلم ميمونة وهو محرم .٣٤٥٣حدثتني ميمونة بنت الحارث، أن رسول الله صلى الله عليه وسلم تزوجها وهو…٣٤٥٤لا يبع بعضكم على بيع بعض ولا يخطب بعضكم على خطبة بعض٣٤٥٥لا يبع الرجل على بيع أخيه ولا يخطب على خطبة أخيه إلا أن يأذن له٣٤٥٦بهذا الإسناد٣٤٥٧بهذا الإسناد .٣٤٥٨وزهير بن حرب، وابن أبي عمر، قال زهير حدثنا سفيان، بن عيينة عن الزهري،…٣٤٥٩لا تناجشوا ولا يبع المرء على بيع أخيه ولا يبع حاضر لباد ولا يخطب…٣٤٦٠ولا يزد الرجل على بيع أخيه٣٤٦١لا يسم المسلم على سوم أخيه ولا يخطب على خطبته٣٤٦٢على سوم أخيه وخطبة أخيه٣٤٦٤المؤمن أخو المؤمن فلا يحل للمؤمن أن يبتاع على بيع أخيه ولا يخطب على…٣٤٦٥حدثنا يحيى بن يحيى، قال قرأت على مالك عن نافع، عن ابن عمر، أن رسول…٣٤٦٦ومحمد بن المثنى، وعبيد الله بن سعيد، قالوا حدثنا يحيى، عن عبيد الله،…٣٤٦٧وحدثنا يحيى بن يحيى، أخبرنا حماد بن زيد، عن عبد الرحمن السراج، عن…٣٤٦٨لا شغار في الإسلام٣٤٦٩حدثنا أبو بكر بن أبي شيبة، حدثنا ابن نمير، وأبو أسامة عن عبيد الله،…٣٤٧٠- وهو ابن عمر - بهذا الإسناد ولم يذكر زيادة ابن نمير .٣٤٧١ومحمد بن رافع، عن عبد الرزاق، أخبرنا ابن جريج، أخبرني أبو الزبير، أنه…٣٤٧٢إن أحق الشرط أن يوفى به ما استحللتم به الفروج٣٤٧٣لا تنكح الأيم حتى تستأمر ولا تنكح البكر حتى تستأذن٣٤٧٤. بمثل معنى حديث هشام وإسناده . واتفق لفظ حديث هشام وشيبان ومعاوية بن…٣٤٧٥ومحمد بن رافع، جميعا عن عبد الرزاق، - واللفظ لابن رافع - حدثنا عبد…٣٤٧٦الأيم أحق بنفسها من وليها والبكر تستأذن في نفسها وإذنها صماتها٣٤٧٧الثيب أحق بنفسها من وليها والبكر تستأمر وإذنها سكوتها٣٤٧٨الثيب أحق بنفسها من وليها والبكر يستأذنها أبوها في نفسها وإذنها صماتها٣٤٧٩حدثنا أبو كريب، محمد بن العلاء حدثنا أبو أسامة، ح وحدثنا أبو بكر بن…٣٤٨٠وحدثنا يحيى بن يحيى، أخبرنا أبو معاوية، عن هشام بن عروة، ح وحدثنا ابن…٣٤٨١أن النبي صلى الله عليه وسلم تزوجها وهى بنت سبع سنين وزفت إليه وهي بنت…٣٤٨٢وإسحاق بن إبراهيم، وأبو بكر بن أبي شيبة وأبو كريب قال يحيى وإسحاق…٣٤٨٣وزهير بن حرب، - واللفظ لزهير - قالا حدثنا وكيع، حدثنا سفيان، عن…٣٤٨٤بهذا الإسناد ولم يذكر فعل عائشة .٣٤٨٥هريرة قال كنت عند النبي صلى الله عليه وسلم فأتاه رجل فأخبره أنه تزوج…٣٤٨٦هل نظرت إليها فإن في عيون الأنصار شيئا٣٤٨٧فهل عندك من شىء٣٤٨٨انطلق فقد زوجتكها فعلمها من القرآن٣٤٨٩أنه قال سألت عائشة زوج النبي صلى الله عليه وسلم كم كان صداق رسول الله…٣٤٩٠وأبو الربيع، سليمان بن داود العتكي وقتيبة بن سعيد واللفظ ليحيى قال…٣٤٩١أن عبد الرحمن بن عوف، تزوج على عهد رسول الله صلى الله عليه وسلم على…٣٤٩٢وحميد، عن أنس، أن عبد الرحمن بن عوف، تزوج امرأة على وزن نواة من ذهب…٣٤٩٣وهارون، بن عبد الله قالا حدثنا وهب بن جرير، ح وحدثنا أحمد بن خراش،…٣٤٩٤ومحمد بن قدامة، قالا أخبرنا النضر بن شميل، حدثنا شعبة، حدثنا عبد…٣٤٩٥أن عبد الرحمن، تزوج امرأة على وزن نواة من ذهب .٣٤٩٦بهذا الإسناد غير أنه قال فقال رجل من ولد عبد الرحمن بن عوف من ذهب٣٤٩٧الله أكبر خربت خيبر إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٣٤٩٨وعبد، العزيز بن صهيب عن أنس، ح وحدثناه قتيبة بن سعيد، حدثنا حماد، -…٣٥٠٠خربت خيبر إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٣٥٠١وحدثنا أبو بكر بن أبي شيبة، حدثنا شبابة، حدثنا سليمان، عن ثابث، عن…٣٥٠٢سلام عليكم كيف أنتم يا أهل البيت٣٥٠٣حدثنا محمد بن حاتم بن ميمون، حدثنا بهز، ح وحدثني محمد بن رافع، حدثنا…٣٥٠٤- قال ما رأيت رسول الله صلى الله عليه وسلم أولم على امرأة - وقال أبو…٣٥٠٥ومحمد بن بشار، قالا حدثنا محمد، - وهو ابن جعفر - حدثنا شعبة، عن عبد…٣٥٠٦وعاصم بن النضر التيمي، ومحمد بن عبد الأعلى، كلهم عن معتمر، - واللفظ…٣٥٠٧بالحجاب لقد كان أبى بن كعب يسألني عنه . قال أنس أصبح رسول الله صلى…٣٥٠٨ضعه - ثم قال - اذهب فادع لي فلانا وفلانا وفلانا ومن لقيت٣٥٠٩إذا دعي أحدكم إلى الوليمة فليأتها٣٥١٠إذا دعي أحدكم إلى الوليمة فليجب٣٥١١إذا دعي أحدكم إلى وليمة عرس فليجب٣٥١٢ائتوا الدعوة إذا دعيتم٣٥١٣إذا دعا أحدكم أخاه فليجب عرسا كان أو نحوه٣٥١٤من دعي إلى عرس أو نحوه فليجب٣٥١٥ائتوا الدعوة إذا دعيتم٣٥١٦أجيبوا هذه الدعوة إذا دعيتم لها٣٥١٧إذا دعيتم إلى كراع فأجيبوا٣٥١٨إذا دعي أحدكم إلى طعام فليجب فإن شاء طعم وإن شاء ترك٣٥١٩بهذا الإسناد بمثله٣٥٢٠إذا دعي أحدكم فليجب فإن كان صائما فليصل وإن كان مفطرا فليطعم٣٥٢١أنه كان يقول بئس الطعام طعام الوليمة يدعى إليه الأغنياء ويترك…٣٥٢٢وحدثنا ابن أبي عمر، حدثنا سفيان، قال قلت للزهري يا أبا بكر كيف هذا…٣٥٢٣وعبد بن حميد، عن عبد الرزاق، أخبرنا معمر، عن الزهري، عن سعيد بن…٣٥٢٤شر الطعام طعام الوليمة يمنعها من يأتيها ويدعى إليها من يأباها ومن لم…٣٥٢٦أتريدين أن ترجعي إلى رفاعة لا حتى تذوقي عسيلته ويذوق عسيلتك٣٥٢٧لعلك تريدين أن ترجعي إلى رفاعة لا حتى يذوق عسيلتك وتذوقي عسيلته٣٥٢٨أن رفاعة القرظي، طلق امرأته فتزوجها عبد الرحمن بن الزبير فجاءت النبي…٣٥٢٩لا حتى يذوق عسيلتها٣٥٣٠بهذا الإسناد .٣٥٣١لا حتى يذوق الآخر من عسيلتها ما ذاق الأول٣٥٣٢وحدثناه محمد بن عبد الله بن نمير، حدثنا أبي ح، وحدثناه محمد بن…٣٥٣٣لو أن أحدهم إذا أراد أن يأتي أهله قال باسم الله اللهم جنبنا الشيطان…٣٥٣٤وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، ح وحدثنا ابن نمير،…٣٥٣٥حدثنا قتيبة بن سعيد، وأبو بكر بن أبي شيبة وعمرو الناقد - واللفظ لأبي…٣٥٣٦أن يهود، كانت تقول إذا أتيت المرأة من دبرها في قبلها ثم حملت كان…٣٥٣٧وحدثناه قتيبة بن سعيد، حدثنا أبو عوانة، ح وحدثنا عبد الوارث بن عبد…٣٥٣٨إذا باتت المرأة هاجرة فراش زوجها لعنتها الملائكة حتى تصبح٣٥٣٩بهذا الإسناد وقال " حتى ترجع " .٣٥٤٠والذي نفسي بيده ما من رجل يدعو امرأته إلى فراشها فتأبى عليه إلا كان…٣٥٤١إذا دعا الرجل امرأته إلى فراشه فلم تأته فبات غضبان عليها لعنتها…٣٥٤٢إن من أشر الناس عند الله منزلة يوم القيامة الرجل يفضي إلى امرأته…٣٥٤٣إن من أعظم الأمانة عند الله يوم القيامة الرجل يفضي إلى امرأته وتفضي…٣٥٤٤لا عليكم أن لا تفعلوا ما كتب الله خلق نسمة هي كائنة إلى يوم القيامة…٣٥٤٥فإن الله كتب من هو خالق إلى يوم القيامة٣٥٤٦وإنكم لتفعلون وإنكم لتفعلون وإنكم لتفعلون ما من نسمة كائنة إلى يوم…٣٥٤٧لا عليكم أن لا تفعلوا فإنما هو القدر٣٥٤٨لا عليكم أن لا تفعلوا ذاكم فإنما هو القدر٣٥٤٩لا عليكم أن لا تفعلوا ذاكم فإنما هو القدر
