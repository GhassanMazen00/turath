كتاب الذكر والدعاء والتوبة والاستغفار

١٢٤ حديثًا
٦٨٠٥يقول الله عز وجل أنا عند ظن عبدي بي وأنا معه حين يذكرني إن ذكرني في…٦٨٠٦وإن تقرب إلى ذراعا تقربت منه باعا٦٨٠٧إن الله قال إذا تلقاني عبدي بشبر تلقيته بذراع وإذا تلقاني بذراع…٦٨٠٨سيروا هذا جمدان سبق المفردون٦٨٠٩لله تسعة وتسعون اسما من حفظها دخل الجنة وإن الله وتر يحب الوتر٦٨١٠إن لله تسعة وتسعين اسما مائة إلا واحدا من أحصاها دخل الجنة٦٨١١إذا دعا أحدكم فليعزم في الدعاء ولا يقل اللهم إن شئت فأعطني فإن الله…٦٨١٢إذا دعا أحدكم فلا يقل اللهم اغفر لي إن شئت ولكن ليعزم المسألة وليعظم…٦٨١٣لا يقولن أحدكم اللهم اغفر لي إن شئت اللهم ارحمني إن شئت . ليعزم في…٦٨١٤لا يتمنين أحدكم الموت لضر نزل به فإن كان لا بد متمنيا فليقل اللهم…٦٨١٥حدثنا ابن أبي خلف، حدثنا روح، حدثنا شعبة، ح وحدثني زهير بن حرب، حدثنا…٦٨١٦لا يتمنين أحدكم الموت٦٨١٧حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن إدريس، عن إسماعيل بن…٦٨١٨وجرير بن عبد الحميد، ووكيع ح وحدثنا ابن نمير، حدثنا أبي ح، وحدثنا…٦٨١٩لا يتمنى أحدكم الموت ولا يدع به من قبل أن يأتيه إنه إذا مات أحدكم…٦٨٢٠من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه٦٨٢١وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن قتادة، قال سمعت…٦٨٢٢من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه٦٨٢٣بهذا الإسناد .٦٨٢٤من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه…٦٨٢٥أن عائشة، أخبرته أن رسول الله صلى الله عليه وسلم قال بمثله .٦٨٢٦من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه٦٨٢٧بهذا الإسناد نحو حديث عبثر .٦٨٢٨من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه٦٨٢٩إن الله يقول أنا عند ظن عبدي بي وأنا معه إذا دعاني٦٨٣٠قال الله عز وجل إذا تقرب عبدي مني شبرا تقربت منه ذراعا وإذا تقرب مني…٦٨٣١إذا أتاني يمشي أتيته هرولة٦٨٣٢يقول الله عز وجل أنا عند ظن عبدي وأنا معه حين يذكرني فإن ذكرني في…٦٨٣٣يقول الله عز وجل من جاء بالحسنة فله عشر أمثالها وأزيد ومن جاء بالسيئة…٦٨٣٤فله عشر أمثالها أو أزيد٦٨٣٥هل كنت تدعو بشىء أو تسأله إياه٦٨٣٦حدثناه عاصم بن النضر التيمي، حدثنا خالد بن الحارث، حدثنا حميد، بهذا…٦٨٣٧لا طاقة لك بعذاب الله٦٨٣٨وابن، بشار قالا حدثنا سالم بن نوح العطار، عن سعيد، بن أبي عروبة عن…٦٨٣٩إن لله تبارك وتعالى ملائكة سيارة فضلا يتبعون مجالس الذكر فإذا وجدوا…٦٨٤٠اللهم آتنا في الدنيا حسنة وفي الآخرة حسنة وقنا عذاب النار٦٨٤١ربنا آتنا في الدنيا حسنة وفي الآخرة حسنة وقنا عذاب النار٦٨٤٢من قال لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل…٦٨٤٣من قال حين يصبح وحين يمسي سبحان الله وبحمده مائة مرة . لم يأت أحد يوم…٦٨٤٦كلمتان خفيفتان على اللسان ثقيلتان في الميزان حبيبتان إلى الرحمن سبحان…٦٨٤٧لأن أقول سبحان الله والحمد لله ولا إله إلا الله والله أكبر أحب إلى…٦٨٤٨قل لا إله إلا الله وحده لا شريك له الله أكبر كبيرا والحمد لله كثيرا…٦٨٤٩اللهم اغفر لي وارحمني واهدني وارزقني٦٨٥٠اللهم اغفر لي وارحمني واهدني وعافني وارزقني٦٨٥١قل اللهم اغفر لي وارحمني وعافني وارزقني٦٨٥٢أيعجز أحدكم أن يكسب كل يوم ألف حسنة٦٨٥٣من نفس عن مؤمن كربة من كرب الدنيا نفس الله عنه كربة من كرب يوم…٦٨٥٤قال قال رسول الله صلى الله عليه وسلم بمثل حديث أبي معاوية غير أن حديث…٦٨٥٧حدثنا أبو بكر بن أبي شيبة، حدثنا مرحوم بن عبد العزيز، عن أبي نعامة…٦٨٥٨إنه ليغان على قلبي وإني لأستغفر الله في اليوم مائة مرة٦٨٥٩يا أيها الناس توبوا إلى الله فإني أتوب في اليوم إليه مائة مرة٦٨٦٠في هذا الإسناد .٦٨٦١من تاب قبل أن تطلع الشمس من مغربها تاب الله عليه٦٨٦٢أيها الناس اربعوا على أنفسكم إنكم ليس تدعون أصم ولا غائبا إنكم تدعون…٦٨٦٣وإسحاق بن إبراهيم، وأبو سعيد الأشج جميعا عن حفص بن، غياث عن عاصم،…٦٨٦٤إنكم لا تنادون أصم ولا غائبا٦٨٦٥قال بينما رسول الله صلى الله عليه وسلم فذكر نحوه .٦٨٦٦وأبو الربيع، قالا حدثنا حماد بن زيد، عن أيوب، عن أبي، عثمان عن أبي…٦٨٦٧والذي تدعونه أقرب إلى أحدكم من عنق راحلة أحدكم٦٨٦٨ألا أدلك على كلمة من كنوز الجنة - أو قال - على كنز من كنوز الجنة٦٨٦٩قل اللهم إني ظلمت نفسي ظلما كبيرا - وقال قتيبة كثيرا - ولا يغفر…٦٨٧٠أنه سمع عبد الله بن عمرو بن العاص، يقول إن أبا بكر الصديق قال لرسول…٦٨٧١اللهم فإني أعوذ بك من فتنة النار وعذاب النار وفتنة القبر وعذاب القبر…٦٨٧٢ووكيع، عن هشام، بهذا الإسناد .٦٨٧٣اللهم إني أعوذ بك من العجز والكسل والجبن والهرم والبخل وأعوذ بك من…٦٨٧٤ومن فتنة المحيا والممات٦٨٧٥حدثنا أبو كريب، محمد بن العلاء أخبرنا ابن مبارك، عن سليمان التيمي، عن…٦٨٧٦اللهم إني أعوذ بك من البخل والكسل وأرذل العمر وعذاب القبر وفتنة…٦٨٧٧وزهير بن حرب، قالا حدثنا سفيان بن عيينة، حدثني سمى، عن أبي صالح، عن…٦٨٧٨من نزل منزلا ثم قال أعوذ بكلمات الله التامات من شر ما خلق . لم يضره…٦٨٨٠إذا نزل أحدكم منزلا فليقل أعوذ بكلمات الله التامات من شر ما خلق .…٦٨٨١أنه ذكر له أن أبا صالح، مولى غطفان أخبره أنه، سمع أبا هريرة، يقول قال…٦٨٨٢إذا أخذت مضجعك فتوضأ وضوءك للصلاة ثم اضطجع على شقك الأيمن ثم قل اللهم…٦٨٨٣وإن أصبح أصاب خيرا٦٨٨٤اللهم أسلمت نفسي إليك ووجهت وجهي إليك وألجأت ظهري إليك وفوضت أمري…٦٨٨٥يا فلان إذا أويت إلى فراشك٦٨٨٦وإن أصبحت أصبت خيرا٦٨٨٧اللهم باسمك أحيا وباسمك أموت٦٨٨٨اللهم خلقت نفسي وأنت توفاها لك مماتها ومحياها إن أحييتها فاحفظها وإن…٦٨٨٩اللهم رب السموات ورب الأرض ورب العرش العظيم ربنا ورب كل شىء فالق الحب…٦٨٩٠من شر كل دابة أنت آخذ بناصيتها٦٨٩١قولي اللهم رب السموات السبع٦٨٩٢إذا أوى أحدكم إلى فراشه فليأخذ داخلة إزاره فلينفض بها فراشه وليسم…٦٨٩٣ثم ليقل باسمك ربي وضعت جنبي فإن أحييت نفسي فارحمها٦٨٩٤الحمد لله الذي أطعمنا وسقانا وكفانا وآوانا فكم ممن لا كافي له ولا مئوي٦٨٩٥اللهم إني أعوذ بك من شر ما عملت ومن شر ما لم أعمل٦٨٩٦اللهم إني أعوذ بك من شر ما عملت وشر ما لم أعمل٦٨٩٧ومن شر ما لم أعمل٦٨٩٨اللهم إني أعوذ بك من شر ما عملت وشر ما لم أعمل٦٨٩٩اللهم لك أسلمت وبك آمنت وعليك توكلت وإليك أنبت وبك خاصمت اللهم إني…٦٩٠٠سمع سامع بحمد الله وحسن بلائه علينا ربنا صاحبنا وأفضل علينا عائذا…٦٩٠١اللهم اغفر لي خطيئتي وجهلي وإسرافي في أمري وما أنت أعلم به مني اللهم…٦٩٠٢في هذا الإسناد .٦٩٠٣اللهم أصلح لي ديني الذي هو عصمة أمري وأصلح لي دنياى التي فيها معاشي…٦٩٠٤اللهم إني أسألك الهدى والتقى والعفاف والغنى٦٩٠٥وابن، بشار قالا حدثنا عبد الرحمن، عن سفيان، عن أبي، إسحاق بهذا…٦٩٠٦اللهم إني أعوذ بك من العجز والكسل والجبن والبخل والهرم وعذاب القبر…٦٩٠٧أمسينا وأمسى الملك لله والحمد لله لا إله إلا الله وحده لا شريك له٦٩٠٨أمسينا وأمسى الملك لله والحمد لله لا إله إلا الله وحده لا شريك له٦٩٠٩أمسينا وأمسى الملك لله والحمد لله لا إله إلا الله وحده لا شريك له…٦٩١٠لا إله إلا الله وحده أعز جنده ونصر عبده وغلب الأحزاب وحده فلا شىء بعده٦٩١١قل اللهم اهدني وسددني واذكر بالهدى هدايتك الطريق والسداد سداد السهم٦٩١٢قل اللهم إني أسألك الهدى والسداد٦٩١٣ما زلت على الحال التي فارقتك عليها٦٩١٤سبحان الله عدد خلقه سبحان الله رضا نفسه سبحان الله زنة عرشه سبحان…٦٩١٥ومحمد بن بشار، - واللفظ لابن المثنى - قالا حدثنا محمد بن جعفر، حدثنا…٦٩١٦أخذتما مضجعكما من الليل٦٩١٧وعبيد بن يعيش، عن عبد الله بن نمير، حدثنا عبد الملك، عن عطاء بن أبي…٦٩٢٠إذا سمعتم صياح الديكة فاسألوا الله من فضله فإنها رأت ملكا وإذا سمعتم…٦٩٢١لا إله إلا الله العظيم الحليم لا إله إلا الله رب العرش العظيم لا إله…٦٩٢٢بهذا الإسناد وحديث معاذ بن هشام أتم .٦٩٢٣رب السموات والأرض٦٩٢٤لا إله إلا الله رب العرش الكريم٦٩٢٥ما اصطفى الله لملائكته أو لعباده سبحان الله وبحمده٦٩٢٦ألا أخبرك بأحب الكلام إلى الله٦٩٢٧ما من عبد مسلم يدعو لأخيه بظهر الغيب إلا قال الملك ولك بمثل٦٩٢٨من دعا لأخيه بظهر الغيب قال الملك الموكل به آمين ولك بمثل٦٩٣٠دعوة المرء المسلم لأخيه بظهر الغيب مستجابة عند رأسه ملك موكل كلما دعا…٦٩٣١.٦٩٣٢إن الله ليرضى عن العبد أن يأكل الأكلة فيحمده عليها أو يشرب الشربة…٦٩٣٣بهذا الإسناد .٦٩٣٤يستجاب لأحدكم ما لم يعجل فيقول قد دعوت فلا أو فلم يستجب لي٦٩٣٥يستجاب لأحدكم ما لم يعجل فيقول قد دعوت ربي فلم يستجب لي٦٩٣٦لا يزال يستجاب للعبد ما لم يدع بإثم أو قطيعة رحم ما لم يستعجل
