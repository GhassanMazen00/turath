٧٩ حديثًا
٧٤١٧الدنيا سجن المؤمن وجنة الكافر٧٤١٨أيكم يحب أن هذا له بدرهم٧٤١٩وإبراهيم بن محمد بن عرعرة السامي، قالا حدثنا عبد الوهاب، - يعنيان…٧٤٢٠يقول ابن آدم مالي مالي - قال - وهل لك يا ابن آدم من مالك إلا ما أكلت…٧٤٢١وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، وقالا، جميعا حدثنا…٧٤٢٢يقول العبد مالي مالي إنما له من ماله ثلاث ما أكل فأفنى أو لبس فأبلى…٧٤٢٣بهذا الإسناد مثله .٧٤٢٤يتبع الميت ثلاثة فيرجع اثنان ويبقى واحد يتبعه أهله وماله وعمله فيرجع…٧٤٢٥أظنكم سمعتم أن أبا عبيدة قدم بشىء من البحرين٧٤٢٦وتلهيكم كما ألهتهم٧٤٢٧إذا فتحت عليكم فارس والروم أى قوم أنتم٧٤٢٨إذا نظر أحدكم إلى من فضل عليه في المال والخلق فلينظر إلى من هو أسفل…٧٤٢٩حدثنا محمد بن رافع، حدثنا عبد الرزاق، حدثنا معمر، عن همام بن منبه، عن…٧٤٣٠انظروا إلى من أسفل منكم ولا تنظروا إلى من هو فوقكم فهو أجدر أن لا…٧٤٣١إن ثلاثة في بني إسرائيل أبرص وأقرع وأعمى فأراد الله أن يبتليهم فبعث…٧٤٣٢إن الله يحب العبد التقي الغني الخفي٧٤٣٣حدثنا يحيى بن حبيب الحارثي، حدثنا المعتمر، قال سمعت إسماعيل، عن قيس،…٧٤٣٤وحدثناه يحيى بن يحيى، أخبرنا وكيع، عن إسماعيل بن أبي خالد، بهذا…٧٤٣٥حدثنا شيبان بن فروخ، حدثنا سليمان بن المغيرة، حدثنا حميد بن هلال، عن…٧٤٣٦وقد أدرك الجاهلية قال خطب عتبة بن غزوان وكان أميرا على البصرة . فذكر…٧٤٣٧وحدثنا أبو كريب، محمد بن العلاء حدثنا وكيع، عن قرة بن خالد، عن حميد،…٧٤٣٨هل تضارون في رؤية الشمس في الظهيرة ليست في سحابة٧٤٣٩هل تدرون مم أضحك٧٤٤٠اللهم اجعل رزق آل محمد قوتا٧٤٤١اللهم اجعل رزق آل محمد قوتا٧٤٤٢بن القعقاع بهذا الإسناد وقال " كفافا " .٧٤٤٣وإسحاق بن إبراهيم، قال إسحاق أخبرنا وقال، زهير حدثنا جرير، عن منصور،…٧٤٤٤قالت ما شبع رسول الله صلى الله عليه وسلم ثلاثة أيام تباعا من خبز بر…٧٤٤٥ومحمد بن بشار، قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي إسحاق، قال…٧٤٤٦قالت ما شبع آل محمد صلى الله عليه وسلم من خبز بر فوق ثلاث .٧٤٦٤لا تدخلوا على هؤلاء القوم المعذبين إلا أن تكونوا باكين فإن لم تكونوا…٧٤٦٥لا تدخلوا مساكن الذين ظلموا أنفسهم إلا أن تكونوا باكين حذرا أن يصيبكم…٧٤٦٦أن عبد الله بن عمر، أخبره أن الناس نزلوا مع رسول الله صلى الله عليه…٧٤٦٧بهذا الإسناد . مثله غير أنه قال فاستقوا من بئارها واعتجنوا به .٧٤٦٨الساعي على الأرملة والمسكين كالمجاهد في سبيل الله - وأحسبه قال -…٧٤٦٩كافل اليتيم له أو لغيره أنا وهو كهاتين في الجنة٧٤٧٠من بنى مسجدا - قال بكير حسبت أنه قال - يبتغي به وجه الله بنى الله له…٧٤٧١من بنى مسجدا لله بنى الله له في الجنة مثله٧٤٧٢بنى الله له بيتا في الجنة٧٤٧٣بينا رجل بفلاة من الأرض فسمع صوتا في سحابة اسق حديقة فلان . فتنحى ذلك…٧٤٧٤وأجعل ثلثه في المساكين والسائلين وابن السبيل٧٤٧٥قال الله تبارك وتعالى أنا أغنى الشركاء عن الشرك من عمل عملا أشرك فيه…٧٤٧٦من سمع سمع الله به ومن راءى راءى الله به٧٤٧٧من يسمع يسمع الله به ومن يرائي يرائي الله به٧٤٧٨بهذا الإسناد وزاد ولم أسمع أحدا غيره يقول قال رسول الله صلى الله عليه…٧٤٧٩حدثنا سعيد بن عمرو الأشعثي، أخبرنا سفيان، عن الوليد بن حرب، - قال…٧٤٨٠بهذا الإسناد .٧٤٨١إن العبد ليتكلم بالكلمة ينزل بها في النار أبعد ما بين المشرق والمغرب٧٤٨٢إن العبد ليتكلم بالكلمة ما يتبين ما فيها يهوي بها في النار أبعد ما…٧٤٨٣يؤتى بالرجل يوم القيامة فيلقى في النار فتندلق أقتاب بطنه فيدور بها…٧٤٨٤حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن الأعمش، عن أبي وائل، قال كنا…٧٤٨٥كل أمتي معافاة إلا المجاهرين وإن من الإجهار أن يعمل العبد بالليل عملا…٧٤٨٦إن هذا حمد الله وإنك لم تحمد الله٧٤٨٧وحدثنا أبو كريب، حدثنا أبو خالد، - يعني الأحمر - عن سليمان التيمي، عن…٧٤٨٨إذا عطس أحدكم فحمد الله فشمتوه فإن لم يحمد الله فلا تشمتوه٧٤٨٩أن أباه، حدثه أنه، سمع النبي صلى الله عليه وسلم وعطس رجل عنده فقال له…٧٤٩٠التثاؤب من الشيطان فإذا تثاءب أحدكم فليكظم ما استطاع٧٤٩١إذا تثاوب أحدكم فليمسك بيده على فيه فإن الشيطان يدخل٧٤٩٢إذا تثاوب أحدكم فليمسك بيده فإن الشيطان يدخل٧٤٩٣إذا تثاوب أحدكم في الصلاة فليكظم ما استطاع فإن الشيطان يدخل٧٤٩٤قال قال رسول الله صلى الله عليه وسلم بمثل حديث بشر وعبد العزيز .٧٤٩٥خلقت الملائكة من نور وخلق الجان من مارج من نار وخلق آدم مما وصف لكم٧٤٩٦فقدت أمة من بني إسرائيل لا يدرى ما فعلت ولا أراها إلا الفأر ألا…٧٤٩٧الفأرة مسخ وآية ذلك أنه يوضع بين يديها لبن الغنم فتشربه ويوضع بين…٧٥٠٠عجبا لأمر المؤمن إن أمره كله خير وليس ذاك لأحد إلا للمؤمن إن أصابته…٧٥٠١ويحك قطعت عنق صاحبك قطعت عنق صاحبك٧٥٠٢ويحك قطعت عنق صاحبك٧٥٠٣بهذا الإسناد . نحو حديث يزيد بن زريع وليس في حديثهما فقال رجل ما من…٧٥٠٤لقد أهلكتم أو قطعتم ظهر الرجل٧٥٠٥ومحمد بن المثنى، جميعا عن ابن مهدي، - واللفظ لابن المثنى قالا حدثنا…٧٥٠٦إذا رأيتم المداحين فاحثوا في وجوههم التراب٧٥٠٧وابن، بشار قالا حدثنا عبد الرحمن، عن سفيان، عن منصور، ح وحدثنا عثمان…٧٥٠٨أراني في المنام أتسوك بسواك فجذبني رجلان أحدهما أكبر من الآخر فناولت…٧٥٠٩قال كان أبو هريرة يحدث ويقول اسمعي يا ربة الحجرة اسمعي يا ربة الحجرة…٧٥١٠لا تكتبوا عني ومن كتب عني غير القرآن فليمحه وحدثوا عني ولا حرج ومن…٧٥١١كان ملك فيمن كان قبلكم وكان له ساحر فلما كبر قال للملك إني قد كبرت…٧٥٢٠حديث ٧٥٢٠٧٥٢١ألم يأن للرحيل٧٥٢٢لا حاجة لي في إبلك
