١٧٨ حديثًا•صفحة ١ من ٢
٣٩٦٢وزهير بن حرب، - واللفظ لزهير - قالا حدثنا يحيى، - وهو القطان - عن…٣٩٦٣قال أعطى رسول الله صلى الله عليه وسلم خيبر بشطر ما يخرج من ثمر أو زرع…٣٩٦٤أن رسول الله صلى الله عليه وسلم عامل أهل خيبر بشطر ما خرج منها من زرع…٣٩٦٥أقركم فيها على ذلك ما شئنا٣٩٦٦وحدثنا ابن رمح، أخبرنا الليث، عن محمد بن عبد الرحمن، عن نافع، عن عبد،…٣٩٦٧نقركم بها على ذلك ما شئنا٣٩٦٨ما من مسلم يغرس غرسا إلا كان ما أكل منه له صدقة وما سرق منه له صدقة…٣٩٦٩من غرس هذا النخل أمسلم أم كافر٣٩٧٠لا يغرس رجل مسلم غرسا ولا زرعا فيأكل منه سبع أو طائر أو شىء إلا كان…٣٩٧١يا أم معبد من غرس هذا النخل أمسلم أم كافر٣٩٧٢وإسحاق، بن إبراهيم جميعا عن أبي معاوية، ح وحدثنا عمرو الناقد، حدثنا…٣٩٧٣ما من مسلم يغرس غرسا أو يزرع زرعا فيأكل منه طير أو إنسان أو بهيمة إلا…٣٩٧٤من غرس هذا النخل أمسلم أم كافر٣٩٧٥إن بعت من أخيك ثمرا٣٩٧٦بهذا الإسناد مثله .٣٩٧٧وقتيبة، وعلي بن حجر، قالوا حدثنا إسماعيل بن جعفر، عن حميد، عن أنس، أن…٣٩٧٨حدثني أبو الطاهر، أخبرنا ابن وهب، أخبرني مالك، عن حميد الطويل، عن…٣٩٧٩إن لم يثمرها الله فبم يستحل أحدكم مال أخيه٣٩٨٠وإبراهيم بن دينار، وعبد الجبار بن العلاء، - واللفظ لبشر - قالوا حدثنا…٣٩٨١سعيد الخدري قال أصيب رجل في عهد رسول الله صلى الله عليه وسلم في ثمار…٣٩٨٢بهذا الإسناد مثله .٣٩٨٣أين المتألي على الله لا يفعل المعروف٣٩٨٤حدثنا حرملة بن يحيى، أخبرنا عبد الله بن وهب، أخبرني يونس، عن ابن…٣٩٨٥أن كعب بن مالك، أخبره أنه، تقاضى دينا له على ابن أبي حدرد بمثل حديث…٣٩٨٦قال مسلم وروى الليث بن سعد، حدثني جعفر بن ربيعة، عن عبد الرحمن بن،…٣٩٨٧من أدرك ماله بعينه عند رجل قد أفلس - أو إنسان قد أفلس - فهو أحق به من…٣٩٨٨ومحمد بن رمح، جميعا عن الليث بن سعد، ح وحدثنا أبو الربيع، ويحيى بن…٣٩٨٩أنه لصاحبه الذي باعه٣٩٩٠إذا أفلس الرجل فوجد الرجل متاعه بعينه فهو أحق به٣٩٩١فهو أحق به من الغرماء٣٩٩٢إذا أفلس الرجل فوجد الرجل عنده سلعته بعينها فهو أحق بها٣٩٩٣تلقت الملائكة روح رجل ممن كان قبلكم فقالوا أعملت من الخير شيئا قال لا…٣٩٩٤رجل لقي ربه فقال ما عملت قال ما عملت من الخير إلا أني كنت رجلا ذا مال…٣٩٩٥أن رجلا مات فدخل الجنة فقيل له ما كنت تعمل قال فإما ذكر وإما ذكر .…٣٩٩٦أتي الله بعبد من عباده آتاه الله مالا فقال له ماذا عملت في الدنيا -…٣٩٩٧حوسب رجل ممن كان قبلكم فلم يوجد له من الخير شىء إلا أنه كان يخالط…٣٩٩٨كان رجل يداين الناس فكان يقول لفتاه إذا أتيت معسرا فتجاوز عنه لعل…٣٩٩٩أن عبيد الله بن عبد الله بن عتبة، حدثه أنه، سمع أبا هريرة، يقول سمعت…٤٠٠٠من سره أن ينجيه الله من كرب يوم القيامة فلينفس عن معسر أو يضع عنه٤٠٠١بهذا الإسناد نحوه .٤٠٠٢مطل الغني ظلم وإذا أتبع أحدكم على مليء فليتبع٤٠٠٣حدثنا إسحاق بن إبراهيم، أخبرنا عيسى بن يونس، ح وحدثنا محمد بن رافع،…٤٠٠٤وحدثنا أبو بكر بن أبي شيبة، أخبرنا وكيع، ح وحدثني محمد بن حاتم، حدثنا…٤٠٠٥أنه سمع جابر بن عبد الله، يقول نهى رسول الله صلى الله عليه وسلم عن…٤٠٠٦لا يمنع فضل الماء ليمنع به الكلأ٤٠٠٧لا تمنعوا فضل الماء لتمنعوا به الكلأ٤٠٠٨لا يباع فضل الماء ليباع به الكلأ٤٠٠٩حدثنا يحيى بن يحيى، قال قرأت على مالك عن ابن شهاب، عن أبي بكر بن، عبد…٤٠١٠ومحمد بن رمح، عن الليث بن سعد، ح وحدثنا أبو بكر، بن أبي شيبة حدثنا…٤٠١١شر الكسب مهر البغي وثمن الكلب وكسب الحجام٤٠١٢ثمن الكلب خبيث ومهر البغي خبيث وكسب الحجام خبيث٤٠١٣كثير بهذا الإسناد مثله .٤٠١٤وحدثنا إسحاق بن إبراهيم، أخبرنا النضر بن شميل، حدثنا هشام، عن يحيى،…٤٠١٥والسنور، قال زجر النبي صلى الله عليه وسلم عن ذلك .٤٠١٦أن رسول الله صلى الله عليه وسلم أمر بقتل الكلاب .٤٠١٧قال أمر رسول الله صلى الله عليه وسلم بقتل الكلاب فأرسل في أقطار…٤٠١٨وحدثني حميد بن مسعدة، حدثنا بشر، - يعني ابن المفضل - حدثنا إسماعيل، -…٤٠١٩أن رسول الله صلى الله عليه وسلم أمر بقتل الكلاب إلا كلب صيد أو كلب…٤٠٢٠عليكم بالأسود البهيم ذي النقطتين فإنه شيطان٤٠٢١ما بالهم وبال الكلاب٤٠٢٢ورخص، في كلب الغنم والصيد والزرع .٤٠٢٣من اقتنى كلبا إلا كلب ماشية أو ضاري نقص من عمله كل يوم قيراطان٤٠٢٤من اقتنى كلبا إلا كلب صيد أو ماشية نقص من أجره كل يوم قيراطان٤٠٢٥من اقتنى كلبا إلا كلب ضارية أو ماشية نقص من عمله كل يوم قيراطان٤٠٢٦من اقتنى كلبا إلا كلب ماشية أو كلب صيد نقص من عمله كل يوم قيراط٤٠٢٧من اقتنى كلبا إلا كلب ضار أو ماشية نقص من عمله كل يوم قيراطان٤٠٢٨أيما أهل دار اتخذوا كلبا إلا كلب ماشية أو كلب صائد نقص من عملهم كل…٤٠٢٩من اتخذ كلبا إلا كلب زرع أو غنم أو صيد ينقص من أجره كل يوم قيراط٤٠٣٠من اقتنى كلبا ليس بكلب صيد ولا ماشية ولا أرض فإنه ينقص من أجره…٤٠٣١من اتخذ كلبا إلا كلب ماشية أو صيد أو زرع انتقص من أجره كل يوم قيراط٤٠٣٢من أمسك كلبا فإنه ينقص من عمله كل يوم قيراط إلا كلب حرث أو ماشية٤٠٣٣حدثنا إسحاق بن إبراهيم، أخبرنا شعيب بن إسحاق، حدثنا الأوزاعي، حدثني…٤٠٣٤بهذا الإسناد مثله .٤٠٣٥من اتخذ كلبا ليس بكلب صيد ولا غنم نقص من عمله كل يوم قيراط٤٠٣٦من اقتنى كلبا لا يغني عنه زرعا ولا ضرعا نقص من عمله كل يوم قيراط٤٠٣٧وقتيبة، وابن، حجر قالوا حدثنا إسماعيل، عن يزيد بن، خصيفة أخبرني…٤٠٣٨إن أفضل ما تداويتم به الحجامة أو هو من أمثل دوائكم٤٠٣٩إن أفضل ما تداويتم به الحجامة والقسط البحري ولا تعذبوا صبيانكم بالغمز٤٠٤٠حدثنا أحمد بن الحسن بن خراش، حدثنا شبابة، حدثنا شعبة، عن حميد، قال…٤٠٤١أن رسول الله صلى الله عليه وسلم احتجم وأعطى الحجام أجره واستعط .٤٠٤٢وعبد بن حميد، - واللفظ لعبد - قالا أخبرنا عبد، الرزاق أخبرنا معمر، عن…٤٠٤٣يا أيها الناس إن الله تعالى يعرض بالخمر ولعل الله سينزل فيها أمرا فمن…٤٠٤٤حدثنا سويد بن سعيد، حدثنا حفص بن ميسرة، عن زيد بن أسلم، عن عبد…٤٠٤٥حدثني أبو الطاهر، أخبرنا ابن وهب، أخبرني سليمان بن بلال، عن يحيى بن،…٤٠٤٦وإسحاق بن إبراهيم، قال زهير حدثنا وقال، إسحاق أخبرنا جرير، عن منصور،…٤٠٤٧قالت لما أنزلت الآيات من آخر سورة البقرة في الربا - قالت - خرج رسول…٤٠٤٨إن الله ورسوله حرم بيع الخمر والميتة والخنزير والأصنام٤٠٤٩وابن، نمير قالا حدثنا أبو أسامة، عن عبد الحميد، بن جعفر عن يزيد بن…٤٠٥٠لعن الله اليهود حرمت عليهم الشحوم فجملوها فباعوها٤٠٥١بهذا الإسناد . مثله .٤٠٥٢قاتل الله اليهود حرم الله عليهم الشحوم فباعوها وأكلوا أثمانها٤٠٥٣قاتل الله اليهود حرم عليهم الشحم فباعوه وأكلوا ثمنه٤٠٥٤لا تبيعوا الذهب بالذهب إلا مثلا بمثل ولا تشفوا بعضها على بعض ولا…٤٠٥٥لا تبيعوا الذهب بالذهب ولا تبيعوا الورق بالورق إلا مثلا بمثل ولا…٤٠٥٦حدثنا شيبان بن فروخ، حدثنا جرير يعني ابن حازم، ح وحدثنا محمد بن…٤٠٥٧لا تبيعوا الذهب بالذهب ولا الورق بالورق إلا وزنا بوزن مثلا بمثل سواء…٤٠٥٨لا تبيعوا الدينار بالدينارين ولا الدرهم بالدرهمين٤٠٥٩الورق بالذهب ربا إلا هاء وهاء والبر بالبر ربا إلا هاء وهاء والشعير…٤٠٦٠وزهير بن حرب، وإسحاق، عن ابن عيينة، عن الزهري، بهذا الإسناد .٤٠٦١حدثنا عبيد الله بن عمر القواريري، حدثنا حماد بن زيد، عن أيوب، عن أبي…٤٠٦٢وابن أبي عمر، جميعا عن عبد الوهاب الثقفي، عن أيوب، بهذا الإسناد نحوه .٤٠٦٣الذهب بالذهب والفضة بالفضة والبر بالبر والشعير بالشعير والتمر بالتمر…٤٠٦٤الذهب بالذهب والفضة بالفضة والبر بالبر والشعير بالشعير والتمر بالتمر…٤٠٦٥الذهب بالذهب مثلا بمثل٤٠٦٦التمر بالتمر والحنطة بالحنطة والشعير بالشعير والملح بالملح مثلا بمثل…٤٠٦٧بهذا الإسناد . ولم يذكر " يدا بيد " .٤٠٦٨الذهب بالذهب وزنا بوزن مثلا بمثل والفضة بالفضة وزنا بوزن مثلا بمثل…٤٠٦٩الدينار بالدينار لا فضل بينهما والدرهم بالدرهم لا فضل بينهما٤٠٧٠بهذا الإسناد مثله .٤٠٧١ما كان يدا بيد فلا بأس به وما كان نسيئة فهو ربا٤٠٧٢أنه سمع أبا المنهال، يقول سألت البراء بن عازب عن الصرف، فقال سل زيد…٤٠٧٣حدثنا أبو الربيع العتكي، حدثنا عباد بن العوام، أخبرنا يحيى بن أبي…٤٠٧٤أن عبد الرحمن بن أبي بكرة، أخبره أن أبا بكرة قال نهانا رسول الله صلى…٤٠٧٥الذهب بالذهب وزنا بوزن٤٠٧٦لا تباع حتى تفصل٤٠٧٧يزيد بهذا الإسناد نحوه .٤٠٧٨لا تبيعوا الذهب بالذهب إلا وزنا بوزن٤٠٧٩من كان يؤمن بالله واليوم الآخر فلا يأخذن إلا مثلا بمثل٤٠٨٠الطعام بالطعام مثلا بمثل٤٠٨١أكل تمر خيبر هكذا٤٠٨٢أكل تمر خيبر هكذا٤٠٨٣وعبد الله بن عبد الرحمن الدارمي، - واللفظ لهما - جميعا عن يحيى بن…٤٠٨٤ما هذا التمر من تمرنا٤٠٨٥لا صاعى تمر بصاع ولا صاعى حنطة بصاع ولا درهم بدرهمين٤٠٨٦كأن هذا ليس من تمر أرضنا٤٠٨٧حدثنا إسحاق بن إبراهيم، أخبرنا عبد الأعلى، أخبرنا داود، عن أبي نضرة،…٤٠٨٨الربا في النسيئة٤٠٨٩إنما الربا في النسيئة٤٠٩٠لا ربا فيما كان يدا بيد٤٠٩١ألا إنما الربا في النسيئة٤٠٩٢وإسحاق بن إبراهيم، - واللفظ لعثمان - قال إسحاق أخبرنا وقال، عثمان…٤٠٩٣وزهير بن حرب، وعثمان بن أبي شيبة، قالوا حدثنا هشيم، أخبرنا أبو…٤٠٩٤إن الحلال بين وإن الحرام بين وبينهما مشتبهات لا يعلمهن كثير من الناس…٤٠٩٥بهذا الإسناد مثله .٤٠٩٦وأبي، فروة الهمداني ح وحدثنا قتيبة بن سعيد، حدثنا يعقوب، - يعني ابن…٤٠٩٧الحلال بين والحرام بين٤٠٩٨أنه كان يسير على جمل له قد أعيا فأراد أن يسيبه قال فلحقني النبي صلى…٤٠٩٩بمثل حديث ابن نمير .٤١٠٠وإسحاق بن إبراهيم، - واللفظ لعثمان - قال إسحاق أخبرنا وقال، عثمان…٤١٠١حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن الأعمش، عن سالم بن أبي الجعد،…٤١٠٢حدثنا أبو كامل الجحدري، حدثنا عبد الواحد بن زياد، حدثنا الجريري، عن…٤١٠٣قال لما أتى على النبي صلى الله عليه وسلم وقد أعيا بعيري - قال - فنخسه…٤١٠٤يا جابر أتوفيت الثمن٤١٠٥أنه سمع جابر بن عبد الله، يقول اشترى مني رسول الله صلى الله عليه وسلم…٤١٠٦حدثني يحيى بن حبيب الحارثي، حدثنا خالد بن الحارث، حدثنا شعبة، أخبرنا…٤١٠٧قد أخذت جملك بأربعة دنانير ولك ظهره إلى المدينة٤١٠٨أعطه إياه إن خيار الناس أحسنهم قضاء٤١٠٩فإن خير عباد الله أحسنهم قضاء٤١١٠إن لصاحب الحق مقالا - فقال لهم - اشتروا له سنا فأعطوه إياه٤١١١خياركم محاسنكم قضاء
