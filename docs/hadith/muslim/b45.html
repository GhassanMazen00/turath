٦٥٠٠وزهير بن حرب، قالا حدثنا جرير، عن عمارة بن القعقاع، عن أبي زرعة، عن…٦٥٠١أمك ثم أمك ثم أمك ثم أبوك ثم أدناك أدناك٦٥٠٢نعم وأبيك لتنبأن٦٥٠٣بهذا الإسناد في حديث وهيب من أبر وفي حديث محمد بن طلحة أى الناس أحق…٦٥٠٤وزهير بن حرب، قالا حدثنا وكيع، عن سفيان، عن حبيب، ح وحدثنا محمد بن…٦٥٠٧فهل من والديك أحد حى٦٥٠٨حدثنا شيبان بن فروخ، حدثنا سليمان بن المغيرة، حدثنا حميد بن هلال، عن…٦٥٠٩لم يتكلم في المهد إلا ثلاثة عيسى ابن مريم وصاحب جريج وكان جريج رجلا…٦٥١٠رغم أنف ثم رغم أنف ثم رغم أنف٦٥١١رغم أنفه ثم رغم أنفه ثم رغم أنفه٦٥١٢قال قال رسول الله صلى الله عليه وسلم " رغم أنفه " . ثلاثا ثم ذكر مثله…٦٥١٣إن أبر البر صلة الولد أهل ود أبيه٦٥١٤أبر البر أن يصل الرجل ود أبيه٦٥١٥إن من أبر البر صلة الرجل أهل ود أبيه بعد أن يولي٦٥١٦البر حسن الخلق والإثم ما حاك في صدرك وكرهت أن يطلع عليه الناس٦٥١٧البر حسن الخلق والإثم ما حاك في نفسك وكرهت أن يطلع عليه الناس٦٥١٨إن الله خلق الخلق حتى إذا فرغ منهم قامت الرحم فقالت هذا مقام العائذ…٦٥١٩الرحم معلقة بالعرش تقول من وصلني وصله الله ومن قطعني قطعه الله٦٥٢٠لا يدخل الجنة قاطع٦٥٢١لا يدخل الجنة قاطع رحم٦٥٢٢وعبد بن حميد، عن عبد الرزاق، عن معمر، عن الزهري، بهذا الإسناد . مثله…٦٥٢٣من سره أن يبسط عليه رزقه أو ينسأ في أثره فليصل رحمه٦٥٢٤من أحب أن يبسط له في رزقه وينسأ له في أثره فليصل رحمه٦٥٢٥لئن كنت كما قلت فكأنما تسفهم المل ولا يزال معك من الله ظهير عليهم ما…٦٥٢٦لا تباغضوا ولا تحاسدوا ولا تدابروا وكونوا عباد الله إخوانا ولا يحل…٦٥٣٠حدثنا حاجب بن الوليد، حدثنا محمد بن حرب، حدثنا محمد بن الوليد…٦٥٣١وابن أبي عمر، وعمرو الناقد، جميعا عن ابن عيينة، عن الزهري، بهذا…٦٥٣٢لا يحل لمسلم أن يهجر أخاه فوق ثلاث ليال يلتقيان فيعرض هذا ويعرض هذا…٦٥٣٣فيعرض هذا ويعرض هذا٦٥٣٤لا يحل للمؤمن أن يهجر أخاه فوق ثلاثة أيام٦٥٣٥لا هجرة بعد ثلاث٦٥٣٦إياكم والظن فإن الظن أكذب الحديث ولا تحسسوا ولا تجسسوا ولا تنافسوا…٦٥٣٧لا تهجروا ولا تدابروا ولا تحسسوا ولا يبع بعضكم على بيع بعض وكونوا…٦٥٣٨لا تحاسدوا ولا تباغضوا ولا تجسسوا ولا تحسسوا ولا تناجشوا وكونوا عباد…٦٥٣٩لا تقاطعوا ولا تدابروا ولا تباغضوا ولا تحاسدوا وكونوا إخوانا كما…٦٥٤٠لا تباغضوا ولا تدابروا ولا تنافسوا وكونوا عباد الله إخوانا٦٥٤١لا تحاسدوا ولا تناجشوا ولا تباغضوا ولا تدابروا ولا يبع بعضكم على بيع…٦٥٤٢إن الله لا ينظر إلى أجسادكم ولا إلى صوركم ولكن ينظر إلى قلوبكم٦٥٤٣إن الله لا ينظر إلى صوركم وأموالكم ولكن ينظر إلى قلوبكم وأعمالكم٦٥٤٦تعرض الأعمال في كل يوم خميس واثنين فيغفر الله عز وجل في ذلك اليوم لكل…٦٥٤٧تعرض أعمال الناس في كل جمعة مرتين يوم الاثنين ويوم الخميس فيغفر لكل…٦٥٤٨إن الله يقول يوم القيامة أين المتحابون بجلالي اليوم أظلهم في ظلي يوم…٦٥٤٩أن رجلا زار أخا له في قرية أخرى فأرصد الله له على مدرجته ملكا فلما…٦٥٥٠بهذا الإسناد نحوه .٦٥٥١عائد المريض في مخرفة الجنة حتى يرجع٦٥٥٢من عاد مريضا لم يزل في خرفة الجنة حتى يرجع٦٥٥٣إن المسلم إذا عاد أخاه المسلم لم يزل في خرفة الجنة حتى يرجع٦٥٥٤من عاد مريضا لم يزل في خرفة الجنة٦٥٥٥بهذا الإسناد .٦٥٥٦إن الله عز وجل يقول يوم القيامة يا ابن آدم مرضت فلم تعدني . قال يا رب…٦٥٥٧وإسحاق بن إبراهيم، قال إسحاق أخبرنا وقال، عثمان حدثنا جرير، عن…٦٥٥٨وابن، بشار قالا حدثنا ابن أبي عدي، ح وحدثني بشر بن خالد، أخبرنا محمد،…٦٥٥٩أجل إني أوعك كما يوعك رجلان منكم٦٥٦٠نعم والذي نفسي بيده ما على الأرض مسلم٦٥٦١ما من مسلم يشاك شوكة فما فوقها إلا كتبت له بها درجة ومحيت عنه بها…٦٥٦٢ما يصيب المؤمن من شوكة فما فوقها إلا رفعه الله بها درجة أو حط عنه بها…٦٥٦٣لا تصيب المؤمن شوكة فما فوقها إلا قص الله بها من خطيئته٦٥٦٤بهذا الإسناد .٦٥٦٥ما من مصيبة يصاب بها المسلم إلا كفر بها عنه حتى الشوكة يشاكها٦٥٦٦لا يصيب المؤمن من مصيبة حتى الشوكة إلا قص بها من خطاياه أو كفر بها من…٦٥٦٧ما من شىء يصيب المؤمن حتى الشوكة تصيبه إلا كتب الله له بها حسنة أو…٦٥٦٨ما يصيب المؤمن من وصب ولا نصب ولا سقم ولا حزن حتى الهم يهمه إلا كفر…٦٥٦٩قاربوا وسددوا ففي كل ما يصاب به المسلم كفارة حتى النكبة ينكبها أو…٦٥٧٠ما لك يا أم السائب أو يا أم المسيب تزفزفين٦٥٧١إن شئت صبرت ولك الجنة وإن شئت دعوت الله أن يعافيك . قالت أصبر . قالت…٦٥٧٢يا عبادي إني حرمت الظلم على نفسي وجعلته بينكم محرما فلا تظالموا يا…٦٥٧٣بهذا الإسناد غير أن مروان أتمهما حديثا .٦٥٧٤والحسين، ابنا بشر ومحمد بن يحيى قالوا حدثنا أبو مسهر، . فذكروا الحديث…٦٥٧٥إني حرمت على نفسي الظلم وعلى عبادي فلا تظالموا٦٥٧٦اتقوا الظلم فإن الظلم ظلمات يوم القيامة واتقوا الشح فإن الشح أهلك من…٦٥٧٧إن الظلم ظلمات يوم القيامة٦٥٧٨المسلم أخو المسلم لا يظلمه ولا يسلمه من كان في حاجة أخيه كان الله في…٦٥٧٩أتدرون ما المفلس٦٥٨٠لتؤدن الحقوق إلى أهلها يوم القيامة حتى يقاد للشاة الجلحاء من الشاة…٦٥٨١إن الله عز وجل يملي للظالم فإذا أخذه لم يفلته٦٥٨٢ما هذا دعوى أهل الجاهلية٦٥٨٣ما بال دعوى الجاهلية٦٥٨٤دعوها فإنها منتنة٦٥٨٥المؤمن للمؤمن كالبنيان يشد بعضه بعضا٦٥٨٦مثل المؤمنين في توادهم وتراحمهم وتعاطفهم مثل الجسد إذا اشتكى منه عضو…٦٥٨٧حدثنا إسحاق الحنظلي، أخبرنا جرير، عن مطرف، عن الشعبي، عن النعمان بن،…٦٥٨٨المؤمنون كرجل واحد إن اشتكى رأسه تداعى له سائر الجسد بالحمى والسهر٦٥٨٩المسلمون كرجل واحد إن اشتكى عينه اشتكى كله وإن اشتكى رأسه اشتكى كله٦٥٩٠حدثنا ابن نمير، حدثنا حميد بن عبد الرحمن، عن الأعمش، عن الشعبي، عن…٦٥٩١المستبان ما قالا فعلى البادئ ما لم يعتد المظلوم٦٥٩٢ما نقصت صدقة من مال وما زاد الله عبدا بعفو إلا عزا وما تواضع أحد لله…٦٥٩٣أتدرون ما الغيبة٦٥٩٤لا يستر الله على عبد في الدنيا إلا ستره الله يوم القيامة٦٥٩٥لا يستر عبد عبدا في الدنيا إلا ستره الله يوم القيامة٦٥٩٦ائذنوا له فلبئس ابن العشيرة أو بئس رجل العشيرة٦٥٩٧بئس أخو القوم وابن العشيرة٦٥٩٨من يحرم الرفق يحرم الخير٦٥٩٩من يحرم الرفق يحرم الخير٦٦٠٠من حرم الرفق حرم الخير أو من يحرم الرفق يحرم الخير٦٦٠١يا عائشة إن الله رفيق يحب الرفق ويعطي على الرفق ما لا يعطي على العنف…٦٦٠٢إن الرفق لا يكون في شىء إلا زانه ولا ينزع من شىء إلا شانه٦٦٠٣وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، سمعت المقدام بن شريح…٦٦٠٤خذوا ما عليها ودعوها فإنها ملعونة٦٦٠٥خذوا ما عليها وأعروها فإنها ملعونة٦٦٠٦لا تصاحبنا ناقة عليها لعنة٦٦٠٧لا ايم الله لا تصاحبنا راحلة عليها لعنة من الله٦٦٠٨لا ينبغي لصديق أن يكون لعانا٦٦٠٩الرحمن بهذا الإسناد مثله .٦٦١٠لا يكون اللعانون شفعاء ولا شهداء يوم القيامة٦٦١١في هذا الإسناد بمثل معنى حديث حفص بن ميسرة .٦٦١٢إن اللعانين لا يكونون شهداء ولا شفعاء يوم القيامة٦٦١٣إني لم أبعث لعانا وإنما بعثت رحمة٦٦١٤حدثنا زهير بن حرب، حدثنا جرير، عن الأعمش، عن أبي الضحى، عن مسروق، عن…٦٦١٥وإسحاق بن إبراهيم، وعلي بن خشرم، جميعا عن عيسى بن يونس، كلاهما عن…٦٦١٦اللهم إنما أنا بشر فأيما رجل من المسلمين سببته أو لعنته أو جلدته…٦٦١٧وحدثنا ابن نمير، حدثنا أبي، حدثنا الأعمش، عن أبي سفيان، عن جابر، عن…٦٦١٨بإسناد عبد الله بن نمير . مثل حديثه غير أن في، حديث عيسى جعل " وأجرا…٦٦١٩اللهم إني أتخذ عندك عهدا لن تخلفنيه فإنما أنا بشر فأى المؤمنين آذيته…٦٦٢٠بهذا الإسناد نحوه إلا أنه قال " أو جلده " . قال أبو الزناد وهي لغة…٦٦٢١حدثني سليمان بن معبد، حدثنا سليمان بن حرب، حدثنا حماد بن زيد، عن…٦٦٢٢اللهم إنما محمد بشر يغضب كما يغضب البشر وإني قد اتخذت عندك عهدا لن…٦٦٢٣اللهم فأيما عبد مؤمن سببته فاجعل ذلك له قربة إليك يوم القيامة٦٦٢٤اللهم إني اتخذت عندك عهدا لن تخلفنيه فأيما مؤمن سببته أو جلدته فاجعل…٦٦٢٧آنت هيه لقد كبرت لا كبر سنك٦٦٢٨اذهب وادع لي معاوية٦٦٢٩يقول كنت ألعب مع الصبيان فجاء رسول الله صلى الله عليه وسلم فاختبأت…٦٦٣٠إن من شر الناس ذا الوجهين الذي يأتي هؤلاء بوجه وهؤلاء بوجه٦٦٣١إن شر الناس ذو الوجهين الذي يأتي هؤلاء بوجه وهؤلاء بوجه٦٦٣٢تجدون من شر الناس ذا الوجهين الذي يأتي هؤلاء بوجه وهؤلاء بوجه٦٦٣٣ليس الكذاب الذي يصلح بين الناس ويقول خيرا وينمي خيرا٦٦٣٤بهذا الإسناد . مثله غير أنفي حديث صالح وقالت ولم أسمعه يرخص في شىء…٦٦٣٥بهذا الإسناد إلى قوله " ونمى خيرا " . ولم يذكر ما بعده .٦٦٣٦ألا أنبئكم ما العضه هي النميمة القالة بين الناس٦٦٣٧إن الصدق يهدي إلى البر وإن البر يهدي إلى الجنة وإن الرجل ليصدق حتى…٦٦٣٨إن الصدق بر وإن البر يهدي إلى الجنة وإن العبد ليتحرى الصدق حتى يكتب…٦٦٣٩عليكم بالصدق فإن الصدق يهدي إلى البر وإن البر يهدي إلى الجنة وما يزال…٦٦٤٠ويتحرى الصدق ويتحرى الكذب٦٦٤١ما تعدون الرقوب فيكم٦٦٤٢بهذا الإسناد مثل معناه .٦٦٤٣ليس الشديد بالصرعة إنما الشديد الذي يملك نفسه عند الغضب٦٦٤٤ليس الشديد بالصرعة٦٦٤٥وعبد بن حميد، جميعا عن عبد الرزاق، أخبرنا معمر، ح وحدثنا عبد الله بن…٦٦٤٦إني لأعرف كلمة لو قالها لذهب عنه الذي يجد أعوذ بالله من الشيطان الرجيم٦٦٤٧إني لأعلم كلمة لو قالها لذهب ذا عنه أعوذ بالله من الشيطان الرجيم٦٦٤٨بهذا الإسناد .٦٦٤٩لما صور الله آدم في الجنة تركه ما شاء الله أن يتركه فجعل إبليس يطيف…٦٦٥٠بهذا الإسناد نحوه .٦٦٥١إذا قاتل أحدكم أخاه فليجتنب الوجه٦٦٥٢وزهير بن حرب، قالا حدثنا سفيان بن عيينة، عن أبي الزناد، بهذا الإسناد…٦٦٥٣إذا قاتل أحدكم أخاه فليتق الوجه٦٦٥٤إذا قاتل أحدكم أخاه فلا يلطمن الوجه٦٦٥٥إذا قاتل أحدكم أخاه فليجتنب الوجه فإن الله خلق آدم على صورته٦٦٥٦إذا قاتل أحدكم أخاه فليجتنب الوجه٦٦٥٧إن الله يعذب الذين يعذبون في الدنيا٦٦٥٨إن الله يعذب الذين يعذبون الناس في الدنيا
