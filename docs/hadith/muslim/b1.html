كتاب الإيمان

٤٣٦ حديثًا•صفحة ١ من ٣
٩٣الإسلام أن تشهد أن لا إله إلا الله وأن محمدا رسول الله وتقيم الصلاة…٩٤حدثني محمد بن عبيد الغبري، وأبو كامل الجحدري وأحمد بن عبدة قالوا…٩٥وحميد بن عبد الرحمن، قالا لقينا عبد الله بن عمر فذكرنا القدر وما…٩٦وحدثني حجاج بن الشاعر، حدثنا يونس بن محمد، حدثنا المعتمر، عن أبيه، عن…٩٧أن تؤمن بالله وملائكته وكتابه ولقائه ورسله وتؤمن بالبعث الآخر٩٨إذا ولدت الأمة بعلها٩٩حدثني زهير بن حرب، حدثنا جرير، عن عمارة، - وهو ابن القعقاع - عن أبي…١٠٠خمس صلوات في اليوم والليلة١٠١أفلح وأبيه إن صدق١٠٢حدثني عمرو بن محمد بن بكير الناقد، حدثنا هاشم بن القاسم أبو النضر،…١٠٣حدثني عبد الله بن هاشم العبدي، حدثنا بهز، حدثنا سليمان بن المغيرة، عن…١٠٤لقد وفق - أو لقد هدي - قال كيف قلت١٠٥وعبد الرحمن بن بشر، قالا حدثنا بهز، حدثنا شعبة، حدثنا محمد بن عثمان…١٠٦تعبد الله لا تشرك به شيئا وتقيم الصلاة وتؤتي الزكاة وتصل ذا رحمك١٠٧تعبد الله لا تشرك به شيئا وتقيم الصلاة المكتوبة وتؤدي الزكاة المفروضة…١٠٨قال أتى النبي صلى الله عليه وسلم النعمان بن قوقل فقال يا رسول الله…١٠٩والقاسم بن زكرياء، قالا حدثنا عبيد الله بن موسى، عن شيبان، عن الأعمش،…١١٠أن رجلا، سأل رسول الله صلى الله عليه وسلم فقال أرأيت إذا صليت الصلوات…١١١بني الإسلام على خمسة على أن يوحد الله وإقام الصلاة وإيتاء الزكاة…١١٢بني الإسلام على خمس على أن يعبد الله ويكفر بما دونه وإقام الصلاة…١١٣بني الإسلام على خمس شهادة أن لا إله إلا الله وأن محمدا عبده ورسوله…١١٤إن الإسلام بني على خمس شهادة أن لا إله إلا الله وإقام الصلاة وإيتاء…١١٥آمركم بأربع وأنهاكم عن أربع الإيمان بالله - ثم فسرها لهم فقال - شهادة…١١٦من الوفد أو من القوم١١٧أنهاكم عما ينبذ في الدباء والنقير والحنتم والمزفت١١٨آمركم بأربع وأنهاكم عن أربع اعبدوا الله ولا تشركوا به شيئا وأقيموا…١١٩وتذيفون فيه من القطيعاء أو التمر والماء١٢٠لا تشربوا في النقير١٢١إنك تأتي قوما من أهل الكتاب . فادعهم إلى شهادة أن لا إله إلا الله…١٢٢إنك ستأتي قوما١٢٣إنك تقدم على قوم أهل كتاب فليكن أول ما تدعوهم إليه عبادة الله عز وجل…١٢٤أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…١٢٥أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…١٢٦أمرت أن أقاتل الناس حتى يشهدوا أن لا إله إلا الله ويؤمنوا بي وبما جئت…١٢٧أمرت أن أقاتل الناس١٢٩أمرت أن أقاتل الناس حتى يشهدوا أن لا إله إلا الله وأن محمدا رسول الله…١٣٠من قال لا إله إلا الله وكفر بما يعبد من دون الله حرم ماله ودمه وحسابه…١٣١أنه سمع النبي صلى الله عليه وسلم يقول " من وحد الله " ثم ذكر بمثله .١٣٢يا عم قل لا إله إلا الله . كلمة أشهد لك بها عند الله١٣٣وعبد بن حميد، قالا أخبرنا عبد الرزاق، أخبرنا معمر، ح وحدثنا حسن…١٣٤قل لا إله إلا الله أشهد لك بها يوم القيامة١٣٥قل لا إله إلا الله أشهد لك بها يوم القيامة١٣٦من مات وهو يعلم أنه لا إله إلا الله دخل الجنة١٣٧حدثنا محمد بن أبي بكر المقدمي، حدثنا بشر بن المفضل، حدثنا خالد…١٣٨أشهد أن لا إله إلا الله وأني رسول الله لا يلقى الله بهما عبد غير شاك…١٣٩حدثنا سهل بن عثمان، وأبو كريب محمد بن العلاء جميعا عن أبي معاوية، -…١٤٠من قال أشهد أن لا إله إلا الله وحده لا شريك له وأن محمدا عبده ورسوله…١٤١أدخله الله الجنة على ما كان من عمل١٤٢من شهد أن لا إله إلا الله وأن محمدا رسول الله حرم الله عليه النار١٤٣قال كنت ردف النبي صلى الله عليه وسلم ليس بيني وبينه إلا مؤخرة الرحل…١٤٤يا معاذ تدري ما حق الله على العباد وما حق العباد على الله١٤٥يا معاذ أتدري ما حق الله على العباد١٤٦هل تدري ما حق الله على الناس١٤٧حدثني زهير بن حرب، حدثنا عمر بن يونس الحنفي، حدثنا عكرمة بن عمار، قال…١٤٨أن نبي الله صلى الله عليه وسلم ومعاذ بن جبل رديفه على الرحل قال " يا…١٤٩أليس يشهد أن لا إله إلا الله وأني رسول الله١٥٠أنه عمي فأرسل إلى رسول الله صلى الله عليه وسلم فقال تعال فخط لي مسجدا…١٥١ذاق طعم الإيمان من رضي بالله ربا وبالإسلام دينا وبمحمد رسولا١٥٢الإيمان بضع وسبعون شعبة والحياء شعبة من الإيمان١٥٣الإيمان بضع وسبعون أو بضع وستون شعبة فأفضلها قول لا إله إلا الله…١٥٤الحياء من الإيمان١٥٥بهذا الإسناد وقال مر برجل من الأنصار يعظ أخاه .١٥٦الحياء خير كله١٥٧حدثنا إسحاق بن إبراهيم، أخبرنا النضر، حدثنا أبو نعامة العدوي، قال…١٥٨الحياء لا يأتي إلا بخير١٥٩قل آمنت بالله فاستقم١٦٠تطعم الطعام وتقرأ السلام على من عرفت ومن لم تعرف١٦١من سلم المسلمون من لسانه ويده١٦٢المسلم من سلم المسلمون من لسانه ويده١٦٣من سلم المسلمون من لسانه ويده١٦٤بهذا الإسناد قال سئل رسول الله صلى الله عليه وسلم أى المسلمين أفضل…١٦٥ثلاث من كن فيه وجد بهن حلاوة الإيمان من كان الله ورسوله أحب إليه مما…١٦٦ثلاث من كن فيه وجد طعم الإيمان من كان يحب المرء لا يحبه إلا لله ومن…١٦٧من أن يرجع يهوديا أو نصرانيا١٦٨لا يؤمن عبد - وفي حديث عبد الوارث الرجل - حتى أكون أحب إليه من أهله…١٦٩لا يؤمن أحدكم حتى أكون أحب إليه من ولده ووالده والناس أجمعين١٧٠لا يؤمن أحدكم حتى يحب لأخيه - أو قال لجاره - ما يحب لنفسه١٧١والذي نفسي بيده لا يؤمن عبد حتى يحب لجاره - أو قال لأخيه - ما يحب…١٧٢لا يدخل الجنة من لا يأمن جاره بوائقه١٧٣من كان يؤمن بالله واليوم الآخر فليقل خيرا أو ليصمت ومن كان يؤمن بالله…١٧٤من كان يؤمن بالله واليوم الآخر فلا يؤذي جاره ومن كان يؤمن بالله…١٧٥فليحسن إلى جاره١٧٦من كان يؤمن بالله واليوم الآخر فليحسن إلى جاره ومن كان يؤمن بالله…١٧٧من رأى منكم منكرا فليغيره بيده فإن لم يستطع فبلسانه فإن لم يستطع…١٧٨حدثنا أبو كريب، محمد بن العلاء حدثنا أبو معاوية، حدثنا الأعمش، عن…١٧٩ما من نبي بعثه الله في أمة قبلي إلا كان له من أمته حواريون وأصحاب…١٨٠ما كان من نبي إلا وقد كان له حواريون يهتدون بهديه ويستنون بسنته١٨١ألا إن الإيمان ها هنا وإن القسوة وغلظ القلوب في الفدادين عند أصول…١٨٢جاء أهل اليمن هم أرق أفئدة الإيمان يمان والفقه يمان والحكمة يمانية١٨٣قال قال رسول الله صلى الله عليه وسلم بمثله .١٨٤أتاكم أهل اليمن هم أضعف قلوبا وأرق أفئدة الفقه يمان والحكمة يمانية١٨٥رأس الكفر نحو المشرق والفخر والخيلاء في أهل الخيل والإبل الفدادين أهل…١٨٦الإيمان يمان والكفر قبل المشرق والسكينة في أهل الغنم والفخر والرياء…١٨٧الفخر والخيلاء في الفدادين أهل الوبر والسكينة في أهل الغنم١٨٨الإيمان يمان والحكمة يمانية١٨٩جاء أهل اليمن هم أرق أفئدة وأضعف قلوبا الإيمان يمان والحكمة يمانية…١٩٠أتاكم أهل اليمن هم ألين قلوبا وأرق أفئدة الإيمان يمان والحكمة يمانية…١٩١رأس الكفر قبل المشرق١٩٢والفخر والخيلاء في أصحاب الإبل والسكينة والوقار في أصحاب الشاء١٩٣غلظ القلوب والجفاء في المشرق والإيمان في أهل الحجاز١٩٤لا تدخلون الجنة حتى تؤمنوا ولا تؤمنوا حتى تحابوا . أولا أدلكم على شىء…١٩٥والذي نفسي بيده لا تدخلون الجنة حتى تؤمنوا١٩٦قال ورجوت أن يسقط، عني رجلا قال فقال سمعته من الذي سمعه منه أبي كان…١٩٧حدثني محمد بن حاتم، حدثنا ابن مهدي، حدثنا سفيان، عن سهيل بن أبي صالح،…١٩٨سمعه وهو، يحدث أبا صالح عن تميم الداري، عن رسول الله صلى الله عليه…١٩٩قال بايعت رسول الله صلى الله عليه وسلم على إقام الصلاة وإيتاء الزكاة…٢٠٠وزهير بن حرب، وابن، نمير قالوا حدثنا سفيان، عن زياد بن علاقة، سمع…٢٠١ويعقوب الدورقي، قالا حدثنا هشيم، عن سيار، عن الشعبي، عن جرير، قال…٢٠٢لا يزني الزاني حين يزني وهو مؤمن ولا يسرق السارق حين يسرق وهو مؤمن…٢٠٣وحدثني عبد الملك بن شعيب بن الليث بن سعد، قال حدثني أبي، عن جدي، قال…٢٠٤وأبي، سلمة وأبي بكر بن عبد الرحمن بن الحارث بن هشام عن أبي هريرة، عن…٢٠٥وحدثني حسن بن علي الحلواني، حدثنا يعقوب بن إبراهيم، حدثنا عبد العزيز…٢٠٦يرفع الناس إليه فيها أبصارهم٢٠٧لا يزني الزاني حين يزني وهو مؤمن ولا يسرق حين يسرق وهو مؤمن ولا يشرب…٢٠٨رفعه قال " لا يزني الزاني " . ثم ذكر بمثل حديث شعبة .٢١٠أربع من كن فيه كان منافقا خالصا ومن كانت فيه خلة منهن كانت فيه خلة من…٢١١آية المنافق ثلاث إذا حدث كذب وإذا وعد أخلف وإذا ائتمن خان٢١٢من علامات المنافق ثلاثة إذا حدث كذب وإذا وعد أخلف وإذا ائتمن خان٢١٣آية المنافق ثلاث وإن صام وصلى وزعم أنه مسلم٢١٤وإن صام وصلى وزعم أنه مسلم٢١٥إذا كفر الرجل أخاه فقد باء بها أحدهما٢١٦أيما امرئ قال لأخيه يا كافر . فقد باء بها أحدهما إن كان كما قال وإلا…٢١٧ليس من رجل ادعى لغير أبيه وهو يعلمه إلا كفر ومن ادعى ما ليس له فليس…٢١٨لا ترغبوا عن آبائكم فمن رغب عن أبيه فهو كفر٢١٩من ادعى أبا في الإسلام غير أبيه يعلم أنه غير أبيه فالجنة عليه حرام٢٢٠من ادعى إلى غير أبيه وهو يعلم أنه غير أبيه فالجنة عليه حرام٢٢١سباب المسلم فسوق وقتاله كفر٢٢٢وابن المثنى، عن محمد بن جعفر، عن شعبة، عن منصور، ح وحدثنا ابن نمير،…٢٢٣ومحمد بن المثنى، وابن، بشار جميعا عن محمد بن جعفر، عن شعبة، ح وحدثنا…٢٢٤وحدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا شعبة، عن واقد بن محمد، عن…٢٢٥ويحكم - أو قال ويلكم - لا ترجعوا بعدي كفارا يضرب بعضكم رقاب بعض٢٢٦أن أباه، حدثه عن ابن عمر، عن النبي صلى الله عليه وسلم بمثل حديث شعبة…٢٢٧اثنتان في الناس هما بهم كفر الطعن في النسب والنياحة على الميت٢٢٨أيما عبد أبق من مواليه فقد كفر حتى يرجع إليهم٢٢٩أيما عبد أبق فقد برئت منه الذمة٢٣٠إذا أبق العبد لم تقبل له صلاة٢٣١هل تدرون ماذا قال ربكم٢٣٢ألم تروا إلى ما قال ربكم قال ما أنعمت على عبادي من نعمة إلا أصبح فريق…٢٣٣ما أنزل الله من السماء من بركة إلا أصبح فريق من الناس بها كافرين ينزل…٢٣٤أصبح من الناس شاكر ومنهم كافر قالوا هذه رحمة الله . وقال بعضهم لقد…٢٣٥آية المنافق بغض الأنصار وآية المؤمن حب الأنصار٢٣٦حب الأنصار آية الإيمان وبغضهم آية النفاق٢٣٧لا يحبهم إلا مؤمن ولا يبغضهم إلا منافق من أحبهم أحبه الله ومن أبغضهم…٢٣٨لا يبغض الأنصار رجل يؤمن بالله واليوم الآخر٢٣٩لا يبغض الأنصار رجل يؤمن بالله واليوم الآخر٢٤٠قال قال علي والذي فلق الحبة وبرأ النسمة إنه لعهد النبي الأمي صلى الله…٢٤١يا معشر النساء تصدقن وأكثرن الاستغفار فإني رأيتكن أكثر أهل النار٢٤٢بهذا الإسناد مثله .٢٤٣وحدثني الحسن بن علي الحلواني، وأبو بكر بن إسحاق قالا حدثنا ابن أبي…٢٤٤إذا قرأ ابن آدم السجدة فسجد اعتزل الشيطان يبكي يقول يا ويله -
