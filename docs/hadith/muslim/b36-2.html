٢٥٥ حديثًا•صفحة ٢ من ٢
٥٢٧٩لا يشربن أحد منكم قائما فمن نسي فليستقئ٥٢٨٠عباس قال سقيت رسول الله صلى الله عليه وسلم من زمزم فشرب وهو قائم .٥٢٨١أن النبي صلى الله عليه وسلم شرب من زمزم من دلو منها وهو قائم .٥٢٨٢ومغيرة، عن الشعبي، عن ابن عباس، أن رسول الله صلى الله عليه وسلم شرب…٥٢٨٣سمع الشعبي، سمع ابن عباس، قال سقيت رسول الله صلى الله عليه وسلم من…٥٢٨٤بهذا الإسناد وفي حديثهما فأتيته بدلو .٥٢٨٥أن النبي صلى الله عليه وسلم نهى أن يتنفس في الإناء .٥٢٨٦أن رسول الله صلى الله عليه وسلم كان يتنفس في الإناء ثلاثا .٥٢٨٧إنه أروى وأبرأ وأمرأ٥٢٨٨وحدثناه قتيبة بن سعيد، وأبو بكر بن أبي شيبة قالا حدثنا وكيع، عن هشام،…٥٢٨٩حدثنا يحيى بن يحيى، قال قرأت على مالك عن ابن شهاب، عن أنس بن مالك، أن…٥٢٩٠وعمرو الناقد، وزهير بن حرب، ومحمد بن عبد، الله بن نمير - واللفظ لزهير…٥٢٩١الأيمنون الأيمنون الأيمنون٥٢٩٢أتأذن لي أن أعطي هؤلاء٥٢٩٣حدثنا يحيى بن يحيى، أخبرنا عبد العزيز بن أبي حازم، ح وحدثناه قتيبة…٥٢٩٤إذا أكل أحدكم طعاما فلا يمسح يده حتى يلعقها أو يلعقها٥٢٩٥إذا أكل أحدكم من الطعام فلا يمسح يده حتى يلعقها أو يلعقها٥٢٩٦وزهير بن حرب، ومحمد بن حاتم، قالوا حدثنا ابن، مهدي عن سفيان، عن سعد…٥٢٩٧قال كان رسول الله صلى الله عليه وسلم يأكل بثلاث أصابع ويلعق يده قبل…٥٢٩٨بن سعد أن عبد الرحمن بن كعب بن مالك، - أو عبد الله بن كعب - أخبره عن…٥٢٩٩أنحدثاه - أو، أحدهما - عن أبيه، كعب بن مالك عن النبي صلى الله عليه…٥٣٠٠إنكم لا تدرون في أيه البركة٥٣٠١إذا وقعت لقمة أحدكم فليأخذها فليمط ما كان بها من أذى وليأكلها ولا…٥٣٠٢ولا يمسح يده بالمنديل حتى يلعقها أو يلعقها٥٣٠٣إن الشيطان يحضر أحدكم عند كل شىء من شأنه حتى يحضره عند طعامه فإذا…٥٣٠٤إذا سقطت لقمة أحدكم٥٣٠٥وحدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن فضيل، عن الأعمش، عن أبي،…٥٣٠٦إذا سقطت لقمة أحدكم فليمط عنها الأذى وليأكلها ولا يدعها للشيطان٥٣٠٧إذا أكل أحدكم فليلعق أصابعه فإنه لا يدري في أيتهن البركة٥٣٠٨وليسلت أحدكم الصحفة٥٣٠٩إن هذا اتبعنا فإن شئت أن تأذن له وإن شئت رجع٥٣١٠وإسحاق بن إبراهيم، جميعا عن أبي معاوية، ح وحدثناه نصر بن علي الجهضمي،…٥٣١١بهذا الحديث .٥٣١٢أن جارا، لرسول الله صلى الله عليه وسلم فارسيا كان طيب المرق فصنع…٥٣١٣ما أخرجكما من بيوتكما هذه الساعة٥٣١٤ما أقعدكما ها هنا٥٣١٥يا أهل الخندق إن جابرا قد صنع لكم سورا فحيهلا بكم٥٣١٦أرسلك أبو طلحة٥٣١٧قال بعثني أبو طلحة إلى رسول الله صلى الله عليه وسلم لأدعوه وقد جعل…٥٣١٨قال بعثني أبو طلحة إلى رسول الله صلى الله عليه وسلم . وساق الحديث…٥٣١٩قال أمر أبو طلحة أم سليم أن تصنع، للنبي صلى الله عليه وسلم طعاما…٥٣٢٠هلمه فإن الله سيجعل فيه البركة٥٣٢١وحدثنا عبد بن حميد، حدثنا خالد بن مخلد البجلي، حدثني محمد بن موسى،…٥٣٢٢وحدثنا الحسن بن علي الحلواني، حدثنا وهب بن جرير، حدثنا أبي قال، سمعت…٥٣٢٣أنحدثه أنه، سمع أنس بن مالك، يقول جئت رسول الله صلى الله عليه وسلم…٥٣٢٤وحدثني حجاج بن الشاعر، حدثنا يونس بن محمد، حدثنا حرب بن ميمون، عن…٥٣٢٥الله بن أبي طلحة أنه سمع أنس بن مالك، يقول إن خياطا دعا رسول الله صلى…٥٣٢٦قال دعا رسول الله صلى الله عليه وسلم رجل فانطلقت معه فجيء بمرقة فيها…٥٣٢٧وعبد بن حميد، جميعا عن عبد الرزاق، أخبرنا معمر، عن ثابت البناني،…٥٣٢٨اللهم بارك لهم في ما رزقتهم واغفر لهم وارحمهم٥٣٢٩بهذا الإسناد ولم يشكا في إلقاء النوى بين الإصبعين .٥٣٣٠وعبد الله بن عون الهلالي، - قال يحيى أخبرنا وقال ابن عون، حدثنا -…٥٣٣١قال رأيت النبي صلى الله عليه وسلم مقعيا يأكل تمرا .٥٣٣٢وابن أبي عمر، جميعا عن سفيان، قال ابن أبي عمر حدثنا سفيان بن عيينة،…٥٣٣٣حدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت جبلة،…٥٣٣٤بهذا الإسناد وليس في حديثهما قول شعبة ولا قوله وقد كان أصاب الناس…٥٣٣٥ومحمد بن المثنى، قالا حدثنا عبد الرحمن، عن سفيان، عن جبلة بن سحيم،…٥٣٣٦لا يجوع أهل بيت عندهم التمر٥٣٣٧يا عائشة بيت لا تمر فيه جياع أهله يا عائشة بيت لا تمر فيه جياع أهله…٥٣٣٨من أكل سبع تمرات مما بين لابتيها حين يصبح لم يضره سم حتى يمسي٥٣٣٩من تصبح بسبع تمرات عجوة لم يضره ذلك اليوم سم ولا سحر٥٣٤٠وحدثناه ابن أبي عمر، حدثنا مروان بن معاوية الفزاري، ح وحدثناه إسحاق،…٥٣٤١إن في عجوة العالية شفاء أو إنها ترياق أول البكرة٥٣٤٢الكمأة من المن وماؤها شفاء للعين٥٣٤٣الكمأة من المن وماؤها شفاء للعين٥٣٤٤وحدثنا محمد بن المثنى، حدثني محمد بن جعفر، حدثنا شعبة، قال وأخبرني…٥٣٤٥الكمأة من المن الذي أنزل الله تبارك وتعالى على بني إسرائيل وماؤها…٥٣٤٦الكمأة من المن الذي أنزل الله على موسى وماؤها شفاء للعين٥٣٤٧الكمأة من المن الذي أنزل الله عز وجل على بني إسرائيل وماؤها شفاء للعين٥٣٤٨الكمأة من المن وماؤها شفاء للعين٥٣٤٩عليكم بالأسود منه٥٣٥٠نعم الأدم - أو الإدام - الخل٥٣٥١بهذا الإسناد وقال " نعم الأدم " . ولم يشك .٥٣٥٢نعم الأدم الخل نعم الأدم الخل٥٣٥٣أنه سمع جابر بن عبد الله، يقول أخذ رسول الله صلى الله عليه وسلم بيدي…٥٣٥٤فنعم الأدم الخل٥٣٥٥وحدثنا أبو بكر بن أبي شيبة، حدثنا يزيد بن هارون، أخبرنا حجاج بن أبي،…٥٣٥٦لا ولكني أكرهه من أجل ريحه٥٣٥٧في هذا الإسناد .٥٣٥٨وأحمد بن سعيد بن صخر، - واللفظ منهما قريب - قالا حدثنا أبو النعمان،…٥٣٥٩من يضيف هذا الليلة رحمه الله٥٣٦٠. أن رجلا، من الأنصار بات به ضيف فلم يكن عنده إلا قوته وقوت صبيانه…٥٣٦١ألا رجل يضيف هذا رحمه الله٥٣٦٢احتلبوا هذا اللبن بيننا٥٣٦٣بهذا الإسناد .٥٣٦٤هل مع أحد منكم طعام٥٣٦٥من كان عنده طعام اثنين فليذهب بثلاثة ومن كان عنده طعام أربعة فليذهب…٥٣٦٦بل أنت أبرهم وأخيرهم٥٣٦٧طعام الاثنين كافي الثلاثة وطعام الثلاثة كافي الأربعة٥٣٦٨طعام الواحد يكفي الاثنين وطعام الاثنين يكفي الأربعة وطعام الأربعة…٥٣٦٩حدثنا ابن نمير، حدثنا أبي، حدثنا سفيان، ح وحدثني محمد بن المثنى،…٥٣٧٠طعام الواحد يكفي الاثنين وطعام الاثنين يكفي الأربعة٥٣٧١طعام الرجل يكفي رجلين وطعام رجلين يكفي أربعة وطعام أربعة يكفي ثمانية٥٣٧٢الكافر يأكل في سبعة أمعاء والمؤمن يأكل في معى واحد٥٣٧٣وابن، نمير قالا حدثنا عبيد الله، ح وحدثني محمد بن رافع، وعبد بن، حميد…٥٣٧٤إن الكافر يأكل في سبعة أمعاء٥٣٧٥المؤمن يأكل في معى واحد والكافر يأكل في سبعة أمعاء٥٣٧٦وحدثنا ابن نمير، حدثنا أبي، حدثنا سفيان، عن أبي الزبير، عن جابر، عن…٥٣٧٧المؤمن يأكل في معى واحد والكافر يأكل في سبعة أمعاء٥٣٧٨حدثنا قتيبة بن سعيد، حدثنا عبد العزيز، - يعني ابن محمد - عن العلاء،…٥٣٧٩المؤمن يشرب في معى واحد والكافر يشرب في سبعة أمعاء٥٣٨٠وزهير بن حرب، وإسحاق بن إبراهيم، قال زهير حدثنا وقال الآخران، أخبرنا…٥٣٨١بهذا الإسناد مثله .٥٣٨٢وعبد الملك بن عمرو، وعمر بن سعد، أبو داود الحفري كلهم عن سفيان، عن…٥٣٨٣ومحمد بن المثنى، قالا حدثنا أبو معاوية، عن الأعمش، عن أبي حازم، عن…
