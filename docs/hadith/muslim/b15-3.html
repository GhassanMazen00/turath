كتاب الحج

٥٨٣ حديثًا•صفحة ٣ من ٤
٣١٠٤أنه كان رديف رسول الله صلى الله عليه وسلم حين أفاض من عرفة فلما جاء…٣١٠٥أن رسول الله صلى الله عليه وسلم أفاض من عرفة وأسامة ردفه قال أسامة…٣١٠٦وقتيبة بن سعيد، جميعا عن حماد بن زيد، - قال أبو الربيع حدثنا حماد، -…٣١٠٧وعبد الله بن نمير، وحميد، بن عبد الرحمن عن هشام بن عروة، بهذا الإسناد…٣١١٠أن رسول الله صلى الله عليه وسلم صلى المغرب والعشاء بالمزدلفة جميعا .٣١١١أنأخبره أن أباه قال جمع رسول الله صلى الله عليه وسلم بين المغرب…٣١١٢حدثنا محمد بن المثنى، حدثنا عبد الرحمن بن مهدي، حدثنا شعبة، عن الحكم،…٣١١٣بهذا الإسناد وقال صلاهما بإقامة واحدة .٣١١٤قال جمع رسول الله صلى الله عليه وسلم بين المغرب والعشاء بجمع صلى…٣١١٥وحدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، حدثنا إسماعيل بن…٣١١٦قال ما رأيت رسول الله صلى الله عليه وسلم صلى صلاة إلا لميقاتها إلا…٣١١٧وإسحاق بن إبراهيم، جميعا عن جرير، عن الأعمش، بهذا الإسناد وقال قبل…٣١١٨وحدثنا عبد الله بن مسلمة بن قعنب، حدثنا أفلح، - يعني ابن حميد - عن…٣١١٩ومحمد بن المثنى، جميعا عن الثقفي، - قال ابن المثنى حدثنا عبد الوهاب،…٣١٢٠قالت وددت أني كنت استأذنت رسول الله صلى الله عليه وسلم كما استأذنته…٣١٢١بهذا الإسناد . نحوه .٣١٢٢حدثنا محمد بن أبي بكر المقدمي، حدثنا يحيى، - وهو القطان - عن ابن…٣١٢٣بهذا الإسناد وفي روايته قالت لا أى بنى إن نبي الله صلى الله عليه وسلم…٣١٢٤أن ابن شوال، أخبره أنه، دخل على أم حبيبة فأخبرته أن النبي صلى الله…٣١٢٥وحدثنا أبو بكر بن أبي شيبة، حدثنا سفيان بن عيينة، حدثنا عمرو بن…٣١٢٦وقتيبة بن سعيد، جميعا عن حماد، - قال يحيى أخبرنا حماد بن زيد، - عن…٣١٢٧يزيد أنه سمع ابن عباس، يقول أنا ممن، قدم رسول الله صلى الله عليه وسلم…٣١٢٨قال كنت فيمن قدم رسول الله صلى الله عليه وسلم في ضعفة أهله .٣١٢٩أن ابن عباس، قال بعث بي رسول الله صلى الله عليه وسلم بسحر من جمع في…٣١٣٠وحرملة بن يحيى، قالا أخبرنا ابن وهب، أخبرني يونس، عن ابن شهاب، أن…٣١٣١قال رمى عبد الله بن مسعود جمرة العقبة من بطن الوادي بسبع حصيات يكبر…٣١٣٢وحدثنا منجاب بن الحارث التميمي، أخبرنا ابن مسهر، عن الأعمش، قال سمعت…٣١٣٣يقول لا تقولوا سورة البقرة . واقتصا الحديث بمثل حديث ابن مسهر .٣١٣٤وحدثنا أبو بكر بن أبي شيبة، حدثنا غندر، عن شعبة، ح وحدثنا محمد بن…٣١٣٥بهذا الإسناد غير أنه قال فلما أتى جمرة العقبة .٣١٣٦وحدثنا أبو بكر بن أبي شيبة، حدثنا أبو المحياة، ح وحدثنا يحيى بن يحيى،…٣١٣٧لتأخذوا مناسككم فإني لا أدري لعلي لا أحج بعد حجتي هذه٣١٣٨إن أمر عليكم عبد مجدع - حسبتها قالت - أسود يقودكم بكتاب الله تعالى…٣١٣٩جدته قالت حججت مع رسول الله صلى الله عليه وسلم حجة الوداع فرأيت أسامة…٣١٤٠وعبد بن حميد، قال ابن حاتم حدثنا محمد بن بكر، أخبرنا ابن جريج، أخبرنا…٣١٤١وابن، إدريس عن ابن، جريج عن أبي الزبير، عن جابر، قال رمى رسول الله…٣١٤٢أنه سمع جابر بن عبد الله، يقول كان النبي صلى الله عليه وسلم . بمثله .٣١٤٣الاستجمار تو ورمى الجمار تو والسعى بين الصفا والمروة تو والطواف تو…٣١٤٤رحم الله المحلقين - مرة أو مرتين ثم قال - والمقصرين٣١٤٥اللهم ارحم المحلقين٣١٤٦رحم الله المحلقين٣١٤٧بهذا الإسناد وقال في الحديث فلما كانت الرابعة قال " والمقصرين " .٣١٤٨اللهم اغفر للمحلقين٣١٤٩وحدثني أمية بن بسطام، حدثنا يزيد بن زريع، حدثنا روح، عن العلاء، عن…٣١٥٠حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، وأبو داود الطيالسي عن شعبة، عن…٣١٥١أن رسول الله صلى الله عليه وسلم حلق رأسه في حجة الوداع .٣١٥٢أن رسول الله صلى الله عليه وسلم أتى منى فأتى الجمرة فرماها ثم أتى…٣١٥٣وابن، نمير وأبو كريب قالوا أخبرنا حفص بن، غياث عن هشام، بهذا الإسناد…٣١٥٤وحدثنا محمد بن المثنى، حدثنا عبد الأعلى، حدثنا هشام، عن محمد، عن أنس،…٣١٥٥قال لما رمى رسول الله صلى الله عليه وسلم الجمرة ونحر نسكه وحلق ناول…٣١٥٦حدثنا يحيى بن يحيى، قال قرأت على مالك عن ابن شهاب، عن عيسى بن، طلحة…٣١٥٧أنه سمع عبد الله بن عمرو بن العاص، يقول وقف رسول الله صلى الله عليه…٣١٥٨إلى آخره .٣١٥٩وحدثنا علي بن خشرم، أخبرنا عيسى، عن ابن جريج، قال سمعت ابن شهاب، يقول…٣١٦٠بهذا الإسناد أما رواية ابن بكر فكرواية عيسى إلا قوله لهؤلاء الثلاث .…٣١٦١وزهير بن حرب، قال أبو بكر حدثنا ابن عيينة، عن الزهري، عن عيسى بن…٣١٦٢وعبد بن حميد، عن عبد الرزاق، عن معمر، عن الزهري، بهذا الإسناد رأيت…٣١٦٣وحدثني محمد بن عبد الله بن قهزاذ، حدثنا علي بن الحسن، عن عبد الله بن،…٣١٦٤أن النبي صلى الله عليه وسلم قيل له في الذبح والحلق والرمى والتقديم…٣١٦٥أن رسول الله صلى الله عليه وسلم أفاض يوم النحر ثم رجع فصلى الظهر بمنى…٣١٦٦حدثني زهير بن حرب، حدثنا إسحاق بن يوسف الأزرق، أخبرنا سفيان، عن عبد…٣١٦٧أن النبي صلى الله عليه وسلم وأبا بكر وعمر كانوا ينزلون الأبطح٣١٦٨أن ابن عمر، كان يرى التحصيب سنة وكان يصلي الظهر يوم النفر بالحصبة .…٣١٦٩قالت نزول الأبطح ليس بسنة إنما نزله رسول الله صلى الله عليه وسلم لأنه…٣١٧٠بهذا الإسناد مثله .٣١٧١أن أبا بكر، وعمر، وابن، عمر كانوا ينزلون الأبطح . قال الزهري وأخبرني…٣١٧٢وإسحاق بن إبراهيم، وابن أبي عمر، وأحمد بن، عبدة - واللفظ لأبي بكر -…٣١٧٣حدثنا قتيبة بن سعيد، وأبو بكر بن أبي شيبة وزهير بن حرب جميعا عن ابن،…٣١٧٤ننزل غدا إن شاء الله بخيف بني كنانة حيث تقاسموا على الكفر٣١٧٥نحن نازلون غدا بخيف بني كنانة حيث تقاسموا على الكفر٣١٧٦منزلنا - إن شاء الله إذا فتح الله - الخيف حيث تقاسموا على الكفر٣١٧٧أن العباس بن عبد المطلب، استأذن رسول الله صلى الله عليه وسلم أن يبيت…٣١٧٨بن عمر بهذا الإسناد مثله .٣١٧٩أحسنتم وأجملتم كذا فاصنعوا٣١٨٠نحن نعطيه من عندنا٣١٨١وعمرو الناقد، وزهير بن حرب، قالوا حدثنا ابن عيينة، عن عبد الكريم…٣١٨٢وقال، إسحاق بن إبراهيم أخبرنا معاذ بن هشام، قال أخبرني أبي كلاهما، عن…٣١٨٣ومحمد بن مرزوق، وعبد بن حميد، قال عبد أخبرنا وقال الآخران، حدثنا محمد…٣١٨٤الكريم بن مالك الجزري أن مجاهدا، أخبره أن عبد الرحمن بن أبي ليلى…٣١٨٥حدثنا قتيبة بن سعيد، حدثنا مالك، ح وحدثنا يحيى بن يحيى، - واللفظ له -…٣١٨٦وحدثنا يحيى بن يحيى، أخبرنا أبو خيثمة، عن أبي الزبير، عن جابر، ح.…٣١٨٧وحدثني محمد بن حاتم، حدثنا وكيع، حدثنا عزرة بن ثابت، عن أبي الزبير،…٣١٨٨أنه سمع جابر بن عبد الله، قال اشتركنا مع النبي صلى الله عليه وسلم في…٣١٨٩أنه سمع جابر بن عبد الله، يحدث عن حجة النبي، صلى الله عليه وسلم قال…٣١٩٠حدثنا يحيى بن يحيى، أخبرنا هشيم، عن عبد الملك، عن عطاء، عن جابر بن،…٣١٩١حدثنا عثمان بن أبي شيبة، حدثنا يحيى بن زكرياء بن أبي زائدة، عن ابن…٣١٩٢أنه سمع جابر بن عبد، الله يقول نحر رسول الله صلى الله عليه وسلم عن…٣١٩٣أن ابن عمر، أتى على رجل وهو ينحر بدنته باركة فقال ابعثها قياما مقيدة…٣١٩٤ومحمد بن رمح، قالا أخبرنا الليث، ح وحدثنا قتيبة، حدثنا ليث، عن ابن…٣١٩٥بهذا الإسناد مثله .٣١٩٦وزهير بن حرب، قالا حدثنا سفيان، عن الزهري، عن عروة، عن عائشة، عن…٣١٩٧تقول كنت أفتل قلائد هدى رسول الله صلى الله عليه وسلم بيدى هاتين ثم لا…٣١٩٨قالت فتلت قلائد بدن رسول الله صلى الله عليه وسلم بيدى ثم أشعرها…٣١٩٩ويعقوب بن إبراهيم الدورقي، قال ابن حجر حدثنا إسماعيل بن إبراهيم، عن…٣٢٠٠وحدثنا محمد بن المثنى، حدثنا حسين بن الحسن، حدثنا ابن عون، عن القاسم،…٣٢٠١وحدثنا زهير بن حرب، حدثنا جرير، عن منصور، عن إبراهيم، عن الأسود، عن…٣٢٠٢قالت ربما فتلت القلائد لهدى رسول الله صلى الله عليه وسلم فيقلد هديه…٣٢٠٣قالت أهدى رسول الله صلى الله عليه وسلم مرة إلى البيت غنما فقلدها .٣٢٠٤وحدثنا إسحاق بن منصور، حدثنا عبد الصمد، حدثني أبي، حدثني محمد بن…٣٢٠٥حدثنا يحيى بن يحيى، قال قرأت على مالك عن عبد الله بن أبي بكر، عن…٣٢٠٦وحدثنا سعيد بن منصور، حدثنا هشيم، أخبرنا إسماعيل بن أبي خالد، عن…٣٢٠٧وحدثنا محمد بن المثنى، حدثنا عبد الوهاب، حدثنا داود، ح وحدثنا ابن…٣٢٠٨أن رسول الله صلى الله عليه وسلم رأى رجلا يسوق بدنة فقال " اركبها " .…٣٢٠٩بهذا الإسناد وقال بينما رجل يسوق بدنة مقلدة .٣٢١٠رسول الله صلى الله عليه وسلم فذكر أحاديث منها وقال بينما رجل يسوق…٣٢١١وسريج بن يونس، قالا حدثنا هشيم، أخبرنا حميد، عن ثابت، عن أنس، قال…٣٢١٢وحدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن مسعر، عن بكير بن الأخنس،…٣٢١٣يقول مر على النبي صلى الله عليه وسلم ببدنة . فذكر مثله .٣٢١٤اركبها بالمعروف إذا ألجئت إليها حتى تجد ظهرا٣٢١٥اركبها بالمعروف حتى تجد ظهرا٣٢١٦انحرها ثم اصبغ نعليها في دمها ثم اجعله على صفحتها ولا تأكل منها أنت…٣٢١٧أن رسول الله صلى الله عليه وسلم بعث بثمان عشرة بدنة مع رجل . ثم ذكر…٣٢١٨إن عطب منها شىء فخشيت عليه موتا فانحرها ثم اغمس نعلها في دمها ثم اضرب…٣٢١٩لا ينفرن أحد حتى يكون آخر عهده بالبيت٣٢٢٠قال أمر الناس أن يكون، آخر عهدهم بالبيت إلا أنه خفف عن المرأة الحائض .٣٢٢١قال كنت مع ابن عباس إذ قال زيد بن ثابت تفتي أن تصدر الحائض، قبل أن…٣٢٣٠حدثنا يحيى بن يحيى التميمي، قال قرأت على مالك عن نافع، عن ابن عمر، أن…٣٢٣١وقتيبة بن سعيد، وأبو كامل الجحدري كلهم عن حماد بن زيد، - قال أبو كامل…٣٢٣٢عمر قال أقبل رسول الله صلى الله عليه وسلم عام الفتح على ناقة لأسامة…٣٢٣٣قال دخل رسول الله صلى الله عليه وسلم البيت ومعه أسامة وبلال وعثمان بن…٣٢٣٤وحدثني حميد بن مسعدة، حدثنا خالد، - يعني ابن الحارث - حدثنا عبد الله،…٣٢٣٥أنه قال دخل رسول الله صلى الله عليه وسلم البيت هو وأسامة بن زيد وبلال…٣٢٣٦وحدثني حرملة بن يحيى، أخبرنا ابن وهب، أخبرني يونس، عن ابن شهاب،…٣٢٣٧وعبد بن حميد، جميعا عن ابن بكر، قال عبد أخبرنا محمد بن بكر، أخبرنا…٣٢٣٨أن النبي صلى الله عليه وسلم دخل الكعبة وفيها ست سوار فقام عند سارية…٣٢٣٩قال قلت لعبد الله بن أبي أوفى صاحب رسول الله صلى الله عليه وسلم أدخل…٣٢٤٠لولا حداثة عهد قومك بالكفر لنقضت الكعبة ولجعلتها على أساس إبراهيم فإن…٣٢٤١بهذا الإسناد .٣٢٤٢ألم ترى أن قومك حين بنوا الكعبة اقتصروا عن قواعد إبراهيم٣٢٤٣لولا أن قومك حديثو عهد بجاهلية - أو قال بكفر - لأنفقت كنز الكعبة في…٣٢٤٤يا عائشة لولا أن قومك حديثو عهد بشرك لهدمت الكعبة فألزقتها بالأرض…٣٢٤٥لولا أن الناس حديث عهدهم بكفر وليس عندي من النفقة ما يقوي على بنائه…٣٢٤٦إن قومك استقصروا من بنيان البيت ولولا حداثة عهدهم بالشرك أعدت ما…٣٢٤٧بهذا الإسناد . مثل حديث ابن بكر .٣٢٤٨يا عائشة لولا حدثان قومك بالكفر لنقضت البيت حتى أزيد فيه من الحجر فإن…٣٢٤٩حدثنا سعيد بن منصور، حدثنا أبو الأحوص، حدثنا أشعث بن أبي الشعثاء، عن…٣٢٥٠مخافة أن تنفر قلوبهم٣٢٥١أنه قال كان الفضل بن عباس رديف رسول الله صلى الله عليه وسلم فجاءته…٣٢٥٢أن امرأة، من خثعم قالت يا رسول الله إن أبي شيخ كبير عليه فريضة الله…٣٢٥٣وزهير بن حرب، وابن أبي عمر، جميعا عن ابن، عيينة - قال أبو بكر حدثنا…٣٢٥٤قال رفعت امرأة صبيا لها فقالت يا رسول الله ألهذا حج قال " نعم ولك أجر…٣٢٥٥أن امرأة، رفعت صبيا فقالت يا رسول الله ألهذا حج قال " نعم ولك أجر " .٣٢٥٦بمثله .٣٢٥٧أيها الناس قد فرض الله عليكم الحج فحجوا٣٢٥٨لا تسافر المرأة ثلاثا إلا ومعها ذو محرم٣٢٥٩ثلاثة إلا ومعها ذو محرم٣٢٦٠لا يحل لامرأة تؤمن بالله واليوم الآخر تسافر مسيرة ثلاث ليال إلا ومعها…٣٢٦١لا تسافر امرأة فوق ثلاث ليال إلا مع ذي محرم٣٢٦٣لا تشدوا الرحال إلا إلى ثلاثة مساجد مسجدي هذا والمسجد الحرام والمسجد…٣٢٦٤رسول الله صلى الله عليه وسلم أربعا فأعجبنني وآنقنني نهى أن تسافر…
