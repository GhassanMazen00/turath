كتاب الحج

٥٨٣ حديثًا•صفحة ٤ من ٤
٣٢٦٥أكثر من ثلاث إلا مع ذي محرم٣٢٦٦لا يحل لامرأة مسلمة تسافر مسيرة ليلة إلا ومعها رجل ذو حرمة منها٣٢٦٧لا يحل لامرأة تؤمن بالله واليوم الآخر تسافر مسيرة يوم إلا مع ذي محرم٣٢٦٨لا يحل لامرأة تؤمن بالله واليوم الآخر تسافر مسيرة يوم وليلة إلا مع ذي…٣٢٦٩لا يحل لامرأة أن تسافر ثلاثا إلا ومعها ذو محرم منها٣٢٧٠لا يحل لامرأة تؤمن بالله واليوم الآخر أن تسافر سفرا يكون ثلاثة أيام…٣٢٧١بهذا الإسناد مثله .٣٢٧٢لا يخلون رجل بامرأة إلا ومعها ذو محرم ولا تسافر المرأة إلا مع ذي محرم٣٢٧٣بهذا الإسناد نحوه .٣٢٧٤لا يخلون رجل بامرأة إلا ومعها ذو محرم٣٢٧٥سبحان الذي سخر لنا هذا وما كنا له مقرنين وإنا إلى ربنا لمنقلبون اللهم…٣٢٧٦الله بن سرجس قال كان رسول الله صلى الله عليه وسلم إذا سافر يتعوذ من…٣٢٧٧اللهم إني أعوذ بك من وعثاء السفر٣٢٧٨لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…٣٢٧٩وحدثني زهير بن حرب، حدثنا إسماعيل، - يعني ابن علية - عن أيوب، ح…٣٢٨٠آيبون تائبون عابدون لربنا حامدون٣٢٨١وحدثنا حميد بن مسعدة، حدثنا بشر بن المفضل، حدثنا يحيى بن أبي إسحاق،…٣٢٨٢أن رسول الله صلى الله عليه وسلم أناخ بالبطحاء التي بذي الحليفة فصلى…٣٢٨٣قال كان ابن عمر ينيخ بالبطحاء التي بذي الحليفة التي كان رسول الله صلى…٣٢٨٤أن عبد الله بن عمر، كان إذا صدر من الحج أو العمرة أناخ بالبطحاء التي…٣٢٨٥أن رسول الله صلى الله عليه وسلم أتي في معرسه بذي الحليفة فقيل له إنك…٣٢٨٦وسريج بن يونس، - واللفظ لسريج - قالا حدثنا إسماعيل بن جعفر، أخبرني…٣٢٨٧أن ابن شهاب، أخبره عن حميد بن عبد الرحمن بن عوف، عن أبي هريرة، قال…٣٢٨٨ما من يوم أكثر من أن يعتق الله فيه عبدا من النار من يوم عرفة وإنه…٣٢٨٩العمرة إلى العمرة كفارة لما بينهما والحج المبرور ليس له جزاء إلا الجنة٣٢٩٠وحدثناه سعيد بن منصور، وأبو بكر بن أبي شيبة وعمرو الناقد وزهير بن حرب…٣٢٩١من أتى هذا البيت فلم يرفث ولم يفسق رجع كما ولدته أمه٣٢٩٢من حج فلم يرفث ولم يفسق٣٢٩٣حدثنا سعيد بن منصور، حدثنا هشيم، عن سيار، عن أبي حازم، عن أبي هريرة،…٣٢٩٤وهل ترك لنا عقيل من رباع أو دور٣٢٩٥وهل ترك لنا عقيل منزلا٣٢٩٦وهل ترك لنا عقيل من منزل٣٢٩٧للمهاجر إقامة ثلاث بعد الصدر بمكة٣٢٩٨يقيم المهاجر بمكة بعد قضاء نسكه ثلاثا٣٢٩٩ثلاث ليال يمكثهن المهاجر بمكة بعد الصدر٣٣٠٠مكث المهاجر بمكة بعد قضاء نسكه ثلاث٣٣٠١بهذا الإسناد مثله .٣٣٠٢لا هجرة ولكن جهاد ونية وإذا استنفرتم فانفروا٣٣٠٣يوم خلق السموات والأرض٣٣٠٤إن مكة حرمها الله ولم يحرمها الناس فلا يحل لامرئ يؤمن بالله واليوم…٣٣٠٥إن الله حبس عن مكة الفيل وسلط عليها رسوله والمؤمنين وإنها لن تحل لأحد…٣٣٠٦إن الله عز وجل حبس عن مكة الفيل وسلط عليها رسوله والمؤمنين ألا وإنها…٣٣٠٧لا يحل لأحدكم أن يحمل بمكة السلاح٣٣٠٨ويحيى بن يحيى، وقتيبة بن سعيد، أما القعنبي فقال قرأت على مالك بن أنس…٣٣٠٩وقتيبة بن سعيد الثقفي، قال يحيى أخبرنا وقال، قتيبة حدثنا معاوية بن…٣٣١٠أن النبي صلى الله عليه وسلم دخل يوم فتح مكة وعليه عمامة سوداء .٣٣١١وإسحاق بن إبراهيم، قالا أخبرنا وكيع، عن مساور، الوراق عن جعفر بن عمرو…٣٣١٢والحسن الحلواني، قالا حدثنا أبو أسامة، عن مساور الوراق، قال حدثني…٣٣١٣إن إبراهيم حرم مكة ودعا لأهلها وإني حرمت المدينة كما حرم إبراهيم مكة…٣٣١٤بمثلى ما دعا به إبراهيم٣٣١٥إن إبراهيم حرم مكة وإني أحرم ما بين لابتيها٣٣١٦أن مروان بن الحكم، خطب الناس فذكر مكة وأهلها وحرمتها ولم يذكر المدينة…٣٣١٧إن إبراهيم حرم مكة وإني حرمت المدينة ما بين لابتيها لا يقطع عضاهها…٣٣١٨إني أحرم ما بين لابتى المدينة أن يقطع عضاهها أو يقتل صيدها - وقال -…٣٣١٩ولا يريد أحد أهل المدينة بسوء إلا أذابه الله في النار ذوب الرصاص أو…٣٣٢٠وعبد بن حميد، جميعا عن العقدي، - قال عبد أخبرنا عبد الملك بن عمرو، -…٣٣٢١التمس لي غلاما من غلمانكم يخدمني٣٣٢٢إني أحرم ما بين لابتيها٣٣٢٣من أحدث فيها حدثا فعليه لعنة الله والملائكة والناس أجمعين لا يقبل…٣٣٢٤قال سألت أنسا أحرم رسول الله صلى الله عليه وسلم المدينة قال نعم هي…٣٣٢٥اللهم بارك لهم في مكيالهم وبارك لهم في صاعهم وبارك لهم في مدهم٣٣٢٦اللهم اجعل بالمدينة ضعفى ما بمكة من البركة٣٣٢٧المدينة حرم ما بين عير إلى ثور فمن أحدث فيها حدثا أو آوى محدثا فعليه…٣٣٢٨فمن أخفر مسلما فعليه لعنة الله والملائكة والناس أجمعين لا يقبل منه…٣٣٢٩من تولى غير مواليه٣٣٣٠المدينة حرم فمن أحدث فيها حدثا أو آوى محدثا فعليه لعنة الله والملائكة…٣٣٣١بهذا الإسناد . مثله ولم يقل " يوم القيامة " وزاد " وذمة المسلمين…٣٣٣٢ما بين لابتيها حرام٣٣٣٣ومحمد بن رافع، وعبد بن حميد، قال إسحاق أخبرنا عبد الرزاق، حدثنا معمر،…٣٣٣٤اللهم بارك لنا في ثمرنا وبارك لنا في مدينتنا وبارك لنا في صاعنا وبارك…٣٣٣٥اللهم بارك لنا في مدينتنا وفي ثمارنا وفي مدنا وفي صاعنا بركة مع بركة٣٣٣٦ما هذا الذي بلغني من حديثكم - ما أدري كيف قال - والذي أحلف به أو…٣٣٣٧اللهم بارك لنا في صاعنا ومدنا واجعل مع البركة بركتين٣٣٣٨بهذا الإسناد . مثله .٣٣٣٩لا يصبر أحد على لأوائها فيموت إلا كنت له شفيعا أو شهيدا يوم القيامة…٣٣٤٠إني حرمت ما بين لابتى المدينة كما حرم إبراهيم مكة٣٣٤١قال أهوى رسول الله صلى الله عليه وسلم بيده إلى المدينة فقال " إنها…٣٣٤٢اللهم حبب إلينا المدينة كما حببت مكة أو أشد وصححها وبارك لنا في صاعها…٣٣٤٣وابن، نمير عن هشام بن عروة، بهذا الإسناد . نحوه .٣٣٤٤من صبر على لأوائها كنت له شفيعا أو شهيدا يوم القيامة٣٣٤٥لا يصبر على لأوائها وشدتها أحد إلا كنت له شهيدا أو شفيعا يوم القيامة٣٣٤٦من صبر على لأوائها وشدتها كنت له شهيدا أو شفيعا يوم القيامة٣٣٤٧لا يصبر على لأواء المدينة وشدتها أحد من أمتي إلا كنت له شفيعا يوم…٣٣٤٨موسى بن أبي عيسى أنه سمع أبا عبد الله القراظ، يقول سمعت أبا هريرة،…٣٣٤٩لا يصبر أحد على لأواء المدينة٣٣٥٠على أنقاب المدينة ملائكة لا يدخلها الطاعون ولا الدجال٣٣٥١يأتي المسيح من قبل المشرق همته المدينة حتى ينزل دبر أحد ثم تصرف…٣٣٥٢يأتي على الناس زمان يدعو الرجل ابن عمه وقريبه هلم إلى الرخاء هلم إلى…٣٣٥٣أمرت بقرية تأكل القرى يقولون يثرب وهى المدينة تنفي الناس كما ينفي…٣٣٥٤كما ينفي الكير الخبث٣٣٥٥إنما المدينة كالكير تنفي خبثها وينصع طيبها٣٣٥٦إنها طيبة - يعني المدينة - وإنها تنفي الخبث كما تنفي النار خبث الفضة٣٣٥٧إن الله تعالى سمى المدينة طابة٣٣٥٨من أراد أهل هذه البلدة بسوء - يعني المدينة - أذابه الله كما يذوب…٣٣٥٩من أراد أهلها بسوء - يريد المدينة - أذابه الله كما يذوب الملح في الماء٣٣٦٠جميعا سمعا أبا عبد الله، القراظ سمع أبا هريرة، عن النبي صلى الله عليه…٣٣٦١من أراد أهل المدينة بسوء أذابه الله كما يذوب الملح في الماء٣٣٦٢أنه سمع سعد بن مالك، يقول قال رسول الله صلى الله عليه وسلم . بمثله…٣٣٦٣اللهم بارك لأهل المدينة في مدهم٣٣٦٤يفتح الشام فيخرج من المدينة قوم بأهليهم يبسون والمدينة خير لهم لو…٣٣٦٥يفتح اليمن فيأتي قوم يبسون فيتحملون بأهليهم ومن أطاعهم والمدينة خير…٣٣٦٦ليتركنها أهلها على خير ما كانت مذللة للعوافي٣٣٦٧يتركون المدينة على خير ما كانت لا يغشاها إلا العوافي - يريد عوافي…٣٣٦٨ما بين بيتي ومنبري روضة من رياض الجنة٣٣٦٩ما بين منبري وبيتي روضة من رياض الجنة٣٣٧٠ما بين بيتي ومنبري روضة من رياض الجنة ومنبري على حوضي٣٣٧١إني مسرع فمن شاء منكم فليسرع معي ومن شاء فليمكث٣٣٧٢إن أحدا جبل يحبنا ونحبه٣٣٧٣إن أحدا جبل يحبنا ونحبه٣٣٧٤صلاة في مسجدي هذا أفضل من ألف صلاة فيما سواه إلا المسجد الحرام٣٣٧٥صلاة في مسجدي هذا خير من ألف صلاة في غيره من المساجد إلا المسجد الحرام٣٣٧٦فإني آخر الأنبياء وإن مسجدي آخر المساجد٣٣٧٧صلاة في مسجدي هذا خير من ألف صلاة - أو كألف صلاة - فيما سواه من…٣٣٧٨وعبيد الله بن سعيد، ومحمد بن حاتم، قالوا حدثنا يحيى، القطان عن يحيى…٣٣٧٩صلاة في مسجدي هذا أفضل من ألف صلاة فيما سواه إلا المسجد الحرام٣٣٨٠بهذا الإسناد .٣٣٨١وحدثني إبراهيم بن موسى، أخبرنا ابن أبي زائدة، عن موسى الجهني، عن…٣٣٨٢وحدثناه ابن أبي عمر، حدثنا عبد الرزاق، أخبرنا معمر، عن أيوب، عن نافع،…٣٣٨٣صلاة فيه أفضل من ألف صلاة فيما سواه من المساجد إلا مسجد الكعبة٣٣٨٤لا تشد الرحال إلا إلى ثلاثة مساجد مسجدي هذا ومسجد الحرام ومسجد الأقصى٣٣٨٥تشد الرحال إلى ثلاثة مساجد٣٣٨٦إنما يسافر إلى ثلاثة مساجد مسجد الكعبة ومسجدي ومسجد إيلياء٣٣٨٧حدثني محمد بن حاتم، حدثنا يحيى بن سعيد، عن حميد الخراط، قال سمعت أبا…٣٣٨٨وسعيد بن عمرو الأشعثي، قال سعيد أخبرنا وقال أبو بكر، حدثنا حاتم بن…٣٣٨٩أن رسول الله صلى الله عليه وسلم كان يزور قباء راكبا وماشيا .٣٣٩٠عمر قال كان رسول الله صلى الله عليه وسلم يأتي مسجد قباء راكبا وماشيا…٣٣٩١عمر أن رسول الله صلى الله عليه وسلم كان يأتي قباء راكبا وماشيا .٣٣٩٢وحدثني أبو معن الرقاشي، زيد بن يزيد الثقفي - بصري ثقة - حدثنا خالد، -…٣٣٩٣الله بن عمر أن رسول الله صلى الله عليه وسلم كان يأتي قباء راكبا…٣٣٩٤وقتيبة، وابن، حجر قال ابن أيوب حدثنا إسماعيل بن، جعفر أخبرني عبد الله…٣٣٩٥أن ابن، عمر كان يأتي قباء كل سبت وكان يقول رأيت النبي صلى الله عليه…٣٣٩٦عمر أن رسول الله صلى الله عليه وسلم كان يأتي قباء يعني كل سبت كان…٣٣٩٧بهذا الإسناد . ولم يذكر كل سبت .
