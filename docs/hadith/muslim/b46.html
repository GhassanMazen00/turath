٥٢ حديثًا
٦٧٢٣إن أحدكم يجمع خلقه في بطن أمه أربعين يوما ثم يكون في ذلك علقة مثل ذلك…٦٧٢٤إن خلق أحدكم يجمع في بطن أمه أربعين ليلة٦٧٢٥يدخل الملك على النطفة بعد ما تستقر في الرحم بأربعين أو خمسة وأربعين…٦٧٢٦إذا مر بالنطفة ثنتان وأربعون ليلة بعث الله إليها ملكا فصورها وخلق…٦٧٢٧أن أبا الطفيل، أخبره أنه، سمع عبد الله بن مسعود، يقول . وساق الحديث…٦٧٢٨إن النطفة تقع في الرحم أربعين ليلة ثم يتصور عليها الملك٦٧٢٩أن ملكا موكلا بالرحم إذا أراد الله أن يخلق شيئا بإذن الله لبضع…٦٧٣٠إن الله عز وجل قد وكل بالرحم ملكا فيقول أى رب نطفة أى رب علقة أى رب…٦٧٣١ما منكم من أحد ما من نفس منفوسة إلا وقد كتب الله مكانها من الجنة…٦٧٣٢وهناد بن السري، قالا حدثنا أبو الأحوص، عن منصور، بهذا الإسناد في…٦٧٣٣ما منكم من نفس إلا وقد علم منزلها من الجنة والنار٦٧٣٤وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن منصور، والأعمش،…٦٧٣٥لا . بل فيما جفت به الأقلام وجرت به المقادير٦٧٣٦كل عامل ميسر لعمله٦٧٣٧قال قيل يا رسول الله أعلم أهل الجنة من أهل النار قال فقال " نعم " .…٦٧٣٨في هذا الإسناد . بمعنى حديث حماد وفي حديث عبد الوارث قال قلت يا رسول…٦٧٣٩لا بل شىء قضي عليهم ومضى فيهم وتصديق ذلك في كتاب الله عز وجل { ونفس…٦٧٤٠إن الرجل ليعمل الزمن الطويل بعمل أهل الجنة ثم يختم له عمله بعمل أهل…٦٧٤١إن الرجل ليعمل عمل أهل الجنة فيما يبدو للناس وهو من أهل النار . وإن…٦٧٤٢احتج آدم وموسى فقال موسى يا آدم أنت أبونا خيبتنا وأخرجتنا من الجنة…٦٧٤٣تحاج آدم وموسى فحج آدم موسى فقال له موسى أنت آدم الذي أغويت الناس…٦٧٤٤احتج آدم وموسى عليهما السلام عند ربهما فحج آدم موسى قال موسى أنت آدم…٦٧٤٥احتج آدم وموسى فقال له موسى أنت آدم الذي أخرجتك خطيئتك من الجنة فقال…٦٧٤٦حدثني عمرو الناقد، حدثنا أيوب بن النجار اليمامي، حدثنا يحيى بن أبي…٦٧٤٧وحدثنا ابن رافع، حدثنا عبد الرزاق، أخبرنا معمر، عن همام بن منبه، عن…٦٧٤٨وحدثنا محمد بن منهال الضرير، حدثنا يزيد بن زريع، حدثنا هشام بن حسان،…٦٧٤٩كتب الله مقادير الخلائق قبل أن يخلق السموات والأرض بخمسين ألف سنة -…٦٧٥٠. بهذا الإسناد مثله غير أنهما لم يذكرا وعرشه على الماء .٦٧٥١إن قلوب بني آدم كلها بين إصبعين من أصابع الرحمن كقلب واحد يصرفه حيث…٦٧٥٢كل شىء بقدر حتى العجز والكيس أو الكيس والعجز٦٧٥٣حدثنا أبو بكر بن أبي شيبة، وأبو كريب قالا حدثنا وكيع، عن سفيان، عن…٦٧٥٤إن الله كتب على ابن آدم حظه من الزنى أدرك ذلك لا محالة فزنى العينين…٦٧٥٥كتب على ابن آدم نصيبه من الزنى مدرك ذلك لا محالة فالعينان زناهما…٦٧٥٦ما من مولود إلا يولد على الفطرة فأبواه يهودانه وينصرانه ويمجسانه كما…٦٧٥٧كما تنتج البهيمة بهيمة٦٧٥٨ما من مولود إلا يولد على الفطرة٦٧٥٩ما من مولود إلا يلد على الفطرة فأبواه يهودانه وينصرانه ويشركانه٦٧٦٠ما من مولود يولد إلا وهو على الملة٦٧٦١من يولد يولد على هذه الفطرة فأبواه يهودانه وينصرانه كما تنتجون الإبل…٦٧٦٢كل إنسان تلده أمه على الفطرة وأبواه بعد يهودانه وينصرانه ويمجسانه فإن…٦٧٦٣الله أعلم بما كانوا عاملين٦٧٦٤بإسناد يونس وابن أبي ذئب . مثل حديثهما غير أن في، حديث شعيب ومعقل سئل…٦٧٦٥الله أعلم بما كانوا عاملين٦٧٦٦الله أعلم بما كانوا عاملين إذ خلقهم٦٧٦٧إن الغلام الذي قتله الخضر طبع كافرا ولو عاش لأرهق أبويه طغيانا وكفرا٦٧٦٨أولا تدرين أن الله خلق الجنة وخلق النار فخلق لهذه أهلا ولهذه أهلا٦٧٦٩أوغير ذلك يا عائشة إن الله خلق للجنة أهلا خلقهم لها وهم في أصلاب…٦٧٧٠بإسناد وكيع نحو حديثه .٦٧٧١اللهم أمتعني بزوجي رسول الله صلى الله عليه وسلم وبأبي أبي سفيان وبأخي…٦٧٧٢من عذاب في النار وعذاب في القبر٦٧٧٣إنك سألت الله لآجال مضروبة وآثار موطوءة وأرزاق مقسومة لا يعجل شيئا…٦٧٧٤المؤمن القوي خير وأحب إلى الله من المؤمن الضعيف وفي كل خير احرص على…
