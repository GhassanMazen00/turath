كتاب اللباس والزينة

١٨٤ حديثًا•صفحة ٢ من ٢
٥٥٤٦لا تصحب الملائكة رفقة فيها كلب ولا جرس٥٥٤٧بهذا الإسناد .٥٥٤٨الجرس مزامير الشيطان٥٥٤٩لا يبقين في رقبة بعير قلادة من وتر أو قلادة إلا قطعت٥٥٥٠حدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن ابن جريج، عن أبي…٥٥٥١أنه سمع جابر، بن عبد الله يقول نهى رسول الله صلى الله عليه وسلم بمثله…٥٥٥٢لعن الله الذي وسمه٥٥٥٣بن أبي حبيب أن ناعما أبا عبد الله، مولى أم سلمة حدثه أنه، سمع ابن…٥٥٥٤قال لما ولدت أم سليم قالت لي يا أنس انظر هذا الغلام فلا يصيبن شيئا…٥٥٥٥يحدث أن أمه، حين ولدت انطلقوا بالصبي إلى النبي صلى الله عليه وسلم…٥٥٥٨قال رأيت في يد رسول الله صلى الله عليه وسلم الميسم وهو يسم إبل الصدقة…٥٥٥٩حدثني زهير بن حرب، حدثني يحيى، - يعني ابن سعيد - عن عبيد الله، أخبرني…٥٥٦٠بهذا الإسناد . وجعل التفسير في حديث أبي أسامة من قول عبيد الله .٥٥٦١نافع بإسناد عبيد الله . مثله وألحقا التفسير في الحديث .٥٥٦٢وحجاج بن الشاعر، وعبد بن حميد، عن عبد الرزاق، عن معمر، عن أيوب، ح…٥٥٦٣إياكم والجلوس في الطرقات٥٥٦٤أسلم بهذا الإسناد مثله .٥٥٦٥لعن الله الواصلة والمستوصلة٥٥٦٦بهذا الإسناد . نحو حديث أبي معاوية غير أنحديثهما فتمرط شعرها .٥٥٦٧أن امرأة، أتت النبي صلى الله عليه وسلم فقالت إني زوجت ابنتي فتمرق شعر…٥٥٦٨وابن، بشار قالا حدثنا أبو داود، حدثنا شعبة، ح وحدثنا أبو بكر بن أبي…٥٥٧٣وعثمان بن أبي شيبة، - واللفظ لإسحاق - أخبرنا جرير، عن منصور، عن…٥٥٧٤وابن، بشار قالا حدثنا عبد الرحمن، وهو ابن مهدي حدثنا سفيان، ح وحدثنا…٥٥٧٥ومحمد بن المثنى، وابن، بشار قالوا حدثنا محمد، بن جعفر حدثنا شعبة، عن…٥٥٧٦وحدثنا شيبان بن فروخ، حدثنا جرير، - يعني ابن حازم - حدثنا الأعمش، عن…٥٥٧٧ومحمد بن رافع، قالا أخبرنا عبد الرزاق، أخبرنا ابن جريج، أخبرني أبو…٥٥٧٨إنما هلكت بنو إسرائيل حين اتخذ هذه نساؤهم٥٥٧٩إنما عذب بنو إسرائيل٥٥٨٠وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن عمرو بن مرة، عن…٥٥٨١ومحمد بن المثنى، قالا أخبرنا معاذ، - وهو ابن هشام - حدثني أبي، عن…٥٥٨٢صنفان من أهل النار لم أرهما قوم معهم سياط كأذناب البقر يضربون بها…٥٥٨٣المتشبع بما لم يعط كلابس ثوبى زور٥٥٨٤المتشبع بما لم يعط كلابس ثوبى زور٥٥٨٥بهذا الإسناد .
