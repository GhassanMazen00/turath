كتاب الحج

٥٨٣ حديثًا•صفحة ١ من ٤
٢٧٩١لا تلبسوا القمص ولا العمائم ولا السراويلات ولا البرانس ولا الخفاف إلا…٢٧٩٢لا يلبس المحرم القميص ولا العمامة ولا البرنس ولا السراويل ولا ثوبا…٢٧٩٣من لم يجد نعلين فليلبس الخفين وليقطعهما أسفل من الكعبين٢٧٩٤السراويل لمن لم يجد الإزار والخفان لمن لم يجد النعلين٢٧٩٥بهذا الإسناد أنه سمع النبي صلى الله عليه وسلم يخطب بعرفات . فذكر هذا…٢٧٩٦بهذا الإسناد ولم يذكر أحد منهم يخطب بعرفات . غير شعبة وحده .٢٧٩٧من لم يجد نعلين فليلبس خفين ومن لم يجد إزارا فليلبس سراويل٢٧٩٨أين السائل عن العمرة اغسل عنك أثر الصفرة - أو قال أثر الخلوق - واخلع…٢٧٩٩ما كنت صانعا في حجك٢٨٠٠أين الذي سألني عن العمرة آنفا٢٨٠١انزع عنك الجبة واغسل عنك الصفرة وما كنت صانعا في حجك فاصنعه في عمرتك٢٨٠٢أين السائل آنفا عن العمرة٢٨٠٣فهن لهن ولمن أتى عليهن من غير أهلهن ممن أراد الحج والعمرة فمن كان…٢٨٠٤هن لهم ولكل آت أتى عليهن من غيرهن ممن أراد الحج والعمرة ومن كان دون…٢٨٠٥يهل أهل المدينة من ذي الحليفة وأهل الشام من الجحفة وأهل نجد من قرن٢٨٠٦يهل أهل المدينة من ذي الحليفة ويهل أهل الشام من الجحفة ويهل أهل نجد…٢٨٠٧مهل أهل المدينة ذو الحليفة ومهل أهل الشام مهيعة وهي الجحفة ومهل أهل…٢٨٠٨أنه سمع جابر بن عبد الله، - رضى الله عنهما - يسأل عن المهل، فقال سمعت…٢٨٠٩مهل أهل المدينة من ذي الحليفة والطريق الآخر الجحفة ومهل أهل العراق من…٢٨١١لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك والملك لا…٢٨١٢لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك والملك لا…٢٨١٣- رضى الله عنهما - قال تلقفت التلبية من في رسول الله صلى الله عليه…٢٨١٤لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك والملك لا…٢٨١٥- رضى الله عنهما - قال كان المشركون يقولون لبيك لا شريك لك - قال -…٢٨١٦بن عبد الله أنه سمع أباه، - رضى الله عنه - يقول بيداؤكم هذه التي…٢٨١٧قال كان ابن عمر - رضى الله عنهما - إذا قيل له الإحرام من البيداء قال…٢٨١٨وحدثنا يحيى بن يحيى، قال قرأت على مالك عن سعيد بن أبي سعيد المقبري،…٢٨١٩قال حججت مع عبد الله بن عمر بن الخطاب - رضى الله عنهما - بين حج وعمرة…٢٨٢٠- رضى الله عنهما - قال كان رسول الله صلى الله عليه وسلم إذا وضع رجله…٢٨٢١- رضى الله عنهما - أنه كان يخبر أن النبي صلى الله عليه وسلم أهل حين…٢٨٢٢أنأخبره أن عبد الله بن عمر - رضى الله عنهما - قال رأيت رسول الله صلى…٢٨٢٣وأحمد بن عيسى، قال أحمد حدثنا وقال، حرملة أخبرنا ابن وهب، أخبرني…٢٨٢٤- رضى الله عنها - قالت طيبت رسول الله صلى الله عليه وسلم لحرمه حين…٢٨٢٥- رضى الله عنها - زوج النبي صلى الله عليه وسلم قالت طيبت رسول الله…٢٨٢٦- رضى الله عنها - أنها قالت كنت أطيب رسول الله صلى الله عليه وسلم…٢٨٢٧- رضى الله عنها - قالت طيبت رسول الله صلى الله عليه وسلم لحله ولحرمه .٢٨٢٨وعبد بن حميد، قال عبد أخبرنا وقال ابن حاتم، حدثنا محمد بن بكر، أخبرنا…٢٨٢٩وزهير بن حرب، جميعا عن ابن عيينة، - قال زهير حدثنا سفيان، - حدثنا…٢٨٣٠- رضى الله عنها - قالت كنت أطيب رسول الله صلى الله عليه وسلم بأطيب ما…٢٨٣١- رضى الله عنها - أنها قالت طيبت رسول الله صلى الله عليه وسلم لحرمه…٢٨٣٢وسعيد بن منصور، وأبو الربيع، وخلف بن هشام، وقتيبة، بن سعيد قال يحيى…٢٨٣٣- رضى الله عنها - قالت لكأني أنظر إلى وبيص الطيب في مفارق رسول الله…٢٨٣٤وزهير بن حرب، وأبو سعيد الأشج قالوا حدثنا وكيع، حدثنا الأعمش، عن أبي…٢٨٣٥- رضى الله عنها - قالت لكأني أنظر . بمثل حديث وكيع .٢٨٣٦وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن الحكم، قال سمعت…٢٨٣٧- رضى الله عنها - قالت إن كنت لأنظر إلى وبيص الطيب في مفارق رسول الله…٢٨٣٨إسحاق سمع ابن الأسود، يذكر عن أبيه، عن عائشة، - رضى الله عنها - قالت…٢٨٣٩قال قالت عائشة رضى الله عنها كأني أنظر إلى وبيص المسك في مفرق رسول…٢٨٤٠بهذا الإسناد . مثله .٢٨٤١ويعقوب الدورقي، قالا حدثنا هشيم، أخبرنا منصور، عن عبد الرحمن بن…٢٨٤٢حدثنا سعيد بن منصور، وأبو كامل جميعا عن أبي عوانة، - قال سعيد حدثنا…٢٨٤٣- رضى الله عنها - أنها قالت كنت أطيب رسول الله صلى الله عليه وسلم ثم…٢٨٤٤وسفيان، عن إبراهيم بن محمد بن، المنتشر عن أبيه، قال سمعت ابن عمر، -…٢٨٤٥إنا لم نرده عليك إلا أنا حرم٢٨٤٦ومحمد بن رمح، وقتيبة، جميعا عن الليث بن سعد، ح وحدثنا عبد بن حميد،…٢٨٤٧بهذا الإسناد وقال أهديت له من لحم حمار وحش .٢٨٤٨لولا أنا محرمون لقبلناه منك٢٨٤٩وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن الحكم، ح وحدثنا…٢٨٥٠إنا لا نأكله إنا حرم٢٨٥١وحدثنا قتيبة بن سعيد، حدثنا سفيان، عن صالح بن كيسان، ح وحدثنا ابن…٢٨٥٢إنما هي طعمة أطعمكموها الله٢٨٥٣هل معكم من لحمه شىء٢٨٥٤وحدثنا صالح بن مسمار السلمي، حدثنا معاذ بن هشام، حدثني أبي، عن يحيى،…٢٨٥٥خذوا ساحل البحر حتى تلقوني٢٨٥٦أمنكم أحد أمره أن يحمل عليها أو أشار إليها٢٨٥٧أن أباه، - رضى الله عنه - أخبره أنه، غزا مع رسول الله صلى الله عليه…٢٨٥٨هل معكم منه شىء٢٨٥٩هل أشار إليه إنسان منكم أو أمره بشىء٢٨٦٠حدثني زهير بن حرب، حدثنا يحيى بن سعيد، عن ابن جريج، أخبرني محمد، بن…٢٨٦١أربع كلهن فاسق يقتلن في الحل والحرم الحدأة والغراب والفارة والكلب…٢٨٦٢خمس فواسق يقتلن في الحل والحرم الحية والغراب الأبقع والفارة والكلب…٢٨٦٣خمس فواسق يقتلن في الحرم العقرب والفارة والحديا والغراب والكلب العقور٢٨٦٤بهذا الإسناد .٢٨٦٥خمس فواسق يقتلن في الحرم الفارة والعقرب والغراب والحديا والكلب العقور٢٨٦٦بهذا الإسناد قالت أمر رسول الله صلى الله عليه وسلم بقتل خمس فواسق في…٢٨٦٧خمس من الدواب كلها فواسق تقتل في الحرم الغراب والحدأة والكلب العقور…٢٨٦٨خمس لا جناح على من قتلهن في الحرم والإحرام الفارة والعقرب والغراب…٢٨٦٩خمس من الدواب كلها فاسق لا حرج على من قتلهن العقرب والغراب والحدأة…٢٨٧٠أن رجلا، سأل ابن عمر ما يقتل المحرم من الدواب فقال أخبرتني إحدى نسوة…٢٨٧١قال سأل رجل ابن عمر ما يقتل الرجل من الدواب وهو محرم قال حدثتني إحدى…٢٨٧٢خمس من الدواب ليس على المحرم في قتلهن جناح الغراب والحدأة والعقرب…٢٨٧٣خمس من الدواب لا جناح على من قتلهن في قتلهن الغراب والحدأة والعقرب…٢٨٧٤وابن، رمح عن الليث بن سعد، ح وحدثنا شيبان بن فروخ، حدثنا جرير، - يعني…٢٨٧٥خمس لا جناح في قتل ما قتل منهن في الحرم٢٨٧٦خمس من قتلهن وهو حرام فلا جناح عليه فيهن العقرب والفارة والكلب العقور…٢٨٧٧أيؤذيك هوام رأسك٢٨٧٨وزهير بن حرب، ويعقوب بن إبراهيم، جميعا عن ابن علية، عن أيوب، في هذا…٢٨٧٩- رضى الله عنه - قال في أنزلت هذه الآية { فمن كان منكم مريضا أو به…٢٨٨٠- رضى الله عنه - أن رسول الله صلى الله عليه وسلم وقف عليه ورأسه…٢٨٨١أيؤذيك هوامك هذه٢٨٨٢رضى الله عنه أن رسول الله صلى الله عليه وسلم مر به زمن الحديبية فقال…٢٨٨٣ما كنت أرى أن الجهد بلغ منك ما أرى أتجد شاة٢٨٨٤- رضى الله عنه - أنه خرج مع النبي صلى الله عليه وسلم محرما فقمل رأسه…٢٨٨٥وزهير بن حرب، وإسحاق بن إبراهيم، قال إسحاق أخبرنا وقال الآخران، حدثنا…٢٨٨٦أن النبي صلى الله عليه وسلم احتجم بطريق مكة وهو محرم وسط رأسه .٢٨٨٧وعمرو الناقد، وزهير بن حرب، جميعا عن ابن، عيينة - قال أبو بكر حدثنا…٢٨٨٨أن عمر بن عبيد الله بن معمر، رمدت عينه فأراد أن يكحلها، فنهاه أبان بن…٢٨٨٩وعمرو الناقد، وزهير بن حرب، وقتيبة بن سعيد، قالوا حدثنا سفيان بن…٢٨٩٠وعلي بن خشرم، قالا أخبرنا عيسى بن يونس، حدثنا ابن جريج، أخبرني زيد بن…٢٨٩١اغسلوه بماء وسدر وكفنوه في ثوبيه ولا تخمروا رأسه فإن الله يبعثه يوم…٢٨٩٤اغسلوه بماء وسدر وألبسوه ثوبيه ولا تخمروا رأسه فإنه يأتي يوم القيامة…٢٨٩٥فإنه يبعث يوم القيامة ملبيا٢٨٩٦اغسلوه بماء وسدر وكفنوه في ثوبيه ولا تخمروا رأسه ولا وجهه فإنه يبعث…٢٨٩٧اغسلوه بماء وسدر وكفنوه في ثوبيه ولا تمسوه بطيب ولا تخمروا رأسه فإنه…٢٨٩٨- رضى الله عنهما - أن رجلا، وقصه بعيره وهو محرم مع رسول الله صلى الله…٢٨٩٩أنه سمع ابن عباس، - رضى الله عنهما - يحدث أن رجلا أتى النبي صلى الله…٢٩٠٠يقول قال ابن عباس - رضى الله عنهما - وقصت رجلا راحلته وهو مع رسول…٢٩٠١اغسلوه ولا تقربوه طيبا ولا تغطوا وجهه فإنه يبعث يلبي٢٩٠٢- رضى الله عنها - قالت دخل رسول الله صلى الله عليه وسلم على ضباعة بنت…٢٩٠٣حجي واشترطي أن محلي حيث حبستني٢٩٠٤- رضى الله عنها - مثله .٢٩٠٥أهلي بالحج واشترطي أن محلي حيث تحبسني٢٩٠٦وعكرمة، عن ابن عباس، رضى الله عنهما أن ضباعة، أرادت الحج فأمرها النبي…٢٩٠٧حجي واشترطي أن محلي حيث تحبسني٢٩٠٨وزهير بن حرب، وعثمان بن أبي شيبة، كلهم عن عبدة، - قال زهير حدثنا عبدة…٢٩٠٩- رضى الله عنهما - في حديث أسماء بنت عميس حين نفست بذي الحليفة أن…٢٩١٠من كان معه هدى فليهل بالحج مع العمرة ثم لا يحل حتى يحل منهما جميعا٢٩١١من أحرم بعمرة ولم يهد فليحلل ومن أحرم بعمرة وأهدى فلا يحل حتى ينحر…٢٩١٢من كان معه هدى فليهلل بالحج مع عمرته ثم لا يحل حتى يحل منهما جميعا٢٩١٣من أراد منكم أن يهل بحج وعمرة فليفعل ومن أراد أن يهل بحج فليهل ومن…٢٩١٤من أراد منكم أن يهل بعمرة فليهل فلولا أني أهديت لأهللت بعمرة٢٩١٥من أحب منكم أن يهل بعمرة فليهل بعمرة٢٩١٦وحدثنا أبو كريب، حدثنا وكيع، حدثنا هشام، عن أبيه، عن عائشة، - رضى…٢٩١٧حدثنا يحيى بن يحيى، قال قرأت على مالك عن أبي الأسود، محمد بن عبد…٢٩١٨وعمرو الناقد، وزهير بن حرب، جميعا عن ابن، عيينة - قال عمرو حدثنا…٢٩١٩حدثني سليمان بن عبيد الله أبو أيوب الغيلاني، حدثنا أبو عامر عبد…٢٩٢٠وحدثني أبو أيوب الغيلاني، حدثنا بهز، حدثنا حماد، عن عبد الرحمن، عن…٢٩٢١رضى الله عنها أن رسول الله صلى الله عليه وسلم أفرد الحج .٢٩٢٢من لم يكن معه منكم هدى فأحب أن يجعلها عمرة فليفعل ومن كان معه هدى فلا٢٩٢٣حدثني يحيى بن أيوب، حدثنا عباد بن عباد المهلبي، حدثنا عبيد الله بن…٢٩٢٤قال جاءت عائشة حاجة .٢٩٢٥وحدثنا عبد الله بن مسلمة بن قعنب، حدثنا سليمان، - يعني ابن بلال - عن…٢٩٢٦يقول أخبرتني عمرة، أنها سمعت عائشة، رضى الله عنها ح. وحدثناه ابن أبي…٢٩٢٧انتظري فإذا طهرت فاخرجي إلى التنعيم فأهلي منه ثم القينا عند كذا وكذا…٢٩٢٨وإبراهيم، - قال لا أعرف حديث أحدهما من الآخر - أن أم المؤمنين - رضى…٢٩٢٩أوما كنت طفت ليالي قدمنا مكة٢٩٣٠وحدثناه سويد بن سعيد، عن علي بن مسهر، عن الأعمش، عن إبراهيم، عن…٢٩٣١أوما شعرت أني أمرت الناس بأمر فإذا هم يترددون قال الحكم كأنهم يترددون…٢٩٣٢سمع علي بن، الحسين عن ذكوان، عن عائشة، - رضى الله عنها - قالت قدم…٢٩٣٣يسعك طوافك لحجك وعمرتك٢٩٣٤يجزئ عنك طوافك بالصفا والمروة عن حجك وعمرتك٢٩٣٥وحدثنا يحيى بن حبيب الحارثي، حدثنا خالد بن الحارث، حدثنا قرة، حدثنا…٢٩٣٦وابن، نمير قالا حدثنا سفيان، عن عمرو، أخبره عمرو بن أوس، أخبرني عبد…٢٩٣٧ومحمد بن رمح، جميعا عن الليث بن سعد، - قال قتيبة حدثنا ليث، - عن أبي…٢٩٣٨وعبد بن حميد، قال ابن حاتم حدثنا وقال عبد، أخبرنا محمد بن بكر، أخبرنا…٢٩٣٩أن عائشة، - رضى الله عنها - في حجة النبي صلى الله عليه وسلم أهلت…٢٩٤٠من لم يكن معه هدى فليحلل٢٩٤١وحدثني محمد بن حاتم، حدثنا يحيى بن سعيد، عن ابن جريج، أخبرني أبو…٢٩٤٢أنه سمع جابر، بن عبد الله - رضى الله عنه - يقول لم يطف النبي صلى الله…٢٩٤٣حلوا وأصيبوا النساء
