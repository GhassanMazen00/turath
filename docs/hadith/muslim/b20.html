٣٠ حديثًا
٣٧٧٠من أعتق شركا له في عبد فكان له مال يبلغ ثمن العبد قوم عليه قيمة العدل…٣٧٧١ومحمد بن رمح، جميعا عن الليث بن سعد، ح وحدثنا شيبان بن فروخ، حدثنا…٣٧٧٢وابن، بشار - واللفظ لابن المثنى - قالا حدثنا محمد، بن جعفر حدثنا…٣٧٧٣من أعتق شقصا له في عبد فخلاصه في ماله إن كان له مال فإن لم يكن له مال…٣٧٧٤إن لم يكن له مال قوم عليه العبد قيمة عدل ثم يستسعى في نصيب الذي لم…٣٧٧٥يحدث بهذا الإسناد بمعنى حديث ابن أبي عروبة وذكر في الحديث قوم عليه…٣٧٧٦لا يمنعك ذلك فإنما الولاء لمن أعتق٣٧٧٧ابتاعي فأعتقي . فإنما الولاء لمن أعتق٣٧٧٨لا يمنعك ذلك منها ابتاعي وأعتقي٣٧٧٩اشتريها وأعتقيها واشترطي لهم الولاء فإن الولاء لمن أعتق٣٧٨٠وإسحاق بن إبراهيم، جميعا عن جرير، كلهم عن هشام بن عروة، بهذا الإسناد…٣٧٨١اشتريها وأعتقيها فإن الولاء لمن أعتق٣٧٨٢الولاء لمن ولي النعمة٣٧٨٣اشتريها وأعتقيها فإن الولاء لمن أعتق٣٧٨٤بهذا الإسناد نحوه .٣٧٨٥وابن، بشار جميعا عن أبي هشام، قال ابن المثنى حدثنا مغيرة بن سلمة…٣٧٨٦ألم أر برمة على النار فيها لحم٣٧٨٧لا يمنعك ذلك فإنما الولاء لمن أعتق٣٧٨٨حدثنا يحيى بن يحيى التميمي، أخبرنا سليمان بن بلال، عن عبد الله بن…٣٧٨٩وزهير بن حرب، قالا حدثنا ابن عيينة، ح وحدثنا يحيى بن أيوب، وقتيبة،…٣٧٩٠أنه لا يحل لمسلم أن يتوالى مولى رجل مسلم بغير إذنه٣٧٩١من تولى قوما بغير إذن مواليه فعليه لعنة الله والملائكة لا يقبل منه…٣٧٩٣من تولى قوما بغير إذن مواليه فعليه لعنة الله والملائكة والناس أجمعين…٣٧٩٤المدينة حرم ما بين عير إلى ثور فمن أحدث فيها حدثا أو آوى محدثا فعليه…٣٧٩٥من أعتق رقبة مؤمنة أعتق الله بكل إرب منها إربا منه من النار٣٧٩٦من أعتق رقبة أعتق الله بكل عضو منها عضوا من أعضائه من النار حتى فرجه…٣٧٩٧من أعتق رقبة مؤمنة أعتق الله بكل عضو منه عضوا من النار حتى يعتق فرجه…٣٧٩٨أيما امرئ مسلم أعتق امرأ مسلما استنقذ الله بكل عضو منه عضوا منه من…٣٧٩٩لا يجزي ولد والدا إلا أن يجده مملوكا فيشتريه فيعتقه٣٨٠٠بهذا الإسناد مثله وقالوا " ولد والده " .
