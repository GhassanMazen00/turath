كتاب الحج

٥٨٣ حديثًا•صفحة ٢ من ٤
٢٩٤٤أيها الناس أحلوا فلولا الهدى الذي معي فعلت كما فعلتم٢٩٤٥أحلوا من إحرامكم فطوفوا بالبيت وبين الصفا والمروة وقصروا وأقيموا…٢٩٤٦وحدثنا محمد بن معمر بن ربعي القيسي، حدثنا أبو هشام المغيرة بن سلمة،…٢٩٤٧وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت…٢٩٤٨بهذا الإسناد وقال في الحديث فافصلوا حجكم من عمرتكم فإنه أتم لحجكم…٢٩٤٩وأبو الربيع، وقتيبة، جميعا عن حماد، - قال خلف حدثنا حماد بن زيد، - عن…٢٩٥٠رسول الله صلى الله عليه وسلم وساق الحديث بنحو حديث حاتم بن إسماعيل…٢٩٥١اغتسلي واستثفري بثوب وأحرمي٢٩٥٢نحرت ها هنا ومنى كلها منحر فانحروا في رحالكم ووقفت ها هنا وعرفة كلها…٢٩٥٣- رضى الله عنهما - أن رسول الله صلى الله عليه وسلم لما قدم مكة أتى…٢٩٥٤- رضى الله عنها - قالت كان قريش ومن دان دينها يقفون بالمزدلفة وكانوا…٢٩٥٥وحدثنا أبو كريب، حدثنا أبو أسامة، حدثنا هشام، عن أبيه، قال كانت العرب…٢٩٥٦وعمرو الناقد، جميعا عن ابن عيينة، - قال عمرو حدثنا سفيان بن عيينة، -…٢٩٥٧وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، أخبرنا شعبة، عن قيس بن…٢٩٥٨في هذا الإسناد نحوه .٢٩٥٩وحدثنا محمد بن المثنى، حدثنا عبد الرحمن، - يعني ابن مهدي - حدثنا…٢٩٦٠يا أبا موسى كيف قلت حين أحرمت٢٩٦١وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، حدثنا شعبة، عن الحكم،…٢٩٦٢وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، حدثنا شعبة، عن قتادة،…٢٩٦٣بهذا الإسناد مثله .٢٩٦٤ومحمد بن بشار، قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن عمرو بن مرة،…٢٩٦٥- رضى الله عنه - قال كانت المتعة في الحج لأصحاب محمد صلى الله عليه…٢٩٦٦وحدثنا أبو بكر بن أبي شيبة، حدثنا عبد الرحمن بن مهدي، عن سفيان، عن…٢٩٦٧وحدثنا قتيبة بن سعيد، حدثنا جرير، عن فضيل، عن زبيد، عن إبراهيم…٢٩٦٨حدثنا قتيبة، حدثنا جرير، عن بيان، عن عبد الرحمن بن أبي الشعثاء، قال…٢٩٦٩وابن أبي عمر، جميعا عن الفزاري، - قال سعيد حدثنا مروان بن معاوية، -…٢٩٧٠بهذا الإسناد وقال في روايته يعني معاوية .٢٩٧١بهذا الإسناد . مثل حديثهما وفي حديث سفيان المتعة في الحج .٢٩٧٢قال قال لي عمران بن حصين إني لأحدثك بالحديث اليوم ينفعك الله به بعد…٢٩٧٣ومحمد بن حاتم، كلاهما عن وكيع، حدثنا سفيان، عن الجريري، في هذا…٢٩٧٤قال قال لي عمران بن حصين أحدثك حديثا عسى الله أن ينفعك به إن رسول…٢٩٧٥وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن حميد بن هلال، قال…٢٩٧٦وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، عن شعبة، عن قتادة، عن…٢٩٧٧وحدثنا إسحاق بن إبراهيم، حدثنا عيسى بن يونس، حدثنا سعيد بن أبي عروبة،…٢٩٨٠ومحمد بن أبي بكر المقدمي، قالا حدثنا بشر، بن المفضل حدثنا عمران بن…٢٩٨١وحدثنيه محمد بن حاتم، حدثنا يحيى بن سعيد، عن عمران القصير، حدثنا أبو…٢٩٨٢من كان منكم أهدى فإنه لا يحل من شىء حرم منه حتى يقضي حجه ومن لم يكن…٢٩٨٣أن عائشة، زوج النبي صلى الله عليه وسلم أخبرته عن رسول الله صلى الله…٢٩٨٤إني لبدت رأسي وقلدت هديي فلا أحل حتى أنحر٢٩٨٥- رضى الله عنهم - قالت قلت يا رسول الله ما لك لم تحل بنحوه .٢٩٨٦إني قلدت هديي ولبدت رأسي فلا أحل حتى أحل من الحج٢٩٨٧فلا أحل حتى أنحر٢٩٨٨إني لبدت رأسي وقلدت هديي فلا أحل حتى أنحر هديي٢٩٨٩أن عبد الله بن عمر، - رضى الله عنهما - خرج في الفتنة معتمرا وقال إن…٢٩٩٠أن عبد الله بن عبد الله، وسالم بن عبد الله، كلما عبد الله حين نزل…٢٩٩١قال أراد ابن عمر الحج حين نزل الحجاج بابن الزبير . واقتص الحديث بمثل…٢٩٩٢أن ابن عمر، أراد الحج عام نزل الحجاج بابن الزبير فقيل له إن الناس…٢٩٩٣حدثنا أبو الربيع الزهراني، وأبو كامل قالا حدثنا حماد، ح وحدثني زهير…٢٩٩٤وعبد الله بن عون الهلالي، قالا حدثنا عباد بن عباد، المهلبي حدثنا عبيد…٢٩٩٥لبيك عمرة وحجا٢٩٩٦وحدثني أمية بن بسطام العيشي، حدثنا يزيد، - يعني ابن زريع - حدثنا…٢٩٩٧قال كنت جالسا عند ابن عمر فجاءه رجل فقال أيصلح لي أن أطوف بالبيت قبل…٢٩٩٨وحدثنا قتيبة بن سعيد، حدثنا جرير، عن بيان، عن وبرة، قال سأل رجل ابن…٢٩٩٩قدم بعمرة فطاف بالبيت ولم يطف بين الصفا والمروة أيأتي امرأته فقال قدم…٣٠٠٠وأبو الربيع الزهراني، عن حماد بن زيد، ح وحدثنا عبد، بن حميد أخبرنا…٣٠٠١أن رجلا، من أهل العراق قال له سل لي عروة بن الزبير عن رجل يهل بالحج…٣٠٠٢من كان معه هدى فليقم على إحرامه ومن لم يكن معه هدى فليحلل٣٠٠٣وحدثني عباس بن عبد العظيم العنبري، حدثنا أبو هشام المغيرة بن سلمة،…٣٠٠٤وأحمد بن عيسى، قالا حدثنا ابن وهب، أخبرني عمرو، عن أبي الأسود، أن عبد…٣٠٠٥حدثنا محمد بن حاتم، حدثنا روح بن عبادة، حدثنا شعبة، عن مسلم القري،…٣٠٠٦بهذا الإسناد فأما عبد الرحمن ففي حديثه المتعة ولم يقل متعة الحج .…٣٠٠٧سمع ابن عباس، - رضى الله عنهما - يقول أهل النبي صلى الله عليه وسلم…٣٠٠٨بهذا الإسناد غير أنه قال وكان ممن لم يكن معه الهدى طلحة بن عبيد الله…٣٠٠٩- رضى الله عنهما - قال كانوا يرون أن العمرة في أشهر الحج من أفجر…٣٠١٠من شاء أن يجعلها عمرة فليجعلها عمرة٣٠١١وحدثناه إبراهيم بن دينار، حدثنا روح، ح وحدثنا أبو داود المباركي،…٣٠١٢- رضى الله عنهما - قال قدم النبي صلى الله عليه وسلم وأصحابه لأربع…٣٠١٣- رضى الله عنهما - قال صلى رسول الله صلى الله عليه وسلم الصبح بذي طوى…٣٠١٤هذه عمرة استمتعنا بها فمن لم يكن عنده الهدى فليحل الحل كله فإن العمرة…٣٠١٥وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت أبا جمرة…٣٠١٦وابن، بشار جميعا عن ابن أبي عدي، - قال ابن المثنى حدثنا ابن أبي عدي،…٣٠١٧في هذا الإسناد . بمعنى حديث شعبة غير أنه قال إن نبي الله صلى الله…٣٠١٨وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، قال حدثنا شعبة، عن…٣٠١٩قال قيل لابن عباس إن هذا الأمر قد تفشغ بالناس من طاف بالبيت فقد حل…٣٠٢٠قال كان ابن عباس يقول لا يطوف بالبيت حاج ولا غير حاج إلا حل . قلت…٣٠٢١قال قال ابن عباس قال لي معاوية أعلمت أني قصرت من رأس رسول الله صلى…٣٠٢٢أن معاوية بن أبي سفيان، أخبره قال قصرت عن رسول الله صلى الله عليه…٣٠٢٤وحدثنا حجاج بن الشاعر، حدثنا معلى بن أسد، حدثنا وهيب بن خالد، عن…٣٠٢٥حدثني حامد بن عمر البكراوي، حدثنا عبد الواحد، عن عاصم، عن أبي نضرة،…٣٠٢٦- رضى الله عنه - أن عليا، قدم من اليمن فقال له النبي صلى الله عليه…٣٠٢٧بهذا الإسناد . مثله غير أن في رواية بهز " لحللت " .٣٠٣٠والذي نفسي بيده ليهلن ابن مريم بفج الروحاء حاجا أو معتمرا أو ليثنينهما٣٠٣١والذي نفسي بيده٣٠٣٢أن أنسا، - رضى الله عنه - أخبره أن رسول الله صلى الله عليه وسلم اعتمر…٣٠٣٣قال سألت أنسا كم حج رسول الله صلى الله عليه وسلم قال حجة واحدة واعتمر…٣٠٣٤وحدثني زهير بن حرب، حدثنا الحسن بن موسى، أخبرنا زهير، عن أبي إسحاق،…٣٠٣٥وحدثنا هارون بن عبد الله، أخبرنا محمد بن بكر البرساني، أخبرنا ابن…٣٠٣٦وحدثنا إسحاق بن إبراهيم، أخبرنا جرير، عن منصور، عن مجاهد، قال دخلت…٣٠٣٧ما منعك أن تحجي معنا٣٠٤٠أن رسول الله صلى الله عليه وسلم كان يخرج من طريق الشجرة ويدخل من طريق…٣٠٤١ومحمد بن المثنى، قالا حدثنا يحيى، - وهو القطان - عن عبيد الله، بهذا…٣٠٤٢وابن أبي عمر، جميعا عن ابن عيينة، - قال ابن المثنى حدثنا سفيان، - عن…٣٠٤٣أن رسول الله صلى الله عليه وسلم دخل عام الفتح من كداء من أعلى مكة .…٣٠٤٤وعبيد الله بن سعيد، قالا حدثنا يحيى، - وهو القطان - عن عبيد الله،…٣٠٤٥أن ابن عمر، كان لا يقدم مكة إلا بات بذي طوى حتى يصبح ويغتسل ثم يدخل…٣٠٤٦أن عبد الله، حدثه أن رسول الله صلى الله عليه وسلم كان ينزل بذي طوى…٣٠٤٧أن عبد الله، أخبره أن رسول الله صلى الله عليه وسلم استقبل فرضتى الجبل…٣٠٤٨أن رسول الله صلى الله عليه وسلم كان إذا طاف بالبيت الطواف الأول خب…٣٠٤٩أن رسول الله صلى الله عليه وسلم كان إذا طاف في الحج والعمرة أول ما…٣٠٥٠وحرملة بن يحيى، قال حرملة أخبرنا ابن وهب، أخبرني يونس، عن ابن شهاب،…٣٠٥١- رضى الله عنهما - قال رمل رسول الله صلى الله عليه وسلم من الحجر إلى…٣٠٥٢أن ابن عمر، رمل من الحجر إلى الحجر وذكر أن رسول الله صلى الله عليه…٣٠٥٣وحدثنا عبد الله بن مسلمة بن قعنب، حدثنا مالك، ح وحدثنا يحيى بن يحيى،…٣٠٥٤وابن، جريج عن جعفر بن محمد، عن أبيه، عن جابر بن عبد الله، أن رسول…٣٠٥٥حدثنا أبو كامل، فضيل بن حسين الجحدري حدثنا عبد الواحد بن زياد، حدثنا…٣٠٥٦بهذا الإسناد نحوه غير أنه قال وكان أهل مكة قوم حسد . ولم يقل يحسدونه .٣٠٥٧قال قلت لابن عباس إن قومك يزعمون أن رسول الله صلى الله عليه وسلم رمل…٣٠٥٨قال قلت لابن عباس أراني قد رأيت رسول الله صلى الله عليه وسلم . قال…٣٠٥٩قال قدم رسول الله صلى الله عليه وسلم وأصحابه مكة وقد وهنتهم حمى يثرب…٣٠٦٠وابن أبي عمر، وأحمد بن عبدة، جميعا عن ابن عيينة، - قال ابن عبدة حدثنا…٣٠٦١أنه قال لم أر رسول الله صلى الله عليه وسلم يمسح من البيت إلا الركنين…٣٠٦٢وحرملة، قال أبو الطاهر أخبرنا عبد الله بن وهب، أخبرني يونس، عن ابن…٣٠٦٣ذكر أن رسول الله صلى الله عليه وسلم كان لا يستلم إلا الحجر والركن…٣٠٦٤وزهير بن حرب، وعبيد الله بن سعيد، جميعا عن يحيى، القطان - قال ابن…٣٠٦٥وابن، نمير جميعا عن أبي خالد، - قال أبو بكر حدثنا أبو خالد الأحمر، -…٣٠٦٦أن قتادة بن، دعامة حدثه أن أبا الطفيل البكري حدثه أنه، سمع ابن عباس،…٣٠٦٧وعمرو، ح وحدثني هارون بن سعيد الأيلي، حدثني ابن وهب، أخبرني عمرو، عن…٣٠٦٨أن عمر، قبل الحجر وقال إني لأقبلك وإني لأعلم أنك حجر ولكني رأيت رسول…٣٠٦٩والمقدمي، وأبو كامل وقتيبة بن سعيد كلهم عن حماد، - قال خلف حدثنا حماد…٣٠٧٠بن ربيعة قال رأيت عمر يقبل الحجر ويقول إني لأقبلك وأعلم أنك حجر ولولا…٣٠٧١وزهير بن حرب، جميعا عن وكيع، - قال أبو بكر حدثنا وكيع، - عن سفيان، عن…٣٠٧٢بهذا الإسناد قال ولكني رأيت أبا القاسم صلى الله عليه وسلم بك حفيا .…٣٠٧٣وحرملة بن يحيى، قالا أخبرنا ابن وهب، أخبرني يونس، عن ابن شهاب، عن…٣٠٧٤قال طاف رسول الله صلى الله عليه وسلم بالبيت في حجة الوداع على راحلته…٣٠٧٥أنه سمع جابر بن عبد الله، يقول طاف النبي صلى الله عليه وسلم في حجة…٣٠٧٦قالت طاف النبي صلى الله عليه وسلم في حجة الوداع حول الكعبة على بعيره…٣٠٧٧يقول رأيت رسول الله صلى الله عليه وسلم يطوف بالبيت ويستلم الركن بمحجن…٣٠٧٨طوفي من وراء الناس وأنت راكبة٣٠٧٩قال قلت لها إني لأظن رجلا لو لم يطف بين الصفا والمروة ما ضره . قالت…٣٠٨٠قلت لعائشة ما أرى على جناحا أن لا أتطوف بين الصفا والمروة . قالت لم…٣٠٨١وابن أبي عمر، جميعا عن ابن عيينة، - قال ابن أبي عمر حدثنا سفيان، -…٣٠٨٢وحدثني محمد بن رافع، حدثنا حجين بن المثنى، حدثنا ليث، عن عقيل، عن…٣٠٨٣أن عائشة، أخبرته أن الأنصار كانوا قبل أن يسلموا هم وغسان يهلون لمناة…٣٠٨٤قال كانت الأنصار يكرهون أن يطوفوا بين الصفا والمروة حتى نزلت { إن…٣٠٨٥أنه سمع جابر بن عبد الله، يقول لم يطف النبي صلى الله عليه وسلم ولا…٣٠٨٦بهذا الإسناد . مثله وقال إلا طوافا واحدا طوافه الأول .٣٠٨٨وعلي بن خشرم، كلاهما عن عيسى بن يونس، - قال ابن خشرم أخبرنا عيسى، -…٣٠٨٩عليكم بالسكينة٣٠٩٠بهذا الإسناد غير أنه لم يذكر في الحديث ولم يزل رسول الله صلى الله…٣٠٩١لبيك اللهم لبيك٣٠٩٤لبيك اللهم لبيك٣٠٩٥ومحمد بن المثنى، قالا حدثنا عبد الله بن نمير، ح وحدثنا سعيد بن يحيى…٣٠٩٦وهارون بن عبد الله، ويعقوب الدورقي، قالوا أخبرنا يزيد بن هارون،…٣٠٩٧وحدثنا يحيى بن يحيى، قال قرأت على مالك عن محمد بن أبي بكر الثقفي، أنه…٣٠٩٨وحدثني سريج بن يونس، حدثنا عبد الله بن رجاء، عن موسى بن عقبة، حدثني…٣٠٩٩أنه سمعه يقول دفع رسول الله صلى الله عليه وسلم من عرفة حتى إذا كان…٣١٠٠قال انصرف رسول الله صلى الله عليه وسلم بعد الدفعة من عرفات إلى بعض…٣١٠١يقول أفاض رسول الله صلى الله عليه وسلم من عرفات فلما انتهى إلى الشعب…٣١٠٢وحدثنا إسحاق بن إبراهيم، أخبرنا يحيى بن آدم، حدثنا زهير أبو خيثمة،…٣١٠٣أن رسول الله صلى الله عليه وسلم لما أتى النقب الذي ينزله الأمراء نزل…
