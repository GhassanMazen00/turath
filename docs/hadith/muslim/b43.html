٢١٧ حديثًا•صفحة ١ من ٢
٣٨٤حديث ٣٨٤٥٩٣٨إن الله اصطفى كنانة من ولد إسماعيل واصطفى قريشا من كنانة واصطفى من…٥٩٣٩إني لأعرف حجرا بمكة كان يسلم على قبل أن أبعث إني لأعرفه الآن٥٩٤٠أنا سيد ولد آدم يوم القيامة وأول من ينشق عنه القبر وأول شافع وأول مشفع٥٩٤١أن النبي صلى الله عليه وسلم دعا بماء فأتي بقدح رحراح فجعل القوم…٥٩٤٢أنه قال رأيت رسول الله صلى الله عليه وسلم وحانت صلاة العصر فالتمس…٥٩٤٣أن نبي الله صلى الله عليه وسلم وأصحابه بالزوراء - قال والزوراء…٥٩٤٤أن النبي صلى الله عليه وسلم كان بالزوراء فأتي بإناء ماء لا يغمر…٥٩٤٥أن أم مالك، كانت تهدي للنبي صلى الله عليه وسلم في عكة لها سمنا…٥٩٤٦لو لم تكله لأكلتم منه ولقام لكم٥٩٤٧إنكم ستأتون غدا إن شاء الله عين تبوك وإنكم لن تأتوها حتى يضحي النهار…٥٩٤٨حدثنا عبد الله بن مسلمة بن قعنب، حدثنا سليمان بن بلال، عن عمرو بن…٥٩٤٩وفي كل دور الأنصار خير٥٩٥٠إن رجلا أتاني وأنا نائم فأخذ السيف فاستيقظت وهو قائم على رأسي فلم…٥٩٥١وأبو سلمة بن عبد الرحمن أن جابر بن عبد الله الأنصاري، وكان، من أصحاب…٥٩٥٢حدثنا أبو بكر بن أبي شيبة، حدثنا عفان، حدثنا أبان بن يزيد، حدثنا…٥٩٥٣إن مثل ما بعثني الله به عز وجل من الهدى والعلم كمثل غيث أصاب أرضا…٥٩٥٤إن مثلي ومثل ما بعثني الله به كمثل رجل أتى قومه فقال يا قوم إني رأيت…٥٩٥٥إنما مثلي ومثل أمتي كمثل رجل استوقد نارا فجعلت الدواب والفراش يقعن…٥٩٥٦وابن أبي عمر، قالا حدثنا سفيان، عن أبي الزناد، بهذا الإسناد نحوه .٥٩٥٧مثلي كمثل رجل استوقد نارا فلما أضاءت ما حولها جعل الفراش وهذه الدواب…٥٩٥٨مثلي ومثلكم كمثل رجل أوقد نارا فجعل الجنادب والفراش يقعن فيها وهو…٥٩٥٩مثلي ومثل الأنبياء كمثل رجل بنى بنيانا فأحسنه وأجمله فجعل الناس…٥٩٦٠مثلي ومثل الأنبياء من قبلي كمثل رجل ابتنى بيوتا فأحسنها وأجملها…٥٩٦١مثلي ومثل الأنبياء من قبلي كمثل رجل بنى بنيانا فأحسنه وأجمله إلا موضع…٥٩٦٢مثلي ومثل النبيين٥٩٦٣مثلي ومثل الأنبياء كمثل رجل بنى دارا فأتمها وأكملها إلا موضع لبنة…٥٩٦٤بهذا الإسناد مثله وقال بدل أتمها أحسنها .٥٩٦٥إن الله عز وجل إذا أراد رحمة أمة من عباده قبض نبيها قبلها فجعله لها…٥٩٦٦أنا فرطكم، على الحوض٥٩٦٧حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، ح وحدثنا أبو كريب، حدثنا ابن…٥٩٦٩أنا فرطكم، على الحوض من ورد شرب ومن شرب لم يظمأ أبدا وليردن على أقوام…٥٩٧٠وحدثنا هارون بن سعيد الأيلي، حدثنا ابن وهب، أخبرني أسامة، عن أبي…٥٩٧٢حوضي مسيرة شهر وزواياه سواء وماؤه أبيض من الورق وريحه أطيب من المسك…٥٩٧٣إني على الحوض أنتظر من يرد على منكم فوالله ليقتطعن دوني رجال فلأقولن…٥٩٧٤- وهو ابن الحارث - أن بكيرا، حدثه عن القاسم بن عباس الهاشمي، عن عبد…٥٩٧٥وحدثني أبو معن الرقاشي، وأبو بكر بن نافع وعبد بن حميد قالوا حدثنا أبو…٥٩٧٦إني فرط لكم وأنا شهيد عليكم وإني والله لأنظر إلى حوضي الآن وإني قد…٥٩٧٧إني فرطكم على الحوض وإن عرضه كما بين أيلة إلى الجحفة إني لست أخشى…٥٩٧٨أنا فرطكم على الحوض ولأنازعن أقواما ثم لأغلبن عليهم فأقول يا رب…٥٩٧٩وإسحاق بن إبراهيم، عن جرير، عن الأعمش، بهذا الإسناد . ولم يذكر "…٥٩٨٠وإسحاق بن إبراهيم، كلاهما عن جرير، ح وحدثنا ابن المثنى، حدثنا محمد بن…٥٩٨١وحدثناه سعيد بن عمرو الأشعثي، أخبرنا عبثر، ح وحدثنا أبو بكر بن أبي،…٥٩٨٢حوضه ما بين صنعاء والمدينة٥٩٨٣أنه سمع حارثة بن وهب الخزاعي، يقول سمعت رسول الله صلى الله عليه وسلم…٥٩٨٤إن أمامكم حوضا ما بين ناحيتيه كما بين جربا وأذرح٥٩٨٥إن أمامكم حوضا كما بين جربا وأذرح٥٩٨٦بهذا الإسناد . مثله وزاد قال عبيد الله فسألته فقال قريتين بالشام…٥٩٨٧وحدثني سويد بن سعيد، حدثنا حفص بن ميسرة، عن موسى بن عقبة، عن نافع، عن…٥٩٨٨إن أمامكم حوضا كما بين جربا وأذرح فيه أباريق كنجوم السماء من ورده…٥٩٨٩والذي نفس محمد بيده لآنيته أكثر من عدد نجوم السماء وكواكبها ألا في…٥٩٩٢وحدثنا محمد بن بشار، حدثنا يحيى بن حماد، حدثنا شعبة، عن قتادة، عن…٥٩٩٣لأذودن عن حوضي رجالا كما تذاد الغريبة من الإبل٥٩٩٤سمع أبا هريرة، يقول قال رسول الله صلى الله عليه وسلم بمثله .٦٠٠٤حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن بشر، وأبو أسامة عن مسعر، عن…٦٠٠٥وحدثني إسحاق بن منصور، أخبرنا عبد الصمد بن عبد الوارث، حدثنا إبراهيم،…٦٠٠٦لم تراعوا لم تراعوا٦٠٠٧ما رأينا من فزع وإن وجدناه لبحرا٦٠٠٨وابن، بشار قالا حدثنا محمد بن جعفر، ح وحدثنيه يحيى بن حبيب، حدثنا…٦٠٠٩قال كان رسول الله صلى الله عليه وسلم أجود الناس بالخير وكان أجود ما…٦٠١٠بهذا الإسناد نحوه .٦٠١١وأبو الربيع، قالا حدثنا حماد بن زيد، عن ثابت البناني، عن أنس بن مالك،…٦٠١٢بمثله .٦٠١٣وزهير بن حرب، جميعا عن إسماعيل، - واللفظ لأحمد - قالا حدثنا إسماعيل…٦٠١٤وابن، نمير قالا حدثنا محمد بن بشر، حدثنا زكرياء، حدثني سعيد، - وهو…٦٠١٦يا أنيس أذهبت حيث أمرتك٦٠١٧وأبو الربيع، قالا حدثنا عبد الوارث، عن أبي التياح، عن أنس بن مالك،…٦٠١٨وعمرو الناقد، قالا حدثنا سفيان بن عيينة، عن ابن المنكدر، سمع جابر بن…٦٠١٩الله يقول مثله سواء .٦٠٢٠قال ما سئل رسول الله صلى الله عليه وسلم على الإسلام شيئا إلا أعطاه -…٦٠٢١أن رجلا، سأل النبي صلى الله عليه وسلم غنما بين جبلين فأعطاه إياه فأتى…٦٠٢٢وحدثني أبو الطاهر، أحمد بن عمرو بن سرح أخبرنا عبد الله بن وهب، أخبرني…٦٠٢٣لو قد جاءنا مال البحرين لقد أعطيتك هكذا وهكذا وهكذا٦٠٢٤حدثنا محمد بن حاتم بن ميمون، حدثنا محمد بن بكر، أخبرنا ابن جريج،…٦٠٢٥ولد لي الليلة غلام فسميته باسم أبي إبراهيم٦٠٢٦إن إبراهيم ابني وإنه مات في الثدى وإن له لظئرين تكملان رضاعه في الجنة٦٠٢٧وأملك إن كان الله نزع منكم الرحمة٦٠٢٨إنه من لا يرحم لا يرحم٦٠٢٩حدثنا عبد بن حميد، أخبرنا عبد الرزاق، أخبرنا معمر، عن الزهري، حدثني…٦٠٣٠من لا يرحم الناس لا يرحمه الله عز وجل٦٠٣١وعبد الله بن نمير، عن إسماعيل، عن قيس، عن جرير، عن النبي صلى الله…٦٠٣٢سمع عبد الله، بن أبي عتبة يحدث عن أبي سعيد الخدري، ح وحدثنا زهير بن…٦٠٣٣إن من خياركم أحاسنكم أخلاقا٦٠٣٤ووكيع، ح وحدثنا ابن نمير، حدثنا أبي ح، وحدثنا أبو سعيد الأشج، حدثنا…٦٠٣٥قال قلت لجابر بن سمرة أكنت تجالس رسول الله صلى الله عليه وسلم قال نعم…٦٠٣٦يا أنجشة رويدك سوقا بالقوارير٦٠٣٧وحامد بن عمر، وأبو كامل قالوا حدثنا حماد، عن ثابت، عن أنس، بنحوه .٦٠٣٨وحدثناه ابن بشار، حدثنا أبو داود، حدثنا هشام، عن قتادة، عن أنس، عن…٦٠٣٩ويحك يا أنجشة رويدا سوقك بالقوارير٦٠٤٠أى أنجشة رويدا سوقك بالقوارير٦٠٤١رويدا يا أنجشة لا تكسر القوارير٦٠٤٢قال كان رسول الله صلى الله عليه وسلم إذا صلى الغداة جاء خدم المدينة…٦٠٤٣قال لقد رأيت رسول الله صلى الله عليه وسلم والحلاق يحلقه وأطاف به…٦٠٤٤يا أم فلان انظري أى السكك شئت حتى أقضي لك حاجتك٦٠٤٥زوج النبي صلى الله عليه وسلم أنها قالت ما خير رسول الله صلى الله عليه…٦٠٤٦وإسحاق بن إبراهيم، جميعا عن جرير، ح وحدثنا أحمد، بن عبدة حدثنا فضيل…٦٠٤٧بهذا الإسناد . نحو حديث مالك .٦٠٤٨قالت ما خير رسول الله صلى الله عليه وسلم بين أمرين أحدهما أيسر من…٦٠٤٩وابن، نمير جميعا عن عبد الله بن نمير، عن هشام، بهذا الإسناد إلى قوله…٦٠٥٠قالت ما ضرب رسول الله صلى الله عليه وسلم شيئا قط بيده ولا امرأة ولا…٦٠٥١وابن، نمير قالا حدثنا عبدة، ووكيع، ح وحدثنا أبو كريب، حدثنا أبو…٦٠٥٢حدثنا عمرو بن حماد بن طلحة القناد، حدثنا أسباط، - وهو ابن نصر…٦٠٥٣قال أنس ما شممت عنبرا قط ولا مسكا ولا شيئا أطيب من ريح رسول الله صلى…٦٠٥٤قال كان رسول الله صلى الله عليه وسلم أزهر اللون كأن عرقه اللؤلؤ إذا…٦٠٥٥يا أم سليم ما هذا الذي تصنعين٦٠٥٦ما تصنعين يا أم سليم٦٠٥٧يا أم سليم ما هذا٦٠٥٨قالت إن كان لينزل على رسول الله صلى الله عليه وسلم في الغداة الباردة…٦٠٥٩أحيانا يأتيني في مثل صلصلة الجرس وهو أشده على ثم يفصم عني وقد وعيته…٦٠٦٠قال كان نبي الله صلى الله عليه وسلم إذا أنزل عليه الوحى كرب لذلك…٦٠٦١قال كان النبي صلى الله عليه وسلم إذا أنزل عليه الوحى نكس رأسه ونكس…٦٠٦٢ومحمد بن جعفر بن زياد، قال منصور حدثنا وقال، ابن جعفر أخبرنا إبراهيم،…٦٠٦٣بهذا الإسناد نحوه .٦٠٦٤ومحمد بن بشار، قالا حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت أبا…٦٠٦٥قال ما رأيت من ذي لمة أحسن في حلة حمراء من رسول الله صلى الله عليه…٦٠٦٦يقول كان رسول الله صلى الله عليه وسلم أحسن الناس وجها وأحسنهم خلقا…٦٠٦٧قال قلت لأنس بن مالك كيف كان شعر رسول الله صلى الله عليه وسلم قال كان…٦٠٦٨أن رسول الله صلى الله عليه وسلم كان يضرب شعره منكبيه .٦٠٦٩قال كان شعر رسول الله صلى الله عليه وسلم إلى أنصاف أذنيه .٦٠٧٠ومحمد بن بشار، - واللفظ لابن المثنى - قالا حدثنا محمد بن جعفر، حدثنا…٦٠٧١قال قلت له أرأيت رسول الله صلى الله عليه وسلم قال نعم كان أبيض مليح…٦٠٧٢قال رأيت رسول الله صلى الله عليه وسلم وما على وجه الأرض رجل رآه غيري…٦٠٧٣وابن، نمير وعمرو الناقد جميعا عن ابن إدريس، - قال عمرو حدثنا عبد الله…٦٠٧٤قال سألت أنس بن مالك هل كان رسول الله صلى الله عليه وسلم خضب فقال لم…٦٠٧٥قال سألت أنس بن مالك أخضب رسول الله صلى الله عليه وسلم قال إنه لم ير…٦٠٧٦صلى الله عليه وسلم فقال لو شئت أن أعد شمطات كن في رأسه فعلت . وقال لم…٦٠٧٧قال يكره أن ينتف الرجل، الشعرة البيضاء من رأسه ولحيته - قال - ولم…٦٠٧٨بهذا الإسناد .٦٠٧٩وابن، بشار وأحمد بن إبراهيم الدورقي وهارون بن عبد الله جميعا عن أبي…٦٠٨٠قال رأيت رسول الله صلى الله عليه وسلم هذه منه بيضاء ووضع زهير بعض…٦٠٨١قال رأيت رسول الله صلى الله عليه وسلم أبيض قد شاب كان الحسن بن علي…٦٠٨٢وخالد بن عبد الله، ح وحدثنا ابن نمير، حدثنا محمد بن بشر، كلهم عن…٦٠٨٣صلى الله عليه وسلم فقال كان إذا دهن رأسه لم ير منه شىء وإذا لم يدهن…٦٠٨٤أنه سمع جابر بن سمرة، يقول كان رسول الله صلى الله عليه وسلم قد شمط…٦٠٨٥قال رأيت خاتما في ظهر رسول الله صلى الله عليه وسلم كأنه بيضة حمام .٦٠٨٦بهذا الإسناد مثله .٦٠٨٧ومحمد بن عباد، قالا حدثنا حاتم، - وهو ابن إسماعيل - عن الجعد بن عبد…٦٠٨٨قال رأيت النبي صلى الله عليه وسلم وأكلت معه خبزا ولحما - أو قال ثريدا…٦٠٨٩أنه سمعه يقول كان رسول الله صلى الله عليه وسلم ليس بالطويل البائن ولا…٦٠٩٠وقتيبة بن سعيد، وعلي بن حجر، قالوا حدثنا إسماعيل، يعنون ابن جعفر ح…٦٠٩١قال قبض رسول الله صلى الله عليه وسلم وهو ابن ثلاث وستين وأبو بكر وهو…٦٠٩٢وحدثني عبد الملك بن شعيب بن الليث، حدثني أبي، عن جدي، قال حدثني عقيل،…٦٠٩٣وعباد بن موسى، قالا حدثنا طلحة بن يحيى، عن يونس بن يزيد، عن ابن شهاب،…٦٠٩٤قال قلت لعروة كم كان النبي صلى الله عليه وسلم بمكة قال عشرا . قال قلت…٦٠٩٥قال قلت لعروة كم لبث النبي صلى الله عليه وسلم بمكة قال عشرا . قلت فإن…٦٠٩٦وهارون بن عبد الله، عن روح بن عبادة، حدثنا زكرياء بن إسحاق، عن عمرو…٦٠٩٧قال أقام رسول الله صلى الله عليه وسلم بمكة ثلاث عشرة سنة يوحى إليه…٦٠٩٨وحدثنا عبد الله بن عمر بن محمد بن أبان الجعفي، حدثنا سلام أبو الأحوص،…٦٠٩٩وابن، بشار - واللفظ لابن المثنى - قالا حدثنا محمد بن، جعفر حدثنا…٦١٠٠مولى بني هاشم قال سألت ابن عباس كم أتى لرسول الله صلى الله عليه وسلم…
