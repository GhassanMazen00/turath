كتاب الصيام

٢٨٥ حديثًا•صفحة ١ من ٢
٢٤٩٥إذا جاء رمضان فتحت أبواب الجنة وغلقت أبواب النار وصفدت الشياطين٢٤٩٦إذا كان رمضان فتحت أبواب الرحمة وغلقت أبواب جهنم وسلسلت الشياطين٢٤٩٧والحلواني، قالا حدثنا يعقوب، حدثنا أبي، عن صالح، عن ابن شهاب، حدثني…٢٤٩٨لا تصوموا حتى تروا الهلال ولا تفطروا حتى تروه فإن أغمي عليكم فاقدروا…٢٤٩٩الشهر هكذا وهكذا وهكذا - ثم عقد إبهامه في الثالثة - فصوموا لرؤيته…٢٥٠٠فإن غم عليكم فاقدروا ثلاثين٢٥٠١الشهر تسع وعشرون الشهر هكذا وهكذا وهكذا٢٥٠٢إنما الشهر تسع وعشرون فلا تصوموا حتى تروه ولا تفطروا حتى تروه فإن غم…٢٥٠٣الشهر تسع وعشرون فإذا رأيتم الهلال فصوموا وإذا رأيتموه فأفطروا فإن غم…٢٥٠٤إذا رأيتموه فصوموا وإذا رأيتموه فأفطروا فإن غم عليكم فاقدروا له٢٥٠٥الشهر تسع وعشرون ليلة لا تصوموا حتى تروه ولا تفطروا حتى تروه إلا أن…٢٥٠٦الشهر هكذا وهكذا وهكذا٢٥٠٧الشهر تسع وعشرون٢٥٠٨الشهر هكذا وهكذا وهكذا عشرا وعشرا وتسعا٢٥٠٩الشهر كذا وكذا وكذا٢٥١٠الشهر تسع وعشرون٢٥١١إنا أمة أمية لا نكتب ولا نحسب الشهر هكذا وهكذا وهكذا - وعقد الإبهام…٢٥١٢بهذا الإسناد ولم يذكر للشهر الثاني ثلاثين .٢٥١٣الشهر هكذا وهكذا٢٥١٤إذا رأيتم الهلال فصوموا وإذا رأيتموه فأفطروا فإن غم عليكم فصوموا…٢٥١٥صوموا لرؤيته وأفطروا لرؤيته فإن غمي عليكم فأكملوا العدد٢٥١٦صوموا لرؤيته وأفطروا لرؤيته فإن غمي عليكم الشهر فعدوا ثلاثين٢٥١٧إذا رأيتموه فصوموا وإذا رأيتموه فأفطروا فإن أغمي عليكم فعدوا ثلاثين٢٥١٨لا تقدموا رمضان بصوم يوم ولا يومين إلا رجل كان يصوم صوما فليصمه٢٥١٩وابن أبي عمر، قالا حدثنا عبد الوهاب بن عبد المجيد، حدثنا أيوب، ح…٢٥٢٠إن الشهر تسع وعشرون٢٥٢١حدثنا محمد بن رمح، أخبرنا الليث، ح وحدثنا قتيبة بن سعيد، - واللفظ له…٢٥٢٢إن الشهر يكون تسعا وعشرين٢٥٢٣إن الشهر يكون تسعة وعشرين يوما٢٥٢٤بهذا الإسناد مثله .٢٥٢٥الشهر هكذا وهكذا٢٥٢٦الشهر هكذا وهكذا وهكذا٢٥٢٧وسلمة، بن سليمان قالا أخبرنا عبد الله، - يعني ابن المبارك - أخبرنا…٢٥٢٨ويحيى بن أيوب، وقتيبة، وابن، حجر قال يحيى بن يحيى أخبرنا وقال…٢٥٢٩إن الله مده للرؤية فهو لليلة رأيتموه٢٥٣٠إن الله قد أمده لرؤيته فإن أغمي عليكم فأكملوا العدة٢٥٣١شهرا عيد لا ينقصان رمضان وذو الحجة٢٥٣٢شهرا عيد لا ينقصان٢٥٣٣إن وسادتك لعريض إنما هو سواد الليل وبياض النهار٢٥٣٤قال لما نزلت هذه الآية { وكلوا واشربوا حتى يتبين لكم الخيط الأبيض من…٢٥٣٥- رضى الله عنه - قال لما نزلت هذه الآية { وكلوا واشربوا حتى يتبين لكم…٢٥٣٦إن بلالا يؤذن بليل فكلوا واشربوا حتى تسمعوا تأذين ابن أم مكتوم٢٥٣٧إن بلالا يؤذن بليل فكلوا واشربوا حتى تسمعوا أذان ابن أم مكتوم٢٥٣٨إن بلالا يؤذن بليل فكلوا واشربوا حتى يؤذن ابن أم مكتوم٢٥٣٩وحدثنا ابن نمير، حدثنا أبي، حدثنا عبيد الله، حدثنا القاسم، عن عائشة،…٢٥٤٠بالإسنادين كليهما . نحو حديث ابن نمير .٢٥٤١لا يمنعن أحدا منكم أذان بلال - أو قال نداء بلال - من سحوره فإنه يؤذن…٢٥٤٢إن الفجر ليس الذي يقول هكذا٢٥٤٣ينبه نائمكم ويرجع قائمكم٢٥٤٤أنه سمع سمرة بن جندب، يقول سمعت محمدا، صلى الله عليه وسلم يقول٢٥٤٥لا يغرنكم أذان بلال ولا هذا البياض - لعمود الصبح - حتى يستطير هكذا٢٥٤٦لا يغرنكم من سحوركم أذان بلال ولا بياض الأفق المستطيل هكذا حتى يستطير…٢٥٤٧لا يغرنكم نداء بلال ولا هذا البياض حتى يبدو الفجر - أو قال - حتى…٢٥٤٨- رضى الله عنه - يقول قال رسول الله صلى الله عليه وسلم . فذكر هذا .٢٥٤٩تسحروا فإن في السحور بركة٢٥٥٠فصل ما بين صيامنا وصيام أهل الكتاب أكلة السحر٢٥٥١بهذا الإسناد .٢٥٥٢حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن هشام، عن قتادة، عن أنس، عن…٢٥٥٣بهذا الإسناد .٢٥٥٤لا يزال الناس بخير ما عجلوا الفطر٢٥٥٥وحدثناه قتيبة، حدثنا يعقوب، ح وحدثني زهير بن حرب، حدثنا عبد الرحمن،…٢٥٥٦حدثنا يحيى بن يحيى، وأبو كريب محمد بن العلاء قالا أخبرنا أبو معاوية،…٢٥٥٧وحدثنا أبو كريب، أخبرنا ابن أبي زائدة، عن الأعمش، عن عمارة، عن أبي…٢٥٥٨إذا أقبل الليل وأدبر النهار وغابت الشمس فقد أفطر الصائم٢٥٥٩يا فلان انزل فاجدح لنا٢٥٦٠وعباد بن العوام، عن الشيباني، عن ابن أبي أوفى، - رضى الله عنه - قال…٢٥٦١يا فلان انزل فاجدح لنا٢٥٦٢وجاء الليل من ها هنا٢٥٦٣إني لست كهيئتكم إني أطعم وأسقى٢٥٦٤إني لست مثلكم إني أطعم وأسقى٢٥٦٥وحدثنا عبد الوارث بن عبد الصمد، حدثني أبي، عن جدي، عن أيوب، عن نافع،…٢٥٦٦وأيكم مثلي إني أبيت يطعمني ربي ويسقيني٢٥٦٧وإسحاق، قال زهير حدثنا جرير، عن عمارة، عن أبي، زرعة عن أبي هريرة، -…٢٥٦٨فاكلفوا ما لكم به طاقة٢٥٦٩وحدثنا ابن نمير، حدثنا أبي، حدثنا الأعمش، عن أبي صالح، عن أبي هريرة،…٢٥٧٠نعم ذاك الذي حملني على الذي صنعت٢٥٧١لو مد لنا الشهر لواصلنا وصالا يدع المتعمقون تعمقهم إنكم لستم مثلي -…٢٥٧٢إني لست كهيئتكم إني يطعمني ربي ويسقيني٢٥٧٣- رضى الله عنها - قالت كان رسول الله صلى الله عليه وسلم يقبل إحدى…٢٥٧٤وابن أبي عمر، قالا حدثنا سفيان، قال قلت لعبد الرحمن بن القاسم أسمعت…٢٥٧٥- رضى الله عنها - قالت كان رسول الله صلى الله عليه وسلم يقبلني وهو…٢٥٧٦وعلقمة، عن عائشة، رضى الله عنها ح . وحدثنا شجاع بن مخلد، حدثنا يحيى…٢٥٧٧وزهير بن حرب، قالا حدثنا سفيان، عن منصور، عن إبراهيم، عن علقمة، عن…٢٥٧٨وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن منصور، عن إبراهيم،…٢٥٧٩وحدثنا محمد بن المثنى، حدثنا أبو عاصم، قال سمعت ابن عون، عن إبراهيم،…٢٥٨٠ومسروق أنهما دخلا على أم المؤمنين يسألانها . فذكر نحوه .٢٥٨١أن عمر بن عبد العزيز، أخبره أن عروة بن الزبير أخبره أن عائشة أم…٢٥٨٢بن أبي كثير بهذا الإسناد مثله .٢٥٨٣وقتيبة بن سعيد، وأبو بكر بن أبي شيبة قال يحيى أخبرنا وقال الآخران،…٢٥٨٤رضى الله عنها قالت كان رسول الله صلى الله عليه وسلم يقبل في رمضان وهو…٢٥٨٥رضى الله عنها أن النبي صلى الله عليه وسلم كان يقبل وهو صائم .٢٥٨٦- رضى الله عنها - قالت كان رسول الله صلى الله عليه وسلم يقبل وهو صائم…٢٥٨٧وحدثنا أبو الربيع الزهراني، حدثنا أبو عوانة، ح وحدثنا أبو بكر بن أبي…٢٥٨٨أنه سأل رسول الله صلى الله عليه وسلم أيقبل الصائم فقال له رسول الله…٢٥٨٩حدثني محمد بن حاتم، حدثنا يحيى بن سعيد، عن ابن جريج، ح وحدثني محمد،…٢٥٩٠وأبي، بكر بن عبد الرحمن أن عائشة، زوج النبي صلى الله عليه وسلم قالت…٢٥٩١أن أبا بكر، حدثه أن مروان أرسله إلى أم سلمة - رضى الله عنها - يسأل عن…٢٥٩٢وأم سلمة زوجى النبي صلى الله عليه وسلم أنهما قالتا إن كان رسول الله…٢٥٩٣وأنا تدركني الصلاة وأنا جنب فأصوم٢٥٩٤حدثنا أحمد بن عثمان النوفلي، حدثنا أبو عاصم، حدثنا ابن جريج، أخبرني…٢٥٩٥حدثنا يحيى بن يحيى، وأبو بكر بن أبي شيبة وزهير بن حرب وابن نمير كلهم…٢٥٩٦بهذا الإسناد . مثل رواية ابن عيينة وقال بعرق فيه تمر - وهو الزنبيل -…٢٥٩٧ومحمد بن رمح، قالا أخبرنا الليث، ح وحدثنا قتيبة، حدثنا ليث، عن ابن…٢٥٩٨بهذا الإسناد أن رجلا، أفطر في رمضان فأمره رسول الله صلى الله عليه…٢٥٩٩أن أبا هريرة، حدثه أن النبي صلى الله عليه وسلم أمر رجلا أفطر في رمضان…٢٦٠٠بهذا الإسناد نحو حديث ابن عيينة .٢٦٠١- رضى الله عنها - أنها قالت جاء رجل إلى رسول الله صلى الله عليه وسلم…٢٦٠٢أن محمد بن جعفر بن الزبير، أخبره أن عباد بن عبد الله بن الزبير حدثه…٢٦٠٣أن عبد الرحمن، بن القاسم حدثه أن محمد بن جعفر بن الزبير حدثه أن عباد…٢٦٠٤ومحمد بن رمح، قالا أخبرنا الليث، ح وحدثنا قتيبة بن، سعيد حدثنا ليث،…٢٦٠٥بهذا الإسناد . مثله . قال يحيى قال سفيان لا أدري من قول من هو يعني…٢٦٠٦بهذا الإسناد قال الزهري وكان الفطر آخر الأمرين وإنما يؤخذ من أمر رسول…٢٦٠٧بهذا الإسناد مثل حديث الليث . قال ابن شهاب فكانوا يتبعون الأحدث…٢٦٠٨- رضى الله عنهما - قال سافر رسول الله صلى الله عليه وسلم في رمضان…٢٦٠٩عباس - رضى الله عنهما - قال لا تعب على من صام ولا على من أفطر قد صام…٢٦١٠أولئك العصاة أولئك العصاة٢٦١١بهذا الإسناد وزاد فقيل له إن الناس قد شق عليهم الصيام وإنما ينظرون…٢٦١٢ومحمد بن المثنى، وابن، بشار جميعا عن محمد، بن جعفر - قال أبو بكر…٢٦١٣يحدث أنه سمع جابر بن عبد الله، - رضى الله عنهما - يقول رأى رسول الله…٢٦١٤عليكم برخصة الله الذي رخص لكم٢٦١٥حدثنا هداب بن خالد، حدثنا همام بن يحيى، حدثنا قتادة، عن أبي نضرة، عن…٢٦١٦وقال ابن المثنى، حدثنا أبو عامر، حدثنا هشام، وقال ابن المثنى، حدثنا…٢٦١٧حدثنا نصر بن علي الجهضمي، حدثنا بشر، - يعني ابن مفضل - عن أبي مسلمة،…٢٦١٨حدثني عمرو الناقد، حدثنا إسماعيل بن إبراهيم، عن الجريري، عن أبي نضرة،…٢٦١٩وسهل بن عثمان، وسويد بن سعيد، وحسين، بن حريث كلهم عن مروان، - قال…٢٦٢٠حدثنا يحيى بن يحيى، أخبرنا أبو خيثمة، عن حميد، قال سئل أنس - رضى الله…٢٦٢١وحدثنا أبو بكر بن أبي شيبة، حدثنا أبو خالد الأحمر، عن حميد، قال خرجت…٢٦٢٢ذهب المفطرون اليوم بالأجر٢٦٢٣ذهب المفطرون اليوم بالأجر٢٦٢٤إنكم قد دنوتم من عدوكم والفطر أقوى لكم٢٦٢٥إن شئت فصم وإن شئت فأفطر٢٦٢٦صم إن شئت وأ��طر إن شئت٢٦٢٧بهذا الإسناد . مثل حديث حماد بن زيد إني رجل أسرد الصوم .٢٦٢٨وقال أبو بكر، حدثنا عبد الرحيم بن سليمان، كلاهما عن هشام، بهذا…٢٦٢٩هي رخصة من الله فمن أخذ بها فحسن ومن أحب أن يصوم فلا جناح عليه٢٦٣٠حدثنا داود بن رشيد، حدثنا الوليد بن مسلم، عن سعيد بن عبد العزيز، عن…٢٦٣١حدثنا عبد الله بن مسلمة القعنبي، حدثنا هشام بن سعد، عن عثمان بن حيان،…٢٦٣٢أن ناسا، تماروا عندها يوم عرفة في صيام رسول الله صلى الله عليه وسلم…٢٦٣٣وابن أبي عمر، عن سفيان، عن أبي النضر، بهذا الإسناد . ولم يذكر وهو…٢٦٣٤مولى أم الفضل .٢٦٣٥أن أبا النضر، حدثه أن عميرا مولى ابن عباس - رضى الله عنهما - حدثه…٢٦٣٦زوج النبي صلى الله عليه وسلم أنها قالت إن الناس شكوا في صيام رسول…٢٦٣٧من شاء صامه ومن شاء تركه٢٦٣٨بهذا الإسناد ولم يذكر في أول الحديث وكان رسول الله صلى الله عليه وسلم…٢٦٣٩- رضى الله عنها - أن يوم، عاشوراء كان يصام في الجاهلية فلما جاء…٢٦٤٠أن عائشة، - رضى الله عنها - قالت كان رسول الله صلى الله عليه وسلم…٢٦٤١من شاء فليصمه ومن شاء فليفطره٢٦٤٢إن عاشوراء يوم من أيام الله فمن شاء صامه ومن شاء تركه٢٦٤٣وزهير بن حرب، قالا حدثنا يحيى، وهو القطان ح وحدثنا أبو بكر بن أبي…٢٦٤٤كان يوما يصومه أهل الجاهلية فمن أحب منكم أن يصومه فليصمه ومن كره…
