٢٩ حديثًا
٢٠٨٩إن الشمس والقمر من آيات الله وإنهما لا ينخسفان لموت أحد ولا لحياته…٢٠٩٠أما بعد فإن الشمس والقمر من آيات الله٢٠٩١سمع الله لمن حمده ربنا ولك الحمد٢٠٩٢أن الشمس، خسفت على عهد رسول الله صلى الله عليه وسلم فبعث مناديا "…٢٠٩٤أنه سمع ابن شهاب، يخبر عن عروة، عن عائشة، أن النبي صلى الله عليه وسلم…٢٠٩٥قال كان كثير بن عباس يحدث أن ابن عباس، كان يحدث عن صلاة، رسول الله…٢٠٩٦أصدق - حسبته يريد عائشة - أن الشمس انكسفت على عهد رسول الله صلى الله…٢٠٩٧صلى ست ركعات وأربع سجدات٢٠٩٨إني قد رأيتكم تفتنون في القبور كفتنة الدجال٢٠٩٩في هذا الإسناد بمثل معنى حديث سليمان بن بلال٢١٠٢يا أيها الناس إنما الشمس والقمر آيتان من آيات الله وإنهما لا ينكسفان…٢١٠٣أما بعد ما من شىء لم أكن رأيته إلا قد رأيته في مقامي هذا حتى الجنة…٢١٠٦حدثنا يحيى بن حبيب الحارثي، حدثنا خالد بن الحارث، حدثنا ابن جريج،…٢١٠٧بهذا الإسناد مثله وقال قياما طويلا يقوم ثم يركع وزاد فجعلت أنظر إلى…٢١٠٨قالت كسفت الشمس على عهد النبي صلى الله عليه وسلم ففزع فأخطأ بدرع حتى…٢١٠٩إن الشمس والقمر آيتان من آيات الله لا ينكسفان لموت أحد ولا لحياته…٢١١٠في هذا الإسناد بمثله غير أنه قال ثم رأيناك تكعكعت .٢١١١حدثنا أبو بكر بن أبي شيبة، حدثنا إسماعيل ابن علية، عن سفيان، عن حبيب،…٢١١٢وحدثنا محمد بن المثنى، وأبو بكر بن خلاد كلاهما عن يحيى القطان، - قال…٢١١٣حدثني محمد بن رافع، حدثنا أبو النضر، حدثنا أبو معاوية، - وهو شيبان…٢١١٤إن الشمس والقمر آيتان من آيات الله يخوف الله بهما عباده وإنهما لا…٢١١٥إن الشمس والقمر ليس ينكسفان لموت أحد من الناس ولكنهما آيتان من آيات…٢١١٦ووكيع، ح وحدثنا ابن أبي عمر، حدثنا سفيان، ومروان، كلهم عن إسماعيل،…٢١١٧إن هذه الآيات التي يرسل الله لا تكون لموت أحد ولا لحياته ولكن الله…٢١١٨وحدثني عبيد الله بن عمر القواريري، حدثنا بشر بن المفضل، حدثنا…٢١١٩- وكان من أصحاب رسول الله صلى الله عليه وسلم - قال كنت أرتمي بأسهم لي…٢١٢٠بأسهم لي على عهد رسول الله صلى الله عليه وسلم إذ خسفت الشمس . ثم ذكر…٢١٢١إن الشمس والقمر لا يخسفان لموت أحد ولا لحياته ولكنهما آية من آيات…٢١٢٢إن الشمس والقمر آيتان من آيات الله لا ينكسفان لموت أحد ولا لحياته…
