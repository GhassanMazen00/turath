٨٤ حديثًا
٣٥٦٨أن عائشة، أخبرتها أن رسول الله صلى الله عليه وسلم كان عندها وإنها…٣٥٦٩يحرم من الرضاعة ما يحرم من الولادة٣٥٧٠بهذا الإسناد . مثل حديث هشام بن عروة .٣٥٧١أنها أخبرته أن أفلح - أخا أبي القعيس - جاء يستأذن عليها وهو عمها من…٣٥٧٢تربت يداك أو يمينك٣٥٧٣أن عائشة، أخبرته أنه، جاء أفلح أخو أبي القعيس يستأذن عليها بعد ما نزل…٣٥٧٤فإنه عمك تربت يمينك٣٥٧٥فليلج عليك عمك٣٥٧٦بهذا الإسناد أن أخا أبي القعيس، استأذن عليها . فذكر نحوه .٣٥٧٧بهذا الإسناد نحوه غير أنه قال استأذن عليها أبو القعيس .٣٥٧٨فهلا أذنت له تربت يمينك أو يدك٣٥٧٩لا تحتجبي منه فإنه يحرم من الرضاعة ما يحرم من النسب٣٥٨٠ليدخل عليك فإنه عمك٣٥٨١وزهير بن حرب، ومحمد بن العلاء، - واللفظ لأبي بكر - قالوا حدثنا أبو…٣٥٨٢وإسحاق بن إبراهيم، عن جرير، ح وحدثنا ابن، نمير حدثنا أبي ح، وحدثنا…٣٥٨٣إنها لا تحل لي إنها ابنة أخي من الرضاعة ويحرم من الرضاعة ما يحرم من…٣٥٨٤ابنة أخي من الرضاعة٣٥٨٥إن حمزة أخي من الرضاعة٣٥٨٦قالت دخل على رسول الله صلى الله عليه وسلم فقلت له هل لك في أختي بنت…٣٥٨٧بهذا الإسناد سواء .٣٥٨٨وحدثنا محمد بن رمح بن المهاجر، أخبرنا الليث، عن يزيد بن أبي حبيب،…٣٥٨٩بإسناد ابن أبي حبيب عنه نحو حديثه ولم يسم أحد منهم في حديثه عزة غير…٣٥٩٠لا تحرم المصة والمصتان٣٥٩١لا تحرم الإملاجة والإملاجتان٣٥٩٢وابن، بشار قالا حدثنا معاذ بن هشام، حدثني أبي، عن قتادة، عن صالح بن…٣٥٩٣لا تحرم الرضعة أو الرضعتان أو المصة أو المصتان٣٥٩٤أو الرضعتان أو المصتان٣٥٩٥لا تحرم الإملاجة والإملاجتان٣٥٩٦سأل رجل النبي صلى الله عليه وسلم أتحرم المصة فقال " لا " .٣٥٩٧أنها قالت كان فيما أنزل من القرآن عشر رضعات معلومات يحرمن . ثم نسخن…٣٥٩٨تقول - وهى تذكر الذي يحرم من الرضاعة - قالت عمرة فقالت عائشة نزل في…٣٥٩٩قال أخبرتني عمرة، أنها سمعت عائشة، تقول . بمثله .٣٦٠٠وابن أبي عمر، قالا حدثنا سفيان بن عيينة، عن عبد الرحمن، بن القاسم عن…٣٦٠١أرضعيه تحرمي عليه ويذهب الذي في نفس أبي حذيفة٣٦٠٢أرضعيه تحرمي عليه٣٦٠٣أرضعيه حتى يدخل عليك٣٦٠٤وهارون بن سعيد الأيلي، - واللفظ لهارون - قالا حدثنا ابن وهب، أخبرني…٣٦٠٥أن أمه، زينب بنت أبي سلمة أخبرته أن أمها أم سلمة زوج النبي صلى الله…٣٦٠٦انظرن إخوتكن من الرضاعة فإنما الرضاعة من المجاعة٣٦٠٧وابن، بشار قالا حدثنا محمد بن جعفر، ح وحدثنا عبيد الله بن معاذ، حدثنا…٣٦٠٨الخدري أن رسول الله صلى الله عليه وسلم يوم حنين بعث جيشا إلى أوطاس…٣٦٠٩ومحمد بن المثنى، وابن، بشار قالوا حدثنا عبد، الأعلى عن سعيد، عن…٣٦١٠بهذا الإسناد نحوه .٣٦١١قال أصابوا سبيا يوم أوطاس لهن أزواج فتخوفوا فأنزلت هذه الآية {…٣٦١٢بهذا الإسناد نحوه .٣٦١٣هو لك يا عبد الولد للفراش وللعاهر الحجر واحتجبي منه يا سودة بنت زمعة٣٦١٤بهذا الإسناد نحوه . غير أن معمرا وابن عيينة في حديثهما " الولد للفراش…٣٦١٥الولد للفراش وللعاهر الحجر٣٦١٦وزهير بن حرب، وعبد الأعلى بن حماد، وعمرو الناقد، قالوا حدثنا سفيان،…٣٦١٧ألم ترى أن مجززا نظر آنفا إلى زيد بن حارثة وأسامة بن زيد فقال إن بعض…٣٦١٨يا عائشة ألم ترى أن مجززا المدلجي دخل على فرأى أسامة وزيدا وعليهما…٣٦١٩قالت دخل قائف ورسول الله صلى الله عليه وسلم شاهد وأسامة بن زيد وزيد…٣٦٢٠وابن، جريج كلهم عن الزهري، بهذا الإسناد . بمعنى حديثهم . وزاد في حديث…٣٦٢١إنه ليس بك على أهلك هوان إن شئت سبعت لك وإن سبعت لك سبعت لنسائي٣٦٢٢ليس بك على أهلك هوان إن شئت سبعت عندك وإن شئت ثلثت ثم درت٣٦٢٣إن شئت زدتك وحاسبتك به للبكر سبع وللثيب ثلاث٣٦٢٤بهذا الإسناد مثله .٣٦٢٥إن شئت أن أسبع لك وأسبع لنسائي وإن سبعت لك سبعت لنسائي٣٦٢٦مالك قال إذا تزوج البكر على الثيب أقام عندها سبعا وإذا تزوج الثيب على…٣٦٢٧وخالد، الحذاء عن أبي قلابة، عن أنس، قال من السنة أن يقيم، عند البكر…٣٦٢٨حدثنا أبو بكر بن أبي شيبة، حدثنا شبابة بن سوار، حدثنا سليمان بن…٣٦٢٩قالت ما رأيت امرأة أحب إلى أن أكون في مسلاخها من سودة بنت زمعة من…٣٦٣٠بهذا الإسناد أن سودة، لما كبرت . بمعنى حديث جرير وزاد في حديث شريك…٣٦٣١قالت كنت أغار على اللاتي وهبن أنفسهن لرسول الله صلى الله عليه وسلم…٣٦٣٢أنها كانت تقول أما تستحيي امرأة تهب نفسها لرجل حتى أنزل الله عز وجل {…٣٦٣٣ومحمد بن حاتم، قال محمد بن حاتم حدثنا محمد، بن بكر أخبرنا ابن جريج،…٣٦٣٤وعبد بن حميد، جميعا عن عبد الرزاق، عن ابن جريج، بهذا الإسناد وزاد قال…٣٦٣٥تنكح المرأة لأربع لمالها ولحسبها ولجمالها ولدينها فاظفر بذات الدين…٣٦٣٦قال تزوجت امرأة في عهد رسول الله صلى الله عليه وسلم فلقيت النبي صلى…٣٦٣٧حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا شعبة، عن محارب، عن جابر بن،…٣٦٣٨وأبو الربيع الزهراني، قال يحيى أخبرنا حماد بن زيد، عن عمرو بن دينار،…٣٦٣٩هل نكحت يا جابر٣٦٤٠ما يعجلك يا جابر٣٦٤١حدثنا محمد بن المثنى، حدثنا عبد الوهاب، - يعني ابن عبد المجيد الثقفي…٣٦٤٢أتبيعنيه بكذا وكذا والله يغفر لك٣٦٤٣الدنيا متاع وخير متاع الدنيا المرأة الصالحة٣٦٤٤إن المرأة كالضلع إذا ذهبت تقيمها كسرتها وإن تركتها استمتعت بها وفيها…٣٦٤٥وعبد بن حميد، كلاهما عن يعقوب بن إبراهيم بن سعد، عن ابن أخي الزهري،…٣٦٤٦إن المرأة خلقت من ضلع لن تستقيم لك على طريقة فإن استمتعت بها استمتعت…٣٦٤٧من كان يؤمن بالله واليوم الآخر فإذا شهد أمرا فليتكلم بخير أو ليسكت…٣٦٤٨لا يفرك مؤمن مؤمنة إن كره منها خلقا رضي منها آخر٣٦٤٩وحدثنا محمد بن المثنى، حدثنا أبو عاصم، حدثنا عبد الحميد بن جعفر،…٣٦٥٠لولا حواء لم تخن أنثى زوجها الدهر٣٦٥١لولا بنو إسرائيل لم يخبث الطعام ولم يخنز اللحم ولولا حواء لم تخن أنثى…
