٤٥١٩حدثنا يحيى بن يحيى التميمي، حدثنا سليم بن أخضر، عن ابن عون، قال كتبت…٤٥٢٠بهذا الإسناد . مثله وقال جويرية بنت الحارث . ولم يشك .٤٥٢٣أن سليمان بن بريدة، حدثه عن أبيه، قال كان رسول الله صلى الله عليه…٤٥٢٤بهذا .٤٥٢٥بشروا ولا تنفروا ويسروا ولا تعسروا٤٥٢٦يسرا ولا تعسرا وبشرا ولا تنفرا وتطاوعا ولا تختلفا٤٥٢٧وتطاوعا ولا تختلفا٤٥٢٨يسروا ولا تعسروا وسكنوا ولا تنفروا٤٥٢٩إذا جمع الله الأولين والآخرين يوم القيامة يرفع لكل غادر لواء فقيل هذه…٤٥٣٠حدثنا أبو الربيع العتكي، حدثنا حماد، حدثنا أيوب، ح وحدثنا عبد الله بن…٤٥٣١إن الغادر ينصب الله له لواء يوم القيامة فيقال ألا هذه غدرة فلان٤٥٣٢لكل غادر لواء يوم القيامة٤٥٣٣لكل غادر لواء يوم القيامة يقال هذه غدرة فلان٤٥٣٤يقال هذه غدرة فلان٤٥٣٥لكل غادر لواء يوم القيامة يعرف به يقال هذه غدرة فلان٤٥٣٦لكل غادر لواء يوم القيامة يعرف به٤٥٣٧لكل غادر لواء عند استه يوم القيامة٤٥٣٨لكل غادر لواء يوم القيامة يرفع له بقدر غدره ألا ولا غادر أعظم غدرا من…٤٥٣٩وعمرو الناقد، وزهير بن حرب، - واللفظ لعلي وزهير - قال علي أخبرنا وقال…٤٥٤٠قال قال رسول الله صلى الله عليه وسلم " الحرب خدعة " .٤٥٤١لا تمنوا لقاء العدو فإذا لقيتموهم فاصبروا٤٥٤٢يا أيها الناس لا تتمنوا لقاء العدو واسألوا الله العافية فإذا لقيتموهم…٤٥٤٣اللهم منزل الكتاب سريع الحساب اهزم الأحزاب اللهم اهزمهم وزلزلهم٤٥٤٤يقول دعا رسول الله صلى الله عليه وسلم . بمثل حديث خالد غير أنه قال "…٤٥٤٥وابن أبي عمر، جميعا عن ابن عيينة، عن إسماعيل، بهذا الإسناد وزاد ابن…٤٥٤٦اللهم إنك إن تشأ لا تعبد في الأرض٤٥٤٧ومحمد بن رمح، قالا أخبرنا الليث، ح وحدثنا قتيبة بن، سعيد حدثنا ليث،…٤٥٤٨حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن بشر، وأبو أسامة قالا حدثنا…٤٥٤٩وسعيد بن منصور، وعمرو الناقد، جميعا عن ابن عيينة، قال يحيى أخبرنا…٤٥٥٠حدثنا عبد بن حميد، أخبرنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن…٤٥٥١بن دينار أن ابن شهاب، أخبره عن عبيد الله بن عبد الله بن عتبة، عن ابن…٤٥٥٢ومحمد بن رمح، قالا أخبرنا الليث، ح وحدثنا قتيبة بن، سعيد حدثنا ليث،…٤٥٥٣وهناد بن السري، قالا حدثنا ابن المبارك، عن موسى، بن عقبة عن نافع، عن…٤٥٥٤قال حرق رسول الله صلى الله عليه وسلم نخل بني النضير .٤٥٥٥غزا نبي من الأنبياء فقال لقومه لا يتبعني رجل قد ملك بضع امرأة وهو…٤٥٥٦وحدثنا قتيبة بن سعيد، حدثنا أبو عوانة، عن سماك، عن مصعب بن سعد، عن…٤٥٥٧وابن، بشار - واللفظ لابن المثنى - قالا حدثنا محمد، بن جعفر حدثنا…٤٥٥٨حدثنا يحيى بن يحيى، قال قرأت على مالك عن نافع، عن ابن عمر، قال بعث…٤٥٥٩أن رسول الله صلى الله عليه وسلم بعث سرية قبل نجد وفيهم ابن عمر وأن…٤٥٦٠وعبد الرحيم بن سليمان، عن عبيد الله بن عمر، عن نافع، عن ابن عمر، قال…٤٥٦١ومحمد بن المثنى، قالا حدثنا يحيى، - وهو القطان - عن عبيد الله، بهذا…٤٥٦٢بهذا الإسناد نحو حديثهم .٤٥٦٣وعمرو الناقد، - واللفظ لسريج - قالا حدثنا عبد، الله بن رجاء عن يونس،…٤٥٦٤قال نفل رسول الله صلى الله عليه وسلم سرية بنحو حديث ابن رجاء .٤٥٦٥أن رسول الله صلى الله عليه وسلم قد كان ينفل بعض من يبعث من السرايا…٤٥٦٦وكان، جليسا لأبي قتادة قال قال أبو قتادة . واقتص الحديث .٤٥٦٧مولى أبي قتادة أن أبا قتادة، قال . وساق الحديث .٤٥٦٨من قتل قتيلا له عليه بينة فله سلبه٤٥٦٩وشمالي، فإذا أنا بين، غلامين من الأنصار حديثة أسنانهما تمنيت لو كنت…٤٥٧٠ما منعك أن تعطيه سلبه٤٥٧١وحدثني زهير بن حرب، حدثنا الوليد بن مسلم، حدثنا صفوان بن عمرو، عن…٤٥٧٢حدثنا زهير بن حرب، حدثنا عمر بن يونس الحنفي، حدثنا عكرمة بن عمار،…٤٥٧٣يا سلمة هب لي المرأة٤٥٧٤أيما قرية أتيتموها وأقمتم فيها فسهمكم فيها وأيما قرية عصت الله ورسوله…٤٥٧٥ومحمد بن عباد، وأبو بكر بن أبي شيبة وإسحاق بن إبراهيم - واللفظ لابن…٤٥٧٦بهذا الإسناد .٤٥٧٧لا نورث ما تركنا صدقة٤٥٧٨ومحمد بن رافع، وعبد بن حميد، - قال ابن رافع حدثنا وقال الآخران،…٤٥٧٩لا نورث ما تركنا فهو صدقة٤٥٨٠لا نورث ما تركنا صدقة إنما يأكل آل محمد - صلى الله عليه وسلم - في هذا…٤٥٨١حدثنا إسحاق بن إبراهيم، و محمد بن رافع وعبد بن حميد قال ابن رافع…٤٥٨٢لا نورث ما تركنا صدقة٤٥٨٣لا يقتسم ورثتي دينارا ما تركت بعد نفقة نسائي ومئونة عاملي فهو صدقة٤٥٨٤بهذا الإسناد . نحوه .٤٥٨٥لا نورث ما تركنا صدقة٤٥٨٦أن رسول الله صلى الله عليه وسلم قسم في النفل للفرس سهمين وللرجل سهما .٤٥٨٧بهذا الإسناد مثله ولم يذكر في النفل .٤٥٨٨اللهم أنجز لي ما وعدتني اللهم آت ما وعدتني اللهم إن تهلك هذه العصابة…٤٥٨٩ماذا عندك يا ثمامة٤٥٩٠أنه سمع أبا هريرة، يقول بعث رسول الله صلى الله عليه وسلم خيلا له نحو…٤٥٩١انطلقوا إلى يهود٤٥٩٢وإسحاق بن منصور، قال ابن رافع حدثنا وقال، إسحاق أخبرنا عبد الرزاق،…٤٥٩٣بهذا الإسناد هذا الحديث وحديث ابن جريج أكثر وأتم .٤٥٩٤لأخرجن اليهود والنصارى من جزيرة العرب حتى لا أدع إلا مسلما٤٥٩٥بهذا الإسناد مثله .٤٥٩٦قوموا إلى سيدكم - أو خيركم٤٥٩٧لقد حكمت فيهم بحكم الله٤٥٩٨ومحمد بن العلاء الهمداني، كلاهما عن ابن نمير، قال ابن العلاء حدثنا…٤٥٩٩لقد حكمت فيهم بحكم الله عز وجل٤٦٠٠أن سعدا، قال وتحجر كلمه للبرء فقال اللهم إنك تعلم أن ليس أحد أحب إلى…٤٦٠١بهذا الإسناد نحوه غير أنه قال فانفجر من ليلته فمازال يسيل حتى مات…٤٦٠٢أن لا يصلين أحد الظهر إلا في بني قريظة٤٦٠٣وحرملة، قالا أخبرنا ابن وهب، أخبرني يونس، عن ابن، شهاب عن أنس بن…٤٦٠٤يا أم أيمن اتركيه ولك كذا وكذا٤٦٠٥قال أصبت جرابا من شحم يوم خيبر - قال - فالتزمته فقلت لا أعطي اليوم…٤٦٠٦حدثنا محمد بن بشار العبدي، حدثنا بهز بن أسد، حدثنا شعبة، حدثني حميد،…٤٦٠٧بسم الله الرحمن الرحيم من محمد رسول الله إلى هرقل عظيم الروم سلام على…٤٦٠٨من محمد عبد الله ورسوله٤٦٠٩أن نبي الله صلى الله عليه وسلم كتب إلى كسرى وإلى قيصر وإلى النجاشي…٤٦١٠وحدثناه محمد بن عبد الله الرزي، حدثنا عبد الوهاب بن عطاء، عن سعيد، عن…٤٦١١ولم يذكر وليس بالنجاشي الذي صلى عليه النبي صلى الله عليه وسلم .٤٦١٢أى عباس ناد أصحاب السمرة٤٦١٣انهزموا ورب الكعبة انهزموا ورب الكعبة٤٦١٤قال كنت مع النبي صلى الله عليه وسلم يوم حنين . وساق الحديث . غير أن…٤٦١٥أنا النبي لا كذب أنا ابن عبد المطلب٤٦١٦أنا النبي لا كذب أنا ابن عبد المطلب اللهم نزل نصرك٤٦١٧أنا النبي لا كذب أنا ابن عبد المطلب٤٦١٨ومحمد بن المثنى، وأبو بكر بن خلاد قالوا حدثنا يحيى، بن سعيد عن سفيان،…٤٦١٩لقد رأى ابن الأكوع فزعا٤٦٢٠إنا قافلون إن شاء الله٤٦٢١والذي نفسي بيده لتضربوه إذا صدقكم وتتركوه إذا كذبكم٤٦٢٢حدثنا شيبان بن فروخ، حدثنا سليمان بن المغيرة، حدثنا ثابت البناني، عن…٤٦٢٣وحدثنيه عبد الله بن هاشم، حدثنا بهز، حدثنا سليمان بن المغيرة، بهذا…٤٦٢٤يا أبا هريرة ادع لي الأنصار٤٦٢٥{ جاء الحق وزهق الباطل إن الباطل كان زهوقا } { جاء الحق وما يبدئ…٤٦٢٦وعبد بن حميد، كلاهما عن عبد الرزاق، أخبرنا الثوري، عن ابن أبي نجيح،…٤٦٢٧لا يقتل قرشي صبرا بعد هذا اليوم إلى يوم القيامة٤٦٢٨بهذا الإسناد . وزاد قال ولم يكن أسلم أحد من عصاة قريش غير مطيع كان…٤٦٢٩هذا ما كاتب عليه محمد رسول الله٤٦٣٠محمد رسول الله٤٦٣١اكتب الشرط بيننا بسم الله الرحمن الرحيم هذا ما قاضى عليه محمد رسول…٤٦٣٢اكتب بسم الله الرحمن الرحيم٤٦٣٣حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، ح وحدثنا ابن نمير،…٤٦٣٤وحدثنا أبو كريب، محمد بن العلاء ومحمد بن عبد الله بن نمير قالا حدثنا…٤٦٣٥وإسحاق، جميعا عن جرير، ح وحدثني أبو سعيد، الأشج حدثنا وكيع، كلاهما عن…٤٦٣٦وحدثني إبراهيم بن سعيد الجوهري، حدثنا أبو أسامة، عن مالك بن مغول، عن…٤٦٣٧لقد أنزلت على آية هي أحب إلى من الدنيا جميعا٤٦٣٨نحو حديث ابن أبي عروبة .٤٦٣٩انصرفا نفي لهم بعهدهم ونستعين الله عليهم٤٦٤٠ألا رجل يأتيني بخبر القوم جعله الله معي يوم القيامة٤٦٤١من يردهم عنا وله الجنة أو هو رفيقي في الجنة٤٦٤٢أنه سمع سهل بن سعد، يسأل عن جرح، رسول الله صلى الله عليه وسلم يوم أحد…٤٦٤٣أنه سمع سهل بن سعد، وهو يسأل عن جرح، رسول الله صلى الله عليه وسلم…٤٦٤٤وزهير بن حرب، وإسحاق بن إبراهيم، وابن، أبي عمر جميعا عن ابن عيينة، ح…٤٦٤٥كيف يفلح قوم شجوا نبيهم وكسروا رباعيته وهو يدعوهم إلى الله٤٦٤٦رب اغفر لقومي فإنهم لا يعلمون٤٦٤٧ومحمد بن بشر، عن الأعمش، بهذا الإسناد غير أنه قال فهو ينضح الدم عن…٤٦٤٨اشتد غضب الله على قوم فعلوا هذا برسول الله صلى الله عليه وسلم٤٦٤٩اللهم عليك بقريش٤٦٥٠اللهم عليك الملأ من قريش أبا جهل بن هشام وعتبة بن ربيعة وعقبة بن أبي…٤٦٥١اللهم عليك بقريش اللهم عليك بقريش اللهم عليك بقريش٤٦٥٢قال استقبل رسول الله صلى الله عليه وسلم البيت فدعا على ستة نفر من…٤٦٥٣لقد لقيت من قومك وكان أشد ما لقيت منهم يوم العقبة إذ عرضت نفسي على…٤٦٥٤هل أنت إلا إصبع دميت وفي سبيل الله ما لقيت٤٦٥٥وإسحاق بن إبراهيم، جميعا عن ابن عيينة، عن الأسود بن قيس، بهذا الإسناد…٤٦٥٦أنه سمع جندبا، يقول أبطأ جبريل على رسول الله صلى الله عليه وسلم فقال…٤٦٥٧ومحمد بن رافع، - واللفظ لابن رافع - قال إسحاق أخبرنا وقال ابن رافع،…٤٦٥٨ومحمد بن المثنى، وابن، بشار قالوا حدثنا محمد، بن جعفر عن شعبة، ح…٤٦٥٩أى سعد ألم تسمع إلى ما قال أبو حباب - يريد عبد الله بن أبى - قال كذا…٤٦٦٠في هذا الإسناد بمثله وزاد وذلك قبل أن يسلم عبد الله .٤٦٦١حدثنا محمد بن عبد الأعلى القيسي، حدثنا المعتمر، عن أبيه، عن أنس بن…٤٦٦٢من ينظر لنا ما صنع أبو جهل٤٦٦٣من يعلم لي ما فعل أبو جهل٤٦٦٤من لكعب بن الأشرف فإنه قد آذى الله ورسوله٤٦٦٥الله أكبر خربت خيبر إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٤٦٦٦خربت خيبر إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٤٦٦٧إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٤٦٦٨ومحمد بن عباد، - واللفظ لابن عباد - قالا حدثنا حاتم، - وهو ابن…٤٦٦٩- ونسبه غير ابن وهب فقال ابن عبد الله بن كعب بن مالك - أن سلمة، بن…٤٦٧٠والله لولا أنت ما اهتدينا ولا تصدقنا ولا صلينا فأنزلن سكينة علينا إن…
