٣١ حديثًا
٤٢٠٤ما حق امرئ مسلم له شىء يريد أن يوصي فيه يبيت ليلتين إلا ووصيته مكتوبة…٤٢٠٥وله شىء يوصي فيه٤٢٠٦وحدثنا أبو كامل الجحدري، حدثنا حماد يعني ابن زيد، ح وحدثني زهير بن،…٤٢٠٧ما حق امرئ مسلم له شىء يوصي فيه يبيت ثلاث ليال إلا ووصيته عنده مكتوبة٤٢٠٨وحرملة، قالا أخبرنا ابن وهب، أخبرني يونس، ح وحدثني عبد الملك بن شعيب…٤٢٠٩حدثنا يحيى بن يحيى التميمي، أخبرنا إبراهيم بن سعد، عن ابن شهاب، عن…٤٢١٠وحرملة، قالا أخبرنا ابن وهب، أخبرني يونس، ح وحدثنا إسحاق، بن إبراهيم…٤٢١١قال دخل النبي صلى الله عليه وسلم على يعودني . فذكر بمعنى حديث الزهري…٤٢١٢قال مرضت فأرسلت إلى النبي صلى الله عليه وسلم فقلت دعني أقسم مالي حيث…٤٢١٣وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن سماك، بهذا الإسناد…٤٢١٤قال عادني النبي صلى الله عليه وسلم فقلت أوصي بمالي كله . قال " لا " .…٤٢١٥أن النبي صلى الله عليه وسلم دخل على سعد يعوده بمكة فبكى قال " ما…٤٢١٦من ولد سعد قالوا مرض سعد بمكة فأتاه رسول الله صلى الله عليه وسلم…٤٢١٧وحدثني محمد بن المثنى، حدثنا عبد الأعلى، حدثنا هشام، عن محمد، عن…٤٢١٨الثلث والثلث كثير٤٢١٩وقتيبة بن سعيد، وعلي بن حجر، قالوا حدثنا إسماعيل، - وهو ابن جعفر - عن…٤٢٢٠أن رجلا، قال للنبي صلى الله عليه وسلم إن أمي افتلتت نفسها وإني أظنها…٤٢٢١أن رجلا، أتى النبي صلى الله عليه وسلم فقال يا رسول الله إن أمي افتلتت…٤٢٢٢بهذا الإسناد أما أبو أسامة وروح ففي حديثهما فهل لي أجر كما قال يحيى…٤٢٢٣إذا مات الإنسان انقطع عنه عمله إلا من ثلاثة إلا من صدقة جارية أو علم…٤٢٢٤إن شئت حبست أصلها وتصدقت بها٤٢٢٥أو يطعم صديقا غير متمول فيه٤٢٢٦قال أصبت أرضا من أرض خيبر فأتيت رسول الله صلى الله عليه وسلم فقلت…٤٢٢٧قال سألت عبد الله بن أبي أوفى هل أوصى رسول الله صلى الله عليه وسلم…٤٢٢٨بهذا الإسناد . مثله غير أن في حديث وكيع قلت فكيف أمر الناس بالوصية…٤٢٢٩قالت ما ترك رسول الله صلى الله عليه وسلم دينارا ولا درهما ولا شاة ولا…٤٢٣٠وعثمان بن أبي شيبة، وإسحاق بن إبراهيم، كلهم عن جرير، ح وحدثنا علي بن…٤٢٣١قال ذكروا عند عائشة أن عليا كان وصيا فقالت متى أوصى إليه فقد كنت…٤٢٣٢ائتوني أكتب لكم كتابا لا تضلوا بعدي٤٢٣٣ائتوني بالكتف والدواة - أو اللوح والدواة - أكتب لكم كتابا لن تضلوا…٤٢٣٤هلم أكتب لكم كتابا لا تضلون بعده
