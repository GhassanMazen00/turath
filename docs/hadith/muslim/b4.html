كتاب الصلاة

٣٢١ حديثًا•صفحة ١ من ٣
٨٣٧يا بلال قم فناد بالصلاة٨٣٨قال أمر بلال أن يشفع، الأذان ويوتر الإقامة . زاد يحيى في حديثه عن ابن…٨٣٩قال ذكروا أن يعلموا، وقت الصلاة بشىء يعرفونه فذكروا أن ينوروا نارا أو…٨٤٠بهذا الإسناد لما كثر الناس ذكروا أن يعلموا . بمثل حديث الثقفي غير أنه…٨٤١وعبد الوهاب بن عبد المجيد، قالا حدثنا أيوب، عن أبي قلابة، عن أنس، قال…٨٤٢الله أكبر الله أكبر أشهد أن لا إله إلا الله أشهد أن لا إله إلا الله…٨٤٣قال كان لرسول الله صلى الله عليه وسلم مؤذنان بلال وابن أم مكتوم…٨٤٤مثله .٨٤٥قالت كان ابن أم مكتوم يؤذن لرسول الله صلى الله عليه وسلم وهو أعمى .٨٤٦وسعيد بن عبد الرحمن، عن هشام، بهذا الإسناد مثله .٨٤٧وحدثني زهير بن حرب، حدثنا يحيى، - يعني ابن سعيد - عن حماد بن سلمة،…٨٤٨إذا سمعتم النداء فقولوا مثل ما يقول المؤذن٨٤٩إذا سمعتم المؤذن فقولوا مثل ما يقول ثم صلوا على فإنه من صلى على صلاة…٨٥٠إذا قال المؤذن الله أكبر الله أكبر . فقال أحدكم الله أكبر الله أكبر .…٨٥١من قال حين يسمع المؤذن أشهد أن لا إله إلا الله وحده لا شريك له وأن…٨٥٤إن الشيطان إذا سمع النداء بالصلاة ذهب حتى يكون مكان الروحاء٨٥٥بهذا الإسناد .٨٥٦إن الشيطان إذا سمع النداء بالصلاة أحال له ضراط حتى لا يسمع صوته فإذا…٨٥٧إذا أذن المؤذن أدبر الشيطان وله حصاص٨٥٨إن الشيطان إذا نودي بالصلاة ولى وله حصاص٨٥٩إذا نودي للصلاة أدبر الشيطان له ضراط حتى لا يسمع التأذين فإذا قضي…٨٦٠حتى يظل الرجل إن يدري كيف صلى٨٦١وسعيد بن منصور، وأبو بكر بن أبي شيبة وعمرو الناقد وزهير بن حرب وابن…٨٦٢أن ابن عمر، قال كان رسول الله صلى الله عليه وسلم إذا قام للصلاة رفع…٨٦٣حدثني محمد بن رافع، حدثنا حجين، - وهو ابن المثنى - حدثنا الليث، عن…٨٦٤أنه رأى مالك بن الحويرث إذا صلى كبر ثم رفع يديه وإذا أراد أن يركع رفع…٨٦٥سمع الله لمن حمده٨٦٦بهذا الإسناد أنه رأى نبي الله صلى الله عليه وسلم وقال حتى يحاذي بهما…٨٦٧أن أبا هريرة، كان يصلي لهم فيكبر كلما خفض ورفع فلما انصرف قال والله…٨٦٨سمع الله لمن حمده٨٦٩أنه سمع أبا هريرة، يقول كان رسول الله صلى الله عليه وسلم إذا قام إلى…٨٧٠أن أبا هريرة، كان - حين يستخلفه مروان على المدينة - إذا قام للصلاة…٨٧١أن أبا هريرة، كان يكبر في الصلاة كلما رفع ووضع . فقلنا يا أبا هريرة…٨٧٢أنه كان يكبر كلما خفض ورفع ويحدث أن رسول الله صلى الله عليه وسلم كان…٨٧٣وخلف بن هشام، جميعا عن حماد، - قال يحيى أخبرنا حماد بن زيد، - عن…٨٧٤لا صلاة لمن لم يقرأ بفاتحة الكتاب٨٧٥لا صلاة لمن لم يقترئ بأم القرآن٨٧٦لا صلاة لمن لم يقرأ بأم القرآن٨٧٧وعبد بن حميد، قالا أخبرنا عبد الرزاق، أخبرنا معمر، عن الزهري، بهذا…٨٧٨من صلى صلاة لم يقرأ فيها بأم القرآن فهى خداج - ثلاثا - غير تمام٨٧٩من صلى صلاة فلم يقرأ فيها بأم القرآن٨٨٠من صلى صلاة لم يقرأ فيها بفاتحة الكتاب فهى خداج٨٨٢لا صلاة إلا بقراءة٨٨٣وزهير بن حرب، - واللفظ لعمرو - قالا حدثنا إسماعيل بن إبراهيم، أخبرنا…٨٨٤حدثنا يحيى بن يحيى، أخبرنا يزيد، - يعني ابن زريع - عن حبيب المعلم، عن…٨٨٥ارجع فصل فإنك لم تصل٨٨٦إذا قمت إلى الصلاة فأسبغ الوضوء ثم استقبل القبلة فكبر٨٨٧أيكم قرأ خلفي بسبح اسم ربك الأعلى٨٨٨ومحمد بن بشار، قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن قتادة، قال…٨٨٩قد علمت أن بعضكم خالجنيها٨٩٠وابن، بشار كلاهما عن غندر، قال ابن المثنى حدثنا محمد بن جعفر، حدثنا…٨٩١حدثنا محمد بن المثنى، حدثنا أبو داود، حدثنا شعبة، في هذا الإسناد .…٨٩٢أن عمر بن الخطاب، كان يجهر بهؤلاء الكلمات يقول سبحانك اللهم وبحمدك…٨٩٣أنه سمع أنس بن مالك، يذكر ذلك .٨٩٤أنزلت على آنفا سورة٨٩٥نهر وعدنيه ربي عز وجل في الجنة عليه حوض٨٩٦سمع الله لمن حمده٨٩٧إن الله هو السلام فإذا قعد أحدكم في الصلاة فليقل التحيات لله والصلوات…٨٩٨ثم يتخير من المسألة ما شاء٨٩٩ثم ليتخير بعد من المسألة ما شاء أو ما أحب٩٠٠ثم يتخير بعد من الدعاء٩٠١يقول علمني رسول الله صلى الله عليه وسلم التشهد كفي بين كفيه كما…٩٠٢التحيات المباركات الصلوات الطيبات لله السلام عليك أيها النبي ورحمة…٩٠٣حدثنا أبو بكر بن أبي شيبة، حدثنا يحيى بن آدم، حدثنا عبد الرحمن بن…٩٠٤إذا صليتم فأقيموا صفوفكم ثم ليؤمكم أحدكم فإذا كبر فكبروا وإذا قال غير…٩٠٥وإذا قرأ فأنصتوا٩٠٦فإن الله عز وجل قضى على لسان نبيه صلى الله عليه وسلم سمع الله لمن حمده٩٠٧اللهم صل على محمد وعلى آل محمد كما صليت على آل إبراهيم وبارك على محمد…٩٠٨قولوا اللهم صل على محمد وعلى آل محمد كما صليت على آل إبراهيم إنك حميد…٩٠٩ومسعر، عن الحكم، بهذا الإسناد مثله . وليس في حديث مسعر ألا أهدي لك…٩١٠وبارك على محمد٩١١قولوا اللهم صل على محمد وعلى أزواجه وذريته كما صليت على آل إبراهيم…٩١٢من صلى على واحدة صلى الله عليه عشرا٩١٣إذا قال الإمام سمع الله لمن حمده . فقولوا اللهم ربنا لك الحمد . فإنه…٩١٤حدثنا قتيبة بن سعيد، حدثنا يعقوب، - يعني ابن عبد الرحمن - عن سهيل، عن…٩١٥إذا أمن الإمام فأمنوا فإنه من وافق تأمينه تأمين الملائكة غفر له ما…٩١٦وأبو سلمة بن عبد الرحمن أن أبا هريرة، قال سمعت رسول الله صلى الله…٩١٧إذا قال أحدكم في الصلاة آمين . والملائكة في السماء آمين . فوافق…٩١٨إذا قال أحدكم آمين . والملائكة في السماء آمين . فوافقت إحداهما الأخرى…٩١٩حدثنا محمد بن رافع، حدثنا عبد الرزاق، حدثنا معمر، عن همام بن منبه، عن…٩٢٠إذا قال القارئ غير المغضوب عليهم ولا الضالين . فقال من خلفه آمين .…٩٢١إنما جعل الإمام ليؤتم به فإذا كبر فكبروا وإذا سجد فاسجدوا وإذا رفع…٩٢٢حدثنا قتيبة بن سعيد، حدثنا ليث، ح وحدثنا محمد بن رمح، أخبرنا الليث،…٩٢٣فإذا صلى قائما فصلوا قياما٩٢٤إذا صلى قائما فصلوا قياما٩٢٥أن النبي صلى الله عليه وسلم سقط من فرسه فجحش شقه الأيمن . وساق الحديث…٩٢٦إنما جعل الإمام ليؤتم به فإذا ركع فاركعوا وإذا رفع فارفعوا وإذا صلى…٩٢٧بهذا الإسناد نحوه .٩٢٨إن كدتم آنفا لتفعلون فعل فارس والروم يقومون على ملوكهم وهم قعود فلا…٩٢٩حدثنا يحيى بن يحيى، أخبرنا حميد بن عبد الرحمن الرؤاسي، عن أبيه، عن…٩٣٠إنما الإمام ليؤتم به فلا تختلفوا عليه فإذا كبر فكبروا وإذا ركع…٩٣١حدثنا محمد بن رافع، حدثنا عبد الرزاق، حدثنا معمر، عن همام بن منبه، عن…٩٣٢لا تبادروا الإمام إذا كبر فكبروا وإذا قال ولا الضالين . فقولوا آمين .…٩٣٣ولا الضالين . فقولوا آمين٩٣٤إنما الإمام جنة فإذا صلى قاعدا فصلوا قعودا وإذا قال سمع الله لمن حمده…٩٣٥إنما جعل الإمام ليؤتم به فإذا كبر فكبروا وإذا ركع فاركعوا وإذا قال…٩٣٦حدثنا أحمد بن عبد الله بن يونس، حدثنا زائدة، حدثنا موسى بن أبي عائشة،…٩٣٧وعبد بن حميد، - واللفظ لابن رافع - قالا حدثنا عبد الرزاق، أخبرنا…٩٣٨أن عائشة، زوج النبي صلى الله عليه وسلم قالت لما ثقل رسول الله صلى…٩٣٩أن عائشة، زوج النبي صلى الله عليه وسلم قالت لقد راجعت رسول الله صلى…٩٤٠مروا أبا بكر فليصل بالناس٩٤١مروا أبا بكر فليصل بالناس٩٤٢بهذا الإسناد نحوه وفي حديثهما لما مرض رسول الله صلى الله عليه وسلم…٩٤٣قالت أمر رسول الله صلى الله عليه وسلم أبا بكر أن يصلي بالناس في مرضه…٩٤٤وحسن الحلواني، وعبد بن حميد، - قال عبد أخبرني وقال الآخران، حدثنا…٩٤٥وزهير بن حرب، قالا حدثنا سفيان بن عيينة، عن الزهري، عن أنس، قال آخر…٩٤٦وعبد بن حميد، جميعا عن عبد الرزاق، أخبرنا معمر، عن الزهري، قال أخبرني…٩٤٧وهارون بن عبد الله، قالا حدثنا عبد الصمد، قال سمعت أبي يحدث، قال…٩٤٨مروا أبا بكر فليصل بالناس٩٤٩يا أبا بكر ما منعك أن تثبت إذ أمرتك٩٥٠بمثل حديث مالك . وفي حديثهما فرفع أبو بكر يديه فحمد الله ورجع القهقرى…٩٥١قال ذهب نبي الله صلى الله عليه وسلم يصلح بين بني عمرو بن عوف . بمثل…٩٥٢وحسن بن علي الحلواني، جميعا عن عبد الرزاق، قال ابن رافع حدثنا عبد…٩٥٣والحلواني، قالا حدثنا عبد الرزاق، عن ابن جريج، حدثني ابن شهاب، عن…٩٥٤التسبيح للرجال والتصفيق للنساء٩٥٥وحدثنا قتيبة بن سعيد، حدثنا الفضيل يعني ابن عياض، ح وحدثنا أبو كريب،…٩٥٦حدثنا محمد بن رافع، حدثنا عبد الرزاق، أخبرنا معمر، عن همام، عن أبي…٩٥٧يا فلان ألا تحسن صلاتك ألا ينظر المصلي إذا صلى كيف يصلي فإنما يصلي…٩٥٨هل ترون قبلتي ها هنا فوالله ما يخفى على ركوعكم ولا سجودكم إني لأراكم…٩٥٩أقيموا الركوع والسجود فوالله إني لأراكم من بعدي - وربما قال من بعد…٩٦٠أتموا الركوع والسجود فوالله إني لأراكم من بعد ظهري إذا ما ركعتم وإذا…٩٦١أيها الناس إني إمامكم فلا تسبقوني بالركوع ولا بالسجود ولا بالقيام ولا…٩٦٢وإسحاق بن إبراهيم، عن ابن فضيل، جميعا عن المختار، عن أنس، عن النبي…٩٦٣أما يخشى الذي يرفع رأسه قبل الإمام أن يحول الله رأسه رأس حمار٩٦٤ما يأمن الذي يرفع رأسه في صلاته قبل الإمام أن يحول الله صورته في صورة…٩٦٥أن يجعل الله وجهه وجه حمار٩٦٦لينتهين أقوام يرفعون أبصارهم إلى السماء في الصلاة أو لا ترجع إليهم٩٦٧لينتهين أقوام عن رفعهم أبصارهم عند الدعاء في الصلاة إلى السماء أو…٩٦٨ما لي أراكم رافعي أيديكم كأنها أذناب خيل شمس اسكنوا في الصلاة٩٦٩بهذا الإسناد نحوه .٩٧٠علام تومئون بأيديكم كأنها أذناب خيل شمس إنما يكفي أحدكم أن يضع يده…٩٧١ما شأنكم تشيرون بأيديكم كأنها أذناب خيل شمس إذا سلم أحدكم فليلتفت إلى…٩٧٢استووا ولا تختلفوا فتختلف قلوبكم ليلني منكم أولو الأحلام والنهى ثم…٩٧٣بهذا الإسناد نحوه .٩٧٤ليلني منكم أولو الأحلام والنهى ثم الذين يلونهم - ثلاثا - وإياكم…٩٧٥سووا صفوفكم فإن تسوية الصف من تمام الصلاة٩٧٦أتموا الصفوف فإني أراكم خلف ظهري٩٧٧أقيموا الصف في الصلاة فإن إقامة الصف من حسن الصلاة٩٧٨لتسون صفوفكم أو ليخالفن الله بين وجوهكم٩٧٩عباد الله لتسون صفوفكم أو ليخالفن الله بين وجوهكم٩٨٠بهذا الإسناد نحوه .٩٨١لو يعلم الناس ما في النداء والصف الأول ثم لم يجدوا إلا أن يستهموا…٩٨٢تقدموا فائتموا بي وليأتم بكم من بعدكم لا يزال قوم يتأخرون حتى يؤخرهم…٩٨٣قال رأى رسول الله صلى الله عليه وسلم قوما في مؤخر المسجد . فذكر مثله .٩٨٤لو تعلمون - أو يعلمون - ما في الصف المقدم لكانت قرعة٩٨٥خير صفوف الرجال أولها وشرها آخرها وخير صفوف النساء آخرها وشرها أولها٩٨٦بهذا الإسناد .٩٨٧حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن سفيان، عن أبي حازم، عن سهل…٩٨٨إذا استأذنت أحدكم امرأته إلى المسجد فلا يمنعها٩٨٩لا تمنعوا نساءكم المساجد إذا استأذنكم إليها
