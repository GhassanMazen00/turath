كتاب اللباس والزينة

١٨٤ حديثًا•صفحة ١ من ٢
٥٣٨٥الذي يشرب في آنية الفضة إنما يجرجر في بطنه نار جهنم٥٣٨٦أن الذي، يأكل أو يشرب في آنية الفضة والذهب٥٣٨٧من شرب في إناء من ذهب أو فضة فإنما يجرجر في بطنه نارا من جهنم٥٣٨٨حدثنا يحيى بن يحيى التميمي، أخبرنا أبو خيثمة، عن أشعث بن أبي الشعثاء،…٥٣٨٩بهذا الإسناد مثله إلا قوله وإبرار القسم أو المقسم . فإنه لم يذكر هذا…٥٣٩٠وحدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، ح وحدثنا عثمان بن أبي،…٥٣٩١وليث بن، أبي سليم عن أشعث بن أبي الشعثاء، بإسنادهم ولم يذكر زيادة…٥٣٩٢وعمرو بن محمد، قالا حدثنا سفيان، عن أشعث بن أبي الشعثاء، بإسنادهم…٥٣٩٤لا تشربوا في إناء الذهب والفضة ولا تلبسوا الديباج والحرير فإنه لهم في…٥٣٩٥وحدثناه ابن أبي عمر، حدثنا سفيان، عن أبي فروة الجهني، قال سمعت عبد…٥٣٩٦سمعه من ابن أبي ليلى، عن حذيفة، ثم حدثنا أبو فروة، قال سمعت ابن عكيم،…٥٣٩٧أنه سمع عبد الرحمن، - يعني ابن أبي ليلى - قال شهدت حذيفة استسقى…٥٣٩٨وابن، بشار قالا حدثنا محمد بن جعفر، ح وحدثنا محمد بن المثنى، حدثنا…٥٣٩٩وحدثنا إسحاق بن إبراهيم، أخبرنا جرير، عن منصور، ح وحدثنا محمد بن…٥٤٠٠لا تلبسوا الحرير ولا الديباج ولا تشربوا في آنية الذهب والفضة ولا…٥٤٠١إنما يلبس هذه من لا خلاق له في الآخرة٥٤٠٢وحدثنا ابن نمير، حدثنا أبي ح، وحدثنا أبو بكر بن أبي شيبة، حدثنا أبو…٥٤٠٣إنما يلبس الحرير في الدنيا من لا خلاق له في الآخرة٥٤٠٤إنما هذه لباس من لا خلاق له٥٤٠٥بهذا الإسناد مثله .٥٤٠٦إنما يلبس هذا من لا خلاق له٥٤٠٧إنما بعثت بها إليك لتنتفع بها ولم أبعث بها إليك لتلبسها٥٤٠٨إنما بعثت بها إليك لتصيب بها مالا٥٤٠٩إنما يلبس الحرير من لا خلاق له٥٤١٠لا تلبسوا الحرير فإنه من لبسه في الدنيا لم يلبسه في الآخرة٥٤١١حدثنا أحمد بن عبد الله بن يونس، حدثنا زهير، حدثنا عاصم الأحول، عن…٥٤١٢حدثني زهير بن حرب، حدثنا جرير بن عبد الحميد، ح وحدثنا ابن نمير، حدثنا…٥٤١٣لا يلبس الحرير إلا من ليس له منه شىء في الآخرة إلا هكذا٥٤١٤حدثنا محمد بن عبد الأعلى، حدثنا المعتمر، عن أبيه، حدثنا أبو عثمان،…٥٤١٥وابن، بشار - واللفظ لابن المثنى - قالا حدثنا محمد، بن جعفر حدثنا…٥٤١٦ومحمد بن المثنى، قالا حدثنا معاذ، - وهو ابن هشام - حدثني أبي، عن…٥٤١٧أن عمر، بن الخطاب خطب بالجابية فقال نهى نبي الله صلى الله عليه وسلم…٥٤١٨بهذا الإسناد مثله .٥٤١٩نهاني عنه جبريل٥٤٢٠إني لم أبعث بها إليك لتلبسها إنما بعثت بها إليك لتشققها خمرا بين…٥٤٢١بهذا الإسناد في حديث معاذ فأمرني فأطرتها بين نسائي . وفي حديث محمد بن…٥٤٢٢شققه خمرا بين الفواطم٥٤٢٣قال كساني رسول الله صلى الله عليه وسلم حلة سيراء فخرجت فيها فرأيت…٥٤٢٤إني لم أبعث بها إليك لتلبسها وإنما بعثت بها إليك لتنتفع بثمنها٥٤٢٥من لبس الحرير في الدنيا لم يلبسه في الآخرة٥٤٢٦من لبس الحرير في الدنيا لم يلبسه في الآخرة٥٤٢٧لا ينبغي هذا للمتقين٥٤٢٨بهذا الإسناد .٥٤٢٩أن أنس بن مالك، أنبأهم أن رسول الله صلى الله عليه وسلم رخص لعبد…٥٤٣٠بهذا الإسناد ولم يذكر في السفر .٥٤٣١قال رخص رسول الله صلى الله عليه وسلم - أو رخص - للزبير بن العوام وعبد…٥٤٣٢وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، بهذا الإسناد مثله .٥٤٣٣أن أنسا، أخبره أن عبد الرحمن بن عوف والزبير بن العوام شكوا إلى رسول…٥٤٣٤إن هذه من ثياب الكفار فلا تلبسها٥٤٣٥.٥٤٣٦أأمك أمرتك بهذا٥٤٣٧حدثنا يحيى بن يحيى، قال قرأت على مالك عن نافع، عن إبراهيم بن عبد،…٥٤٣٨أن أباه، حدثه أنه، سمع علي بن أبي طالب، يقول نهاني النبي صلى الله…٥٤٣٩حدثنا عبد بن حميد، حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن…٥٤٤٠حدثنا هداب بن خالد، حدثنا همام، حدثنا قتادة، قال قلنا لأنس بن مالك أى…٥٤٤١قال كان أحب الثياب إلى رسول الله صلى الله عليه وسلم الحبرة .٥٤٤٢حدثنا شيبان بن فروخ، حدثنا سليمان بن المغيرة، حدثنا حميد، عن أبي…٥٤٤٣ومحمد بن حاتم، ويعقوب بن إبراهيم، جميعا عن ابن علية، قال ابن حجر…٥٤٤٤بهذا الإسناد مثله وقال إزارا غليظا .٥٤٤٥قالت خرج النبي صلى الله عليه وسلم ذات غداة وعليه مرط مرحل من شعر أسود…٥٤٤٦قالت كان وسادة رسول الله صلى الله عليه وسلم التي يتكئ عليها من أدم…٥٤٤٧قالت إنما كان فراش رسول الله صلى الله عليه وسلم الذي ينام عليه أدما…٥٤٤٨بهذا الإسناد وقالا ضجاع رسول الله صلى الله عليه وسلم . في حديث أبي…٥٤٤٩وعمرو الناقد، وإسحاق بن إبراهيم، - واللفظ لعمرو - قال عمرو وقتيبة…٥٤٥٠حدثنا محمد بن عبد الله بن نمير، حدثنا وكيع، عن سفيان، عن محمد بن…٥٤٥١بهذا الإسناد وزاد فأدعها .٥٤٥٢فراش للرجل وفراش لامرأته والثالث للضيف والرابع للشيطان٥٤٥٣لا ينظر الله إلى من جر ثوبه خيلاء٥٤٥٤وعبيد الله بن سعيد، قالا حدثنا يحيى، - وهو القطان - كلهم عن عبيد…٥٤٥٥إن الذي يجر ثيابه من الخيلاء لا ينظر الله إليه يوم القيامة٥٤٥٦وجبلة بن، سحيم عن ابن عمر، عن النبي صلى الله عليه وسلم بمثل حديثهم .٥٤٥٧من جر ثوبه من الخيلاء لم ينظر الله إليه يوم القيامة٥٤٥٨وحدثنا ابن نمير، حدثنا إسحاق بن سليمان، حدثنا حنظلة بن أبي سفيان، قال…٥٤٥٩من جر إزاره لا يريد بذلك إلا المخيلة فإن الله لا ينظر إليه يوم القيامة٥٤٦٠وحدثنا ابن نمير، حدثنا أبي، حدثنا عبد الملك، - يعني ابن أبي سليمان -…٥٤٦١لا ينظر الله إليه يوم القيامة٥٤٦٢يا عبد الله ارفع إزارك٥٤٦٣إن الله لا ينظر إلى من يجر إزاره بطرا٥٤٦٤بهذا الإسناد وفي حديث ابن جعفر كان مروان يستخلف أبا هريرة . وفي حديث…٥٤٦٥بينما رجل يمشي قد أعجبته جمته وبرداه إذ خسف به الأرض فهو يتجلجل في…٥٤٦٦وحدثنا عبيد الله بن معاذ، حدثنا أبي ح، وحدثنا محمد بن بشار، عن محمد…٥٤٦٧بينما رجل يتبختر يمشي في برديه قد أعجبته نفسه فخسف الله به الأرض فهو…٥٤٦٨بينما رجل يتبختر في بردين٥٤٦٩إن رجلا ممن كان قبلكم يتبختر في حلة٥٤٧٠حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا شعبة، عن قتادة، عن النضر بن…٥٤٧١وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، بهذا الإسناد .٥٤٧٢يعمد أحدكم إلى جمرة من نار فيجعلها في يده٥٤٧٣إني كنت ألبس هذا الخاتم وأجعل فصه من داخل٥٤٧٤وحدثناه أبو بكر بن أبي شيبة، حدثنا محمد بن بشر، ح وحدثنيه زهير بن…٥٤٧٥وحدثنيه أحمد بن عبدة، حدثنا عبد الوارث، حدثنا أيوب، ح وحدثنا محمد بن،…٥٤٧٦قال اتخذ رسول الله صلى الله عليه وسلم خاتما من ورق فكان في يده ثم كان…٥٤٧٧لا ينقش أحد على نقش خاتمي هذا٥٤٧٨إني اتخذت خاتما من فضة ونقشت فيه محمد رسول الله . فلا ينقش أحد على…٥٤٧٩وحدثنا أحمد بن حنبل، وأبو بكر بن أبي شيبة وزهير بن حرب قالوا حدثنا…٥٤٨٠وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، حدثنا شعبة، قال سمعت…٥٤٨١أن نبي الله صلى الله عليه وسلم كان أراد أن يكتب إلى العجم فقيل له إن…٥٤٨٢أن النبي صلى الله عليه وسلم أراد أن يكتب إلى كسرى وقيصر والنجاشي .…٥٤٨٣أنه أبصر في يد رسول الله صلى الله عليه وسلم خاتما من ورق يوما واحدا -…٥٤٨٤أن ابن شهاب، أخبره أن أنس بن مالك أخبره أنه، رأى في يد رسول الله صلى…٥٤٨٥بهذا الإسناد مثله .٥٤٨٦قال كان خاتم رسول الله صلى الله عليه وسلم من ورق وكان فصه حبشيا .٥٤٨٧وعباد بن موسى، قالا حدثنا طلحة بن يحيى، - وهو الأنصاري ثم الزرقي - عن…٥٤٨٨بهذا الإسناد . مثل حديث طلحة بن يحيى .٥٤٨٩قال كان خاتم النبي صلى الله عليه وسلم في هذه . وأشار إلى الخنصر من…٥٤٩٠حدثني محمد بن عبد الله بن نمير، وأبو كريب جميعا عن ابن إدريس، -…٥٤٩١وحدثنا ابن أبي عمر، حدثنا سفيان، عن عاصم بن كليب، عن ابن لأبي، موسى…٥٤٩٢وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن عاصم بن كليب، قال…٥٤٩٣قال قال علي نهاني رسول الله صلى الله عليه وسلم أن أتختم في إصبعي هذه…٥٤٩٤استكثروا من النعال فإن الرجل لا يزال راكبا ما انتعل٥٤٩٥إذا انتعل أحدكم فليبدأ باليمنى وإذا خلع فليبدأ بالشمال ولينعلهما…٥٤٩٦لا يمش أحدكم في نعل واحدة لينعلهما جميعا أو ليخلعهما جميعا٥٤٩٧إذا انقطع شسع أحدكم فلا يمش في الأخرى حتى يصلحها٥٤٩٨وأبي، صالح عن أبي هريرة، عن النبي صلى الله عليه وسلم بهذا المعنى .٥٤٩٩وحدثنا قتيبة بن سعيد، عن مالك بن أنس، - فيما قرئ عليه - عن أبي…٥٥٠٠إذا انقطع شسع أحدكم - أو من انقطع شسع نعله - فلا يمش في نعل واحدة حتى…٥٥٠١حدثنا قتيبة، حدثنا ليث، ح وحدثنا ابن رمح، أخبرنا الليث، عن أبي…٥٥٠٢لا تمش في نعل واحد ولا تحتب في إزار واحد ولا تأكل بشمالك ولا تشتمل…٥٥٠٣لا يستلقين أحدكم ثم يضع إحدى رجليه على الأخرى٥٥٠٤أنه رأى رسول الله صلى الله عليه وسلم مستلقيا في المسجد واضعا إحدى…٥٥٠٥وحرملة، قالا أخبرنا ابن وهب، أخبرني يونس، ح وحدثنا إسحاق بن إبراهيم،…٥٥٠٦وأبو الربيع، وقتيبة بن سعيد، قال يحيى أخبرنا حماد، بن زيد وقال…٥٥٠٧وعمرو الناقد، وزهير بن حرب، وابن، نمير وأبو كريب قالوا حدثنا إسماعيل،…٥٥٠٨قال أتي بأبي قحافة أو جاء عام الفتح أو يوم الفتح ورأسه ولحيته مثل…٥٥٠٩غيروا هذا بشىء واجتنبوا السواد٥٥١٠إن اليهود والنصارى لا يصبغون فخالفوهم٥٥١١ما يخلف الله وعده ولا رسله٥٥١٢حازم بهذا الإسناد أن جبريل، وعد رسول الله صلى الله عليه وسلم أن يأتيه…٥٥١٣إن جبريل كان وعدني أن يلقاني الليلة فلم يلقني أم والله ما أخلفني٥٥١٤لا تدخل الملائكة بيتا فيه كلب ولا صورة٥٥١٥لا تدخل الملائكة بيتا فيه كلب ولا صورة٥٥١٦وعبد بن حميد، قالا أخبرنا عبد الرزاق، أخبرنا معمر، عن الزهري، بهذا…٥٥١٧إن الملائكة لا تدخل بيتا فيه صورة٥٥١٨لا تدخل الملائكة بيتا فيه صورة٥٥٢٠لا تدخل الملائكة بيتا فيه كلب ولا تماثيل٥٥٢٣قالت قدم رسول الله صلى الله عليه وسلم من سفر وقد سترت على بابي درنوكا…٥٥٢٤بهذا الإسناد وليس في حديث عبدة قدم من سفر .٥٥٢٥إن من أشد الناس عذابا يوم القيامة الذين يشبهون بخلق الله٥٥٢٦أن عائشة، حدثته أن رسول الله صلى الله عليه وسلم دخل عليها . بمثل حديث…٥٥٢٧إن أشد الناس عذابا٥٥٢٨يا عائشة أشد الناس عذابا عند الله يوم القيامة الذين يضاهون بخلق الله٥٥٢٩أنه كان لها ثوب فيه تصاوير ممدود إلى سهوة فكان النبي صلى الله عليه…٥٥٣٠وعقبة بن مكرم، عن سعيد بن عامر، ح وحدثناه إسحاق بن إبراهيم، أخبرنا…٥٥٣١قالت دخل النبي صلى الله عليه وسلم على وقد سترت نمطا فيه تصاوير فنحاه…٥٥٣٢أن بكيرا، حدثه أن عبد الرحمن بن القاسم حدثه أن أباه حدثه عن عائشة،…٥٥٣٣ما بال هذه النمرقة٥٥٣٤وابن، رمح عن الليث بن سعد، ح وحدثنا إسحاق بن إبراهيم، أخبرنا الثقفي،…٥٥٣٥الذين يصنعون الصور يعذبون يوم القيامة يقال لهم أحيوا ما خلقتم٥٥٣٦حدثنا أبو الربيع، وأبو كامل قالا حدثنا حماد، ح وحدثني زهير بن حرب،…٥٥٣٧إن أشد الناس عذابا يوم القيامة المصورون٥٥٣٨إن من أشد أهل النار يوم القيامة عذابا المصورون
