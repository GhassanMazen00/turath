٦٣٢٣قال مات ابن لأبي طلحة . واقتص الحديث بمثله .٦٣٢٤يا بلال حدثني بأرجى عمل عملته عندك في الإسلام منفعة فإني سمعت الليلة…٦٣٢٥قيل لي أنت منهم٦٣٢٦ومحمد بن رافع، - واللفظ لابن رافع - قال إسحاق أخبرنا وقال ابن رافع،…٦٣٢٧أنه سمع الأسود، يقول سمعت أبا موسى، يقول لقد قدمت أنا وأخي، من اليمن…٦٣٢٨ومحمد بن المثنى، وابن، بشار قالوا حدثنا عبد الرحمن، عن سفيان، عن أبي…٦٣٢٩وابن، بشار - واللفظ لابن المثنى - قالا حدثنا محمد، بن جعفر حدثنا…٦٣٣٠حدثنا أبو كريب، محمد بن العلاء حدثنا يحيى بن آدم، حدثنا قطبة، هو ابن…٦٣٣١قال كنت جالسا مع حذيفة وأبي موسى وساق الحديث وحديث قطبة أتم وأكثر .٦٣٣٢حدثنا إسحاق بن إبراهيم الحنظلي، أخبرنا عبدة بن سليمان، حدثنا الأعمش،…٦٣٣٣حدثنا أبو كريب، حدثنا يحيى بن آدم، حدثنا قطبة، عن الأعمش، عن مسلم، عن…٦٣٣٤خذوا القرآن من أربعة من ابن أم عبد - فبدأ به - ومعاذ بن جبل وأبى بن…٦٣٣٥اقرءوا القرآن من أربعة نفر من ابن أم عبد - فبدأ به - ومن أبى بن كعب…٦٣٣٦قدم معاذا قبل أبى . وفي رواية أبي كريب أبى قبل معاذ .٦٣٣٧وابن، بشار قالا حدثنا ابن أبي عدي، ح وحدثني بشر بن، خالد أخبرنا محمد،…٦٣٣٨استقرئوا القرآن من أربعة من ابن مسعود وسالم مولى أبي حذيفة وأبى بن…٦٣٣٩بهذا الإسناد وزاد قال شعبة بدأ بهذين لا أدري بأيهما بدأ .٦٣٤٠يقول جمع القرآن على عهد رسول الله صلى الله عليه وسلم أربعة كلهم من…٦٣٤١قال قلت لأنس بن مالك من جمع القرآن على عهد رسول الله صلى الله عليه…٦٣٤٢إن الله عز وجل أمرني أن أقرأ عليك٦٣٤٣إن الله أمرني أن أقرأ عليك { لم يكن الذين كفروا }٦٣٤٤يقول قال رسول الله صلى الله عليه وسلم لأبى بمثله .٦٣٤٥اهتز لها عرش الرحمن٦٣٤٦اهتز عرش الرحمن لموت سعد بن معاذ٦٣٤٧اهتز لها عرش الرحمن٦٣٤٨أتعجبون من لين هذه لمناديل سعد بن معاذ في الجنة خير منها وألين٦٣٤٩حدثنا أحمد بن عبدة الضبي، حدثنا أبو داود، حدثنا شعبة، أنبأني أبو…٦٣٥٠بهذا الحديث بالإسنادين جميعا كرواية أبي داود .٦٣٥١والذي نفس محمد بيده إن مناديل سعد بن معاذ في الجنة أحسن من هذا٦٣٥٢أن أكيدر، دومة الجندل أهدى لرسول الله صلى الله عليه وسلم . حلة فذكر…٦٣٥٣من يأخذ مني هذا٦٣٥٤وعمرو الناقد، كلاهما عن سفيان، قال عبيد الله حدثنا سفيان بن عيينة،…٦٣٥٥تبكيه أو لا تبكيه ما زالت الملائكة تظله بأجنحتها حتى رفعتموه٦٣٥٦. بهذا الحديث غير أن ابن جريج، ليس في حديثه ذكر الملائكة وبكاء…٦٣٥٧قال جيء بأبي يوم أحد مجدعا فوضع بين يدى النبي صلى الله عليه وسلم .…٦٣٥٨هل تفقدون من أحد٦٣٥٩حدثنا هداب بن خالد الأزدي، حدثنا سليمان بن المغيرة، أخبرنا حميد بن…٦٣٦٠بهذا الإسناد وزاد بعد قوله قلت فاكفني حتى أذهب فأنظر . قال نعم وكن…٦٣٦١وعليك السلام من أنت٦٣٦٢ارجع إلى قومك فأخبرهم حتى يأتيك أمري٦٣٦٣حازم يقول قال جرير بن عبد الله ما حجبني رسول الله صلى الله عليه وسلم…٦٣٦٤اللهم ثبته واجعله هاديا مهديا٦٣٦٥هل أنت مريحي من ذي الخلصة والكعبة اليمانية والشامية٦٣٦٦يا جرير ألا تريحني من ذي الخلصة٦٣٦٧بهذا الإسناد وقال في حديث مروان فجاء بشير جرير أبو أرطاة حصين بن…٦٣٦٨أنوسلم أتى الخلاء فوضعت له وضوءا فلما خرج قال " من وضع هذا " . في…٦٣٦٩أرى عبد الله رجلا صالحا٦٣٧٠نعم الرجل عبد الله لو كان يصلي من الليل٦٣٧١حدثنا عبد الله بن عبد الرحمن الدارمي، أخبرنا موسى بن خالد، ختن…٦٣٧٢اللهم أكثر ماله وولده وبارك له فيما أعطيته٦٣٧٣يقول قالت أم سليم يا رسول الله خادمك أنس . فذكر نحوه .٦٣٧٤يقول مثل ذلك .٦٣٧٥اللهم أكثر ماله وولده وبارك له فيه٦٣٧٦اللهم أكثر ماله وولده٦٣٧٧حدثنا قتيبة بن سعيد، حدثنا جعفر، - يعني ابن سليمان - عن الجعد أبي…٦٣٧٨حدثنا أبو بكر بن نافع، حدثنا بهز، حدثنا حماد، أخبرنا ثابت، عن أنس،…٦٣٧٩قال أسر إلى نبي الله صلى الله عليه وسلم سرا فما أخبرت به أحدا بعد .…٦٣٨٠حدثني زهير بن حرب، حدثنا إسحاق بن عيسى، حدثني مالك، عن أبي النضر، عن…٦٣٨١تلك الروضة الإسلام وذلك العمود عمود الإسلام وتلك العروة عروة الوثقى…٦٣٨٢يموت عبد الله وهو آخذ بالعروة الوثقى٦٣٨٣أما الطرق التي رأيت عن يسارك فهي طرق أصحاب الشمال - قال - وأما الطرق…٦٣٨٤أجب عني اللهم أيده بروح القدس٦٣٨٥ومحمد بن رافع، وعبد بن حميد، عن عبد الرزاق، أخبرنا معمر، عن الزهري،…٦٣٨٦يا حسان أجب عن رسول الله صلى الله عليه وسلم اللهم أيده بروح القدس٦٣٨٧اهجهم أو هاجهم وجبريل معك٦٣٨٨وعبد الرحمن، كلهم عن شعبة، بهذا الإسناد مثله .٦٣٨٩أن حسان بن ثابت، كان ممن كثر على عائشة فسببته فقالت يا ابن أختي دعه…٦٣٩٠بهذا الإسناد .٦٣٩١حدثني بشر بن خالد، أخبرنا محمد، - يعني ابن جعفر - عن شعبة، عن سليمان،…٦٣٩٢حدثناه ابن المثنى، حدثنا ابن أبي عدي، عن شعبة، في هذا الإسناد وقال…٦٣٩٣كيف بقرابتي منه٦٣٩٤بهذا الإسناد قالت استأذن حسان بن ثابت النبي صلى الله عليه وسلم في…٦٣٩٥اهجوا قريشا فإنه أشد عليها من رشق بالنبل٦٣٩٦اللهم اهد أم أبي هريرة٦٣٩٧من يبسط ثوبه فلن ينسى شيئا سمعه مني٦٣٩٨هريرة بهذا الحديث غير أن مالكا، انتهى حديثه عند انقضاء قول أبي هريرة…٦٣٩٩شهاب أن عروة بن الزبير، حدثه أن عائشة قالت ألا يعجبك أبو هريرة جاء…٦٤٠٠أيكم يبسط ثوبه فيأخذ من حديثي هذا ثم يجمعه إلى صدره فإنه لم ينس شيئا…٦٤٠١ائتوا روضة خاخ فإن بها ظعينة معها كتاب فخذوه منها٦٤٠٢انطلقوا حتى تأتوا روضة خاخ فإن بها امرأة من المشركين معها كتاب من…٦٤٠٣كذبت لا يدخلها فإنه شهد بدرا والحديبية٦٤٠٤لا يدخل النار إن شاء الله من أصحاب الشجرة أحد . الذين بايعوا تحتها٦٤٠٥حدثنا أبو عامر الأشعري، وأبو كريب جميعا عن أبي أسامة، قال أبو عامر…٦٤٠٦اللهم اغفر لعبيد أبي عامر٦٤٠٧إني لأعرف أصوات رفقة الأشعريين بالقرآن حين يدخلون بالليل وأعرف…٦٤٠٨إن الأشعريين إذا أرملوا في الغزو أو قل طعام عيالهم بالمدينة جمعوا ما…٦٤٠٩وأحمد بن جعفر المعقري، قالا حدثنا النضر، - وهو ابن محمد اليمامي -…٦٤١١ليس بأحق بي منكم وله ولأصحابه هجرة واحدة ولكم أنتم أهل السفينة هجرتان٦٤١٢يا أبا بكر لعلك أغضبتهم لئن كنت أغضبتهم لقد أغضبت ربك٦٤١٣وأحمد بن عبدة، - واللفظ لإسحاق - قالا أخبرنا سفيان، عن عمرو، عن جابر…٦٤١٦ولذراري الأنصار ولموالي الأنصار٦٤١٧اللهم أنتم من أحب الناس إلى اللهم أنتم من أحب الناس إلى٦٤١٨والذي نفسي بيده إنكم لأحب الناس إلى٦٤١٩بهذا الإسناد .٦٤٢٠إن الأنصار كرشي وعيبتي وإن الناس سيكثرون ويقلون فاقبلوا من محسنهم…٦٤٢١خير دور الأنصار بنو النجار ثم بنو عبد الأشهل ثم بنو الحارث بن الخزرج…٦٤٢٢حدثناه محمد بن المثنى، حدثنا أبو داود، حدثنا شعبة، عن قتادة، سمعت…٦٤٢٣وابن، رمح عن الليث بن سعد، ح وحدثنا قتيبة، حدثنا عبد العزيز، يعني ابن…٦٤٢٤خير دور الأنصار دار بني النجار ودار بني عبد الأشهل ودار بني الحارث بن…٦٤٢٥خير دور الأنصار بنو النجار ثم بنو عبد الأشهل ثم بنو الحارث بن الخزرج…٦٤٢٦خير الأنصار أو خير دور الأنصار٦٤٢٧أحدثكم بخير دور الأنصار٦٤٢٨ومحمد بن المثنى، وابن، بشار جميعا عن ابن، عرعرة - واللفظ للجهضمي -…٦٤٢٩غفار غفر الله لها وأسلم سالمها الله٦٤٣٠ائت قومك فقل إن رسول الله صلى الله عليه وسلم قال أسلم سالمها الله…٦٤٣١وابن، بشار قالا حدثنا أبو داود، حدثنا شعبة، في هذا الإسناد .٦٤٣٢أسلم سالمها الله وغفار غفر الله لها٦٤٣٣أسلم سالمها الله وغفار غفر الله لها أما إني لم أقلها ولكن قالها الله…٦٤٣٤اللهم العن بني لحيان ورعلا وذكوان وعصية عصوا الله ورسوله غفار غفر…٦٤٣٥غفار غفر الله لها وأسلم سالمها الله وعصية عصت الله ورسوله٦٤٣٦والحلواني، وعبد بن حميد، عن يعقوب بن إبراهيم بن سعد، حدثنا أبي، عن…٦٤٣٧وحدثنيه حجاج بن الشاعر، حدثنا أبو داود الطيالسي، حدثنا حرب بن شداد،…٦٤٣٨الأنصار ومزينة وجهينة وغفار وأشجع ومن كان من بني عبد الله موالي دون…٦٤٣٩قريش والأنصار ومزينة وجهينة وأسلم وغفار وأشجع موالي ليس لهم مولى دون…٦٤٤٠بهذا الإسناد . مثله غير أن في الحديث قال سعد في بعض هذا فيما أعلم .٦٤٤١أسلم وغفار ومزينة ومن كان من جهينة أو جهينة خير من بني تميم وبني عامر…٦٤٤٢والذي نفس محمد بيده لغفار وأسلم ومزينة ومن كان من جهينة أو قال جهينة…٦٤٤٣لأسلم وغفار وشىء من مزينة وجهينة أو شىء من جهينة ومزينة خير عند الله…٦٤٤٤أرأيت إن كان أسلم وغفار ومزينة - وأحسب جهينة - خيرا من بني تميم وبني…٦٤٤٥تميم محمد بن عبد الله بن أبي يعقوب الضبي بهذا الإسناد . مثله قال "…٦٤٤٦أسلم وغفار ومزينة وجهينة خير من بني تميم ومن بني عامر والحليفين بني…٦٤٤٧وهارون بن عبد الله، قالا حدثنا عبد الصمد، ح وحدثنيه عمرو الناقد،…٦٤٤٨أرأيتم إن كان جهينة وأسلم وغفار خيرا من بني تميم وبني عبد الله بن…٦٤٤٩قال أتيت عمر بن الخطاب فقال لي إن أول صدقة بيضت وجه رسول الله صلى…٦٤٥٠اللهم اهد دوسا وائت بهم٦٤٥١هم أشد أمتي على الدجال٦٤٥٢رسول الله صلى الله عليه وسلم يقولها فيهم . فذكر مثله .٦٤٥٣هم أشد الناس قتالا في الملاحم٦٤٥٤تجدون الناس معادن فخيارهم في الجاهلية خيارهم في الإسلام إذا فقهوا…٦٤٥٥تجدون الناس معادن٦٤٥٦خير نساء ركبن الإبل - قال أحدهما صالح نساء قريش . وقال الآخر نساء…٦٤٥٧أرعاه على ولد في صغره٦٤٥٨نساء قريش خير نساء ركبن الإبل أحناه على طفل وأرعاه على زوج في ذات يده٦٤٥٩وعبد بن حميد، قال عبد أخبرنا وقال ابن رافع، حدثنا عبد الرزاق، أخبرنا…٦٤٦٠خير نساء ركبن الإبل صالح نساء قريش أحناه على ولد في صغره وأرعاه على…٦٤٦١حدثني أحمد بن عثمان بن حكيم الأودي، حدثنا خالد، - يعني ابن مخلد -…٦٤٦٢أن رسول الله صلى الله عليه وسلم آخى بين أبي عبيدة بن الجراح وبين أبي…٦٤٦٣لا حلف في الإسلام٦٤٦٤ومحمد بن عبد الله بن نمير، قالا حدثنا عبدة بن، سليمان عن عاصم، عن…٦٤٦٥لا حلف في الإسلام وأيما حلف كان في الجاهلية لم يزده الإسلام إلا شدة٦٤٦٦وإسحاق بن إبراهيم، وعبد الله بن عمر بن أبان، كلهم عن حسين، - قال أبو…٦٤٦٧يأتي على الناس زمان يغزو فئام من الناس فيقال لهم فيكم من رأى رسول…٦٤٦٨يأتي على الناس زمان يبعث منهم البعث فيقولون انظروا هل تجدون فيكم أحدا…٦٤٦٩خير أمتي القرن الذين يلوني ثم الذين يلونهم ثم الذين يلونهم ثم يجيء…٦٤٧٠قرني ثم الذين يلونهم ثم الذين يلونهم ثم يجيء قوم تبدر شهادة أحدهم…٦٤٧١وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، ح وحدثنا محمد بن…٦٤٧٢خير الناس قرني ثم الذين يلونهم ثم الذين يلونهم٦٤٧٣خير أمتي القرن الذين بعثت فيهم ثم الذين يلونهم٦٤٧٤بهذا الإسناد . مثله غير أن في حديث شعبة قال أبو هريرة فلا أدري مرتين…٦٤٧٥إن خيركم قرني ثم
