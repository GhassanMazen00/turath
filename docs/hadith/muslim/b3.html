كتاب الحيض

١٥٧ حديثًا•صفحة ١ من ٢
٦٧٩وزهير بن حرب، وإسحاق بن إبراهيم، قال إسحاق أخبرنا وقال الآخران، حدثنا…٦٨٠وحدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن الشيباني، ح وحدثني…٦٨١قالت كان رسول الله صلى الله عليه وسلم يباشر نساءه فوق الإزار وهن حيض .٦٨٢وأحمد بن عيسى، قالا حدثنا ابن وهب، أخبرني مخرمة، عن أبيه، عن كريب،…٦٨٣أن زينب بنت أم سلمة، حدثته أن أم سلمة حدثتها قالت، بينما أنا مضطجعة،…٦٨٤قالت كان النبي صلى الله عليه وسلم إذا اعتكف يدني إلى رأسه فأرجله وكان…٦٨٥وعمرة بنت عبد الرحمن، أن عائشة، زوج النبي صلى الله عليه وسلم قالت إن…٦٨٦وحدثني هارون بن سعيد الأيلي، حدثنا ابن وهب، أخبرني عمرو بن الحارث، عن…٦٨٧وحدثنا يحيى بن يحيى، أخبرنا أبو خيثمة، عن هشام، أخبرنا عروة، عن…٦٨٨حدثنا أبو بكر بن أبي شيبة، حدثنا حسين بن علي، عن زائدة، عن منصور، عن…٦٨٩ناوليني الخمرة من المسجد٦٩٠تناوليها فإن الحيضة ليست في يدك٦٩١يا عائشة ناوليني الثوب٦٩٢وزهير بن حرب، قالا حدثنا وكيع، عن مسعر، وسفيان، عن المقدام بن شريح،…٦٩٣حدثنا يحيى بن يحيى، أخبرنا داود بن عبد الرحمن المكي، عن منصور، عن…٦٩٤اصنعوا كل شىء إلا النكاح٦٩٥يغسل ذكره ويتوضأ٦٩٦أنه قال استحييت أن أسأل، النبي صلى الله عليه وسلم عن المذى من أجل…٦٩٧توضأ وانضح فرجك٦٩٨أن النبي صلى الله عليه وسلم قام من الليل فقضى حاجته ثم غسل وجهه ويديه…٦٩٩ومحمد بن رمح، قالا أخبرنا الليث، ح وحدثنا قتيبة بن سعيد، حدثنا ليث،…٧٠٠ووكيع، وغندر، عن شعبة، عن الحكم، عن إبراهيم، عن الأسود، عن عائشة،…٧٠١وابن، بشار قالا حدثنا محمد بن جعفر، ح وحدثنا عبيد الله بن معاذ، قال…٧٠٢وزهير بن حرب، قالا حدثنا يحيى، - وهو ابن سعيد - عن عبيد الله، ح…٧٠٣نعم ليتوضأ ثم لينم حتى يغتسل إذا شاء٧٠٤توضأ واغسل ذكرك ثم نم٧٠٥رسول الله صلى الله عليه وسلم . فذكر الحديث قلت كيف كان يصنع في…٧٠٦بهذا الإسناد مثله .٧٠٧إذا أتى أحدكم أهله ثم أراد أن يعود فليتوضأ٧٠٨أن النبي صلى الله عليه وسلم كان يطوف على نسائه بغسل واحد .٧٠٩بل أنت فتربت يمينك نعم فلتغتسل يا أم سليم إذا رأت ذاك٧١٠إذا رأت ذلك المرأة فلتغتسل٧١١إذا كان منها ما يكون من الرجل فلتغتسل٧١٢نعم إذا رأت الماء٧١٣وزهير بن حرب، قالا حدثنا وكيع، ح وحدثنا ابن أبي عمر، حدثنا سفيان،…٧١٤أن عائشة، زوج النبي صلى الله عليه وسلم أخبرته أن أم سليم أم بني أبي…٧١٥وسهل بن عثمان، وأبو كريب - واللفظ لأبي كريب - قال سهل حدثنا وقال…٧١٦إن اسمي محمد الذي سماني به أهلي٧١٧في هذا الإسناد بمثله غير أنه قال كنت قاعدا عند رسول الله صلى الله…٧١٨قالت كان رسول الله صلى الله عليه وسلم إذا اغتسل من الجنابة يبدأ فيغسل…٧١٩وزهير بن حرب، قالا حدثنا جرير، ح وحدثنا علي بن حجر، حدثنا علي بن…٧٢٠أن النبي صلى الله عليه وسلم اغتسل من الجنابة فبدأ فغسل كفيه ثلاثا ثم…٧٢١أن رسول الله صلى الله عليه وسلم كان إذا اغتسل من الجنابة بدأ فغسل…٧٢٢قال حدثتني خالتي، ميمونة قالت أدنيت لرسول الله صلى الله عليه وسلم…٧٢٣بهذا الإسناد . وليس في حديثهما إفراغ ثلاث حفنات على الرأس وفي حديث…٧٢٤أن النبي صلى الله عليه وسلم أتي بمنديل فلم يمسه وجعل يقول بالماء هكذا…٧٢٥قالت كان رسول الله صلى الله عليه وسلم إذا اغتسل من الجنابة دعا بشىء…٧٢٦أن رسول الله صلى الله عليه وسلم كان يغتسل من إناء هو الفرق من الجنابة…٧٢٧حدثنا قتيبة بن سعيد، حدثنا ليث، ح وحدثنا ابن رمح، أخبرنا الليث، ح…٧٢٨وحدثني عبيد الله بن معاذ العنبري، قال حدثنا أبي قال، حدثنا شعبة، عن…٧٢٩حدثنا هارون بن سعيد الأيلي، حدثنا ابن وهب، أخبرني مخرمة بن بكير، عن…٧٣٠- وكانت تحت المنذر بن الزبير - أن عائشة، أخبرتها أنها، كانت تغتسل هي…٧٣١حدثنا عبد الله بن مسلمة بن قعنب، قال حدثنا أفلح بن حميد، عن القاسم بن…٧٣٢صلى الله عليه وسلم من إناء - بيني وبينه - واحد فيبادرني حتى أقول دع…٧٣٣قال أخبرتني ميمونة، أنها كانت تغتسل هي والنبي صلى الله عليه وسلم في…٧٣٤ومحمد بن حاتم، قال إسحاق أخبرنا وقال ابن حاتم، حدثنا محمد بن بكر،…٧٣٥أن زينب بنت أم سلمة، حدثته أن أم سلمة حدثتها قالت، كانت هي ورسول الله…٧٣٦حدثنا عبيد الله بن معاذ، حدثنا أبي ح، وحدثنا محمد بن المثنى، حدثنا…٧٣٧قال كان النبي صلى الله عليه وسلم يتوضأ بالمد ويغتسل بالصاع إلى خمسة…٧٣٨وعمرو بن علي، كلاهما عن بشر بن المفضل، - قال أبو كامل حدثنا بشر، -…٧٣٩- قال أبو بكر - صاحب رسول الله صلى الله عليه وسلم قال كان رسول الله…٧٤٠أما أنا فإني أفيض على رأسي ثلاث أكف٧٤١أما أنا فأفرغ على رأسي ثلاثا٧٤٢أما أنا فأفرغ على رأسي ثلاثا٧٤٣قال كان رسول الله صلى الله عليه وسلم إذا اغتسل من جنابة صب على رأسه…٧٤٤لا إنما يكفيك أن تحثي على رأسك ثلاث حثيات ثم تفيضين عليك الماء فتطهرين٧٤٥في هذا الإسناد وفي حديث عبد الرزاق فأنقضه للحيضة والجنابة فقال " لا "…٧٤٦بهذا الإسناد وقال أفأحله فأغسله من الجنابة . ولم يذكر الحيضة .٧٤٧قال بلغ عائشة أن عبد، الله بن عمرو يأمر النساء إذا اغتسلن أن ينقضن…٧٤٨تطهري بها . سبحان الله٧٤٩خذي فرصة ممسكة فتوضئي بها٧٥٠تأخذ إحداكن ماءها وسدرتها فتطهر فتحسن الطهور ثم تصب على رأسها فتدلكه…٧٥١سبحان الله تطهري بها٧٥٢وحدثنا يحيى بن يحيى، وأبو بكر بن أبي شيبة كلاهما عن أبي الأحوص، عن…٧٥٣لا إنما ذلك عرق وليس بالحيضة فإذا أقبلت الحيضة فدعي الصلاة وإذا أدبرت…٧٥٤حدثنا يحيى بن يحيى، أخبرنا عبد العزيز بن محمد، وأبو معاوية ح وحدثنا…٧٥٥إنما ذلك عرق فاغتسلي ثم صلي٧٥٦إن هذه ليست بالحيضة ولكن هذا عرق فاغتسلي وصلي٧٥٧قالت جاءت أم حبيبة بنت جحش إلى رسول الله صلى الله عليه وسلم وكانت…٧٥٨أن ابنة جحش، كانت تستحاض سبع سنين بنحو حديثهم .٧٥٩امكثي قدر ما كانت تحبسك حيضتك ثم اغتسلي وصلي٧٦٠امكثي قدر ما كانت تحبسك حيضتك ثم اغتسلي٧٦١أن امرأة، سألت عائشة فقالت أتقضي إحدانا الصلاة أيام محيضها فقالت…٧٦٢أنها سألت عائشة أتقضي الحائض الصلاة فقالت عائشة أحرورية أنت قد كن…٧٦٣وحدثنا عبد بن حميد، أخبرنا عبد الرزاق، أخبرنا معمر، عن عاصم، عن…٧٦٤أن أبا مرة، مولى أم هانئ بنت أبي طالب أخبره أنه، سمع أم هانئ بنت أبي…٧٦٥أن أبا مرة، مولى عقيل حدثه أن أم هانئ بنت أبي طالب حدثته أنه، لما كان…٧٦٦بهذا الإسناد وقال فسترته ابنته فاطمة بثوبه فلما اغتسل أخذه فالتحف به…٧٦٧قالت وضعت للنبي صلى الله عليه وسلم ماء وسترته فاغتسل .٧٦٨لا ينظر الرجل إلى عورة الرجل ولا المرأة إلى عورة المرأة ولا يفضي…٧٦٩ومحمد بن رافع، قالا حدثنا ابن أبي فديك، أخبرنا الضحاك بن عثمان، بهذا…٧٧٠كانت بنو إسرائيل يغتسلون عراة ينظر بعضهم إلى سوأة بعض وكان موسى -…٧٧١ومحمد بن حاتم بن ميمون، جميعا عن محمد بن بكر، قال أخبرنا ابن جريج، ح…٧٧٢وحدثنا زهير بن حرب، حدثنا روح بن عبادة، حدثنا زكرياء بن إسحاق، حدثنا…٧٧٣ارجع إلى ثوبك فخذه ولا تمشوا عراة٧٧٤وعبد الله بن محمد بن أسماء الضبعي، قالا حدثنا مهدي، - وهو ابن ميمون -…٧٧٥ويحيى بن أيوب، وقتيبة، وابن، حجر - قال يحيى بن يحيى أخبرنا وقال…٧٧٦إنما الماء من الماء٧٧٧قال كان رسول الله صلى الله عليه وسلم ينسخ حديثه بعضه بعضا كما ينسخ…٧٧٨وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن الحكم، عن ذكوان،…٧٧٩يغسل ما أصابه من المرأة ثم يتوضأ ويصلي٧٨٠يغسل ذكره ويتوضأ٧٨١يتوضأ كما يتوضأ للصلاة ويغسل ذكره٧٨٢أن عروة بن الزبير، أخبره أن أبا أيوب أخبره أنه، سمع ذلك، من رسول الله…٧٨٣إذا جلس بين شعبها الأربع ثم جهدها فقد وجب عليه الغسل٧٨٤بهذا الإسناد مثله غير أن في حديث شعبة " ثم اجتهد " ولم يقل " وإن لم…٧٨٥إذا جلس بين شعبها الأربع ومس الختان الختان فقد وجب الغسل٧٨٦إني لأفعل ذلك أنا وهذه ثم نغتسل٧٨٧الوضوء مما مست النار٧٨٨توضئوا مما مست النار٧٨٩توضئوا مما مست النار٧٩٠أن رسول الله صلى الله عليه وسلم أكل كتف شاة ثم صلى ولم يتوضأ .٧٩١أن النبي صلى الله عليه وسلم أكل عرقا - أو لحما - ثم صلى ولم يتوضأ ولم…٧٩٢أنه رأى رسول الله صلى الله عليه وسلم يحتز من كتف يأكل منها ثم صلى ولم…٧٩٣حدثني أحمد بن عيسى، حدثنا ابن وهب، أخبرني عمرو بن الحارث، عن ابن…٧٩٥زوج النبي صلى الله عليه وسلم أن النبي صلى الله عليه وسلم أكل عندها…٧٩٦زوج النبي صلى الله عليه وسلم بذلك .٧٩٧قال أشهد لكنت أشوي لرسول الله صلى الله عليه وسلم بطن الشاة ثم صلى ولم…٧٩٨حدثنا قتيبة بن سعيد، حدثنا ليث، عن عقيل، عن الزهري، عن عبيد الله بن…٧٩٩مثله .٨٠٠أن رسول الله صلى الله عليه وسلم جمع عليه ثيابه ثم خرج إلى الصلاة فأتي…٨٠١قال كنت مع ابن عباس . وساق الحديث بمعنى حديث ابن حلحلة وفيه أن ابن…٨٠٢إن شئت فتوضأ وإن شئت فلا توضأ٨٠٣وأشعث بن أبي الشعثاء، كلهم عن جعفر بن أبي ثور، عن جابر بن سمرة، عن…٨٠٤لا ينصرف حتى يسمع صوتا أو يجد ريحا٨٠٥إذا وجد أحدكم في بطنه شيئا فأشكل عليه أخرج منه شىء أم لا فلا يخرجن من…٨٠٦هلا أخذتم إهابها فدبغتموه فانتفعتم به٨٠٧هلا انتفعتم بجلدها٨٠٨وعبد بن حميد، جميعا عن يعقوب بن إبراهيم بن سعد، حدثني أبي، عن صالح،…٨٠٩ألا أخذوا إهابها فدبغوه فانتفعوا به٨١٠ألا أخذتم إهابها فاستمتعتم به٨١١ألا انتفعتم بإهابها٨١٢إذا دبغ الإهاب فقد طهر٨١٣وعمرو الناقد، قالا حدثنا ابن عيينة، ح وحدثنا قتيبة بن سعيد، حدثنا عبد…٨١٤أن أبا الخير، حدثه قال رأيت على ابن وعلة السبئي فروا فمسسته فقال ما…٨١٥وحدثني إسحاق بن منصور، وأبو بكر بن إسحاق عن عمرو بن الربيع، أخبرنا…٨١٦حدثنا يحيى بن يحيى، قال قرأت على مالك عن عبد الرحمن بن القاسم، عن…٨١٧وابن، بشر عن هشام، عن أبيه، عن عائشة، أنها استعارت من أسماء قلادة…٨١٨إنما كان يكفيك أن تقول بيديك هكذا٨١٩إنما كان يكفيك أن تقول هكذا٨٢٠إنما كان يكفيك أن تضرب بيديك الأرض ثم تنفخ ثم تمسح بهما وجهك وكفيك٨٢١أن رجلا، أتى عمر فقال إني أجنبت فلم أجد ماء . وساق الحديث وزاد فيه…٨٢٢قال مسلم وروى الليث بن سعد، عن جعفر بن ربيعة، عن عبد الرحمن بن هرمز،…٨٢٣أن رجلا، مر ورسول الله صلى الله عليه وسلم يبول فسلم فلم يرد عليه .٨٢٤أين كنت يا أبا هريرة٨٢٥إن المسلم لا ينجس٨٢٦قالت كان النبي صلى الله عليه وسلم يذكر الله على كل أحيانه .٨٢٧أريد أن أصلي فأتوضأ٨٢٨لم أأصلي فأتوضأ٨٢٩مولى آل السائب أنه سمع عبد الله بن عباس، قال ذهب رسول الله صلى الله…
