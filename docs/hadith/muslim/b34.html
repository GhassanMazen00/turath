كتاب الصيد والذبائح وما يؤكل من الحيوان

٩٢ حديثًا
٤٩٧٢إذا أرسلت كلبك المعلم وذكرت اسم الله عليه فكل٤٩٧٣إذا أرسلت كلابك المعلمة وذكرت اسم الله عليها فكل مما أمسكن عليك وإن…٤٩٧٤إذا أصاب بحده فكل وإذا أصاب بعرضه فقتل فإنه وقيذ فلا تأكل٤٩٧٥وحدثنا يحيى بن أيوب، حدثنا ابن علية، قال وأخبرني شعبة، عن عبد الله…٤٩٧٦وحدثني أبو بكر بن نافع العبدي، حدثنا غندر، حدثنا شعبة، حدثنا عبد الله…٤٩٧٧ما أصاب بحده فكله وما أصاب بعرضه فهو وقيذ٤٩٧٨زائدة بهذا الإسناد .٤٩٧٩فلا تأكل فإنما سميت على كلبك ولم تسم على غيره٤٩٨٠وحدثنا محمد بن الوليد، حدثنا محمد بن جعفر، حدثنا شعبة، عن الحكم، عن…٤٩٨١إذا أرسلت كلبك فاذكر اسم الله فإن أمسك عليك فأدركته حيا فاذبحه وإن…٤٩٨٢إذا رميت سهمك فاذكر اسم الله فإن وجدته قد قتل فكل إلا أن تجده قد وقع…٤٩٨٣أما ما ذكرت أنكم بأرض قوم من أهل الكتاب تأكلون في آنيتهم فإن وجدتم…٤٩٨٤بهذا الإسناد . نحو حديث ابن المبارك غير أن حديث ابن وهب، لم يذكر فيه…٤٩٨٥إذا رميت بسهمك فغاب عنك فأدركته فكله ما لم ينتن٤٩٨٦فكله ما لم ينتن٤٩٨٧كله بعد ثلاث إلا أن ينتن فدعه٤٩٨٨وإسحاق بن إبراهيم، وابن أبي عمر، قال إسحاق أخبرنا وقال الآخران، حدثنا…٤٩٨٩أنه سمع أبا ثعلبة الخشني، يقول نهى رسول الله صلى الله عليه وسلم عن…٤٩٩٠- يعني ابن الحارث - أن ابن شهاب، حدثه عن أبي إدريس الخولاني، عن أبي…٤٩٩١وابن أبي ذئب، وعمرو بن الحارث ويونس بن يزيد وغيرهم ح وحدثني محمد بن…٤٩٩٢كل ذي ناب من السباع فأكله حرام٤٩٩٣بهذا الإسناد مثله .٤٩٩٤وحدثنا عبيد الله بن معاذ العنبري، حدثنا أبي، حدثنا شعبة، عن الحكم، عن…٤٩٩٥بهذا الإسناد مثله .٤٩٩٦وحدثنا أحمد بن حنبل، حدثنا سليمان بن داود، حدثنا أبو عوانة، حدثنا…٤٩٩٧وحدثنا يحيى بن يحيى، أخبرنا هشيم، عن أبي بشر، ح وحدثنا أحمد بن حنبل،…٤٩٩٨هو رزق أخرجه الله لكم فهل معكم من لحمه شىء فتطعمونا٤٩٩٩قال سمع عمرو، جابر بن عبد الله يقول بعثنا رسول الله صلى الله عليه…٥٠٠٠قال سمع عمرو، جابرا يقول في جيش الخبط إن رجلا نحر ثلاث جزائر ثم ثلاثا…٥٠٠١وحدثنا عثمان بن أبي شيبة، حدثنا عبدة، - يعني ابن سليمان - عن هشام، بن…٥٠٠٢وهب بن كيسان أن جابر بن عبد الله، أخبره قال بعث رسول الله صلى الله…٥٠٠٣وحدثنا أبو كريب، حدثنا أبو أسامة، حدثنا الوليد، - يعني ابن كثير - قال…٥٠٠٤عبد الله قال بعث رسول الله صلى الله عليه وسلم بعثا إلى أرض جهينة…٥٠٠٥حدثنا يحيى بن يحيى، قال قرأت على مالك بن أنس عن ابن شهاب، عن عبد،…٥٠٠٦وابن، نمير وزهير بن حرب قالوا حدثنا سفيان، ح وحدثنا ابن نمير، حدثنا…٥٠٠٧وعبد بن حميد، كلاهما عن يعقوب بن إبراهيم، بن سعد حدثنا أبي، عن صالح،…٥٠٠٨وحدثنا محمد بن عبد الله بن نمير، حدثنا أبي، حدثنا عبيد الله، حدثني…٥٠٠٩وحدثني هارون بن عبد الله، حدثنا محمد بن بكر، أخبرنا ابن جريج، أخبرني…٥٠١٠وحدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن الشيباني، قال سألت…٥٠١١وحدثنا أبو كامل، فضيل بن حسين حدثنا عبد الواحد، - يعني ابن زياد -…٥٠١٢وعبد الله بن أبي أوفى، يقولان أصبنا حمرا فطبخناها فنادى منادي رسول…٥٠١٣وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي إسحاق، قال قال…٥٠١٤وإسحاق بن إبراهيم، قال أبو كريب حدثنا ابن بشر، عن مسعر، عن ثابت بن…٥٠١٥وحدثنا زهير بن حرب، حدثنا جرير، عن عاصم، عن الشعبي، عن البراء بن…٥٠١٦بهذا الإسناد نحوه .٥٠١٧قال لا أدري إنما نهى عنه رسول الله صلى الله عليه وسلم من أجل أنه كان…٥٠١٨ما هذه النيران على أى شىء توقدون٥٠١٩وصفوان بن عيسى، ح وحدثنا أبو بكر بن النضر، حدثنا أبو عاصم النبيل،…٥٠٢٠وحدثنا ابن أبي عمر، حدثنا سفيان، عن أيوب، عن محمد، عن أنس، قال لما…٥٠٢١حدثنا محمد بن منهال الضرير، حدثنا يزيد بن زريع، حدثنا هشام بن حسان،…٥٠٢٢وأبو الربيع العتكي، وقتيبة بن سعيد، - واللفظ ليحيى - قال يحيى أخبرنا…٥٠٢٣وحدثني محمد بن حاتم، حدثنا محمد بن بكر، أخبرنا ابن جريج، أخبرني أبو…٥٠٢٤وأحمد بن، عثمان النوفلي قالا حدثنا أبو عاصم، كلاهما عن ابن جريج، بهذا…٥٠٢٥ووكيع، عن هشام، عن فاطمة، عن أسماء، قالت نحرنا فرسا على عهد رسول الله…٥٠٢٦بهذا الإسناد .٥٠٢٧لست بآكله ولا محرمه٥٠٢٨لا آكله ولا أحرمه٥٠٢٩لا آكله ولا أحرمه٥٠٣٠بمثله في هذا الإسناد .٥٠٣١وقتيبة، قالا حدثنا حماد، ح وحدثني زهير بن حرب، حدثنا إسماعيل، كلاهما…٥٠٣٢كلوا فإنه حلال ولكنه ليس من طعامي٥٠٣٣وحدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، عن توبة…٥٠٣٤لا ولكنه لم يكن بأرض قومي فأجدني أعافه٥٠٣٥لا ولكنه لم يكن بأرض قومي فأجدني أعافه٥٠٣٦وعبد بن حميد، قال عبد أخبرني وقال أبو بكر، حدثنا يعقوب بن إبراهيم بن…٥٠٣٧وحدثنا عبد بن حميد، أخبرنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن…٥٠٣٨أن أبا أمامة بن سهل، أخبره عن ابن، عباس قال أتي رسول الله صلى الله…٥٠٣٩وحدثنا محمد بن بشار، وأبو بكر بن نافع قال ابن نافع أخبرنا غندر، حدثنا…٥٠٤٠لا آكله ولا أنهى عنه ولا أحرمه٥٠٤١لا أدري لعله من القرون التي مسخت٥٠٤٢فقال لا تطعموه . وقذره وقال قال عمر بن الخطاب إن النبي صلى الله عليه…٥٠٤٣ذكر لي أن أمة من بني إسرائيل مسخت٥٠٤٤يا أعرابي إن الله لعن أو غضب على سبط من بني إسرائيل فمسخهم دواب يدبون…٥٠٤٥حدثنا أبو كامل الجحدري، حدثنا أبو عوانة، عن أبي يعفور، عن عبد الله…٥٠٤٦وإسحاق بن إبراهيم، وابن أبي عمر، جميعا عن ابن عيينة، عن أبي يعفور،…٥٠٤٧بهذا الإسناد وقال سبع غزوات .٥٠٤٨حدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، عن هشام بن زيد،…٥٠٤٩بهذا الإسناد وفي حديث يحيى بوركها أو فخذيها .٥٠٥٠حدثنا عبيد الله بن معاذ العنبري، حدثنا أبي، حدثنا كهمس، عن ابن بريدة،…٥٠٥١بهذا الإسناد نحوه .٥٠٥٢وعبد الرحمن بن مهدي، قالا حدثنا شعبة، عن قتادة، عن عقبة بن صهبان، عن…٥٠٥٣إنها لا تصيد صيدا ولا تنكأ عدوا ولكنها تكسر السن وتفقأ العين٥٠٥٤بهذا الإسناد نحوه .٥٠٥٥إن الله كتب الإحسان على كل شىء فإذا قتلتم فأحسنوا القتلة وإذا ذبحتم…٥٠٥٦بإسناد حديث ابن علية ومعنى حديثه .٥٠٥٧بن زيد بن أنس بن مالك قال دخلت مع جدي أنس بن مالك دار الحكم بن أيوب…٥٠٥٨وعبد الرحمن بن مهدي، ح وحدثني يحيى بن حبيب، حدثنا خالد بن الحارث، ح…٥٠٥٩لا تتخذوا شيئا فيه الروح غرضا٥٠٦٠وعبد الرحمن بن مهدي، عن شعبة، بهذا الإسناد مثله .٥٠٦١وحدثنا شيبان بن فروخ، وأبو كامل - واللفظ لأبي كامل - قالا حدثنا أبو…٥٠٦٢وحدثني زهير بن حرب، حدثنا هشيم، أخبرنا أبو بشر، عن سعيد بن جبير، قال…٥٠٦٣أنه سمع جابر بن عبد الله، يقول نهى رسول الله صلى الله عليه وسلم أن…
