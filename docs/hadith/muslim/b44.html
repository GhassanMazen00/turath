خليلا لاتخذت أبا بكر خليلا ولكنه أخي وصاحبي وقد اتخذ…٦١٧٣لو كنت متخذا من أمتي أحدا خليلا لاتخذت أبا بكر٦١٧٤لو كنت متخذا خليلا لاتخذت ابن أبي قحافة خليلا٦١٧٥لو كنت متخذا من أهل الأرض خليلا لاتخذت ابن أبي قحافة خليلا ولكن…٦١٧٦ألا إني أبرأ إلى كل خل من خله ولو كنت متخذا خليلا لاتخذت أبا بكر…٦١٧٧أن رسول الله صلى الله عليه وسلم بعثه على جيش ذات السلاسل فأتيته فقلت…٦١٧٨وسئلت، من كان رسول الله صلى الله عليه وسلم مستخلفا لو استخلفه قالت…٦١٨١ادعي لي أبا بكر وأخاك حتى أكتب كتابا فإني أخاف أن يتمنى متمن ويقول…٦١٨٢من أصبح منكم اليوم صائما٦١٨٣بينما رجل يسوق بقرة له قد حمل عليها التفتت إليه البقرة فقالت إني لم…٦١٨٤بهذا الإسناد . قصة الشاة والذئب ولم يذكر قصة البقرة .٦١٨٥فإني أومن به أنا وأبو بكر وعمر٦١٨٦وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، ح وحدثنا محمد بن…٦١٨٧جئت أنا وأبو بكر وعمر ودخلت أنا وأبو بكر وعمر وخرجت أنا وأبو بكر وعمر٦١٨٨في هذا الإسناد بمثله .٦١٨٩بينا أنا نائم رأيت الناس يعرضون وعليهم قمص منها ما يبلغ الثدي ومنها…٦١٩٠بينا أنا نائم إذ رأيت قدحا أتيت به فيه لبن فشربت منه حتى إني لأرى…٦١٩١وعبد بن، حميد كلاهما عن يعقوب بن إبراهيم بن سعد، حدثنا أبي، عن صالح،…٦١٩٢بينا أنا نائم رأيتني على قليب عليها دلو فنزعت منها ما شاء الله ثم…٦١٩٣والحلواني، وعبد بن حميد، عن يعقوب بن إبراهيم بن سعد، حدثنا أبي، عن…٦١٩٤رأيت ابن أبي قحافة ينزع٦١٩٥بينا أنا نائم أريت أني أنزع على حوضي أسقي الناس فجاءني أبو بكر فأخذ…٦١٩٦أريت كأني أنزع بدلو بكرة على قليب فجاء أبو بكر فنزع ذنوبا أو ذنوبين…٦١٩٧رسول الله صلى الله عليه وسلم في أبي بكر وعمر بن الخطاب رضى الله عنهما…٦١٩٨دخلت الجنة فرأيت فيها دارا أو قصرا فقلت لمن هذا فقالوا لعمر بن الخطاب…٦١٩٩وابن المنكدر، عن جابر، ح وحدثنا أبو بكر بن أبي شيبة، حدثنا سفيان، عن…٦٢٠٢عجبت من هؤلاء اللاتي كن عندي فلما سمعن صوتك ابتدرن الحجاب٦٢٠٣أن عمر بن الخطاب، جاء إلى رسول الله صلى الله عليه وسلم وعنده نسوة قد…٦٢٠٤قد كان يكون في الأمم قبلكم محدثون فإن يكن في أمتي منهم أحد فإن عمر بن…٦٢٠٥وزهير بن حرب، قالا حدثنا ابن عيينة، كلاهما عن ابن عجلان، عن سعد بن…٦٢٠٦قال قال عمر وافقت ربي في ثلاث في مقام إبراهيم وفي الحجاب وفي أسارى…٦٢٠٧إنما خيرني الله فقال { استغفر لهم أو لا تستغفر لهم إن تستغفر لهم…٦٢٠٨وعبيد الله بن سعيد، قالا حدثنا يحيى، - وهو القطان - عن عبيد الله،…٦٢٠٩ألا أستحي من رجل تستحي منه الملائكة٦٢١٠اجمعي عليك ثيابك٦٢١١والحسن بن علي الحلواني، وعبد بن حميد، كلهم عن يعقوب، بن إبراهيم بن…٦٢١٢افتح وبشره بالجنة٦٢١٣أن رسول الله صلى الله عليه وسلم دخل حائطا وأمرني أن أحفظ الباب .…٦٢١٤ائذن له وبشره بالجنة٦٢١٥حدثنيه أبو بكر بن إسحاق، حدثنا سعيد بن عفير، حدثني سليمان بن بلال،…٦٢١٦حدثنا حسن بن علي الحلواني، وأبو بكر بن إسحاق قالا حدثنا سعيد بن أبي،…٦٢١٧أنت مني بمنزلة هارون من موسى إلا أنه لا نبي بعدي٦٢١٨أما ترضى أن تكون مني بمنزلة هارون من موسى غير أنه لا نبي بعدي٦٢١٩في هذا الإسناد .٦٢٢٠أما ترضى أن تكون مني بمنزلة هارون من موسى إلا أنه لا نبوة بعدي٦٢٢١أما ترضى أن تكون مني بمنزلة هارون من موسى٦٢٢٢لأعطين هذه الراية رجلا يحب الله ورسوله يفتح الله على يديه٦٢٢٣لأعطين هذه الراية رجلا يفتح الله على يديه يحب الله ورسوله ويحبه الله…٦٢٢٤لأعطين الراية - أو ليأخذن بالراية - غدا رجل يحبه الله ورسوله أو قال…٦٢٢٥أما بعد ألا أيها الناس فإنما أنا بشر يوشك أن يأتي رسول ربي فأجيب وأنا…٦٢٢٦وحدثنا محمد بن بكار بن الريان، حدثنا حسان، - يعني ابن إبراهيم - عن…٦٢٢٧كتاب الله فيه الهدى والنور من استمسك به وأخذ به كان على الهدى ومن…٦٢٢٨ألا وإني تارك فيكم ثقلين أحدهما كتاب الله عز وجل هو حبل الله من اتبعه…٦٢٢٩حدثنا قتيبة بن سعيد، حدثنا عبد العزيز، - يعني ابن أبي حازم - عن أبي،…٦٢٣٠حدثنا عبد الله بن مسلمة بن قعنب، حدثنا سليمان بن بلال، عن يحيى بن…٦٢٣١ليت رجلا صالحا من أصحابي يحرسني الليلة٦٢٣٢يقول قالت عائشة أرق رسول الله صلى الله عليه وسلم ذات ليلة . بمثل حديث…٦٢٣٣ارم فداك أبي وأمي٦٢٣٤وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، ح وحدثنا أبو بكر بن…٦٢٣٥قال لقد جمع لي رسول الله صلى الله عليه وسلم أبويه يوم أحد .٦٢٣٦وابن، رمح عن الليث بن سعد، ح وحدثنا ابن المثنى، حدثنا عبد الوهاب،…٦٢٣٧ارم فداك أبي وأمي٦٢٣٨رده من حيث أخذته٦٢٣٩ومحمد بن بشار، قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن سماك بن حرب،…٦٢٤٠حدثنا زهير بن حرب، حدثنا عبد الرحمن، عن سفيان، عن المقدام بن شريح، عن…٦٢٤١حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن عبد الله الأسدي، عن سعيد، كلاهما عن ابن مسهر، قال إسماعيل أخبرنا علي بن مسهر، عن…٦٢٤٦وحدثنا أبو كريب، حدثنا أبو أسامة، عن هشام، عن أبيه، عن عبد الله بن…٦٢٤٧اهدأ فما عليك إلا نبي أو صديق أو شهيد٦٢٤٨اسكن حراء فما عليك إلا نبي أو صديق أو شهيد٦٢٤٩وعبدة، قالا حدثنا هشام، عن أبيه، قال قالت لي عائشة أبواك والله من…٦٢٥٠بهذا الإسناد وزاد تعني أبا بكر والزبير .٦٢٥١قال قالت لي عائشة كان أبواك من الذين استجابوا لله والرسول من بعد ما…٦٢٥٢إن لكل أمة أمينا وإن أميننا أيتها الأمة أبو عبيدة بن الجراح٦٢٥٣هذا أمين هذه الأمة٦٢٥٤لأبعثن إليكم رجلا أمينا حق أمين حق أمين٦٢٥٥بهذا الإسناد نحوه .٦٢٥٦اللهم إني أحبه فأحبه وأحبب من
