٥٦ حديثًا
٤٣٤٢حدثنا قتيبة بن سعيد، حدثنا ليث، عن يحيى، - وهو ابن سعيد - عن بشير، بن…٤٣٤٣كبر الكبر - أو قال - ليبدإ الأكبر٤٣٤٤وحدثنا القواريري، حدثنا بشر بن المفضل، حدثنا يحيى بن سعيد، عن بشير…٤٣٤٥بن أبي حثمة بنحو حديثهم .٤٣٤٦تحلفون خمسين يمينا وتستحقون قاتلكم٤٣٤٧أن رجلا، من الأنصار من بني حارثة يقال له عبد الله بن سهل بن زيد انطلق…٤٣٤٨أنه أخبره أن نفرا منهم انطلقوا إلى خيبر فتفرقوا فيها فوجدوا أحدهم…٤٣٤٩من كبراء قومه أن عبد، الله بن سهل ومحيصة خرجا إلى خيبر من جهد أصابهم…٤٣٥٠وحرملة بن يحيى، قال أبو الطاهر حدثنا وقال، حرملة أخبرنا ابن وهب،…٤٣٥١شهاب بهذا الإسناد مثله . وزاد وقضى بها رسول الله صلى الله عليه وسلم…٤٣٥٢أن أبا سلمة بن عبد الرحمن، وسليمان بن يسار، أخبراه عن ناس، من الأنصار…٤٣٥٣إن شئتم أن تخرجوا إلى إبل الصدقة فتشربوا من ألبانها وأبوالها٤٣٥٤ألا تخرجون مع راعينا في إبله فتصيبون من أبوالها وألبانها٤٣٥٥قال قدم على رسول الله صلى الله عليه وسلم قوم من عكل أو عرينة فاجتووا…٤٣٥٦وحدثنا محمد بن المثنى، حدثنا معاذ بن معاذ، ح وحدثنا أحمد بن عثمان…٤٣٥٧قال قدم على رسول الله صلى الله عليه وسلم ثمانية نفر من عكل . بنحو…٤٣٥٨قال أتى رسول الله صلى الله عليه وسلم نفر من عرينة فأسلموا وبايعوه وقد…٤٣٥٩وفي حديث همام قدم على النبي صلى الله عليه وسلم رهط من عرينة وفي حديث…٤٣٦٠قال إنما سمل النبي صلى الله عليه وسلم أعين أولئك لأنهم سملوا أعين…٤٣٦١ومحمد بن بشار، - واللفظ لابن المثنى - قالا حدثنا محمد بن جعفر، حدثنا…٤٣٦٢بهذا الإسناد . نحوه وفي حديث ابن إدريس فرضخ رأسه بين حجرين .٤٣٦٣أن رجلا، من اليهود قتل جارية من الأنصار على حلي لها ثم ألقاها في…٤٣٦٤بهذا الإسناد مثله .٤٣٦٥أن جارية، وجد رأسها قد رض بين حجرين فسألوها من صنع هذا بك فلان فلان…٤٣٦٦أيعض أحدكم كما يعض الفحل لا دية له٤٣٦٧وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن قتادة، عن عطاء، عن…٤٣٦٨أردت أن تأكل لحمه٤٣٦٩أردت أن تقضمها كما يقضم الفحل٤٣٧٠ما تأمرني تأمرني أن آمره أن يدع يده في فيك تقضمها كما يقضم الفحل ادفع…٤٣٧١أردت أن تقضمه كما يقضم الفحل٤٣٧٢حدثنا أبو بكر بن أبي شيبة، حدثنا أبو أسامة، أخبرنا ابن جريج، أخبرني…٤٣٧٣بهذا الإسناد نحوه .٤٣٧٤أن أخت الربيع أم حارثة، جرحت إنسانا فاختصموا إلى النبي صلى الله عليه…٤٣٧٥لا يحل دم امرئ مسلم يشهد أن لا إله إلا الله وأني رسول الله إلا بإحدى…٤٣٧٦وعلي بن خشرم، قالا أخبرنا عيسى بن يونس، كلهم عن الأعمش، بهذا الإسناد…٤٣٧٧والذي لا إله غيره لا يحل دم رجل مسلم يشهد أن لا إله إلا الله وأني…٤٣٧٨والذي لا إله غيره٤٣٧٩لا تقتل نفس ظلما إلا كان على ابن آدم الأول كفل من دمها لأنه كان أول…٤٣٨٠وعيسى بن يونس، ح وحدثنا ابن أبي عمر، حدثنا سفيان، كلهم عن الأعمش،…٤٣٨١أول ما يقضى بين الناس يوم القيامة في الدماء٤٣٨٢حدثنا عبيد الله بن معاذ، حدثنا أبي ح، وحدثني يحيى بن حبيب، حدثنا…٤٣٨٣إن الزمان قد استدار كهيئته يوم خلق الله السموات والأرض السنة اثنا عشر…٤٣٨٤أتدرون أى يوم هذا٤٣٨٥حدثنا محمد بن المثنى، حدثنا حماد بن مسعدة، عن ابن عون، قال قال محمد…٤٣٨٦حدثني محمد بن حاتم بن ميمون، حدثنا يحيى بن سعيد، حدثنا قرة بن خالد،…٤٣٨٧حرب أن علقمة بن وائل، حدثه أن أباه حدثه قال إني لقاعد مع النبي صلى…٤٣٨٨القاتل والمقتول في النار٤٣٨٩أن امرأتين، من هذيل رمت إحداهما الأخرى فطرحت جنينها فقضى فيه النبي…٤٣٩٠هريرة أنه قال قضى رسول الله صلى الله عليه وسلم في جنين امرأة من بني…٤٣٩١إنما هذا من إخوان الكهان٤٣٩٢قال اقتتلت امرأتان . وساق الحديث بقصته ولم يذكر وورثها ولدها ومن معهم…٤٣٩٣أسجع كسجع الأعراب٤٣٩٤سجع كسجع الأعراب٤٣٩٥ومحمد بن بشار، قالا حدثنا عبد الرحمن بن مهدي، عن سفيان، عن منصور،…٤٣٩٦ومحمد بن المثنى، وابن، بشار قالوا حدثنا محمد، بن جعفر عن شعبة، عن…٤٣٩٧قال استشار عمر بن الخطاب الناس في إملاص المرأة فقال المغيرة بن شعبة…
