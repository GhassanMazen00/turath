كتاب صلاة المسافرين وقصرها

٣٧٧ حديثًا•صفحة ٣ من ٣
١٨٧٤اقرءوا القرآن فإنه يأتي يوم القيامة شفيعا لأصحابه اقرءوا الزهراوين…١٨٧٥بهذا الإسناد . مثله غير أنه قال " وكأنهما " . في كليهما ولم يذكر قول…١٨٧٦يؤتى بالقرآن يوم القيامة وأهله الذين كانوا يعملون به تقدمه سورة…١٨٧٧وأحمد بن جواس الحنفي، قالا حدثنا أبو الأحوص، عن عمار بن رزيق، عن عبد…١٨٧٨الآيتان من آخر سورة البقرة من قرأهما في ليلة كفتاه١٨٧٩وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، كلاهما عن منصور، بهذا…١٨٨٠من قرأ هاتين الآيتين من آخر سورة البقرة في ليلة كفتاه١٨٨١وعبد، الرحمن بن يزيد عن أبي مسعود، عن النبي صلى الله عليه وسلم مثله .١٨٨٢وحدثنا أبو بكر بن أبي شيبة، حدثنا حفص، وأبو معاوية عن الأعمش، عن…١٨٨٣من حفظ عشر آيات من أول سورة الكهف عصم من الدجال١٨٨٤وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، ح وحدثني زهير بن حرب،…١٨٨٥يا أبا المنذر أتدري أى آية من كتاب الله معك أعظم١٨٨٦أيعجز أحدكم أن يقرأ في ليلة ثلث القرآن١٨٨٧إن الله جزأ القرآن ثلاثة أجزاء فجعل قل هو الله أحد جزءا من أجزاء…١٨٨٨احشدوا فإني سأقرأ عليكم ثلث القرآن١٨٨٩أقرأ عليكم ثلث القرآن١٨٩٠سلوه لأى شىء يصنع ذلك١٨٩١ألم تر آيات أنزلت الليلة لم ير مثلهن قط { قل أعوذ برب الفلق } و { قل…١٨٩٢أنزل - أو أنزلت - على آيات لم ير مثلهن قط المعوذتين١٨٩٣بن عامر الجهني وكان من رفعاء أصحاب محمد صلى الله عليه وسلم .١٨٩٤لا حسد إلا في اثنتين رجل آتاه الله القرآن فهو يقوم به آناء الليل…١٨٩٥لا حسد إلا على اثنتين رجل آتاه الله هذا الكتاب فقام به آناء الليل…١٨٩٦لا حسد إلا في اثنتين رجل آتاه الله مالا فسلطه على هلكته في الحق ورجل…١٨٩٧إن الله يرفع بهذا الكتاب أقواما ويضع به آخرين١٨٩٨أن نافع بن، عبد الحارث الخزاعي لقي عمر بن الخطاب بعسفان . بمثل حديث…١٨٩٩حدثنا يحيى بن يحيى، قال قرأت على مالك عن ابن شهاب، عن عروة بن الزبير،…١٩٠٠أن المسور بن مخرمة، وعبد الرحمن بن عبد القاري، أخبراه أنهما، سمعا عمر…١٩٠١وعبد بن حميد، قالا أخبرنا عبد الرزاق، أخبرنا معمر، عن الزهري، كرواية…١٩٠٢أقرأني جبريل - عليه السلام - على حرف فراجعته فلم أزل أستزيده فيزيدني…١٩٠٣بهذا الإسناد١٩٠٤يا أبى أرسل إلى أن اقرإ القرآن على حرف فرددت إليه أن هون على أمتي .١٩٠٥أنه كان جالسا في المسجد إذ دخل رجل فصلى فقرأ قراءة واقتص الحديث بمثل…١٩٠٦أسأل الله معافاته ومغفرته وإن أمتي لا تطيق ذلك١٩٠٧بهذا الإسناد مثله .١٩٠٨وابن، نمير جميعا عن وكيع، - قال أبو بكر حدثنا وكيع، - عن الأعمش، عن…١٩٠٩وحدثنا أبو كريب، حدثنا أبو معاوية، عن الأعمش، عن أبي وائل، قال جاء…١٩١٠في هذا الإسناد . بنحو حديثهما وقال إني لأعرف النظائر التي كان يقرأ…١٩١١حدثنا شيبان بن فروخ، حدثنا مهدي بن ميمون، حدثنا واصل الأحدب، عن أبي،…١٩١٢قال جاء رجل من بني بجيلة يقال له نهيك بن سنان إلى عبد الله فقال إني…١٩١٣وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، حدثنا شعبة، عن عمرو بن…١٩١٤حدثنا أحمد بن عبد الله بن يونس، حدثنا زهير، حدثنا أبو إسحاق، قال رأيت…١٩١٥وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي…١٩١٦وحدثنا أبو بكر بن أبي شيبة، وأبو كريب - واللفظ لأبي بكر - قالا حدثنا…١٩١٧قال أتى علقمة الشام فدخل مسجدا فصلى فيه ثم قام إلى حلقة فجلس فيها -…١٩١٨حدثنا علي بن حجر السعدي، حدثنا إسماعيل بن إبراهيم، عن داود بن أبي،…١٩١٩قال أتيت الشام فلقيت أبا الدرداء فذكر بمثل حديث ابن علية .١٩٢٠حدثنا يحيى بن يحيى، قال قرأت على مالك عن محمد بن يحيى بن حبان، عن…١٩٢١وإسماعيل بن سالم، جميعا عن هشيم، - قال داود حدثنا هشيم، - أخبرنا…١٩٢٢بهذا الإسناد غير أن في، حديث سعيد وهشام بعد الصبح حتى تشرق الشمس .١٩٢٣لا صلاة بعد صلاة العصر حتى تغرب الشمس ولا صلاة بعد صلاة الفجر حتى…١٩٢٤لا يتحرى أحدكم فيصلي عند طلوع الشمس ولا عند غروبها١٩٢٥لا تحروا بصلاتكم طلوع الشمس ولا غروبها فإنها تطلع بقرنى شيطان١٩٢٦إذا بدا حاجب الشمس فأخروا الصلاة حتى تبرز وإذا غاب حاجب الشمس فأخروا…١٩٢٧إن هذه الصلاة عرضت على من كان قبلكم فضيعوها فمن حافظ عليها كان له…١٩٢٨وحدثني زهير بن حرب، حدثنا يعقوب بن إبراهيم، حدثنا أبي، عن ابن إسحاق،…١٩٢٩وحدثنا يحيى بن يحيى، حدثنا عبد الله بن وهب، عن موسى بن على، عن أبيه،…١٩٣٠ويحيى بن أبي كثير، عن أبي أمامة، - قال عكرمة ولقي شداد أبا أمامة…١٩٣١أنها قالت وهم عمر إنما نهى رسول الله صلى الله عليه وسلم أن يتحرى طلوع…١٩٣٢لا تتحروا طلوع الشمس ولا غروبها فتصلوا عند ذلك١٩٣٣يا بنت أبي أمية سألت عن الركعتين بعد العصر إنه أتاني ناس من عبد القيس…١٩٣٤وقتيبة، وعلي بن حجر، قال ابن أيوب حدثنا إسماعيل، - وهو ابن جعفر -…١٩٣٥قالت ما ترك رسول الله صلى الله عليه وسلم ركعتين بعد العصر عندي قط .١٩٣٦قالت صلاتان ما تركهما رسول الله صلى الله عليه وسلم في بيتي قط سرا ولا…١٩٣٧وابن، بشار قال ابن المثنى حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي…١٩٣٨وحدثنا أبو بكر بن أبي شيبة، وأبو كريب جميعا عن ابن فضيل، - قال أبو…١٩٣٩وحدثنا شيبان بن فروخ، حدثنا عبد الوارث، عن عبد العزيز، - وهو ابن صهيب…١٩٤٠بين كل أذانين صلاة - قالها ثلاثا قال في الثالثة - لمن شاء١٩٤١وحدثنا أبو بكر بن أبي شيبة، حدثنا عبد الأعلى، عن الجريري، عن عبد…١٩٤٢قال صلى رسول الله صلى الله عليه وسلم صلاة الخوف بإحدى الطائفتين ركعة…١٩٤٣رسول الله صلى الله عليه وسلم في الخوف ويقول صليتها مع رسول الله صلى…١٩٤٤قال صلى رسول الله صلى الله عليه وسلم صلاة الخوف في بعض أيامه فقامت…١٩٤٥حدثنا محمد بن عبد الله بن نمير، حدثنا أبي، حدثنا عبد الملك بن أبي…١٩٤٦حدثنا أحمد بن عبد الله بن يونس، حدثنا زهير، حدثنا أبو الزبير، عن…١٩٤٧أن رسول الله صلى الله عليه وسلم صلى بأصحابه في الخوف فصفهم خلفه صفين…١٩٤٨بن خوات عمن صلى مع رسول الله صلى الله عليه وسلم يوم ذات الرقاع صلاة…١٩٤٩حدثنا أبو بكر بن أبي شيبة، حدثنا عفان، حدثنا أبان بن يزيد، حدثنا…١٩٥٠أن جابرا، أخبره أنه، صلى مع رسول الله صلى الله عليه وسلم صلاة الخوف…
