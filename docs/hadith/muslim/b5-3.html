كتاب الْمَسَاجِدِ وَمَوَاضِعِ الصَّلاَةِ

٣٩٦ حديثًا•صفحة ٣ من ٣
١٤٧٤بخمس وعشرين جزءا١٤٧٥صلاة الجماعة تعدل خمسا وعشرين من صلاة الفذ١٤٧٦صلاة مع الإمام أفضل من خمس وعشرين صلاة يصليها وحده١٤٧٧صلاة الجماعة أفضل من صلاة الفذ بسبع وعشرين درجة١٤٧٨صلاة الرجل في الجماعة تزيد على صلاته وحده سبعا وعشرين١٤٧٩وابن، نمير ح قال وحدثنا ابن نمير، حدثنا أبي قالا، حدثنا عبيد الله،…١٤٨٠وحدثناه ابن رافع، أخبرنا ابن أبي فديك، أخبرنا الضحاك، عن نافع، عن ابن…١٤٨١لقد هممت أن آمر رجلا يصلي بالناس ثم أخالف إلى رجال يتخلفون عنها فآمر…١٤٨٢إن أثقل صلاة على المنافقين صلاة العشاء وصلاة الفجر ولو يعلمون ما…١٤٨٣لقد هممت أن آمر فتياني أن يستعدوا لي بحزم من حطب ثم آمر رجلا يصلي…١٤٨٤وحدثنا زهير بن حرب، وأبو كريب وإسحاق بن إبراهيم عن وكيع، عن جعفر بن…١٤٨٥لقد هممت أن آمر رجلا يصلي بالناس ثم أحرق على رجال يتخلفون عن الجمعة…١٤٨٦هل تسمع النداء بالصلاة١٤٨٧حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن بشر العبدي، حدثنا زكرياء بن…١٤٨٨حدثنا أبو بكر بن أبي شيبة، حدثنا الفضل بن دكين، عن أبي العميس، عن علي…١٤٨٩حدثنا أبو بكر بن أبي شيبة، حدثنا أبو الأحوص، عن إبراهيم بن المهاجر،…١٤٩٠ورأى، رجلا يجتاز المسجد خارجا بعد الأذان فقال أما هذا فقد عصى أبا…١٤٩١من صلى العشاء في جماعة فكأنما قام نصف الليل ومن صلى الصبح في جماعة…١٤٩٢عثمان بن حكيم بهذا الإسناد . مثله .١٤٩٣من صلى الصبح فهو في ذمة الله فلا يطلبنكم الله من ذمته بشىء فيدركه…١٤٩٤من صلى صلاة الصبح فهو في ذمة الله فلا يطلبنكم الله من ذمته بشىء فإنه…١٤٩٥فيكبه في نار جهنم١٤٩٦سأفعل إن شاء الله١٤٩٧وعبد بن حميد، كلاهما عن عبد الرزاق، قال أخبرنا معمر، عن الزهري، قال…١٤٩٨وحدثنا إسحاق بن إبراهيم، أخبرنا الوليد بن مسلم، عن الأوزاعي، قال…١٤٩٩قوموا فأصلي لكم١٥٠٠وأبو الربيع، كلاهما عن عبد الوارث، قال شيبان حدثنا عبد الوارث، عن أبي…١٥٠١قوموا فلأصلي بكم١٥٠٢سمع موسى بن أنس، يحدث عن أنس بن مالك، أن رسول الله صلى الله عليه وسلم…١٥٠٣بهذا الإسناد .١٥٠٤قال حدثتني ميمونة، زوج النبي صلى الله عليه وسلم قالت كان رسول الله…١٥٠٥أنه دخل على رسول الله صلى الله عليه وسلم فوجده يصلي على حصير يسجد…١٥٠٦صلاة الرجل في جماعة تزيد على صلاته في بيته وصلاته في سوقه بضعا وعشرين…١٥٠٧في هذا الإسناد بمثل معناه .١٥٠٨إن الملائكة تصلي على أحدكم ما دام في مجلسه تقول اللهم اغفر له اللهم…١٥٠٩لا يزال العبد في صلاة ما كان في مصلاه ينتظر الصلاة وتقول الملائكة…١٥١٠لا يزال أحدكم في صلاة ما دامت الصلاة تحبسه لا يمنعه أن ينقلب إلى أهله…١٥١١أحدكم ما قعد ينتظر الصلاة في صلاة ما لم يحدث تدعو له الملائكة اللهم…١٥١٢وحدثنا محمد بن رافع، حدثنا عبد الرزاق، حدثنا معمر، عن همام بن منبه،…١٥١٣إن أعظم الناس أجرا في الصلاة أبعدهم إليها ممشى فأبعدهم والذي ينتظر…١٥١٤قد جمع الله لك ذلك كله١٥١٥بهذا الإسناد . بنحوه .١٥١٦إن لك ما احتسبت١٥١٧ومحمد بن أبي عمر، كلاهما عن ابن عيينة، ح وحدثنا سعيد بن أزهر الواسطي،…١٥١٨إن لكم بكل خطوة درجة١٥١٩إنه بلغني أنكم تريدون أن تنتقلوا قرب المسجد١٥٢٠يا بني سلمة دياركم تكتب آثاركم١٥٢١من تطهر في بيته ثم مشى إلى بيت من بيوت الله ليقضي فريضة من فرائض الله…١٥٢٢أرأيتم لو أن نهرا بباب أحدكم يغتسل منه كل يوم خمس مرات هل يبقى من…١٥٢٣مثل الصلوات الخمس كمثل نهر جار غمر على باب أحدكم يغتسل منه كل يوم خمس…١٥٢٤من غدا إلى المسجد أو راح أعد الله له في الجنة نزلا كلما غدا أو راح١٥٢٥قال قلت لجابر بن سمرة أكنت تجالس رسول الله صلى الله عليه وسلم قال نعم…١٥٢٦وحدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن سفيان، قال أبو بكر وحدثنا…١٥٢٧وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، كلاهما عن سماك، بهذا…١٥٢٨أحب البلاد إلى الله مساجدها وأبغض البلاد إلى الله أسواقها١٥٢٩إذا كانوا ثلاثة فليؤمهم أحدهم وأحقهم بالإمامة أقرؤهم١٥٣٠بهذا الإسناد مثله .١٥٣١وحدثنا محمد بن المثنى، حدثنا سالم بن نوح، ح وحدثنا حسن بن عيسى، حدثنا…١٥٣٢يؤم القوم أقرؤهم لكتاب الله فإن كانوا في القراءة سواء فأعلمهم بالسنة…١٥٣٣بهذا الإسناد مثله .١٥٣٤يؤم القوم أقرؤهم لكتاب الله وأقدمهم قراءة فإن كانت قراءتهم سواء…١٥٣٥ارجعوا إلى أهليكم فأقيموا فيهم وعلموهم ومروهم فإذا حضرت الصلاة فليؤذن…١٥٣٦وخلف بن هشام، قالا حدثنا حماد، عن أيوب، بهذا الإسناد .١٥٣٧قال أتيت رسول الله صلى الله عليه وسلم في ناس ونحن شببة متقاربون .…١٥٣٨إذا حضرت الصلاة فأذنا ثم أقيما وليؤمكما أكبركما١٥٣٩وحدثناه أبو سعيد الأشج، حدثنا حفص، - يعني ابن غياث - حدثنا خالد…١٥٤٠سمع الله لمن حمده ربنا ولك الحمد١٥٤١واجعلها عليهم كسني يوسف١٥٤٢سمع الله لمن حمده١٥٤٣سمع الله لمن حمده١٥٤٤أنه سمع أبا هريرة، يقول والله لأقربن بكم صلاة رسول الله صلى الله عليه…١٥٤٥وحدثنا يحيى بن يحيى، قال قرأت على مالك عن إسحاق بن عبد الله بن أبي…١٥٤٦وزهير بن حرب، قالا حدثنا إسماعيل، عن أيوب، عن محمد، قال قلت لأنس هل…١٥٤٧عصية عصت الله ورسوله١٥٤٨أن رسول الله صلى الله عليه وسلم قنت شهرا بعد الركوع في صلاة الفجر…١٥٤٩قبل الركوع أو بعد الركوع فقال قبل الركوع . قال قلت فإن ناسا يزعمون أن…١٥٥٠يقول ما رأيت رسول الله صلى الله عليه وسلم وجد على سرية ما وجد على…١٥٥١وابن، فضيل ح وحدثنا ابن أبي عمر، حدثنا مروان، كلهم عن عاصم، عن أنس،…١٥٥٢وحدثنا عمرو الناقد، حدثنا الأسود بن عامر، أخبرنا شعبة، عن قتادة، عن…١٥٥٣وحدثنا عمرو الناقد، حدثنا الأسود بن عامر، أخبرنا شعبة، عن موسى بن…١٥٥٤أن رسول الله صلى الله عليه وسلم قنت شهرا يدعو على أحياء من أحياء…١٥٥٥وابن، بشار قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن عمرو بن مرة، قال…١٥٥٦قال قنت رسول الله صلى الله عليه وسلم في الفجر والمغرب .١٥٥٧اللهم العن بني لحيان ورعلا وذكوان وعصية عصوا الله ورسوله غفار غفر…١٥٥٨غفار غفر الله لها وأسلم سالمها الله وعصية عصت الله ورسوله اللهم العن…١٥٥٩. بمثله إلا أنه لم يقل فجعلت لعنة الكفرة من أجل ذلك .١٥٦٠حدثني حرملة بن يحيى التجيبي، أخبرنا ابن وهب، أخبرني يونس، عن ابن…١٥٦١ليأخذ كل رجل برأس راحلته فإن هذا منزل حضرنا فيه الشيطان١٥٦٢إنكم تسيرون عشيتكم وليلتكم وتأتون الماء إن شاء الله غدا١٥٦٣وحدثني أحمد بن سعيد بن صخر الدارمي، حدثنا عبيد الله بن عبد المجيد،…١٥٦٤حدثنا إسحاق بن إبراهيم الحنظلي، أخبرنا النضر بن شميل، حدثنا عوف بن…١٥٦٥قال كان رسول الله صلى الله عليه وسلم إذا كان في سفر فعرس بليل اضطجع…١٥٦٦من نسي صلاة فليصلها إذا ذكرها لا كفارة لها إلا ذلك١٥٦٧لا كفارة لها إلا ذلك١٥٦٨من نسي صلاة أو نام عنها فكفارتها أن يصليها إذا ذكرها١٥٦٩إذا رقد أحدكم عن الصلاة أو غفل عنها فليصلها إذا ذكرها فإن الله يقول…
