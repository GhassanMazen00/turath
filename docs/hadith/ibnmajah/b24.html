١٢٩ حديثًا
٢٧٥٣أعد الله لمن خرج في سبيله لا يخرجه إلا جهاد في سبيلي وإيمان بي وتصديق…٢٧٥٤المجاهد في سبيل الله مضمون على الله إما أن يكفته إلى مغفرته ورحمته…٢٧٥٥غدوة أو روحة في سبيل الله خير من الدنيا وما فيها٢٧٥٦غدوة أو روحة في سبيل الله خير من الدنيا وما فيها٢٧٥٧لغدوة أو روحة في سبيل الله خير من الدنيا وما فيها٢٧٥٨من جهز غازيا في سبيل الله حتى يستقل كان له مثل أجره حتى يموت أو يرجع٢٧٥٩من جهز غازيا في سبيل الله كان له مثل أجره من غير أن ينقص من أجر…٢٧٦٠أفضل دينار ينفقه الرجل دينار ينفقه على عياله ودينار ينفقه على فرس في…٢٧٦١من أرسل بنفقة في سبيل الله وأقام في بيته فله بكل درهم سبعمائة درهم…٢٧٦٢من لم يغز أو يجهز غازيا أو يخلف غازيا في أهله بخير أصابه الله سبحانه…٢٧٦٣من لقي الله وليس له أثر في سبيل الله لقي الله وفيه ثلمة٢٧٦٤إن بالمدينة لقوما ما سرتم من مسير ولا قطعتم واديا إلا كانوا معكم فيه٢٧٦٥إن بالمدينة رجالا ما قطعتم واديا ولا سلكتم طريقا إلا شركوكم في الأجر…٢٧٦٦من رابط ليلة في سبيل الله سبحانه كانت كألف ليلة صيامها وقيامها٢٧٦٧من مات مرابطا في سبيل الله أجرى عليه أجر عمله الصالح الذي كان يعمل…٢٧٦٨لرباط يوم في سبيل الله من وراء عورة المسلمين محتسبا من غير شهر رمضان…٢٧٦٩رحم الله حارس الحرس٢٧٧٠حرس ليلة في سبيل الله أفضل من صيام رجل وقيامه في أهله ألف سنة السنة…٢٧٧١أوصيك بتقوى الله والتكبير على كل شرف٢٧٧٢يا أيها الناس لن تراعوا٢٧٧٣إذا استنفرتم فانفروا٢٧٧٤لا يجتمع غبار في سبيل الله ودخان جهنم في جوف عبد مسلم٢٧٧٥من راح روحة في سبيل الله كان له بمثل ما أصابه من الغبار مسكا يوم…٢٧٧٦ناس من أمتي عرضوا على يركبون ظهر هذا البحر كالملوك على الأسرة٢٧٧٧غزوة في البحر مثل عشر غزوات في البر والذي يسدر في البحر كالمتشحط في…٢٧٧٨شهيد البحر مثل شهيدى البر والمائد في البحر كالمتشحط في دمه في البر…٢٧٧٩لو لم يبق من الدنيا إلا يوم لطوله الله عز وجل حتى يملك رجل من أهل…٢٧٨٠ستفتح عليكم الآفاق وستفتح عليكم مدينة يقال لها قزوين من رابط فيها…٢٧٨١ويحك أحية أمك٢٧٨٢فارجع إليهما فأضحكهما كما أبكيتهما٢٧٨٣من قاتل لتكون كلمة الله هي العليا فهو في سبيل الله٢٧٨٤ألا قلت خذها مني وأنا الغلام الأنصاري٢٧٨٥ما من غازية تغزو في سبيل الله فيصيبوا غنيمة إلا تعجلوا ثلثى أجرهم فإن…٢٧٨٦الخير معقود بنواصي الخيل إلى يوم القيامة٢٧٨٧الخيل في نواصيها الخير إلى يوم القيامة٢٧٨٨الخيل في نواصيها الخير - أو قال الخيل معقود في نواصيها قال سهيل أنا…٢٧٨٩خير الخيل الأدهم الأقرح المحجل الأرثم طلق اليد اليمنى فإن لم يكن أدهم…٢٧٩٠قال كان النبي صلى الله عليه وسلم يكره الشكال من الخيل .٢٧٩١من ارتبط فرسا في سبيل الله ثم عالج علفه بيده كان له بكل حبة حسنة٢٧٩٢من قاتل في سبيل الله عز وجل من رجل مسلم فواق ناقة وجبت له الجنة٢٧٩٣قال حضرت حربا فقال عبد الله بن رواحة يا نفس ألا أراك تكرهين الجنه…٢٧٩٤من أهريق دمه وعقر جواده٢٧٩٥ما من مجروح يجرح في سبيل الله - والله أعلم بمن يجرح في سبيله - إلا…٢٧٩٦اللهم منزل الكتاب سريع الحساب اهزم الأحزاب اللهم اهزمهم وزلزلهم٢٧٩٧من سأل الله الشهادة بصدق من قلبه بلغه الله منازل الشهداء وإن مات على…٢٧٩٨لا تجف الأرض من دم الشهيد حتى تبتدره زوجتاه كأنهما ظئران أضلتا…٢٧٩٩للشهيد عند الله ست خصال يغفر له في أول دفعة من دمه ويرى مقعده من…٢٨٠٠يا جابر ألا أخبرك ما قال الله عز وجل لأبيك٢٨٠١أرواحهم كطير خضر تسرح في الجنة في أيها شاءت ثم تأوي إلى قناديل معلقة…٢٨٠٢ما يجد الشهيد مس القتل إلا كما يجد أحدكم مس القرصة٢٨٠٣إن شهداء أمتي إذا لقليل القتل في سبيل الله شهادة والمطعون شهادة…٢٨٠٤ما تقولون في الشهيد فيكم٢٨٠٥وسويد بن سعيد، قالا حدثنا مالك بن أنس، حدثني الزهري، عن أنس بن مالك،…٢٨٠٦إن شاء الله تعالى . أن النبي صلى الله عليه وسلم يوم أحد أخذ درعين…٢٨٠٧حدثنا عبد الرحمن بن إبراهيم الدمشقي، حدثنا الوليد بن مسلم، حدثنا…٢٨٠٨أن رسول الله صلى الله عليه وسلم تنفل سيفه ذا الفقار يوم بدر .٢٨٠٩لا تفعل فإنك إن فعلت لم ترفع ضالة٢٨١٠ما هذه ألقها وعليكم بهذه وأشباهها ورماح القنا فإنهما يزيد الله بهما…٢٨١١إن الله ليدخل بالسهم الواحد الثلاثة الجنة صانعه يحتسب في صنعته الخير…٢٨١٢من رمى العدو بسهم فبلغ سهمه العدو أصاب أو أخطأ فيعدل رقبة٢٨١٣{وأعدوا لهم ما استطعتم من قوة} ألا وإن القوة الرمى٢٨١٤من تعلم الرمى ثم تركه فقد عصاني٢٨١٥رميا بني إسماعيل فإن أباكم كان راميا٢٨١٦قال قدمت المدينة فرأيت النبي صلى الله عليه وسلم قائما على المنبر…٢٨١٧وعبدة بن عبد الله، قال حدثنا يحيى بن آدم، حدثنا شريك، عن عمار الدهني،…٢٨١٨أن راية، رسول الله صلى الله عليه وسلم كانت سوداء ولواؤه أبيض .٢٨١٩أنها أخرجت جبة مزررة بالديباج فقالت كان النبي صلى الله عليه وسلم يلبس…٢٨٢٠والديباج، إلا ما كان هكذا ثم أشار بإصبعه ثم الثانية ثم الثالثة ثم…٢٨٢١قال كأني أنظر إلى رسول الله صلى الله عليه وسلم وعليه عمامة سوداء قد…٢٨٢٢أن النبي صلى الله عليه وسلم دخل مكة وعليه عمامة سوداء .٢٨٢٣حدثنا عبيد الله بن عبد الكريم، حدثنا سنيد بن داود، عن خالد بن حيان…٢٨٢٤لأن أشيع مجاهدا في سبيل الله فأكفه على رحله غدوة أو روحة أحب إلى من…٢٨٢٥أستودعك الله الذي لا تضيع ودائعه٢٨٢٦أستودع الله دينك وأمانتك وخواتيم عملك٢٨٢٧يا أكثم اغز مع غير قومك يحسن خلقك وتكرم على رفقائك يا أكثم خير…٢٨٢٨رسول الله صلى الله عليه وسلم كانوا يوم بدر ثلاثمائة وبضعة عشر على عدة…٢٨٢٩صاحب النبي صلى الله عليه وسلم يقول إياكم والسرية التي إن لقيت فرت وإن…٢٨٣٠لا يختلجن في صدرك طعام ضارعت فيه نصرانية٢٨٣١رويم اللخمي، عن أبي ثعلبة الخشني، - قال ولقيه وكلمه - قال أتيت رسول…٢٨٣٢إنا لا نستعين بمشرك٢٨٣٣أن النبي صلى الله عليه وسلم قال " الحرب خدعة " .٢٨٣٤أن النبي صلى الله عليه وسلم قال " الحرب خدعة " .٢٨٣٥وحفص بن عمرو، قالا حدثنا عبد الرحمن بن مهدي، ح وحدثنا محمد بن…٢٨٣٦وعكرمة بن عمار، عن إياس بن سلمة بن الأكوع، عن أبيه، قال بارزت رجلا…٢٨٣٧أن رسول الله صلى الله عليه وسلم نفله سلب قتيل قتله يوم حنين .٢٨٣٨من قتل فله السلب٢٨٣٩حدثنا أبو بكر بن أبي شيبة، حدثنا سفيان بن عيينة، عن الزهري، عن عبيد…٢٨٤٠حدثنا محمد بن إسماعيل، أنبأنا وكيع، عن عكرمة بن عمار، عن إياس بن سلمة…٢٨٤١حدثنا يحيى بن حكيم، حدثنا عثمان بن عمر، أنبأنا مالك بن أنس، عن نافع،…٢٨٤٢حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن سفيان، عن أبي الزناد، عن…٢٨٤٣ائت أبنى صباحا ثم حرق٢٨٤٤أن رسول الله صلى الله عليه وسلم حرق نخل بني النضير وقطع . وهي البويرة…٢٨٤٥أن النبي صلى الله عليه وسلم حرق نخل بني النضير وقطع . وفيه يقول…٢٨٤٦لله أبوك هبها لي٢٨٤٧قال ذهبت فرس له فأخذها العدو فظهر عليهم المسلمون فرد عليه في زمن رسول…٢٨٤٨صلوا على صاحبكم٢٨٤٩قال كان على ثقل النبي صلى الله عليه وسلم رجل يقال له كركرة . فمات…٢٨٥٠يا أيها الناس إن هذا من غنائمكم أدوا الخيط والمخيط فما فوق ذلك فما…٢٨٥١وعلي بن محمد، قالا حدثنا وكيع، عن سفيان، عن يزيد بن يزيد بن جابر، عن…٢٨٥٢أن النبي صلى الله عليه وسلم نفل في البدأة الربع وفي الرجعة الثلث .٢٨٥٣حدثنا علي بن محمد، حدثنا أبو الحسين، أنبأنا رجاء بن أبي سلمة، حدثنا…٢٨٥٤أن النبي صلى الله عليه وسلم أسهم يوم خيبر للفارس ثلاثة أسهم للفرس…٢٨٥٥حدثنا علي بن محمد، حدثنا وكيع، حدثنا هشام بن سعد، عن محمد بن زيد بن…٢٨٥٦قالت غزوت مع رسول الله صلى الله عليه وسلم سبع غزوات أخلفهم في رحالهم…٢٨٥٧سيروا باسم الله وفي سبيل الله قاتلوا من كفر بالله ولا تمثلوا ولا…٢٨٥٨اغزوا باسم الله
