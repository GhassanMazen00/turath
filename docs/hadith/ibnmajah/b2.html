سنن ابن ماجه

كتاب الصلاة

٣٩ حديثًا
٦٦٧صل معنا هذين اليومين٦٦٨نزل جبريل فأمني فصليت معه ثم صليت معه ثم صليت معه ثم صليت معه ثم صليت…٦٦٩حدثنا أبو بكر بن أبي شيبة، حدثنا سفيان بن عيينة، عن الزهري، عن عروة،…٦٧٠تشهده ملائكة الليل والنهار٦٧١حدثنا عبد الرحمن بن إبراهيم الدمشقي، حدثنا الوليد بن مسلم، حدثنا…٦٧٢أصبحوا بالصبح فإنه أعظم للأجر أو لأجركم٦٧٣أن النبي صلى الله عليه وسلم كان يصلي الظهر إذا دحضت الشمس .٦٧٤قال كان النبي صلى الله عليه وسلم يصلي صلاة الهجير التي تدعونها الظهر…٦٧٥نحوه .٦٧٦حدثنا أبو كريب، حدثنا معاوية بن هشام، عن سفيان، عن زيد بن جبير، عن…٦٧٧إذا اشتد الحر فأبردوا بالصلاة فإن شدة الحر من فيح جهنم٦٧٨إذا اشتد الحر فأبردوا بالظهر فإن شدة الحر من فيح جهنم٦٧٩أبردوا بالظهر فإن شدة الحر من فيح جهنم٦٨٠أبردوا بالصلاة فإن شدة الحر من فيح جهنم٦٨١قال قال رسول الله صلى الله عليه وسلم " أبردوا بالظهر " .٦٨٢أنه أخبره أن رسول الله صلى الله عليه وسلم كان يصلي العصر والشمس…٦٨٣قالت صلى النبي صلى الله عليه وسلم العصر والشمس في حجرتي لم يظهرها…٦٨٤ملأ الله بيوتهم وقبورهم نارا كما شغلونا عن الصلاة الوسطى٦٨٥إن الذي تفوته صلاة العصر فكأنما وتر أهله وماله٦٨٦حبسونا عن صلاة الوسطى ملأ الله قبورهم وبيوتهم نارا٦٨٧نحوه .٦٨٨أنه كان يصلي مع النبي صلى الله عليه وسلم المغرب إذا توارت بالحجاب .٦٨٩لا تزال أمتي على الفطرة ما لم يؤخروا المغرب حتى تشتبك النجوم٦٩٠لولا أن أشق على أمتي لأمرتهم بتأخير العشاء٦٩١لولا أن أشق على أمتي لأخرت صلاة العشاء إلى ثلث الليل أو نصف الليل٦٩٢إن الناس قد صلوا وناموا وإنكم لن تزالوا في صلاة ما انتظرتم الصلاة٦٩٣إن الناس قد صلوا وناموا وأنتم لم تزالوا في صلاة ما انتظرتم الصلاة…٦٩٤بكروا بالصلاة في اليوم الغيم فإنه من فاتته صلاة العصر حبط عمله٦٩٥يصليها إذا ذكرها٦٩٦من نسي صلاة فليصلها إذا ذكرها٦٩٧حدثنا حرملة بن يحيى، حدثنا عبد الله بن وهب، حدثنا يونس، عن ابن شهاب،…٦٩٨ليس في النوم تفريط إنما التفريط في اليقظة فإذا نسي أحدكم صلاة أو نام…٦٩٩من أدرك من العصر ركعة قبل أن تغرب الشمس فقد أدركها ومن أدرك من الصبح…٧٠٠من أدرك من الصبح ركعة قبل أن تطلع الشمس فقد أدركها ومن أدرك من العصر…٧٠١ومحمد بن جعفر، وعبد الوهاب، قالوا حدثنا عوف، عن أبي المنهال، سيار بن…٧٠٢قالت ما نام رسول الله صلى الله عليه وسلم قبل العشاء ولا سمر بعدها .٧٠٣وإسحاق بن إبراهيم بن حبيب، وعلي بن المنذر، قالوا حدثنا محمد بن فضيل،…٧٠٤لا تغلبنكم الأعراب على اسم صلاتكم فإنها العشاء وإنهم ليعتمون بالإبل٧٠٥لا تغلبنكم الأعراب على اسم صلاتكم - زاد ابن حرملة - فإنما هي العشاء…
