سنن ابن ماجه

كتاب الفتن

١٧٥ حديثًا•صفحة ١ من ٢
٣٩٢٧أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوها عصموا مني…٣٩٢٨أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوا لا إله إلا…٣٩٢٩اذهبوا به فاقتلوه٣٩٣٠حدثنا سويد بن سعيد، حدثنا علي بن مسهر، عن عاصم، عن السميط بن السمير،…٣٩٣٠.٢إن الأرض لتقبل من هو أشر منه ولكن الله أحب أن يريكم تعظيم حرمة لا إله…٣٩٣١ألا إن أحرم الأيام يومكم هذا ألا وإن أحرم الشهور شهركم هذا ألا وإن…٣٩٣٢ما أطيبك وأطيب ريحك ما أعظمك وأعظم حرمتك والذي نفس محمد بيده لحرمة…٣٩٣٣كل المسلم على المسلم حرام دمه وماله وعرضه٣٩٣٤المؤمن من أمنه الناس على أموالهم وأنفسهم والمهاجر من هجر الخطايا…٣٩٣٥من انتهب نهبة مشهورة فليس منا٣٩٣٦لا يزني الزاني حين يزني وهو مؤمن ولا يشرب الخمر حين يشربها وهو مؤمن…٣٩٣٧من انتهب نهبة فليس منا٣٩٣٨إن النهبة لا تحل٣٩٣٩سباب المسلم فسوق وقتاله كفر٣٩٤٠سباب المسلم فسوق وقتاله كفر٣٩٤١سباب المسلم فسوق وقتاله كفر٣٩٤٢وعبد الرحمن بن مهدي، قالا حدثنا شعبة، عن علي بن مدرك، قال سمعت أبا…٣٩٤٣ويحكم - أو ويلكم - لا ترجعوا بعدي كفارا يضرب بعضكم رقاب بعض٣٩٤٤ألا إني فرطكم على الحوض وإني مكاثر بكم الأمم فلا تقتتلن بعدي٣٩٤٥من صلى الصبح فهو في ذمة الله فلا تخفروا الله في عهده فمن قتله طلبه…٣٩٤٦من صلى الصبح فهو في ذمة الله عز وجل٣٩٤٧المؤمن أكرم على الله عز وجل من بعض ملائكته٣٩٤٨من قاتل تحت راية عمية يدعو إلى عصبية أو يغضب لعصبية فقتلته جاهلية٣٩٤٩لا ولكن من العصبية أن يعين الرجل قومه على الظلم٣٩٥٠إن أمتي لن تجتمع على ضلالة فإذا رأيتم اختلافا فعليكم بالسواد الأعظم٣٩٥١إني صليت صلاة رغبة ورهبة سألت الله عز وجل لأمتي ثلاثا فأعطاني اثنتين…٣٩٥٢زويت لي الأرض حتى رأيت مشارقها ومغاربها وأعطيت الكنزين الأصفر - أو…٣٩٥٣لا إله إلا الله ويل للعرب من شر قد اقترب فتح اليوم من ردم يأجوج ومأجوج٣٩٥٤ستكون فتن يصبح الرجل فيها مؤمنا ويمسي كافرا إلا من أحياه الله بالعلم٣٩٥٥فتنة الرجل في أهله وولده وجاره تكفرها الصلاة والصيام والصدقة والأمر…٣٩٥٦إنه لم يكن نبي قبلي إلا كان حقا عليه أن يدل أمته على ما يعلمه خيرا…٣٩٥٧كيف بكم وبزمان يوشك أن يأتي يغربل الناس فيه غربلة وتبقى حثالة من…٣٩٥٨كيف أنت يا أبا ذر وموتا يصيب الناس حتى يقوم البيت بالوصيف٣٩٥٩إن بين يدى الساعة لهرجا٣٩٦٠مؤذن مسجد جردان قال حدثتني عديسة بنت أهبان، قالت لما جاء علي بن أبي…٣٩٦١إن بين يدى الساعة فتنا كقطع الليل المظلم يصبح الرجل فيها مؤمنا ويمسي…٣٩٦٢إنها ستكون فتنة وفرقة واختلاف فإذا كان كذلك فأت بسيفك أحدا فاضربه حتى…٣٩٦٣ما من مسلمين التقيا بأسيافهما إلا كان القاتل والمقتول في النار٣٩٦٤إذا التقى المسلمان بسيفيهما فالقاتل والمقتول في النار٣٩٦٥إذا المسلمان حمل أحدهما على أخيه السلاح فهما على جرف جهنم فإذا قتل…٣٩٦٦من شر الناس منزلة عند الله يوم القيامة عبد أذهب آخرته بدنيا غيره٣٩٦٧تكون فتنة تستنظف العرب قتلاها في النار اللسان فيها أشد من وقع السيف٣٩٦٨إياكم والفتن فإن اللسان فيها مثل وقع السيف٣٩٦٩إن أحدكم ليتكلم بالكلمة من رضوان الله ما يظن أن تبلغ ما بلغت فيكتب…٣٩٧٠إن الرجل ليتكلم بالكلمة من سخط الله لا يرى بها بأسا فيهوي بها في نار…٣٩٧١من كان يؤمن بالله واليوم الآخر فليقل خيرا أو ليسكت٣٩٧٢قل ربي الله ثم استقم٣٩٧٣لقد سألت عظيما وإنه ليسير على من يسره الله عليه تعبد الله لا تشرك به…٣٩٧٤كلام ابن آدم عليه لا له إلا الأمر بالمعروف والنهى عن المنكر وذكر الله…٣٩٧٥حدثنا علي بن محمد، حدثنا خالي، يعلى عن الأعمش، عن إبراهيم، عن أبي…٣٩٧٦من حسن إسلام المرء تركه ما لا يعنيه٣٩٧٧خير معايش الناس لهم رجل ممسك بعنان فرسه في سبيل الله ويطير على متنه…٣٩٧٨رجل مجاهد في سبيل الله بنفسه وماله٣٩٧٩يكون دعاة على أبواب جهنم من أجابهم إليها قذفوه فيها٣٩٨٠يوشك أن يكون خير مال المسلم غنم يتبع بها شعف الجبال ومواقع القطر يفر…٣٩٨١تكون فتن على أبوابها دعاة إلى النار فأن تموت وأنت عاض على جذل شجرة…٣٩٨٢لا يلدغ المؤمن من جحر مرتين٣٩٨٣لا يلدغ المؤمن من جحر مرتين٣٩٨٤الحلال بين والحرام بين وبينهما مشتبهات لا يعلمها كثير من الناس فمن…٣٩٨٥العبادة في الهرج كهجرة إلى٣٩٨٦بدأ الإسلام غريبا وسيعود غريبا فطوبى للغرباء٣٩٨٧إن الإسلام بدأ غريبا وسيعود غريبا فطوبى للغرباء٣٩٨٨إن الإسلام بدأ غريبا وسيعود غريبا فطوبى للغرباء٣٩٨٩إن يسير الرياء شرك وإن من عادى لله وليا فقد بارز الله بالمحاربة إن…٣٩٩٠الناس كإبل مائة لا تكاد تجد فيها راحلة٣٩٩١تفرقت اليهود على إحدى وسبعين فرقة وتفترق أمتي على ثلاث وسبعين فرقة٣٩٩٢افترقت اليهود على إحدى وسبعين فرقة فواحدة في الجنة وسبعون في النار…٣٩٩٣إن بني إسرائيل افترقت على إحدى وسبعين فرقة وإن أمتي ستفترق على ثنتين…٣٩٩٤لتتبعن سنة من كان قبلكم باعا بباع وذراعا بذراع وشبرا بشبر حتى لو…٣٩٩٥لا والله ما أخشى عليكم أيها الناس إلا ما يخرج الله لكم من زهرة الدنيا٣٩٩٦إذا فتحت عليكم خزائن فارس والروم أى قوم أنتم٣٩٩٧أظنكم سمعتم أن أبا عبيدة قدم بشىء من البحرين٣٩٩٨ما أدع بعدي فتنة أضر على الرجال من النساء٣٩٩٩ما من صباح إلا وملكان يناديان ويل للرجال من النساء وويل للنساء من…٤٠٠٠إن الدنيا خضرة حلوة وإن الله مستخلفكم فيها فناظر كيف تعملون ألا…٤٠٠١يا أيها الناس انهوا نساءكم عن لبس الزينة والتبختر في المسجد فإن بني…٤٠٠٢أيما امرأة تطيبت ثم خرجت إلى المسجد لم تقبل لها صلاة حتى تغتسل٤٠٠٣يا معشر النساء تصدقن وأكثرن من الاستغفار فإني رأيتكن أكثر أهل النار٤٠٠٤مروا بالمعروف وانهوا عن المنكر قبل أن تدعوا فلا يستجاب لكم٤٠٠٥إن الناس إذا رأوا المنكر لا يغيرونه أوشك أن يعمهم الله بعقابه٤٠٠٦إن بني إسرائيل لما وقع فيهم النقص كان الرجل يرى أخاه على الذنب فينهاه…٤٠٠٧ألا لا يمنعن رجلا هيبة الناس أن يقول بحق إذا علمه٤٠٠٨لا يحقر أحدكم نفسه٤٠٠٩ما من قوم يعمل فيهم بالمعاصي هم أعز منهم وأمنع لا يغيرون إلا عمهم…٤٠١٠ألا تحدثوني بأعاجيب ما رأيتم بأرض الحبشة٤٠١١أفضل الجهاد كلمة عدل عند سلطان جائر٤٠١٢حدثنا راشد بن سعيد الرملي، حدثنا الوليد بن مسلم، حدثنا حماد بن سلمة،…٤٠١٣من رأى منكم منكرا فاستطاع أن يغيره بيده فليغيره بيده فإن لم يستطع…٤٠١٤بل ائتمروا بالمعروف وتناهوا عن المنكر حتى إذا رأيت شحا مطاعا وهوى…٤٠١٥إذا ظهر فيكم ما ظهر في الأمم قبلكم٤٠١٦لا ينبغي للمؤمن أن يذل نفسه٤٠١٧إن الله ليسأل العبد يوم القيامة حتى يقول ما منعك إذ رأيت المنكر أن…٤٠١٨إن الله يملي للظالم فإذا أخذه لم يفلته٤٠١٩يا معشر المهاجرين خمس إذا ابتليتم بهن وأعوذ بالله أن تدركوهن لم تظهر…٤٠٢٠ليشربن ناس من أمتي الخمر يسمونها بغير اسمها يعزف على رءوسهم بالمعازف…٤٠٢١{يلعنهم الله ويلعنهم اللاعنون }٤٠٢٢لا يزيد في العمر إلا البر ولا يرد القدر إلا الدعاء وإن الرجل ليحرم…٤٠٢٣الأنبياء ثم الأمثل فالأمثل يبتلى العبد على حسب دينه فإن كان في دينه…٤٠٢٤إنا كذلك يضعف لنا البلاء ويضعف لنا الأجر٤٠٢٥حدثنا محمد بن عبد الله بن نمير، حدثنا وكيع، حدثنا الأعمش، عن شقيق، عن…٤٠٢٦نحن أحق بالشك من إبراهيم إذ قال {رب أرني كيف تحيي الموتى قال أو لم…٤٠٢٧كيف يفلح قوم خضبوا وجه نبيهم بالدم وهو يدعوهم إلى الله٤٠٢٨فعل بي هؤلاء وفعلوا٤٠٢٩أحصوا لي كل من تلفظ بالإسلام٤٠٣٠يا جبريل ما هذه الريح الطيبة قال هذه ريح قبر الماشطة وابنيها وزوجها .…٤٠٣١عظم الجزاء مع عظم البلاء وإن الله إذا أحب قوما ابتلاهم فمن رضي فله…٤٠٣٢المؤمن الذي يخالط الناس ويصبر على أذاهم أعظم أجرا من المؤمن الذي لا…٤٠٣٣ثلاث من كن فيه وجد طعم الإيمان - وقال بندار حلاوة الإيمان - من كان…٤٠٣٤لا تشرك بالله شيئا وإن قطعت وحرقت ولا تترك صلاة مكتوبة متعمدا فمن…٤٠٣٥لم يبق من الدنيا إلا بلاء وفتنة٤٠٣٦سيأتي على الناس سنوات خداعات يصدق فيها الكاذب ويكذب فيها الصادق…٤٠٣٧والذي نفسي بيده لا تذهب الدنيا حتى يمر الرجل على القبر فيتمرغ عليه…٤٠٣٨لتنتقون كما ينتقى التمر من أغفاله فليذهبن خياركم وليبقين شراركم…٤٠٣٩لا يزداد الأمر إلا شدة ولا الدنيا إلا إدبارا ولا الناس إلا شحا ولا…٤٠٤٠بعثت أنا والساعة كهاتين٤٠٤١لا تقوم الساعة حتى تكون عشر آيات الدجال والدخان وطلوع الشمس من مغربها٤٠٤٢حدثنا عبد الرحمن بن إبراهيم، حدثنا الوليد بن مسلم، حدثنا عبد الله بن…٤٠٤٣لا تقوم الساعة حتى تقتلوا إمامكم وتجتلدوا بأسيافكم ويرث دنياكم شراركم٤٠٤٤ما المسئول عنها بأعلم من السائل ولكن سأخبرك عن أشراطها إذا ولدت الأمة…٤٠٤٥إن من أشراط الساعة أن يرفع العلم ويظهر الجهل ويفشو الزنا ويشرب الخمر…٤٠٤٦لا تقوم الساعة حتى يحسر الفرات عن جبل من ذهب فيقتتل الناس عليه فيقتل…٤٠٤٧لا تقوم الساعة حتى يفيض المال وتظهر الفتن ويكثر الهرج٤٠٤٨ذاك عند أوان ذهاب العلم٤٠٤٩يدرس الإسلام كما يدرس وشى الثوب حتى لا يدرى ما صيام ولا صلاة ولا نسك…٤٠٥٠يكون بين يدى الساعة أيام يرفع فيها العلم وينزل فيها الجهل ويكثر فيها…٤٠٥١إن من ورائكم أياما ينزل فيها الجهل ويرفع فيها العلم ويكثر فيها الهرج٤٠٥٢يتقارب الزمان وينقص العلم ويلقى الشح وتظهر الفتن ويكثر الهرج٤٠٥٣أن الأمانة نزلت في جذر قلوب الرجال٤٠٥٤إن الله عز وجل إذا أراد أن يهلك عبدا نزع منه الحياء فإذا نزع منه…٤٠٥٥لا تقوم الساعة حتى تكون عشر آيات طلوع الشمس من مغربها والدجال والدخان…٤٠٥٦بادروا بالأعمال ستا طلوع الشمس من مغربها والدخان ودابة الأرض والدجال…٤٠٥٧الآيات بعد المائتين٤٠٥٨أمتي على خمس طبقات فأربعون سنة أهل بر وتقوى ثم الذين يلونهم إلى عشرين…٤٠٥٩بين يدى الساعة مسخ وخسف وقذف٤٠٦٠يكون في آخر أمتي خسف ومسخ وقذف٤٠٦١يكون في أمتي - أو في هذه الأمة - مسخ وخسف وقذف٤٠٦٢يكون في أمتي خسف ومسخ وقذف٤٠٦٣ليؤمن هذا البيت جيش يغزونه حتى إذا كانوا ببيداء من الأرض خسف بأوسطهم…٤٠٦٤لا ينتهي الناس عن غزو هذا البيت حتى يغزو جيش حتى إذا كانوا بالبيداء -…٤٠٦٥إنهم يبعثون على نياتهم٤٠٦٦تخرج الدابة ومعها خاتم سليمان بن داود وعصا موسى بن عمران عليهما…٤٠٦٧تخرج الدابة من هذا الموضع٤٠٦٨لا تقوم الساعة حتى تطلع الشمس من مغربها فإذا طلعت ورآها الناس آمن من…٤٠٦٩أول الآيات خروجا طلوع الشمس من مغربها وخروج الدابة على الناس ضحى٤٠٧٠إن من قبل مغرب الشمس بابا مفتوحا عرضه سبعون سنة فلا يزال ذلك الباب…٤٠٧١الدجال أعور عين اليسرى جفال الشعر معه جنة ونار فناره جنة وجنته نار٤٠٧٢أن الدجال يخرج من أرض بالمشرق يقال لها خراسان يتبعه أقوام كأن وجوههم…٤٠٧٣وعلي بن محمد، قالا حدثنا وكيع، حدثنا إسماعيل بن أبي خالد، عن قيس بن…٤٠٧٤فإني والله ما قمت مقامي هذا لأمر ينفعكم لرغبة ولا لرهبة ولكن تميما…٤٠٧٥سمع النواس بن سمعان الكلابي، يقول ذكر رسول الله صلى الله عليه وسلم…
