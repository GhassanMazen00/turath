سنن ابن ماجه

كتاب الزهد

٢٤٢ حديثًا•صفحة ١ من ٢
٤١٠٠ليس الزهادة في الدنيا بتحريم الحلال ولا في إضاعة المال ولكن الزهادة…٤١٠١إذا رأيتم الرجل قد أعطي زهدا في الدنيا وقلة منطق فاقتربوا منه فإنه…٤١٠٢ازهد في الدنيا يحبك الله وازهد فيما في أيدي الناس يحبوك٤١٠٣إنك لعلك تدرك أموالا تقسم بين أقوام وإنما يكفيك من ذلك خادم ومركب في…٤١٠٤قال اشتكى سلمان فعاده سعد فرآه يبكي فقال له سعد ما يبكيك يا أخي أليس…٤١٠٥من كانت الدنيا همه فرق الله عليه أمره وجعل فقره بين عينيه ولم يأته من…٤١٠٦من جعل الهموم هما واحدا هم المعاد كفاه الله هم دنياه ومن تشعبت به…٤١٠٧يقول الله سبحانه يا ابن آدم تفرغ لعبادتي أملأ صدرك غنى وأسد فقرك وإن…٤١٠٨ما مثل الدنيا في الآخرة إلا مثل ما يجعل أحدكم إصبعه في اليم فلينظر بم…٤١٠٩ما أنا والدنيا إنما أنا والدنيا كراكب استظل تحت شجرة ثم راح وتركها٤١١٠أترون هذه هينة على صاحبها فوالذي نفسي بيده للدنيا أهون على الله من…٤١١١أترون هذه هانت على أهلها٤١١٢الدنيا ملعونة ملعون ما فيها إلا ذكر الله وما والاه أو عالما أو متعلما٤١١٣الدنيا سجن المؤمن وجنة الكافر٤١١٤يا عبد الله كن في الدنيا كأنك غريب أو كأنك عابر سبيل وعد نفسك من أهل…٤١١٥ألا أخبرك عن ملوك الجنة٤١١٦ألا أنبئكم بأهل الجنة كل ضعيف متضعف ألا أنبئكم بأهل النار كل عتل جواظ…٤١١٧إن أغبط الناس عندي مؤمن خفيف الحاذ ذو حظ من صلاة غامض في الناس لا…٤١١٨البذاذة من الإيمان٤١١٩ألا أنبئكم بخياركم٤١٢٠ما تقولون في هذا الرجل قالوا رأيك في . هذا نقول هذا من أشراف الناس…٤١٢١إن الله يحب عبده المؤمن الفقير المتعفف أبا العيال٤١٢٢يدخل فقراء المؤمنين الجنة قبل الأغنياء بنصف يوم خمسمائة عام٤١٢٣إن فقراء المهاجرين يدخلون الجنة قبل أغنيائهم بمقدار خمسمائة سنة٤١٢٤يا معشر الفقراء ألا أبشركم أن فقراء المؤمنين يدخلون الجنة قبل…٤١٢٥قال كان جعفر بن أبي طالب يحب المساكين ويجلس إليهم ويحدثهم ويحدثونه…٤١٢٦اللهم أحيني مسكينا وأمتني مسكينا واحشرني في زمرة المساكين٤١٢٧وكان، قارئ الأزد عن أبي الكنود، عن خباب، في قوله تعالى {ولا تطرد…٤١٢٨حدثنا يحيى بن حكيم، حدثنا أبو داود، حدثنا قيس بن الربيع، عن المقدام…٤١٢٩ويل للمكثرين إلا من قال بالمال هكذا وهكذا وهكذا وهكذا٤١٣٠الأكثرون هم الأسفلون يوم القيامة إلا من قال بالمال هكذا وهكذا وكسبه…٤١٣١الأكثرون هم الأسفلون إلا من قال هكذا وهكذا وهكذا٤١٣٢ما أحب أن أحدا عندي ذهبا فتأتي على ثالثة وعندي منه شىء إلا شىء أرصده…٤١٣٣اللهم من آمن بي وصدقني وعلم أن ما جئت به هو الحق من عندك - فأقلل ماله…٤١٣٤اللهم بارك فيها وفيمن بعث بها٤١٣٥تعس عبد الدينار وعبد الدرهم وعبد القطيفة وعبد الخميصة إن أعطي رضي وإن…٤١٣٦تعس عبد الدينار وعبد الدرهم وعبد الخميصة تعس وانتكس وإذا شيك فلا انتقش٤١٣٧ليس الغنى عن كثرة العرض ولكن الغنى غنى النفس٤١٣٨قد أفلح من هدي إلى الإسلام ورزق الكفاف وقنع به٤١٣٩اللهم اجعل رزق آل محمد قوتا٤١٤٠ما من غني ولا فقير إلا ود يوم القيامة أنه أتي من الدنيا قوتا٤١٤١من أصبح منكم معافى في جسده آمنا في سربه عنده قوت يومه فكأنما حيزت له…٤١٤٢انظروا إلى من هو أسفل منكم ولا تنظروا إلى من هو فوقكم فإنه أجدر أن لا…٤١٤٣إن الله لا ينظر إلى صوركم وأموالكم ولكن إنما ينظر إلى أعمالكم وقلوبكم٤١٤٤حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، وأبو أسامة عن هشام…٤١٤٥حدثنا أبو بكر بن أبي شيبة، حدثنا يزيد بن هارون، حدثنا محمد بن عمرو،…٤١٤٦يقول رأيت رسول الله صلى الله عليه وسلم يلتوي في اليوم من الجوع ما يجد…٤١٤٧والذي نفس محمد بيده ما أصبح عند آل محمد صاع حب ولا صاع تمر٤١٤٨ما أصبح في آل محمد إلا مد من طعام٤١٤٩حدثنا نصر بن علي، أخبرني أبي، عن شعبة، عن عبد الأكرم، - رجل من أهل…٤١٥٠الحمد لله ما دخل بطني طعام سخن منذ كذا وكذا٤١٥١قالت كان ضجاع رسول الله صلى الله عليه وسلم أدما حشوه ليف .٤١٥٢أن رسول الله صلى الله عليه وسلم أتى عليا وفاطمة وهما في خميل لهما -…٤١٥٣ما يبكيك يا ابن الخطاب٤١٥٤وإسحاق بن إبراهيم بن حبيب، قالا حدثنا محمد بن فضيل، عن مجالد، عن…٤١٥٥حدثنا محمد بن عبد الله بن نمير، وأبو كريب قالا حدثنا أبو أسامة، عن…٤١٥٦سمعه من، خالد بن عمير قال خطبنا عتبة بن غزوان على المنبر فقال لقد…٤١٥٧أنهم أصابهم جوع وهم سبعة قال فأعطاني النبي صلى الله عليه وسلم سبع…٤١٥٨حدثنا محمد بن يحيى بن أبي عمر العدني، حدثنا سفيان بن عيينة، عن محمد…٤١٥٩حدثنا عثمان بن أبي شيبة، حدثنا عبدة بن سليمان، عن هشام بن عروة، عن…٤١٦٠حدثنا أبو كريب، حدثنا أبو معاوية، عن الأعمش، عن أبي السفر، عن عبد…٤١٦١قال مر رسول الله صلى الله عليه وسلم بقبة على باب رجل من الأنصار فقال…٤١٦٢قال لقد رأيتني مع رسول الله صلى الله عليه وسلم بنيت بيتا يكنني من…٤١٦٣لا تتمنوا الموت٤١٦٤لو أنكم توكلتم على الله حق توكله لرزقكم كما يرزق الطير تغدو خماصا…٤١٦٥لا تيأسا من الرزق ما تهززت رءوسكما فإن الإنسان تلده أمه أحمر ليس عليه…٤١٦٦إن من قلب ابن آدم بكل واد شعبة فمن اتبع قلبه الشعب كلها لم يبال الله…٤١٦٧لا يموتن أحد منكم إلا وهو يحسن الظن بالله٤١٦٨المؤمن القوي خير وأحب إلى الله من المؤمن الضعيف وفي كل خير احرص على…٤١٦٩الكلمة الحكمة ضالة المؤمن حيثما وجدها فهو أحق بها٤١٧٠نعمتان مغبون فيهما كثير من الناس الصحة والفراغ٤١٧١إذا قمت في صلاتك فصل صلاة مودع ولا تكلم بكلام تعتذر منه وأجمع اليأس…٤١٧٢مثل الذي يجلس يسمع الحكمة ثم لا يحدث عن صاحبه إلا بشر ما يسمع كمثل…٤١٧٣لا يدخل الجنة من كان في قلبه مثقال حبة من خردل من كبر ولا يدخل النار…٤١٧٤يقول الله سبحانه الكبرياء ردائي والعظمة إزاري من نازعني واحدا منهما…٤١٧٥يقول الله سبحانه الكبرياء ردائي والعظمة إزاري فمن نازعني واحدا منهما…٤١٧٦من يتواضع لله سبحانه درجة يرفعه الله به درجة ومن يتكبر على الله درجة…٤١٧٧وسلم بن قتيبة، قالا حدثنا شعبة، عن علي بن زيد، عن أنس بن مالك، قال إن…٤١٧٨قال كان رسول الله صلى الله عليه وسلم يعود المريض ويشيع الجنازة ويجيب…٤١٧٩إن الله عز وجل أوحى إلى أن تواضعوا حتى لا يفخر أحد على أحد٤١٨٠وعبد الرحمن بن مهدي، قالا حدثنا شعبة، عن قتادة، عن عبد الله بن أبي…٤١٨١إن لكل دين خلقا وخلق الإسلام الحياء٤١٨٢إن لكل دين خلقا وإن خلق الإسلام الحياء٤١٨٣إن مما أدرك الناس من كلام النبوة الأولى إذا لم تستحي فاصنع ما شئت٤١٨٤الحياء من الإيمان والإيمان في الجنة والبذاء من الجفاء والجفاء في النار٤١٨٥ما كان الفحش في شىء قط إلا شانه ولا كان الحياء في شىء قط إلا زانه٤١٨٦من كظم غيظا وهو قادر على أن ينفذه دعاه الله على رءوس الخلائق يوم…٤١٨٧أتتكم وفود عبد القيس٤١٨٨إن فيك خصلتين يحبهما الله الحلم والحياء٤١٨٩ما من جرعة أعظم أجرا عند الله من جرعة غيظ كظمها عبد ابتغاء وجه الله٤١٩٠إني أرى ما لا ترون وأسمع ما لا تسمعون إن السماء أطت وحق لها أن تئط ما…٤١٩١لو تعلمون ما أعلم لضحكتم قليلا ولبكيتم كثيرا٤١٩٢أن عامر بن عبد الله بن الزبير، أخبره أن أباه أخبره أنه، لم يكن بين…٤١٩٣لا تكثروا الضحك فإن كثرة الضحك تميت القلب٤١٩٤حدثنا هناد بن السري، حدثنا أبو الأحوص، عن الأعمش، عن إبراهيم، عن…٤١٩٥يا إخواني لمثل هذا فأعدوا٤١٩٦ابكوا فإن لم تبكوا فتباكوا٤١٩٧ما من عبد مؤمن يخرج من عينيه دموع وإن كان مثل رأس الذباب من خشية الله…٤١٩٨لا يا بنت أبي بكر - أو يا بنت الصديق - ولكنه الرجل يصوم ويتصدق ويصلي…٤١٩٩إنما الأعمال كالوعاء إذا طاب أسفله طاب أعلاه وإذا فسد أسفله فسد أعلاه٤٢٠٠إن العبد إذا صلى في العلانية فأحسن وصلى في السر فأحسن - قال الله عز…٤٢٠١قاربوا وسددوا فإنه ليس أحد منكم بمنجيه عمله٤٢٠٢قال الله عز وجل أنا أغنى الشركاء عن الشرك فمن عمل لي عملا أشرك فيه…٤٢٠٣إذا جمع الله الأولين والآخرين ليوم القيامة ليوم لا ريب فيه نادى مناد…٤٢٠٤ألا أخبركم بما هو أخوف عليكم عندي من المسيح الدجال٤٢٠٥إن أخوف ما أتخوف على أمتي الإشراك بالله أما إني لست أقول يعبدون شمسا…٤٢٠٦من يسمع يسمع الله به ومن يراء يراء الله به٤٢٠٧من يراء يراء الله به ومن يسمع يسمع الله به٤٢٠٨لا حسد إلا في اثنتين رجل آتاه الله مالا فسلطه على هلكته في الحق ورجل…٤٢٠٩لا حسد إلا في اثنتين رجل آتاه الله القرآن فهو يقوم به آناء الليل…٤٢١٠الحسد يأكل الحسنات كما تأكل النار الحطب والصدقة تطفئ الخطيئة كما يطفئ…٤٢١١ما من ذنب أجدر أن يعجل الله لصاحبه العقوبة في الدنيا مع ما يدخر له في…٤٢١٢أسرع الخير ثوابا البر وصلة الرحم وأسرع الشر عقوبة البغى وقطيعة الرحم٤٢١٣حسب امرئ من الشر أن يحقر أخاه المسلم٤٢١٤إن الله أوحى إلى أن تواضعوا ولا يبغي بعضكم على بعض٤٢١٥لا يبلغ العبد أن يكون من المتقين حتى يدع ما لا بأس به حذرا لما به…٤٢١٦كل مخموم القلب صدوق اللسان٤٢١٧يا أبا هريرة كن ورعا تكن أعبد الناس وكن قنعا تكن أشكر الناس وأحب…٤٢١٨لا عقل كالتدبير ولا ورع كالكف ولا حسب كحسن الخلق٤٢١٩الحسب المال والكرم التقوى٤٢٢٠إني لأعرف كلمة - وقال عثمان آية - لو أخذ الناس كلهم بها لكفتهم٤٢٢١يوشك أن تعرفوا أهل الجنة من أهل النار٤٢٢٢إذا قال جيرانك إنك قد أحسنت فقد أحسنت وإذا قالوا إنك قد أسأت فقد أسأت٤٢٢٣إذا سمعت جيرانك يقولون قد أحسنت فقد أحسنت وإذا سمعتهم يقولون : قد…٤٢٢٤أهل الجنة من ملأ الله أذنيه من ثناء الناس خيرا وهو يسمع، وأهل النار…٤٢٢٥ذلك عاجل بشرى المؤمن٤٢٢٦لك أجران : أجر السر وأجر العلانية٤٢٢٧إنما الأعمال بالنيات ولكل امرئ ما نوى فمن كانت هجرته إلى الله وإلى…٤٢٢٨مثل هذه الأمة كمثل أربعة نفر : رجل آتاه الله مالا وعلما فهو يعمل…٤٢٢٩إنما يبعث الناس على نياتهم٤٢٣٠يحشر الناس على نياتهم٤٢٣١حدثنا أبو بشر، : بكر بن خلف وأبو بكر بن خلاد الباهلي قالا حدثنا يحيى…٤٢٣٢هذا ابن آدم وهذا أجله عند قفاه٤٢٣٣قلب الشيخ شاب في اثنتين : في حب الحياة وكثرة المال٤٢٣٤يهرم ابن آدم ويشب منه اثنتان : الحرص على المال والحرص على العمر٤٢٣٥لو أن لابن آدم واديين من مال لأحب أن يكون معهما ثالث ولا يملأ نفسه…٤٢٣٦أعمار أمتي ما بين الستين إلى السبعين وأقلهم من يجوز ذلك٤٢٣٧قالت : والذي ذهب بنفسه صلى الله عليه وسلم ما مات حتى كان أكثر صلاته…٤٢٣٨قالت : كانت عندي امرأة فدخل على النبي صلى الله عليه وسلم فقال : " من…٤٢٣٩يا حنظلة لو كنتم كما تكونون عندي لصافحتكم الملائكة على فرشكم - أو على…٤٢٤٠اكلفوا من العمل ما تطيقون فإن خير العمل أدومه وإن قل٤٢٤١يا أيها الناس عليكم بالقصد٤٢٤٢من أحسن في الإسلام لم يؤاخذ بما كان في الجاهلية ومن أساء أخذ بالأول…٤٢٤٣يا عائشة إياك ومحقرات الأعمال فإن لها من الله طالبا٤٢٤٤إن المؤمن إذا أذنب كانت نكتة سوداء في قلبه فإن تاب ونزع واستغفر صقل…٤٢٤٥لأعلمن أقواما من أمتي يأتون يوم القيامة بحسنات أمثال جبال تهامة بيضا…٤٢٤٦التقوى وحسن الخلق٤٢٤٧إن الله عز وجل أفرح بتوبة أحدكم منه بضالته إذا وجدها٤٢٤٨لو أخطأتم حتى تبلغ خطاياكم السماء ثم تبتم لتاب عليكم٤٢٤٩لله أفرح بتوبة عبده من رجل أضل راحلته بفلاة من الأرض فالتمسها حتى إذا…
