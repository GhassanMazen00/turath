سنن ابن ماجه

كتاب الوصايا

٢٤ حديثًا
٢٦٩٥وعلي بن محمد، قالا حدثنا أبو معاوية، - قال أبو بكر وعبد الله بن نمير…٢٦٩٦قال قلت لعبد الله بن أبي أوفى أوصى رسول الله صلى الله عليه وسلم بشىء…٢٦٩٧الصلاة وما ملكت أيمانكم٢٦٩٨الصلاة وما ملكت أيمانكم٢٦٩٩ما حق امرئ مسلم أن يبيت ليلتين وله شىء يوصي فيه إلا ووصيته مكتوبة عنده٢٧٠٠المحروم من حرم وصيته٢٧٠١من مات على وصية مات على سبيل وسنة ومات على تقى وشهادة ومات مغفورا له٢٧٠٢ما حق امرئ مسلم يبيت ليلتين وله شىء يوصي به إلا ووصيته مكتوبة عنده٢٧٠٣من فر من ميراث وارثه قطع الله ميراثه من الجنة يوم القيامة٢٧٠٤إن الرجل ليعمل بعمل أهل الخير سبعين سنة فإذا أوصى حاف في وصيته فيختم…٢٧٠٥من حضرته الوفاة فأوصى وكانت وصيته على كتاب الله كانت كفارة لما ترك من…٢٧٠٦نعم وأبيك لتنبأن أمك٢٧٠٧يقول الله عز وجل أنى تعجزني ابن آدم وقد خلقتك من مثل هذه فإذا بلغت…٢٧٠٨والحسين بن الحسن المروزي، وسهل، قالوا حدثنا سفيان بن عيينة، عن…٢٧٠٩إن الله تصدق عليكم عند وفاتكم بثلث أموالكم زيادة لكم في أعمالكم٢٧١٠يا ابن آدم اثنتان لم تكن لك واحدة منهما جعلت لك نصيبا من مالك حين…٢٧١١الثلث كبير - أو كثير -٢٧١٢إن الله قسم لكل وارث نصيبه من الميراث فلا يجوز لوارث وصية الولد…٢٧١٣إن الله قد أعطى كل ذي حق حقه فلا وصية لوارث٢٧١٤إن الله قد أعطى كل ذي حق حقه ألا لا وصية لوارث٢٧١٥قال قضى رسول الله صلى الله عليه وسلم بالدين قبل الوصية وأنتم تقرءونها…٢٧١٦أن رجلا، سأل رسول الله صلى الله عليه وسلم قال إن أبي مات وترك مالا…٢٧١٧أن رجلا، أتى النبي صلى الله عليه وسلم فقال إن أمي افتلتت نفسها ولم…٢٧١٨كل من مال يتيمك غير مسرف ولا متأثل مالا
