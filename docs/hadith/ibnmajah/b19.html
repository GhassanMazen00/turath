سنن ابن ماجه

كتاب العتق

٢٠ حديثًا
٢٥١٢وعلي بن محمد، قالا حدثنا وكيع، حدثنا إسماعيل بن أبي خالد، عن سلمة بن…٢٥١٣حدثنا هشام بن عمار، حدثنا سفيان بن عيينة، عن عمرو بن دينار، عن جابر…٢٥١٤المدبر من الثلث٢٥١٥أيما رجل ولدت أمته منه فهي معتقة عن دبر منه٢٥١٦قال ذكرت أم إبراهيم عند رسول الله صلى الله عليه وسلم . فقال " أعتقها…٢٥١٧وإسحاق بن منصور، قالا حدثنا عبد الرزاق، عن ابن جريج، أخبرني أبو…٢٥١٨ثلاثة كلهم حق على الله عونه الغازي في سبيل الله والمكاتب الذي يريد…٢٥١٩أيما عبد كوتب على مائة أوقية فأداها إلا عشر أوقيات فهو رقيق٢٥٢٠إذا كان لإحداكن مكاتب وكان عنده ما يؤدي فلتحتجب منه٢٥٢١وعلي بن محمد، قالا حدثنا وكيع، عن هشام بن عروة، عن أبيه، عن عائشة،…٢٥٢٢من أعتق امرأ مسلما كان فكاكه من النار يجزئ بكل عظم منه عظم منه ومن…٢٥٢٣أنفسها عند أهلها وأغلاها ثمنا٢٥٢٤من ملك ذا رحم محرم فهو حر٢٥٢٥من ملك ذا رحم محرم فهو حر٢٥٢٦قال أعتقتني أم سلمة واشترطت على أن أخدم النبي صلى الله عليه وسلم ما…٢٥٢٧من أعتق نصيبا له في مملوك أو شقصا فعليه خلاصه من ماله إن كان له مال…٢٥٢٩من أعتق عبدا وله مال فمال العبد له إلا أن يشترط السيد ماله فيكون له٢٥٣٠أيما رجل أعتق غلاما ولم يسم ماله فالمال له٢٥٣١نعلان أجاهد فيهما خير من أن أعتق ولد الزنا٢٥٣٢إن أعتقتهما فابدئي بالرجل قبل المرأة
