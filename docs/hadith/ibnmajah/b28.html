سنن ابن ماجه

كتاب الصيد

٥١ حديثًا
٣٢٠٠أن رسول الله صلى الله عليه وسلم أمر بقتل الكلاب ثم قال " ما لهم…٣٢٠١أن رسول الله صلى الله عليه وسلم أمر بقتل الكلاب ثم قال " ما لهم…٣٢٠٢قال أمر رسول الله صلى الله عليه وسلم بقتل الكلاب .٣٢٠٣حدثنا أبو طاهر، حدثنا ابن وهب، أخبرني يونس، عن ابن شهاب، عن سالم، عن…٣٢٠٤من اقتنى كلبا فإنه ينقص من عمله كل يوم قيراط إلا كلب حرث أو ماشية٣٢٠٥لولا أن الكلاب أمة من الأمم لأمرت بقتلها فاقتلوا منها الأسود البهيم…٣٢٠٦من اقتنى كلبا لا يغني عنه زرعا ولا ضرعا نقص من عمله كل يوم قيراط٣٢٠٧أما ما ذكرت أنكم في أرض أهل كتاب فلا تأكلوا في آنيتهم إلا أن لا تجدوا…٣٢٠٨إذا أرسلت كلابك المعلمة وذكرت اسم الله عليها فكل ما أمسكن عليك وإن…٣٢٠٩كلبهم وطائرهم يعني المجوس .٣٢١٠حدثنا عمرو بن عبد الله، حدثنا وكيع، عن سليمان بن المغيرة، عن حميد بن…٣٢١١كل ما ردت عليك قوسك٣٢١٢إذا رميت وخزقت فكل ما خزقت٣٢١٣إذا وجدت فيه سهمك ولم تجد فيه شيئا غيره فكله٣٢١٤ما أصبت بحده فكل وما أصبت بعرضه فهو وقيذ٣٢١٥لا تأكل إلا أن يخزق٣٢١٦ما قطع من البهيمة وهي حية فما قطع منها فهو ميتة٣٢١٧يكون في آخر الزمان قوم يجبون أسنمة الإبل ويقطعون أذناب الغنم ألا فما…٣٢١٨أحلت لنا ميتتان الحوت والجراد٣٢١٩أكثر جنود الله لا آكله ولا أحرمه٣٢٢٠سمع أنس بن مالك، يقول كن أزواج النبي صلى الله عليه وسلم يتهادين…٣٢٢١اللهم أهلك كباره واقتل صغاره وأفسد بيضه واقطع دابره وخذ بأفواهها عن…٣٢٢٢كلوه فإنه من صيد البحر٣٢٢٣وعبد الرحمن بن عبد الوهاب، قالا حدثنا أبو عامر العقدي، حدثنا إبراهيم…٣٢٢٤حدثنا محمد بن يحيى، حدثنا عبد الرزاق، أنبأنا معمر، عن الزهري، عن عبيد…٣٢٢٥إن نبيا من الأنبياء قرصته نملة فأمر بقرية النمل فأحرقت فأوحى الله عز…٣٢٢٦إنها لا تصيد صيدا ولا تنكأ عدوا ولكنها تكسر السن وتفقأ العين٣٢٢٧إنها لا تقتل الصيد ولا تنكي العدو ولكنها تفقأ العين وتكسر السن٣٢٢٨أن النبي صلى الله عليه وسلم أمرها بقتل الأوزاغ .٣٢٢٩من قتل وزغا في أول ضربة فله كذا وكذا حسنة ومن قتلها في الثانية فله…٣٢٣٠أن رسول الله صلى الله عليه وسلم قال للوزغ " الفويسقة " .٣٢٣١حدثنا أبو بكر بن أبي شيبة، حدثنا يونس بن محمد، عن جرير بن حازم، عن…٣٢٣٢حدثنا محمد بن الصباح، أنبأنا سفيان بن عيينة، عن الزهري، أخبرني أبو…٣٢٣٣أكل كل ذي ناب من السباع حرام٣٢٣٤حدثنا بكر بن خلف، حدثنا ابن أبي عدي، عن سعيد، عن علي بن الحكم، عن…٣٢٣٥ومن يأكل الثعلب٣٢٣٦ومحمد بن الصباح، قالا حدثنا عبد الله بن رجاء المكي، عن إسماعيل بن…٣٢٣٧ومن يأكل الضبع٣٢٣٨إن أمة في بني إسرائيل مسخت دواب في الأرض وإني لا أدري لعلها هي٣٢٣٩حدثنا أبو إسحاق الهروي، إبراهيم بن عبد الله بن حاتم حدثنا إسماعيل ابن…٣٢٤٠بلغني أنه أمة مسخت٣٢٤١لا ولكنه لم يكن بأرضي فأجدني أعافه٣٢٤٢قال قال رسول الله صلى الله عليه وسلم " لا أحرم " . يعني الضب .٣٢٤٣وعبد الرحمن بن مهدي، قالا حدثنا شعبة، عن هشام بن زيد، عن أنس بن مالك،…٣٢٤٤أنه مر على النبي صلى الله عليه وسلم بأرنبين معلقهما فقال يا رسول الله…٣٢٤٥لا آكله ولا أحرمه٣٢٤٦البحر الطهور ماؤه الحل ميتته٣٢٤٧ما ألقى البحر أو جزر عنه فكلوه وما مات فيه فطفا فلا تأكلوه٣٢٤٨قال من يأكل الغراب وقد سماه رسول الله صلى الله عليه وسلم " فاسقا " .…٣٢٤٩الحية فاسقة والعقرب فاسقة والفأرة فاسقة
