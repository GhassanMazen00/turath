سنن ابن ماجه

كتاب الحدود

٨٢ حديثًا
٢٥٣٣لا يحل دم امرئ مسلم إلا في إحدى ثلاث رجل زنى وهو محصن فرجم أو رجل قتل…٢٥٣٤لا يحل دم امرئ مسلم يشهد أن لا إله إلا الله وأني رسول الله إلا أحد…٢٥٣٥قال قال رسول الله صلى الله عليه وسلم ( من بدل دينه فاقتلوه ) .٢٥٣٦لا يقبل الله من مشرك أشرك بعد ما أسلم عملا حتى يفارق المشركين إلى…٢٥٣٧إقامة حد من حدود الله خير من مطر أربعين ليلة في بلاد الله عز وجل٢٥٣٨حد يعمل به في الأرض خير لأهل الأرض من أن يمطروا أربعين صباحا٢٥٣٩من جحد آية من القرآن فقد حل ضرب عنقه ومن قال لا إله إلا الله وحده لا…٢٥٤٠أقيموا حدود الله في القريب والبعيد ولا تأخذكم في الله لومة لائم٢٥٤١وعلي بن محمد، قالا حدثنا وكيع، عن سفيان، عن عبد الملك بن عمير، قال…٢٥٤٢بين أظهركم .٢٥٤٣حدثنا علي بن محمد، حدثنا عبد الله بن نمير، وأبو معاوية وأبو أسامة…٢٥٤٤من ستر مسلما ستره الله في الدنيا والآخرة٢٥٤٥ادفعوا الحدود ما وجدتم له مدفعا٢٥٤٦من ستر عورة أخيه المسلم ستر الله عورته يوم القيامة ومن كشف عورة أخيه…٢٥٤٧أتشفع في حد من حدود الله٢٥٤٨حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، حدثنا محمد بن…٢٥٤٩وهشام بن عمار، ومحمد بن الصباح، قالوا حدثنا سفيان بن عيينة، عن…٢٥٥٠خذوا عني خذوا عني قد جعل الله لهن سبيلا البكر بالبكر جلد مائة وتغريب…٢٥٥١قال أتي النعمان بن بشير برجل غشى جارية امرأته فقال لا أقضي فيها إلا…٢٥٥٢أن رسول الله صلى الله عليه وسلم رفع إليه رجل وطئ جارية امرأته فلم…٢٥٥٣ومحمد بن الصباح، قالا حدثنا سفيان بن عيينة، عن الزهري، عن عبيد الله…٢٥٥٤قال جاء ماعز بن مالك إلى النبي صلى الله عليه وسلم . فقال إني قد زنيت…٢٥٥٥أن امرأة، أتت النبي صلى الله عليه وسلم فاعترفت بالزنا فأمر بها فشكت…٢٥٥٦حدثنا علي بن محمد، حدثنا عبد الله بن نمير، عن عبيد الله بن عمر، عن…٢٥٥٧أن النبي صلى الله عليه وسلم رجم يهوديا ويهودية .٢٥٥٨هكذا تجدون في كتابكم حد الزاني٢٥٥٩لو كنت راجما أحدا بغير بينة لرجمت فلانة فقد ظهر فيها الريبة في منطقها…٢٥٦٠لو كنت راجما أحدا بغير بينة لرجمتها٢٥٦١من وجدتموه يعمل عمل قوم لوط فاقتلوا الفاعل والمفعول به٢٥٦٢ارجموا الأعلى والأسفل ارجموهما جميعا٢٥٦٣إن أخوف ما أخاف على أمتي عمل قوم لوط٢٥٦٤من وقع على ذات محرم فاقتلوه ومن وقع على بهيمة فاقتلوه واقتلوا البهيمة٢٥٦٥اجلدها فإن زنت فاجلدها فإن زنت فاجلدها٢٥٦٦إذا زنت الأمة فاجلدوها فإن زنت فاجلدوها فإن زنت فاجلدوها فإن زنت…٢٥٦٧قالت لما نزل عذري قام رسول الله صلى الله عليه وسلم على المنبر فذكر…٢٥٦٨إذا قال الرجل للرجل يا مخنث فاجلدوه عشرين وإذا قال الرجل للرجل يا…٢٥٦٩قال قال علي بن أبي طالب ما كنت أدي من أقمت عليه الحد إلا شارب الخمر…٢٥٧٠قال كان رسول الله صلى الله عليه وسلم يضرب في الخمر بالنعال والجريد .٢٥٧١قال لما جيء بالوليد بن عقبة إلى عثمان قد شهدوا عليه قال لعلي دونك ابن…٢٥٧٢إذا سكر فاجلدوه فإن عاد فاجلدوه فإن عاد فاجلدوه٢٥٧٣إذا شربوا الخمر فاجلدوهم ثم إذا شربوا فاجلدوهم ثم إذا شربوا فاجلدوهم…٢٥٧٤اجلدوه ضرب مائة سوط٢٥٧٥من حمل علينا السلاح فليس منا٢٥٧٦من حمل علينا السلاح فليس منا٢٥٧٧من شهر علينا السلاح فليس منا٢٥٧٨لو خرجتم إلى ذود لنا فشربتم من ألبانها وأبوالها٢٥٧٩ومحمد بن المثنى، قالا حدثنا إبراهيم بن أبي الوزير، حدثنا الدراوردي،…٢٥٨٠من قتل دون ماله فهو شهيد٢٥٨١من أتي عند ماله فقوتل فقاتل فقتل فهو شهيد٢٥٨٢من أريد ماله ظلما فقتل فهو شهيد٢٥٨٣لعن الله السارق يسرق البيضة فتقطع يده ويسرق الحبل فتقطع يده٢٥٨٤قال قطع النبي صلى الله عليه وسلم في مجن قيمته ثلاثة دراهم .٢٥٨٥لا تقطع اليد إلا في ربع دينار فصاعدا٢٥٨٦تقطع يد السارق في ثمن المجن٢٥٨٧في العنق فقال السنة قطع رسول الله صلى الله عليه وسلم يد رجل ثم علقها…٢٥٨٨أن عمرو بن سمرة بن حبيب بن عبد شمس، جاء إلى رسول الله صلى الله عليه…٢٥٨٩إذا سرق العبد فبيعوه ولو بنش٢٥٩٠مال الله عز وجل سرق بعضه بعضا٢٥٩١لا يقطع الخائن ولا المنتهب ولا المختلس٢٥٩٢ليس على المختلس قطع٢٥٩٣لا قطع في ثمر ولا كثر٢٥٩٤لا قطع في ثمر ولا كثر٢٥٩٥فهلا قبل أن تأتيني به٢٥٩٦ما أخذ في أكمامه فاحتمل فثمنه ومثله معه وما كان في الجران ففيه القطع…٢٥٩٧- مولى أبي ذر - يذكر أن أبا أمية، حدثه أن رسول الله صلى الله عليه…٢٥٩٨وأيوب بن محمد الوزان، وعبد الله بن سعيد، قالوا حدثنا معمر بن سليمان،…٢٥٩٩لا تقام الحدود في المساجد٢٦٠٠أنه سمع عمرو بن شعيب، يحدث عن أبيه، عن جده، أن رسول الله صلى الله…٢٦٠١لا يجلد أحد فوق عشر جلدات إلا في حد من حدود الله٢٦٠٢لا تعزروا فوق عشرة أسواط٢٦٠٣من أصاب منكم حدا فعجلت له عقوبته فهو كفارته وإلا فأمره إلى الله٢٦٠٤من أصاب في الدنيا ذنبا فعوقب به فالله أعدل من أن يثني عقوبته على عبده…٢٦٠٥ومحمد بن عبيد المديني أبو عبيد، قالا حدثنا عبد العزيز بن محمد…٢٦٠٦كفى بالسيف شاهدا٢٦٠٧قال مر بي خالي - سماه هشيم في حديثه الحارث بن عمرو - وقد عقد له النبي…٢٦٠٨قال بعثني رسول الله صلى الله عليه وسلم إلى رجل تزوج امرأة أبيه أن…٢٦٠٩من انتسب إلى غير أبيه أو تولى غير مواليه فعليه لعنة الله والملائكة…٢٦١٠من ادعى إلى غير أبيه وهو يعلم أنه غير أبيه فالجنة عليه حرام٢٦١١من ادعى إلى غير أبيه لم يرح رائحة الجنة وإن ريحها ليوجد من مسيرة…٢٦١٢نحن بنو النضر بن كنانة لا نقفو أمنا ولا ننتفي من أبينا٢٦١٣لا آذن لك ولا كرامة ولا نعمة عين كذبت أى عدو الله لقد رزقك الله طيبا…٢٦١٤أخرجوهم من بيوتكم
