سنن ابن ماجه

كتاب الصدقات

٤٦ حديثًا
٢٣٩٠لا تعد في صدقتك٢٣٩١مثل الذي يتصدق ثم يرجع في صدقته مثل الكلب يقيء ثم يرجع فيأكل قيئه٢٣٩٢حدثنا تميم بن المنتصر الواسطي، حدثنا إسحاق بن يوسف، عن شريك، عن هشام…٢٣٩٣أنه حمل على فرس يقال له غمر أو غمرة فرأى مهرا أو مهرة من أفلائها يباع…٢٣٩٤آجرك الله ورد عليك الميراث٢٣٩٥وجبت صدقتك ورجعت إليك حديقتك٢٣٩٦إن شئت حبست أصلها وتصدقت بها٢٣٩٧احبس أصلها وسبل ثمرتها٢٣٩٨العارية مؤداة والمنحة مردودة٢٣٩٩العارية مؤداة والمنحة مردودة٢٤٠٠على اليد ما أخذت حتى تؤديه٢٤٠١من أودع وديعة فلا ضمان عليه٢٤٠٢قال قدم جلب فأعطاني النبي صلى الله عليه وسلم دينارا فذكر نحوه .٢٤٠٣الظلم مطل الغني وإذا أتبع أحدكم على مليء فليتبع٢٤٠٤مطل الغني ظلم وإذا أحلت على مليء فاتبعه٢٤٠٥الزعيم غارم والدين مقضي٢٤٠٦أن رجلا، لزم غريما له بعشرة دنانير على عهد رسول الله صلى الله عليه…٢٤٠٧صلوا على صاحبكم فإن عليه دينا٢٤٠٨ما من مسلم يدان دينا يعلم الله منه أنه يريد أداءه إلا أداه الله عنه…٢٤٠٩إن الله مع الدائن حتى يقضي دينه ما لم يكن فيما يكره الله٢٤١٠أيما رجل يدين دينا وهو مجمع أن لا يوفيه إياه لقي الله سارقا٢٤١١من أخذ أموال الناس يريد إتلافها أتلفه الله٢٤١٢من فارق الروح الجسد وهو بريء من ثلاث دخل الجنة من الكبر والغلول والدين٢٤١٣نفس المؤمن معلقة بدينه حتى يقضى عنه٢٤١٤من مات وعليه دينار أو درهم قضي من حسناته ليس ثم دينار ولا درهم٢٤١٥هل ترك لدينه من قضاء٢٤١٦من ترك مالا فلورثته ومن ترك دينا أو ضياعا فعلى وإلى وأنا أولى…٢٤١٧من يسر على معسر يسر الله عليه في الدنيا والآخرة٢٤١٨من أنظر معسرا كان له بكل يوم صدقة ومن أنظره بعد حله كان له مثله في كل…٢٤١٩من أحب أن يظله الله في ظله - فلينظر معسرا أو ليضع له٢٤٢٠أن رجلا مات فقيل له ما عملت - فإما ذكر أو ذكر - قال إني كنت أتجوز في…٢٤٢١من طلب حقا فليطلبه في عفاف واف أو غير واف٢٤٢٢خذ حقك في عفاف واف أو غير واف٢٤٢٣إن خيركم - أو من خيركم - أحاسنكم قضاء٢٤٢٤بارك الله لك في أهلك ومالك إنما جزاء السلف الوفاء والحمد٢٤٢٥مه إن صاحب الدين له سلطان على صاحبه حتى يقضيه٢٤٢٦هلا مع صاحب الحق كنتم٢٤٢٧لى الواجد يحل عرضه وعقوبته٢٤٢٨قال أتيت النبي صلى الله عليه وسلم بغريم لي فقال لي " الزمه " . ثم مر…٢٤٢٩ويحيى بن حكيم، قالا حدثنا عثمان بن عمر، أنبأنا يونس بن يزيد، عن…٢٤٣٠ما من مسلم يقرض مسلما قرضا مرتين إلا كان كصدقتها مرة٢٤٣١رأيت ليلة أسري بي على باب الجنة مكتوبا الصدقة بعشر أمثالها والقرض…٢٤٣٢إذا أقرض أحدكم قرضا فأهدى له أو حمله على الدابة فلا يركبها ولا يقبله…٢٤٣٣إن أخاك محتبس بدينه فاقض عنه٢٤٣٤جد له فأوفه الذي له٢٥٢٨من أعتق شركا له في عبد أقيم عليه بقيمة عدل فأعطى شركاءه حصصهم إن كان…
