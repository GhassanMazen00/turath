سنن ابن ماجه

كتاب الطهارة وسننها

٤٠٠ حديثًا•صفحة ٢ من ٣
٤١٧قال كان رسول الله صلى الله عليه وسلم يتوضأ ثلاثا ثلاثا .٤١٨وعلي بن محمد، قالا حدثنا وكيع، عن سفيان، عن عبد الله بن محمد بن عقيل،…٤١٩هذا وضوء من لا يقبل الله منه صلاة إلا به٤٢٠هذا وظيفة الوضوء٤٢١إن للوضوء شيطانا يقال له ولهان فاتقوا وسواس الماء٤٢٢هذا الوضوء فمن زاد على هذا فقد أساء وتعدى أو ظلم٤٢٣سمع كريبا، يقول سمعت ابن عباس، يقول بت عند خالتي ميمونة فقام النبي…٤٢٤لا تسرف لا تسرف٤٢٥أن رسول الله صلى الله عليه وسلم مر بسعد وهو يتوضأ فقال " ما هذا السرف…٤٢٦حدثنا أحمد بن عبدة، حدثنا حماد بن زيد، حدثنا موسى بن سالم أبو جهضم،…٤٢٧ألا أدلكم على ما يكفر الله به الخطايا ويزيد به في الحسنات٤٢٨كفارات الخطايا إسباغ الوضوء على المكاره وإعمال الأقدام إلى المساجد…٤٢٩قال رأيت رسول الله صلى الله عليه وسلم يخلل لحيته .٤٣٠أن رسول الله صلى الله عليه وسلم توضأ فخلل لحيته .٤٣١قال كان رسول الله صلى الله عليه وسلم إذا توضأ خلل لحيته وفرج أصابعه…٤٣٢قال كان رسول الله صلى الله عليه وسلم إذا توضأ عرك عارضيه بعض العرك ثم…٤٣٣قال رأيت رسول الله صلى الله عليه وسلم توضأ فخلل لحيته .٤٣٤وحرملة بن يحيى، قالا أخبرنا محمد بن إدريس الشافعي، قال أنبأنا مالك بن…٤٣٥قال رأيت رسول الله صلى الله عليه وسلم توضأ فمسح رأسه مرة .٤٣٦أن رسول الله صلى الله عليه وسلم مسح رأسه مرة .٤٣٧قال رأيت رسول الله صلى الله عليه وسلم توضأ فمسح رأسه مرة .٤٣٨وعلي بن محمد، قالا حدثنا وكيع، عن سفيان، عن عبد الله بن محمد بن عقيل،…٤٣٩أن رسول الله صلى الله عليه وسلم مسح أذنيه داخلهما بالسبابتين وخالف…٤٤٠أن النبي صلى الله عليه وسلم توضأ فمسح ظاهر أذنيه وباطنهما .٤٤١وعلي بن محمد، قالا حدثنا وكيع، عن الحسن بن صالح، عن عبد الله بن محمد…٤٤٢أن رسول الله صلى الله عليه وسلم توضأ فمسح برأسه وأذنيه ظاهرهما…٤٤٣الأذنان من الرأس٤٤٤الأذنان من الرأس٤٤٥الأذنان من الرأس٤٤٦فذكر نحوه .٤٤٧إذا قمت إلى الصلاة فأسبغ الوضوء واجعل الماء بين أصابع يديك ورجليك٤٤٨أسبغ الوضوء وخلل بين الأصابع٤٤٩أن رسول الله صلى الله عليه وسلم كان إذا توضأ حرك خاتمه .٤٥٠ويل للأعقاب من النار أسبغوا الوضوء٤٥١ويل للأعقاب من النار٤٥٢ويل للعراقيب من النار٤٥٣ويل للأعقاب من النار٤٥٤ويل للعراقيب من النار٤٥٥أتموا الوضوء ويل للأعقاب من النار٤٥٦قال رأيت عليا توضأ فغسل قدميه إلى الكعبين ثم قال أردت أن أريكم طهور…٤٥٧أن رسول الله صلى الله عليه وسلم توضأ فغسل رجليه ثلاثا ثلاثا .٤٥٨- تعني حديثها الذي ذكرت أن رسول الله صلى الله عليه وسلم توضأ وغسل…٤٥٩من أتم الوضوء كما أمره الله فالصلوات المكتوبات كفارات لما بينهن٤٦٠إنها لا تتم صلاة لأحد حتى يسبغ الوضوء كما أمره الله تعالى يغسل وجهه…٤٦١أنه رأى رسول الله صلى الله عليه وسلم توضأ ثم أخذ كفا من ماء فنضح به…٤٦٢علمني جبرائيل الوضوء وأمرني أن أنضح تحت ثوبي لما يخرج من البول بعد…٤٦٣إذا توضأت فانتضح٤٦٤قال توضأ رسول الله صلى الله عليه وسلم فنضح فرجه .٤٦٥أن أبا مرة، مولى عقيل حدثه أن أم هانئ بنت أبي طالب حدثته أنه، لما كان…٤٦٦حدثنا علي بن محمد، حدثنا وكيع، حدثنا ابن أبي ليلى، عن محمد بن عبد…٤٦٧وعلي بن محمد، قالا حدثنا وكيع، حدثنا الأعمش، عن سالم بن أبي الجعد، عن…٤٦٨وأحمد بن الأزهر، قالا حدثنا مروان بن محمد، حدثنا يزيد بن السمط، حدثنا…٤٦٩من توضأ فأحسن الوضوء ثم قال ثلاث مرات أشهد أن لا إله إلا الله وحده لا…٤٧٠ما من مسلم يتوضأ فيحسن الوضوء ثم يقول أشهد أن لا إله إلا الله وأشهد…٤٧١حدثنا أبو بكر بن أبي شيبة، حدثنا أحمد بن عبد الله، عن عبد العزيز بن…٤٧٢أنه كان لها مخضب من صفر قالت فكنت أرجل رأس رسول الله صلى الله عليه…٤٧٣وعلي بن محمد، قالا حدثنا وكيع، عن شريك، عن إبراهيم بن جرير، عن أبي…٤٧٤وعلي بن محمد، قالا حدثنا وكيع، حدثنا الأعمش، عن إبراهيم، عن الأسود،…٤٧٥أن رسول الله صلى الله عليه وسلم نام حتى نفخ ثم قام فصلى .٤٧٦قال كان نومه ذلك وهو جالس . يعني النبي صلى الله عليه وسلم .٤٧٧العين وكاء السه فمن نام فليتوضأ٤٧٨حدثنا أبو بكر بن أبي شيبة، حدثنا سفيان بن عيينة، عن عاصم، عن زر، عن…٤٧٩إذا مس أحدكم ذكره فليتوضأ٤٨٠إذا مس أحدكم ذكره فعليه الوضوء٤٨١من مس فرجه فليتوضأ٤٨٢من مس فرجه فليتوضأ٤٨٣ليس فيه وضوء إنما هو منك٤٨٤إنما هو حذية منك٤٨٥توضئوا مما غيرت النار٤٨٦توضئوا مما مست النار٤٨٧توضئوا مما مست النار٤٨٨قال أكل النبي صلى الله عليه وسلم كتفا ثم مسح يديه بمسح كان تحته ثم…٤٨٩وعمرو بن دينار، وعبد الله بن محمد بن عقيل، عن جابر بن عبد الله، قال…٤٩٠على أبي بمثل ذلك .٤٩١قالت أتي رسول الله صلى الله عليه وسلم بكتف شاة فأكل منه وصلى ولم يمس…٤٩٢حدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن يحيى بن سعيد، عن…٤٩٣أن رسول الله صلى الله عليه وسلم أكل كتف شاة فمضمض وغسل يديه وصلى .٤٩٤حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن إدريس، وأبو معاوية قالا…٤٩٥وإسرائيل، عن أشعث بن أبي الشعثاء، عن جعفر بن أبي ثور، عن جابر بن…٤٩٦لا توضئوا من ألبان الغنم وتوضئوا من ألبان الإبل٤٩٧توضئوا من لحوم الإبل ولا توضئوا من لحوم الغنم وتوضئوا من ألبان الإبل…٤٩٨مضمضوا من اللبن فإن له دسما٤٩٩إذا شربتم اللبن فمضمضوا فإن له دسما٥٠٠مضمضوا من اللبن فإن له دسما٥٠١قال حلب رسول الله صلى الله عليه وسلم شاة وشرب من لبنها ثم دعا بماء…٥٠٢وعلي بن محمد، قالا حدثنا وكيع، حدثنا الأعمش، عن حبيب بن أبي ثابت، عن…٥٠٣أن رسول الله صلى الله عليه وسلم كان يتوضأ ثم يقبل ويصلي ولا يتوضأ…٥٠٤فيه الوضوء وفي المني الغسل٥٠٥إذا وجد أحدكم ذلك فلينضح فرجه - يعني ليغسله - ويتوضأ٥٠٦إنما يجزيك من ذلك الوضوء٥٠٧حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن بشر، حدثنا مسعر، عن مصعب بن…٥٠٨حدثنا علي بن محمد، حدثنا وكيع، سمعت سفيان، يقول لزائدة بن قدامة يا…٥٠٩حدثنا سويد بن سعيد، حدثنا شريك، عن عمرو بن عامر، عن أنس بن مالك، قال…٥١٠وعلي بن محمد، قالا حدثنا وكيع، عن سفيان، عن محارب بن دثار، عن سليمان…٥١١حدثنا إسماعيل بن توبة، حدثنا زياد بن عبد الله، حدثنا الفضل بن مبشر،…٥١٢من توضأ على كل طهر فله عشر حسنات٥١٣لا حتى يجد ريحا أو يسمع صوتا٥١٤لا ينصرف حتى يسمع صوتا أو يجد ريحا٥١٥لا وضوء إلا من صوت أو ريح٥١٦لا وضوء إلا من ريح أو سماع٥١٧إذا بلغ الماء قلتين لم ينجسه شىء٥١٨إذا كان الماء قلتين أو ثلاثا لم ينجسه شىء٥١٩لها ما حملت في بطونها ولنا ما غبر طهور٥٢٠إن الماء لا ينجسه شىء٥٢١إن الماء لا ينجسه شىء إلا ما غلب على ريحه وطعمه ولونه٥٢٢إنما ينضح من بول الذكر ويغسل من بول الأنثى٥٢٣وعلي بن محمد، قالا حدثنا وكيع، حدثنا هشام بن عروة، عن أبيه، عن عائشة،…٥٢٤ومحمد بن الصباح، قالا حدثنا سفيان بن عيينة، عن الزهري، عن عبيد الله…٥٢٥ينضح بول الغلام ويغسل بول الجارية٥٢٦رشه فإنه يغسل بول الجارية ويرش على بول الغلام٥٢٧بول الغلام ينضح وبول الجارية يغسل٥٢٨أن أعرابيا، بال في المسجد فوثب إليه بعض القوم فقال رسول الله صلى الله…٥٢٩لقد احتظرت واسعا٥٣٠لقد حظرت واسعا ويحك - أو ويلك -٥٣١لإبراهيم بن عبد الرحمن بن عوف أنها سألت أم سلمة زوج النبي صلى الله…٥٣٢الأرض يطهر بعضها بعضا٥٣٣فبعدها طريق أنظف منها٥٣٤أين كنت يا أبا هريرة٥٣٥حدثنا علي بن محمد، حدثنا وكيع، ح وحدثنا إسحاق بن منصور، أنبأنا يحيى…٥٣٦حدثنا أبو بكر بن أبي شيبة، حدثنا عبدة بن سليمان، عن عمرو بن ميمون،…٥٣٧قالت ربما فركته من ثوب رسول الله صلى الله عليه وسلم بيدي .٥٣٨وعلي بن محمد، حدثنا أبو معاوية، عن الأعمش، عن إبراهيم، عن همام بن…٥٣٩قالت لقد رأيتني أجده في ثوب رسول الله صلى الله عليه وسلم فأحته عنه .٥٤٠أنه سأل أخته أم حبيبة زوج النبي صلى الله عليه وسلم هل كان رسول الله…٥٤١نعم أصلي فيه وفيه٥٤٢نعم إلا أن يرى فيه شيئا فيغسله٥٤٣قال بال جرير بن عبد الله ثم توضأ ومسح على خفيه فقيل له أتفعل هذا قال…٥٤٤وعلي بن محمد، قالا حدثنا وكيع، ح وحدثنا أبو همام الوليد بن شجاع بن…٥٤٥حدثنا محمد بن رمح، أنبأنا الليث بن سعد، عن يحيى بن سعيد، عن سعد بن…٥٤٦حدثنا عمران بن موسى الليثي، حدثنا محمد بن سواء، حدثنا سعيد بن أبي…٥٤٧حدثنا أبو مصعب المدني، حدثنا عبد المهيمن بن العباس بن سهل بن سعد…٥٤٨قال كنت مع رسول الله صلى الله عليه وسلم في سفر فقال " هل من ماء " .…٥٤٩أن النجاشي، أهدى للنبي صلى الله عليه وسلم خفين أسودين ساذجين فلبسهما…٥٥٠أن رسول الله صلى الله عليه وسلم مسح أعلى الخف وأسفله .٥٥١إنما أمرت بالمسح٥٥٢حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، عن الحكم، قال…٥٥٣قال جعل رسول الله صلى الله عليه وسلم للمسافر ثلاثا ولو مضى السائل على…٥٥٤ثلاثة أيام - أحسبه قال - ولياليهن للمسافر في المسح على الخفين٥٥٥للمسافر ثلاثة أيام ولياليهن وللمقيم يوم وليلة٥٥٦وبشر بن هلال الصواف، قالا حدثنا عبد الوهاب بن عبد المجيد، قال حدثنا…٥٥٧وعمرو بن سواد المصريان، قالا حدثنا عبد الله بن وهب، أنبأنا يحيى بن…٥٥٨أنه قدم على عمر بن الخطاب من مصر فقال منذ كم لم تنزع خفيك قال من…٥٥٩أن رسول الله صلى الله عليه وسلم توضأ ومسح على الجوربين والنعلين .٥٦٠وبشر بن آدم، قالا حدثنا عيسى بن يونس، عن عيسى بن سنان، عن الضحاك بن…٥٦١أن رسول الله صلى الله عليه وسلم مسح على الخفين والخمار .٥٦٢قال رأيت رسول الله صلى الله عليه وسلم يمسح على الخفين والعمامة .٥٦٣مولى زيد بن صوحان قال كنت مع سلمان فرأى رجلا ينزع خفيه للوضوء فقال له…٥٦٤قال رأيت رسول الله صلى الله عليه وسلم توضأ وعليه عمامة قطرية فأدخل…٥٦٥حدثنا محمد بن رمح، حدثنا الليث بن سعد، عن ابن شهاب، عن عبيد الله بن…٥٦٦حدثنا محمد بن أبي عمر العدني، حدثنا سفيان بن عيينة، عن عمرو، عن…
