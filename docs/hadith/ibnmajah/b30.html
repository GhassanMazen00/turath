سنن ابن ماجه

كتاب الأشربة

٦٥ حديثًا
٣٣٧١لا تشرب الخمر فإنها مفتاح كل شر٣٣٧٢إياك والخمر فإن خطيئتها تفرع الخطايا كما أن شجرتها تفرع الشجر٣٣٧٣من شرب الخمر في الدنيا لم يشربها في الآخرة إلا أن يتوب٣٣٧٤من شرب الخمر في الدنيا لم يشربها في الآخرة٣٣٧٥مدمن الخمر كعابد وثن٣٣٧٦لا يدخل الجنة مدمن خمر٣٣٧٧من شرب الخمر وسكر لم تقبل له صلاة أربعين صباحا وإن مات دخل النار فإن…٣٣٧٨الخمر من هاتين الشجرتين النخلة والعنبة٣٣٧٩إن من الحنطة خمرا ومن الشعير خمرا ومن الزبيب خمرا ومن التمر خمرا ومن…٣٣٨٠لعنت الخمر على عشرة أوجه بعينها وعاصرها ومعتصرها وبائعها ومبتاعها…٣٣٨١. حتى عد عشرة من هذا الضرب .٣٣٨٢وعلي بن محمد، قالا حدثنا أبو معاوية، حدثنا الأعمش، عن مسلم، عن مسروق،…٣٣٨٣لعن الله اليهود حرمت عليهم الشحوم فجملوها فباعوها٣٣٨٤لا تذهب الليالي والأيام حتى تشرب طائفة من أمتي الخمر يسمونها بغير…٣٣٨٥يشرب ناس من أمتي الخمر باسم يسمونها إياه٣٣٨٦كل شراب أسكر فهو حرام٣٣٨٧قال قال رسول الله صلى الله عليه وسلم " كل مسكر حرام " .٣٣٨٨أن رسول الله صلى الله عليه وسلم قال " كل مسكر حرام " . قال ابن ماجه…٣٣٨٩كل مسكر حرام على كل مؤمن٣٣٩٠كل مسكر خمر وكل خمر حرام٣٣٩١قال قال رسول الله صلى الله عليه وسلم " كل مسكر حرام " .٣٣٩٢كل مسكر حرام وما أسكر كثيره فقليله حرام٣٣٩٣ما أسكر كثيره فقليله حرام٣٣٩٤ما أسكر كثيره فقليله حرام٣٣٩٥حدثنا محمد بن رمح، أنبأنا الليث بن سعد، عن أبي الزبير، عن جابر بن عبد…٣٣٩٦لا تنبذوا التمر والبسر جميعا وانبذوا كل واحد منهما على حدته٣٣٩٧لا تجمعوا بين الرطب والزهو ولا بين الزبيب والتمر وانبذوا كل واحد…٣٣٩٨حدثنا أبو بكر بن أبي شيبة، حدثنا أبو معاوية، ح وحدثنا محمد بن عبد…٣٣٩٩قال كان ينبذ لرسول الله صلى الله عليه وسلم فيشربه يومه ذلك والغد…٣٤٠٠قال كان ينبذ لرسول الله صلى الله عليه وسلم في تور من حجارة .٣٤٠١قال نهى رسول الله صلى الله عليه وسلم أن ينبذ في النقير والمزفت…٣٤٠٢قال نهى رسول الله صلى الله عليه وسلم أن ينبذ في المزفت والقرع .٣٤٠٣حدثنا نصر بن علي، حدثنا أبي، عن المثنى بن سعيد، عن أبي المتوكل، عن…٣٤٠٤والعباس بن عبد العظيم العنبري، قالا حدثنا شبابة، عن شعبة، عن بكير بن…٣٤٠٥كنت نهيتكم عن الأوعية فانتبذوا فيه واجتنبوا كل مسكر٣٤٠٦إني كنت نهيتكم عن نبيذ الأوعية ألا وإن وعاء لا يحرم شيئا كل مسكر حرام٣٤٠٧حدثتني رميثة، عن عائشة، أنها قالت أتعجز إحداكن أن تتخذ كل عام من جلد…٣٤٠٨قال نهى رسول الله صلى الله عليه وسلم أن ينبذ في الجرار .٣٤٠٩اضرب بهذا الحائط فإن هذا شراب من لا يؤمن بالله واليوم الآخر٣٤١٠غطوا الإناء وأوكوا السقاء وأطفئوا السراج وأغلقوا الباب فإن الشيطان لا…٣٤١١حدثنا عبد الحميد بن بيان الواسطي، حدثنا خالد بن عبد الله، عن سهيل، عن…٣٤١٢قالت كنت أصنع لرسول الله صلى الله عليه وسلم ثلاثة آنية من الليل مخمرة…٣٤١٣إن الذي يشرب في إناء الفضة إنما يجرجر في بطنه نار جهنم٣٤١٤هي لهم في الدنيا وهي لكم في الآخرة٣٤١٥من شرب في إناء فضة فكأنما يجرجر في بطنه نار جهنم٣٤١٦أنه كان يتنفس في الإناء ثلاثا وزعم أنس أن رسول الله صلى الله عليه…٣٤١٧ومحمد بن الصباح، قالا حدثنا مروان بن معاوية، حدثنا رشدين بن كريب، عن…٣٤١٨حدثنا أحمد بن عمرو بن السرح، حدثنا ابن وهب، عن يونس، عن ابن شهاب، عن…٣٤١٩حدثنا محمد بن بشار، حدثنا أبو عامر، حدثنا زمعة بن صالح، عن سلمة بن…٣٤٢٠حدثنا بشر بن هلال الصواف، حدثنا عبد الوارث بن سعيد، عن أيوب، عن…٣٤٢١أن رسول الله صلى الله عليه وسلم نهى أن يشرب من فم السقاء .٣٤٢٢قال سقيت النبي صلى الله عليه وسلم من زمزم فشرب قائما . فذكرت ذلك…٣٤٢٣له يقال لها كبشة الأنصارية أن رسول الله صلى الله عليه وسلم دخل عليها…٣٤٢٤حدثنا حميد بن مسعدة، حدثنا بشر بن المفضل، حدثنا سعيد، عن قتادة، عن…٣٤٢٥حدثنا هشام بن عمار، حدثنا مالك بن أنس، عن الزهري، عن أنس بن مالك، أن…٣٤٢٦أتأذن لي أن أسقي خالدا٣٤٢٧إذا شرب أحدكم فلا يتنفس في الإناء فإذا أراد أن يعود فلينح الإناء ثم…٣٤٢٨حدثنا بكر بن خلف أبو بشر، حدثنا يزيد بن زريع، عن خالد الحذاء، عن…٣٤٢٩قال نهى رسول الله صلى الله عليه وسلم أن ينفخ في الإناء .٣٤٣٠قال لم يكن رسول الله صلى الله عليه وسلم ينفخ في الشراب .٣٤٣١لا يلغ أحدكم كما يلغ الكلب ولا يشرب باليد الواحدة كما يشرب القوم…٣٤٣٢إن كان عندك ماء بات في شن فاسقنا وإلا كرعنا٣٤٣٣لا تكرعوا ولكن اغسلوا أيديكم ثم اشربوا فيها فإنه ليس إناء أطيب من اليد٣٤٣٤ساقي القوم آخرهم شربا٣٤٣٥قال كان لرسول الله صلى الله عليه وسلم قدح من قوارير يشرب فيه .
