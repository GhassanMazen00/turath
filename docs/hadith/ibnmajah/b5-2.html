سنن ابن ماجه

كتاب إقامة الصلاة والسنة فيها

٦٣٠ حديثًا•صفحة ٢ من ٥
٩٥٣قال ذكر عند ابن عباس ما يقطع الصلاة فذكروا الكلب والحمار والمرأة فقال…٩٥٤إذا صلى أحدكم فليصل إلى سترة. وليدن منها. ولا يدع أحدا يمر بين يديه.…٩٥٥إذا كان أحدكم يصلي فلا يدع أحدا يمر بين يديه فإن أبى فليقاتله فإن معه…٩٥٦أن النبي صلى الله عليه وسلم كان يصلي من الليل، وأنا معترضة بينه وبين…٩٥٧وسويد بن سعيد، قالا حدثنا يزيد بن زريع، حدثنا خالد الحذاء، عن أبي…٩٥٨قال حدثتني ميمونة، زوج النبي صلى الله عليه وسلم قالت كان النبي صلى…٩٥٩قال نهى رسول الله صلى الله عليه وسلم أن يصلى خلف المتحدث والنائم .٩٦٠حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن عبيد، عن الأعمش، عن أبي…٩٦١ألا يخشى الذي يرفع رأسه قبل الإمام أن يحول الله رأسه رأس حمار؟٩٦٢إني قد بدنت، فإذا ركعت فاركعوا، وإذا رفعت فارفعوا، وإذا سجدت فاسجدوا،…٩٦٣لا تبادروني بالركوع ولا بالسجود، فمهما أسبقكم به إذا ركعت، تدركوني به…٩٦٤إن من الجفاء أن يكثر الرجل مسح جبهته، قبل الفراغ من صلاته٩٦٥لا تفقع أصابعك وأنت في الصلاة٩٦٦قال نهى رسول الله صلى الله عليه وسلم أن يغطي الرجل فاه في الصلاة .٩٦٧أن رسول الله صلى الله عليه وسلم رأى رجلا قد شبك أصابعه في الصلاة ففرج…٩٦٨إذا تثاءب أحدكم فليضع يده على فيه، ولا يعوي، فإن الشيطان يضحك منه٩٦٩البزاق والمخاط والحيض والنعاس في الصلاة من الشيطان٩٧٠ثلاثة لا تقبل لهم صلاة الرجل يؤم القوم وهم له كارهون، والرجل لا يأتي…٩٧١ثلاثة لا ترتفع صلاتهم فوق رءوسهم شبرا رجل أم قوما وهم له كارهون،…٩٧٢اثنان، فما فوقهما، جماعة٩٧٣حدثنا محمد بن عبد الملك بن أبي الشوارب، حدثنا عبد الواحد بن زياد،…٩٧٤يقول كان رسول الله صلى الله عليه وسلم يصلي المغرب، فجئت فقمت عن…٩٧٥قال صلى رسول الله صلى الله عليه وسلم بامرأة من أهله، وبي، فأقامني عن…٩٧٦لا تختلفوا، فتختلف قلوبكم، ليليني منكم أولو الأحلام والنهى، ثم الذين…٩٧٧قال كان رسول الله يحب أن يليه المهاجرون والأنصار، ليأخذوا عنه .٩٧٨تقدموا فأتموا بي، وليأتم بكم من بعدكم، لا يزال قوم يتأخرون حتى يؤخرهم…٩٧٩إذا حضرت الصلاة فأذنا وأقيما، وليؤمكما أكبركما٩٨٠يؤم القوم أقرؤهم لكتاب الله، فإن كانت قراءتهم سواء، فليؤمهم أقدمهم…٩٨١الإمام ضامن، فإن أحسن، فله ولهم، وإن أساء - يعني - فعليه ولا عليهم٩٨٢يأتي على الناس زمان يقومون ساعة، لا يجدون إماما يصلي بهم٩٨٣من أم الناس فأصاب، فالصلاة له ولهم، ومن انتقص من ذلك شيئا فعليه، ولا…٩٨٤يا أيها الناس إن منكم منفرين فأيكم ما صلى بالناس فليجوز فإن فيهم…٩٨٥وحميد بن مسعدة، قالا حدثنا حماد بن زيد، أنبأنا عبد العزيز بن صهيب، عن…٩٨٦أتريد أن تكون فتانا يا معاذ إذا صليت بالناس فاقرأ بالشمس وضحاها وسبح…٩٨٧يا عثمان تجاوز في الصلاة واقدر الناس بأضعفهم فإن فيهم الكبير والصغير…٩٨٨إذا أممت قوما فأخف بهم٩٨٩إني لأدخل في الصلاة وأنا أريد إطالتها فأسمع بكاء الصبي فأتجوز في…٩٩٠إني لأسمع بكاء الصبي فأتجوز في الصلاة٩٩١إني لأقوم في الصلاة وأنا أريد أن أطول فيها فأسمع بكاء الصبي فأتجوز…٩٩٢ألا تصفون كما تصف الملائكة عند ربها٩٩٣سووا صفوفكم فإن تسوية الصفوف من تمام الصلاة٩٩٤سووا صفوفكم أو ليخالفن الله بين وجوهكم٩٩٥إن الله وملائكته يصلون على الذين يصلون الصفوف ومن سد فرجة رفعه الله…٩٩٦أن رسول الله صلى الله عليه وسلم كان يستغفر للصف المقدم، ثلاثا،…٩٩٧إن الله وملائكته يصلون على الصفوف الأول٩٩٨لو يعلمون ما في الصف الأول لكانت قرعة٩٩٩إن الله وملائكته يصلون على الصف الأول١٠٠٠خير صفوف النساء آخرها، وشرها أولها، وخير صفوف الرجال أولها، وشرها…١٠٠١خير صفوف الرجال مقدمها، وشرها مؤخرها، وخير صفوف النساء مؤخرها، وشرها…١٠٠٢بين السواري على عهد رسول الله صلى الله عليه وسلم ونطرد عنها طردا .١٠٠٣استقبل صلاتك، لا صلاة للذي خلف الصف١٠٠٤قال أخذ بيدي زياد بن أبي الجعد فأوقفني على شيخ بالرقة يقال له وابصة…١٠٠٥إن الله وملائكته يصلون على ميامن الصفوف١٠٠٦حدثنا علي بن محمد، حدثنا وكيع، عن مسعر، عن ثابت بن عبيد، عن ابن…١٠٠٧من عمر ميسرة المسجد كتب له كفلان من الأجر١٠٠٨حدثنا العباس بن عثمان الدمشقي، حدثنا الوليد بن مسلم، حدثنا مالك بن…١٠٠٩قال قال عمر قلت يا رسول الله لو اتخذت من مقام إبراهيم مصلى فنزلت…١٠١٠يا جبريل كيف حالنا في صلاتنا إلى بيت المقدس١٠١١ما بين المشرق والمغرب قبلة١٠١٢إذا دخل أحدكم المسجد فلا يجلس حتى يركع ركعتين١٠١٣إذا دخل أحدكم المسجد فليصل ركعتين قبل أن يجلس١٠١٤أن عمر بن الخطاب، قام يوم الجمعة خطيبا - أو خطب يوم الجمعة - فحمد…١٠١٥من أكل من هذه الشجرة الثوم فلا يؤذينا بها في مسجدنا هذا١٠١٦من أكل من هذه الشجرة شيئا فلا يأتين المسجد١٠١٧قال أتى رسول الله صلى الله عليه وسلم مسجد قباء يصلي فيه، فجاءت رجال…١٠١٨إنك سلمت على آنفا وأنا أصلي١٠١٩حدثنا أحمد بن سعيد الدارمي، حدثنا النضر بن شميل، حدثنا يونس بن أبي…١٠٢٠حدثنا يحيى بن حكيم، حدثنا أبو داود، حدثنا أشعث بن سعيد أبو الربيع…١٠٢١إذا صليت فلا تبزقن بين يديك، ولا عن يمينك، ولكن ابزق عن يسارك، أو تحت…١٠٢٢ما بال أحدكم يقوم مستقبله - يعني ربه - فيتنخع أمامه أيحب أحدكم أن…١٠٢٣إن الرجل إذا قام يصلي أقبل الله عليه بوجهه حتى ينقلب أو يحدث حدث سوء١٠٢٤وعبدة بن عبد الله، قالا حدثنا عبد الصمد، حدثنا حماد بن سلمة، عن ثابت،…١٠٢٥من مس الحصى فقد لغا١٠٢٦إن كنت فاعلا فمرة واحدة١٠٢٧إذا قام أحدكم إلى الصلاة فإن الرحمة تواجهه فلا يمسح الحصى١٠٢٨حدثتني ميمونة، زوج النبي صلى الله عليه وسلم قالت كان رسول الله صلى…١٠٢٩قال صلى رسول الله صلى الله عليه وسلم على حصير .١٠٣٠قال صلى ابن عباس وهو بالبصرة على بساطه ثم حدث أصحابه، أن رسول الله…١٠٣١حدثنا أبو بكر بن أبي شيبة، حدثنا عبد العزيز بن محمد الدراوردي، عن…١٠٣٢أن رسول الله صلى الله عليه وسلم صلى في بني عبد الأشهل وعليه كساء…١٠٣٣حدثنا إسحاق بن إبراهيم بن حبيب، حدثنا بشر بن المفضل، عن غالب القطان،…١٠٣٤التسبيح للرجال والتصفيق للنساء١٠٣٥التسبيح للرجال والتصفيق للنساء١٠٣٦وعبيد الله، عن نافع، أنه كان يقول قال ابن عمر رخص رسول الله صلى الله…١٠٣٧قال كان جدي أوس يصلي فيشير إلى وهو في الصلاة فأعطيه نعليه ويقول رأيت…١٠٣٨قال رأيت رسول الله صلى الله عليه وسلم يصلي حافيا ومنتعلا .١٠٣٩حدثنا علي بن محمد، حدثنا يحيى بن آدم، حدثنا زهير، عن أبي إسحاق، عن…١٠٤٠أمرت أن لا أكف شعرا ولا ثوبا١٠٤١حدثنا محمد بن عبد الله بن نمير، حدثنا عبد الله بن إدريس، عن الأعمش،…١٠٤٢- رجلا من أهل المدينة - يقول رأيت أبا رافع مولى رسول الله صلى الله…١٠٤٣لا ترفعوا أبصاركم إلى السماء أن تلتمع١٠٤٤ما بال أقوام يرفعون أبصارهم إلى السماء١٠٤٥لينتهين أقوام يرفعون أبصارهم إلى السماء أو لا ترجع أبصارهم١٠٤٦حدثنا حميد بن مسعدة، وأبو بكر بن خلاد قالا حدثنا نوح بن قيس، حدثنا…١٠٤٧أوكلكم يجد ثوبين١٠٤٨أنه دخل على رسول الله صلى الله عليه وسلم وهو يصلي في ثوب واحد متوشحا…١٠٤٩قال رأيت رسول الله صلى الله عليه وسلم يصلي في ثوب واحد متوشحا به…١٠٥٠قال رأيت رسول الله صلى الله عليه وسلم يصلي بالبئر العليا في ثوب .١٠٥١قال رأيت النبي صلى الله عليه وسلم يصلي الظهر والعصر في ثوب واحد…١٠٥٢إذا قرأ ابن آدم السجدة فسجد اعتزل الشيطان يبكي يقول يا ويله أمر ابن…١٠٥٣حدثنا أبو بكر بن خلاد الباهلي، حدثنا محمد بن يزيد بن خنيس، عن الحسن…١٠٥٤اللهم لك سجدت وبك آمنت ولك أسلمت أنت ربي سجد وجهي للذي شق سمعه وبصره…١٠٥٥أنه سجد مع النبي صلى الله عليه وسلم إحدى عشرة سجدة منهن النجم .١٠٥٦قال حدثتني عمتي أم الدرداء، عن أبي الدرداء، قال سجدت مع النبي صلى…١٠٥٧أن رسول الله صلى الله عليه وسلم أقرأه خمس عشرة سجدة في القرآن منها…١٠٥٨حدثنا أبو بكر بن أبي شيبة، حدثنا سفيان بن عيينة، عن أيوب بن موسى، عن…١٠٥٩حدثنا أبو بكر بن أبي شيبة، حدثنا سفيان بن عيينة، عن يحيى بن سعيد، عن…١٠٦٠وعليك فارجع فصل فإنك لم تصل١٠٦١سمع الله لمن حمده١٠٦٢قالت سألت عائشة كيف كانت صلاة رسول الله صلى الله عليه وسلم قالت كان…١٠٦٣قال صلاة السفر ركعتان والجمعة ركعتان والعيد ركعتان تمام غير قصر على…١٠٦٤قال صلاة السفر ركعتان وصلاة الجمعة ركعتان والفطر والأضحى ركعتان تمام…١٠٦٥صدقة تصدق الله بها عليكم، فاقبلوا صدقته١٠٦٦حدثنا محمد بن رمح، أنبأنا الليث بن سعد، عن ابن شهاب، عن عبد الله بن…١٠٦٧قال كان رسول الله صلى الله عليه وسلم إذا خرج من هذه المدينة لم يزد…١٠٦٨وجبارة بن المغلس، قالا حدثنا أبو عوانة، عن بكير بن الأخنس، عن مجاهد،…١٠٦٩وسعيد بن جبير، وعطاء بن أبي رباح، وطاوس، أخبروه عن ابن عباس، أنه…١٠٧٠أن النبي صلى الله عليه وسلم جمع بين الظهر والعصر والمغرب والعشاء في…١٠٧١حدثنا أبو بكر بن خلاد الباهلي، حدثنا أبو عامر، عن عيسى بن حفص بن عاصم…١٠٧٢أنه سمع ابن عباس، يقول فرض رسول الله صلى الله عليه وسلم صلاة الحضر…١٠٧٣ثلاثا للمهاجر بعد الصدر١٠٧٤وقرأته، عليه أنبأنا ابن جريج، أخبرني عطاء، حدثني جابر بن عبد الله، في…١٠٧٥قال أقام رسول الله صلى الله عليه وسلم تسعة عشر يوما يصلي ركعتين…١٠٧٦أن رسول الله صلى الله عليه وسلم أقام بمكة عام الفتح خمس عشرة ليلة…١٠٧٧وعبد الأعلى، قالا حدثنا يحيى بن أبي إسحاق، عن أنس قال: خرجنا مع رسول…١٠٧٨بين العبد وبين الكفر ترك الصلاة١٠٧٩العهد الذي بيننا وبينهم الصلاة، فمن تركها فقد كفر١٠٨٠ليس بين العبد والشرك إلا ترك الصلاة، فإذا تركها فقد أشرك١٠٨١يا أيها الناس توبوا إلى الله قبل أن تموتوا وبادروا بالأعمال الصالحة…١٠٨٢قال: كنت قائد أبي حين ذهب بصره، وكنت إذا خرجت به إلى الجمعة فسمع…١٠٨٣أضل الله عن الجمعة من كان قبلنا. كان لليهود يوم السبت. والأحد…١٠٨٤إن يوم الجمعة سيد الأيام، وأعظمها عند الله. وهو أعظم عند الله من يوم…١٠٨٥إن من أفضل أيامكم يوم الجمعة فيه خلق آدم وفيه النفخة وفيه الصعقة…١٠٨٦الجمعة إلى الجمعة كفارة ما بينهما. ما لم تغش الكبائر١٠٨٧من غسل يوم الجمعة واغتسل، وبكر وابتكر، ومشى ولم يركب، ودنا من الإمام،…١٠٨٨من أتى الجمعة فليغتسل١٠٨٩غسل يوم الجمعة واجب على كل محتلم١٠٩٠من توضأ فأحسن الوضوء، ثم أتى الجمعة، فدنا وأنصت واستمع، غفر له ما…١٠٩١من توضأ يوم الجمعة فبها ونعمت يجزئ عنه الفريضة ومن اغتسل فالغسل أفضل١٠٩٢إذا كان يوم الجمعة، كان على كل باب من أبواب المسجد ملائكة يكتبون…١٠٩٣أن رسول الله صلى الله عليه وسلم ضرب مثل الجمعة ثم التبكير، كناحر…١٠٩٤إن الناس يجلسون من الله يوم القيامة على قدر رواحهم إلى الجمعات الأول…١٠٩٥ما على أحدكم لو اشترى ثوبين ليوم الجمعة، سوى ثوب مهنته١٠٩٦ما على أحدكم، إن وجد سعة، أن يتخذ ثوبين لجمعته، سوى ثوبى مهنته١٠٩٧من اغتسل يوم الجمعة فأحسن غسله، وتطهر فأحسن طهوره، ولبس من أحسن…١٠٩٨إن هذا يوم عيد، جعله الله للمسلمين. فمن جاء إلى الجمعة فليغتسل. وإن…١٠٩٩حدثنا محمد بن الصباح، حدثنا عبد العزيز بن أبي حازم، حدثني أبي، عن سهل…١١٠٠فلا نرى للحيطان فيئا نستظل به .١١٠١حدثنا هشام بن عمار، حدثنا عبد الرحمن بن سعد بن عمار بن سعد، مؤذن…١١٠٢حدثنا أحمد بن عبدة، حدثنا المعتمر بن سليمان، حدثنا حميد، عن أنس، قال…
