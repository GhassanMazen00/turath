سنن ابن ماجه

كتاب الرهون

٥٣ حديثًا
٢٤٣٦أن النبي صلى الله عليه وسلم اشترى من يهودي طعاما إلى أجل وأرهنه درعه .٢٤٣٧قال لقد رهن رسول الله صلى الله عليه وسلم درعه عند يهودي بالمدينة فأخذ…٢٤٣٨أن النبي صلى الله عليه وسلم توفي ودرعه مرهونة عند يهودي بطعام .٢٤٣٩أن رسول الله صلى الله عليه وسلم مات ودرعه رهن عند يهودي بثلاثين صاعا…٢٤٤٠الظهر يركب إذا كان مرهونا ولبن الدر يشرب إذا كان مرهونا وعلى الذي…٢٤٤١أن رسول الله صلى الله عليه وسلم قال " لا يغلق الرهن " .٢٤٤٢ثلاثة أنا خصمهم يوم القيامة ومن كنت خصمه خصمته يوم القيامة رجل أعطى…٢٤٤٣أعطوا الأجير أجره قبل أن يجف عرقه٢٤٤٤إن موسى صلى الله عليه وسلم أجر نفسه ثماني سنين أو عشرا على عفة فرجه…٢٤٤٥حدثنا أبو عمر، حفص بن عمرو حدثنا عبد الرحمن بن مهدي، حدثنا سليم بن…٢٤٤٦حدثنا محمد بن عبد الأعلى الصنعاني، حدثنا المعتمر بن سليمان، عن أبيه،…٢٤٤٧قال كنت أدلو الدلو بتمرة وأشترط أنها جلدة .٢٤٤٨قال جاء رجل من الأنصار فقال يا رسول الله ما لي أرى لونك منكفئا . قال…٢٤٤٩إنما يزرع ثلاثة رجل له أرض فهو يزرعها ورجل منح أرضا فهو يزرع ما منح…٢٤٥٠ومحمد بن الصباح، قالا حدثنا سفيان بن عيينة، عن عمرو بن دينار، قال…٢٤٥١من كانت له فضول أرضين فليزرعها أو ليزرعها أخاه فإن أبى فليمسك أرضه٢٤٥٢من كانت له أرض فليزرعها أو ليمنحها أخاه فإن أبى فليمسك أرضه٢٤٥٣حدثنا أبو كريب، حدثنا عبدة بن سليمان، وأبو أسامة ومحمد بن عبيد عن…٢٤٥٤من كانت له أرض فليزرعها أو ليزرعها ولا يؤاجرها٢٤٥٥- مولى ابن أبي أحمد - أنه أخبره أنه، سمع أبا سعيد الخدري، يقول نهى…٢٤٥٦ألا منحها أحدكم أخاه٢٤٥٧لأن يمنح أحدكم أخاه أرضه خير له من أن يأخذ عليها كذا وكذا٢٤٥٨حدثنا محمد بن الصباح، حدثنا سفيان بن عيينة، عن يحيى بن سعيد، عن حنظلة…٢٤٥٩ما تصنعون بمحاقلكم٢٤٦٠من استغنى عن أرضه فليمنحها أخاه أو ليدع٢٤٦١إن كان هذا شأنكم فلا تكروا المزارع٢٤٦٢لأن يمنح أحدكم أخاه خير له من أن يأخذ عليها أجرا معلوما٢٤٦٣أن معاذ بن جبل، أكرى الأرض على عهد رسول الله صلى الله عليه وسلم وأبي…٢٤٦٥من كانت له أرض فلا يكريها بطعام مسمى٢٤٦٦من زرع في أرض قوم بغير إذنهم فليس له من الزرع شىء وترد عليه نفقته٢٤٦٧وسهل بن أبي سهل، وإسحاق بن منصور، قالوا حدثنا يحيى بن سعيد القطان، عن…٢٤٦٨أن رسول الله صلى الله عليه وسلم أعطى خيبر أهلها على النصف نخلها…٢٤٦٩قال لما افتتح رسول الله صلى الله عليه وسلم خيبر أعطاها على النصف .٢٤٧٠أنه سمع موسى بن طلحة بن عبيد الله، يحدث عن أبيه، قال مررت مع رسول…٢٤٧١وهشام بن عروة، عن أبيه، عن عائشة، أن النبي صلى الله عليه وسلم سمع…٢٤٧٢المسلمون شركاء في ثلاث في الماء والكلإ والنار وثمنه حرام٢٤٧٣ثلاث لا يمنعن الماء والكلأ والنار٢٤٧٤. قالت قلت يا رسول الله هذا الماء قد عرفناه فما بال الملح والنار قال٢٤٧٥هو منك صدقة وهو مثل الماء العد من ورده أخذه٢٤٧٨لا يمنع أحدكم فضل ماء ليمنع به الكلأ٢٤٧٩لا يمنع فضل الماء ولا يمنع نقع البئر٢٤٨٠اسق يا زبير ثم أرسل الماء إلى جارك٢٤٨١ثعلبة بن أبي مالك قال قضى رسول الله صلى الله عليه وسلم في سيل مهزور…٢٤٨٢أن رسول الله صلى الله عليه وسلم قضى في سيل مهزور أن يمسك حتى يبلغ…٢٤٨٣أن رسول الله صلى الله عليه وسلم قضى في شرب النخل من السيل أن الأعلى…٢٤٨٤يبدأ بالخيل يوم وردها٢٤٨٥كل قسم قسم في الجاهلية فهو على ما قسم وكل قسم أدركه الإسلام فهو على…٢٤٨٦من حفر بئرا فله أربعون ذراعا عطنا لماشيته٢٤٨٧حريم البئر مد رشائها٢٤٨٨أن رسول الله صلى الله عليه وسلم قضى في النخلة والنخلتين والثلاثة…٢٤٨٩حريم النخلة مد جريدها٢٤٩٠حديث ٢٤٩٠٢٤٩١من باع دارا ولم يجعل ثمنها في مثلها لم يبارك له فيها
