سنن ابن ماجه

كتاب الذبائح

٣٨ حديثًا
٣١٦٢عن الغلام، شاتان مكافئتان وعن الجارية، شاة٣١٦٣حدثنا أبو بكر بن أبي شيبة، حدثنا عفان، حدثنا حماد بن سلمة، أنبأنا عبد…٣١٦٤إن مع الغلام عقيقة فأهريقوا عنه دما وأميطوا عنه الأذى٣١٦٥كل غلام مرتهن بعقيقته تذبح عنه يوم السابع ويحلق رأسه ويسمى٣١٦٦يعق عن الغلام ولا يمس رأسه بدم٣١٦٧اذبحوا لله عز وجل في أى شهر ما كان وبروا الله وأطعموا٣١٦٨لا فرعة ولا عتيرة٣١٦٩لا فرعة ولا عتيرة٣١٧٠إن الله عز وجل كتب الإحسان على كل شىء فإذا قتلتم فأحسنوا القتلة وإذا…٣١٧١دع أذنها وخذ بسالفتها٣١٧٢إذا ذبح أحدكم فليجهز٣١٧٣{إن الشياطين ليوحون إلى أوليائهم} قال كانوا يقولون ما ذكر عليه اسم…٣١٧٤سموا أنتم وكلوا٣١٧٥قال ذبحت أرنبين بمروة فأتيت بهما النبي صلى الله عليه وسلم فأمرني…٣١٧٦أن ذئبا، نيب في شاة فذبحوها بمروة فرخص لهم رسول الله صلى الله عليه…٣١٧٧أمرر الدم بما شئت واذكر اسم الله عليه٣١٧٨ما أنهر الدم وذكر اسم الله عليه فكل غير السن والظفر فإن السن عظم…٣١٧٩حدثنا أبو كريب، حدثنا مروان بن معاوية، حدثنا هلال بن ميمون الجهني، عن…٣١٨٠أن رسول الله صلى الله عليه وسلم أتى رجلا من الأنصار فأخذ الشفرة ليذبح…٣١٨١انطلقوا بنا إلى الواقفي٣١٨٢أن امرأة، ذبحت شاة بحجر فذكر ذلك لرسول الله صلى الله عليه وسلم فلم ير…٣١٨٣إن لها أوابد - أحسبه قال - كأوابد الوحش فما غلبكم منها فاصنعوا به هكذا٣١٨٤لو طعنت في فخذها لأجزأك٣١٨٥وعبد الله بن سعيد، قالا حدثنا عقبة بن خالد، عن موسى بن محمد بن…٣١٨٦حدثنا علي بن محمد، حدثنا وكيع، عن شعبة، عن هشام بن زيد بن أنس بن…٣١٨٧لا تتخذوا شيئا فيه الروح غرضا٣١٨٨أنه سمع جابر بن عبد الله، يقول نهى رسول الله صلى الله عليه وسلم أن…٣١٨٩حدثنا سويد بن سعيد، حدثنا ابن أبي زائدة، عن محمد بن إسحاق، عن ابن أبي…٣١٩٠حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن هشام بن عروة، عن فاطمة بنت…٣١٩١أنه سمع جابر بن عبد الله، يقول أكلنا زمن خيبر الخيل وحمر الوحش .٣١٩٢حدثنا سويد بن سعيد، حدثنا علي بن مسهر، عن أبي إسحاق الشيباني، قال…٣١٩٣أن رسول الله صلى الله عليه وسلم حرم أشياء حتى ذكر الحمر الإنسية .٣١٩٤حدثنا سويد بن سعيد، حدثنا علي بن مسهر، عن عاصم، عن الشعبي، عن البراء…٣١٩٥حدثنا يعقوب بن حميد بن كاسب، حدثنا المغيرة بن عبد الرحمن، عن يزيد بن…٣١٩٦أن منادي النبي، صلى الله عليه وسلم نادى إن الله ورسوله ينهيانكم عن…٣١٩٧ومعمر، جميعا عن عبد الكريم الجزري، عن عطاء، عن جابر بن عبد الله، قال…٣١٩٨حدثنا محمد بن المصفى، حدثنا بقية، حدثني ثور بن يزيد، عن صالح بن يحيى…٣١٩٩كلوه إن شئتم فإن ذكاة الجنين ذكاة أمه
