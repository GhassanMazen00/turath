سنن ابن ماجه

كتاب الهبات

١٥ حديثًا
٢٣٧٥فكل بنيك نحلت مثل الذي نحلت النعمان٢٣٧٦أكل ولدك نحلت٢٣٧٧لا يحل للرجل أن يعطي العطية ثم يرجع فيها إلا الوالد فيما يعطي ولده٢٣٧٨لا يرجع أحدكم في هبته إلا الوالد من ولده٢٣٧٩لا عمرى فمن أعمر شيئا فهو له٢٣٨٠من أعمر رجلا عمرى له ولعقبه فقد قطع قوله حقه فيها فهي لمن أعمر ولعقبه٢٣٨١أن النبي صلى الله عليه وسلم جعل العمرى للوارث .٢٣٨٢لا رقبى فمن أرقب شيئا فهو له حياته ومماته٢٣٨٣العمرى جائزة لمن أعمرها والرقبى جائزة لمن أرقبها٢٣٨٤إن مثل الذي يعود في عطيته كمثل الكلب أكل حتى إذا شبع قاء ثم عاد في…٢٣٨٥العائد في هبته كالعائد في قيئه٢٣٨٦العائد في هبته كالكلب يعود في قيئه٢٣٨٧الرجل أحق بهبته ما لم يثب منها٢٣٨٨لا يجوز لامرأة في مالها إلا بإذن زوجها إذا هو ملك عصمتها٢٣٨٩لا يجوز للمرأة في مالها إلا بإذن زوجها فهل استأذنت كعبا
