سنن ابن ماجه

كتاب اللباس

١٠٧ حديثًا
٣٥٥٠شغلني أعلام هذه اذهبوا بها إلى أبي جهم وائتوني بأنبجانيته٣٥٥١قال دخلت على عائشة فأخرجت لي إزارا غليظا من التي تصنع باليمن وكساء من…٣٥٥٢أن رسول الله صلى الله عليه وسلم صلى في شملة قد عقد عليها .٣٥٥٣قال كنت مع النبي صلى الله عليه وسلم وعليه رداء نجراني غليظ الحاشية .٣٥٥٤قالت ما رأيت رسول الله صلى الله عليه وسلم يسب أحدا ولا يطوى له ثوب .٣٥٥٥أن امرأة، جاءت إلى رسول الله صلى الله عليه وسلم ببردة - قال وما…٣٥٥٦حدثنا يحيى بن عثمان بن سعيد بن كثير بن دينار الحمصي، حدثنا بقية بن…٣٥٥٧من لبس ثوبا جديدا فقال الحمد لله الذي كساني ما أواري به عورتي وأتجمل…٣٥٥٨ثوبك هذا غسيل أم جديد٣٥٥٩حدثنا أبو بكر، حدثنا سفيان بن عيينة، عن الزهري، عن عطاء بن يزيد…٣٥٦٠حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، وأبو أسامة عن عبيد…٣٥٦١حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، وأبو أسامة عن سعد…٣٥٦٢حدثنا أبو بكر بن أبي شيبة، حدثنا الحسن بن موسى، عن شيبان، عن قتادة،…٣٥٦٣حدثنا محمد بن عثمان بن كرامة، حدثنا أبو أسامة، حدثنا الأحوص بن حكيم،…٣٥٦٤وأحمد بن الأزهر، قالا حدثنا مروان بن محمد، حدثنا يزيد بن السمط، حدثني…٣٥٦٥قال رأيت رسول الله صلى الله عليه وسلم يسم غنما في آذانها ورأيته متزرا…٣٥٦٦خير ثيابكم البياض فالبسوها وكفنوا فيها موتاكم٣٥٦٧البسوا ثياب البياض فإنها أطهر وأطيب٣٥٦٨إن أحسن ما زرتم الله به في قبوركم ومساجدكم البياض٣٥٦٩إن الذي يجر ثوبه من الخيلاء لا ينظر الله إليه يوم القيامة٣٥٧٠من جر إزاره من الخيلاء لم ينظر الله إليه يوم القيامة٣٥٧١من جر ثوبه من الخيلاء لم ينظر الله له يوم القيامة٣٥٧٢هذا موضع الإزار فإن أبيت . فأسفل فإن أبيت فأسفل فإن أبيت فلا حق…٣٥٧٣إزرة المؤمن إلى أنصاف ساقيه لا جناح عليه ما بينه وبين الكعبين وما…٣٥٧٤يا سفيان بن سهل لا تسبل فإن الله لا يحب المسبلين٣٥٧٥قالت لم يكن ثوب أحب إلى رسول الله صلى الله عليه وسلم من القميص .٣٥٧٦الإسبال في الإزار والقميص والعمامة من جر شيئا خيلاء لم ينظر الله إليه…٣٥٧٧قال كان رسول الله صلى الله عليه وسلم يلبس قميصا قصير اليدين والطول .٣٥٧٨قال أتيت رسول الله صلى الله عليه وسلم فبايعته وإن زر قميصه لمطلق .…٣٥٧٩وعلي بن محمد، قالا حدثنا وكيع، ح وحدثنا محمد بن بشار، حدثنا يحيى،…٣٥٨٠قالت سئل رسول الله صلى الله عليه وسلم كم تجر المرأة من ذيلها قال "…٣٥٨١أن أزواج النبي، صلى الله عليه وسلم رخص لهن في الذيل ذراعا فكن يأتينا…٣٥٨٢. أن النبي صلى الله عليه وسلم قال لفاطمة أو لأم سلمة " ذيلك ذراع " .٣٥٨٣أن النبي صلى الله عليه وسلم قال في ذيول النساء " شبرا " . فقالت عائشة…٣٥٨٤قال رأيت النبي صلى الله عليه وسلم يخطب على المنبر وعليه عمامة سوداء .٣٥٨٥أن النبي صلى الله عليه وسلم دخل مكة وعليه عمامة سوداء .٣٥٨٦أن النبي صلى الله عليه وسلم دخل يوم فتح مكة وعليه عمامة سوداء .٣٥٨٧قال كأني أنظر إلى رسول الله صلى الله عليه وسلم وعليه عمامة سوداء قد…٣٥٨٨من لبس الحرير في الدنيا لم يلبسه في الآخرة٣٥٨٩حدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن الشيباني، عن أشعث بن…٣٥٩٠هو لهم في الدنيا ولنا في الآخرة٣٥٩١إنما يلبس هذا من لا خلاق له في الآخرة٣٥٩٢. أن أنس بن مالك، نبأهم أن رسول الله صلى الله عليه وسلم رخص للزبير بن…٣٥٩٣والديباج، إلا ما كان هكذا ثم أشار بإصبعه ثم الثانية ثم الثالثة ثم…٣٥٩٤مولى أسماء قال رأيت ابن عمر اشترى عمامة لها علم فدعا بالجلمين فقصه…٣٥٩٥إن هذين حرام على ذكور أمتي حل لإناثهم٣٥٩٦لا ولكن اجعلها خمرا بين الفواطم٣٥٩٧إن هذين محرم على ذكور أمتي حل لإناثهم٣٥٩٨قال رأيت على زينب بنت رسول الله صلى الله عليه وسلم قميص حرير سيراء .٣٥٩٩قال ما رأيت أجمل من رسول الله صلى الله عليه وسلم مترجلا في حلة حمراء .٣٦٠٠صدق الله ورسوله {إنما أموالكم وأولادكم فتنة} رأيت هذين فلم أصبر٣٦٠١حدثنا أبو بكر بن أبي شيبة، حدثنا علي بن مسهر، عن يزيد بن أبي زياد، عن…٣٦٠٢حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن أسامة بن زيد، عن عبد الله…٣٦٠٣حدثنا أبو بكر، حدثنا عيسى بن يونس، عن هشام بن الغاز، عن عمرو بن شعيب،…٣٦٠٤حدثنا علي بن محمد، حدثنا وكيع، عن ابن أبي ليلى، عن محمد بن عبد…٣٦٠٥كلوا واشربوا وتصدقوا والبسوا ما لم يخالطه إسراف أو مخيلة٣٦٠٦من لبس ثوب شهرة ألبسه الله يوم القيامة ثوب مذلة٣٦٠٧من لبس ثوب شهرة في الدنيا ألبسه الله ثوب مذلة يوم القيامة ثم ألهب فيه…٣٦٠٨من لبس ثوب شهرة أعرض الله عنه حتى يضعه متى وضعه٣٦٠٩أيما إهاب دبغ فقد طهر٣٦١٠هلا أخذوا إهابها فدبغوه فانتفعوا به٣٦١١ما ضر أهل هذه لو انتفعوا بإهابها٣٦١٢قالت أمر رسول الله صلى الله عليه وسلم أن يستمتع بجلود الميتة إذا دبغت…٣٦١٣أن لا تنتفعوا من الميتة بإهاب ولا عصب٣٦١٤قال كان لنعل النبي صلى الله عليه وسلم قبالان مثني شراكهما .٣٦١٥قال كان لنعل النبي صلى الله عليه وسلم قبالان .٣٦١٦إذا انتعل أحدكم فليبدأ باليمنى وإذا خلع فليبدأ باليسرى٣٦١٧لا يمشي أحدكم في نعل واحد ولا خف واحد ليخلعهما جميعا أو ليمش فيهما…٣٦١٨قال نهى رسول الله صلى الله عليه وسلم أن ينتعل الرجل قائما .٣٦١٩قال نهى النبي صلى الله عليه وسلم أن ينتعل الرجل قائما .٣٦٢٠أن النجاشي، أهدى لرسول الله صلى الله عليه وسلم خفين ساذجين أسودين…٣٦٢١إن اليهود والنصارى لا يصبغون فخالفوهم٣٦٢٢إن أحسن ما غيرتم به الشيب الحناء والكتم٣٦٢٣قال دخلت على أم سلمة . قال فأخرجت إلى شعرا من شعر رسول الله صلى الله…٣٦٢٤اذهبوا به إلى بعض نسائه فلتغيره وجنبوه السواد٣٦٢٥إن أحسن ما اختضبتم به لهذا السواد أرغب لنسائكم فيكم وأهيب لكم في صدور…٣٦٢٦أن عبيد بن جريج، سأل ابن عمر قال رأيتك تصفر لحيتك بالورس فقال ابن عمر…٣٦٢٧قال مر النبي صلى الله عليه وسلم على رجل قد خضب بالحناء فقال " ما أحسن…٣٦٢٨قال رأيت رسول الله صلى الله عليه وسلم هذه منه بيضاء . يعني عنفقته .٣٦٢٩وابن أبي عدي، عن حميد، قال سئل أنس بن مالك أخضب رسول الله صلى الله…٣٦٣٠قال كان شيب رسول الله صلى الله عليه وسلم نحو عشرين شعرة .٣٦٣١قال قالت أم هانئ دخل رسول الله صلى الله عليه وسلم مكة وله أربع غدائر…٣٦٣٢قال كان أهل الكتاب يسدلون أشعارهم وكان المشركون يفرقون وكان رسول الله…٣٦٣٣قالت كنت أفرق خلف يافوخ رسول الله صلى الله عليه وسلم ثم أسدل ناصيته .٣٦٣٤قال كان شعر رسول الله صلى الله عليه وسلم شعرا رجلا بين أذنيه ومنكبيه٣٦٣٥قالت كان لرسول الله صلى الله عليه وسلم شعر دون الجمة وفوق الوفرة.٣٦٣٦وسفيان بن عقبة، عن سفيان، عن عاصم بن كليب، عن أبيه، عن وائل بن حجر،…٣٦٣٧وعلي بن محمد، قالا حدثنا أبو أسامة، عن عبيد الله بن عمر، عن عمر بن…٣٦٣٨حدثنا أبو بكر بن أبي شيبة، حدثنا شبابة، حدثنا شعبة، عن عبد الله بن…٣٦٣٩لا ينقش أحد على نقش خاتمي هذا٣٦٤٠إنا قد اصطنعنا خاتما ونقشنا فيه نقشا فلا ينقشن عليه أحد٣٦٤١أن رسول الله صلى الله عليه وسلم اتخذ خاتما من فضة له فص حبشي ونقشه…٣٦٤٢حدثنا أبو بكر، حدثنا عبد الله بن نمير، عن عبيد الله، عن نافع بن جبير،…٣٦٤٣حدثنا أبو بكر، حدثنا علي بن مسهر، عن يزيد بن أبي زياد، عن الحسن بن…٣٦٤٤تحلى بهذا يا بنية٣٦٤٥أن النبي صلى الله عليه وسلم كان يجعل فص خاتمه مما يلي كفه .٣٦٤٦أن رسول الله صلى الله عليه وسلم لبس خاتم فضة فيه فص حبشي كان يجعل فصه…٣٦٤٧أن النبي صلى الله عليه وسلم كان يتختم في يمينه .٣٦٤٨قال نهاني رسول الله صلى الله عليه وسلم أن أتختم في هذه وفي هذه يعني…٣٦٤٩لا تدخل الملائكة بيتا فيه كلب ولا صورة٣٦٥٠إن الملائكة لا تدخل بيتا فيه كلب ولا صورة٣٦٥١ما منعك أن تدخل٣٦٥٢أن امرأة، أتت النبي صلى الله عليه وسلم فأخبرته أن زوجها في بعض…٣٦٥٣قالت سترت سهوة لي - تعني الداخل - بستر فيه تصاوير فلما قدم النبي صلى…٣٦٥٤حدثنا أبو بكر، حدثنا أبو الأحوص، عن أبي إسحاق، عن هبيرة، عن علي، قال…٣٦٥٥حدثنا أبو بكر بن أبي شيبة، حدثنا زيد بن الحباب، حدثنا يحيى بن أيوب،…٣٦٥٦حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن أبي المعتمر، عن ابن سيرين،…
