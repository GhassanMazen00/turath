سنن ابن ماجه

كتاب الصيام

١٤٥ حديثًا
١٦٣٨كل عمل ابن آدم يضاعف الحسنة بعشر أمثالها إلى سبعمائة ضعف إلى ما شاء…١٦٣٩الصيام جنة من النار كجنة أحدكم من القتال١٦٤٠إن في الجنة بابا يقال له الريان يدعى يوم القيامة يقال: أين الصائمون؟…١٦٤١من صام رمضان إيمانا واحتسابا غفر له ما تقدم من ذنبه١٦٤٢إذا كانت أول ليلة من رمضان صفدت الشياطين ومردة الجن وغلقت أبواب النار…١٦٤٣إن لله عند كل فطر عتقاء وذلك في كل ليلة١٦٤٤إن هذا الشهر قد حضركم وفيه ليلة خير من ألف شهر من حرمها فقد حرم الخير…١٦٤٥حدثنا محمد بن عبد الله بن نمير، حدثنا أبو خالد الأحمر، عن عمرو بن…١٦٤٦حدثنا أبو بكر بن أبي شيبة، حدثنا حفص بن غياث، عن عبد الله بن سعيد، عن…١٦٤٧الصيام يوم كذا وكذا ونحن متقدمون فمن شاء فليتقدم ومن شاء فليتأخر١٦٤٨قالت كان رسول الله صلى الله عليه وسلم يصل شعبان برمضان .١٦٤٩رسول الله صلى الله عليه وسلم فقالت: كان يصوم شعبان كله حتى يصله…١٦٥٠لا تقدموا صيام رمضان بيوم ولا بيومين إلا رجل كان يصوم صوما فيصومه١٦٥١إذا كان النصف من شعبان فلا صوم حتى يجيء رمضان١٦٥٢أتشهد أن لا إله إلا الله وأن محمدا رسول الله١٦٥٣حدثنا أبو بكر بن أبي شيبة، حدثنا هشيم، عن أبي بشر، عن أبي عمير بن أنس…١٦٥٤إذا رأيتم الهلال فصوموا وإذا رأيتموه فأفطروا . فإن غم عليكم فاقدروا له١٦٥٥إذا رأيتم الهلال فصوموا وإذا رأيتموه فأفطروا فإن غم عليكم فصوموا…١٦٥٦كم مضى من الشهر؟١٦٥٧الشهر هكذا وهكذا وهكذا١٦٥٨حدثنا مجاهد بن موسى، حدثنا القاسم بن مالك المزني، حدثنا الجريري، عن…١٦٥٩شهرا عيد لا ينقصان رمضان وذو الحجة١٦٦٠الفطر يوم تفطرون والأضحى يوم تضحون١٦٦١قال صام رسول الله صلى الله عليه وسلم في السفر وأفطر .١٦٦٢إن شئت فصم وإن شئت فأفطر١٦٦٣وهارون بن عبد الله الحمال، قالا: حدثنا ابن أبي فديك، جميعا عن هشام بن…١٦٦٤ليس من البر الصيام في السفر١٦٦٥ليس من البر الصيام في السفر١٦٦٦صائم رمضان في السفر كالمفطر في الحضر١٦٦٧وعلي بن محمد، قالا: حدثنا وكيع، عن أبي هلال، عن عبد الله بن سوادة، عن…١٦٦٨قال رخص رسول الله صلى الله عليه وسلم للحبلى التي تخاف على نفسها أن…١٦٦٩حدثنا علي بن المنذر، حدثنا سفيان بن عيينة، عن عمرو بن دينار، وعن يحيى…١٦٧٠حدثنا علي بن محمد، حدثنا عبد الله بن نمير، عن عبيدة، عن إبراهيم، عن…١٦٧١حدثنا أبو بكر بن أبي شيبة، حدثنا سفيان بن عيينة، عن الزهري، عن حميد…١٦٧٢من أفطر يوما من رمضان من غير رخصة لم يجزه صيام الدهر١٦٧٣من أكل ناسيا وهو صائم فليتم صومه فإنما أطعمه الله وسقاه١٦٧٤وعلي بن محمد، قالا حدثنا أبو أسامة، عن هشام بن عروة، عن فاطمة بنت…١٦٧٥أجل. ولكني قئت١٦٧٦من ذرعه القىء فلا قضاء عليه ومن استقاء فعليه القضاء١٦٧٧من خير خصال الصائم السواك١٦٧٨قالت: اكتحل رسول الله صلى الله عليه وسلم وهو صائم .١٦٧٩أفطر الحاجم والمحجوم١٦٨٠أفطر الحاجم والمحجوم١٦٨١أفطر الحاجم والمحجوم١٦٨٢قال: احتجم رسول الله صلى الله عليه وسلم وهو صائم محرم .١٦٨٣وعبد الله بن الجراح، قالا: حدثنا أبو الأحوص، عن زياد بن علاقة، عن…١٦٨٤قالت: كان رسول الله صلى الله عليه وسلم يقبل وهو صائم. وأيكم يملك إربه…١٦٨٥وعلي بن محمد، قالا حدثنا أبو معاوية، عن الأعمش، عن مسلم، عن شتير بن…١٦٨٦حدثنا أبو بكر بن أبي شيبة، حدثنا الفضل بن دكين، عن إسرائيل، عن زيد بن…١٦٨٧قال دخل الأسود ومسروق على عائشة فقالا أكان رسول الله صلى الله عليه…١٦٨٨قال: رخص للكبير الصائم في المباشرة وكره للشاب .١٦٨٩من لم يدع قول الزور، والجهل، والعمل به، فلا حاجة لله في أن يدع طعامه…١٦٩٠رب صائم ليس له من صيامه إلا الجوع. ورب قائم ليس له من قيامه إلا السهر١٦٩١إذا كان يوم صوم أحدكم فلا يرفث ولا يجهل. فإن جهل عليه أحد، فليقل: إني…١٦٩٢تسحروا فإن في السحور بركة١٦٩٣استعينوا بطعام السحر على صيام النهار وبالقيلولة على قيام الليل١٦٩٤حدثنا علي بن محمد، حدثنا وكيع، عن هشام الدستوائي، عن قتادة، عن أنس بن…١٦٩٥قال: تسحرت مع رسول الله صلى الله عليه وسلم هو النهار إلا أن الشمس لم…١٦٩٦لا يمنعن أحدكم أذان بلال من سحوره. فإنه يؤذن لينتبه نائمكم، وليرجع…١٦٩٧لا يزال الناس بخير ما عجلوا الإفطار١٦٩٨لا يزال الناس بخير ما عجلوا الفطر. عجلوا الفطر، فإن اليهود يؤخرون١٦٩٩إذا أفطر أحدكم، فليفطر على تمرة. فإن لم يجد، فليفطر على الماء. فإنه…١٧٠٠لا صيام لمن لم يفرضه من الليل١٧٠١حدثنا إسماعيل بن موسى، حدثنا شريك، عن طلحة بن يحيى، عن مجاهد، عن…١٧٠٢من أصبح وهو جنب فليفطر١٧٠٣قالت: كان النبي صلى الله عليه وسلم يبيت جنبا. فيأتيه بلال، فيؤذنه…١٧٠٤يصبح، وهو جنب، يريد الصوم؟ قالت: كان رسول الله صلى الله عليه وسلم…١٧٠٥من صام الأبد، فلا صام ولا أفطر١٧٠٦لا صام من صام الأبد١٧٠٧هو كصوم الدهر أو كهيئة صوم الدهر١٧٠٨من صام ثلاثة أيام من كل شهر فذلك صوم الدهر١٧٠٩أنها قالت: كان رسول الله صلى الله عليه وسلم يصوم ثلاثة أيام من كل شهر…١٧١٠صلى الله عليه وسلم فقالت: كان يصوم حتى نقول قد صام . ويفطر حتى نقول:…١٧١١قال: كان رسول الله صلى الله عليه وسلم يصوم حتى نقول: لا يفطر. ويفطر…١٧١٢أحب الصيام إلى الله صيام داود. كان يصوم يوما ويفطر يوما. وأحب الصلاة…١٧١٣قال: قال عمر بن الخطاب: يا رسول الله! كيف بمن يصوم يومين ويفطر يوما؟…١٧١٤صام نوح الدهر إلا يوم الفطر ويوم الأضحى١٧١٥من صام ستة أيام بعد الفطر كان تمام السنة {من جاء بالحسنة فله عشر…١٧١٦من صام رمضان ثم أتبعه بست من شوال، كان كصوم الدهر١٧١٧من صام يوما في سبيل الله، باعد الله، بذلك اليوم، النار عن وجهه سبعين…١٧١٨من صام يوما في سبيل الله، زحزح الله وجهه عن النار سبعين خريفا١٧١٩أيام منى، أيام أكل وشرب١٧٢٠لا يدخل الجنة إلا نفس مسلمة. وإن هذه الأيام أيام أكل وشرب١٧٢١حدثنا أبو بكر بن أبي شيبة، حدثنا يحيى بن يعلى التيمي، عن عبد الملك بن…١٧٢٢يوم الفطر ويوم الأضحى. أما يوم الفطر، فيوم فطركم من صيامكم. ويوم…١٧٢٣وحفص بن غياث، عن الأعمش، عن أبي صالح، عن أبي هريرة، قال: نهى رسول…١٧٢٤حدثنا هشام بن عمار، حدثنا سفيان بن عيينة، عن عبد الحميد بن جبير بن…١٧٢٥قال: قلما رأيت رسول الله صلى الله عليه وسلم يفطر يوم الجمعة .١٧٢٦لا تصوموا يوم السبت إلا فيما افترض عليكم. فإن لم يجد أحدكم إلا عود…١٧٢٧ما من أيام العمل الصالح فيها أحب إلى الله من هذه الأيام١٧٢٨ما من أيام الدنيا أيام، أحب إلى الله سبحانه أن يتعبد له فيها، من أيام…١٧٢٩قالت: ما رأيت رسول الله صلى الله عليه وسلم صام العشر قط .١٧٣٠صيام يوم عرفة، إني أحتسب على الله أن يكفر السنة التي قبله والتي بعده١٧٣١من صام يوم عرفة، غفر له سنة أمامه وسنة بعده١٧٣٢وعلي بن محمد، قالا حدثنا وكيع، حدثني حوشب بن عقيل، حدثني مهدي العبدي،…١٧٣٣قالت: كان رسول الله صلى الله عليه وسلم يصوم عاشوراء، ويأمر بصيامه .١٧٣٤قال: قدم النبي صلى الله عليه وسلم المدينة فوجد اليهود صياما. فقال: "…١٧٣٥منكم أحد طعم اليوم؟١٧٣٦لئن بقيت إلى قابل لأصومن اليوم التاسع١٧٣٧كان يوما يصومه أهل الجاهلية. فمن أحب منكم أن يصومه فليصمه، ومن كرهه…١٧٣٨صيام يوم عاشوراء إني أحتسب على الله أن يكفر السنة التي قبله١٧٣٩رسول الله صلى الله عليه وسلم فقالت كان يتحرى صيام الاثنين والخميس .١٧٤٠إن يوم الاثنين والخميس يغفر الله فيهما لكل مسلم إلا مهتجرين يقول…١٧٤١فما لي أرى جسمك ناحلا١٧٤٢شهر الله الذي تدعونه المحرم١٧٤٣حدثنا إبراهيم بن المنذر الحزامي، حدثنا داود بن عطاء، حدثني زيد بن عبد…١٧٤٤أن أسامة بن زيد، كان يصوم أشهر الحرم . فقال له رسول الله صلى الله…١٧٤٥لكل شىء زكاة وزكاة الجسد الصوم١٧٤٦من فطر صائما كان له مثل أجرهم من غير أن ينقص من أجورهم شيئا١٧٤٧أفطر عندكم الصائمون وأكل طعامكم الأبرار وصلت عليكم الملائكة١٧٤٨الصائم إذا أكل عنده الطعام صلت عليه الملائكة١٧٤٩حدثنا محمد بن المصفى، حدثنا بقية، حدثنا محمد بن عبد الرحمن، عن سليمان…١٧٥٠إذا دعي أحدكم إلى طعام وهو صائم فليقل إني صائم١٧٥١من دعي إلى طعام وهو صائم فليجب فإن شاء طعم وإن شاء ترك١٧٥٢ثلاثة لا ترد دعوتهم الإمام العادل والصائم حتى يفطر ودعوة المظلوم…١٧٥٣إن للصائم عند فطره لدعوة ما ترد١٧٥٤قال كان النبي صلى الله عليه وسلم لا يخرج يوم الفطر حتى يطعم تمرات .١٧٥٥قال كان النبي صلى الله عليه وسلم لا يغدو يوم الفطر حتى يغدي أصحابه من…١٧٥٦أن رسول الله صلى الله عليه وسلم كان لا يخرج يوم الفطر حتى يأكل وكان…١٧٥٧من مات وعليه صيام شهر فليطعم عنه مكان كل يوم مسكين١٧٥٨أرأيت لو كان على أختك دين أكنت تقضينه١٧٥٩قال جاءت امرأة إلى النبي صلى الله عليه وسلم فقالت يا رسول الله إن أمي…١٧٦٠قدموا على رسول الله صلى الله عليه وسلم بإسلام ثقيف . قال وقدموا عليه…١٧٦١لا تصوم المرأة وزوجها شاهد يوما من غير شهر رمضان إلا بإذنه١٧٦٢قال نهى رسول الله صلى الله عليه وسلم النساء أن يصمن إلا بإذن أزواجهن .١٧٦٣إذا نزل الرجل بقوم فلا يصوم إلا بإذنهم١٧٦٤الطاعم الشاكر بمنزلة الصائم الصابر١٧٦٥الطاعم الشاكر له مثل أجر الصائم الصابر١٧٦٦إني أريت ليلة القدر فأنسيتها فالتمسوها في العشر الأواخر في الوتر١٧٦٧قالت كان النبي صلى الله عليه وسلم يجتهد في العشر الأواخر ما لا يجتهد…١٧٦٨قالت كان النبي صلى الله عليه وسلم إذا دخلت العشر أحيا الليل وشد…١٧٦٩قال كان النبي صلى الله عليه وسلم يعتكف كل عام عشرة أيام فلما كان…١٧٧٠أن النبي صلى الله عليه وسلم كان يعتكف العشر الأواخر من رمضان فسافر…١٧٧١قالت كان النبي صلى الله عليه وسلم إذا أراد أن يعتكف صلى الصبح ثم دخل…١٧٧٢أنه كان عليه نذر ليلة في الجاهلية يعتكفها فسأل النبي صلى الله عليه…١٧٧٣أن نافعا، حدثه عن عبد الله بن عمر، أن رسول الله صلى الله عليه وسلم…١٧٧٤حدثنا محمد بن يحيى، حدثنا نعيم بن حماد، حدثنا ابن المبارك، عن عيسى بن…١٧٧٥أن رسول الله صلى الله عليه وسلم اعتكف في قبة تركية على سدتها قطعة…١٧٧٦وعمرة بنت عبد الرحمن، أن عائشة، قالت إن كنت لأدخل البيت للحاجة…١٧٧٧المعتكف يتبع الجنازة ويعود المريض١٧٧٨حدثنا علي بن محمد، حدثنا وكيع، عن هشام بن عروة، عن أبيه، عن عائشة،…١٧٧٩على رسلكما إنها صفية بنت حيى١٧٨٠قال قالت عائشة اعتكفت مع رسول الله صلى الله عليه وسلم امرأة من نسائه…١٧٨١هو يعكف الذنوب ويجرى له من الحسنات كعامل الحسنات كلها١٧٨٢من قام ليلتى العيدين لله محتسبا لم يمت قلبه يوم تموت القلوب
