سنن ابن ماجه

كتاب الطب

١١٤ حديثًا
٣٤٣٦عباد الله وضع الله الحرج إلا من اقترض من عرض أخيه شيئا فذاك الذي حرج٣٤٣٧قال سئل رسول الله صلى الله عليه وسلم أرأيت أدوية نتداوى بها ورقى…٣٤٣٨ما أنزل الله داء إلا أنزل له دواء٣٤٣٩ما أنزل الله داء إلا أنزل له شفاء٣٤٤٠أن النبي صلى الله عليه وسلم عاد رجلا فقال له " ما تشتهي " . فقال…٣٤٤١أتشتهي شيئا أتشتهي كعكا٣٤٤٢مه يا علي إنك ناقه٣٤٤٣صهيب قال قدمت على النبي صلى الله عليه وسلم وبين يديه خبز وتمر فقال…٣٤٤٤لا تكرهوا مرضاكم على الطعام والشراب فإن الله يطعمهم ويسقيهم٣٤٤٥إنه ليرتو فؤاد الحزين ويسرو عن فؤاد السقيم كما تسرو إحداكن الوسخ عن…٣٤٤٦عليكم بالبغيض النافع التلبينة٣٤٤٧إن في الحبة السوداء شفاء من كل داء إلا السام٣٤٤٨عليكم بهذه الحبة السوداء فإن فيها شفاء من كل داء إلا السام٣٤٤٩إن هذه الحبة السوداء شفاء من كل داء إلا أن يكون السام٣٤٥٠من لعق العسل ثلاث غدوات كل شهر لم يصبه عظيم من البلاء٣٤٥١حدثنا أبو بشر، بكر بن خلف حدثنا عمر بن سهل، حدثنا أبو حمزة العطار، عن…٣٤٥٢عليكم بالشفاءين العسل والقرآن٣٤٥٣الكمأة من المن وماؤها شفاء للعين والعجوة من الجنة وهي شفاء من الجنة٣٤٥٤الكمأة من المن الذي أنزل الله على بني إسرائيل وماؤها شفاء للعين٣٤٥٥الكمأة من المن والعجوة من الجنة وهي شفاء من السم٣٤٥٦العجوة والصخرة من الجنة٣٤٥٧عليكم بالسنى والسنوت فإن فيهما شفاء من كل داء إلا السام . قيل يا رسول…٣٤٥٨فذكر نحوه وقال فيه اشكمت درد . يعني تشتكي بطنك بالفارسية . قال أبو…٣٤٥٩حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن يونس بن أبي إسحاق، عن…٣٤٦٠من شرب سما فقتل نفسه فهو يتحساه في نار جهنم خالدا مخلدا فيها أبدا٣٤٦١بماذا كنت تستمشين٣٤٦٢علام تدغرن أولادكن بهذا العلاق عليكم بهذا العود الهندي فإن فيه سبعة…٣٤٦٣شفاء عرق النسا ألية شاة أعرابية تذاب ثم تجزأ ثلاثة أجزاء ثم يشرب على…٣٤٦٤ومحمد بن الصباح، قالا حدثنا عبد العزيز بن أبي حازم، عن أبيه، عن سهل…٣٤٦٥قال إني لأعرف يوم أحد من جرح وجه رسول الله صلى الله عليه وسلم ومن كان…٣٤٦٦من تطبب ولم يعلم منه طب قبل ذلك فهو ضامن٣٤٦٧قال نعت رسول الله صلى الله عليه وسلم من ذات الجنب ورسا وقسطا وزيتا…٣٤٦٨عليكم بالعود الهندي - يعني به الكست - فإن فيه سبعة أشفية منها ذات…٣٤٦٩لا تسبها فإنها تنفي الذنوب كما تنفي النار خبث الحديد٣٤٧٠أبشر فإن الله يقول هي ناري أسلطها على عبدي المؤمن في الدنيا لتكون حظه…٣٤٧١إن الحمى من فيح جهنم فابردوها بالماء٣٤٧٢إن شدة الحمى من فيح جهنم فابردوها بالماء٣٤٧٣الحمى من فيح جهنم فابردوها بالماء٣٤٧٤أنها كانت تؤتى بالمرأة الموعوكة فتدعو بالماء فتصبه في جيبها وتقول إن…٣٤٧٥الحمى كير من كير جهنم فنحوها عنكم بالماء البارد٣٤٧٦إن كان في شىء مما تداوون به خير فالحجامة٣٤٧٧ما مررت ليلة أسري بي بملإ من الملائكة إلا كلهم يقول لي عليك يا محمد…٣٤٧٨نعم العبد الحجام يذهب بالدم ويخف الصلب ويجلو البصر٣٤٧٩ما مررت ليلة أسري بي بملإ إلا قالوا يا محمد مر أمتك بالحجامة٣٤٨٠أن أم سلمة، زوج النبي صلى الله عليه وسلم استأذنت رسول الله صلى الله…٣٤٨١يقول احتجم رسول الله صلى الله عليه وسلم بلحى جمل وهو محرم وسط رأسه .٣٤٨٢قال نزل جبريل على النبي صلى الله عليه وسلم بحجامة الأخدعين والكاهل .٣٤٨٣أن النبي صلى الله عليه وسلم احتجم في الأخدعين وعلى الكاهل .٣٤٨٤من أهراق منه هذه الدماء فلا يضره أن لا يتداوى بشىء لشىء٣٤٨٥أن النبي صلى الله عليه وسلم سقط من فرسه على جذع فانفكت قدمه . قال…٣٤٨٦من أراد الحجامة فليتحر سبعة عشر أو تسعة عشر أو إحدى وعشرين ولا يتبيغ…٣٤٨٧الحجامة على الريق أمثل وفيه شفاء وبركة وتزيد في العقل وفي الحفظ…٣٤٨٨الحجامة على الريق أمثل وهي تزيد في العقل وتزيد في الحفظ وتزيد الحافظ…٣٤٨٩من اكتوى أو استرقى فقد برئ من التوكل٣٤٩٠ويونس، عن الحسن، عن عمران بن الحصين، قال نهى رسول الله صلى الله عليه…٣٤٩١الشفاء في ثلاث شربة عسل وشرطة محجم وكية بنار وأنهى أمتي عن الكى٣٤٩٢لأبلغن أو لأبلين في أبي أمامة عذرا٣٤٩٣قال مرض أبى بن كعب مرضا فأرسل إليه النبي طبيبا فكواه على أكحله .٣٤٩٤أن رسول الله صلى الله عليه وسلم كوى سعد بن معاذ في أكحله مرتين .٣٤٩٥عليكم بالإثمد فإنه يجلو البصر وينبت الشعر٣٤٩٦عليكم بالإثمد عند النوم فإنه يجلو البصر وينبت الشعر٣٤٩٧خير أكحالكم الإثمد يجلو البصر وينبت الشعر٣٤٩٨من اكتحل فليوتر من فعل فقد أحسن ومن لا فلا حرج٣٤٩٩قال كانت للنبي صلى الله عليه وسلم مكحلة يكتحل منها ثلاثا في كل عين .٣٥٠٠حدثنا أبو بكر بن أبي شيبة، حدثنا عفان، حدثنا حماد بن سلمة، أنبأنا…٣٥٠١خير الدواء القرآن٣٥٠٢عبيد الله حدثتني جدتي، سلمى أم رافع مولاة رسول الله صلى الله عليه…٣٥٠٣لو خرجتم إلى ذود لنا فشربتم من ألبانها وأبوالها٣٥٠٤في أحد جناحى الذباب سم وفي الآخر شفاء فإذا وقع في الطعام فامقلوه فيه…٣٥٠٥إذا وقع الذباب في شرابكم فليغمسه فيه ثم ليطرحه فإن في أحد جناحيه داء…٣٥٠٦حدثنا محمد بن عبد الله بن نمير، حدثنا معاوية بن هشام، حدثنا عمار بن…٣٥٠٧قال قال رسول الله صلى الله عليه وسلم " العين حق " .٣٥٠٨استعيذوا بالله فإن العين حق٣٥٠٩حدثنا هشام بن عمار، حدثنا سفيان، عن الزهري، عن أبي أمامة بن سهل بن…٣٥١٠نعم فلو كان شىء سابق القدر سبقته العين٣٥١١قال كان رسول الله صلى الله عليه وسلم يتعوذ من عين الجان وأعين الإنس…٣٥١٢ومسعر، عن معبد بن خالد، عن عبد الله بن شداد، عن عائشة، أن النبي صلى…٣٥١٣لا رقية إلا من عين أو حمة٣٥١٤أن خالدة بنت أنس أم بني حزم الساعدية، جاءت إلى النبي صلى الله عليه…٣٥١٥حدثنا علي بن أبي الخصيب، حدثنا يحيى بن عيسى، عن الأعمش، عن أبي سفيان،…٣٥١٦أن النبي صلى الله عليه وسلم رخص في الرقية من الحمة والعين والنملة .٣٥١٧وهناد بن السري، قالا حدثنا أبو الأحوص، عن مغيرة، عن إبراهيم، عن…٣٥١٨أما إنه لو قال حين أمسى أعوذ بكلمات الله التامات من شر ما خلق - ما…٣٥١٩قال عرضت النهشة من الحية على رسول الله صلى الله عليه وسلم فأمر بها .٣٥٢٠أذهب الباس رب الناس واشف أنت الشافي لا شفاء إلا شفاؤك شفاء لا يغادر…٣٥٢١بسم الله بتربة أرضنا بريقة بعضنا ليشفى سقيمنا بإذن ربنا٣٥٢٢اجعل يدك اليمنى عليه وقل بسم الله أعوذ بعزة الله وقدرته من شر ما أجد…٣٥٢٣أن جبرائيل، أتى النبي صلى الله عليه وسلم فقال يا محمد اشتكيت قال "…٣٥٢٤ألا أرقيك برقية جاءني بها جبرائيل٣٥٢٥أعوذ بكلمات الله التامة من كل شيطان وهامة ومن كل عين لامة٣٥٢٦بسم الله الكبير أعوذ بالله العظيم من شر عرق نعار ومن شر حر النار٣٥٢٧أنه سمع جنادة بن أبي أمية، قال سمعت عبادة بن الصامت، يقول أتى جبرائيل…٣٥٢٨وعلي بن ميمون الرقي، وسهل بن أبي سهل، قالوا حدثنا وكيع، عن مالك بن…٣٥٢٩أن النبي صلى الله عليه وسلم كان إذا اشتكى يقرأ على نفسه المعوذات…٣٥٣٠إن الرقى والتمائم والتولة شرك٣٥٣١حدثنا علي بن أبي الخصيب، حدثنا وكيع، عن مبارك، عن الحسن، عن عمران بن…٣٥٣٢ائتوني بشىء من ماء٣٥٣٣حدثنا محمد بن عبيد بن عتبة بن عبد الرحمن الكندي حدثنا علي بن ثابت…٣٥٣٤قالت أمر النبي صلى الله عليه وسلم بقتل ذي الطفيتين فإنه يلتمس البصر…٣٥٣٥اقتلوا الحيات واقتلوا ذا الطفيتين والأبتر فإنهما يلتمسان البصر…٣٥٣٦قال كان النبي صلى الله عليه وسلم يعجبه الفأل الحسن ويكره الطيرة .٣٥٣٧لا عدوى ولا طيرة وأحب الفأل الصالح٣٥٣٨الطيرة شرك وما منا إلا ولكن الله يذهبه بالتوكل٣٥٣٩لا عدوى ولا طيرة ولا هامة ولا صفر٣٥٤٠لا عدوى ولا طيرة ولا هامة٣٥٤١لا يورد الممرض على المصح٣٥٤٢كل ثقة بالله وتوكلا على الله٣٥٤٣لا تديموا النظر إلى المجذومين٣٥٤٤ارجع فقد بايعناك٣٥٤٥يا عائشة أشعرت أن الله قد أفتاني فيما استفتيته فيه جاءني رجلان فجلس…٣٥٤٦ما أصابني شىء منها إلا وهو مكتوب على وآدم في طينته٣٥٤٧لو أن أحدكم إذا نزل منزلا قال أعوذ بكلمات الله التامة من شر ما خلق -…٣٥٤٨قال لما استعملني رسول الله صلى الله عليه وسلم على الطائف جعل يعرض لي…٣٥٤٩حدثنا هارون بن حيان، حدثنا إبراهيم بن موسى، أنبأنا عبدة بن سليمان،…
