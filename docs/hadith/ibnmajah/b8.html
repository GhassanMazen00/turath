سنن ابن ماجه

كتاب الزكاة

٦٢ حديثًا
١٧٨٣إنك تأتي قوما أهل كتاب فادعهم إلى شهادة أن لا إله إلا الله وأني رسول…١٧٨٤ما من أحد لا يؤدي زكاة ماله إلا مثل له يوم القيامة شجاعا أقرع حتى…١٧٨٥ما من صاحب إبل ولا غنم ولا بقر لا يؤدي زكاتها إلا جاءت يوم القيامة…١٧٨٦تأتي الإبل التي لم تعط الحق منها تطأ صاحبها بأخفافها وتأتي البقر…١٧٨٧مولى عمر بن الخطاب قال خرجت مع عبد الله بن عمر فلحقه أعرابي فقال له…١٧٨٨إذا أديت زكاة مالك فقد قضيت ما عليك١٧٨٩ليس في المال حق سوى الزكاة١٧٩٠إني قد عفوت لكم عن صدقة الخيل والرقيق ولكن هاتوا ربع العشور من كل…١٧٩١ومحمد بن يحيى، قالا حدثنا عبيد الله بن موسى، أنبأنا إبراهيم بن…١٧٩٢لا زكاة في مال حتى يحول عليه الحول١٧٩٣لا صدقة فيما دون خمسة أوساق من التمر ولا فيما دون خمس أواق ولا فيما…١٧٩٤ليس فيما دون خمس ذود صدقة وليس فيما دون خمس أواق صدقة وليس فيما دون…١٧٩٥أن العباس، سأل النبي صلى الله عليه وسلم في تعجيل صدقته قبل أن تحل…١٧٩٦اللهم صل على آل أبي أوفى١٧٩٧إذا أعطيتم الزكاة فلا تنسوا ثوابها أن تقولوا اللهم اجعلها مغنما ولا…١٧٩٨في خمس من الإبل شاة وفي عشر شاتان وفي خمس عشرة ثلاث شياه وفي عشرين…١٧٩٩ليس فيما دون خمس من الإبل صدقة وليس في الأربع شىء فإذا بلغت خمسا…١٨٠٠ومحمد بن يحيى، ومحمد بن مرزوق، قالوا حدثنا محمد بن عبد الله بن…١٨٠١حدثنا علي بن محمد، حدثنا وكيع، حدثنا شريك، عن عثمان الثقفي، عن أبي…١٨٠٢لا يرجع المصدق إلا عن رضا١٨٠٣قال بعثني رسول الله صلى الله عليه وسلم إلى اليمن وأمرني أن آخذ من…١٨٠٤في ثلاثين من البقر تبيع أو تبيعة وفي أربعين مسنة١٨٠٥في أربعين شاة شاة إلى عشرين ومائة فإذا زادت واحدة ففيها شاتان إلى…١٨٠٦تؤخذ صدقات المسلمين على مياههم١٨٠٧في أربعين شاة شاة إلى عشرين ومائة فإذا زادت واحدة ففيها شاتان إلى…١٨٠٨المعتدي في الصدقة كمانعها١٨٠٩العامل على الصدقة بالحق كالغازي في سبيل الله حتى يرجع إلى بيته١٨١٠أنه من غل منها بعيرا أو شاة أتي به يوم القيامة يحمله١٨١١حدثنا أبو بدر، عباد بن الوليد حدثنا أبو عتاب، حدثني إبراهيم بن عطاء،…١٨١٢ليس على المسلم في عبده ولا في فرسه صدقة١٨١٣تجوزت لكم عن صدقة الخيل والرقيق١٨١٤خذ الحب من الحب والشاة من الغنم والبعير من الإبل والبقرة من البقر١٨١٥قال إنما سن رسول الله صلى الله عليه وسلم الزكاة في هذه الخمسة في…١٨١٦فيما سقت السماء والعيون العشر وفيما سقي بالنضح نصف العشر١٨١٧فيما سقت السماء والأنهار والعيون أو كان بعلا العشر وفيما سقي بالسواني…١٨١٨قال بعثني رسول الله صلى الله عليه وسلم إلى اليمن وأمرني أن آخذ مما…١٨١٩والزبير بن بكار، قالا حدثنا ابن نافع، حدثنا محمد بن صالح التمار، عن…١٨٢٠حدثنا موسى بن مروان الرقي، حدثنا عمر بن أيوب، عن جعفر بن برقان، عن…١٨٢١لو شاء رب هذه الصدقة تصدق بأطيب منها إن رب هذه الصدقة يأكل الحشف يوم…١٨٢٢حدثنا أحمد بن محمد بن يحيى بن سعيد القطان، حدثنا عمرو بن محمد…١٨٢٣وعلي بن محمد، قالا حدثنا وكيع، عن سعيد بن عبد العزيز، عن سليمان بن…١٨٢٤حدثنا محمد بن يحيى، حدثنا نعيم بن حماد، حدثنا ابن المبارك، حدثنا…١٨٢٥أن رسول الله صلى الله عليه وسلم أمر بزكاة الفطر صاعا من تمر أو صاعا…١٨٢٦قال فرض رسول الله صلى الله عليه وسلم صدقة الفطر صاعا من شعير أو صاعا…١٨٢٧وأحمد بن الأزهر، قالا حدثنا مروان بن محمد، حدثنا أبو يزيد الخولاني،…١٨٢٨حدثنا علي بن محمد، حدثنا وكيع، عن سفيان، عن سلمة بن كهيل، عن القاسم…١٨٢٩حدثنا علي بن محمد، حدثنا وكيع، عن داود بن قيس الفراء، عن عياض بن عبد…١٨٣٠مؤذن رسول الله صلى الله عليه وسلم أن رسول الله صلى الله عليه وسلم أمر…١٨٣١قال بعثني رسول الله صلى الله عليه وسلم إلى البحرين - أو إلى هجر -…١٨٣٢الوسق ستون صاعا١٨٣٣الوسق ستون صاعا١٨٣٤لها أجران أجر الصدقة وأجر القرابة١٨٣٥حدثنا أبو بكر بن أبي شيبة، حدثنا يحيى بن آدم، حدثنا حفص بن غياث، عن…١٨٣٦لأن يأخذ أحدكم أحبله فيأتي الجبل فيجيء بحزمة حطب على ظهره فيبيعها…١٨٣٧من يتقبل لي بواحدة وأتقبل له بالجنة١٨٣٨من سأل الناس أموالهم تكثرا فإنما يسأل جمر جهنم فليستقل منه أو ليكثر١٨٣٩لا تحل الصدقة لغني ولا لذي مرة سوي١٨٤٠من سأل وله ما يغنيه جاءت مسألته يوم القيامة خدوشا أو خموشا أو كدوحا…١٨٤١لا تحل الصدقة لغني إلا لخمسة لعامل عليها أو لغاز في سبيل الله أو لغني…١٨٤٢ما تصدق أحد بصدقة من طيب ولا يقبل الله إلا الطيب إلا أخذها الرحمن…١٨٤٣ما منكم من أحد إلا سيكلمه ربه ليس بينه وبينه ترجمان فينظر أمامه…١٨٤٤الصدقة على المسكين صدقة وهي على ذي القرابة اثنتان صدقة وصلة
