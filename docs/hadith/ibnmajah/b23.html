سنن ابن ماجه

كتاب الفرائض

٣٤ حديثًا
٢٧١٩يا أبا هريرة تعلموا الفرائض وعلموها فإنه نصف العلم وهو ينسى وهو أول…٢٧٢٠أعط ابنتى سعد ثلثى ماله وأعط امرأته الثمن وخذ أنت ما بقي٢٧٢١حدثنا علي بن محمد، حدثنا وكيع، حدثنا سفيان، عن أبي قيس الأودي، عن…٢٧٢٢حدثنا أبو بكر بن أبي شيبة، حدثنا شبابة، حدثنا يونس بن أبي إسحاق، عن…٢٧٢٣قال أبو الحسن القطان حدثنا أبو حاتم، حدثنا ابن الطباع، حدثنا هشيم، عن…٢٧٢٤حدثنا أحمد بن عمرو بن السرح المصري، أنبأنا عبد الله بن وهب، أنبأنا…٢٧٢٥أن رسول الله صلى الله عليه وسلم ورث جدة سدسا .٢٧٢٦يا عمر تكفيك آية الصيف التي نزلت في آخر سورة النساء٢٧٢٧قال قال عمر بن الخطاب ثلاث لأن يكون رسول الله صلى الله عليه وسلم…٢٧٢٨سمع جابر بن عبد الله، يقول مرضت فأتاني رسول الله صلى الله عليه وسلم…٢٧٢٩لا يرث المسلم الكافر ولا الكافر المسلم٢٧٣٠وهل ترك لنا عقيل من رباع أو دور٢٧٣١لا يتوارث أهل ملتين٢٧٣٢ما أحرز الولد أو الوالد فهو لعصبته من كان٢٧٣٣أعطوا ميراثه رجلا من أهل قريته٢٧٣٤- قال محمد يعني ابن أبي ليلى وهي أخت ابن شداد لأمه - قالت مات مولاى…٢٧٣٥حدثنا محمد بن رمح، أنبأنا الليث بن سعد، عن إسحاق بن أبي فروة، عن ابن…٢٧٣٦المرأة ترث من دية زوجها وماله وهو يرث من ديتها ومالها ما لم يقتل…٢٧٣٧الله ورسوله مولى من لا مولى له والخال وارث من لا وارث له٢٧٣٨من ترك مالا فلورثته ومن ترك كلا فإلينا - وربما قال فإلى الله وإلى…٢٧٣٩قال قضى رسول الله صلى الله عليه وسلم أن أعيان بني الأم يتوارثون دون…٢٧٤٠اقسموا المال بين أهل الفرائض على كتاب الله فما تركت الفرائض فلأولى…٢٧٤١قال مات رجل على عهد رسول الله صلى الله عليه وسلم ولم يدع له وارثا إلا…٢٧٤٢المرأة تحوز ثلاث مواريث عتيقها ولقيطها وولدها الذي لاعنت عليه٢٧٤٣أيما امرأة ألحقت بقوم من ليس منهم فليست من الله في شىء ولن يدخلها…٢٧٤٤كفر بامرئ ادعاء نسب لا يعرفه أو جحده وإن دق٢٧٤٥من عاهر أمة أو حرة فولده ولد زنا لا يرث ولا يورث٢٧٤٦كل مستلحق استلحق بعد أبيه الذي يدعى له ادعاه ورثته من بعده فقضى أن من…٢٧٤٧وسفيان، عن عبد الله بن دينار، عن ابن عمر، قال نهى رسول الله صلى الله…٢٧٤٨حدثنا محمد بن عبد الملك بن أبي الشوارب، حدثنا يحيى بن سليم الطائفي،…٢٧٤٩ما كان من ميراث قسم في الجاهلية فهو على قسمة الجاهلية وما كان من…٢٧٥٠إذا استهل الصبي صلي عليه وورث٢٧٥١لا يرث الصبي حتى يستهل صارخا٢٧٥٢هو أولى الناس بمحياه ومماته
