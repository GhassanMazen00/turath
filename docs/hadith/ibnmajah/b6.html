كتاب الجنائز

٢٠٥ حديثًا•صفحة ١ من ٢
١٤٣٣للمسلم على المسلم ستة بالمعروف يسلم عليه إذا لقيه ويجيبه إذا دعاه…١٤٣٤للمسلم على المسلم أربع خلال يشمته إذا عطس ويجيبه إذا دعاه ويشهده إذا…١٤٣٥خمس من حق المسلم على المسلم رد التحية وإجابة الدعوة وشهود الجنازة…١٤٣٦حدثنا محمد بن عبد الأعلى الصنعاني، حدثنا سفيان، قال سمعت محمد بن…١٤٣٧قال كان النبي صلى الله عليه وسلم لا يعود مريضا إلا بعد ثلاث .١٤٣٨إذا دخلتم على المريض فنفسوا له في الأجل فإن ذلك لا يرد شيئا وهو يطيب…١٤٣٩أن النبي صلى الله عليه وسلم عاد رجلا فقال " ما تشتهي " . قال أشتهي…١٤٤٠أتشتهي شيئا أتشتهي كعكا١٤٤١إذا دخلت على مريض فمره أن يدعو لك فإن دعاءه كدعاء الملائكة١٤٤٢من أتى أخاه المسلم عائدا مشى في خرافة الجنة حتى يجلس فإذا جلس غمرته…١٤٤٣من عاد مريضا نادى مناد من السماء طبت وطاب ممشاك وتبوأت من الجنة منزلا١٤٤٤لقنوا موتاكم لا إله إلا الله١٤٤٥لقنوا موتاكم لا إله إلا الله١٤٤٦لقنوا موتاكم لا إله إلا الله الحليم الكريم سبحان الله رب العرش العظيم…١٤٤٧إذا حضرتم المريض أو الميت فقولوا خيرا فإن الملائكة يؤمنون على ما…١٤٤٨اقرءوها عند موتاكم١٤٤٩إن أرواح المؤمنين في طير خضر تعلق في شجر الجنة١٤٥٠قال دخلت على جابر بن عبد الله وهو يموت فقلت اقرأ على رسول الله صلى…١٤٥١لا تبتئسي على حميمك فإن ذلك من حسناته١٤٥٢المؤمن يموت بعرق الجبين١٤٥٣قال سألت رسول الله صلى الله عليه وسلم متى تنقطع معرفة العبد من الناس…١٤٥٤إن الروح إذا قبض تبعه البصر١٤٥٥إذا حضرتم موتاكم فأغمضوا البصر فإن البصر يتبع الروح وقولوا خيرا فإن…١٤٥٦وعلي بن محمد، قالا حدثنا وكيع، عن سفيان، عن عاصم بن عبيد الله، عن…١٤٥٧والعباس بن عبد العظيم، وسهل بن أبي سهل، قالوا حدثنا يحيى بن سعيد، عن…١٤٥٨اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك بماء وسدر واجعلن في…١٤٥٩حدثتني حفصة، عن أم عطية، بمثل حديث محمد وكان في حديث حفصة " اغسلنها…١٤٦٠لا تبرز فخذك ولا تنظر إلى فخذ حى ولا ميت١٤٦١ليغسل موتاكم المأمونون١٤٦٢من غسل ميتا وكفنه وحنطه وحمله وصلى عليه ولم يفش عليه ما رأى خرج من…١٤٦٣من غسل ميتا فليغتسل١٤٦٤قالت لو كنت استقبلت من الأمر ما استدبرت ما غسل النبي صلى الله عليه…١٤٦٥بل أنا يا عائشة وارأساه١٤٦٦حدثنا سعيد بن يحيى بن الأزهر الواسطي، حدثنا أبو معاوية، حدثنا أبو…١٤٦٧قال لما غسل النبي صلى الله عليه وسلم ذهب يلتمس منه ما يلتمس من الميت…١٤٦٨إذا أنا مت فاغسلني بسبع قرب من بئري بئر غرس١٤٦٩أن النبي صلى الله عليه وسلم كفن في ثلاثة أثواب بيض يمانية ليس فيها…١٤٧٠قال كفن رسول الله صلى الله عليه وسلم في ثلاث رياط بيض سحولية .١٤٧١قال كفن رسول الله صلى الله عليه وسلم في ثلاثة أثواب قميصه الذي قبض…١٤٧٢خير ثيابكم البياض فكفنوا فيها موتاكم والبسوها١٤٧٣خير الكفن الحلة١٤٧٤إذا ولي أحدكم أخاه فليحسن كفنه١٤٧٥لا تدرجوه في أكفانه حتى أنظر إليه١٤٧٦حدثنا عمرو بن رافع، حدثنا عبد الله بن المبارك، عن حبيب بن سليم، عن…١٤٧٧أسرعوا بالجنازة فإن تكن صالحة فخير تقدمونها إليه وإن تكن غير ذلك فشر…١٤٧٨قال قال عبد الله بن مسعود من اتبع جنازة فليحمل بجوانب السرير كلها…١٤٧٩لتكن عليكم السكينة١٤٨٠ألا تستحيون أن ملائكة الله يمشون على أقدامهم وأنتم ركبان١٤٨١الراكب خلف الجنازة والماشي منها حيث شاء١٤٨٢وهشام بن عمار، وسهل بن أبي سهل، قالوا حدثنا سفيان، عن الزهري، عن…١٤٨٣وهارون بن عبد الله الحمال، قالا حدثنا محمد بن بكر البرساني، أنبأنا…١٤٨٤الجنازة متبوعة وليست بتابعة ليس منها من تقدمها١٤٨٥أبفعل الجاهلية تأخذون - أو بصنع الجاهلية تشبهون - لقد هممت أن أدعو…١٤٨٦لا تؤخروا الجنازة إذا حضرت١٤٨٧أن أبا بردة، حدثه قال أوصى أبو موسى الأشعري حين حضره الموت فقال لا…١٤٨٨من صلى عليه مائة من المسلمين غفر له١٤٨٩ما من أربعين من مؤمن يشفعون لمؤمن إلا شفعهم الله١٤٩٠ما صف صفوف ثلاثة من المسلمين على ميت إلا أوجب١٤٩١قال مر على النبي صلى الله عليه وسلم بجنازة فأثني عليها خيرا فقال "…١٤٩٢قال مر على النبي صلى الله عليه وسلم بجنازة - فأثني عليها خيرا في…١٤٩٣أن رسول الله صلى الله عليه وسلم صلى على امرأة ماتت في نفاسها فقام…١٤٩٤حدثنا نصر بن علي الجهضمي، حدثنا سعيد بن عامر، عن همام، عن أبي غالب،…١٤٩٥أن النبي صلى الله عليه وسلم قرأ على الجنازة بفاتحة الكتاب .١٤٩٦وإبراهيم بن المستمر، قالا حدثنا أبو عاصم، حدثنا حماد بن جعفر العبدي،…١٤٩٧إذا صليتم على الميت فأخلصوا له الدعاء١٤٩٨اللهم اغفر لحينا وميتنا وشاهدنا وغائبنا وصغيرنا وكبيرنا وذكرنا…١٤٩٩اللهم إن فلان بن فلان في ذمتك وحبل جوارك فقه من فتنة القبر وعذاب…١٥٠٠اللهم صل عليه واغفر له وارحمه وعافه واعف عنه واغسله بماء وثلج وبرد…١٥٠١حدثنا عبد الله بن سعيد، حدثنا حفص بن غياث، عن حجاج، عن أبي الزبير، عن…١٥٠٢. أن النبي صلى الله عليه وسلم صلى على عثمان بن مظعون وكبر عليه أربعا .١٥٠٣حدثنا علي بن محمد، حدثنا عبد الرحمن المحاربي، حدثنا الهجري، قال صليت…١٥٠٤ومحمد بن الصباح، وأبو بكر بن خلاد قالوا حدثنا يحيى بن اليمان، عن…١٥٠٥حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، ح وحدثنا يحيى بن…١٥٠٦أن رسول الله صلى الله عليه وسلم كبر خمسا .١٥٠٧الطفل يصلى عليه١٥٠٨إذا استهل الصبي صلي عليه وورث١٥٠٩صلوا على أطفالكم فإنهم من أفراطكم١٥١٠قال قلت لعبد الله بن أبي أوفى رأيت إبراهيم ابن رسول الله صلى الله…١٥١١إن له مرضعا في الجنة ولو عاش لكان صديقا نبيا . ولو عاش لعتقت أخواله…١٥١٢إن تمام رضاعه في الجنة١٥١٣قال أتي بهم رسول الله صلى الله عليه وسلم يوم أحد فجعل يصلي على عشرة…١٥١٤أيهم أكثر أخذا للقرآن١٥١٥أن رسول الله صلى الله عليه وسلم أمر بقتلى أحد أن ينزع عنهم الحديد…١٥١٦وسهل بن أبي سهل، قالا حدثنا سفيان بن عيينة، عن الأسود بن قيس، سمع…١٥١٧من صلى على جنازة في المسجد فليس له شىء١٥١٨قالت والله ما صلى رسول الله صلى الله عليه وسلم على سهيل ابن بيضاء إلا…١٥١٩وحين يقوم قائم الظهيرة حتى تميل الشمس، وحين تضيف للغروب حتى تغرب .١٥٢٠أن رسول الله صلى الله عليه وسلم أدخل رجلا قبره ليلا وأسرج في قبره .١٥٢١لا تدفنوا موتاكم بالليل إلا أن تضطروا١٥٢٢صلوا على موتاكم بالليل والنهار١٥٢٣حدثنا أبو بشر، بكر بن خلف حدثنا يحيى بن سعيد، عن عبيد الله، عن نافع،…١٥٢٤وسهل بن أبي سهل، قالا حدثنا يحيى بن سعيد، عن مجالد، عن عامر، عن جابر،…١٥٢٥صلوا على كل ميت وجاهدوا مع كل أمير١٥٢٦أن رجلا، من أصحاب النبي صلى الله عليه وسلم جرح فآذته الجراحة فدب إلى…١٥٢٧. أن امرأة، سوداء كانت تقم المسجد ففقدها رسول الله صلى الله عليه وسلم…١٥٢٨ألا آذنتموني بها١٥٢٩هلا آذنتموني بها١٥٣٠ما منعكم أن تعلموني١٥٣١ومحمد بن يحيى، قالا حدثنا أحمد بن حنبل، حدثنا غندر، عن شعبة، عن حبيب…١٥٣٢أن النبي صلى الله عليه وسلم صلى على ميت بعد ما دفن .١٥٣٣ألا آذنتموني بها١٥٣٤إن النجاشي قد مات١٥٣٥إن أخاكم النجاشي قد مات فصلوا عليه١٥٣٦إن أخاكم النجاشي قد مات فقوموا فصلوا عليه١٥٣٧صلوا على أخ لكم مات بغير أرضكم١٥٣٨أن النبي صلى الله عليه وسلم صلى على النجاشي فكبر أربعا .١٥٣٩من صلى على جنازة فله قيراط ومن انتظر حتى يفرغ منها فله قيراطان١٥٤٠من صلى على جنازة فله قيراط ومن شهد دفنها فله قيراطان١٥٤١من صلى على جنازة فله قيراط ومن شهدها حتى تدفن فله قيراطان والذي نفس…١٥٤٢إذا رأيتم الجنازة فقوموا لها حتى تخلفكم أو توضع١٥٤٣قوموا فإن للموت فزعا١٥٤٤حدثنا علي بن محمد، حدثنا وكيع، عن شعبة، عن محمد بن المنكدر، عن مسعود…١٥٤٥وعقبة بن مكرم، قالا حدثنا صفوان بن عيسى، حدثنا بشر بن رافع، عن عبد…١٥٤٦السلام عليكم دار قوم مؤمنين أنتم لنا فرط وإنا بكم لاحقون اللهم لا…١٥٤٧حدثنا محمد بن عباد بن آدم، حدثنا أبو أحمد، حدثنا سفيان، عن علقمة بن…١٥٤٨حدثنا محمد بن زياد، حدثنا حماد بن زيد، عن يونس بن خباب، عن المنهال بن…١٥٤٩حدثنا أبو كريب، حدثنا أبو خالد الأحمر، عن عمرو بن قيس، عن المنهال بن…١٥٥٠بسم الله وعلى ملة رسول الله١٥٥١قال سل رسول الله صلى الله عليه وسلم سعدا ورش على قبره ماء .١٥٥٢أن رسول الله صلى الله عليه وسلم أخذ من قبل القبلة واستقبل استقبالا…١٥٥٣حدثنا هشام بن عمار، حدثنا حماد بن عبد الرحمن الكلبي، حدثنا إدريس…١٥٥٤اللحد لنا والشق لغيرنا١٥٥٥اللحد لنا والشق لغيرنا١٥٥٦أنه قال: ألحدوا لي لحدا وانصبوا على اللبن نصبا كما فعل برسول الله صلى…١٥٥٧حدثنا محمود بن غيلان، حدثنا هاشم بن القاسم، حدثنا مبارك بن فضالة،…١٥٥٨قالت لما مات رسول الله صلى الله عليه وسلم اختلفوا في اللحد والشق حتى…١٥٥٩ارفقوا به رفق الله به إنه كان يحب الله ورسوله١٥٦٠احفروا وأوسعوا وأحسنوا١٥٦١أن رسول الله صلى الله عليه وسلم أعلم قبر عثمان بن مظعون بصخرة .١٥٦٢ومحمد بن زياد، قالا حدثنا عبد الوارث، عن أيوب، عن أبي الزبير، عن…١٥٦٣قال نهى رسول الله صلى الله عليه وسلم أن يكتب على القبر شىء .١٥٦٤أن النبي صلى الله عليه وسلم نهى أن يبنى على القبر .١٥٦٥أن رسول الله صلى الله عليه وسلم صلى على جنازة ثم أتى قبر الميت فحثى…١٥٦٦لأن يجلس أحدكم على جمرة تحرقه خير له من أن يجلس على قبر١٥٦٧لأن أمشي على جمرة أو سيف أو أخصف نعلي برجلي أحب إلى من أن أمشي على…١٥٦٨يا ابن الخصاصية ما تنقم على الله أصبحت تماشي رسول الله١٥٦٩زوروا القبور فإنها تذكركم الآخرة١٥٧٠أن رسول الله صلى الله عليه وسلم رخص في زيارة القبور .١٥٧١كنت نهيتكم عن زيارة القبور فزوروا القبور فإنها تزهد في الدنيا وتذكر…١٥٧٢استأذنت ربي في أن أستغفر لها فلم يأذن لي واستأذنت ربي في أن أزور…١٥٧٣قال جاء أعرابي إلى النبي صلى الله عليه وسلم فقال يا رسول الله إن أبي…١٥٧٤وقبيصة، كلهم عن سفيان، عن عبد الله بن عثمان بن خثيم، عن عبد الرحمن بن…١٥٧٥حدثنا أزهر بن مروان، حدثنا عبد الوارث، حدثنا محمد بن جحادة، عن أبي…١٥٧٦حدثنا محمد بن خلف العسقلاني أبو نصر، حدثنا محمد بن طالب، حدثنا أبو…١٥٧٧حدثنا أبو بكر بن أبي شيبة، حدثنا أبو أسامة، عن هشام، عن حفصة، عن أم…١٥٧٨حدثنا محمد بن المصفى الحمصي، حدثنا أحمد بن خالد، حدثنا إسرائيل، عن…١٥٧٩حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن يزيد بن عبد الله، مولى…١٥٨٠حدثنا هشام بن عمار، حدثنا إسماعيل بن عياش، حدثنا عبد الله بن دينار،…١٥٨١النياحة من أمر الجاهلية وإن النائحة إذا ماتت ولم تتب قطع الله لها…١٥٨٢النياحة على الميت من أمر الجاهلية فإن النائحة إن لم تتب قبل أن تموت…
