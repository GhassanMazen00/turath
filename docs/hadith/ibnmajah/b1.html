الله صلاة بغير طهور ولا صدقة من غلول٢٧٥مفتاح الصلاة الطهور وتحريمها التكبير وتحليلها التسليم٢٧٦مفتاح الصلاة الطهور وتحريمها التكبير وتحليلها التسليم٢٧٧استقيموا ولن تحصوا واعلموا أن خير أعمالكم الصلاة ولا يحافظ على الوضوء…٢٧٨استقيموا ولن تحصوا واعلموا أن من أفضل أعمالكم الصلاة ولا يحافظ على…٢٧٩استقيموا ونعما أن تستقيموا وخير أعمالكم الصلاة ولا يحافظ على الوضوء…٢٨٠إسباغ الوضوء شطر الإيمان والحمد لله تملأ الميزان والتسبيح والتكبير…٢٨١إن أحدكم إذا توضأ فأحسن الوضوء ثم أتى المسجد لا ينهزه إلا الصلاة لم…٢٨٢من توضأ فمضمض واستنشق خرجت خطاياه من فمه وأنفه فإذا غسل وجهه خرجت…٢٨٣إن العبد إذا توضأ فغسل يديه خرت خطاياه من يديه فإذا غسل وجهه خرت…٢٨٤غر محجلون بلق من آثار الطهور٢٨٥من توضأ مثل وضوئي هذا غفر له ما تقدم من ذنبه٢٨٦وأبي، عن الأعمش، ح وحدثنا علي بن محمد، حدثنا وكيع، عن سفيان، عن…٢٨٧لولا أن أشق على أمتي لأمرتهم بالسواك عند كل صلاة٢٨٨قال كان رسول الله صلى الله عليه وسلم يصلي بالليل ركعتين ركعتين ثم…٢٨٩تسوكوا فإن السواك مطهرة للفم مرضاة للرب وما جاءني جبريل إلا أوصاني…٢٩٠قال قلت أخبريني بأى، شىء كان النبي صلى الله عليه وسلم يبدأ إذا دخل…٢٩١قال إن أفواهكم طرق للقرآن فطيبوها بالسواك .٢٩٢الفطرة خمس - أو خمس من الفطرة - الختان والاستحداد وتقليم الأظفار ونتف…٢٩٣عشر من الفطرة قص الشارب وإعفاء اللحية والسواك والاستنشاق بالماء وقص…٢٩٤من الفطرة المضمضة والاستنشاق والسواك وقص الشارب وتقليم الأظفار ونتف…٢٩٥حدثنا بشر بن هلال الصواف، حدثنا جعفر بن سليمان، عن أبي عمران الجوني،…٢٩٦إن هذه الحشوش محتضرة فإذا دخل أحدكم فليقل اللهم إني أعوذ بك من الخبث…٢٩٧ستر ما بين الجن وعورات بني آدم إذا دخل الكنيف أن يقول بسم الله٢٩٨أعوذ بالله من الخبث والخبائث٢٩٩لا يعجز أحدكم إذا دخل مرفقه أن يقول اللهم إني أعوذ بك من الرجس النجس…٣٠٠نحوه .٣٠١الحمد لله الذي أذهب عني الأذى وعافاني٣٠٢أن رسول الله صلى الله عليه وسلم كان يذكر الله على كل أحيانه .٣٠٣أن النبي صلى الله عليه وسلم كان إذا دخل الخلاء وضع خاتمه .٣٠٤لا يبولن أحدكم في مستحمه فإن عامة الوسواس منه٣٠٥وهشيم، ووكيع، عن الأعمش، عن أبي وائل، عن حذيفة، أن رسول الله صلى الله…٣٠٦حدثنا إسحاق بن منصور، حدثنا أبو داود، حدثنا شعبة، عن عاصم، عن أبي…٣٠٧وسويد بن سعيد، وإسماعيل بن موسى السدي، قالوا حدثنا شريك، عن المقدام…٣٠٨يا عمر لا تبل قائما٣٠٩حدثنا يحيى بن الفضل، حدثنا أبو عامر، حدثنا عدي بن الفضل، عن علي بن…٣١٠إذا بال أحدكم فلا يمس ذكره بيمينه ولا يستنج بيمينه٣١١يقول ما تغنيت ولا تمنيت ولا مسست ذكري بيميني منذ بايعت بها رسول الله…٣١٢إذا استطاب أحدكم فلا يستطب بيمينه ليستنج بشماله٣١٣إنما أنا لكم مثل الوالد لولده أعلمكم إذا أتيتم الغائط فلا تستقبلوا…٣١٤ائتني بثلاثة أحجار٣١٥في الاستنجاء ثلاثة أحجار ليس فيها رجيع٣١٦والأعمش، عن إبراهيم، عن عبد الرحمن بن يزيد، عن سلمان، قال قال له بعض…٣١٧لا يبولن أحدكم مستقبل القبلة٣١٨أنه سمع أبا أيوب الأنصاري، يقول نهى رسول الله صلى الله عليه وسلم أن…٣١٩وقد صحب النبي صلى الله عليه وسلم قال نهى رسول الله صلى الله عليه وسلم…٣٢٠أنه شهد على رسول الله صلى الله عليه وسلم أنه نهى أن نستقبل القبلة…٣٢١أنه سمع أبا سعيد الخدري، يقول إن رسول الله صلى الله عليه وسلم نهاني…٣٢٢ومحمد بن يحيى، قالا حدثنا يزيد بن هارون، أنبأنا يحيى بن سعيد، أن محمد…٣٢٣فذكر نحوه .٣٢٤أراهم قد فعلوها استقبلوا بمقعدتي القبلة٣٢٥قال نهى رسول الله صلى الله عليه وسلم أن نستقبل القبلة ببول فرأيته قبل…٣٢٦إذا بال أحدكم فلينتر ذكره ثلاث مرات٣٢٧قالت انطلق النبي صلى الله عليه وسلم يبول فاتبعه عمر بماء فقال " ما…٣٢٨اتقوا الملاعن الثلاث البراز في الموارد والظل وقارعة الطريق٣٢٩إياكم والتعريس على جواد الطريق والصلاة عليها فإنها مأوى الحيات…٣٣٠أن النبي صلى الله عليه وسلم نهى أن يصلى على قارعة الطريق أو يضرب…٣٣١قال كان النبي صلى الله عليه وسلم إذا ذهب المذهب أبعد .٣٣٢قال كنت مع النبي صلى الله عليه وسلم في سفر فتنحى لحاجته ثم جاء فدعا…٣٣٣أن النبي صلى الله عليه وسلم كان إذا ذهب إلى الغائط أبعد .٣٣٤ومحمد بن بشار، قالا حدثنا يحيى بن سعيد القطان، عن أبي جعفر الخطمي، -…٣٣٥حدثنا أبو بكر بن أبي شيبة، حدثنا عبيد الله بن موسى، أنبأنا إسماعيل بن…٣٣٦أن رسول الله صلى الله عليه وسلم كان إذا أراد الحاجة أبعد .٣٣٧من استجمر فليوتر من فعل ذلك فقد أحسن ومن لا فلا حرج ومن تخلل فليلفظ…٣٣٨ومن اكتحل فليوتر من فعل فقد أحسن ومن لا فلا حرج ومن لاك فليبتلع٣٣٩ائت تلك الأشاءتين٣٤٠قال كان أحب ما استتر به النبي صلى الله عليه وسلم لحاجته هدف أو حائش…٣٤١قال عدل رسول الله صلى الله عليه وسلم إلى الشعب فبال حتى أني آوي له من…٣٤٢لا يتناجى اثنان على غائطهما ينظر كل واحد منهما إلى عورة صاحبه فإن…٣٤٣حدثنا محمد بن رمح، أنبأنا الليث بن سعد، عن أبي الزبير، عن جابر، عن…٣٤٤لا يبولن أحدكم في الماء الراكد٣٤٥لا يبولن أحدكم في الماء الناقع٣٤٦ويحك أما علمت ما أصاب صاحب بني إسرائيل كانوا إذا أصابهم البول قرضوه…٣٤٧إنهما ليعذبان وما يعذبان في كبير أما أحدهما فكان لا يستنزه من بوله…٣٤٨أكثر عذاب القبر من البول٣٤٩إنهما ليعذبان وما يعذبان في كبير أما أحدهما فيعذب في البول وأما الآخر…٣٥٠إنه لم يمنعني من أن أرد عليك إلا أني كنت على غير وضوء٣٥١قال مر رجل على النبي صلى الله عليه وسلم وهو يبول فسلم عليه فلم يرد…٣٥٢إذا رأيتني على مثل هذه الحالة فلا تسلم على فإنك إن فعلت ذلك لم أرد…٣٥٣والحسين بن أبي السري العسقلاني، قالا حدثنا أبو داود، عن سفيان، عن…٣٥٤قالت ما رأيت رسول الله صلى الله عليه وسلم خرج من غائط قط إلا مس ماء .٣٥٥يا معشر الأنصار إن الله قد أثنى عليكم في الطهور فما طهوركم٣٥٦وإبراهيم بن سليمان الواسطي، قالا حدثنا أبو نعيم، حدثنا شريك، نحوه .٣٥٧نزلت في أهل قباء {فيه رجال يحبون أن يتطهروا والله يحب المطهرين} قال…٣٥٨وعلي بن محمد، قالا حدثنا وكيع، عن شريك، عن إبراهيم بن جرير، عن أبي…٣٥٩أن نبي الله صلى الله عليه وسلم دخل الغيضة فقضى حاجته فأتاه جرير…٣٦٠حدثنا محمد بن يحيى، حدثنا يعلى بن عبيد، حدثنا عبد الملك بن أبي…٣٦١ويحيى بن حكيم، قالا حدثنا حرمي بن عمارة بن أبي حفصة، حدثنا حريش بن…٣٦٢قال كان رسول الله صلى الله عليه وسلم لا يكل طهوره إلى أحد ولا صدقته…٣٦٣إذا ولغ الكلب في إناء أحدكم فليغسله سبع مرات٣٦٤إذا شرب الكلب في إناء أحدكم فليغسله سبع مرات٣٦٥إذا ولغ الكلب في الإناء فاغسلوه سبع مرات وعفروه الثامنة بالتراب٣٦٦إذا ولغ الكلب في إناء أحدكم فليغسله سبع مرات٣٦٧إنها ليست بنجس هي من الطوافين أو الطوافات٣٦٨وإسماعيل بن توبة، قالا حدثنا يحيى بن زكريا بن أبي زائدة، عن حارثة، عن…٣٦٩الهرة لا تقطع الصلاة لأنها من متاع البيت٣٧٠قال اغتسل بعض أزواج النبي صلى الله عليه وسلم في جفنة فجاء النبي صلى…٣٧١أن امرأة، من أزواج النبي صلى الله عليه وسلم اغتسلت من جنابة فتوضأ أو…٣٧٢ومحمد بن يحيى، وإسحاق بن منصور، قالوا حدثنا أبو داود، حدثنا شريك، عن…٣٧٣أن رسول الله صلى الله عليه وسلم نهى أن يتوضأ الرجل بفضل وضوء المرأة .٣٧٤نحوه .٣٧٥قال كان النبي صلى الله عليه وسلم وأهله يغتسلون من إناء واحد ولا يغتسل…٣٧٦صلى الله حدثنا أسامة…٣٨٣حدثنا محمد بن يحيى، حدثنا داود بن شبيب، حدثنا حبيب بن أبي حبيب، عن…٣٨٤وعلي بن محمد، قالا حدثنا وكيع، عن أبيه، ح وحدثنا محمد بن يحيى، حدثنا…٣٨٥أن رسول الله صلى الله عليه وسلم قال لابن
