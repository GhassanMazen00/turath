سنن ابن ماجه

كتاب الأطعمة

١٢٠ حديثًا
٣٢٥١يا أيها الناس أفشوا السلام وأطعموا الطعام وصلوا الأرحام وصلوا بالليل…٣٢٥٢أفشوا السلام وأطعموا الطعام وكونوا إخوانا كما أمركم الله عز وجل٣٢٥٣تطعم الطعام وتقرأ السلام على من عرفت ومن لم تعرف٣٢٥٤طعام الواحد يكفي الاثنين وطعام الاثنين يكفي الأربعة وطعام الأربعة…٣٢٥٥إن طعام الواحد يكفي الاثنين وإن طعام الاثنين يكفي الثلاثة والأربعة…٣٢٥٦المؤمن يأكل في معى واحد والكافر يأكل في سبعة أمعاء٣٢٥٧الكافر يأكل في سبعة أمعاء والمؤمن يأكل في معى واحد٣٢٥٨المؤمن يأكل في معى واحد والكافر يأكل في سبعة أمعاء٣٢٥٩حدثنا محمد بن بشار، حدثنا عبد الرحمن، حدثنا سفيان، عن الأعمش، عن أبي…٣٢٦٠من أحب أن يكثر الله خير بيته فليتوضأ إذا حضر غداؤه وإذا رفع٣٢٦١حدثنا جعفر بن مسافر، حدثنا صاعد بن عبيد الجزري، حدثنا زهير بن معاوية،…٣٢٦٢أن رسول الله صلى الله عليه وسلم قال " لا آكل متكئا " .٣٢٦٣إن الله جعلني عبدا كريما ولم يجعلني جبارا عنيدا٣٢٦٤أما أنه لو كان قال بسم الله لكفاكم فإذا أكل أحدكم طعاما فليقل بسم…٣٢٦٥سم الله عز وجل٣٢٦٦ليأكل أحدكم بيمينه وليشرب بيمينه وليأخذ بيمينه وليعط بيمينه فإن…٣٢٦٧يا غلام سم الله وكل بيمينك وكل مما يليك٣٢٦٨لا تأكلوا بالشمال فإن الشيطان يأكل بالشمال٣٢٦٩إذا أكل أحدكم طعاما فلا يمسح يده حتى يلعقها أو يلعقها٣٢٧٠لا يمسح أحدكم يده حتى يلعقها فإنه لا يدري في أى طعامه البركة٣٢٧١من أكل في قصعة فلحسها استغفرت له القصعة٣٢٧٢من أكل في قصعة ثم لحسها استغفرت له القصعة٣٢٧٣إذا وضعت المائدة فليأكل مما يليه ولا يتناول من بين يدى جليسه٣٢٧٤يا عكراش كل من موضع واحد فإنه طعام واحد٣٢٧٥كلوا من جوانبها ودعوا ذروتها يبارك فيها٣٢٧٦كلوا بسم الله من حواليها واعفوا رأسها فإن البركة تأتيها من فوقها٣٢٧٧إذا وضع الطعام فخذوا من حافته وذروا وسطه فإن البركة تنزل في وسطه٣٢٧٨حدثنا سويد بن سعيد، حدثنا يزيد بن زريع، عن يونس، عن الحسن، عن معقل بن…٣٢٧٩إذا وقعت اللقمة من يد أحدكم فليمسح ما عليها من الأذى وليأكلها٣٢٨٠كمل من الرجال كثير ولم يكمل من النساء إلا مريم بنت عمران وآسية امرأة…٣٢٨١فضل عائشة على النساء كفضل الثريد على سائر الطعام٣٢٨٢حدثنا محمد بن سلمة المصري أبو الحارث المرادي، حدثنا عبد الله بن وهب،…٣٢٨٣الحمد لله الذي أطعمنا وسقانا وجعلنا مسلمين٣٢٨٤الحمد لله حمدا كثيرا طيبا مباركا غير مكفي ولا مودع ولا مستغنى عنه ربنا٣٢٨٥من أكل طعاما فقال الحمد لله الذي أطعمني هذا ورزقنيه من غير حول مني…٣٢٨٦فلعلكم تأكلون متفرقين٣٢٨٧كلوا جميعا ولا تفرقوا فإن البركة مع الجماعة٣٢٨٨قال لم يكن رسول الله صلى الله عليه وسلم ينفخ في طعام ولا شراب ولا…٣٢٨٩إذا جاء أحدكم خادمه بطعامه فليجلسه فليأكل معه فإن أبى فليناوله منه٣٢٩٠إذا أحدكم قرب إليه مملوكه طعاما قد كفاه عناءه وحره فليدعه فليأكل معه…٣٢٩١إذا جاء خادم أحدكم بطعامه فليقعده معه أو ليناوله منه فإنه هو الذي ولي…٣٢٩٢قال ما أكل النبي صلى الله عليه وسلم على خوان ولا في سكرجة . قال فعلام…٣٢٩٣قال ما رأيت رسول الله صلى الله عليه وسلم أكل على خوان حتى مات .٣٢٩٤حدثنا عبد الله بن أحمد بن بشير بن ذكوان الدمشقي، حدثنا الوليد بن…٣٢٩٥إذا وضعت المائدة فلا يقوم رجل حتى ترفع المائدة ولا يرفع يده وإن شبع…٣٢٩٦ألا لا يلومن امرؤ إلا نفسه يبيت وفي يده ريح غمر٣٢٩٧إذا نام أحدكم وفي يده ريح غمر فلم يغسل يده فأصابه شىء فلا يلومن إلا…٣٢٩٨لا تجمعن جوعا وكذبا٣٢٩٩وعلي بن محمد، قالا حدثنا وكيع، عن أبي هلال، عن عبد الله بن سوادة، عن…٣٣٠٠وحرملة بن يحيى، قالا حدثنا عبد الله بن وهب، أخبرني عمرو بن الحارث،…٣٣٠١حدثنا أبو السائب، سلم بن جنادة حدثنا حفص بن غياث، عن عبيد الله بن…٣٣٠٢قال كان النبي صلى الله عليه وسلم يحب القرع .٣٣٠٣حدثنا محمد بن المثنى، حدثنا ابن أبي عدي، عن حميد، عن أنس، قال بعثت…٣٣٠٤هذا القرع هو الدباء نكثر به طعامنا٣٣٠٥سيد طعام أهل الدنيا وأهل الجنة اللحم٣٣٠٦قال ما دعي رسول الله صلى الله عليه وسلم إلى لحم قط إلا أجاب ولا أهدي…٣٣٠٧قال أتي رسول الله صلى الله عليه وسلم ذات يوم بلحم فرفع إليه الذراع…٣٣٠٨أطيب اللحم لحم الظهر٣٣٠٩قال ما أعلم رسول الله صلى الله عليه وسلم رأى شاة سميطا حتى لحق بالله…٣٣١٠قال ما رفع من بين يدى رسول الله صلى الله عليه وسلم فضل شواء قط ولا…٣٣١١حدثنا حرملة بن يحيى، حدثنا يحيى بن بكير، حدثنا ابن لهيعة، أخبرني…٣٣١٢هون عليك فإني لست بملك إنما أنا ابن امرأة تأكل القديد٣٣١٣حدثنا محمد بن يحيى، حدثنا محمد بن يوسف، حدثنا سفيان، عن عبد الرحمن بن…٣٣١٤أحلت لنا ميتتان ودمان فأما الميتتان فالحوت والجراد وأما الدمان فالكبد…٣٣١٥سيد إدامكم الملح٣٣١٦نعم الإدام الخل٣٣١٧نعم الإدام الخل٣٣١٨أنه حدثه قال حدثتني أم سعد، قالت دخل رسول الله صلى الله عليه وسلم على…٣٣١٩ائتدموا بالزيت وادهنوا به فإنه من شجرة مباركة٣٣٢٠كلوا الزيت وادهنوا به فإنه مبارك٣٣٢١بركة أو بركتان٣٣٢٢من أطعمه الله طعاما فليقل اللهم بارك لنا فيه وارزقنا خيرا منه . ومن…٣٣٢٣وعلي بن محمد، وعبد الرحمن بن إبراهيم، قالوا حدثنا أبو أسامة، قال…٣٣٢٤قالت كانت أمي تعالجني للسمنة تريد أن تدخلني على رسول الله صلى الله…٣٣٢٥وإسماعيل بن موسى، قالا حدثنا إبراهيم بن سعد، عن أبيه، عن عبد الله بن…٣٣٢٦وعمرو بن رافع، قالا حدثنا يعقوب بن الوليد بن أبي هلال المدني، عن أبي…٣٣٢٧بيت لا تمر فيه جياع أهله٣٣٢٨بيت لا تمر فيه كالبيت لا طعام فيه٣٣٢٩اللهم بارك لنا في مدينتنا وفي ثمارنا وفي مدنا وفي صاعنا بركة مع بركة٣٣٣٠كلوا البلح بالتمر كلوا الخلق بالجديد فإن الشيطان يغضب ويقول بقي ابن…٣٣٣١يقول نهى رسول الله صلى الله عليه وسلم أن يقرن الرجل بين التمرتين حتى…٣٣٣٢حدثنا محمد بن بشار، حدثنا أبو داود، حدثنا أبو عامر الخزاز، عن الحسن،…٣٣٣٣قال رأيت رسول الله صلى الله عليه وسلم أتي بتمر عتيق فجعل يفتشه .٣٣٣٤حدثنا هشام بن عمار، حدثنا صدقة بن خالد، حدثني ابن جابر، حدثني سليم بن…٣٣٣٥وسويد بن سعيد، قالا حدثنا عبد العزيز بن أبي حازم، حدثني أبي قال، سألت…٣٣٣٦أن حنش بن عبد الله، حدثه عن أم أيمن، أنها غربلت دقيقا فصنعته للنبي…٣٣٣٧قال ما رأى رسول الله صلى الله عليه وسلم رغيفا محورا بواحد من عينيه…٣٣٣٨حدثنا أبو عمير، عيسى بن محمد بن النحاس الرملي حدثنا ضمرة بن ربيعة، عن…٣٣٣٩وأحمد بن سعيد الدارمي، قالا حدثنا عبد الصمد بن عبد الوارث، حدثنا…٣٣٤٠أن جبريل، عليه السلام أتى النبي صلى الله عليه وسلم فقال إن أمتك تفتح…٣٣٤١وددت لو أن عندنا خبزة بيضاء من برة سمراء ملبقة بسمن نأكلها٣٣٤٢قال صنعت أم سليم للنبي صلى الله عليه وسلم خبزة وضعت فيها شيئا من سمن…٣٣٤٣أنه قال والذي نفسي بيده ما شبع نبي الله صلى الله عليه وسلم ثلاثة أيام…٣٣٤٤قالت ما شبع آل محمد صلى الله عليه وسلم منذ قدموا المدينة ثلاث ليال…٣٣٤٥قالت لقد توفي النبي صلى الله عليه وسلم وما في بيتي من شىء يأكله ذو…٣٣٤٦قالت ما شبع آل محمد صلى الله عليه وسلم من خبز الشعير حتى قبض .٣٣٤٧قال كان رسول الله صلى الله عليه وسلم يبيت الليالي المتتابعة طاويا…٣٣٤٨حدثنا يحيى بن عثمان بن سعيد بن كثير بن دينار الحمصي، - وكان يعد من…٣٣٤٩ما ملأ آدمي وعاء شرا من بطن حسب الآدمي لقيمات يقمن صلبه فإن غلبت…٣٣٥٠كف جشاءك عنا فإن أطولكم جوعا يوم القيامة أكثركم شبعا في دار الدنيا٣٣٥١إن أكثر الناس شبعا في الدنيا أطولهم جوعا يوم القيامة٣٣٥٢إن من السرف أن تأكل كل ما اشتهيت٣٣٥٣يا عائشة أكرمي كريمك فإنها ما نفرت عن قوم قط فعادت إليهم٣٣٥٤اللهم إني أعوذ بك من الجوع فإنه بئس الضجيع وأعوذ بك من الخيانة فإنها…٣٣٥٥لا تدعوا العشاء ولو بكف من تمر فإن تركه يهرم٣٣٥٦الخير أسرع إلى البيت الذي يغشى من الشفرة إلى سنام البعير٣٣٥٧الخير أسرع إلى البيت الذي يؤكل فيه من الشفرة إلى سنام البعير٣٣٥٨إن من السنة أن يخرج الرجل مع ضيفه إلى باب الدار٣٣٥٩قال صنعت طعاما فدعوت رسول الله صلى الله عليه وسلم فجاء فرأى في البيت…٣٣٦٠إنه ليس لي أن أدخل بيتا مزوقا٣٣٦١حدثنا أبو كريب، حدثنا يحيى بن عبد الرحمن الأرحبي، حدثنا يونس بن أبي…٣٣٦٢إذا عملت مرقة فأكثر ماءها واغترف لجيرانك منها٣٣٦٣أن عمر بن الخطاب، قام يوم الجمعة خطيبا فحمد الله وأثنى عليه ثم قال يا…٣٣٦٤إني أكره أن أوذي صاحبي٣٣٦٥ألم أكن نهيتكم عن أكل هذه الشجرة إن الملائكة تتأذى مما يتأذى منه…٣٣٦٦أنه سمع عقبة بن عامر الجهني، يقول إن رسول الله صلى الله عليه وسلم قال…٣٣٦٧الحلال ما أحل الله في كتابه والحرام ما حرم الله في كتابه وما سكت عنه…٣٣٦٨خذ هذا العنقود فأبلغه أمك٣٣٦٩دونكها يا طلحة فإنها تجم الفؤاد٣٣٧٠قال نهى رسول الله صلى الله عليه وسلم أن يأكل الرجل وهو منبطح على وجهه…
