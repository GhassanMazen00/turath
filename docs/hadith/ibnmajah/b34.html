سنن ابن ماجه

كتاب الدعاء

٦٦ حديثًا
٣٨٢٧من لم يدع الله سبحانه غضب عليه٣٨٢٨إن الدعاء هو العبادة٣٨٢٩ليس شىء أكرم على الله سبحانه من الدعاء٣٨٣٠رب أعني ولا تعن على وانصرني ولا تنصر على وامكر لي ولا تمكر على واهدني…٣٨٣١ما عندي ما أعطيك٣٨٣٢اللهم إني أسألك الهدى والتقى والعفاف والغنى٣٨٣٣اللهم انفعني بما علمتني وعلمني ما ينفعني وزدني علما والحمد لله على كل…٣٨٣٤اللهم ثبت قلبي على دينك٣٨٣٥قل اللهم إني ظلمت نفسي ظلما كثيرا ولا يغفر الذنوب إلا أنت فاغفر لي…٣٨٣٦لا تفعلوا كما يفعل أهل فارس بعظمائها٣٨٣٧اللهم إني أعوذ بك من الأربع من علم لا ينفع ومن قلب لا يخشع ومن نفس لا…٣٨٣٨اللهم إني أعوذ بك من فتنة النار وعذاب النار ومن فتنة القبر وعذاب…٣٨٣٩اللهم إني أعوذ بك من شر ما عملت ومن شر ما لم أعمل٣٨٤٠اللهم إني أعوذ بك من عذاب جهنم وأعوذ بك من عذاب القبر وأعوذ بك من…٣٨٤١اللهم إني أعوذ برضاك من سخطك وبمعافاتك من عقوبتك وأعوذ بك منك لا أحصي…٣٨٤٢تعوذوا بالله من الفقر والقلة والذلة وأن تظلم أو تظلم٣٨٤٣سلوا الله علما نافعا وتعوذوا بالله من علم لا ينفع٣٨٤٤أن النبي صلى الله عليه وسلم كان يتعوذ من الجبن والبخل وأرذل العمر…٣٨٤٥قل اللهم اغفر لي وارحمني وعافني وارزقني٣٨٤٦اللهم إني أسألك من الخير كله عاجله وآجله ما علمت منه وما لم أعلم…٣٨٤٧ما تقول في الصلاة٣٨٤٨سل ربك العفو والعافية في الدنيا والآخرة٣٨٤٩عليكم بالصدق فإنه مع البر وهما في الجنة وإياكم والكذب فإنه مع الفجور…٣٨٥٠تقولين اللهم إنك عفو تحب العفو فاعف عني٣٨٥١ما من دعوة يدعو بها العبد أفضل من - اللهم إني أسألك المعافاة في…٣٨٥٢يرحمنا الله وأخا عاد٣٨٥٣يستجاب لأحدكم ما لم يعجل٣٨٥٤لا يقولن أحدكم اللهم اغفر لي إن شئت وليعزم في المسألة فإن الله لا…٣٨٥٥اسم الله الأعظم في هاتين الآيتين {وإلهكم إله واحد لا إله إلا هو…٣٨٥٦سمع غيلان بن أنس، يحدث عن القاسم، عن أبي أمامة، عن النبي صلى الله…٣٨٥٧لقد سأل الله باسمه الأعظم الذي إذا سئل به أعطى وإذا دعي به أجاب٣٨٥٨لقد سأل الله باسمه الأعظم الذي إذا سئل به أعطى وإذا دعي به أجاب٣٨٥٩اللهم إني أسألك باسمك الطاهر الطيب المبارك الأحب إليك الذي إذا دعيت…٣٨٦٠إن لله تسعة وتسعين اسما مائة إلا واحدا من أحصاها دخل الجنة٣٨٦١إن لله تسعة وتسعين اسما مائة إلا واحدا إنه وتر يحب الوتر من حفظها دخل…٣٨٦٢ثلاث دعوات يستجاب لهن لا شك فيهن دعوة المظلوم ودعوة المسافر ودعوة…٣٨٦٣دعاء الوالد يفضي إلى الحجاب٣٨٦٤سيكون قوم يعتدون في الدعاء٣٨٦٥إن ربكم حيي كريم يستحيي من عبده أن يرفع إليه يديه فيردهما صفرا - أو…٣٨٦٦إذا دعوت الله فادع ببطون كفيك ولا تدع بظهورهما فإذا فرغت فامسح بهما…٣٨٦٧من قال حين يصبح لا إله إلا الله وحده لا شريك له له الملك وله الحمد…٣٨٦٨إذا أصبحتم فقولوا اللهم بك أصبحنا وبك أمسينا وبك نحيى وبك نموت وإذا…٣٨٦٩ما من عبد يقول في صباح كل يوم ومساء كل ليلة بسم الله الذي لا يضر مع…٣٨٧٠ما من مسلم أو إنسان أو عبد يقول حين يمسي وحين يصبح رضيت بالله ربا…٣٨٧١اللهم إني أسألك العفو والعافية في الدنيا والآخرة اللهم إني أسألك…٣٨٧٢اللهم أنت ربي لا إله إلا أنت خلقتني وأنا عبدك وأنا على عهدك ووعدك ما…٣٨٧٣اللهم رب السموات ورب الأرض ورب كل شىء فالق الحب والنوى منزل التوراة…٣٨٧٤إذا أراد أحدكم أن يضطجع على فراشه فلينزع داخلة إزاره ثم لينفض بها…٣٨٧٥وسعيد بن شرحبيل، أنبأنا الليث بن سعد، عن عقيل، عن ابن شهاب، أن عروة…٣٨٧٦إذا أخذت مضجعك أو أويت إلى فراشك فقل اللهم أسلمت وجهي إليك وألجأت…٣٨٧٧اللهم قني عذابك يوم تبعث - أو تجمع - عبادك٣٨٧٨من تعار من الليل فقال حين يستيقظ لا إله إلا الله وحده لا شريك له له…٣٨٧٩سبحان الله رب العالمين٣٨٨٠الحمد لله الذي أحيانا بعد ما أماتنا وإليه النشور٣٨٨١ما من عبد بات على طهور ثم تعار من الليل فسأل الله شيئا من أمر الدنيا…٣٨٨٢الله الله ربي لا أشرك به شيئا٣٨٨٣لا إله إلا الله الحليم الكريم سبحان الله رب العرش العظيم سبحان الله…٣٨٨٤اللهم إني أعوذ بك أن أضل أو أزل أو أظلم أو أظلم أو أجهل أو يجهل على٣٨٨٥بسم الله لا حول ولا قوة إلا بالله التكلان على الله٣٨٨٦إذا خرج الرجل من باب بيته - أو من باب داره - كان معه ملكان موكلان به…٣٨٨٧إذا دخل الرجل بيته فذكر الله عند دخوله وعند طعامه قال الشيطان لا مبيت…٣٨٨٨اللهم إني أعوذ بك من وعثاء السفر وكآبة المنقلب والحور بعد الكور ودعوة…٣٨٨٩اللهم إنا نعوذ بك من شر ما أرسل به٣٨٩٠اللهم اجعله صيبا هنيئا٣٨٩١وما يدريك لعله كما قال قوم هود {فلما رأوه عارضا مستقبل أوديتهم قالوا…٣٨٩٢من فجئه صاحب بلاء فقال الحمد لله الذي عافاني مما ابتلاك به وفضلني على…
