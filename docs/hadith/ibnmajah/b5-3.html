كتاب إقامة الصلاة والسنة فيها

٦٣٠ حديثًا•صفحة ٣ من ٥
١١٠٣أن النبي صلى الله عليه وسلم كان يخطب خطبتين يجلس بينهما جلسة . زاد…١١٠٤قال رأيت النبي صلى الله عليه وسلم يخطب على المنبر، وعليه عمامة سوداء .١١٠٥ومحمد بن الوليد، قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن سماك بن حرب،…١١٠٦قال كان النبي صلى الله عليه وسلم يخطب قائما. ثم يجلس. ثم يقوم فيقرأ…١١٠٧أن رسول الله صلى الله عليه وسلم كان إذا خطب في الحرب، خطب على قوس.…١١٠٨أنه سئل أكان النبي صلى الله عليه وسلم يخطب قائما أو قاعدا؟ قال أوما…١١٠٩أن النبي صلى الله عليه وسلم كان إذا صعد المنبر سلم .١١١٠إذا قلت لصاحبك: أنصت، يوم الجمعة، والإمام يخطب، فقد لغوت١١١١أن رسول الله صلى الله عليه وسلم قرأ يوم الجمعة تبارك، وهو قائم.…١١١٢سمع جابرا، وأبو الزبير، سمع جابر بن عبد الله، قال دخل سليك الغطفاني…١١١٣قال جاء رجل والنبي صلى الله عليه وسلم يخطب فقال " أصليت؟ " . قال: لا…١١١٤أصليت ركعتين قبل أن تجيء؟١١١٥اجلس فقد آذيت وآنيت١١١٦من تخطى رقاب الناس يوم الجمعة اتخذ جسرا إلى جهنم١١١٧حدثنا محمد بن بشار، حدثنا أبو داود، حدثنا جرير بن حازم، عن ثابت، عن…١١١٨حدثنا أبو بكر بن أبي شيبة، حدثنا حاتم بن إسماعيل المدني، عن جعفر بن…١١١٩شىء كان النبي صلى الله عليه وسلم يقرأ يوم الجمعة مع سورة الجمعة قال…١١٢٠أن النبي صلى الله عليه وسلم كان يقرأ في الجمعة ب {سبح اسم ربك الأعلى}…١١٢١من أدرك من الجمعة ركعة فليصل إليها أخرى١١٢٢من أدرك من الصلاة ركعة فقد أدرك١١٢٣من أدرك ركعة من صلاة الجمعة أو غيرها، فقد أدرك الصلاة١١٢٤قال إن أهل قباء كانوا يجمعون مع رسول الله صلى الله عليه وسلم يوم…١١٢٥من ترك الجمعة ثلاث مرات، تهاونا بها، طبع على قلبه١١٢٦من ترك الجمعة، ثلاثا، من غير ضرورة، طبع الله على قلبه١١٢٧ألا هل عسى أحدكم أن يتخذ الصبة من الغنم على رأس ميل أو ميلين، فيتعذر…١١٢٨من ترك الجمعة متعمدا، فليتصدق بدينار، فإن لم يجد، فبنصف دينار١١٢٩قال: كان النبي صلى الله عليه وسلم يركع من قبل الجمعة أربعا، لا يفصل…١١٣٠أنه كان إذا صلى الجمعة انصرف فصلى سجدتين في بيته ثم قال كان رسول الله…١١٣١أن النبي صلى الله عليه وسلم كان يصلي بعد الجمعة ركعتين .١١٣٢إذا صليتم بعد الجمعة، فصلوا أربعا١١٣٣أن رسول الله صلى الله عليه وسلم نهى أن يحلق في المسجد يوم الجمعة قبل…١١٣٤حدثنا محمد بن المصفى الحمصي، حدثنا بقية، عن عبد الله بن واقد، عن محمد…١١٣٥قال ما كان لرسول الله صلى الله عليه وسلم إلا مؤذن واحد. فإذا خرج أذن،…١١٣٦قال كان النبي صلى الله عليه وسلم إذا قام على المنبر استقبله أصحابه…١١٣٧إن في الجمعة ساعة، لا يوافقها رجل مسلم، قائم يصلي، يسأل الله فيها…١١٣٨في يوم الجمعة ساعة من النهار، لا يسأل الله فيها العبد شيئا إلا أعطي…١١٣٩هي آخر ساعة من ساعات النهار١١٤٠من ثابر على ثنتى عشرة ركعة من السنة، بني له بيت في الجنة. أربع قبل…١١٤١من صلى في يوم وليلة ثنتى عشرة ركعة، بني له بيت في الجنة١١٤٢من صلى في يوم ثنتى عشرة ركعة بني له بيت في الجنة ركعتين قبل الفجر…١١٤٣أن النبي صلى الله عليه وسلم كان إذا أضاء الفجر صلى ركعتين .١١٤٤قال كان رسول الله صلى الله عليه وسلم يصلي الركعتين قبل الغداة كأن…١١٤٥أن رسول الله صلى الله عليه وسلم كان إذا نودي لصلاة الصبح ركع ركعتين…١١٤٦قالت كان النبي صلى الله عليه وسلم إذا توضأ صلى ركعتين ثم خرج إلى…١١٤٧قال كان النبي صلى الله عليه وسلم يصلي الركعتين عند الإقامة .١١٤٨ويعقوب بن حميد بن كاسب، قالا حدثنا مروان بن معاوية، عن يزيد بن كيسان،…١١٤٩ومحمد بن عبادة الواسطيان، قالا حدثنا أبو أحمد، حدثنا سفيان، عن أبي…١١٥٠نعم السورتان هما يقرأ بهما في ركعتى الفجر {قل هو الله أحد } و {قل يا…١١٥١إذا أقيمت الصلاة، فلا صلاة إلا المكتوبة١١٥٢بأى صلاتيك اعتددت؟١١٥٣يوشك أحدكم أن يصلي الفجر أربعا١١٥٤أصلاة الصبح مرتين؟١١٥٥ويعقوب بن حميد بن كاسب، قالا حدثنا مروان بن معاوية، عن يزيد بن كيسان،…١١٥٦قال أرسل أبي إلى عائشة أى صلاة رسول الله صلى الله عليه وسلم كان أحب…١١٥٧إن أبواب السماء تفتح إذا زالت الشمس١١٥٨وزيد بن أخزم، ومحمد بن معمر، قالوا حدثنا موسى بن داود الكوفي، حدثنا…١١٥٩أشغلني أمر الساعي أن أصليهما بعد الظهر فصليتهما بعد العصر١١٦٠من صلى قبل الظهر أربعا، وبعدها أربعا، حرمه الله على النار١١٦١وأبي، وإسرائيل، عن أبي إسحاق، عن عاصم بن ضمرة السلولي، قال سألنا عليا…١١٦٢بين كل أذانين صلاة١١٦٣يقول إن كان المؤذن ليؤذن على عهد رسول الله صلى الله عليه وسلم فيرى…١١٦٤قالت كان النبي صلى الله عليه وسلم يصلي المغرب، ثم يرجع إلى بيتي فيصلي…١١٦٥اركعوا هاتين الركعتين في بيوتكم١١٦٦وأبي، وائل، عن عبد الله بن مسعود أن النبي صلى الله عليه وسلم كان يقرأ…١١٦٧من صلى بعد المغرب ست ركعات لم يتكلم بينهن بسوء عدلن له بعبادة ثنتى…١١٦٨إن الله قد أمدكم بصلاة، لهي خير لكم من حمر النعم. الوتر، جعله الله…١١٦٩يا أهل القرآن أوتروا. فإن الله وتر يحب الوتر١١٧٠إن الله وتر يحب الوتر فأوتروا يا أهل القرآن١١٧١وزبيد، عن ذر، عن سعيد بن عبد الرحمن بن أبزى، عن أبيه، عن أبى بن كعب،…١١٧٢حدثنا نصر بن علي الجهضمي، حدثنا أبو أحمد، حدثنا يونس بن أبي إسحاق، عن…١١٧٣حدثنا محمد بن الصباح، وأبو يوسف الرقي محمد بن أحمد الصيدلاني قالا…١١٧٤قال كان رسول الله صلى الله عليه وسلم يصلي من الليل مثنى مثنى. ويوتر…١١٧٥صلاة الليل مثنى مثنى والوتر ركعة١١٧٦قال سأل ابن عمر رجل فقال كيف أوتر قال أوتر بواحدة . قال إني أخشى أن…١١٧٧قالت كان رسول الله صلى الله عليه وسلم يسلم في كل ثنتين ويوتر بواحدة .١١٧٨اللهم عافني فيمن عافيت وتولني فيمن توليت واهدني فيمن هديت وقني شر ما…١١٧٩اللهم إني أعوذ برضاك من سخطك وأعوذ بمعافاتك من عقوبتك وأعوذ بك منك لا…١١٨٠أن نبي الله صلى الله عليه وسلم كان لا يرفع يديه في شىء من دعائه إلا…١١٨١إذا دعوت الله فادع بباطن كفيك ولا تدع بظهورهما فإذا فرغت فامسح بهما…١١٨٢أن رسول الله صلى الله عليه وسلم كان يوتر فيقنت قبل الركوع .١١٨٣حدثنا نصر بن علي الجهضمي، حدثنا سهل بن يوسف، حدثنا حميد، عن أنس بن…١١٨٤فقال قنت رسول الله صلى الله عليه وسلم بعد الركوع .١١٨٥رسول الله صلى الله عليه وسلم فقالت: من كل الليل قد أوتر. من أوله…١١٨٦وانتهى وتره إلى السحر .١١٨٧من خاف منكم أن لا يستيقظ من آخر الليل، فليوتر من أول الليل ثم ليرقد.…١١٨٨من نام عن الوتر أو نسيه، فليصل إذا أصبح، أو ذكره١١٨٩أوتروا قبل أن تصبحوا١١٩٠الوتر حق. فمن شاء فليوتر بخمس. ومن شاء فليوتر بثلاث. ومن شاء فليوتر…١١٩١حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن بشر، حدثنا سعيد بن أبي…١١٩٢قالت كان رسول الله صلى الله عليه وسلم يوتر بسبع أو بخمس لا يفصل بينهن…١١٩٣وإسحاق بن منصور، قالا حدثنا يزيد بن هارون، أنبأنا شعبة، عن جابر، عن…١١٩٤وابن، عمر قالا سن رسول الله صلى الله عليه وسلم صلاة السفر ركعتين وهما…١١٩٥أن النبي صلى الله عليه وسلم كان يصلي بعد الوتر ركعتين خفيفتين، وهو…١١٩٦قال حدثتني عائشة، قالت كان رسول الله صلى الله عليه وسلم يوتر بواحدة.…١١٩٧وسفيان، عن سعد بن إبراهيم، عن أبي سلمة بن عبد الرحمن، عن عائشة، قالت…١١٩٨قالت كان النبي صلى الله عليه وسلم إذا صلى ركعتى الفجر اضطجع على شقه…١١٩٩قال كان رسول الله صلى الله عليه وسلم إذا صلى ركعتى الفجر اضطجع .١٢٠٠قال كنت مع ابن عمر فتخلفت فأوترت فقال ما خلفك قلت أوترت . فقال أما لك…١٢٠١أن النبي صلى الله عليه وسلم كان يوتر على راحلته .١٢٠٢أن النبي صلى الله عليه وسلم قال لأبي بكر . فذكر نحوه .١٢٠٣إنما أنا بشر. أنسى كما تنسون. فإذا نسي أحدكم فليسجد سجدتين وهو جالس١٢٠٤إذا صلى أحدكم فلم يدر كم صلى، فليسجد سجدتين وهو جالس١٢٠٥قال: صلى النبي صلى الله عليه وسلم الظهر خمسا. فقيل له: أزيد في…١٢٠٦أن النبي صلى الله عليه وسلم صلى صلاة أظن أنها الظهر (العصر) فلما كان…١٢٠٧وابن، فضيل ويزيد بن هارون ح وحدثنا عثمان بن أبي شيبة، حدثنا أبو خالد…١٢٠٨إذا قام أحدكم من الركعتين فلم يستتم قائما فليجلس فإذا استتم قائما فلا…١٢٠٩إذا شك أحدكم في الثنتين والواحدة فليجعلها واحدة وإذا شك في الثنتين…١٢١٠إذا شك أحدكم في صلاته فليلغ الشك وليبن على اليقين فإذا استيقن التمام…١٢١١لو حدث في الصلاة شىء لأنبأتكموه وإنما أنا بشر أنسى كما تنسون فإذا…١٢١٢إذا شك أحدكم في الصلاة فليتحر الصواب ثم يسجد سجدتين١٢١٣ما قصرت وما نسيت١٢١٤لم تقصر ولم أنس١٢١٥وأحمد بن ثابت الجحدري، حدثنا عبد الوهاب، حدثنا خالد الحذاء، عن أبي…١٢١٦إن الشيطان يأتي أحدكم في صلاته، فيدخل بينه وبين نفسه حتى لا يدري زاد…١٢١٧إن الشيطان يدخل بين ابن آدم وبين نفسه فلا يدري كم صلى فإذا وجد ذلك…١٢١٨أن ابن مسعود، سجد سجدتى السهو بعد السلام وذكر أن النبي صلى الله عليه…١٢١٩في كل سهو سجدتان بعد ما يسلم١٢٢٠إني خرجت إليكم جنبا. وإني نسيت حتى قمت في الصلاة١٢٢١من أصابه قىء أو رعاف أو قلس أو مذى، فلينصرف، فليتوضأ. ثم ليبن على…١٢٢٢إذا صلى أحدكم فأحدث، فليمسك على أنفه، ثم لينصرف١٢٢٣صل قائما. فإن لم تستطع فقاعدا. فإن لم تستطع، فعلى جنب١٢٢٤قال رأيت النبي صلى الله عليه وسلم صلى جالسا على يمينه وهو وجع .١٢٢٥قالت والذي ذهب بنفسه صلى الله عليه وسلم ما مات حتى كان أكثر صلاته وهو…١٢٢٦قالت كان النبي صلى الله عليه وسلم يقرأ وهو قاعد فإذا أراد أن يركع قام…١٢٢٧قالت ما رأيت رسول الله صلى الله عليه وسلم يصلي في شىء من صلاة الليل…١٢٢٨رسول الله صلى الله عليه وسلم بالليل فقالت كان يصلي ليلا طويلا قائما…١٢٢٩صلاة الجالس على النصف من صلاة القائم١٢٣٠صلاة القاعد على النصف من صلاة القائم١٢٣١من صلى قائما فهو أفضل. ومن صلى قاعدا فله نصف أجر القائم. ومن صلى…١٢٣٢مروا أبا بكر فليصل بالناس١٢٣٣قالت أمر رسول الله صلى الله عليه وسلم أبا بكر أن يصلي بالناس في مرضه…١٢٣٤حدثنا نصر بن علي الجهضمي، أنبأنا عبد الله بن داود، من كتابه في بيته…١٢٣٥حدثنا علي بن محمد، حدثنا وكيع، عن إسرائيل، عن أبي إسحاق، عن الأرقم بن…١٢٣٦وقد أحسنت كذلك فافعل١٢٣٧إنما جعل الإمام ليؤتم به. فإذا ركع فاركعوا. وإذا رفع فارفعوا. وإذا…١٢٣٨إنما جعل الإمام ليؤتم به فإذا كبر فكبروا وإذا ركع فاركعوا وإذا قال…١٢٣٩إنما جعل الإمام ليؤتم به فإذا كبر فكبروا وإذا ركع فاركعوا وإذا قال…١٢٤٠إن كدتم أن تفعلوا فعل فارس والروم يقومون على ملوكهم وهم قعود فلا…١٢٤١وحفص بن غياث، ويزيد بن هارون، عن أبي مالك الأشجعي، سعد بن طارق قال…١٢٤٢حدثنا حاتم بن بكر الضبي، حدثنا محمد بن يعلى، زنبور حدثنا عنبسة بن عبد…١٢٤٣أن رسول الله صلى الله عليه وسلم كان يقنت في صلاة الصبح. يدعو على حى…١٢٤٤اللهم أنج الوليد بن الوليد، وسلمة بن هشام، وعياش بن أبي ربيعة،…١٢٤٥ومحمد بن الصباح، قالا حدثنا سفيان بن عيينة، عن معمر، عن يحيى بن أبي…١٢٤٦لعن الله العقرب ما تدع المصلي وغير المصلي اقتلوها في الحل والحرم١٢٤٧أن النبي صلى الله عليه وسلم قتل عقربا وهو في الصلاة .١٢٤٨حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، وأبو أسامة عن عبيد…١٢٤٩لا صلاة بعد العصر حتى تغرب الشمس ولا صلاة بعد الفجر حتى تطلع الشمس١٢٥٠لا صلاة بعد الفجر حتى تطلع الشمس ولا صلاة بعد العصر حتى تغرب الشمس١٢٥١نعم. جوف الليل الأوسط. فصل ما بدا لك حتى يطلع الصبح. ثم انته حتى تطلع…١٢٥٢حدثنا الحسن بن داود المنكدري، حدثنا ابن أبي فديك، عن الضحاك بن عثمان،…
