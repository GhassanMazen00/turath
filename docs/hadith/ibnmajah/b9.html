سنن ابن ماجه

كتاب النكاح

١٧١ حديثًا•صفحة ١ من ٢
١٨٤٥يا معشر الشباب من استطاع منكم الباءة فليتزوج فإنه أغض للبصر وأحصن…١٨٤٦النكاح من سنتي فمن لم يعمل بسنتي فليس مني وتزوجوا فإني مكاثر بكم…١٨٤٧لم نر للمتحابين مثل النكاح١٨٤٨حدثنا أبو مروان، محمد بن عثمان العثماني حدثنا إبراهيم بن سعد، عن…١٨٤٩وزيد بن أخزم، قالا حدثنا معاذ بن هشام، حدثنا أبي، عن قتادة، عن الحسن،…١٨٥٠أن يطعمها إذا طعم وأن يكسوها إذا اكتسى ولا يضرب الوجه ولا يقبح ولا…١٨٥١استوصوا بالنساء خيرا فإنما هن عندكم عوان . ليس تملكون منهن شيئا غير…١٨٥٢لو أمرت أحدا أن يسجد لأحد لأمرت المرأة أن تسجد لزوجها ولو أن رجلا أمر…١٨٥٣قال لما قدم معاذ من الشام سجد للنبي صلى الله عليه وسلم قال " ما هذا…١٨٥٤أيما امرأة ماتت وزوجها عنها راض دخلت الجنة١٨٥٥إنما الدنيا متاع وليس من متاع الدنيا شىء أفضل من المرأة الصالحة١٨٥٦ليتخذ أحدكم قلبا شاكرا ولسانا ذاكرا وزوجة مؤمنة تعين أحدكم على أمر…١٨٥٧ما استفاد المؤمن بعد تقوى الله خيرا له من زوجة صالحة إن أمرها أطاعته…١٨٥٨تنكح النساء لأربع لمالها ولحسبها ولجمالها ولدينها . فاظفر بذات الدين…١٨٥٩لا تزوجوا النساء لحسنهن فعسى حسنهن أن يرديهن ولا تزوجوهن لأموالهن…١٨٦٠أتزوجت يا جابر١٨٦١عليكم بالأبكار فإنهن أعذب أفواها وأنتق أرحاما وأرضى باليسير١٨٦٢من أراد أن يلقى الله طاهرا مطهرا فليتزوج الحرائر١٨٦٣انكحوا فإني مكاثر بكم١٨٦٤إذا ألقى الله في قلب امرئ خطبة امرأة فلا بأس أن ينظر إليها١٨٦٥اذهب فانظر إليها فإنه أحرى أن يؤدم بينكما١٨٦٦اذهب فانظر إليها فإنه أجدر أن يؤدم بينكما١٨٦٧لا يخطب الرجل على خطبة أخيه١٨٦٨لا يخطب الرجل على خطبة أخيه١٨٦٩إذا حللت فآذنيني١٨٧٠الأيم أولى بنفسها من وليها والبكر تستأمر في نفسها١٨٧١لا تنكح الثيب حتى تستأمر ولا البكر حتى تستأذن وإذنها الصموت١٨٧٢الثيب تعرب عن نفسها والبكر رضاها صمتها١٨٧٣أن القاسم بن محمد، أخبره أن عبد الرحمن بن يزيد ومجمع بن يزيد…١٨٧٤قال جاءت فتاة إلى النبي صلى الله عليه وسلم فقالت إن أبي زوجني ابن…١٨٧٥. أن جارية، بكرا أتت النبي صلى الله عليه وسلم فذكرت له أن أباها زوجها…١٨٧٦حدثنا سويد بن سعيد، حدثنا علي بن مسهر، حدثنا هشام بن عروة، عن أبيه،…١٨٧٧قال تزوج النبي صلى الله عليه وسلم عائشة وهي بنت سبع وبنى بها وهي بنت…١٨٧٨. أنه حين هلك عثمان بن مظعون ترك ابنة له . قال ابن عمر فزوجنيها خالي…١٨٧٩أيما امرأة لم ينكحها الولي فنكاحها باطل فنكاحها باطل فنكاحها باطل فإن…١٨٨٠لا نكاح إلا بولي١٨٨١لا نكاح إلا بولي١٨٨٢لا تزوج المرأة المرأة ولا تزوج المرأة نفسها فإن الزانية هي التي تزوج…١٨٨٣حدثنا سويد بن سعيد، حدثنا مالك بن أنس، عن نافع، عن ابن عمر، قال نهى…١٨٨٤حدثنا أبو بكر بن أبي شيبة، حدثنا يحيى بن سعيد، وأبو أسامة عن عبيد…١٨٨٥لا شغار في الإسلام١٨٨٦قال سألت عائشة كم كان صداق نساء النبي صلى الله عليه وسلم قالت كان…١٨٨٧قال قال عمر بن الخطاب لا تغالوا صداق النساء فإنها لو كانت مكرمة في…١٨٨٨وهناد بن السري، قالا حدثنا وكيع، عن سفيان، عن عاصم بن عبيد الله، عن…١٨٨٩حدثنا حفص بن عمرو، حدثنا عبد الرحمن بن مهدي، عن سفيان، عن أبي حازم،…١٨٩٠أن النبي صلى الله عليه وسلم تزوج عائشة على متاع بيت قيمته خمسون درهما…١٨٩١مثله .١٨٩٢حدثنا هشام بن عمار، حدثنا عيسى بن يونس، حدثني أبي، عن جدي أبي إسحاق،…١٨٩٣الحمد لله نحمده ونستعينه ونعوذ بالله من شرور أنفسنا ومن سيئات أعمالنا…١٨٩٤كل أمر ذي بال لا يبدأ فيه بالحمد أقطع١٨٩٥أعلنوا هذا النكاح واضربوا عليه بالغربال١٨٩٦فصل ما بين الحلال والحرام الدف والصوت في النكاح١٨٩٧أما هذا فلا تقولوه ما يعلم ما في غد إلا الله١٨٩٨يا أبا بكر إن لكل قوم عيدا وهذا عيدنا١٨٩٩يعلم الله إني لأحبكن١٩٠٠حدثنا إسحاق بن منصور، أنبأنا جعفر بن عون، أنبأنا الأجلح، عن أبي…١٩٠١قال كنت مع ابن عمر فسمع صوت، طبل فأدخل إصبعيه في أذنيه ثم تنحى حتى…١٩٠٢أخرجوه من بيوتكم١٩٠٣حدثنا يعقوب بن حميد بن كاسب، حدثنا عبد العزيز بن أبي حازم، عن سهيل،…١٩٠٤حدثنا أبو بكر بن خلاد الباهلي، حدثنا خالد بن الحارث، حدثنا شعبة، عن…١٩٠٥بارك الله لكم وبارك عليكم وجمع بينكما في خير١٩٠٦اللهم بارك لهم وبارك عليهم١٩٠٧أن النبي صلى الله عليه وسلم رأى على عبد الرحمن بن عوف أثر صفرة فقال "…١٩٠٨قال ما رأيت رسول الله صلى الله عليه وسلم أولم على شىء من نسائه ما…١٩٠٩وغياث بن جعفر الرحبي، قالا حدثنا سفيان بن عيينة، حدثنا وائل بن داود،…١٩١٠قال شهدت للنبي صلى الله عليه وسلم وليمة ما فيها لحم ولا خبز . قال ابن…١٩١١حدثنا سويد بن سعيد، حدثنا المفضل بن عبد الله، عن جابر، عن الشعبي، عن…١٩١٢قال دعا أبو أسيد الساعدي رسول الله صلى الله عليه وسلم إلى عرسه فكانت…١٩١٣قال شر الطعام طعام الوليمة يدعى لها الأغنياء ويترك الفقراء ومن لم يجب…١٩١٤إذا دعي أحدكم إلى وليمة عرس فليجب١٩١٥الوليمة أول يوم حق والثاني معروف والثالث رياء وسمعة١٩١٦إن للثيب ثلاثا وللبكر سبعا١٩١٧ليس بك على أهلك هوان إن شئت سبعت لك وإن سبعت لك سبعت لنسائي١٩١٨إذا أفاد أحدكم امرأة أو خادما أو دابة فليأخذ بناصيتها وليقل اللهم إني…١٩١٩لو أن أحدكم إذا أتى امرأته قال اللهم جنبني الشيطان وجنب الشيطان ما…١٩٢٠احفظ عورتك إلا من زوجتك أو ما ملكت يمينك١٩٢١إذا أتى أحدكم أهله فليستتر ولا يتجرد تجرد العيرين١٩٢٢حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن سفيان، عن منصور، عن موسى بن…١٩٢٣لا ينظر الله إلى رجل جامع امرأته في دبرها١٩٢٤إن الله لا يستحيي من الحق١٩٢٥وجميل بن الحسن، قالا حدثنا سفيان بن عيينة، عن محمد بن المنكدر، أنه…١٩٢٦أوتفعلون لا عليكم أن لا تفعلوا فإنه ليس من نسمة قضى الله لها أن تكون…١٩٢٧حدثنا هارون بن إسحاق الهمداني، حدثنا سفيان، عن عمرو، عن عطاء، عن…١٩٢٨حديث ١٩٢٨١٩٢٩حديث ١٩٢٩١٩٣٠حدثنا أبو كريب، حدثنا عبدة بن سليمان، عن محمد بن إسحاق، عن يعقوب بن…١٩٣١لا تنكح المرأة على عمتها ولا على خالتها١٩٣٢أتريدين أن ترجعي إلى رفاعة لا حتى تذوقي عسيلته ويذوق عسيلتك١٩٣٣لا . حتى يذوق العسيلة١٩٣٤حدثنا محمد بن بشار، حدثنا أبو عامر، عن زمعة بن صالح، عن سلمة بن…١٩٣٥ومجالد، عن الشعبي، عن الحارث، عن علي، قال لعن رسول الله صلى الله عليه…١٩٣٦ألا أخبركم بالتيس المستعار١٩٣٧يحرم من الرضاع ما يحرم من النسب١٩٣٨إنها ابنة أخي من الرضاعة وإنه يحرم من الرضاعة ما يحرم من النسب١٩٣٩أن زينب بنت أبي سلمة، حدثته أن أم حبيبة حدثتها أنها، قالت لرسول الله…١٩٤٠لا تحرم الرضعة ولا الرضعتان أو المصة والمصتان١٩٤١لا تحرم المصة والمصتان١٩٤٢أنها قالت كان فيما أنزل الله من القرآن ثم سقط لا يحرم إلا عشر رضعات…١٩٤٣قالت جاءت سهلة بنت سهيل إلى النبي صلى الله عليه وسلم فقالت يا رسول…١٩٤٤حدثنا أبو سلمة، يحيى بن خلف حدثنا عبد الأعلى، عن محمد بن إسحاق، عن…١٩٤٥أن النبي صلى الله عليه وسلم دخل عليها وعندها رجل فقال " من هذا " .…١٩٤٦لا رضاع إلا ما فتق الأمعاء١٩٤٧وعقيل، عن ابن شهاب، أخبرني أبو عبيدة بن عبد الله بن زمعة، عن أمه،…١٩٤٨إنه عمك فأذني له١٩٤٩فليلج عليك عمك١٩٥٠إذا رجعت فطلق إحداهما١٩٥١طلق أيتهما شئت١٩٥٢اختر منهن أربعا١٩٥٣قال أسلم غيلان بن سلمة وتحته عشر نسوة فقال له النبي صلى الله عليه…١٩٥٤إن أحق الشرط أن يوفى به ما استحللتم به الفروج١٩٥٥حديث ١٩٥٥١٩٥٦من كانت له جارية فأدبها فأحسن أدبها وعلمها فأحسن تعليمها ثم أعتقها…١٩٥٧وعبد العزيز، عن أنس، قال صارت صفية لدحية الكلبي ثم صارت لرسول الله…١٩٥٨. أن رسول الله صلى الله عليه وسلم أعتق صفية وجعل عتقها صداقها وتزوجها…١٩٥٩إذا تزوج العبد بغير إذن سيده كان عاهرا١٩٦٠أيما عبد تزوج بغير إذن مواليه فهو زان١٩٦١والحسن، ابنى محمد بن علي عن أبيهما، عن علي بن أبي طالب، أن رسول الله…١٩٦٢فاستمتعوا من هذه النساء١٩٦٣حدثنا محمد بن خلف العسقلاني، حدثنا الفريابي، عن أبان بن أبي حازم، عن…١٩٦٤حدثتني ميمونة بنت الحارث، . أن رسول الله صلى الله عليه وسلم تزوجها…١٩٦٥أن النبي صلى الله عليه وسلم نكح وهو محرم .١٩٦٦المحرم لا ينكح ولا ينكح ولا يخطب١٩٦٧إذا أتاكم من ترضون خلقه ودينه فزوجوه إلا تفعلوا تكن فتنة في الأرض…١٩٦٨تخيروا لنطفكم وانكحوا الأكفاء وأنكحوا إليهم١٩٦٩من كانت له امرأتان يميل مع إحداهما على الأخرى جاء يوم القيامة وأحد…١٩٧٠أن رسول الله صلى الله عليه وسلم كان إذا سافر أقرع بين نسائه .١٩٧١اللهم هذا فعلي فيما أملك فلا تلمني فيما تملك ولا أملك١٩٧٢قالت لما أن كبرت، سودة بنت زمعة وهبت يومها لعائشة فكان رسول الله صلى…١٩٧٣يا عائشة إليك عني إنه ليس يومك١٩٧٤. أنها قالت نزلت هذه الآية {والصلح خير} في رجل كانت تحته امرأة قد…١٩٧٥من أفضل الشفاعة أن يشفع بين الاثنين في النكاح١٩٧٦حدثنا أبو بكر بن أبي شيبة، حدثنا شريك، عن العباس بن ذريح، عن البهي،…١٩٧٧خيركم خيركم لأهله وأنا خيركم لأهلي١٩٧٨خياركم خياركم لنسائهم١٩٧٩قالت سابقني النبي صلى الله عليه وسلم فسبقته .١٩٨٠قالت لما قدم رسول الله صلى الله عليه وسلم المدينة وهو عروس بصفية بنت…١٩٨١قال قالت عائشة ما علمت حتى دخلت على زينب بغير إذن وهي غضبى . ثم قالت…١٩٨٢رسول الله صلى الله عليه وسلم فكان يسرب إلى صواحباتي يلاعبنني .١٩٨٣إلام يجلد أحدكم امرأته جلد الأمة ولعله أن يضاجعها من آخر يومه١٩٨٤قالت ما ضرب رسول الله صلى الله عليه وسلم خادما له ولا امرأة ولا ضرب…١٩٨٥لا تضربوا إماء الله١٩٨٦لا يسأل الرجل فيم يضرب امرأته ولا تنم إلا على وتر١٩٨٧حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، وأبو أسامة عن عبيد…١٩٨٨لعن الله الواصلة والمستوصلة١٩٨٩حدثنا أبو عمر، حفص بن عمرو وعبد الرحمن بن عمر قالا حدثنا عبد الرحمن…١٩٩٠قالت تزوجني النبي صلى الله عليه وسلم في شوال وبنى بي في شوال . فأى…١٩٩١حدثنا أبو بكر بن أبي شيبة، حدثنا أسود بن عامر، حدثنا زهير، عن محمد بن…١٩٩٢أن رسول الله صلى الله عليه وسلم أمرها أن تدخل على رجل امرأته قبل أن…١٩٩٣لا شؤم وقد يكون اليمن في ثلاثة في المرأة والفرس والدار١٩٩٤إن كان ففي الفرس والمرأة والمسكن
