سنن ابن ماجه

كتاب الديات

٨٠ حديثًا
٢٦١٥أول ما يقضى بين الناس يوم القيامة في الدماء٢٦١٦لا تقتل نفس ظلما إلا كان على ابن آدم الأول كفل من دمها لأنه أول من سن…٢٦١٧أول ما يقضى بين الناس يوم القيامة في الدماء٢٦١٨من لقي الله لا يشرك به شيئا لم يتند بدم حرام دخل الجنة٢٦١٩لزوال الدنيا أهون على الله من قتل مؤمن بغير حق٢٦٢٠من أعان على قتل مؤمن بشطر كلمة لقي الله عز وجل مكتوب بين عينيه آيس من…٢٦٢١يجيء القاتل والمقتول يوم القيامة متعلق برأس صاحبه يقول رب سل هذا لم…٢٦٢٢إن عبدا قتل تسعة وتسعين نفسا ثم عرضت له التوبة فسأل عن أعلم أهل الأرض…٢٦٢٣من أصيب بدم أو خبل - والخبل الجرح - فهو بالخيار بين إحدى ثلاث فإن…٢٦٢٤من قتل له قتيل فهو بخير النظرين إما أن يقتل وإما أن يفدى٢٦٢٥وكانا، شهدا حنينا مع رسول الله صلى الله عليه وسلم قالا صلى النبي صلى…٢٦٢٦من قتل عمدا دفع إلى أولياء القتيل فإن شاءوا قتلوا وإن شاءوا أخذوا…٢٦٢٧قتيل الخطإ شبه العمد قتيل السوط والعصا مائة من الإبل أربعون منها خلفة…٢٦٢٨الحمد لله الذي صدق وعده ونصر عبده وهزم الأحزاب وحده ألا إن قتيل الخطإ…٢٦٢٩حدثنا محمد بن بشار، حدثنا معاذ بن هانئ، حدثنا محمد بن مسلم، عن عمرو…٢٦٣٠من قتل خطأ فديته من الإبل ثلاثون بنت مخاض وثلاثون ابنة لبون وثلاثون…٢٦٣١في دية الخطإ عشرون حقة وعشرون جذعة وعشرون بنت مخاض وعشرون بنت لبون…٢٦٣٢أن النبي صلى الله عليه وسلم جعل الدية اثنى عشر ألفا قال ذلك قوله {وما…٢٦٣٣قال قضى رسول الله صلى الله عليه وسلم بالدية على العاقلة .٢٦٣٤أنا وارث من لا وارث له أعقل عنه وأرثه والخال وارث من لا وارث له يعقل…٢٦٣٥من قتل في عمية أو عصبية بحجر أو سوط أو عصا فعليه عقل الخطإ ومن قتل…٢٦٣٦خذ الدية بارك الله لك فيها٢٦٣٧لا قود في المأمومة ولا الجائفة ولا المنقلة٢٦٣٨حدثنا محمد بن يحيى، حدثنا عبد الرزاق، أنبأنا معمر، عن الزهري، عن…٢٦٣٩إن هذا ليقول بقول شاعر فيه غرة عبد أو أمة٢٦٤٠وعلي بن محمد، قالا حدثنا وكيع، عن هشام بن عروة، عن أبيه، عن المسور بن…٢٦٤١أنه سمع طاوسا، عن ابن عباس، عن عمر بن الخطاب، أنه نشد الناس قضاء…٢٦٤٢أن عمر، كان يقول الدية للعاقلة ولا ترث المرأة من دية زوجها شيئا حتى…٢٦٤٣أن النبي صلى الله عليه وسلم قضى لحمل بن مالك الهذلي اللحياني بميراثه…٢٦٤٤. أن رسول الله صلى الله عليه وسلم قضى أن عقل أهل الكتابين نصف عقل…٢٦٤٥أن رسول الله صلى الله عليه وسلم قال " القاتل لا يرث " .٢٦٤٦ليس لقاتل ميراث٢٦٤٧حدثنا إسحاق بن منصور، أنبأنا يزيد بن هارون، أنبأنا محمد بن راشد، عن…٢٦٤٨لا ميراثها لزوجها وولدها٢٦٤٩يا أنس كتاب الله القصاص٢٦٥٠الأسنان سواء الثنية والضرس سواء٢٦٥١حدثنا إسماعيل بن إبراهيم البالسي، حدثنا علي بن الحسن بن شقيق، حدثنا…٢٦٥٢ومحمد بن جعفر، وابن أبي عدي، قالوا حدثنا شعبة، عن قتادة، عن عكرمة، عن…٢٦٥٣الأصابع سواء كلهن فيهن عشر عشر من الإبل٢٦٥٤حدثنا رجاء بن المرجى السمرقندي، حدثنا النضر بن شميل، حدثنا سعيد بن…٢٦٥٥في المواضح خمس خمس من الإبل٢٦٥٦يعمد أحدكم إلى أخيه فيعضه كعضاض الفحل ثم يأتي يلتمس العقل لا عقل لها٢٦٥٧يقضم أحدكم كما يقضم الفحل٢٦٥٨حدثنا علقمة بن عمرو الدارمي، حدثنا أبو بكر بن عياش، عن مطرف، عن…٢٦٥٩لا يقتل مسلم بكافر٢٦٦٠لا يقتل مؤمن بكافر ولا ذو عهد في عهده٢٦٦١لا يقتل بالولد الوالد٢٦٦٢لا يقتل الوالد بالولد٢٦٦٣من قتل عبده قتلناه ومن جدعه جدعناه٢٦٦٤قال قتل رجل عبده عمدا متعمدا فجلده رسول الله صلى الله عليه وسلم مائة…٢٦٦٥أن يهوديا، رضخ رأس امرأة بين حجرين فقتلها فرضخ رسول الله صلى الله…٢٦٦٦أن يهوديا، قتل جارية على أوضاح لها فقال لها " أقتلك فلان " . فأشارت…٢٦٦٧لا قود إلا بالسيف٢٦٦٨لا قود إلا بالسيف٢٦٦٩ألا لا يجني جان إلا على نفسه لا يجني والد على ولده ولا مولود على والده٢٦٧٠ألا لا تجني أم على ولد ألا لا تجني أم على ولد٢٦٧١لا تجني عليه ولا يجني عليك٢٦٧٢لا تجني نفس على أخرى٢٦٧٣العجماء جرحها جبار والمعدن جبار والبئر جبار٢٦٧٤العجماء جرحها جبار والمعدن جبار٢٦٧٥قال قضى رسول الله صلى الله عليه وسلم أن المعدن جبار والبئر جبار…٢٦٧٦النار جبار والبئر جبار٢٦٧٧من كبراء قومه أن عبد، الله بن سهل ومحيصة خرجا إلى خيبر من جهد أصابهم…٢٦٧٨تقسمون وتستحقون٢٦٧٩أنه قدم على النبي صلى الله عليه وسلم وقد خصى غلاما له فأعتقه النبي…٢٦٨٠قال جاء رجل إلى النبي صلى الله عليه وسلم صارخا فقال له رسول الله صلى…٢٦٨١إن من أعف الناس قتلة أهل الإيمان٢٦٨٢إن أعف الناس قتلة أهل الإيمان٢٦٨٣المسلمون تتكافأ دماؤهم وهم يد على من سواهم يسعى بذمتهم أدناهم ويرد…٢٦٨٤المسلمون يد على من سواهم وتتكافأ دماؤهم٢٦٨٥يد المسلمين على من سواهم تتكافأ دماؤهم ويجير على المسلمين أدناهم ويرد…٢٦٨٦من قتل معاهدا لم يرح رائحة الجنة وإن ريحها ليوجد من مسيرة أربعين عاما٢٦٨٧من قتل معاهدا له ذمة الله وذمة رسوله لم يرح رائحة الجنة وإن ريحها…٢٦٨٨من أمن رجلا على دمه فقتله فإنه يحمل لواء غدر يوم القيامة٢٦٨٩إذا أمنك الرجل على دمه فلا تقتله٢٦٩٠أما إنه إن كان صادقا ثم قتلته دخلت النار٢٦٩١فليس لأحد بعد النبي صلى الله عليه وسلم أن يقول " اقتله فإنك مثله " .…٢٦٩٢قال ما رفع إلى رسول الله صلى الله عليه وسلم شىء فيه القصاص إلا أمر…٢٦٩٣ما من رجل يصاب بشىء من جسده فيتصدق به إلا رفعه الله به درجة أو حط عنه…٢٦٩٤المرأة إذا قتلت عمدا لا تقتل حتى تضع ما في بطنها إن كانت حاملا وحتى…
