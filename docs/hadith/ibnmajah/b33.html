٣٦٥٧أوصي امرأ بأمه أوصي امرأ بأمه أوصي امرأ بأمه - ثلاثا - أوصي امرأ…٣٦٥٨قال قالوا يا رسول الله من أبر قال " أمك " . قال ثم من قال " أمك " .…٣٦٥٩لا يجزي ولد والده إلا أن يجده مملوكا فيشتريه فيعتقه٣٦٦٠القنطار اثنا عشر ألف أوقية كل أوقية خير مما بين السماء والأرض٣٦٦١إن الله يوصيكم بأمهاتكم - ثلاثا - إن الله يوصيكم بآبائكم إن الله…٣٦٦٢هما جنتك ونارك٣٦٦٣الوالد أوسط أبواب الجنة فأضع ذلك الباب أو احفظه٣٦٦٤نعم الصلاة عليهما والاستغفار لهما وإيفاء بعهودهما من بعد موتهما…٣٦٦٥وأملك أن كان الله قد نزع منكم الرحمة٣٦٦٦إن الولد مبخلة مجبنة٣٦٦٧ألا أدلكم على أفضل الصدقة ابنتك مردودة إليك ليس لها كاسب غيرك٣٦٦٨ما عجبك لقد دخلت به الجنة٣٦٦٩من كان له ثلاث بنات فصبر عليهن وأطعمهن وسقاهن وكساهن من جدته - كن له…٣٦٧٠ما من رجل تدرك له ابنتان فيحسن إليهما ما صحبتاه أو صحبهما إلا أدخلتاه…٣٦٧١أكرموا أولادكم وأحسنوا أدبهم٣٦٧٢من كان يؤمن بالله واليوم الآخر فليحسن إلى جاره ومن كان يؤمن بالله…٣٦٧٣ما زال جبريل يوصيني بالجار حتى ظننت أنه سيورثه٣٦٧٤ما زال جبرائيل يوصيني بالجار حتى ظننت أنه سيورثه٣٦٧٥من كان يؤمن بالله واليوم الآخر فليكرم ضيفه وجائزته يوم وليلة ولا يحل…٣٦٧٦إن نزلتم بقوم فأمروا لكم بما ينبغي للضيف فاقبلوا وإن لم يفعلوا فخذوا…٣٦٧٧ليلة الضيف واجبة فإن أصبح بفنائه فهو دين عليه فإن شاء اقتضى وإن شاء…٣٦٧٨اللهم إني أحرج حق الضعيفين اليتيم والمرأة٣٦٧٩خير بيت في المسلمين بيت فيه يتيم يحسن إليه وشر بيت في المسلمين بيت…٣٦٨٠من عال ثلاثة من الأيتام كان كمن قام ليله وصام نهاره وغدا وراح شاهرا…٣٦٨١اعزل الأذى عن طريق المسلمين٣٦٨٢كان على الطريق غصن شجرة يؤذي الناس فأماطها رجل فأدخل الجنة٣٦٨٣حدثنا أبو بكر بن أبي شيبة، حدثنا يزيد بن هارون، أنبأنا هشام بن حسان،…٣٦٨٤قال قلت يا رسول الله أى الصدقة أفضل قال " سقى الماء " .٣٦٨٥يصف الناس يوم القيامة صفوفا - وقال ابن نمير أهل الجنة - فيمر الرجل من…٣٦٨٦نعم في كل ذات كبد حرى أجر٣٦٨٧من يحرم الرفق يحرم الخير٣٦٨٨إن الله رفيق يحب الرفق ويعطي عليه ما لا يعطي على العنف٣٦٨٩إن الله رفيق يحب الرفق في الأمر كله٣٦٩٠إخوانكم جعلهم الله تحت أيديكم فأطعموهم مما تأكلون وألبسوهم مما تلبسون…٣٦٩١لا يدخل الجنة سيئ الملكة٣٦٩٢والذي نفسي بيده لا تدخلوا الجنة حتى تؤمنوا ولا تؤمنوا حتى تحابوا .…٣٦٩٣حدثنا أبو بكر بن أبي شيبة، حدثنا إسماعيل بن عياش، عن محمد بن زياد، عن…٣٦٩٤اعبدوا الرحمن وأفشوا السلام٣٦٩٥أن رجلا، دخل المسجد ورسول الله صلى الله عليه وسلم جالس في ناحية…٣٦٩٦إن جبرائيل يقرأ عليك السلام٣٦٩٧إذا سلم عليكم أحد من أهل الكتاب فقولوا وعليكم٣٦٩٨أنه أتى النبي صلى الله عليه وسلم ناس من اليهود فقالوا السام عليك يا…٣٦٩٩إني راكب غدا إلى اليهود فلا تبدءوهم بالسلام فإذا سلموا عليكم فقولوا…٣٧٠٠حدثنا أبو بكر، حدثنا أبو خالد الأحمر، عن حميد، عن أنس، قال أتانا رسول…٣٧٠١سمعه من، شهر بن حوشب يقول أخبرته أسماء بنت يزيد، قالت مر علينا رسول…٣٧٠٢حدثنا علي بن محمد، حدثنا وكيع، عن جرير بن حازم، عن حنظلة بن عبد…٣٧٠٣ما من مسلمين يلتقيان فيتصافحان إلا غفر لهما قبل أن يتفرقا٣٧٠٤حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن فضيل، حدثنا يزيد بن أبي…٣٧٠٥وغندر، وأبو أسامة عن شعبة، عن عمرو بن مرة، عن عبد الله بن سلمة، عن…٣٧٠٦أن أبا موسى، استأذن على عمر ثلاثا فلم يؤذن له فانصرف فأرسل إليه عمر…٣٧٠٧يتكلم الرجل تسبيحة وتكبيرة وتحميدة ويتنحنح ويؤذن أهل البيت٣٧٠٨قال كان لي من رسول الله صلى الله عليه وسلم مدخلان مدخل بالليل ومدخل…٣٧٠٩حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن شعبة، عن محمد بن المنكدر،…٣٧١٠بخير من رجل لم يصبح صائما ولم يعد سقيما٣٧١١حدثنا أبو إسحاق الهروي، إبراهيم بن عبد الله بن حاتم حدثنا عبد الله بن…٣٧١٢إذا أتاكم كريم قوم فأكرموه٣٧١٣إن هذا حمد الله وإن هذا لم يحمد الله٣٧١٤يشمت العاطس ثلاثا فما زاد فهو مزكوم٣٧١٥إذا عطس أحدكم فليقل الحمد لله . وليرد عليه من حوله يرحمك الله . وليرد…٣٧١٦قال كان النبي صلى الله عليه وسلم إذا لقي الرجل فكلمه لم يصرف وجهه عنه…٣٧١٧إذا قام أحدكم عن مجلسه ثم رجع فهو أحق به٣٧١٨من اعتذر إلى أخيه بمعذرة فلم يقبلها كان عليه مثل خطيئة صاحب مكس٣٧١٩حدثنا أبو بكر، حدثنا وكيع، عن زمعة بن صالح، عن الزهري، عن وهب بن عبد…٣٧٢٠يا أبا عمير ما فعل النغير٣٧٢١حدثنا أبو بكر بن أبي شيبة، حدثنا عبدة بن سليمان، عن محمد بن إسحاق، عن…٣٧٢٢أن النبي صلى الله عليه وسلم نهى أن يقعد بين الظل والشمس .٣٧٢٣ما لك ولهذا النوم هذه نومة يكرهها الله أو يبغضها الله٣٧٢٤يا جنيدب إنما هذه ضجعة أهل النار٣٧٢٥قم واقعد فإنها نومة جهنمية٣٧٢٦من اقتبس علما من النجوم اقتبس شعبة من السحر زاد ما زاد٣٧٢٧لا تسبوا الريح فإنها من روح الله تأتي بالرحمة والعذاب ولكن سلوا الله…٣٧٢٨أحب الأسماء إلى الله عز وجل عبد الله وعبد الرحمن٣٧٢٩لئن عشت إن شاء الله لأنهين أن يسمى رباح ونجيح وأفلح ونافع ويسار٣٧٣٠حدثنا أبو بكر، حدثنا المعتمر بن سليمان، عن الركين، عن أبيه، عن سمرة،…٣٧٣١قال لقيت عمر بن الخطاب فقال من أنت فقلت مسروق بن الأجدع . فقال عمر…٣٧٣٢أن زينب، كان اسمها برة فقيل لها تزكي نفسها فسماها رسول الله صلى الله…٣٧٣٣أن ابنة لعمر، كان يقال لها عاصية فسماها رسول الله صلى الله عليه وسلم…٣٧٣٤حدثنى ابن أخي عبد الله بن سلام، عن عبد الله بن سلام، قال قدمت على…٣٧٣٥تسموا باسمي ولا تكنوا بكنيتي٣٧٣٦تسموا باسمي ولا تكنوا بكنيتي٣٧٣٧تسموا باسمي ولا تكتنوا بكنيتي٣٧٣٨أن عمر، قال لصهيب ما لك تكتني بأبي يحيى وليس لك ولد . قال كناني رسول…٣٧٣٩فأنت أم عبد الله٣٧٤٠حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن شعبة، عن أبي التياح، عن…٣٧٤١حدثنا أبو بكر، حدثنا عبد الله بن إدريس، عن داود، عن الشعبي، عن أبي…٣٧٤٢حدثنا أبو بكر، حدثنا عبد الرحمن بن مهدي، عن سفيان، عن حبيب بن أبي…٣٧٤٣إياكم والتمادح فإنه الذبح٣٧٤٤ويحك قطعت عنق صاحبك٣٧٤٥المستشار مؤتمن٣٧٤٦المستشار مؤتمن٣٧٤٧إذا استشار أحدكم أخاه فليشر عليه٣٧٤٨تفتح لكم أرض الأعاجم وستجدون فيها بيوتا يقال لها الحمامات فلا يدخلها…٣٧٤٩حدثنا علي بن محمد، حدثنا وكيع، ح وحدثنا أبو بكر بن أبي شيبة، حدثنا…٣٧٥٠أيما امرأة وضعت ثيابها في غير بيت زوجها فقد هتكت ستر ما بينها وبين…٣٧٥١أن النبي صلى الله عليه وسلم كان إذا اطلى بدأ بعورته فطلاها بالنورة…٣٧٥٢أن النبي اطلى وولي عانته بيده .٣٧٥٣لا يقص على الناس إلا أمير أو مأمور أو مراء٣٧٥٤قال لم يكن القصص في زمن رسول الله صلى الله عليه وسلم ولا زمن أبي بكر…٣٧٥٥إن من الشعر لحكمة٣٧٥٦إن من الشعر حكما٣٧٥٧أصدق كلمة قالها الشاعر كلمة لبيد ألا كل شىء ما خلا الله باطل وكاد…٣٧٥٨قال أنشدت رسول الله صلى الله عليه وسلم مائة قافية من شعر أمية بن أبي…٣٧٥٩لأن يمتلئ جوف الرجل قيحا حتى يريه خير له من أن يمتلئ شعرا٣٧٦٠لأن يمتلئ جوف أحدكم قيحا حتى يريه خير له من أن يمتلئ شعرا٣٧٦١إن أعظم الناس فرية لرجل هاجى رجلا فهجا القبيلة بأسرها ورجل انتفى من…٣٧٦٢من لعب بالنرد فقد عصى الله ورسوله٣٧٦٣من لعب بالنردشير فكأنما غمس يده في لحم خنزير ودمه٣٧٦٤شيطان يتبع شيطانا٣٧٦٥شيطان يتبع شيطانة٣٧٦٦شيطان يتبع شيطانة٣٧٦٧شيطان يتبع شيطانا٣٧٦٨لو يعلم أحدكم ما في الوحدة ما سار أحد بليل وحده٣٧٦٩لا تتركوا النار في بيوتكم حين تنامون٣٧٧٠إنما هذه النار عدو لكم فإذا نمتم فأطفئوها عنكم٣٧٧١حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، عن عبد الملك، عن…٣٧٧٢لا تنزلوا على جواد الطريق ولا تقضوا عليها الحاجات٣٧٧٣حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الرحيم بن سليمان، عن عاصم، حدثنا…٣٧٧٤تربوا صحفكم أنجح لها إن التراب مبارك٣٧٧٥إذا كنتم ثلاثة فلا يتناجى اثنان دون صاحبهما فإن ذلك يحزنه٣٧٧٦قال نهى رسول الله صلى الله عليه وسلم أن يتناجى اثنان دون الثالث .٣٧٧٧يقول مر رجل بسهام في المسجد فقال رسول الله صلى الله عليه وسلم " أمسك…٣٧٧٨إذا مر أحدكم في مسجدنا أو في سوقنا ومعه نبل فليمسك على نصالها بكفه أن…٣٧٧٩الماهر بالقرآن مع السفرة الكرام البررة والذي يقرؤه يتتعتع فيه وهو…٣٧٨٠يقال لصاحب القرآن إذا دخل الجنة اقرأ واصعد . فيقرأ ويصعد بكل آية درجة…٣٧٨١يجيء القرآن يوم القيامة كالرجل الشاحب فيقول أنا الذي أسهرت ليلك…٣٧٨٢أيحب أحدكم إذا رجع إلى أهله أن يجد فيه ثلاث خلفات عظام سمان٣٧٨٣مثل القرآن مثل الإبل المعقلة إن تعاهدها صاحبها بعقلها أمسكها عليه وإن…٣٧٨٤قال الله عز وجل قسمت الصلاة بيني وبين عبدي شطرين فنصفها لي ونصفها…٣٧٨٥ألا أعلمك أعظم سورة في القرآن قبل أن أخرج من المسجد٣٧٨٦إن سورة في القرآن ثلاثون آية شفعت لصاحبها حتى غفر له {تبارك الذي بيده…٣٧٨٧{قل هو الله أحد} تعدل ثلث القرآن٣٧٨٨{قل هو الله أحد} تعدل ثلث القرآن٣٧٨٩الله أحد الواحد الصمد تعدل ثلث القرآن٣٧٩٠ألا أنبئكم بخير أعمالكم وأرضاها عند مليككم وأرفعها في درجاتكم وخير…٣٧٩١ما جلس قوم مجلسا يذكرون الله فيه إلا حفتهم الملائكة وتغشتهم الرحمة…٣٧٩٢إن الله عز وجل يقول أنا مع عبدي إذا هو ذكرني وتحركت بي شفتاه٣٧٩٣لا يزال لسانك رطبا من ذكر الله عز وجل٣٧٩٤إذا قال العبد لا إله إلا الله والله أكبر . قال يقول الله عز وجل صدق…٣٧٩٥إني لأعلم كلمة لا يقولها أحد عند موته إلا كانت نورا لصحيفته وإن جسده…٣٧٩٦ما من نفس تموت تشهد أن لا إله إلا الله وأني رسول الله صلى الله عليه…٣٧٩٧لا إله إلا الله لا يسبقها عمل ولا تترك ذنبا٣٧٩٨من قال في يوم مائة مرة لا إله إلا الله وحده لا شريك له له الملك وله…٣٧٩٩من قال في دبر صلاة الغداة لا إله إلا الله وحده لا شريك له له الملك…٣٨٠٠أفضل الذكر لا إله إلا الله وأفضل الدعاء الحمد لله٣٨٠١أن عبدا من عباد الله قال يا رب لك الحمد كما ينبغي لجلال وجهك ولعظيم…٣٨٠٢من ذا الذي قال هذا٣٨٠٣الحمد لله الذي بنعمته تتم الصالحات٣٨٠٤الحمد لله على كل حال رب أعوذ بك من حال أهل النار٣٨٠٥ما أنعم الله على عبد نعمة فقال الحمد لله . إلا كان الذي أعطاه أفضل…٣٨٠٦كلمتان خفيفتان على اللسان ثقيلتان في الميزان حبيبتان إلى الرحمن سبحان…
