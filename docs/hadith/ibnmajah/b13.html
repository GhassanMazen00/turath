سنن ابن ماجه

كتاب الأحكام

٦٧ حديثًا
٢٣٠٨من جعل قاضيا بين الناس فقد ذبح بغير سكين٢٣٠٩من سأل القضاء وكل إلى نفسه ومن جبر عليه نزل إليه ملك فسدده٢٣١٠اللهم اهد قلبه وثبت لسانه٢٣١١ما من حاكم يحكم بين الناس إلا جاء يوم القيامة وملك آخذ بقفاه ثم يرفع…٢٣١٢إن الله مع القاضي ما لم يجر فإذا جار وكله إلى نفسه٢٣١٣لعنة الله على الراشي والمرتشي٢٣١٤إذا حكم الحاكم فاجتهد فأصاب فله أجران وإذا حكم فاجتهد فأخطأ فله أجر٢٣١٥القضاة ثلاثة اثنان في النار وواحد في الجنة رجل علم الحق فقضى به فهو…٢٣١٦لا يقضي القاضي بين اثنين وهو غضبان٢٣١٧إنكم تختصمون إلى وإنما أنا بشر ولعل بعضكم أن يكون ألحن بحجته من بعض…٢٣١٨إنما أنا بشر ولعل بعضكم أن يكون ألحن بحجته من بعض فمن قطعت له من حق…٢٣١٩من ادعى ما ليس له فليس منا وليتبوأ مقعده من النار٢٣٢٠من أعان على خصومة بظلم - أو يعين على ظلم - لم يزل في سخط الله حتى ينزع٢٣٢١لو يعطى الناس بدعواهم ادعى ناس دماء رجال وأموالهم ولكن اليمين على…٢٣٢٢وعلي بن محمد، قالا حدثنا وكيع، وأبو معاوية قالا حدثنا الأعمش، عن…٢٣٢٣من حلف على يمين وهو فيها فاجر يقتطع بها مال امرئ مسلم لقي الله وهو…٢٣٢٤لا يقتطع رجل حق امرئ مسلم بيمينه إلا حرم الله عليه الجنة وأوجب له…٢٣٢٥من حلف بيمين آثمة عند منبري هذا فليتبوأ مقعده من النار ولو على سواك…٢٣٢٦لا يحلف عند هذا المنبر عبد ولا أمة على يمين آثمة ولو على سواك رطب إلا…٢٣٢٧أنشدك بالله الذي أنزل التوراة على موسى٢٣٢٨نشدتكما بالله الذي أنزل التوراة على موسى عليه السلام٢٣٢٩أنه ذكر أن رجلين، ادعيا دابة ولم يكن بينهما بينة فأمرهما النبي صلى…٢٣٣٠ومحمد بن معمر، وزهير بن محمد، قالوا حدثنا روح بن عبادة، حدثنا سعيد،…٢٣٣١إذا ضاع للرجل متاع أو سرق له متاع فوجده في يد رجل يبيعه فهو أحق به…٢٣٣٢أن ابن محيصة الأنصاري، أخبره أن ناقة للبراء كانت ضارية دخلت في حائط…٢٣٣٣خذوا ظرفا مكان ظرفكم٢٣٣٤قال كان النبي صلى الله عليه وسلم عند إحدى أمهات المؤمنين فأرسلت أخرى…٢٣٣٥إذا استأذن أحدكم جاره أن يغرز خشبة في جداره فلا يمنعه٢٣٣٦لا يمنع أحدكم جاره أن يغرز خشبة في جداره٢٣٣٧لا يمنع أحدكم جاره أن يضع خشبة على جداره٢٣٣٨اجعلوا الطريق سبعة أذرع٢٣٣٩إذا اختلفتم في الطريق فاجعلوه سبعة أذرع٢٣٤٠أن رسول الله صلى الله عليه وسلم قضى أن " لا ضرر ولا ضرار " .٢٣٤١قال قال رسول الله صلى الله عليه وسلم " لا ضرر ولا ضرار " .٢٣٤٢من ضار أضر الله به ومن شاق شق الله عليه٢٣٤٣وعمار بن خالد الواسطي، قالا حدثنا أبو بكر بن عياش، عن دهثم بن قران،…٢٣٤٤إذا بيع البيع من رجلين فالبيع للأول٢٣٤٥ومحمد بن المثنى، قالا حدثنا عبد الأعلى، حدثنا خالد الحذاء، عن أبي…٢٣٤٦أن رجلين، تدارءا في بيع ليس لواحد منهما بينة فأمرهما رسول الله صلى…٢٣٤٧أن النبي صلى الله عليه وسلم كان إذا سافر أقرع بين نسائه .٢٣٤٨قال أتي علي بن أبي طالب وهو باليمن في ثلاثة قد وقعوا على امرأة في طهر…٢٣٤٩يا عائشة ألم ترى أن مجززا المدلجي دخل على فرأى أسامة وزيدا عليهما…٢٣٥٠أن قريشا، أتوا امرأة كاهنة فقالوا لها أخبرينا أشبهنا أثرا بصاحب…٢٣٥١يا غلام هذه أمك وهذا أبوك٢٣٥٢أن أبويه، اختصما إلى النبي صلى الله عليه وسلم أحدهما كافر والآخر مسلم…٢٣٥٣الصلح جائز بين المسلمين إلا صلحا حرم حلالا أو أحل حراما٢٣٥٤إذا بايعت فقل ها ولا خلابة٢٣٥٥إذا أنت بايعت فقل لا خلابة . ثم أنت في كل سلعة ابتعتها بالخيار ثلاث…٢٣٥٦قال أصيب رجل في عهد رسول الله صلى الله عليه وسلم في ثمار ابتاعها فكثر…٢٣٥٧أن رسول الله صلى الله عليه وسلم خلع معاذ بن جبل من غرمائه ثم استعمله…٢٣٥٨من وجد متاعه بعينه عند رجل قد أفلس فهو أحق به من غيره٢٣٥٩أيما رجل باع سلعة فأدرك سلعته بعينها عند رجل وقد أفلس ولم يكن قبض من…٢٣٦٠أيما رجل مات أو أفلس فصاحب المتاع أحق بمتاعه إذا وجده بعينه٢٣٦١أيما امرئ مات وعنده مال امرئ بعينه اقتضى منه شيئا أو لم يقتض فهو أسوة…٢٣٦٢قرني ثم الذين يلونهم ثم الذين يلونهم ثم يجيء قوم تبدر شهادة أحدهم…٢٣٦٣احفظوني في أصحابي ثم الذين يلونهم ثم الذين يلونهم ثم يفشو الكذب حتى…٢٣٦٤خير الشهود من أدى شهادته قبل أن يسألها٢٣٦٥وجميل بن الحسن العتكي، قالا حدثنا محمد بن مروان العجلي، حدثنا عبد…٢٣٦٦لا تجوز شهادة خائن ولا خائنة ولا محدود في الإسلام ولا ذي غمر على أخيه٢٣٦٧لا تجوز شهادة بدوي على صاحب قرية٢٣٦٨أن رسول الله صلى الله عليه وسلم قضى باليمين مع الشاهد .٢٣٦٩أن النبي صلى الله عليه وسلم قضى باليمين مع الشاهد .٢٣٧٠قال قضى رسول الله صلى الله عليه وسلم بالشاهد واليمين .٢٣٧١أن النبي صلى الله عليه وسلم أجاز شهادة الرجل ويمين الطالب .٢٣٧٢عدلت شهادة الزور بالإشراك بالله٢٣٧٣قال قال رسول الله صلى الله عليه وسلم { لن تزول قدم شاهد الزور حتى…٢٣٧٤أن رسول الله صلى الله عليه وسلم أجاز شهادة أهل الكتاب بعضهم على بعض .
