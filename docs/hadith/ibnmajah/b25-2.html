سنن ابن ماجه

٢٣٨ حديثًا•صفحة ٢ من ٢
٣٠٣٢أنه رمى جمرة العقبة ولم يقف عندها وذكر أن النبي صلى الله عليه وسلم…٣٠٣٣قال كان رسول الله صلى الله عليه وسلم إذا رمى جمرة العقبة مضى ولم يقف .٣٠٣٤أن النبي صلى الله عليه وسلم رمى الجمرة على راحلته .٣٠٣٥قال رأيت رسول الله صلى الله عليه وسلم رمى الجمرة يوم النحر على ناقة…٣٠٣٦أن النبي صلى الله عليه وسلم رخص للرعاء أن يرموا يوما ويدعوا يوما .٣٠٣٧قال رخص رسول الله صلى الله عليه وسلم لرعاء الإبل في البيتوتة أن يرموا…٣٠٣٨حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، عن أشعث، عن أبي…٣٠٣٩أن النبي صلى الله عليه وسلم لبى حتى رمى جمرة العقبة .٣٠٤٠قال قال الفضل بن عباس كنت ردف النبي صلى الله عليه وسلم فما زلت أسمعه…٣٠٤١وعلي بن محمد، قالا حدثنا وكيع، ح وحدثنا أبو بكر بن خلاد الباهلي،…٣٠٤٢قالت طيبت رسول الله صلى الله عليه وسلم لإحرامه حين أحرم ولإحلاله حين…٣٠٤٣اللهم اغفر للمحلقين٣٠٤٤رحم الله المحلقين٣٠٤٥قال قيل يا رسول الله لم ظاهرت للمحلقين ثلاثا وللمقصرين واحدة قال "…٣٠٤٦إني لبدت رأسي وقلدت هديي فلا أحل حتى أنحر٣٠٤٧حدثنا أحمد بن عمرو بن السرح المصري، أنبأنا عبد الله بن وهب، أنبأنا…٣٠٤٨منى كلها منحر وكل فجاج مكة طريق ومنحر وكل عرفة موقف وكل المزدلفة موقف٣٠٤٩قال ما سئل رسول الله صلى الله عليه وسلم عمن قدم شيئا قبل شىء إلا يلقي…٣٠٥٠قال كان رسول الله صلى الله عليه وسلم يسأل يوم منى فيقول " لا حرج لا…٣٠٥١أن النبي صلى الله عليه وسلم سئل عمن ذبح قبل أن يحلق أو حلق قبل أن…٣٠٥٢أنه سمع جابر بن عبد الله، يقول قعد رسول الله صلى الله عليه وسلم بمنى…٣٠٥٣قال رأيت رسول الله صلى الله عليه وسلم رمى جمرة العقبة ضحى وأما بعد…٣٠٥٤أن رسول الله صلى الله عليه وسلم كان يرمي الجمار إذا زالت الشمس قدر ما…٣٠٥٥يا أيها الناس ألا أى يوم أحرم٣٠٥٦نضر الله امرأ سمع مقالتي فبلغها فرب حامل فقه غير فقيه ورب حامل فقه…٣٠٥٧أتدرون أى يوم هذا وأى شهر هذا وأى بلد هذا٣٠٥٨أن رسول الله صلى الله عليه وسلم وقف يوم النحر بين الجمرات في الحجة…٣٠٥٩وأبو الزبير، عن عائشة، وابن، عباس أن النبي صلى الله عليه وسلم أخر…٣٠٦٠أن النبي صلى الله عليه وسلم لم يرمل في السبع الذي أفاض فيه . قال عطاء…٣٠٦١إن آية ما بيننا وبين المنافقين أنهم لا يتضلعون من زمزم٣٠٦٢ماء زمزم لما شرب له٣٠٦٣حدثنا عبد الرحمن بن إبراهيم الدمشقي، حدثنا عمر بن عبد الواحد، عن…٣٠٦٤إني دخلت الكعبة ووددت أني لم أكن فعلت إني أخاف أن أكون أتعبت أمتي من…٣٠٦٥قال استأذن العباس بن عبد المطلب رسول الله صلى الله عليه وسلم أن يبيت…٣٠٦٦وهناد بن السري، قالا حدثنا أبو معاوية، عن إسماعيل بن مسلم، عن عطاء،…٣٠٦٧وعبدة، ووكيع، وأبو معاوية ح وحدثنا علي بن محمد، حدثنا وكيع، وأبو…٣٠٦٨قالت ادلج النبي صلى الله عليه وسلم ليلة النفر من البطحاء ادلاجا .٣٠٦٩قال كان رسول الله صلى الله عليه وسلم وأبو بكر وعمر وعثمان ينزلون…٣٠٧٠لا ينفرن أحد حتى يكون آخر عهده بالبيت٣٠٧١قال نهى رسول الله صلى الله عليه وسلم أن ينفر الرجل حتى يكون آخر عهده…٣٠٧٢وعروة، عن عائشة، قالت حاضت صفية بنت حيى بعدما أفاضت . قالت عائشة…٣٠٧٣عقرى حلقى ما أراها إلا حابستنا٣٠٧٤اغتسلي واستثفري بثوب وأحرمي٣٠٧٥حدثنا أبو بكر بن أبي شيبة، حدثنا محمد بن بشر العبدي، عن محمد بن عمرو،…٣٠٧٦حدثنا القاسم بن محمد بن عباد المهلبي، حدثنا عبد الله بن داود، حدثنا…٣٠٧٧من كسر أو عرج فقد حل وعليه حجة أخرى٣٠٧٨من كسر أو مرض أو عرج فقد حل وعليه الحج من قابل٣٠٧٩ما كنت أرى الجهد بلغ منك ما أرى أتجد شاة٣٠٨٠قال أمرني النبي صلى الله عليه وسلم حين آذاني القمل أن أحلق رأسي وأصوم…٣٠٨١أن رسول الله صلى الله عليه وسلم احتجم وهو صائم محرم .٣٠٨٢أن النبي صلى الله عليه وسلم احتجم وهو محرم من رهصة أخذته .٣٠٨٣أن النبي صلى الله عليه وسلم كان يدهن رأسه بالزيت وهو محرم غير المقتت .٣٠٨٤اغسلوه بماء وسدر وكفنوه في ثوبيه ولا تخمروا وجهه ولا رأسه فإنه يبعث…٣٠٨٥قال جعل رسول الله صلى الله عليه وسلم في الضبع يصيبه المحرم كبشا وجعله…٣٠٨٦أن رسول الله صلى الله عليه وسلم قال في بيض النعام يصيبه المحرم " ثمنه…٣٠٨٧خمس فواسق يقتلن في الحل والحرم الحية والغراب الأبقع والفأرة والكلب…٣٠٨٨قال قال رسول الله صلى الله عليه وسلم خمس من الدواب لا جناح على من…٣٠٨٩يقتل المحرم الحية والعقرب والسبع العادي والكلب العقور والفأرة الفويسقة٣٠٩٠إنه ليس بنا رد عليك ولكنا حرم٣٠٩١قال أتي النبي صلى الله عليه وسلم بلحم صيد وهو محرم فلم يأكله .٣٠٩٢أن النبي صلى الله عليه وسلم أعطاه حمار وحش وأمره أن يفرقه في الرفاق…٣٠٩٣قال خرجت مع رسول الله صلى الله عليه وسلم زمن الحديبية فأحرم أصحابه…٣٠٩٤وعمرة بنت عبد الرحمن، أن عائشة، زوج النبي صلى الله عليه وسلم قالت كان…٣٠٩٥زوج النبي صلى الله عليه وسلم قالت كنت أفتل القلائد لهدى النبي صلى…٣٠٩٦وعلي بن محمد، قالا حدثنا أبو معاوية، عن الأعمش، عن إبراهيم، عن…٣٠٩٧وعلي بن محمد، قالا حدثنا وكيع، عن هشام الدستوائي، عن قتادة، عن أبي…٣٠٩٨أن النبي صلى الله عليه وسلم قلد وأشعر وأرسل بها ولم يجتنب ما يجتنب…٣٠٩٩قال أمرني رسول الله صلى الله عليه وسلم أن أقوم على بدنه وأن أقسم…٣١٠٠وعلي بن محمد، قالا حدثنا وكيع، حدثنا سفيان، عن ابن أبي ليلى، عن…٣١٠١أن النبي صلى الله عليه وسلم كان في بدنه جمل .٣١٠٢أن النبي صلى الله عليه وسلم اشترى هديه من قديد .٣١٠٣. أن النبي صلى الله عليه وسلم رأى رجلا يسوق بدنة فقال " اركبها " .…٣١٠٤أن النبي صلى الله عليه وسلم مر عليه ببدنة فقال " اركبها " . قال إنها…٣١٠٥إذا عطب منها شىء فخشيت عليه موتا فانحرها ثم اغمس نعلها في دمها ثم…٣١٠٦انحره واغمس نعله في دمه ثم اضرب صفحته وخل بينه وبين الناس فليأكلوه٣١٠٧قال توفي رسول الله صلى الله عليه وسلم وأبو بكر وعمر وما تدعى رباع مكة…٣١٠٨والله إنك لخير أرض الله وأحب أرض الله إلي والله لولا أني أخرجت منك ما…٣١٠٩يا أيها الناس إن الله حرم مكة يوم خلق السموات والأرض فهي حرام إلى يوم…٣١١٠لا تزال هذه الأمة بخير ما عظموا هذه الحرمة حق تعظيمها فإذا ضيعوا ذلك…٣١١١إن الإيمان ليأرز إلى المدينة كما تأرز الحية إلى جحرها٣١١٢قال قال رسول الله صلى الله عليه وسلم من استطاع منكم أن يموت بالمدينة…٣١١٣اللهم إن إبراهيم خليلك ونبيك وإنك حرمت مكة على لسان إبراهيم اللهم…٣١١٤من أراد أهل المدينة بسوء أذابه الله كما يذوب الملح في الماء٣١١٥إن أحدا جبل يحبنا ونحبه وهو على ترعة من ترع الجنة وعير على ترعة من…٣١١٦قال بعث رجل معي بدراهم هدية إلى البيت . قال فدخلت البيت وشيبة جالس…٣١١٧من أدرك رمضان بمكة فصام وقام منه ما تيسر له كتب الله له مائة ألف شهر…٣١١٨ائتنفوا العمل فقد غفر لكم٣١١٩اربطوا أوساطكم بأزركم
