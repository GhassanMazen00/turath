كتاب الطهارة وسننها

٤٠٠ حديثًا•صفحة ٣ من ٣
٥٦٧جعلت لي الأرض مسجدا وطهورا٥٦٨أنها استعارت من أسماء قلادة فهلكت فأرسل النبي صلى الله عليه وسلم…٥٦٩أن رجلا، أتى عمر بن الخطاب فقال إني أجنبت فلم أجد الماء . فقال عمر لا…٥٧٠وسلمة بن كهيل، أنهما سألا عبد الله بن أبي أوفى عن التيمم، فقال أمر…٥٧١حين تيمموا مع رسول الله صلى الله عليه وسلم فأمر المسلمين فضربوا…٥٧٢قتلوه قتلهم الله أفلم يكن شفاء العي السؤال٥٧٣وعلي بن محمد، قالا حدثنا وكيع، عن الأعمش، عن سالم بن أبي الجعد، عن…٥٧٤حدثنا محمد بن عبد الملك بن أبي الشوارب، حدثنا عبد الواحد بن زياد،…٥٧٥أما أنا فأفيض على رأسي ثلاث أكف٥٧٦وعلي بن محمد، قالا حدثنا وكيع، ح وحدثنا أبو كريب، حدثنا ابن فضيل،…٥٧٧أما أنا فأحثو على رأسي ثلاثا٥٧٨حدثنا أبو بكر بن أبي شيبة، حدثنا أبو خالد الأحمر، عن ابن عجلان، عن…٥٧٩وعبد الله بن عامر بن زرارة، وإسماعيل بن موسى السدي، قالوا حدثنا شريك،…٥٨٠قالت كان رسول الله صلى الله عليه وسلم يغتسل من الجنابة ثم يستدفئ بي…٥٨١قالت كان رسول الله صلى الله عليه وسلم يجنب ثم ينام ولا يمس ماء حتى…٥٨٢قالت إن رسول الله صلى الله عليه وسلم إن كانت له إلى أهله حاجة قضاها…٥٨٣أن رسول الله صلى الله عليه وسلم كان يجنب ثم ينام كهيئته لا يمس ماء .…٥٨٤قالت كان رسول الله صلى الله عليه وسلم إذا أراد أن ينام وهو جنب توضأ…٥٨٥أن عمر بن الخطاب، قال لرسول الله صلى الله عليه وسلم أيرقد أحدنا وهو…٥٨٦أنه كان تصيبه الجنابة بالليل فيريد أن ينام فأمره رسول الله صلى الله…٥٨٧إذا أتى أحدكم أهله ثم أراد أن يعود فليتوضأ٥٨٨أن النبي صلى الله عليه وسلم كان يطوف على نسائه في غسل واحد .٥٨٩قال وضعت لرسول الله صلى الله عليه وسلم غسلا فاغتسل من جميع نسائه في…٥٩٠هو أزكى وأطيب وأطهر٥٩١وغندر، ووكيع، عن شعبة، عن الحكم، عن إبراهيم، عن الأسود، عن عائشة،…٥٩٢نعم إذا توضأ وضوءه للصلاة٥٩٣أن النبي صلى الله عليه وسلم كان إذا أراد أن يأكل وهو جنب غسل يديه .٥٩٤حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، عن عمرو بن مرة،…٥٩٥لا يقرأ القرآن الجنب ولا الحائض٥٩٦لا يقرأ الجنب والحائض شيئا من القرآن٥٩٧إن تحت كل شعرة جنابة فاغسلوا الشعر وأنقوا البشرة٥٩٨الصلوات الخمس والجمعة إلى الجمعة وأداء الأمانة كفارة لما بينها٥٩٩من ترك موضع شعرة من جسده من جنابة لم يغسلها فعل به كذا وكذا من النار٦٠٠نعم إذا رأت الماء فلتغتسل٦٠١إذا رأت ذلك فأنزلت فعليها الغسل٦٠٢ليس عليها غسل حتى تنزل كما أنه ليس على الرجل غسل حتى ينزل٦٠٣إنما يكفيك أن تحثي عليه ثلاث حثيات من ماء ثم تفيضي عليك من الماء…٦٠٤قال بلغ عائشة أن عبد، الله بن عمرو يأمر نساءه إذا اغتسلن أن ينقضن…٦٠٥لا يغتسل أحدكم في الماء الدائم وهو جنب٦٠٦ومحمد بن بشار، قالا حدثنا غندر، محمد بن جعفر عن شعبة، عن الحكم، عن…٦٠٧قال قال رسول الله صلى الله عليه وسلم " الماء من الماء " .٦٠٨وعبد الرحمن بن إبراهيم الدمشقي، قالا حدثنا الوليد بن مسلم، حدثنا…٦٠٩حدثنا محمد بن بشار، حدثنا عثمان بن عمر، أنبأنا يونس، عن الزهري، قال…٦١٠إذا جلس الرجل بين شعبها الأربع ثم جهدها فقد وجب الغسل٦١١إذا التقى الختانان وتوارت الحشفة فقد وجب الغسل٦١٢إذا استيقظ أحدكم من نومه فرأى بللا ولم ير أنه احتلم اغتسل وإذا رأى…٦١٣قال كنت أخدم النبي صلى الله عليه وسلم فكان إذا أراد أن يغتسل قال "…٦١٤أنه قال سألت فلم أجد أحدا يخبرني أن رسول الله صلى الله عليه وسلم سبح…٦١٥لا يغتسلن أحدكم بأرض فلاة ولا فوق سطح لا يواريه فإن لم يكن يرى فإنه…٦١٦إذا أراد أحدكم الغائط وأقيمت الصلاة فليبدأ به٦١٧أن رسول الله صلى الله عليه وسلم نهى أن يصلي الرجل وهو حاقن .٦١٨لا يقوم أحدكم إلى الصلاة وبه أذى٦١٩لا يقوم أحد من المسلمين وهو حاقن حتى يتخفف٦٢٠إنما ذلك عرق فانظري إذا أتى قرؤك فلا تصلي فإذا مر القرء فتطهري ثم صلي…٦٢١لا إنما ذلك عرق وليس بالحيضة فإذا أقبلت الحيضة فدعي الصلاة وإذا أدبرت…٦٢٢وما هي أى هنتاه٦٢٣لا ولكن دعي قدر الأيام والليالي التي كنت تحيضين٦٢٤لا إنما ذلك عرق وليس بالحيضة اجتنبي الصلاة أيام محيضك ثم اغتسلي…٦٢٥المستحاضة تدع الصلاة أيام أقرائها ثم تغتسل وتتوضأ لكل صلاة وتصوم وتصلي٦٢٦إن هذه ليست بالحيضة وإنما هو عرق فإذا أقبلت الحيضة فدعي الصلاة وإذا…٦٢٧حمنة بنت جحش أنها استحيضت على عهد رسول الله صلى الله عليه وسلم فأتت…٦٢٨اغسليه بالماء والسدر وحكيه ولو بضلع٦٢٩اقرصيه واغسليه وصلي فيه٦٣٠حدثنا حرملة بن يحيى، حدثنا ابن وهب، أخبرني عمرو بن الحارث، عن عبد…٦٣١أن امرأة، سألتها أتقضي الحائض الصلاة قالت لها عائشة أحرورية أنت قد…٦٣٢ناوليني الخمرة من المسجد٦٣٣وعلي بن محمد، قالا حدثنا وكيع، عن هشام بن عروة، عن أبيه، عن عائشة،…٦٣٤حدثنا محمد بن يحيى، حدثنا عبد الرزاق، أنبأنا سفيان، عن منصور ابن…٦٣٥حدثنا عبد الله بن الجراح، حدثنا أبو الأحوص، عن عبد الكريم، ح وحدثنا…٦٣٦حدثنا أبو بكر بن أبي شيبة، حدثنا جرير، عن منصور، عن إبراهيم، عن…٦٣٧قالت كنت مع رسول الله صلى الله عليه وسلم في لحافه فوجدت ما تجد النساء…٦٣٨حدثنا الخليل بن عمرو، حدثنا ابن سلمة، عن محمد بن إسحاق، عن يزيد بن…٦٣٩من أتى حائضا أو امرأة في دبرها أو كاهنا فصدقه بما يقول فقد كفر بما…٦٤٠يتصدق بدينار أو بنصف دينار٦٤١انقضي شعرك واغتسلي٦٤٢تأخذ إحداكن ماءها وسدرها فتطهر فتحسن الطهور أو تبلغ في الطهور ثم تصب…٦٤٣حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، عن المقدام بن…٦٤٤اصنعوا كل شىء إلا الجماع٦٤٥إن المسجد لا يحل لجنب ولا لحائض٦٤٦إنما هي عرق أو عروق٦٤٧حدثنا محمد بن يحيى، حدثنا عبد الرزاق، أنبأنا معمر، عن أيوب، عن ابن…٦٤٨حدثنا نصر بن علي الجهضمي، حدثنا شجاع بن الوليد، عن علي بن عبد الأعلى،…٦٤٩قال كان رسول الله صلى الله عليه وسلم وقت للنفساء أربعين يوما إلا أن…٦٥٠قال كان الرجل إذا وقع على امرأته وهي حائض أمره النبي صلى الله عليه…٦٥١حدثنا أبو بشر، بكر بن خلف حدثنا عبد الرحمن بن مهدي، عن معاوية بن…٦٥٢حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن طلحة بن يحيى، عن عبيد الله…٦٥٣أن رسول الله صلى الله عليه وسلم صلى وعليه مرط عليه بعضه وعليها بعضه…٦٥٤وعلي بن محمد، قالا حدثنا وكيع، عن سفيان، عن عبد الكريم، عن عمرو بن…٦٥٥لا يقبل الله صلاة حائض إلا بخمار٦٥٦. أن امرأة، سألت عائشة قالت تختضب الحائض فقالت قد كنا عند النبي صلى…٦٥٧نحوه .٦٥٨قال رأيت النبي صلى الله عليه وسلم حامل الحسين بن علي على عاتقه ولعابه…٦٥٩قال رأيت النبي صلى الله عليه وسلم أتي بدلو فمضمض منه فمج فيه مسكا أو…٦٦٠وكان، قد عقل مجة مجها رسول الله صلى الله عليه وسلم في دلو من بئر لهم .٦٦١لا تنظر المرأة إلى عورة المرأة ولا ينظر الرجل إلى عورة الرجل٦٦٢حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن سفيان، عن منصور، عن موسى بن…٦٦٣وإسحاق بن منصور، قالا حدثنا يزيد بن هارون، أنبأنا مستلم بن سعيد، عن…٦٦٤لو كنت مسحت عليه بيدك أجزأك٦٦٥ارجع فأحسن
