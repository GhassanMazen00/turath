١٢٥٣إن الشمس تطلع بين قرنى الشيطان أو قال: يطلع معها قرنا الشيطان فإذا…١٢٥٤يا بني عبد مناف لا تمنعوا أحدا طاف بهذا البيت وصلى أية ساعة شاء من…١٢٥٥لعلكم ستدركون أقواما يصلون الصلاة لغير وقتها. فإن لا ينكسفان لموت أحد من الناس. فإذا رأيتموه فقوموا…١٢٦٢إن أناسا يزعمون أن الشمس والقمر لا ينكسفان إلا لموت عظيم من العظماء…١٢٦٣سمع الله لمن حمده ربنا ولك الحمد١٢٦٤ومحمد بن إسماعيل، قالا حدثنا وكيع، عن سبعا وخمسا .١٢٧٩أن رسول الله صلى الله عليه وسلم كبر في العيدين سبعا في الأولى وخمسا…١٢٨٠وعقيل، عن ابن شهاب، عن عروة، عن عائشة، أن رسول الله صلى الله عليه…١٢٨١أن رسول الله صلى الله عليه وسلم كان يقرأ صلى الله عليه وسلم…١٢٩٦قال إن من السنة أن يمشي إلى العيد .١٢٩٧أن رسول الله صلى الله عليه وسلم كان يأتي العيد ماشيا .١٢٩٨أن النبي صلى الله عليه وسلم كان إذا خرج إلى العيدين سلك على دارى سعيد…١٢٩٩أنه كان يخرج إلى العيد في طريق ويرجع في أخرى ويزعم أن رسول الله صلى…١٣٠٠أن النبي صلى الله عليه وسلم كان يأتي العيد ماشيا ويرجع ماشيا في غير…١٣٠١أن النبي صلى الله عليه وسلم كان إذا خرج إلى العيد رجع في غير صلى يوم عيد أو غيره نصبت الحربة…١٣٠٦أن رسول الله صلى الله عليه وسلم صلى العيد بالمصلى مستترا بحربة .١٣٠٧فلتلبسها أختها من جلبابها١٣٠٨أخرجوا العواتق وذوات الخدور ليشهدن العيد ودعوة المسلمين . وليجتنبن…١٣٠٩أن النبي صلى الله عليه وسلم كان يخرج بناته ونساءه في العيدين .١٣١٠من شاء أن يصلي فليصل١٣١١اجتمع عيدان في يومكم هذا فمن شاء أجزأه من الجمعة وإنا مجمعون إن شاء…١٣١٢من شاء أن يأتي الجمعة فليأتها ومن شاء أن يتخلف فليتخلف١٣١٣قال أصاب الناس مطر في يوم عيد على عهد رسول الله صلى الله عليه وسلم…١٣١٤أن النبي صلى الله عليه وسلم نهى أن يلبس السلاح في بلاد الإسلام في…١٣١٥قال كان رسول الله صلى الله عليه وسلم يغتسل يوم الفطر ويوم الأضحى .١٣١٦- وكانت له صحبة - أن رسول الله صلى الله عليه وسلم كان يغتسل يوم الفطر…١٣١٧حدثنا عبد الوهاب بن الضحاك، حدثنا إسماعيل بن عياش، حدثنا صفوان بن…١٣١٨قال كان الليل والنهار مثنى مثنى١٣٢٣أن رسول الله صلى الله عليه وسلم يوم الفتح صلى سبحة الضحى ثماني ركعات…١٣٢٤في كل ركعتين تسليمة١٣٢٥صلاة الليل مثنى مثنى. وتشهد في كل ركعتين. وتباءس وتمسكن وتقنع. وتقول:…١٣٢٦من قافية رأس أحدكم بالليل بحبل فيه ثلاث عقد فإن استيقظ…١٣٣٠ذلك الشيطان بال في أذنيه١٣٣١لا تكن مثل فلان كان يقوم الليل فترك قيام الليل١٣٣٢قالت أم سليمان بن داود لسليمان يا بنى لا تكثر النوم بالليل فإن كثرة…١٣٣٣من كثرت صلاته بالليل حسن وجهه بالنهار١٣٣٤يا أيها الناس أفشوا السلام وأطعموا الطعام وصلوا بالليل والناس نيام…١٣٣٥إذا استيقظ الرجل من الليل وأيقظ امرأته فصليا ركعتين كتبا من الذاكرين…١٣٣٦رحم الله رجلا قام من الليل فصلى وأيقظ امرأته فصلت فإن أبت رش في وجهها…١٣٣٧إن هذا القرآن نزل بحزن فإذا قرأتموه فابكوا فإن لم تبكوا فتباكوا…١٣٣٨أنه سمع عبد الرحمن بن سابط الجمحي، يحدث عن عائشة، زوج النبي
