بشر بن…٨١٥حدثنا أبو بكر بن أبي شيبة، حدثنا إسماعيل ابن علية، عن الجريري، عن قيس…٨١٦وسفيان بن عيينة، عن زياد بن علاقة، عن قطبة بن مالك، سمع النبي صلى…٨١٧حدثنا محمد بن عبد الله بن نمير، حدثنا أبي، حدثنا إسماعيل بن أبي خالد،…٨١٨حدثه أبو المنهال، عن أبي برزة، أن رسول الله صلى الله عليه وسلم كان…٨١٩حدثنا أبو بشر، بكر بن خلف حدثنا ابن أبي عدي، عن حجاج الصواف، عن يحيى…٨٢٠قال قرأ رسول الله صلى الله عليه وسلم في صلاة الصبح ب {المؤمنون} فلما…٨٢١وعبد الرحمن بن مهدي، قالا حدثنا سفيان، عن مخول، عن مسلم البطين، عن…٨٢٢قال كان رسول الله صلى الله عليه وسلم يقرأ في صلاة الفجر يوم الجمعة…٨٢٣أن رسول الله صلى الله عليه وسلم كان يقرأ في صلاة الفجر يوم الجمعة…٨٢٤حدثنا إسحاق بن منصور، أنبأنا إسحاق بن سليمان، أنبأنا عمرو بن أبي قيس،…٨٢٥حدثنا أبو بكر بن أبي شيبة، حدثنا زيد بن الحباب، حدثنا معاوية بن صالح،…٨٢٦قال قلت لخباب بأى شىء كنتم تعرفون قراءة رسول الله صلى الله عليه وسلم…٨٢٧قال ما رأيت أحدا أشبه صلاة برسول الله صلى الله عليه وسلم من فلان .…٨٢٨قال اجتمع ثلاثون بدريا من أصحاب رسول الله صلى الله عليه وسلم فقالوا…٨٢٩حدثنا بشر بن هلال الصواف، حدثنا يزيد بن زريع، حدثنا هشام الدستوائي،…٨٣٠حدثنا عقبة بن مكرم، حدثنا سلم بن قتيبة، عن هاشم بن البريد، عن أبي…٨٣١وهشام بن عمار، قالا حدثنا سفيان بن عيينة، عن الزهري، عن عبيد الله بن…٨٣٢حدثنا محمد بن الصباح، أنبأنا سفيان، عن الزهري، عن محمد بن جبير بن…٨٣٣قال كان النبي صلى الله عليه وسلم يقرأ في المغرب {قل يا أيها الكافرون}…٨٣٤حدثنا محمد بن الصباح، أنبأنا سفيان بن عيينة، ح وحدثنا عبد الله بن…٨٣٥أحسن صوتا أو قراءة منه .٨٣٦اقرأ بالشمس وضحاها وسبح اسم ربك الأعلى والليل إذا يغشى واقرأ بسم ربك٨٣٧لا صلاة لمن لم يقرأ فيها بفاتحة الكتاب٨٣٨من صلى صلاة لم يقرأ فيها بأم القرآن فهي خداج غير تمام٨٣٩لا صلاة لمن لم يقرأ في كل ركعة ب {الحمد لله } وسورة في فريضة أو غيرها٨٤٠كل صلاة لا يقرأ فيها بأم الكتاب فهي خداج٨٤١كل صلاة لا يقرأ فيها بفاتحة الكتاب فهي خداج فهي خداج٨٤٢قال سأله رجل فقال أقرأ والإمام يقرأ قال سأل رجل النبي صلى الله عليه…٨٤٣حدثنا محمد بن يحيى، حدثنا سعيد بن عامر، حدثنا شعبة، عن مسعر، عن يزيد…٨٤٤حدثنا جميل بن الحسن بن جميل العتكي، حدثنا عبد الأعلى، حدثنا سعيد، عن…٨٤٥وعلي بن الحسين بن إشكاب، قالا حدثنا إسماعيل ابن علية، عن يونس، عن…٨٤٦إنما جعل الإمام ليؤتم به فإذا كبر فكبروا وإذا قرأ فأنصتوا وإذا قال…٨٤٧إذا قرأ الإمام فأنصتوا فإذا كان عند القعدة فليكن أول ذكر أحدكم التشهد٨٤٨هل قرأ منكم من أحد٨٤٩حدثنا جميل بن الحسن، حدثنا عبد الأعلى، حدثنا معمر، عن الزهري، عن ابن…٨٥٠من كان له إمام فإن قراءة الإمام له قراءة٨٥١إذا أمن القارئ فأمنوا فإن الملائكة تؤمن فمن وافق تأمينه تأمين…٨٥٢إذا أمن القارئ فأمنوا فمن وافق تأمينه تأمين الملائكة غفر له ما تقدم…٨٥٣{غير المغضوب عليهم ولا الضالين}٨٥٤حدثنا عثمان بن أبي شيبة، حدثنا حميد بن عبد الرحمن، حدثنا ابن أبي…٨٥٥وعمار بن خالد الواسطي، قالا حدثنا أبو بكر بن عياش، عن أبي إسحاق، عن…٨٥٦ما حسدتكم اليهود على شىء ما حسدتكم على السلام والتأمين٨٥٧ما حسدتكم اليهود على شىء ما حسدتكم على آمين فأكثروا من قول آمين٨٥٨وهشام بن عمار، وأبو عمر الضرير قالوا حدثنا سفيان بن عيينة، عن الزهري،…٨٥٩أن رسول الله صلى الله عليه وسلم كان إذا كبر رفع يديه حتى يجعلهما…٨٦٠وهشام بن عمار، قالا حدثنا إسماعيل بن عياش، عن صالح بن كيسان، عن عبد…٨٦١عمير بن حبيب قال كان رسول الله صلى الله عليه وسلم يرفع يديه مع كل…٨٦٢وهو في عشرة من أصحاب رسول الله صلى الله عليه وسلم ، أحدهم أبو قتادة…٨٦٣ثم رفع حين كبر للركوع، ثم قام فرفع يديه، واستوى حتى رجع كل عظم إلى…٨٦٤وإذا رفع رأسه من الركوع فعل مثل ذلك، وإذا قام من السجدتين فعل مثل ذلك…٨٦٥أن رسول الله صلى الله عليه وسلم كان يرفع يديه عند كل تكبيرة .٨٦٦أن رسول الله صلى الله عليه وسلم كان يرفع يديه إذا دخل في الصلاة، وإذا…٨٦٧قال قلت لأنظرن إلى رسول الله صلى الله عليه وسلم كيف يصلي، فقام…٨٦٨أن جابر بن عبد الله، كان إذا افتتح الصلاة رفع يديه وإذا ركع وإذا رفع…٨٦٩قالت كان رسول الله صلى الله عليه وسلم إذا ركع لم يشخص رأسه ولم يصوبه،…٨٧٠لا تجزئ صلاة لا يقيم الرجل فيها صلبه في الركوع والسجود٨٧١يا معشر المسلمين لا صلاة لمن لا يقيم صلبه في الركوع والسجود٨٧٢يقول رأيت رسول الله صلى الله عليه وسلم يصلي فكان إذا ركع سوى ظهره حتى…٨٧٣حدثنا محمد بن عبد الله بن نمير، حدثنا محمد بن بشر، حدثنا إسماعيل بن…٨٧٤قالت كان رسول الله صلى الله عليه وسلم يركع فيضع يديه على ركبتيه…٨٧٥سمع الله لمن حمده٨٧٦إذا قال الإمام سمع الله لمن حمده . فقولوا ربنا ولك الحمد٨٧٧إذا قال الإمام سمع الله لمن حمده . فقولوا اللهم ربنا ولك الحمد٨٧٨سمع الله لمن حمده اللهم ربنا لك الحمد ملء السموات وملء الأرض وملء ما…٨٧٩اللهم ربنا لك الحمد ملء السموات وملء الأرض وملء ما شئت من شىء بعد…٨٨٠أن النبي صلى الله عليه وسلم كان إذا سجد جافى يديه فلو أن بهمة أرادت…٨٨١وصفوان بن عيسى، وأبو داود قالوا حدثنا داود بن قيس، عن عبيد الله بن…٨٨٢قال رأيت النبي صلى الله عليه وسلم إذا سجد وضع ركبتيه قبل يديه وإذا…٨٨٣أمرت أن أسجد على سبعة أعظم٨٨٤أمرت أن أسجد على سبع ولا أكف شعرا ولا ثوبا٨٨٥إذا سجد العبد سجد معه سبعة آراب: وجهه وكفاه وركبتاه وقدماه٨٨٦حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، حدثنا عباد بن راشد، عن الحسن،…٨٨٧اجعلوها في ركوعكم٨٨٨سبحان ربي العظيم٨٨٩سبحانك اللهم وبحمدك. اللهم اغفر لي٨٩٠إذا ركع أحدكم فليقل في ركوعه سبحان ربي العظيم . ثلاثا فإذا فعل ذلك…٨٩١إذا سجد أحدكم فليعتدل، ولا يفترش ذراعيه افتراش الكلب٨٩٢اعتدلوا في السجود، ولا يسجد أحدكم وهو باسط ذراعيه كالكلب٨٩٣قالت كان رسول الله صلى الله عليه وسلم إذا رفع رأسه من الركوع لم يسجد…٨٩٤لا تقع بين السجدتين٨٩٥يا علي، لا تقع إقعاء الكلب٨٩٦إذا رفعت رأسك من السجود فلا تقع كما يقعي الكلب ضع أليتيك بين قدميك،…٨٩٧رب اغفر لي، رب اغفر لي٨٩٨رب اغفر لي وارحمني واجبرني وارزقني وارفعني٨٩٩لا تقولوا السلام على الله فإن الله هو السلام فإذا جلستم فقولوا…٩٠٠التحيات المباركات الصلوات الطيبات لله السلام عليك أيها النبي ورحمة…٩٠١إذا صليتم، فكان عند القعدة، فليكن من أول قول أحدكم التحيات الطيبات…٩٠٢باسم الله وبالله، التحيات لله والصلوات والطيبات لله، السلام عليك أيها…٩٠٣قولوا اللهم صل على محمد عبدك ورسولك كما صليت على إبراهيم، وبارك على…٩٠٤قولوا اللهم صل على محمد وعلى آل محمد كما صليت على، إبراهيم إنك حميد…٩٠٥قولوا اللهم صل على محمد وأزواجه وذريته، كما صليت على إبراهيم، وبارك…٩٠٦حدثنا الحسين بن بيان، حدثنا زياد بن عبد الله، حدثنا المسعودي، عن عون…٩٠٧ما من مسلم يصلي أدركتم من قبلكم وفتم من بعدكم، تحمدون…٩٢٨اللهم أنت السلام ومنك السلام تباركت يا ذا الجلال والإكرام٩٢٩حدثنا عثمان بن أبي شيبة، حدثنا أبو الأحوص، عن سماك، عن قبيصة بن هلب،…٩٣٠حدثنا علي بن محمد، حدثنا
