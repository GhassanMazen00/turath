سنن ابن ماجه

كتاب الكفارات

٤٧ حديثًا
٢٠٩٠والذي نفس محمد بيده٢٠٩١والذي نفسي بيده٢٠٩٢لا ومصرف القلوب٢٠٩٣لا وأستغفر الله٢٠٩٤إن الله ينهاكم أن تحلفوا بآبائكم٢٠٩٥لا تحلفوا بالطواغي ولا بآبائكم٢٠٩٦من حلف فقال في يمينه باللات والعزى فليقل لا إله إلا الله٢٠٩٧قل لا إله إلا الله وحده لا شريك له ثم انفث عن يسارك ثلاثا وتعوذ ولا…٢٠٩٨من حلف بملة سوى الإسلام كاذبا متعمدا فهو كما قال٢٠٩٩حدثنا هشام بن عمار، حدثنا بقية، عن عبد الله بن محرر، عن قتادة، عن…٢١٠٠من قال إني بريء من الإسلام فإن كان كاذبا فهو كما قال وإن كان صادقا لم…٢١٠١لا تحلفوا بآبائكم من حلف بالله فليصدق ومن حلف له بالله فليرض ومن لم…٢١٠٢رأى عيسى ابن مريم رجلا يسرق فقال أسرقت قال لا والذي لا إله إلا هو .…٢١٠٣إنما الحلف حنث أو ندم٢١٠٤من حلف فقال إن شاء الله فله ثنياه٢١٠٥من حلف واستثنى إن شاء رجع وإن شاء ترك غير حانث٢١٠٦من حلف واستثنى فلم يحنث٢١٠٧والله ما أحملكم وما عندي ما أحملكم عليه٢١٠٨من حلف على يمين فرأى غيرها خيرا منها فليأت الذي هو خير وليكفر عن يمينه٢١٠٩قال قلت يا رسول الله يأتيني ابن عمي فأحلف أن لا، أعطيه ولا أصله . قال…٢١١٠من حلف في قطيعة رحم أو فيما لا يصلح فبره أن لا يتم على ذلك٢١١١من حلف على يمين فرأى غيرها خيرا منها فليتركها فإن تركها كفارتها٢١١٢قال كفر رسول الله صلى الله عليه وسلم بصاع من تمر وأمر الناس بذلك فمن…٢١١٣قال كان الرجل يقوت أهله قوتا فيه سعة وكان الرجل يقوت أهله قوتا فيه…٢١١٤إذا استلج أحدكم في اليمين فإنه آثم له عند الله من الكفارة التي أمر بها٢١١٥حدثنا علي بن محمد، حدثنا وكيع، عن علي بن صالح، عن أشعث بن أبي…٢١١٦بإسناده نحوه . قال يزيد بن أبي زياد يعني لا هجرة من دار قد أسلم أهلها…٢١١٧إذا حلف أحدكم فلا يقل ما شاء الله وشئت . ولكن ليقل ما شاء الله ثم شئت٢١١٨أما والله إن كنت لأعرفها لكم قولوا ما شاء الله ثم شاء محمد٢١١٩صدقت المسلم أخو المسلم٢١٢٠إنما اليمين على نية المستحلف٢١٢١يمينك على ما يصدقك به صاحبك٢١٢٢إنما يستخرج به من اللئيم٢١٢٣إن النذر لا يأتي ابن آدم بشىء إلا ما قدر له ولكن يغلبه القدر ما قدر…٢١٢٤لا نذر في معصية ولا نذر فيما لا يملك ابن آدم٢١٢٥لا نذر في معصية وكفارته كفارة يمين٢١٢٦من نذر أن يطيع الله فليطعه ومن نذر أن يعصي الله فلا يعصه٢١٢٧من نذر نذرا ولم يسمه فكفارته كفارة يمين٢١٢٨من نذر نذرا ولم يسمه فكفارته كفارة يمين ومن نذر نذرا لم يطقه فكفارته…٢١٢٩قال نذرت نذرا في الجاهلية فسألت النبي صلى الله عليه وسلم بعد ما أسلمت…٢١٣٠في نفسك شىء من أمر الجاهلية٢١٣١أن أباها، لقي النبي صلى الله عليه وسلم وهي رديفة له فقال إني نذرت أن…٢١٣٢أن سعد بن عبادة، استفتى رسول الله صلى الله عليه وسلم في نذر كان على…٢١٣٣ليصم عنها الولي٢١٣٤مرها فلتركب ولتختمر ولتصم ثلاثة أيام٢١٣٥حدثنا يعقوب بن حميد بن كاسب، حدثنا عبد العزيز بن محمد، عن عمرو بن أبي…٢١٣٦حدثنا محمد بن يحيى، حدثنا إسحاق بن محمد الفروي، حدثنا عبد الله بن…
