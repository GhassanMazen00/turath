سنن ابن ماجه

كتاب الجنائز

٢٠٥ حديثًا•صفحة ٢ من ٢
١٥٨٣قال: نهى رسول الله صلى الله عليه وسلم أن تتبع جنازة معها رانة .١٥٨٤ليس منا من شق الجيوب وضرب الخدود ودعا بدعوى الجاهلية١٥٨٥ومحمد بن كرامة، قالا حدثنا أبو أسامة، عن عبد الرحمن بن يزيد بن جابر،…١٥٨٦أنا بريء ممن حلق وسلق وخرق١٥٨٧دعها يا عمر فإن العين دامعة والنفس مصابة والعهد قريب١٥٨٨لله ما أخذ وله ما أعطى وكل شىء عنده إلى أجل مسمى فلتصبر ولتحتسب١٥٨٩تدمع العين ويحزن القلب ولا نقول ما يسخط الرب لولا أنه وعد صادق وموعود…١٥٩٠إن للزوج من المرأة لشعبة ما هي لشىء١٥٩١لكن حمزة لا بواكي له١٥٩٢حدثنا هشام بن عمار، حدثنا سفيان، عن إبراهيم الهجري، عن ابن أبي أوفى،…١٥٩٣الميت يعذب بما نيح عليه١٥٩٤الميت يعذب ببكاء الحى إذا قالوا واعضداه واكاسياه . واناصراه واجبلاه…١٥٩٥فإن أهلها يبكون عليها وإنها تعذب في قبرها١٥٩٦إنما الصبر عند الصدمة الأولى١٥٩٧يقول الله سبحانه: ابن آدم إن صبرت واحتسبت عند الصدمة الأولى لم أرض لك…١٥٩٨ما من مسلم يصاب بمصيبة فيفزع إلى ما أمر الله به من قوله: إنا لله وإنا…١٥٩٩يا أيها الناس أيما أحد من الناس أو من المؤمنين أصيب بمصيبة فليتعز…١٦٠٠من أصيب بمصيبة فذكر مصيبته فأحدث استرجاعا - وإن تقادم عهدها - كتب…١٦٠١ما من مؤمن يعزي أخاه بمصيبة إلا كساه الله سبحانه من حلل الكرامة يوم…١٦٠٢من عزى مصابا فله مثل أجره١٦٠٣لا يموت لرجل ثلاثة من الولد فيلج النار إلا تحلة القسم١٦٠٤ما من مسلم يموت له ثلاثة من الولد لم يبلغوا الحنث إلا تلقوه من أبواب…١٦٠٥ما من مسلمين يتوفى لهما ثلاثة من الولد لم يبلغوا الحنث إلا أدخلهم…١٦٠٦من قدم ثلاثة من الولد لم يبلغوا الحنث كانوا له حصنا حصينا من النار١٦٠٧لسقط أقدمه بين يدى أحب إلى من فارس أخلفه خلفي١٦٠٨إن السقط ليراغم ربه إذا أدخل أبويه النار . فيقال أيها السقط المراغم…١٦٠٩والذي نفسي بيده إن السقط ليجر أمه بسرره إلى الجنة إذا احتسبته١٦١٠اصنعوا لآل جعفر طعاما. فقد أتاهم ما يشغلهم أو أمر يشغلهم١٦١١إن آل جعفر قد شغلوا بشأن ميتهم فاصنعوا لهم طعاما١٦١٢حدثنا محمد بن يحيى، قال حدثنا سعيد بن منصور، حدثنا هشيم، ح وحدثنا…١٦١٣موت غربة شهادة١٦١٤يا ليته مات في غير مولده١٦١٥من مات مريضا مات شهيدا ووقي فتنة القبر وغدي وريح عليه برزقه من الجنة١٦١٦كسر عظم الميت ككسره حيا١٦١٧كسر عظم الميت ككسر عظم الحى في الإثم١٦١٨حدثنا سهل بن أبي سهل، حدثنا سفيان بن عيينة، عن الزهري، عن عبيد الله…١٦١٩أذهب الباس رب الناس . واشف أنت الشافي لا شفاء إلا شفاؤك شفاء لا يغادر…١٦٢٠ما من نبي يمرض إلا خير بين الدنيا والآخرة١٦٢١حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، عن زكريا، عن فراس،…١٦٢٢قال: قالت عائشة: ما رأيت أحدا أشد عليه الوجع من رسول الله صلى الله…١٦٢٣اللهم أعني على سكرات الموت١٦٢٤سمع أنس بن مالك، يقول آخر نظرة نظرتها إلى رسول الله صلى الله عليه…١٦٢٥الصلاة وما ملكت أيمانكم١٦٢٦قال: ذكروا عند عائشة أن عليا كان وصيا . فقالت: متى أوصى إليه؟ فلقد…١٦٢٧حدثنا علي بن محمد، حدثنا أبو معاوية، عن عبد الرحمن بن أبي بكر، عن ابن…١٦٢٨ما قبض نبي إلا دفن حيث يقبض١٦٢٩لا كرب على أبيك بعد اليوم إنه قد حضر من أبيك ما ليس بتارك منه أحدا…١٦٣٠أن فاطمة، قالت حين قبض رسول الله صلى الله عليه وسلم وا أبتاه إلى…١٦٣١حدثنا بشر بن هلال الصواف، حدثنا جعفر بن سليمان الضبعي، حدثنا ثابت، عن…١٦٣٢حدثنا محمد بن بشار، أخبرنا عبد الرحمن بن مهدي، حدثنا سفيان، عن عبد…١٦٣٣حدثنا إسحاق بن منصور، أنبأنا عبد الوهاب بن عطاء العجلي، عن ابن عون،…١٦٣٤حدثنا إبراهيم بن المنذر الحزامي، حدثنا خالي، محمد بن إبراهيم بن…١٦٣٥حدثنا الحسن بن علي الخلال، حدثنا عمرو بن عاصم، حدثنا سليمان بن…١٦٣٦إن من أفضل أيامكم يوم الجمعة فيه خلق آدم وفيه النفخة وفيه الصعقة…١٦٣٧أكثروا الصلاة على يوم الجمعة فإنه مشهود تشهده الملائكة وإن أحدا لن…
