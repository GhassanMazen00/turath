سنن ابن ماجه

كتاب الطلاق

٧٨ حديثًا
٢٠١٦وعبد الله بن عامر بن زرارة، ومسروق بن المرزبان، قالوا حدثنا يحيى بن…٢٠١٧ما بال أقوام يلعبون بحدود الله يقول أحدهم قد طلقتك قد راجعتك . قد…٢٠١٨أبغض الحلال إلى الله الطلاق٢٠١٩مره فليراجعها حتى تطهر ثم تحيض ثم تطهر ثم إن شاء طلقها قبل أن يجامعها…٢٠٢٠قال طلاق السنة أن يطلقها طاهرا من غير جماع .٢٠٢١قال طلاق السنة يطلقها عند كل طهر تطليقة فإذا طهرت الثالثة طلقها…٢٠٢٢طلق امرأته وهي حائض فقال تعرف عبد الله بن عمر طلق امرأته وهي حائض…٢٠٢٣مره فليراجعها ثم ليطلقها وهي طاهر أو حامل٢٠٢٤. قالت طلقني زوجي ثلاثا وهو خارج إلى اليمن فأجاز ذلك رسول الله صلى…٢٠٢٥أن عمران بن الحصين، سئل عن رجل، يطلق امرأته ثم يقع بها ولم يشهد على…٢٠٢٦سبق الكتاب أجله اخطبها إلى نفسها٢٠٢٧إن تفعل فقد مضى أجلها٢٠٢٨وعمرو بن عتبة، أنهما كتبا إلى سبيعة بنت الحارث يسألانها عن أمرها،…٢٠٢٩ومحمد بن بشار، قالا حدثنا عبد الله بن داود، حدثنا هشام بن عروة، عن…٢٠٣٠قال والله لمن شاء لاعناه لأنزلت سورة النساء القصرى بعد أربعة أشهر…٢٠٣١وكانت، تحت أبي سعيد الخدري أن أخته الفريعة بنت مالك، قالت خرج زوجي في…٢٠٣٢حدثنا محمد بن يحيى، حدثنا عبد العزيز بن عبد الله، حدثنا ابن أبي…٢٠٣٣قالت قالت فاطمة بنت قيس يا رسول الله إني أخاف أن يقتحم على . فأمرها…٢٠٣٤بلى فجدي نخلك فإنك عسى أن تصدقي أو تفعلي معروفا٢٠٣٥وعلي بن محمد، قالا حدثنا وكيع، حدثنا سفيان، عن أبي بكر بن أبي الجهم…٢٠٣٦لا سكنى لك ولا نفقة٢٠٣٧أن عمرة بنت الجون، تعوذت من رسول الله صلى الله عليه وسلم حين أدخلت…٢٠٣٨إذا ادعت المرأة طلاق زوجها فجاءت على ذلك بشاهد عدل استحلف زوجها فإن…٢٠٣٩ثلاث جدهن جد وهزلهن جد النكاح والطلاق والرجعة٢٠٤٠إن الله تجاوز لأمتي عما حدثت به أنفسها ما لم تعمل به أو تكلم به٢٠٤١رفع القلم عن ثلاثة عن النائم حتى يستيقظ وعن الصغير حتى يكبر وعن…٢٠٤٢يرفع القلم عن الصغير وعن المجنون وعن النائم٢٠٤٣إن الله تجاوز عن أمتي الخطأ والنسيان وما استكرهوا عليه٢٠٤٤إن الله تجاوز لأمتي عما توسوس به صدورها . ما لم تعمل به أو تتكلم به…٢٠٤٥إن الله وضع عن أمتي الخطأ والنسيان وما استكرهوا عليه٢٠٤٦لا طلاق ولا عتاق في إغلاق٢٠٤٧لا طلاق فيما لا يملك٢٠٤٨لا طلاق قبل نكاح ولا عتق قبل ملك٢٠٤٩لا طلاق قبل النكاح٢٠٥٠عذت بعظيم . الحقي بأهلك٢٠٥١وعلي بن محمد، قالا حدثنا وكيع، عن جرير بن حازم، عن الزبير بن سعيد، عن…٢٠٥٢حدثنا أبو بكر بن أبي شيبة، حدثنا أبو معاوية، عن الأعمش، عن مسلم، عن…٢٠٥٣يا عائشة إني ذاكر لك أمرا فلا عليك أن لا تعجلي فيه حتى تستأمري أبويك…٢٠٥٤لا تسأل المرأة زوجها الطلاق في غير كنهه فتجد ريح الجنة . وإن ريحها…٢٠٥٥أيما امرأة سألت زوجها الطلاق في غير ما بأس فحرام عليها رائحة الجنة٢٠٥٦أتردين عليه حديقته٢٠٥٧أتردين عليه حديقته٢٠٥٨قال قلت لها حدثيني حديثك، . قالت اختلعت من زوجي ثم جئت عثمان . فسألت…٢٠٥٩حدثنا هشام بن عمار، حدثنا عبد الرحمن بن أبي الرجال، عن أبيه، عن عمرة،…٢٠٦٠أن رسول الله صلى الله عليه وسلم إنما آلى لأن زينب ردت عليه هديته .…٢٠٦١الشهر تسع وعشرون٢٠٦٢حدثنا أبو بكر بن أبي شيبة، حدثنا عبد الله بن نمير، حدثنا محمد بن…٢٠٦٣قال قالت عائشة تبارك الذي وسع سمعه كل شىء . إني لأسمع كلام خولة بنت…٢٠٦٤حدثنا عبد الله بن سعيد، حدثنا عبد الله بن إدريس، عن محمد بن إسحاق، عن…٢٠٦٥ما حملك على ذلك٢٠٦٦انظروها فإن جاءت به أسحم أدعج العينين عظيم الأليتين فلا أراه إلا قد…٢٠٦٧البينة أو حد في ظهرك٢٠٦٨عسى أن تجيء به أسود٢٠٦٩أن رجلا، لاعن امرأته وانتفى من ولدها ففرق رسول الله صلى الله عليه…٢٠٧٠حدثنا علي بن سلمة النيسابوري، حدثنا يعقوب بن إبراهيم بن سعد، حدثنا…٢٠٧١أربع من النساء لا ملاعنة بينهن النصرانية تحت المسلم واليهودية تحت…٢٠٧٢قالت آلى رسول الله صلى الله عليه وسلم من نسائه وحرم فجعل الحرام حلالا…٢٠٧٣قال قال ابن عباس في الحرام يمين . وكان ابن عباس يقول لقد كان لكم في…٢٠٧٤أنها أعتقت بريرة فخيرها رسول الله صلى الله عليه وسلم وكان لها زوج حر .٢٠٧٥يا عباس ألا تعجب من حب مغيث بريرة ومن بغض بريرة مغيثا٢٠٧٦هو عليها صدقة وهو لنا هدية٢٠٧٧قالت أمرت بريرة أن تعتد، بثلاث حيض .٢٠٧٨أن رسول الله صلى الله عليه وسلم خير بريرة .٢٠٧٩طلاق الأمة اثنتان وعدتها حيضتان٢٠٨٠طلاق الأمة تطليقتان وقرؤها حيضتان٢٠٨١يا أيها الناس ما بال أحدكم يزوج عبده أمته ثم يريد أن يفرق بينهما إنما…٢٠٨٢امرأته تطليقتين ثم أعتقا أيتزوجها قال نعم . فقيل له عمن قال قضى بذلك…٢٠٨٣حدثنا علي بن محمد، حدثنا وكيع، عن سعيد بن أبي عروبة، عن مطر الوراق،…٢٠٨٤قد كانت إحداكن ترمي بالبعرة عند رأس الحول وإنما هي أربعة أشهر وعشرا٢٠٨٥لا يحل لامرأة أن تحد على ميت فوق ثلاث إلا على زوج٢٠٨٦لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد على ميت فوق ثلاث إلا على…٢٠٨٧لا تحد على ميت فوق ثلاث إلا المرأة تحد على زوجها أربعة أشهر وعشرا ولا…٢٠٨٨وعثمان بن عمر، قالا حدثنا ابن أبي ذئب، عن خاله الحارث بن عبد الرحمن،…٢٠٨٩الوالد أوسط أبواب الجنة فحافظ على والديك أو اترك٢٤٣٥إن الدين يقضى من صاحبه يوم القيامة إذا مات إلا من يدين في ثلاث خلال…٢٤٦٤لأن يمنح أحدكم أخاه الأرض خير له من أن يأخذ لها خراجا معلوما٢٤٧٦ورأى، أناسا يبيعون الماء فقال لا تبيعوا الماء فإني سمعت رسول الله صلى…٢٤٧٧وإبراهيم بن سعيد الجوهري، قالا حدثنا وكيع، حدثنا ابن جريج، عن أبي…
