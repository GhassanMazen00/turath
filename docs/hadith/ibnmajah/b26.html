سنن ابن ماجه

كتاب الأضاحي

٤٢ حديثًا
٣١٢٠أن رسول الله صلى الله عليه وسلم كان يضحي بكبشين أملحين أقرنين ويسمي…٣١٢١إني وجهت وجهي للذي فطر السموات والأرض حنيفا وما أنا من المشركين إن…٣١٢٢حدثنا محمد بن يحيى، حدثنا عبد الرزاق، أنبأنا سفيان الثوري، عن عبد…٣١٢٣من كان له سعة ولم يضح فلا يقربن مصلانا٣١٢٤قال سألت ابن عمر . فذكر مثله سواء .٣١٢٥يا أيها الناس إن على كل أهل بيت في كل عام أضحية وعتيرة٣١٢٦ما عمل ابن آدم يوم النحر عملا أحب إلى الله عز وجل من هراقة دم وإنه…٣١٢٧سنة أبيكم إبراهيم٣١٢٨قال ضحى رسول الله صلى الله عليه وسلم بكبش أقرن فحيل يأكل في سواد…٣١٢٩حدثنا عبد الرحمن بن إبراهيم، حدثنا محمد بن شعيب، أخبرني سعيد بن عبد…٣١٣٠خير الكفن الحلة وخير الضحايا الكبش الأقرن٣١٣١حدثنا هدية بن عبد الوهاب، أنبأنا الفضل بن موسى، أنبأنا الحسين بن…٣١٣٢حدثنا محمد بن يحيى، حدثنا عبد الرزاق، عن مالك بن أنس، عن أبي الزبير،…٣١٣٣قال ذبح رسول الله صلى الله عليه وسلم عمن اعتمر من نسائه في حجة الوداع…٣١٣٤قال قلت الإبل على عهد رسول الله صلى الله عليه وسلم فأمرهم أن ينحروا…٣١٣٥حدثنا أحمد بن عمرو بن السرح المصري أبو طاهر، أنبأنا ابن وهب، أنبأنا…٣١٣٦حدثنا محمد بن معمر، حدثنا محمد بن بكر البرساني، حدثنا ابن جريج، قال…٣١٣٧وعبد الرحيم، عن سفيان الثوري، عن سعيد بن مسروق، وحدثنا الحسين بن علي،…٣١٣٨أن رسول الله صلى الله عليه وسلم أعطاه غنما فقسمها على أصحابه ضحايا…٣١٣٩يجوز الجذع من الضأن أضحية٣١٤٠إن الجذع يوفي مما توفي منه الثنية٣١٤١لا تذبحوا إلا مسنة إلا أن يعسر عليكم فتذبحوا جذعة من الضأن٣١٤٢قال نهى رسول الله صلى الله عليه وسلم أن يضحى بمقابلة أو مدابرة أو…٣١٤٣حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، حدثنا سفيان، عن سلمة بن كهيل،…٣١٤٤أربع لا تجزئ في الأضاحي العوراء البين عورها والمريضة البين مرضها…٣١٤٥أنه ذكر أنه سمع جرى بن كليب، يحدث أنه سمع عليا، يحدث أن رسول الله صلى…٣١٤٦ومحمد بن عبد الملك أبو بكر، قالا حدثنا عبد الرزاق، عن الثوري، عن جابر…٣١٤٧حدثنا عبد الرحمن بن إبراهيم، حدثنا ابن أبي فديك، حدثني الضحاك بن…٣١٤٨ومحمد بن يوسف، ح وحدثنا محمد بن يحيى، حدثنا عبد الرزاق، جميعا عن…٣١٤٩إذا دخل العشر وأراد أحدكم أن يضحي فلا يمس من شعره ولا بشره شيئا٣١٥٠من رأى منكم هلال ذي الحجة فأراد أن يضحي فلا يقربن له شعرا ولا ظفرا٣١٥١أن رجلا، ذبح يوم النحر قبل الصلاة فأمره النبي صلى الله عليه وسلم أن…٣١٥٢من كان ذبح منكم قبل الصلاة فليعد أضحيته ومن لا فليذبح على اسم الله٣١٥٣أنه ذبح قبل الصلاة فذكره للنبي صلى الله عليه وسلم فقال " أعد أضحيتك "…٣١٥٤من هذا الذي ذبح٣١٥٥قال لقد رأيت رسول الله صلى الله عليه وسلم يذبح أضحيته بيده واضعا قدمه…٣١٥٦حدثنا هشام بن عمار، حدثنا عبد الرحمن بن سعد بن عمار بن سعد، مؤذن رسول…٣١٥٧أن مجاهدا، أخبره أن عبد الرحمن بن أبي ليلى أخبره أن علي بن أبي طالب…٣١٥٨أن رسول الله صلى الله عليه وسلم أمر من كل جزور ببضعة فجعلت في قدر…٣١٥٩حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن سفيان، عن عبد الرحمن بن…٣١٦٠كنت نهيتكم عن لحوم الأضاحي فوق ثلاثة أيام فكلوا وادخروا٣١٦١حدثنا محمد بن بشار، حدثنا أبو بكر الحنفي، حدثنا أسامة بن زيد، عن…
