كتاب الأطعمة عن رسول الله صلى الله عليه وسلم

٧٣ حديثًا
١٧٨٨حدثنا محمد بن بشار، حدثنا معاذ بن هشام، حدثني أبي، عن يونس، عن قتادة،…١٧٨٩حدثنا محمود بن غيلان، حدثنا أبو داود، أخبرنا شعبة، عن هشام بن زيد بن…١٧٩٠لا آكله ولا أحرمه١٧٩١حدثنا أحمد بن منيع، حدثنا إسماعيل بن إبراهيم، أخبرنا ابن جريج، عن عبد…١٧٩٢أويأكل الضبع أحد١٧٩٣ونصر بن علي، قالا حدثنا سفيان، عن عمرو بن دينار، عن جابر، قال أطعمنا…١٧٩٤حديث ١٧٩٤١٧٩٥حدثنا أبو كريب، حدثنا حسين بن علي الجعفي، عن زائدة، عن محمد بن عمرو،…١٧٩٦أنقوها غسلا واطبخوا فيها١٧٩٧إن لم تجدوا غيرها فارحضوها بالماء١٧٩٨ألقوها وما حولها وكلوه١٧٩٩لا يأكل أحدكم بشماله ولا يشرب بشماله فإن الشيطان يأكل بشماله ويشرب…١٨٠٠إذا أكل أحدكم فليأكل بيمينه وليشرب بيمينه فإن الشيطان يأكل بشماله…١٨٠١إذا أكل أحدكم فليلعق أصابعه فإنه لا يدري في أيتهن البركة١٨٠٢إذا أكل أحدكم طعاما فسقطت لقمة فليمط ما رابه منها ثم ليطعمها ولا…١٨٠٣إذا ما وقعت لقمة أحدكم فليمط عنها الأذى وليأكلها ولا يدعها للشيطان١٨٠٤من أكل في قصعة ثم لحسها استغفرت له القصعة١٨٠٥إن البركة تنزل وسط الطعام فكلوا من حافتيه ولا تأكلوا من وسطه١٨٠٦من أكل من هذه - قال أول مرة الثوم ثم قال الثوم والبصل والكراث فلا…١٨٠٧سمع جابر بن سمرة، يقول نزل رسول الله صلى الله عليه وسلم على أبي أيوب…١٨٠٨والد، وكيع، عن أبي إسحاق، عن شريك بن حنبل، عن علي، أنه قال نهي عن أكل…١٨٠٩حدثنا هناد، حدثنا وكيع، عن أبيه، عن أبي إسحاق، عن شريك بن حنبل، عن…١٨١٠كلوه فإني لست كأحدكم إني أخاف أن أوذي صاحبي١٨١١قال الثوم من طيبات الرزق . وأبو خلدة اسمه خالد بن دينار وهو ثقة عند…١٨١٢أغلقوا الباب وأوكئوا السقاء وأكفئوا الإناء أو خمروا الإناء وأطفئوا…١٨١٣لا تتركوا النار في بيوتكم حين تنامون١٨١٤وعبيد الله، عن الثوري، عن جبلة بن سحيم، عن ابن عمر، قال نهى رسول الله…١٨١٥بيت لا تمر فيه جياع أهله١٨١٦إن الله ليرضى عن العبد أن يأكل الأكلة أو يشرب الشربة فيحمده عليها١٨١٧كل بسم الله ثقة بالله وتوكلا عليه١٨١٨الكافر يأكل في سبعة أمعاء والمؤمن يأكل في معى واحد١٨١٩المؤمن يشرب في معى واحد والكافر يشرب في سبعة أمعاء١٨٢٠حديث ١٨٢٠١٨٢١حدثنا أحمد بن منيع، حدثنا سفيان، عن أبي يعفور العبدي، عن عبد الله بن…١٨٢٢والمؤمل، قالا حدثنا سفيان، عن أبي يعفور، عن ابن أبي أوفى، قال غزونا…١٨٢٣اللهم أهلك الجراد اقتل كباره وأهلك صغاره وأفسد بيضه واقطع دابره وخذ…١٨٢٤حدثنا هناد، حدثنا عبدة، عن محمد بن إسحاق، عن ابن أبي نجيح، عن مجاهد،…١٨٢٥حديث ١٨٢٥١٨٢٦حدثنا زيد بن أخزم الطائي، حدثنا أبو قتيبة، عن أبي العوام، عن قتادة،…١٨٢٧حدثنا هناد، حدثنا وكيع، عن سفيان، عن أيوب، عن أبي قلابة، عن زهدم، عن…١٨٢٨قال أكلت مع رسول الله صلى الله عليه وسلم لحم حبارى . قال أبو عيسى هذا…١٨٢٩أن عطاء بن يسار، أخبره أن أم سلمة أخبرته أنها، قربت إلى رسول الله صلى…١٨٣٠أما أنا فلا آكل متكئا١٨٣١ومحمود بن غيلان، وأحمد بن إبراهيم الدورقي، قالوا حدثنا أبو أسامة، عن…١٨٣٢إذا اشترى أحدكم لحما فليكثر مرقته فإن لم يجد لحما أصاب مرقة وهو أحد…١٨٣٣لا يحقرن أحدكم شيئا من المعروف وإن لم يجد فليلق أخاه بوجه طليق وإن…١٨٣٤كمل من الرجال كثير ولم يكمل من النساء إلا مريم ابنة عمران وآسية امرأة…١٨٣٥انهسوا اللحم نهسا فإنه أهنأ وأمرأ١٨٣٦حدثنا محمود بن غيلان، حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن…١٨٣٧حدثنا واصل بن عبد الأعلى، حدثنا محمد بن فضيل، عن أبي حيان التيمي، عن…١٨٣٨قالت ما كان الذراع أحب اللحم إلى رسول الله صلى الله عليه وسلم ولكن…١٨٣٩نعم الإدام الخل١٨٤٠نعم الإدام الخل١٨٤١حدثنا أبو كريب، محمد بن العلاء حدثنا أبو بكر بن عياش، عن أبي حمزة…١٨٤٢نعم الإدام الخل١٨٤٣حدثنا عبدة بن عبد الله الخزاعي، حدثنا معاوية بن هشام، عن سفيان، عن…١٨٤٤قال كان النبي صلى الله عليه وسلم يأكل القثاء بالرطب . قال أبو عيسى…١٨٤٥اشربوا من أبوالها وألبانها١٨٤٦بركة الطعام الوضوء قبله والوضوء بعده١٨٤٧إنما أمرت بالوضوء إذا قمت إلى الصلاة١٨٤٨حدثنا محمد بن بشار، حدثنا العلاء بن الفضل بن عبد الملك بن أبي سوية…١٨٤٩حدثنا قتيبة بن سعيد، حدثنا الليث، عن معاوية بن صالح، عن أبي طالوت،…١٨٥٠هذا الدباء نكثر به طعامنا١٨٥١حديث ١٨٥١١٨٥٢كلوا الزيت وادهنوا به فإنه من شجرة مباركة١٨٥٣إذا كفى أحدكم خادمه طعامه حره ودخانه فليأخذ بيده فليقعده معه فإن أبى…١٨٥٤أفشوا السلام وأطعموا الطعام واضربوا الهام تورثوا الجنان١٨٥٥اعبدوا الرحمن وأطعموا الطعام وأفشوا السلام تدخلوا الجنة بسلام١٨٥٦تعشوا ولو بكف من حشف فإن ترك العشاء مهرمة١٨٥٧ادن يا بنى وسم الله وكل بيمينك وكل مما يليك١٨٥٨حديث ١٨٥٨١٨٥٩إن الشيطان حساس لحاس فاحذروه على أنفسكم من بات وفي يده ريح غمر فأصابه…١٨٦٠من بات وفي يده ريح غمر فأصابه شيء فلا يلومن إلا نفسه
