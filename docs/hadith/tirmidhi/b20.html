كتاب النذور والأيمان عن رسول الله صلى الله عليه وسلم

٢٤ حديثًا
١٥٢٤لا نذر في معصية وكفارته كفارة يمين١٥٢٥لا نذر في معصية الله وكفارته كفارة يمين١٥٢٦حديث ١٥٢٦١٥٢٧ليس على العبد نذر فيما لا يملك١٥٢٨كفارة النذر إذا لم يسم كفارة يمين١٥٢٩يا عبد الرحمن لا تسأل الإمارة فإنك إن أتتك عن مسألة وكلت إليها وإن…١٥٣٠من حلف على يمين فرأى غيرها خيرا منها فليكفر عن يمينه وليفعل١٥٣١من حلف على يمين فقال إن شاء الله فقد استثنى فلا حنث عليه١٥٣٢من حلف على يمين فقال إن شاء الله لم يحنث١٥٣٣ألا إن الله ينهاكم أن تحلفوا بآبائكم١٥٣٤إن الله ينهاكم أن تحلفوا بآبائكم ليحلف حالف بالله أو ليسكت١٥٣٥من حلف بغير الله فقد كفر أو أشرك١٥٣٦إن الله لغني عن مشيها مروها فلتركب١٥٣٧حديث ١٥٣٧١٥٣٨لا تنذروا فإن النذر لا يغني من القدر شيئا وإنما يستخرج به من البخيل١٥٣٩حدثنا إسحاق بن منصور، أخبرنا يحيى بن سعيد القطان، عن عبيد الله بن…١٥٤٠لا ومقلب القلوب١٥٤١من أعتق رقبة مؤمنة أعتق الله منه بكل عضو منه عضوا من النار حتى يعتق…١٥٤٢حدثنا أبو كريب، حدثنا المحاربي، عن شعبة، عن حصين، عن هلال بن يساف، عن…١٥٤٣من حلف بملة غير الإسلام كاذبا فهو كما قال١٥٤٤إن الله لا يصنع بشقاء أختك شيئا فلتركب ولتختمر ولتصم ثلاثة أيام١٥٤٥من حلف منكم فقال في حلفه واللات والعزى فليقل لا إله إلا الله ومن قال…١٥٤٦أن سعد بن عبادة، استفتى رسول الله صلى الله عليه وسلم في نذر كان على…١٥٤٧حدثنا محمد بن عبد الأعلى حدثنا عمران بن عيينة هو أخو سفيان بن عيينة…
