كتاب الطب عن رسول اللَّهِ صلى الله عليه وسلم

٥٤ حديثًا
٢٠٣٦إذا أحب الله عبدا حماه الدنيا كما يظل أحدكم يحمي سقيمه الماء٢٠٣٧مه مه يا علي فإنك ناقه٢٠٣٨نعم يا عباد الله تداووا فإن الله لم يضع داء إلا وضع له شفاء أو قال…٢٠٣٩إنه ليرتو فؤاد الحزين ويسرو عن فؤاد السقيم كما تسرو إحداكن الوسخ…٢٠٤٠لا تكرهوا مرضاكم على الطعام فإن الله تبارك و تعالى يطعمهم ويسقيهم٢٠٤١عليكم بهذه الحبة السوداء فإن فيها شفاء من كل داء إلا السام٢٠٤٢اشربوا من ألبانها وأبوالها٢٠٤٣من قتل نفسه بحديدة جاء يوم القيامة وحديدته في يده يتوجأ بها في بطنه…٢٠٤٤من قتل نفسه بحديدة فحديدته في يده يتوجأ بها في بطنه في نار جهنم خالدا…٢٠٤٥حدثنا سويد بن نصر، قال أخبرنا عبد الله بن المبارك، عن يونس بن أبي…٢٠٤٦إنها ليست بدواء ولكنها داء٢٠٤٧إن خير ما تداويتم به السعوط واللدود والحجامة والمشي٢٠٤٨إن خير ما تداويتم به اللدود والسعوط والحجامة والمشي وخير ما اكتحلتم…٢٠٤٩وعقبة بن عامر، وابن، عباس . وهذا حديث حسن صحيح .٢٠٥٠حدثنا حميد بن مسعدة، قال حدثنا يزيد بن زريع، قال أخبرنا معمر، عن…٢٠٥١وجرير بن حازم، قالا حدثنا قتادة، عن أنس، قال كان رسول الله صلى الله…٢٠٥٢أسري به أنه لم يمر على ملإ من الملائكة إلا أمروه أن مر أمتك بالحجامة…٢٠٥٣نعم العبد الحجام يذهب الدم ويخف الصلب ويجلو عن البصر٢٠٥٤حدثنا أحمد بن منيع، قال حدثنا حماد بن خالد الخياط، قال حدثنا فائد،…٢٠٥٥من اكتوى أو استرقى فقد برئ من التوكل٢٠٥٦حدثنا عبدة بن عبد الله الخزاعي، قال حدثنا معاوية بن هشام، عن سفيان،…٢٠٥٧لا رقية إلا من عين أو حمة٢٠٥٨حدثنا هشام بن يونس الكوفي، قال حدثنا القاسم بن مالك المزني، عن…٢٠٥٩نعم فإنه لو كان شيء سابق القدر لسبقته العين٢٠٦٠أعيذكما بكلمات الله التامة من كل شيطان وهامة ومن كل عين لامة٢٠٦١لا شىء في الهام والعين حق٢٠٦٢لو كان شيء سابق القدر لسبقته العين وإذا استغسلتم فاغسلوا٢٠٦٣وما علمت أنها رقية اقبضوا الغنم واضربوا لي معكم بسهم٢٠٦٤وما يدريك أنها رقية٢٠٦٥حدثنا ابن أبي عمر، قال حدثنا سفيان، عن الزهري، عن أبي خزامة، عن أبيه،…٢٠٦٦العجوة من الجنة وفيها شفاء من السم والكمأة من المن وماؤها شفاء للعين٢٠٦٧الكمأة من المن وماؤها شفاء للعين٢٠٦٨الكمأة من المن وماؤها شفاء للعين والعجوة من الجنة وهي شفاء من السم٢٠٦٩قال حدثت أن أبا هريرة قال أخذت ثلاثة أكمئ أو خمسا أو سبعا فعصرتهن…٢٠٧٠قال حدثت أن أبا هريرة قال الشونيز دواء من كل داء إلا السام . قال…٢٠٧١حدثنا قتيبة، قال حدثنا الليث، عن ابن شهاب، عن أبي بكر بن عبد الرحمن،…٢٠٧٢من تعلق شيئا وكل إليه٢٠٧٣الحمى فور من النار فأبردوها بالماء٢٠٧٤إن الحمى من فيح جهنم فأبردوها بالماء٢٠٧٥بسم الله الكبير أعوذ بالله العظيم من شر كل عرق نعار ومن شر حر النار٢٠٧٦أردت أن أنهى عن الغيال فإذا فارس والروم يفعلون ولا يقتلون أولادهم٢٠٧٧لقد هممت أن أنهى عن الغيلة حتى ذكرت أن الروم وفارس يصنعون ذلك فلا يضر…٢٠٧٨أن النبي صلى الله عليه وسلم كان ينعت الزيت والورس من ذات الجنب . قال…٢٠٧٩حدثنا رجاء بن محمد العذري البصري، قال حدثنا عمرو بن محمد بن أبي رزين،…٢٠٨٠امسح بيمينك سبع مرات وقل أعوذ بعزة الله وقوته من شر ما أجد٢٠٨١حدثنا محمد بن بشار، قال حدثنا محمد بن بكر، حدثنا عبد الحميد بن جعفر،…٢٠٨٢قال جاء رجل إلى النبي صلى الله عليه وسلم فقال إن أخي استطلق بطنه .…٢٠٨٣ما من عبد مسلم يعود مريضا لم يحضر أجله فيقول سبع مرات أسأل الله…٢٠٨٤إذا أصاب أحدكم الحمى فإن الحمى قطعة من النار فليطفئها عنه بالماء…٢٠٨٥بأى شيء دووي جرح رسول الله صلى الله عليه وسلم فقال ما بقي أحد أعلم به…٢٠٨٦إنما مثل المريض إذا برأ وصح كالبردة تقع من السماء في صفائها ولونها٢٠٨٧إذا دخلتم على المريض فنفسوا له في أجله فإن ذلك لا يرد شيئا ويطيب نفسه٢٠٨٨أبشر فإن الله يقول هي ناري أسلطها على عبدي المذنب لتكون حظه من النار٢٠٨٩حديث ٢٠٨٩
