٣٧ حديثًا
١٣٨٦حدثنا علي بن سعيد الكندي الكوفي، أخبرنا ابن أبي زائدة، عن الحجاج، عن…١٣٨٧من قتل مؤمنا متعمدا دفع إلى أولياء المقتول فإن شاءوا قتلوا وإن شاءوا…١٣٨٨حدثنا محمد بن بشار، حدثنا معاذ بن هانئ، حدثنا محمد بن مسلم الطائفي،…١٣٨٩حدثنا سعيد بن عبد الرحمن المخزومي، حدثنا سفيان بن عيينة، عن عمرو بن…١٣٩٠في المواضح خمس خمس١٣٩١دية الأصابع اليدين والرجلين سواء عشر من الإبل لكل أصبع١٣٩٢ومحمد بن جعفر، قالا حدثنا شعبة، عن قتادة، عن عكرمة، عن ابن عباس، عن…١٣٩٣ما من رجل يصاب بشيء في جسده فيتصدق به إلا رفعه الله به درجة وحط عنه…١٣٩٤قال خرجت جارية عليها أوضاح فأخذها يهودي فرضخ رأسها بحجر وأخذ ما عليها…١٣٩٥حديث ١٣٩٥١٣٩٦إن أول ما يحكم بين العباد في الدماء١٣٩٧إن أول ما يقضى بين العباد في الدماء١٣٩٨لو أن أهل السماء وأهل الأرض اشتركوا في دم مؤمن لأكبهم الله في النار١٣٩٩حدثنا علي بن حجر، حدثنا إسماعيل بن عياش، حدثنا المثنى بن الصباح، عن…١٤٠٠لا يقاد الوالد بالولد١٤٠١لا تقام الحدود في المساجد ولا يقتل الوالد بالولد١٤٠٢لا يحل دم امرئ مسلم يشهد أن لا إله إلا الله وأني رسول الله إلا بإحدى…١٤٠٣ألا من قتل نفسا معاهدة له ذمة الله وذمة رسوله فقد أخفر بذمة الله فلا…١٤٠٤أن النبي صلى الله عليه وسلم ودى العامريين بدية المسلمين وكان لهما عهد…١٤٠٥ومن قتل له قتيل فهو بخير النظرين إما أن يعفو وإما أن يقتل١٤٠٦إن الله حرم مكة ولم يحرمها الناس من كان يؤمن بالله واليوم الآخر فلا…١٤٠٧أما إنه إن كان صادقا فقتلته دخلت النار١٤٠٨اغزوا بسم الله وفي سبيل الله قاتلوا من كفر بالله اغزوا ولا تغلوا ولا…١٤٠٩إن الله كتب الإحسان على كل شيء فإذا قتلتم فأحسنوا القتلة وإذا ذبحتم…١٤١٠إن هذا ليقول بقول شاعر بل فيه غرة عبد أو أمة١٤١١أن امرأتين، كانتا ضرتين فرمت إحداهما الأخرى بحجر أو عمود فسطاط فألقت…١٤١٢حدثنا أحمد بن منيع، حدثنا هشيم، أنبأنا مطرف، عن الشعبي، حدثنا أبو…١٤١٣لا يقتل مسلم بكافر١٤١٤من قتل عبده قتلناه ومن جدع عبده جدعناه١٤١٥أن ورث امرأة أشيم الضبابي من دية زوجها١٤١٦يعض أحدكم أخاه كما يعض الفحل لا دية لك١٤١٧حدثنا علي بن سعيد الكندي، حدثنا ابن المبارك، عن معمر، عن بهز بن حكيم،…١٤١٨من قتل دون ماله فهو شهيد ومن سرق من الأرض شبرا طوقه يوم القيامة من…١٤١٩من قتل دون ماله فهو شهيد١٤٢٠حديث ١٤٢٠١٤٢١من قتل دون ماله فهو شهيد ومن قتل دون دينه فهو شهيد ومن قتل دون دمه…١٤٢٢ورافع بن خديج، نحو هذا الحديث بمعناه . قال أبو عيسى هذا حديث حسن صحيح…
