٤٢٤ حديثًا•صفحة ٣ من ٣
٣٢٤٧الدعاء هو العبادة٣٢٤٨حدثنا ابن أبي عمر، حدثنا سفيان، عن منصور، عن مجاهد، عن أبي معمر، عن…٣٢٤٩نحوه .٣٢٥٠قد قال الناس ثم كفر أكثرهم فمن مات عليها فهو ممن استقام٣٢٥١حدثنا بندار، حدثنا محمد بن جعفر، حدثنا شعبة، عن عبد الملك بن ميسرة،…٣٢٥٢لا يصيب عبدا نكبة فما فوقها أو دونها إلا بذنب وما يعفو الله عنه أكثر٣٢٥٣ما ضل قوم بعد هدى كانوا عليه إلا أوتوا الجدل٣٢٥٤. إن رسول الله صلى الله عليه وسلم لما رأى قريشا استعصوا عليه قال٣٢٥٥ما من مؤمن إلا وله بابان باب يصعد منه عمله وباب ينزل منه رزقه فإذا…٣٢٥٦حدثنا علي بن سعيد الكندي، حدثنا أبو محياة، عن عبد الملك بن عمير، عن…٣٢٥٧وما أدري لعله كما قال الله تعالى : ( فلما رأوه عارضا مستقبل أوديتهم…٣٢٥٨أتاني داعي الجن فأتيتهم فقرأت عليهم٣٢٥٩إني لأستغفر الله في اليوم سبعين مرة٣٢٦٠هذا وقومه هذا وقومه٣٢٦١هذا وأصحابه والذي نفسي بيده لو كان الإيمان منوطا بالثريا لتناوله رجال…٣٢٦٢يا ابن الخطاب لقد أنزل على هذه الليلة سورة ما أحب أن لي بها ما طلعت…٣٢٦٣لقد أنزلت على آية أحب إلى مما على الأرض ثم قرأها النبي صلى الله عليه…٣٢٦٤أن ثمانين، هبطوا على رسول الله صلى الله عليه وسلم وأصحابه من جبل…٣٢٦٥حدثنا الحسن بن قزعة البصري، حدثنا سفيان بن حبيب، عن شعبة، عن ثوير، عن…٣٢٦٦أن الأقرع بن حابس، قدم على النبي صلى الله عليه وسلم فقال أبو بكر يا…٣٢٦٧في قوله : ( إن الذين ينادونك من وراء الحجرات أكثرهم لا يعقلون ) قال…٣٢٦٨نحوه . قال أبو عيسى هذا حديث حسن صحيح .٣٢٦٩قال قرأ أبو سعيد الخدري : ( واعلموا أن فيكم، رسول الله لو يطيعكم في…٣٢٧٠يا أيها الناس إن الله قد أذهب عنكم عبية الجاهلية وتعاظمها بآبائها…٣٢٧١الحسب المال والكرم التقوى٣٢٧٢لا تزال جهنم تقول هل من مزيد حتى يضع فيها رب العزة قدمه فتقول قط قط…٣٢٧٣حدثنا ابن أبي عمر، حدثنا سفيان بن عيينة، عن سلام، عن عاصم بن أبي…٣٢٧٤قال قدمت المدينة فدخلت المسجد فإذا هو غاص بالناس وإذا رايات سود تخفق…٣٢٧٥إدبار النجوم الركعتان قبل الفجر وإدبار السجود الركعتان بعد المغرب٣٢٧٦انتهى إليها ما يعرج من الأرض وما ينزل من فوق . قال فأعطاه الله عندها…٣٢٧٧أن النبي صلى الله عليه وسلم رأى جبريل وله ستمائة جناح . قال أبو عيسى…٣٢٧٨حدثنا ابن أبي عمر، حدثنا سفيان، عن مجالد، عن الشعبي، قال لقي ابن عباس…٣٢٧٩قال رأى محمد ربه . قلت أليس الله يقول : ( لا تدركه الأبصار وهو يدرك…٣٢٨٠في قول الله ( ولقد رآه نزلة * أخرى عند سدرة المنتهى ) ( فأوحى إلى…٣٢٨١وابن أبي رزمة، وأبو نعيم عن إسرائيل، عن سماك، عن عكرمة، عن ابن عباس (…٣٢٨٢ويزيد بن هارون، عن يزيد بن إبراهيم التستري، عن قتادة، عن عبد الله بن…٣٢٨٣وابن أبي رزمة، عن إسرائيل، عن أبي إسحاق، عن عبد الرحمن بن يزيد، عن…٣٢٨٤إن تغفر اللهم تغفر جما وأى عبد لك لا ألما٣٢٨٥حدثنا علي بن حجر، أخبرنا علي بن مسهر، عن الأعمش، عن إبراهيم، عن أبي…٣٢٨٦قال سأل أهل مكة النبي صلى الله عليه وسلم آية فانشق القمر بمكة مرتين…٣٢٨٧حدثنا ابن أبي عمر، حدثنا سفيان، عن ابن أبي نجيح، عن مجاهد، عن أبي…٣٢٨٨قال انفلق القمر على عهد رسول الله صلى الله عليه وسلم فقال رسول الله…٣٢٨٩حدثنا عبد بن حميد، حدثنا محمد بن كثير، حدثنا سليمان بن كثير، عن حصين،…٣٢٩٠حدثنا أبو كريب، وأبو بكر بندار قالا حدثنا وكيع، عن سفيان، عن زياد بن…٣٢٩١لقد قرأتها على الجن ليلة الجن فكانوا أحسن مردودا منكم كنت كلما أتيت…٣٢٩٢يقول الله أعددت لعبادي الصالحين ما لا عين رأت ولا أذن سمعت ولا خطر…٣٢٩٣إن في الجنة لشجرة يسير الراكب في ظلها مائة عام لا يقطعها وإن شئتم…٣٢٩٤ارتفاعها كما بين السماء والأرض ومسيرة ما بينهما خمسمائة عام٣٢٩٥: ( أتجعلون رزقكم أنكم تكذبون ) قال شكركم تقولون مطرنا بنوء كذا وكذا…٣٢٩٦إن من المنشآت اللائي كن في الدنيا عجائز عمشا رمصا٣٢٩٧شيبتني هود والواقعة والمرسلات و عم يتساءلون وإذا الشمس كورت٣٢٩٨وغير، واحد، - المعنى واحد قالوا حدثنا يونس بن محمد، حدثنا شيبان بن…٣٢٩٩والحسن بن علي الحلواني، حدثنا يزيد بن هارون، حدثنا محمد بن إسحاق، عن…٣٣٠٠حدثنا سفيان بن وكيع، حدثنا يحيى بن آدم، حدثنا عبيد الله الأشجعي، عن…٣٣٠١هل تدرون ما قال هذا٣٣٠٢رضى الله عنهما قال حرق رسول الله صلى الله عليه وسلم نخل بني النضير…٣٣٠٣حدثنا الحسن بن محمد الزعفراني، حدثنا عفان بن مسلم، حدثنا حفص بن غياث،…٣٣٠٤أن رجلا، من الأنصار بات به ضيف فلم يكن عنده إلا قوته وقوت صبيانه فقال…٣٣٠٥انطلقوا حتى تأتوا روضة خاخ فإن بها ظعينة معها كتاب فخذوه منها فائتوني…٣٣٠٦قال ما مست يد رسول الله صلى الله عليه وسلم يد امرأة إلا امرأة يملكها…٣٣٠٧حدثنا عبد بن حميد، حدثنا أبو نعيم، حدثنا يزيد بن عبد الله الشيباني،…٣٣٠٨في قوله تعالى : ( إذا جاءكم المؤمنات مهاجرات فامتحنوهن ) قال كانت…٣٣٠٩حدثنا عبد الله بن عبد الرحمن، أخبرنا محمد بن كثير، عن الأوزاعي، عن…٣٣١٠والذي نفسي بيده لو كان الإيمان بالثريا لتناوله رجال من هؤلاء٣٣١١حدثنا أحمد بن منيع، حدثنا هشيم، أخبرنا حصين، عن أبي سفيان، عن جابر،…٣٣١٢إن الله قد صدقك٣٣١٣حدثنا عبد بن حميد، حدثنا عبيد الله بن موسى، عن إسرائيل، عن السدي، عن…٣٣١٤إن الله قد صدقك٣٣١٥ما بال دعوى الجاهلية٣٣١٦وغير، واحد، هذا الحديث عن أبي جناب، عن الضحاك، عن ابن عباس، قوله ولم…٣٣١٧وسأله، رجل عن هذه الآية، ( يا أيها الذين آمنوا إن من أزواجكم وأولادكم…٣٣١٨حدثنا عبد بن حميد، أخبرنا عبد الرزاق، عن معمر، عن الزهري، عن عبيد…٣٣١٩إن أول ما خلق الله القلم فقال له اكتب فجرى بما هو كائن إلى الأبد٣٣٢٠هل تدرون ما اسم هذه٣٣٢١وهو الدشتكي أن أباه، أخبره أن أباه رحمه الله أخبره كذا، قال أخبره قال…٣٣٢٢كعكر الزيت فإذا قربه إلى وجهه سقطت فروة وجهه فيه٣٣٢٣قال قول الجن لقومهم ( لما قام عبد الله يدعوه كادوا يكونون عليه لبدا )…٣٣٢٤قال كان الجن يصعدون إلى السماء يستمعون الوحى فإذا سمعوا الكلمة زادوا…٣٣٢٥بينما أنا أمشي سمعت صوتا من السماء فرفعت رأسي فإذا الملك الذي جاءني…٣٣٢٦الصعود جبل من نار يتصعد فيه الكافر سبعين خريفا ثم يهوي به كذلك فيه…٣٣٢٧حدثنا ابن أبي عمر، حدثنا سفيان، عن مجالد، عن الشعبي، عن جابر بن عبد…٣٣٢٨حدثنا الحسن بن الصباح البزار، حدثنا زيد بن حباب، أخبرنا سهيل بن عبد…٣٣٢٩قال كان رسول الله صلى الله عليه وسلم إذا أنزل عليه القرآن يحرك به…٣٣٣٠إن أدنى أهل الجنة منزلة لمن ينظر إلى جنانه وأزواجه وخدمه وسرره مسيرة…٣٣٣١حدثنا سعيد بن يحيى بن سعيد الأموي، حدثني أبي قال، هذا ما عرضنا على…٣٣٣٢تحشرون حفاة عراة غرلا٣٣٣٣من سره أن ينظر إلى يوم القيامة كأنه رأى عين فليقرأ ( إذا الشمس كورت )…٣٣٣٤إن العبد إذا أخطأ خطيئة نكتت في قلبه نكتة سوداء فإذا هو نزع واستغفر…٣٣٣٥حدثنا يحيى بن درست، - بصري - حدثنا حماد بن زيد، عن أيوب، عن نافع، عن…٣٣٣٦يقوم أحدهم في الرشح إلى أنصاف أذنيه٣٣٣٧من نوقش الحساب هلك٣٣٣٨حدثنا محمد بن عبيد الهمداني، حدثنا علي بن أبي بكر، عن همام، عن قتادة،…٣٣٣٩اليوم الموعود يوم القيامة واليوم المشهود يوم عرفة والشاهد يوم الجمعة…٣٣٤٠إن نبيا من الأنبياء كان أعجب بأمته فقال من يقول لهؤلاء فأوحى الله…٣٣٤١أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوها عصموا مني…٣٣٤٢هي الصلاة بعضها شفع وبعضها وتر٣٣٤٣: ( إذ انبعث أشقاها ) انبعث لها رجل عارم عزيز منيع في رهطه مثل أبي…٣٣٤٤ما من نفس منفوسة إلا قد كتب مدخلها٣٣٤٥هل أنت إلا إصبع دميت وفي سبيل الله ما لقيت٣٣٤٦بينما أنا عند البيت بين النائم واليقظان إذ سمعت قائلا يقول أحد بين…٣٣٤٧حدثنا ابن أبي عمر، حدثنا سفيان، عن إسماعيل بن أمية، قال سمعت رجلا،…٣٣٤٨لو فعل لأخذته الملائكة عيانا٣٣٤٩حدثنا عبد الله بن سعيد أبو سعيد الأشج، حدثنا أبو خالد الأحمر، عن داود…٣٣٥٠حدثنا محمود بن غيلان، حدثنا أبو داود الطيالسي، حدثنا القاسم بن الفضل…٣٣٥١وعاصم، هو ابن بهدلة سمعا زر بن حبيش، وزر بن حبيش يكنى أبا مريم يقول…٣٣٥٢يقول قال رجل للنبي صلى الله عليه وسلم يا خير البرية . قال " ذلك…٣٣٥٣أتدرون ما أخبارها٣٣٥٤يقول ابن آدم مالي مالي وهل لك من مالك إلا ما تصدقت فأمضيت أو أكلت…٣٣٥٥حدثنا أبو كريب، حدثنا حكام بن سلم الرازي، عن عمرو بن أبي قيس، عن…٣٣٥٦) قال الزبير يا رسول الله فأى النعيم نسأل عنه وإنما هما الأسودان…٣٣٥٧حدثنا عبد بن حميد، حدثنا أحمد بن يونس، عن أبي بكر بن عياش، عن محمد بن…٣٣٥٨إن أول ما يسأل عنه يوم القيامة يعني العبد من النعيم أن يقال له ألم…٣٣٥٩هو نهر في الجنة٣٣٦٠بينا أنا أسير في الجنة إذ عرض لي نهر حافتاه قباب اللؤلؤ . قلت للملك…٣٣٦١الكوثر نهر في الجنة حافتاه من ذهب ومجراه على الدر والياقوت تربته أطيب…٣٣٦٢حدثنا عبد بن حميد، حدثنا سليمان بن داود، عن شعبة، عن أبي بشر، عن سعيد…٣٣٦٣وأحمد بن منيع، حدثنا أبو معاوية، حدثنا الأعمش، عن عمرو بن مرة، عن…٣٣٦٤لم يكن له شبيه ولا عدل وليس كمثله شيء٣٣٦٥حدثنا عبد بن حميد، حدثنا عبيد الله بن موسى، عن أبي جعفر الرازي، عن…٣٣٦٦يا عائشة استعيذي بالله من شر هذا فإن هذا الغاسق إذا وقب٣٣٦٧حديث ٣٣٦٧٣٣٦٨حديث ٣٣٦٨٣٣٦٩لما خلق الله الأرض جعلت تميد فخلق الجبال فعاد بها عليها فاستقرت فعجبت…٣٧٢٣أنا دار الحكمة وعلي بابها
