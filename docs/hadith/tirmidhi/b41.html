كتاب العلم عن رسول الله صلى الله عليه وسلم

٤٣ حديثًا
٢٦٤٥من يرد الله به خيرا يفقهه في الدين٢٦٤٦من سلك طريقا يلتمس فيه علما سهل الله له طريقا إلى الجنة٢٦٤٧من خرج في طلب العلم فهو في سبيل الله حتى يرجع٢٦٤٨من طلب العلم كان كفارة لما مضى٢٦٤٩من سئل عن علم علمه ثم كتمه ألجم يوم القيامة بلجام من نار٢٦٥٠إن الناس لكم تبع وإن رجالا يأتونكم من أقطار الأرضين يتفقهون في الدين…٢٦٥١يأتيكم رجال من قبل المشرق يتعلمون فإذا جاءوكم فاستوصوا بهم خيرا٢٦٥٢إن الله لا يقبض العلم انتزاعا ينتزعه من الناس ولكن يقبض العلم بقبض…٢٦٥٣هذا أوان يختلس العلم من الناس حتى لا يقدروا منه على شيء٢٦٥٤من طلب العلم ليجاري به العلماء أو ليماري به السفهاء أو يصرف به وجوه…٢٦٥٥من تعلم علما لغير الله أو أراد به غير الله فليتبوأ مقعده من النار٢٦٥٦نضر الله امرأ سمع منا حديثا فحفظه حتى يبلغه غيره فرب حامل فقه إلى من…٢٦٥٧نضر الله امرأ سمع منا شيئا فبلغه كما سمع فرب مبلغ أوعى من سامع٢٦٥٨نضر الله امرأ سمع مقالتي فوعاها وحفظها وبلغها فرب حامل فقه إلى من هو…٢٦٥٩من كذب على متعمدا فليتبوأ مقعده من النار٢٦٦٠لا تكذبوا على فإنه من كذب على يلج في النار٢٦٦١من كذب على - حسبت أنه قال متعمدا فليتبوأ بيته من النار٢٦٦٢من حدث عني حديثا وهو يرى أنه كذب فهو أحد الكاذبين٢٦٦٣لا ألفين أحدكم متكئا على أريكته يأتيه أمر مما أمرت به أو نهيت عنه…٢٦٦٤ألا هل عسى رجل يبلغه الحديث عني وهو متكئ على أريكته فيقول بيننا…٢٦٦٥حدثنا سفيان بن وكيع، حدثنا سفيان بن عيينة، عن زيد بن أسلم، عن عطاء بن…٢٦٦٦حدثنا قتيبة، حدثنا الليث، عن الخليل بن مرة، عن يحيى بن أبي صالح، عن…٢٦٦٧ومحمود بن غيلان، قالا حدثنا الوليد بن مسلم، حدثنا الأوزاعي، عن يحيى…٢٦٦٨حدثنا قتيبة، حدثنا سفيان بن عيينة، عن عمرو بن دينار، عن وهب بن منبه،…٢٦٦٩بلغوا عني ولو آية وحدثوا عن بني إسرائيل ولا حرج ومن كذب على متعمدا…٢٦٧٠إن الدال على الخير كفاعله٢٦٧١أن رجلا، أتى النبي صلى الله عليه وسلم يستحمله فقال إنه قد أبدع بي .…٢٦٧٢اشفعوا ولتؤجروا وليقض الله على لسان نبيه ما شاء٢٦٧٣ما من نفس تقتل ظلما إلا كان على ابن آدم كفل من دمها وذلك لأنه أول من…٢٦٧٤من دعا إلى هدى كان له من الأجر مثل أجور من يتبعه لا ينقص ذلك من…٢٦٧٥من سن سنة خير فاتبع عليها فله أجره ومثل أجور من اتبعه غير منقوص من…٢٦٧٦أوصيكم بتقوى الله والسمع والطاعة وإن عبد حبشي فإنه من يعش منكم يرى…٢٦٧٧أن النبي صلى الله عليه وسلم قال لبلال بن الحارث " اعلم " . قال ما…٢٦٧٨يا بنى إن قدرت أن تصبح وتمسي ليس في قلبك غش لأحد فافعل٢٦٧٩اتركوني ما تركتكم فإذا حدثتكم فخذوا عني فإنما هلك من كان قبلكم بكثرة…٢٦٨٠يوشك أن يضرب الناس، أكباد الإبل يطلبون العلم فلا يجدون أحدا أعلم من…٢٦٨١فقيه أشد على الشيطان من ألف عابد٢٦٨٢من سلك طريقا يبتغي فيه علما سلك الله به طريقا إلى الجنة وإن الملائكة…٢٦٨٣اتق الله فيما تعلم٢٦٨٤خصلتان لا تجتمعان في منافق حسن سمت ولا فقه في الدين٢٦٨٥فضل العالم على العابد كفضلي على أدناكم٢٦٨٦لن يشبع المؤمن من خير يسمعه حتى يكون منتهاه الجنة٢٦٨٧الكلمة الحكمة ضالة المؤمن فحيث وجدها فهو أحق بها
