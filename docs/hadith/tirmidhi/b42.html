كتاب الاستئذان والآداب عن رسول الله صلى الله عليه وسلم

٤٧ حديثًا
٢٦٨٨والذي نفسي بيده لا تدخلوا الجنة حتى تؤمنوا ولا تؤمنوا حتى تحابوا ألا…٢٦٨٩والحسين بن محمد الحريري، بلخي قالا حدثنا محمد بن كثير، عن جعفر بن…٢٦٩٠الاستئذان ثلاث فإن أذن لك وإلا فارجع٢٦٩١الاستئذان ثلاث فإن أذن لك وإلا فارجع٢٦٩٢وعليك ارجع فصل فإنك لم تصل٢٦٩٣إن جبريل يقرئك السلام٢٦٩٤قال قيل يا رسول الله الرجلان يلتقيان أيهما يبدأ بالسلام فقال "…٢٦٩٥ليس منا من تشبه بغيرنا لا تشبهوا باليهود ولا بالنصارى فإن تسليم…٢٦٩٦حدثنا أبو الخطاب، زياد بن يحيى البصري حدثنا أبو عتاب، سهل بن حماد…٢٦٩٧أنه سمع شهر بن حوشب، يقول سمعت أسماء بنت يزيد، تحدث أن رسول الله صلى…٢٦٩٨يا بنى إذا دخلت على أهلك فسلم يكون بركة عليك وعلى أهل بيتك٢٦٩٩السلام قبل الكلام٢٧٠٠لا تبدءوا اليهود والنصارى بالسلام وإذا لقيتم أحدهم في الطريق فاضطروهم…٢٧٠١حدثنا سعيد بن عبد الرحمن المخزومي، حدثنا سفيان بن عيينة، عن الزهري،…٢٧٠٢أن أسامة بن زيد، أخبره أن النبي صلى الله عليه وسلم مر بمجلس وفيه…٢٧٠٣يسلم الراكب على الماشي والماشي على القاعد والقليل على الكثير٢٧٠٤يسلم الصغير على الكبير والمار على القاعد والقليل على الكثير٢٧٠٥يسلم الفارس على الماشي والماشي على القائم والقليل على الكثير٢٧٠٦إذا انتهى أحدكم إلى مجلس فليسلم فإن بدا له أن يجلس فليجلس ثم إذا قام…٢٧٠٧من كشف سترا فأدخل بصره في البيت قبل أن يؤذن له فرأى عورة أهله فقد أتى…٢٧٠٨أن النبي صلى الله عليه وسلم كان في بيته فاطلع عليه رجل فأهوى إليه…٢٧٠٩لو علمت أنك تنظر لطعنت بها في عينك إنما جعل الاستئذان من أجل البصر٢٧١٠ارجع فقل السلام عليكم أأدخل٢٧١١حدثنا سويد بن نصر، أخبرنا ابن المبارك، أنبأنا شعبة، عن محمد بن…٢٧١٢أخبرنا أحمد بن منيع، حدثنا سفيان بن عيينة، عن الأسود بن قيس، عن نبيح…٢٧١٣إذا كتب أحدكم كتابا فليتربه فإنه أنجح للحاجة٢٧١٤ضع القلم على أذنك فإنه أذكر للمملي٢٧١٥إني والله ما آمن يهود على كتاب٢٧١٦أن رسول الله صلى الله عليه وسلم كتب قبل موته إلى كسرى وإلى قيصر وإلى…٢٧١٧أنه أخبره أن أبا سفيان بن حرب أخبره أن هرقل أرسل إليه في نفر من قريش…٢٧١٨قال لما أراد نبي الله صلى الله عليه وسلم أن يكتب إلى العجم قيل له إن…٢٧١٩احتلبوا هذا اللبن بيننا٢٧٢٠ونصر بن علي، قالا حدثنا أبو أحمد، عن سفيان، عن الضحاك بن عثمان، عن…٢٧٢١إن عليك السلام تحية الميت إن عليك السلام تحية الميت٢٧٢٢لا تقل عليك السلام ولكن قل السلام عليك٢٧٢٣أن رسول الله صلى الله عليه وسلم كان إذا سلم سلم ثلاثا وإذا تكلم بكلمة…٢٧٢٤ألا أخبركم عن النفر الثلاثة أما أحدهم فأوى إلى الله فآواه الله وأما…٢٧٢٥حدثنا علي بن حجر، أخبرنا شريك، عن سماك بن حرب، عن جابر بن سمرة، قال…٢٧٢٦إن كنتم لا بد فاعلين فردوا السلام وأعينوا المظلوم واهدوا السبيل٢٧٢٧ما من مسلمين يلتقيان فيتصافحان إلا غفر لهما قبل أن يفترقا٢٧٢٨حدثنا سويد، أخبرنا عبد الله، أخبرنا حنظلة بن عبيد الله، عن أنس بن…٢٧٢٩قال قلت لأنس بن مالك هل كانت المصافحة في أصحاب رسول الله صلى الله…٢٧٣٠من تمام التحية الأخذ باليد٢٧٣١تمام عيادة المريض أن يضع أحدكم يده على جبهته أو قال على يده فيسأله…٢٧٣٢حدثنا محمد بن إسماعيل، حدثنا إبراهيم بن يحيى بن محمد بن عباد المدني،…٢٧٣٣لا تشركوا بالله شيئا ولا تسرقوا ولا تزنوا ولا تقتلوا النفس التي حرم…٢٧٣٤أن أبا مرة، مولى أم هانئ بنت أبي طالب أخبره أنه، سمع أم هانئ، تقول…
