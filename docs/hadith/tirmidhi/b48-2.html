كتاب الدعوات عن رسول الله صلى الله عليه وسلم

٢٤٤ حديثًا•صفحة ٢ من ٢
٣٥٢٠اللهم لك الحمد كالذي تقول وخيرا مما نقول اللهم لك صلاتي ونسكي ومحياى…٣٥٢١ألا أدلكم على ما يجمع ذلك كله تقول اللهم إنا نسألك من خير ما سألك منه…٣٥٢٢يا مقلب القلوب ثبت قلبي على دينك٣٥٢٣إذا أويت إلى فراشك فقل اللهم رب السموات السبع وما أظلت ورب الأرضين…٣٥٢٤يا حى يا قيوم برحمتك أستغيث٣٥٢٥ألظوا بيا ذا الجلال والإكرام٣٥٢٦من أوى إلى فراشه طاهرا يذكر الله حتى يدركه النعاس لم ينقلب ساعة من…٣٥٢٧أى شيء تمام النعمة٣٥٢٨إذا فزع أحدكم في النوم فليقل أعوذ بكلمات الله التامة من غضبه وعقابه…٣٥٢٩يا أبا بكر قل اللهم فاطر السموات والأرض عالم الغيب والشهادة لا إله…٣٥٣٠لا أحد أغير من الله ولذلك حرم الفواحش ما ظهر منها وما بطن ولا أحد أحب…٣٥٣١قل اللهم إني ظلمت نفسي ظلما كثيرا ولا يغفر الذنوب إلا أنت فاغفر لي…٣٥٣٢حدثنا محمود بن غيلان حدثنا أبو أحمد حدثنا سفيان عن يزيد بن أبي زياد…٣٥٣٣إن الحمد لله وسبحان الله ولا إله إلا الله والله أكبر لتساقط من ذنوب…٣٥٣٤من قال لا إله إلا الله وحده لا شريك له له الملك وله الحمد يحيي ويميت…٣٥٣٥المرء مع من أحب يوم القيامة٣٥٣٦المرء مع من أحب٣٥٣٧إن الله يقبل توبة العبد ما لم يغرغر٣٥٣٨لله أفرح بتوبة أحدكم من أحدكم بضالته إذا وجدها٣٥٣٩لولا أنكم تذنبون لخلق الله خلقا يذنبون ويغفر لهم٣٥٤٠قال الله يا ابن آدم إنك ما دعوتني ورجوتني غفرت لك على ما كان فيك ولا…٣٥٤١خلق الله مائة رحمة فوضع رحمة واحدة بين خلقه يتراحمون بها وعند الله…٣٥٤٢لو يعلم المؤمن ما عند الله من العقوبة ما طمع في الجنة أحد ولو يعلم…٣٥٤٣إن الله حين خلق الخلق كتب بيده على نفسه إن رحمتي تغلب غضبي٣٥٤٤تدرون بم دعا الله دعا الله باسمه الأعظم الذي إذا دعي به أجاب وإذا سئل…٣٥٤٥رغم أنف رجل ذكرت عنده فلم يصل على ورغم أنف رجل دخل عليه رمضان ثم…٣٥٤٦البخيل الذي من ذكرت عنده فلم يصل على٣٥٤٧اللهم برد قلبي بالثلج والبرد والماء البارد اللهم نق قلبي من الخطايا…٣٥٤٨من فتح له منكم باب الدعاء فتحت له أبواب الرحمة وما سئل الله شيئا يعني…٣٥٤٩عليكم بقيام الليل فإنه دأب الصالحين قبلكم وإن قيام الليل قربة إلى…٣٥٥٠أعمار أمتي ما بين ستين إلى سبعين وأقلهم من يجوز ذلك٣٥٥١رب أعني ولا تعن على وانصرني ولا تنصر على وامكر لي ولا تمكر على واهدني…٣٥٥٢من دعا على من ظلمه فقد انتصر٣٥٥٣من قال عشر مرات لا إله إلا الله وحده لا شريك له له الملك وله الحمد…٣٥٥٤ألا أعلمك بأكثر مما سبحت به٣٥٥٥أن النبي صلى الله عليه وسلم مر عليها وهي في مسجد ثم مر النبي صلى الله…٣٥٥٦إن الله حيي كريم يستحي إذا رفع الرجل إليه يديه أن يردهما صفرا خائبتين٣٥٥٧أن رجلا، كان يدعو بأصبعيه فقال رسول الله صلى الله عليه وسلم " أحد أحد…٣٥٥٨سلوا الله العفو والعافية فإن أحدا لم يعط بعد اليقين خيرا من العافية٣٥٥٩ما أصر من استغفر ولو فعله في اليوم سبعين مرة٣٥٦٠من لبس ثوبا جديدا فقال الحمد لله الذي كساني ما أواري به عورتي وأتجمل…٣٥٦١ألا أدلكم على قوم أفضل غنيمة وأسرع رجعة قوم شهدوا صلاة الصبح ثم جلسوا…٣٥٦٢أى أخى أشركنا في دعائك ولا تنسنا٣٥٦٣قل اللهم اكفني بحلالك عن حرامك وأغنني بفضلك عمن سواك٣٥٦٤حدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، عن عمرو بن مرة،…٣٥٦٥اللهم أذهب البأس رب الناس واشف فأنت الشافي لا شفاء إلا شفاؤك شفاء لا…٣٥٦٦اللهم إني أعوذ برضاك من سخطك وأعوذ بمعافاتك من عقوبتك وأعوذ بك منك لا…٣٥٦٧اللهم إني أعوذ بك من الجبن وأعوذ بك من البخل وأعوذ بك من أرذل العمر…٣٥٦٨ألا أخبرك بما هو أيسر عليك من هذا أو أفضل سبحان الله عدد ما خلق في…٣٥٦٩ما من صباح يصبح العباد فيه إلا ومناد ينادي سبحوا الملك القدوس٣٥٧٠يا أبا الحسن أفلا أعلمك كلمات ينفعك الله بهن وينفع بهن من علمته ويثبت…٣٥٧١سلوا الله من فضله فإن الله عز وجل يحب أن يسأل وأفضل العبادة انتظار…٣٥٧٢اللهم إني أعوذ بك من الكسل والعجز والبخل٣٥٧٣ما على الأرض مسلم يدعو الله بدعوة إلا آتاه الله إياها أو صرف عنه من…٣٥٧٤إذا أخذت مضجعك فتوضأ وضوءك للصلاة ثم اضطجع على شقك الأيمن ثم قل اللهم…٣٥٧٥حدثنا عبد بن حميد، حدثنا محمد بن إسماعيل بن أبي فديك، حدثنا ابن أبي…٣٥٧٦اللهم بارك لهم فيما رزقتهم واغفر لهم وارحمهم٣٥٧٧من قال أستغفر الله العظيم الذي لا إله إلا هو الحى القيوم وأتوب إليه .…٣٥٧٨إن شئت دعوت وإن شئت صبرت فهو خير لك٣٥٧٩أقرب ما يكون الرب من العبد في جوف الليل الآخر فإن استطعت أن تكون ممن…٣٥٨٠إن الله عز وجل يقول إن عبدي كل عبدي الذي يذكرني وهو ملاق قرنه٣٥٨١ألا أدلك على باب من أبواب الجنة٣٥٨٢قال ما نهض ملك من الأرض حتى قال لا حول ولا قوة إلا بالله .٣٥٨٣عليكن بالتسبيح والتهليل والتقديس واعقدن بالأنامل فإنهن مسئولات…٣٥٨٤اللهم أنت عضدي وأنت نصيري وبك أقاتل٣٥٨٥خير الدعاء دعاء يوم عرفة وخير ما قلت أنا والنبيون من قبلي لا إله إلا…٣٥٨٦قل اللهم اجعل سريرتي خيرا من علانيتي واجعل علانيتي صالحة اللهم إني…٣٥٨٧يا مقلب القلوب ثبت قلبي على دينك٣٥٨٨حدثنا عبد الوارث بن عبد الصمد، حدثني أبي، حدثنا محمد بن سالم، حدثنا…٣٥٨٩قولي اللهم هذا استقبال ليلك وإدبار نهارك وأصوات دعاتك وحضور صلواتك…٣٥٩٠ما قال عبد لا إله إلا الله قط مخلصا إلا فتحت له أبواب السماء حتى تفضي…٣٥٩١اللهم إني أعوذ بك من منكرات الأخلاق والأعمال والأهواء٣٥٩٢من القائل كذا وكذا٣٥٩٣ما اصطفى الله لملائكته سبحان ربي وبحمده سبحان ربي وبحمده٣٥٩٤الدعاء لا يرد بين الأذان والإقامة٣٥٩٥الدعاء لا يرد بين الأذان والإقامة٣٥٩٦قال قال رسول الله صلى الله عليه وسلم " سبق المفردون " . قالوا وما…٣٥٩٧لأن أقول سبحان الله والحمد لله ولا إله إلا الله والله أكبر أحب إلى…٣٥٩٨ثلاثة لا ترد دعوتهم الصائم حتى يفطر والإمام العادل ودعوة المظلوم…٣٥٩٩اللهم انفعني بما علمتني وعلمني ما ينفعني وزدني علما الحمد لله على كل…٣٦٠٠إن لله ملائكة سياحين في الأرض فضلا عن كتاب الناس فإذا وجدوا أقواما…٣٦٠١أكثر من قول لا حول ولا قوة إلا بالله فإنها كنز من كنوز الجنة٣٦٠٢لكل نبي دعوة مستجابة وإني اختبأت دعوتي شفاعة لأمتي وهي نائلة إن شاء…٣٦٠٣يقول الله عز وجل أنا عند ظن عبدي بي وأنا معه حين يذكرني فإن ذكرني في…٣٦٠٤استعيذوا بالله من عذاب جهنم استعيذوا بالله من عذاب القبر استعيذوا…٣٦٠٤.٠٢من قال حين يمسي ثلاث مرات أعوذ بكلمات الله التامات من شر ما خلق لم…٣٦٠٤.٠٣اللهم اجعلني أعظم شكرك وأكثر ذكرك وأتبع نصيحتك وأحفظ وصيتك٣٦٠٤.٠٤ما من رجل يدعو الله بدعاء إلا استجيب له فإما أن يعجل له في الدنيا…٣٦٠٤.٠٥ما من عبد يرفع يديه حتى يبدو إبطه يسأل الله مسألة إلا آتاها إياه ما…٣٦٠٤.٠٦إن حسن الظن بالله من حسن عبادة الله٣٦٠٤.٠٧لينظرن أحدكم ما الذي يتمنى فإنه لا يدري ما يكتب له من أمنيته٣٦٠٤.٠٨اللهم متعني بسمعي وبصري واجعلهما الوارث مني وانصرني على من يظلمني وخذ…٣٦٠٤.٠٩ليسأل أحدكم ربه حاجته كلها حتى يسأل شسع نعله إذا انقطع٣٦٠٤.١ليسأل أحدكم ربه حاجته حتى يسأله الملح وحتى يسأله شسع نعله إذا انقطع
