كتاب الدعوات عن رسول الله صلى الله عليه وسلم

٢٤٤ حديثًا•صفحة ١ من ٢
٣٣٧٠ليس شيء أكرم على الله تعالى من الدعاء٣٣٧١الدعاء مخ العبادة٣٣٧٢الدعاء هو العبادة٣٣٧٣من لم يسأل الله يغضب عليه٣٣٧٤حدثنا محمد بن بشار حدثنا مرحوم بن عبد العزيز العطار حدثنا أبو نعامة…٣٣٧٥لا يزال لسانك رطبا من ذكر الله٣٣٧٦الذاكرون الله كثيرا والذاكرات٣٣٧٧ألا أنبئكم بخير أعمالكم وأزكاها عند مليككم وأرفعها في درجاتكم وخير…٣٣٧٨ما من قوم يذكرون الله إلا حفت بهم الملائكة وغشيتهم الرحمة ونزلت عليهم…٣٣٧٩حدثنا محمد بن بشار، حدثنا مرحوم بن عبد العزيز العطار، حدثنا أبو…٣٣٨٠ما جلس قوم مجلسا لم يذكروا الله فيه ولم يصلوا على نبيهم إلا كان عليهم…٣٣٨١ما من أحد يدعو بدعاء إلا آتاه الله ما سأل أو كف عنه من السوء مثله ما…٣٣٨٢من سره أن يستجيب الله له عند الشدائد والكرب فليكثر الدعاء في الرخاء٣٣٨٣أفضل الذكر لا إله إلا الله وأفضل الدعاء الحمد لله٣٣٨٤ومحمد بن عبيد المحاربي، قالا حدثنا يحيى بن زكريا بن أبي زائدة، عن…٣٣٨٥أن رسول الله صلى الله عليه وسلم كان إذا ذكر أحدا فدعا له بدأ بنفسه .…٣٣٨٦رضي الله عنه قال كان رسول الله صلى الله عليه وسلم إذا رفع يديه في…٣٣٨٧يستجاب لأحدكم ما لم يعجل يقول دعوت فلم يستجب لي٣٣٨٨ما من عبد يقول في صباح كل يوم ومساء كل ليلة بسم الله الذي لا يضر مع…٣٣٨٩من قال حين يمسي رضيت بالله ربا وبالإسلام دينا وبمحمد نبيا كان حقا على…٣٣٩٠أمسينا وأمسى الملك لله والحمد لله ولا إله إلا الله وحده لا شريك له…٣٣٩١إذا أصبح أحدكم فليقل اللهم بك أصبحنا وبك أمسينا وبك نحيا وبك نموت…٣٣٩٢قل اللهم عالم الغيب والشهادة فاطر السموات والأرض رب كل شيء ومليكه…٣٣٩٣ألا أدلك على سيد الاستغفار اللهم أنت ربي لا إله إلا أنت خلقتني وأنا…٣٣٩٤ألا أعلمك كلمات تقولها إذا أويت إلى فراشك فإن مت من ليلتك مت على…٣٣٩٥إذا اضطجع أحدكم على جنبه الأيمن ثم قال اللهم إني أسلمت نفسي إليك…٣٣٩٦الحمد لله الذي أطعمنا وسقانا وكفانا وآوانا وكم ممن لا كافي له ولا مؤوي٣٣٩٧من قال حين يأوي إلى فراشه أستغفر الله العظيم الذي لا إله إلا هو الحى…٣٣٩٨اللهم قني عذابك يوم تجمع عبادك أو تبعث عبادك٣٣٩٩رب قني عذابك يوم تبعث عبادك٣٤٠٠اللهم رب السموات ورب الأرضين وربنا ورب كل شيء وفالق الحب والنوى ومنزل…٣٤٠١إذا قام أحدكم عن فراشه ثم رجع إليه فلينفضه بصنفة إزاره ثلاث مرات فإنه…٣٤٠٢أن النبي صلى الله عليه وسلم كان إذا أوى إلى فراشه كل ليلة جمع كفيه ثم…٣٤٠٣اقرأ : ( قل يا أيها الكافرون ) فإنها براءة من الشرك٣٤٠٤حدثنا هشام بن يونس الكوفي، حدثنا المحاربي، عن ليث، عن أبي الزبير، عن…٣٤٠٥حدثنا صالح بن عبد الله، حدثنا حماد بن زيد، عن أبي لبابة، قال قالت…٣٤٠٦فيها آية خير من ألف آية٣٤٠٧ما من مسلم يأخذ مضجعه يقرأ سورة من كتاب الله إلا وكل الله به ملكا فلا…٣٤٠٨ألا أدلكما على ما هو خير لكما من الخادم إذا أخذتما مضجعكما تقولان…٣٤٠٩رضى الله عنه قال جاءت فاطمة إلى النبي صلى الله عليه وسلم تشكو مجلا…٣٤١٠خلتان لا يحصيهما رجل مسلم إلا دخل الجنة ألا وهما يسير ومن يعمل بهما…٣٤١١رضى الله عنهما قال رأيت رسول الله صلى الله عليه وسلم يعقد التسبيح .…٣٤١٢معقبات لا يخيب قائلهن يسبح الله في دبر كل صلاة ثلاثا وثلاثين ويحمده…٣٤١٣حدثنا يحيى بن خلف، حدثنا ابن أبي عدي، عن هشام بن حسان، عن محمد بن…٣٤١٤من تعار من الليل فقال لا إله إلا الله وحده لا شريك له له الملك وله…٣٤١٥حدثنا علي بن حجر حدثنا مسلمة بن عمرو قال كان عمير بن هانئ يصلي كل يوم…٣٤١٦سمع الله لمن حمده٣٤١٧اللهم باسمك أموت وأحيا٣٤١٨اللهم لك الحمد أنت نور السموات والأرض ولك الحمد أنت قيام السموات…٣٤١٩اللهم إني أسألك رحمة من عندك تهدي بها قلبي وتجمع بها أمري وتلم بها…٣٤٢٠اللهم رب جبريل وميكائيل وإسرافيل فاطر السموات والأرض وعالم الغيب…٣٤٢١وجهت وجهي للذي فطر السموات والأرض حنيفا وما أنا من المشركين إن صلاتي…٣٤٢٢وجهت وجهي للذي فطر السموات والأرض حنيفا وما أنا من المشركين إن صلاتي…٣٤٢٣وجهت وجهي للذي فطر السموات والأرض حنيفا وما أنا من المشركين إن صلاتي…٣٤٢٤حدثنا قتيبة، حدثنا محمد بن يزيد بن خنيس، حدثنا الحسن بن محمد بن عبيد…٣٤٢٥سجد وجهي للذي خلقه وشق سمعه وبصره بحوله وقوته٣٤٢٦من قال - يعني إذا خرج من بيته - بسم الله توكلت على الله لا حول ولا…٣٤٢٧بسم الله توكلت على الله اللهم إنا نعوذ بك من أن نزل أو نضل أو نظلم أو…٣٤٢٨من دخل السوق فقال لا إله إلا الله وحده لا شريك له له الملك وله الحمد…٣٤٢٩من قال في السوق لا إله إلا الله وحده لا شريك له له الملك وله الحمد…٣٤٣٠من قال لا إله إلا الله والله أكبر . صدقه ربه فقال لا إله إلا أنا وأنا…٣٤٣١من رأى صاحب بلاء فقال الحمد لله الذي عافاني مما ابتلاك به وفضلني على…٣٤٣٢من رأى مبتلى فقال الحمد لله الذي عافاني مما ابتلاك به وفضلني على كثير…٣٤٣٣من جلس في مجلس فكثر فيه لغطه فقال قبل أن يقوم من مجلسه ذلك سبحانك…٣٤٣٤رب اغفر لي وتب على إنك أنت التواب الغفور٣٤٣٥لا إله إلا الله العلي الحليم لا إله إلا الله رب العرش العظيم لا إله…٣٤٣٦سبحان الله العظيم٣٤٣٧من نزل منزلا ثم قال أعوذ بكلمات الله التامات من شر ما خلق لم يضره شيء…٣٤٣٨اللهم أنت الصاحب في السفر والخليفة في الأهل اللهم اصحبنا بنصحك…٣٤٣٩اللهم أنت الصاحب في السفر والخليفة في الأهل اللهم اصحبنا في سفرنا…٣٤٤٠آيبون تائبون عابدون لربنا حامدون٣٤٤١أن النبي صلى الله عليه وسلم كان إذا قدم من سفر فنظر إلى جدرات المدينة…٣٤٤٢أستودع الله دينك وأمانتك وآخر عملك٣٤٤٣أستودع الله دينك وأمانتك وخواتيم عملك٣٤٤٤زودك الله التقوى٣٤٤٥عليك بتقوى الله والتكبير على كل شرف٣٤٤٦إن ربك ليعجب من عبده إذا قال رب اغفر لي ذنوبي إنه لا يغفر الذنوب غيرك٣٤٤٧اللهم إني أسألك في سفري هذا من البر والتقوى ومن العمل ما ترضى اللهم…٣٤٤٨ثلاث دعوات مستجابات دعوة المظلوم ودعوة المسافر ودعوة الوالد على ولده٣٤٤٩اللهم إني أسألك من خيرها وخير ما فيها وخير ما أرسلت به وأعوذ بك من…٣٤٥٠اللهم لا تقتلنا بغضبك ولا تهلكنا بعذابك وعافنا قبل ذلك٣٤٥١اللهم أهلله علينا باليمن والإيمان والسلامة والإسلام ربي وربك الله٣٤٥٢إني لأعلم كلمة لو قالها لذهب غضبه أعوذ بالله من الشيطان الرجيم٣٤٥٣إذا رأى أحدكم الرؤيا يحبها فإنما هي من الله فليحمد الله عليها وليحدث…٣٤٥٤اللهم بارك لنا في ثمارنا وبارك لنا في مدينتنا وبارك لنا في صاعنا…٣٤٥٥الشربة لك فإن شئت آثرت بها خالدا٣٤٥٦الحمد لله حمدا كثيرا طيبا مباركا فيه غير مودع ولا مستغنى عنه ربنا٣٤٥٧الحمد لله الذي أطعمنا وسقانا وجعلنا مسلمين٣٤٥٨من أكل طعاما فقال الحمد لله الذي أطعمني هذا ورزقنيه من غير حول مني…٣٤٥٩إذا سمعتم صياح الديكة فاسألوا الله من فضله فإنها رأت ملكا وإذا سمعتم…٣٤٦٠ما على الأرض أحد يقول لا إله إلا الله والله أكبر ولا حول ولا قوة إلا…٣٤٦١إن ربكم ليس بأصم ولا غائب هو بينكم وبين رءوس رحالكم٣٤٦٢لقيت إبراهيم ليلة أسري بي فقال يا محمد أقرئ أمتك مني السلام وأخبرهم…٣٤٦٣أيعجز أحدكم أن يكسب ألف حسنة٣٤٦٤من قال سبحان الله العظيم وبحمده . غرست له نخلة في الجنة٣٤٦٥من قال سبحان الله العظيم وبحمده غرست له نخلة في الجنة٣٤٦٦من قال سبحان الله وبحمده مائة مرة غفرت له ذنوبه وإن كانت مثل زبد البحر٣٤٦٧كلمتان خفيفتان على اللسان ثقيلتان في الميزان حبيبتان إلى الرحمن سبحان…٣٤٦٨من قال لا إله إلا الله وحده لا شريك له له الملك وله الحمد يحيي ويميت…٣٤٦٩من قال حين يصبح وحين يمسي سبحان الله وبحمده مائة مرة لم يأت أحد يوم…٣٤٧٠قولوا سبحان الله وبحمده مائة مرة من قالها مرة كتبت له عشرا ومن قالها…٣٤٧١من سبح الله مائة بالغداة ومائة بالعشي كان كمن حج مائة مرة ومن حمد…٣٤٧٢قال تسبيحة في رمضان أفضل من ألف تسبيحة في غيره .٣٤٧٣من قال أشهد أن لا إله إلا الله وحده لا شريك له إلها واحدا أحدا صمدا…٣٤٧٤من قال في دبر صلاة الفجر وهو ثاني رجليه قبل أن يتكلم لا إله إلا الله…٣٤٧٥والذي نفسي بيده لقد سأل الله باسمه الأعظم الذي إذا دعي به أجاب وإذا…٣٤٧٦عجلت أيها المصلي إذا صليت فقعدت فاحمد الله بما هو أهله وصل على ثم ادعه٣٤٧٧أن عمرو بن مالك الجنبي، أخبره أنه، سمع فضالة بن عبيد، يقول سمع النبي…٣٤٧٨اسم الله الأعظم في هاتين الآيتين : ( وإلهكم إله واحد لا إله إلا هو…٣٤٧٩ادعوا الله وأنتم موقنون بالإجابة واعلموا أن الله لا يستجيب دعاء من…٣٤٨٠اللهم عافني في جسدي وعافني في بصري واجعله الوارث مني لا إله إلا الله…٣٤٨١قولي اللهم رب السموات السبع ورب العرش العظيم ربنا ورب كل شيء منزل…٣٤٨٢اللهم إني أعوذ بك من قلب لا يخشع ودعاء لا يسمع ومن نفس لا تشبع ومن…٣٤٨٣يا حصين كم تعبد اليوم إلها٣٤٨٤اللهم إني أعوذ بك من الهم والحزن والعجز والكسل والبخل وضلع الدين…٣٤٨٥اللهم إني أعوذ بك من الكسل والهرم والجبن والبخل وفتنة المسيح وعذاب…٣٤٨٦يا معشر النساء اعقدن بالأنامل فإنهن مسئولات مستنطقات٣٤٨٧أما كنت تدعو أما كنت تسأل ربك العافية٣٤٨٨حدثنا هارون بن عبد الله البزاز، حدثنا روح بن عبادة، عن هشام بن حسان،…٣٤٨٩اللهم إني أسألك الهدى والتقى والعفاف والغنى٣٤٩٠كان من دعاء داود يقول اللهم إني أسألك حبك وحب من يحبك والعمل الذي…٣٤٩١اللهم ارزقني حبك وحب من ينفعني حبه عندك اللهم ما رزقتني مما أحب…٣٤٩٢قل اللهم إني أعوذ بك من شر سمعي ومن شر بصري ومن شر لساني ومن شر قلبي…٣٤٩٣أعوذ برضاك من سخطك وبمعافاتك من عقوبتك لا أحصي ثناء عليك أنت كما…٣٤٩٤اللهم إني أعوذ بك من عذاب جهنم ومن عذاب القبر وأعوذ بك من فتنة المسيح…٣٤٩٥اللهم إني أعوذ بك من فتنة النار وعذاب النار وعذاب القبر وفتنة القبر…٣٤٩٦اللهم اغفر لي وارحمني وألحقني بالرفيق الأعلى٣٤٩٧لا يقول أحدكم اللهم اغفر لي إن شئت اللهم ارحمني إن شئت ليعزم المسألة…٣٤٩٨ينزل ربنا كل ليلة إلى السماء الدنيا حين يبقى ثلث الليل الآخر فيقول من…٣٤٩٩جوف الليل الآخر ودبر الصلوات المكتوبات٣٥٠٠اللهم اغفر لي ذنبي ووسع لي في رزقي وبارك لي فيما رزقتني٣٥٠١من قال حين يصبح اللهم أصبحنا نشهدك ونشهد حملة عرشك وملائكتك وجميع…٣٥٠٢اللهم اقسم لنا من خشيتك ما يحول بيننا وبين معاصيك ومن طاعتك ما تبلغنا…٣٥٠٣قال سمعني أبي، وأنا أقول اللهم، إني أعوذ بك من الهم والكسل وعذاب…٣٥٠٤ألا أعلمك كلمات إذا قلتهن غفر الله لك وإن كنت مغفورا لك٣٥٠٥دعوة ذي النون إذ دعا وهو في بطن الحوت لا إله إلا أنت سبحانك إني كنت…٣٥٠٦إن لله تسعة وتسعين اسما مائة غير واحد من أحصاها دخل الجنة٣٥٠٧إن لله تعالى تسعة وتسعين اسما مائة غير واحدة من أحصاها دخل الجنة هو…٣٥٠٨إن لله تسعة وتسعين اسما من أحصاها دخل الجنة٣٥٠٩إذا مررتم برياض الجنة فارتعوا٣٥١٠إذا مررتم برياض الجنة فارتعوا٣٥١١إذا أصاب أحدكم مصيبة فليقل : ( إنا لله وإنا إليه راجعون ) اللهم عندك…٣٥١٢سل ربك العافية والمعافاة في الدنيا والآخرة٣٥١٣قولي اللهم إنك عفو كريم تحب العفو فاعف عني٣٥١٤سل الله العافية٣٥١٥ما سئل الله شيئا أحب إليه من أن يسأل العافية٣٥١٦اللهم خر لي واختر لي٣٥١٧الوضوء شطر الإيمان والحمد لله تملأ الميزان وسبحان الله والحمد لله…٣٥١٨التسبيح نصف الميزان والحمد لله يملؤه ولا إله إلا الله ليس لها دون…٣٥١٩التسبيح نصف الميزان والحمد يملؤه والتكبير يملأ ما بين السماء والأرض…
