أَبْوَابُ الْوِتْرِ

٣٦ حديثًا
٤٥٢إن الله أمدكم بصلاة هي خير لكم من حمر النعم الوتر جعله الله لكم فيما…٤٥٣إن الله وتر يحب الوتر فأوتروا يا أهل القرآن٤٥٤وروى سفيان الثوري، وغيره، عن أبي إسحاق، عن عاصم بن ضمرة، عن علي، قال…٤٥٥حديث ٤٥٥٤٥٦حدثنا أحمد بن منيع، حدثنا أبو بكر بن عياش، حدثنا أبو حصين، عن يحيى بن…٤٥٧حديث ٤٥٧٤٥٨حديث ٤٥٨٤٥٩حدثنا إسحاق بن منصور الكوسج، حدثنا عبد الله بن نمير، حدثنا هشام بن…٤٦٠حدثنا هناد، حدثنا أبو بكر بن عياش، عن أبي إسحاق، عن الحارث، عن علي،…٤٦١حدثنا قتيبة، حدثنا حماد بن زيد، عن أنس بن سيرين، قال سألت ابن عمر…٤٦٢حدثنا علي بن حجر، أخبرنا شريك، عن أبي إسحاق، عن سعيد بن جبير، عن ابن…٤٦٣حدثنا إسحاق بن إبراهيم بن حبيب بن الشهيد البصري، حدثنا محمد بن سلمة…٤٦٤اللهم اهدني فيمن هديت وعافني فيمن عافيت وتولني فيمن توليت وبارك لي…٤٦٥من نام عن الوتر أو نسيه فليصل إذا ذكر وإذا استيقظ٤٦٦من نام عن وتره فليصل إذا أصبح٤٦٧بادروا الصبح بالوتر٤٦٨أوتروا قبل أن تصبحوا٤٦٩إذا طلع الفجر فقد ذهب كل صلاة الليل والوتر فأوتروا قبل طلوع الفجر٤٧٠لا وتران في ليلة٤٧١حدثنا محمد بن بشار، حدثنا حماد بن مسعدة، عن ميمون بن موسى المرئي، عن…٤٧٢حدثنا قتيبة، حدثنا مالك بن أنس، عن أبي بكر بن عمر بن عبد الرحمن، عن…٤٧٣من صلى الضحى ثنتى عشرة ركعة بنى الله له قصرا من ذهب في الجنة٤٧٤حدثنا أبو موسى، محمد بن المثنى حدثنا محمد بن جعفر، أخبرنا شعبة، عن…٤٧٥عن الله، عز وجل أنه قال ابن آدم اركع لي من أول النهار أربع ركعات أكفك…٤٧٦من حافظ على شفعة الضحى غفر له ذنوبه وإن كانت مثل زبد البحر٤٧٧قال كان نبي الله صلى الله عليه وسلم يصلي الضحى حتى نقول لا يدع ويدعها…٤٧٨إنها ساعة تفتح فيها أبواب السماء وأحب أن يصعد لي فيها عمل صالح٤٧٩من كانت له إلى الله حاجة أو إلى أحد من بني آدم فليتوضأ وليحسن الوضوء…٤٨٠إذا هم أحدكم بالأمر فليركع ركعتين من غير الفريضة ثم ليقل اللهم إني…٤٨١كبري الله عشرا وسبحي الله عشرا واحمديه عشرا ثم سلي ما شئت يقول نعم نعم٤٨٢يا عم ألا أصلك ألا أحبوك ألا أنفعك٤٨٣قولوا اللهم صل على محمد وعلى آل محمد كما صليت على إبراهيم إنك حميد…٤٨٤أولى الناس بي يوم القيامة أكثرهم على صلاة٤٨٥من صلى على صلاة صلى الله عليه بها عشرا٤٨٦قال إن الدعاء موقوف بين السماء والأرض لا يصعد منه شيء حتى تصلي على…٤٨٧حدثنا عباس العنبري، حدثنا عبد الرحمن بن مهدي، عن مالك بن أنس، عن…
