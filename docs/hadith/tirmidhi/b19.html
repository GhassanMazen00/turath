٣١ حديثًا
١٤٩٣ما عمل آدمي من عمل يوم النحر أحب إلى الله من إهراق الدم إنها لتأتي…١٤٩٤حدثنا قتيبة، حدثنا أبو عوانة، عن قتادة، عن أنس بن مالك، قال ضحى رسول…١٤٩٥حدثنا محمد بن عبيد المحاربي الكوفي، حدثنا شريك، عن أبي الحسناء، عن…١٤٩٦قال ضحى رسول الله صلى الله عليه وسلم بكبش أقرن فحيل يأكل في سواد…١٤٩٧حديث ١٤٩٧١٤٩٨حديث ١٤٩٨١٤٩٩نعم - أو نعمت - الأضحية الجذع من الضأن١٥٠٠حديث ١٥٠٠١٥٠١حدثنا أبو عمار الحسين بن حريث، حدثنا الفضل بن موسى، عن الحسين بن…١٥٠٢حدثنا قتيبة، حدثنا مالك بن أنس، عن أبي الزبير، عن جابر، قال نحرنا مع…١٥٠٣حدثنا علي بن حجر، أخبرنا شريك، عن سلمة بن كهيل، عن حجية بن عدي، عن…١٥٠٤قال نهى رسول الله صلى الله عليه وسلم أن يضحى بأعضب القرن والأذن . قال…١٥٠٥هذا عمن لم يضح من أمتي١٥٠٦أن رجلا، سأل ابن عمر عن الأضحية، أواجبة هي فقال ضحى رسول الله صلى…١٥٠٧وهناد، قالا حدثنا ابن أبي زائدة، عن حجاج بن أرطاة، عن نافع، عن ابن…١٥٠٨لا يذبحن أحدكم حتى يصلي١٥٠٩لا يأكل أحدكم من لحم أضحيته فوق ثلاثة أيام١٥١٠كنت نهيتكم عن لحوم الأضاحي فوق ثلاث ليتسع ذو الطول على من لا طول له…١٥١١حدثنا قتيبة، حدثنا أبو الأحوص، عن أبي إسحاق، عن عابس بن ربيعة، قال…١٥١٢لا فرع ولا عتيرة١٥١٣حدثنا أبو سلمة، يحيى بن خلف البصري حدثنا بشر بن المفضل، أخبرنا عبد…١٥١٤وعبد الرحمن بن مهدي، قالا أخبرنا سفيان، عن عاصم بن عبيد الله، عن عبيد…١٥١٥حديث ١٥١٥١٥١٦عن الغلام شاتان وعن الجارية واحدة ولا يضركم ذكرانا كن أم إناثا١٥١٧خير الأضحية الكبش وخير الكفن الحلة١٥١٨يا أيها الناس على كل أهل بيت في كل عام أضحية وعتيرة هل تدرون ما…١٥١٩يا فاطمة احلقي رأسه وتصدقي بزنة شعره فضة١٥٢٠أن النبي صلى الله عليه وسلم خطب ثم نزل فدعا بكبشين فذبحهما . قال أبو…١٥٢١بسم الله والله أكبر هذا عني وعمن لم يضح من أمتي١٥٢٢حديث ١٥٢٢١٥٢٣من رأى هلال ذي الحجة وأراد أن يضحي فلا يأخذن من شعره ولا من أظفاره
