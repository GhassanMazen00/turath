٢٩ حديثًا
١٤٦٤إذا أرسلت كلبك وذكرت اسم الله عليه فأمسك عليك فكل١٤٦٥حديث ١٤٦٥١٤٦٦كلب المجوس . قال أبو عيسى هذا حديث غريب لا نعرفه إلا من هذا الوجه .…١٤٦٧ما أمسك عليك فكل١٤٦٨إذا علمت أن سهمك قتله ولم تر فيه أثر سبع فكل١٤٦٩إذا رميت بسهمك فاذكر اسم الله فإن وجدته قد قتل فكل إلا أن تجده قد وقع…١٤٧٠إذا أرسلت كلبك المعلم وذكرت اسم الله فكل ما أمسك عليك فإن أكل فلا…١٤٧١حديث ١٤٧١١٤٧٢أن رجلا، من قومه صاد أرنبا أو اثنين فذبحهما بمروة فتعلقهما حتى لقي…١٤٧٣حدثنا أبو كريب، حدثنا عبد الرحيم بن سليمان، عن أبي أيوب الأفريقي، عن…١٤٧٤وغير، واحد، قالوا حدثنا أبو عاصم، عن وهب أبي خالد، قال حدثتني أم…١٤٧٥قال نهى رسول الله صلى الله عليه وسلم أن يتخذ شيء فيه الروح غرضا . قال…١٤٧٦ذكاة الجنين ذكاة أمه١٤٧٧حديث ١٤٧٧١٤٧٨حدثنا محمود بن غيلان، حدثنا أبو النضر، هاشم بن القاسم حدثنا عكرمة بن…١٤٧٩أن رسول الله صلى الله عليه وسلم حرم كل ذي ناب من السباع . قال أبو…١٤٨٠حديث ١٤٨٠١٤٨١لو طعنت في فخذها لأجزأ عنك١٤٨٢من قتل وزغة بالضربة الأولى كان له كذا وكذا حسنة فإن قتلها في الضربة…١٤٨٣اقتلوا الحيات واقتلوا ذا الطفيتين والأبتر فإنهما يلتمسان البصر…١٤٨٤حديث ١٤٨٤١٤٨٥إذا ظهرت الحية في المسكن فقولوا لها إنا نسألك بعهد نوح وبعهد سليمان…١٤٨٦لولا أن الكلاب أمة من الأمم لأمرت بقتلها كلها فاقتلوا منها كل أسود…١٤٨٧من اقتنى كلبا ليس بضار ولا كلب ماشية نقص من أجره كل يوم قيراطان١٤٨٨أن رسول الله صلى الله عليه وسلم أمر بقتل الكلاب إلا كلب صيد أو كلب…١٤٨٩لولا أن الكلاب أمة من الأمم لأمرت بقتلها فاقتلوا منها كل أسود بهيم…١٤٩٠من اتخذ كلبا إلا كلب ماشية أو صيد أو زرع انتقص من أجره كل يوم قيراط١٤٩١حديث ١٤٩١١٤٩٢حديث ١٤٩٢
