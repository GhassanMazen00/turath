كتاب المناقب عن رسول الله صلى الله عليه وسلم

٣٥١ حديثًا•صفحة ١ من ٣
٣٦٠٥إن الله اصطفى من ولد إبراهيم إسماعيل واصطفى من ولد إسماعيل بني كنانة…٣٦٠٦إن الله اصطفى كنانة من ولد إسماعيل واصطفى قريشا من كنانة واصطفى هاشما…٣٦٠٧إن الله خلق الخلق فجعلني من خيرهم من خير فرقهم وخير الفريقين ثم تخير…٣٦٠٨حدثنا محمود بن غيلان، حدثنا أبو أحمد، حدثنا سفيان، عن يزيد بن أبي…٣٦٠٩وآدم بين الروح والجسد٣٦١٠أنا أول الناس خروجا إذا بعثوا وأنا خطيبهم إذا وفدوا وأنا مبشرهم إذا…٣٦١١أنا أول من تنشق عنه الأرض فأكسى الحلة من حلل الجنة ثم أقوم عن يمين…٣٦١٢سلوا الله لي الوسيلة٣٦١٣مثلي في النبيين كمثل رجل بنى دارا فأحسنها وأكملها وأجملها وترك منها…٣٦١٤إذا سمعتم المؤذن فقولوا مثل ما يقول المؤذن ثم صلوا على فإنه من صلى…٣٦١٥أنا سيد ولد آدم يوم القيامة ولا فخر وبيدي لواء الحمد ولا فخر وما من…٣٦١٦قد سمعت كلامكم وعجبكم إن إبراهيم خليل الله وهو كذلك وموسى نجي الله…٣٦١٧قال مكتوب في التوراة صفة محمد وعيسى ابن مريم يدفن معه . قال فقال أبو…٣٦١٨حدثنا بشر بن هلال الصواف البصري، حدثنا جعفر بن سليمان الضبعي، عن…٣٦١٩حدثنا محمد بن بشار العبدي، حدثنا وهب بن جرير، حدثنا أبي قال، سمعت…٣٦٢٠حدثنا الفضل بن سهل أبو العباس الأعرج البغدادي، حدثنا عبد الرحمن بن…٣٦٢١قال أنزل على رسول الله صلى الله عليه وسلم وهو ابن أربعين فأقام بمكة…٣٦٢٢حدثنا محمد بن بشار، حدثنا ابن أبي عدي، عن هشام، عن عكرمة، عن ابن…٣٦٢٣أنه سمع أنس بن مالك، يقول لم يكن رسول الله صلى الله عليه وسلم بالطويل…٣٦٢٤إن بمكة حجرا كان يسلم على ليالي بعثت إني لأعرفه الآن٣٦٢٥حدثنا محمد بن بشار، حدثنا يزيد بن هارون، حدثنا سليمان التيمي، عن أبي…٣٦٢٦حدثنا عباد بن يعقوب الكوفي، حدثنا الوليد بن أبي ثور، عن السدي، عن…٣٦٢٧حدثنا محمود بن غيلان، حدثنا عمر بن يونس، عن عكرمة بن عمار، عن إسحاق…٣٦٢٨إن دعوت هذا العذق من هذه النخلة أتشهد أني رسول الله٣٦٢٩قال مسح رسول الله صلى الله عليه وسلم يده على وجهي ودعا لي قال عزرة…٣٦٣٠أرسلك أبو طلحة٣٦٣١حدثنا إسحاق بن موسى الأنصاري، حدثنا معن، حدثنا مالك بن أنس، عن إسحاق…٣٦٣٢أنها قالت أول ما ابتدئ به رسول الله صلى الله عليه وسلم من النبوة حين…٣٦٣٣حى على الوضوء المبارك والبركة من السماء٣٦٣٤أحيانا يأتيني في مثل صلصلة الجرس وهو أشد على وأحيانا يتمثل لي الملك…٣٦٣٥قال ما رأيت من ذي لمة في حلة حمراء أحسن من رسول الله صلى الله عليه…٣٦٣٦قال سأل رجل البراء أكان وجه رسول الله صلى الله عليه وسلم مثل السيف…٣٦٣٧بهذا الإسناد نحوه .٣٦٣٨حدثنا أبو جعفر، محمد بن الحسين بن أبي حليمة من قصر الأحنف وأحمد بن…٣٦٣٩حدثنا حميد بن مسعدة، حدثنا حميد بن الأسود، عن أسامة بن زيد، عن…٣٦٤٠قال كان رسول الله صلى الله عليه وسلم يعيد الكلمة ثلاثا لتعقل عنه .…٣٦٤١قال ما رأيت أحدا أكثر تبسما من رسول الله صلى الله عليه وسلم . قال أبو…٣٦٤٢قال ما كان ضحك رسول الله صلى الله عليه وسلم إلا تبسما . قال أبو عيسى…٣٦٤٣حدثنا قتيبة، حدثنا حاتم بن إسماعيل، عن الجعد بن عبد الرحمن، قال سمعت…٣٦٤٤قال كان خاتم رسول الله صلى الله عليه وسلم - يعني الذي بين كتفيه غدة…٣٦٤٥قال كان في ساقى رسول الله صلى الله عليه وسلم حموشة وكان لا يضحك إلا…٣٦٤٦قال كان النبي صلى الله عليه وسلم ضليع الفم أشكل العينين منهوس العقب .…٣٦٤٧قال كان رسول الله صلى الله عليه وسلم ضليع الفم أشكل العينين منهوش…٣٦٤٨حدثنا قتيبة، حدثنا ابن لهيعة، عن أبي يونس، عن أبي هريرة، قال ما رأيت…٣٦٤٩عرض على الأنبياء فإذا موسى ضرب من الرجال كأنه من رجال شنوءة ورأيت…٣٦٥٠ويعقوب بن إبراهيم الدورقي، قالا حدثنا إسماعيل ابن علية، عن خالد…٣٦٥١أن النبي صلى الله عليه وسلم توفي وهو ابن خمس وستين . قال أبو عيسى هذا…٣٦٥٢حدثنا أحمد بن منيع، قال حدثنا روح بن عبادة، قال حدثنا زكريا بن إسحاق،…٣٦٥٣حدثنا محمد بن بشار، قال حدثنا محمد بن جعفر، قال حدثنا شعبة، عن أبي…٣٦٥٤وقال الحسين بن مهدي، في حديثه ابن جريج عن الزهري، عن عروة، عن عائشة،…٣٦٥٥أبرأ إلى كل خليل من خله ولو كنت متخذا خليلا لاتخذت ابن أبي قحافة…٣٦٥٦حدثنا إبراهيم بن سعيد الجوهري، قال حدثنا إسماعيل بن أبي أويس، عن…٣٦٥٧قال قلت لعائشة أى أصحاب رسول الله صلى الله عليه وسلم كان أحب إلى رسول…٣٦٥٨إن أهل الدرجات العلى ليراهم من تحتهم كما ترون النجم الطالع في أفق…٣٦٥٩إن رجلا خيره ربه بين أن يعيش في الدنيا ما شاء أن يعيش ويأكل في الدنيا…٣٦٦٠إن عبدا خيره الله بين أن يؤتيه من زهرة الدنيا ما شاء وبين ما عنده…٣٦٦١ما لأحد عندنا يد إلا وقد كافيناه ما خلا أبا بكر فإن له عندنا يدا…٣٦٦٢اقتدوا باللذين من بعدي أبي بكر وعمر٣٦٦٣إني لا أدري ما بقائي فيكم فاقتدوا باللذين من بعدي٣٦٦٤هذان سيدا كهول أهل الجنة من الأولين والآخرين إلا النبيين والمرسلين٣٦٦٥هذان سيدا كهول أهل الجنة من الأولين والآخرين إلا النبيين والمرسلين يا…٣٦٦٦أبو بكر وعمر سيدا كهول أهل الجنة من الأولين والآخرين ما خلا النبيين…٣٦٦٧قال قال أبو بكر ألست أحق الناس بها ألست أول من أسلم ألست صاحب كذا…٣٦٦٨حدثنا محمود بن غيلان، قال حدثنا أبو داود، قال حدثنا الحكم بن عطية، عن…٣٦٦٩هكذا نبعث يوم القيامة٣٦٧٠أنت صاحبي على الحوض وصاحبي في الغار٣٦٧١هذان السمع والبصر٣٦٧٢مروا أبا بكر فليصل بالناس٣٦٧٣لا ينبغي لقوم فيهم أبو بكر أن يؤمهم غيره٣٦٧٤من أنفق زوجين في سبيل الله نودي في الجنة يا عبد الله هذا خير فمن كان…٣٦٧٥حدثنا هارون بن عبد الله البزاز البغدادي، قال حدثنا الفضل بن دكين، قال…٣٦٧٦فإن لم تجديني فأت أبا بكر٣٦٧٧بينا رجل راكب بقرة إذ قالت لم أخلق لهذا إنما خلقت للحرث٣٦٧٨حدثنا محمد بن حميد، قال حدثنا إبراهيم بن المختار، عن إسحاق بن راشد،…٣٦٧٩أنت عتيق الله من النار٣٦٨٠ما من نبي إلا له وزيران من أهل السماء ووزيران من أهل الأرض فأما…٣٦٨١اللهم أعز الإسلام بأحب هذين الرجلين إليك بأبي جهل أو بعمر بن الخطاب٣٦٨٢إن الله جعل الحق على لسان عمر وقلبه٣٦٨٣اللهم أعز الإسلام بأبي جهل بن هشام أو بعمر٣٦٨٤ما طلعت الشمس على رجل خير من عمر٣٦٨٥قال ما أظن رجلا ينتقص أبا بكر وعمر يحب النبي صلى الله عليه وسلم . قال…٣٦٨٦لو كان بعدي نبي لكان عمر بن الخطاب٣٦٨٧رأيت كأني أتيت بقدح من لبن فشربت منه فأعطيت فضلي عمر بن الخطاب٣٦٨٨دخلت الجنة فإذا أنا بقصر من ذهب فقلت لمن هذا القصر قالوا لشاب من قريش…٣٦٨٩يا بلال بم سبقتني إلى الجنة ما دخلت الجنة قط إلا سمعت خشخشتك أمامي…٣٦٩٠إن كنت نذرت فاضربي وإلا فلا٣٦٩١يا عائشة تعالى فانظري٣٦٩٢أنا أول من تنشق عنه الأرض ثم أبو بكر ثم عمر ثم آتي أهل البقيع فيحشرون…٣٦٩٣قد كان يكون في الأمم محدثون فإن يك في أمتي أحد فعمر بن الخطاب٣٦٩٤يطلع عليكم رجل من أهل الجنة٣٦٩٥بينما رجل يرعى غنما له إذ جاء ذئب فأخذ شاة فجاء صاحبها فانتزعها منه…٣٦٩٦اهدأ إنما عليك نبي أو صديق أو شهيد٣٦٩٧اثبت أحد فإنما عليك نبي وصديق وشهيدان٣٦٩٨لكل نبي رفيق ورفيقي - يعني في الجنة عثمان٣٦٩٩اثبت حراء فليس عليك إلا نبي أو صديق أو شهيد٣٧٠٠ما على عثمان ما عمل بعد هذه ما على عثمان ما عمل بعد هذه٣٧٠١ما ضر عثمان ما عمل بعد اليوم٣٧٠٢إن عثمان في حاجة الله وحاجة رسوله٣٧٠٣من يشتري بئر رومة فيجعل دلوه مع دلاء المسلمين بخير له منها في الجنة٣٧٠٤أن خطباء، قامت بالشام وفيهم رجال من أصحاب رسول الله صلى الله عليه…٣٧٠٥يا عثمان إنه لعل الله يقمصك قميصا فإن أرادوك على خلعه فلا تخلعه لهم٣٧٠٦لك أجر رجل شهد بدرا وسهمه٣٧٠٧حدثنا أحمد بن إبراهيم الدورقي، قال حدثنا العلاء بن عبد الجبار، قال…٣٧٠٨يقتل فيها هذا مظلوما٣٧٠٩إنه كان يبغض عثمان فأبغضه الله٣٧١٠يا أبا موسى أملك على الباب فلا يدخلن على أحد إلا بإذن٣٧١١حدثنا سفيان بن وكيع، قال حدثنا أبي ويحيى بن سعيد، عن إسماعيل بن أبي…٣٧١٢ما تريدون من علي ما تريدون من علي ما تريدون من علي إن عليا مني وأنا…٣٧١٣من كنت مولاه فعلي مولاه٣٧١٤رحم الله أبا بكر زوجني ابنته وحملني إلى دار الهجرة وأعتق بلالا من…٣٧١٥فإن لم يكن لهم فقه في الدين سنفقههم٣٧١٦أنت مني وأنا منك٣٧١٧حديث ٣٧١٧٣٧١٨إن الله أمرني بحب أربعة وأخبرني أنه يحبهم٣٧١٩علي مني وأنا من علي ولا يؤدي عني إلا أنا أو علي٣٧٢٠أنت أخي في الدنيا والآخرة٣٧٢١اللهم ائتني بأحب خلقك إليك يأكل معي هذا الطير٣٧٢٢قال قال علي كنت إذا سألت رسول الله صلى الله عليه وسلم أعطاني وإذا سكت…٣٧٢٤أما ترضى أن تكون مني بمنزلة هارون من موسى إلا أنه لا نبوة بعدي٣٧٢٥إذا كان القتال فعلي٣٧٢٦ما انتجيته ولكن الله انتجاه٣٧٢٧يا علي لا يحل لأحد أن يجنب في هذا المسجد غيري وغيرك٣٧٢٨حدثنا إسماعيل بن موسى، قال حدثنا علي بن عابس، عن مسلم الملائي، عن أنس…٣٧٢٩حدثنا خلاد بن أسلم البغدادي، حدثنا النضر بن شميل، أخبرنا عوف، عن عبد…٣٧٣٠أنت مني بمنزلة هارون من موسى إلا أنه لا نبي بعدي٣٧٣١أنت مني بمنزلة هارون من موسى إلا أنه لا نبي بعدي٣٧٣٢حدثنا محمد بن حميد الرازي، قال حدثنا إبراهيم بن المختار، عن شعبة، عن…٣٧٣٣من أحبني وأحب هذين وأباهما وأمهما كان معي في درجتي يوم القيامة٣٧٣٤حدثنا محمد بن حميد، قال حدثنا إبراهيم بن المختار، عن شعبة، عن أبي…٣٧٣٥ومحمد بن المثنى، قالا حدثنا محمد بن جعفر، قال حدثنا شعبة، عن عمرو بن…٣٧٣٦أنه لا يحبك إلا مؤمن ولا يبغضك إلا منافق٣٧٣٧اللهم لا تمتني حتى تريني عليا٣٧٣٨حدثنا أبو سعيد الأشج، قال حدثنا يونس بن بكير، عن محمد بن إسحاق، عن…٣٧٣٩من سره أن ينظر إلى شهيد يمشي على وجه الأرض فلينظر إلى طلحة بن عبيد…٣٧٤٠طلحة ممن قضى نحبه٣٧٤١طلحة والزبير جاراى في الجنة٣٧٤٢أين السائل عمن قضى نحبه٣٧٤٣قال جمع لي رسول الله صلى الله عليه وسلم أبويه يوم قريظة فقال " بأبي…٣٧٤٤إن لكل نبي حواريا وإن حواري الزبير بن العوام٣٧٤٥إن لكل نبي حواريا وإن حواري الزبير بن العوام٣٧٤٦قال أوصى الزبير إلى ابنه عبد الله صبيحة الجمل فقال ما مني عضو إلا وقد…٣٧٤٧أبو بكر في الجنة وعمر في الجنة وعثمان في الجنة وعلي في الجنة وطلحة في…٣٧٤٨عشرة في الجنة أبو بكر في الجنة وعمر في الجنة وعثمان وعلي والزبير…٣٧٤٩إن أمركن مما يهمني بعدي ولن يصبر عليكن إلا الصابرون٣٧٥٠وإسحاق بن إبراهيم بن حبيب البصري، قالا حدثنا قريش بن أنس، عن محمد بن…٣٧٥١اللهم استجب لسعد إذا دعاك٣٧٥٢هذا خالي فليرني امرؤ خاله٣٧٥٣ارم فداك أبي وأمي وقال له ارم أيها الغلام الحزور٣٧٥٤وعبد العزيز بن محمد، عن يحيى بن سعيد، عن سعيد بن المسيب، عن سعد بن…٣٧٥٥ارم سعد فداك أبي وأمي
