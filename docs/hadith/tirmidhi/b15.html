كتاب الأحكام عن رسول الله صلى الله عليه وسلم

٦٤ حديثًا
١٣٢٢حديث ١٣٢٢١٣٢٣من سأل القضاء وكل إلى نفسه ومن أجبر عليه ينزل الله عليه ملكا فيسدده١٣٢٤من ابتغى القضاء وسأل فيه شفعاء وكل إلى نفسه ومن أكره عليه أنزل الله…١٣٢٥من ولي القضاء أو جعل قاضيا بين الناس فقد ذبح بغير سكين١٣٢٦إذا حكم الحاكم فاجتهد فأصاب فله أجران وإذا حكم فأخطأ فله أجر واحد١٣٢٧أن رسول الله صلى الله عليه وسلم بعث معاذا إلى اليمن فقال " كيف تقضي "…١٣٢٨وعبد الرحمن بن مهدي، قالا حدثنا شعبة، عن أبي عون، عن الحارث بن عمرو…١٣٢٩إن أحب الناس إلى الله يوم القيامة وأدناهم منه مجلسا إمام عادل وأبغض…١٣٣٠إن الله مع القاضي ما لم يجر فإذا جار تخلى عنه ولزمه الشيطان١٣٣١إذا تقاضى إليك رجلان فلا تقض للأول حتى تسمع كلام الآخر فسوف تدري كيف…١٣٣٢ما من إمام يغلق بابه دون ذوي الحاجة والخلة والمسكنة إلا أغلق الله…١٣٣٣حدثنا علي بن حجر، حدثنا يحيى بن حمزة، عن يزيد بن أبي مريم، عن القاسم…١٣٣٤لا يحكم الحاكم بين اثنين وهو غضبان١٣٣٥أتدري لم بعثت إليك لا تصيبن شيئا بغير إذني فإنه غلول ومن يغلل يأت بما…١٣٣٦حدثنا قتيبة، حدثنا أبو عوانة، عن عمر بن أبي سلمة، عن أبيه، عن أبي…١٣٣٧حدثنا أبو موسى، محمد بن المثنى حدثنا أبو عامر العقدي، حدثنا ابن أبي…١٣٣٨لو أهدي إلى كراع لقبلت ولو دعيت عليه لأجبت١٣٣٩إنكم تختصمون إلى وإنما أنا بشر ولعل بعضكم أن يكون ألحن بحجته من بعض…١٣٤٠حدثنا قتيبة، حدثنا أبو الأحوص، عن سماك بن حرب، عن علقمة بن وائل بن…١٣٤١البينة على المدعي واليمين على المدعى عليه١٣٤٢أن رسول الله صلى الله عليه وسلم قضى أن اليمين على المدعى عليه . قال…١٣٤٣حدثنا يعقوب بن إبراهيم الدورقي، حدثنا عبد العزيز بن محمد، قال حدثني…١٣٤٤ومحمد بن أبان، قالا حدثنا عبد الوهاب الثقفي، عن جعفر بن محمد، عن…١٣٤٥حدثنا علي بن حجر، أخبرنا إسماعيل بن جعفر، حدثنا جعفر بن محمد، عن…١٣٤٦من أعتق نصيبا - أو قال شقصا أو قال شركا له في عبد فكان له من المال ما…١٣٤٧من أعتق نصيبا له في عبد فكان له من المال ما يبلغ ثمنه فهو عتيق من ماله١٣٤٨من أعتق نصيبا - أو قال شقصا في مملوك فخلاصه في ماله إن كان له مال فإن…١٣٤٩العمرى جائزة لأهلها أو ميراث لأهلها١٣٥٠أيما رجل أعمر عمرى له ولعقبه فإنها للذي يعطاها لا ترجع إلى الذي…١٣٥١العمرى جائزة لأهلها والرقبى جائزة لأهلها١٣٥٢الصلح جائز بين المسلمين إلا صلحا حرم حلالا أو أحل حراما والمسلمون على…١٣٥٣إذا استأذن أحدكم جاره أن يغرز خشبة في جداره فلا يمنعه١٣٥٤اليمين على ما يصدقك به صاحبك١٣٥٥اجعلوا الطريق سبعة أذرع١٣٥٦إذا تشاجرتم في الطريق فاجعلوه سبعة أذرع١٣٥٧حدثنا نصر بن علي، حدثنا سفيان، عن زياد بن سعد، عن هلال بن أبي ميمونة…١٣٥٨إن أطيب ما أكلتم من كسبكم وإن أولادكم من كسبكم١٣٥٩طعام بطعام وإناء بإناء١٣٦٠أن النبي صلى الله عليه وسلم استعار قصعة فضاعت فضمنها لهم . قال أبو…١٣٦١حدثنا محمد بن وزير الواسطي، حدثنا إسحاق بن يوسف الأزرق، عن سفيان، عن…١٣٦٢حدثنا أبو سعيد الأشج، حدثنا حفص بن غياث، عن أشعث، عن عدي بن ثابت، عن…١٣٦٣اسق يا زبير ثم أرسل الماء إلى جارك١٣٦٤أن رجلا، من الأنصار أعتق ستة أعبد له عند موته ولم يكن له مال غيرهم…١٣٦٥حديث ١٣٦٥١٣٦٦من زرع في أرض قوم بغير إذنهم فليس له من الزرع شيء وله نفقته١٣٦٧أكل ولدك نحلته مثل ما نحلت هذا١٣٦٨جار الدار أحق بالدار١٣٦٩الجار أحق بشفعته ينتظر به وإن كان غائبا إذا كان طريقهما واحدا١٣٧٠إذا وقعت الحدود وصرفت الطرق فلا شفعة١٣٧١الشريك شفيع والشفعة في كل شيء١٣٧٢عرفها سنة ثم اعرف وكاءها ووعاءها وعفاصها ثم استنفق بها فإن جاء ربها…١٣٧٣عرفها سنة فإن اعترفت فأدها وإلا فاعرف وعاءها وعفاصها ووكاءها وعددها…١٣٧٤ويزيد بن هارون، عن سفيان الثوري، عن سلمة بن كهيل، عن سويد بن غفلة،…١٣٧٥إن شئت حبست أصلها وتصدقت بها١٣٧٦إذا مات الإنسان انقطع عمله إلا من ثلاث صدقة جارية وعلم ينتفع به وولد…١٣٧٧العجماء جرحها جبار والبئر جبار والمعدن جبار وفي الركاز الخمس١٣٧٨من أحيى أرضا ميتة فهي له وليس لعرق ظالم حق١٣٧٩من أحيا أرضا ميتة فهي له١٣٨٠ما لم تنله خفاف الإبل١٣٨١وزاد، فيه وبعث معه معاوية ليقطعها إياه . قال أبو عيسى هذا حديث حسن…١٣٨٢ما من مسلم يغرس غرسا أو يزرع زرعا فيأكل منه إنسان أو طير أو بهيمة إلا…١٣٨٣حدثنا إسحاق بن منصور، أخبرنا يحيى بن سعيد، عن عبيد الله، عن نافع، عن…١٣٨٤إذا كانت لأحدكم أرض فليمنحها أخاه أو ليزرعها١٣٨٥حدثنا محمود بن غيلان، أخبرنا الفضل بن موسى الشيباني، أخبرنا شريك، عن…
