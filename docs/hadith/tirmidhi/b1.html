١٤٨ حديثًا
١لا تقبل صلاة بغير طهور ولا صدقة من غلول٢إذا توضأ العبد المسلم أو المؤمن فغسل وجهه خرجت من وجهه كل خطيئة نظر…٣مفتاح الصلاة الطهور وتحريمها التكبير وتحليلها التسليم٤مفتاح الجنة الصلاة ومفتاح الصلاة الوضوء٥اللهم إني أعوذ بك قال شعبة وقد قال مرة أخرى أعوذ بك من الخبث والخبيث…٦اللهم إني أعوذ بك من الخبث والخبائث٧حدثنا محمد بن إسماعيل، حدثنا مالك بن إسماعيل، عن إسرائيل بن يونس، عن…٨إذا أتيتم الغائط فلا تستقبلوا القبلة بغائط ولا بول ولا تستدبروها ولكن…٩ومحمد بن المثنى، قالا حدثنا وهب بن جرير، حدثنا أبي، عن محمد بن إسحاق،…١٠وقد روى هذا الحديث ابن لهيعة، عن أبي الزبير، عن جابر، عن أبي قتادة،…١١واسع بن حبان، عن ابن عمر، قال رقيت يوما على بيت حفصة فرأيت النبي صلى…١٢يا عمر لا تبل قائما١٣حدثنا هناد، حدثنا وكيع، عن الأعمش، عن أبي وائل، عن حذيفة، أن النبي…١٤حدثنا قتيبة بن سعيد، حدثنا عبد السلام بن حرب الملائي، عن الأعمش، عن…١٥حدثنا محمد بن أبي عمر المكي، حدثنا سفيان بن عيينة، عن معمر، عن يحيى…١٦حدثنا هناد، حدثنا أبو معاوية، عن الأعمش، عن إبراهيم، عن عبد الرحمن بن…١٧التمس لي ثلاثة أحجار١٨لا تستنجوا بالروث ولا بالعظام فإنه زاد إخوانكم من الجن١٩. وفي الباب عن جرير بن عبد الله البجلي وأنس وأبي هريرة . قال أبو عيسى…٢٠حدثنا محمد بن بشار، حدثنا عبد الوهاب الثقفي، عن محمد بن عمرو، عن أبي…٢١إن عامة الوسواس منه٢٢لولا أن أشق على أمتي لأمرتهم بالسواك عند كل صلاة٢٣لولا أن أشق على أمتي لأمرتهم بالسواك عند كل صلاة ولأخرت صلاة العشاء…٢٤إذا استيقظ أحدكم من الليل فلا يدخل يده في الإناء حتى يفرغ عليها مرتين…٢٥لا وضوء لمن لم يذكر اسم الله عليه٢٦حدثنا الحسن بن علي الحلواني، حدثنا يزيد بن هارون، عن يزيد بن عياض، عن…٢٧إذا توضأت فانتثر وإذا استجمرت فأوتر٢٨حدثنا يحيى بن موسى، حدثنا إبراهيم بن موسى الرازي، حدثنا خالد بن عبد…٢٩قال رأيت عمار بن ياسر توضأ فخلل لحيته فقيل له أو قال فقلت له أتخلل…٣٠حدثنا ابن أبي عمر، حدثنا سفيان بن عيينة، عن سعيد بن أبي عروبة، عن…٣١أن النبي صلى الله عليه وسلم كان يخلل لحيته . قال أبو عيسى هذا حديث…٣٢حدثنا إسحاق بن موسى الأنصاري، حدثنا معن بن عيسى القزاز، حدثنا مالك بن…٣٣. قال أبو عيسى هذا حديث حسن وحديث عبد الله بن زيد أصح من هذا وأجود…٣٤. قال وفي الباب عن علي وجد طلحة بن مصرف بن عمرو . قال أبو عيسى وحديث…٣٥. ورواية عمرو بن الحارث عن حبان أصح لأنه قد روي من غير وجه هذا الحديث…٣٦حدثنا هناد، حدثنا عبد الله بن إدريس، عن محمد بن عجلان، عن زيد بن…٣٧الأذنان من الرأس٣٨إذا توضأت فخلل الأصابع٣٩إذا توضأت فخلل بين أصابع يديك ورجليك٤٠قال رأيت النبي صلى الله عليه وسلم إذا توضأ دلك أصابع رجليه بخنصره .…٤١ويل للأعقاب من النار٤٢وهناد، وقتيبة، قالوا حدثنا وكيع، عن سفيان، ح قال وحدثنا محمد بن بشار،…٤٣ومحمد بن رافع، قالا حدثنا زيد بن حباب، عن عبد الرحمن بن ثابت بن…٤٤حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، عن سفيان، عن أبي إسحاق،…٤٥قال قلت لأبي جعفر حدثك جابر، أن النبي صلى الله عليه وسلم توضأ مرة مرة…٤٦قال أبو عيسى وروى وكيع، هذا الحديث عن ثابت بن أبي صفية، قال قلت لأبي…٤٧أن النبي صلى الله عليه وسلم توضأ فغسل وجهه ثلاثا وغسل يديه مرتين…٤٨وقتيبة، قالا حدثنا أبو الأحوص، عن أبي إسحاق، عن أبي حية، قال رأيت…٤٩وهناد، قالا حدثنا أبو الأحوص، عن أبي إسحاق، عن عبد خير، ذكر عن علي،…٥٠جاءني جبريل فقال يا محمد إذا توضأت فانتضح٥١ألا أدلكم على ما يمحو الله به الخطايا ويرفع به الدرجات٥٢فذلكم الرباط فذلكم الرباط فذلكم الرباط٥٣حدثنا سفيان بن وكيع بن الجراح، حدثنا عبد الله بن وهب، عن زيد بن حباب،…٥٤حدثنا قتيبة، حدثنا رشدين بن سعد، عن عبد الرحمن بن زياد بن أنعم، عن…٥٥من توضأ فأحسن الوضوء ثم قال أشهد أن لا إله إلا الله وحده لا شريك له…٥٦وعلي بن حجر، قالا حدثنا إسماعيل ابن علية، عن أبي ريحانة، عن سفينة، أن…٥٧إن للوضوء شيطانا يقال له الولهان فاتقوا وسواس الماء٥٨حدثنا محمد بن حميد الرازي، حدثنا سلمة بن الفضل، عن محمد بن إسحاق، عن…٥٩من توضأ على طهر كتب الله له به عشر حسنات٦٠وعبد الرحمن، هو ابن مهدي قالا حدثنا سفيان بن سعيد، عن عمرو بن عامر…٦١حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، عن سفيان، عن علقمة بن…٦٢. قال أبو عيسى هذا حديث حسن صحيح . وهو قول عامة الفقهاء أن لا بأس أن…٦٣حدثنا محمود بن غيلان، قال حدثنا وكيع، عن سفيان، عن سليمان التيمي، عن…٦٤ومحمود بن غيلان، قالا حدثنا أبو داود، عن شعبة، عن عاصم، قال سمعت أبا…٦٥إن الماء لا يجنب٦٦إن الماء طهور لا ينجسه شيء٦٧إذا كان الماء قلتين لم يحمل الخبث٦٨لا يبولن أحدكم في الماء الدائم ثم يتوضأ منه٦٩هو الطهور ماؤه الحل ميتته٧٠إنهما يعذبان وما يعذبان في كبير أما هذا فكان لا يستتر من بوله وأما…٧١وأحمد بن منيع، قالا حدثنا سفيان بن عيينة، عن الزهري، عن عبيد الله بن…٧٢اشربوا من ألبانها وأبوالها٧٣حدثنا الفضل بن سهل الأعرج البغدادي، حدثنا يحيى بن غيلان، قال حدثنا…٧٤لا وضوء إلا من صوت أو ريح٧٥إذا كان أحدكم في المسجد فوجد ريحا بين أليتيه فلا يخرج حتى يسمع صوتا…٧٦إن الله لا يقبل صلاة أحدكم إذا أحدث حتى يتوضأ٧٧إن الوضوء لا يجب إلا على من نام مضطجعا فإنه إذا اضطجع استرخت مفاصله٧٨حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، عن شعبة، عن قتادة، عن أنس بن…٧٩الوضوء مما مست النار ولو من ثور أقط٨٠سمع جابرا، . قال سفيان وحدثنا محمد بن المنكدر، عن جابر، قال خرج رسول…٨١حدثنا هناد، حدثنا أبو معاوية، عن الأعمش، عن عبد الله بن عبد الله…٨٢من مس ذكره فلا يصل حتى يتوضأ٨٣وروى أبو أسامة، وغير، واحد، هذا الحديث عن هشام بن عروة، عن أبيه، عن…٨٤وروى هذا الحديث أبو الزناد، عن عروة، عن بسرة، عن النبي صلى الله عليه…٨٥وهل هو إلا مضغة منه أو بضعة منه٨٦وهناد، وأبو كريب وأحمد بن منيع ومحمود بن غيلان وأبو عمار الحسين بن…٨٧حدثنا أبو عبيدة بن أبي السفر، - وهو أحمد بن عبد الله الهمداني الكوفي…٨٨حدثنا هناد، حدثنا شريك، عن أبي فزارة، عن أبي زيد، عن عبد الله بن…٨٩حدثنا قتيبة، حدثنا الليث، عن عقيل، عن الزهري، عن عبيد الله بن عبد…٩٠ومحمد بن بشار، قالا حدثنا أبو أحمد، محمد بن عبد الله الزبيري عن…٩١يغسل الإناء إذا ولغ فيه الكلب سبع مرات أولاهن أو أخراهن بالتراب وإذا…٩٢إنها ليست بنجس إنما هي من الطوافين عليكم أو الطوافات٩٣حدثنا هناد، حدثنا وكيع، عن الأعمش، عن إبراهيم، عن همام بن الحارث، قال…٩٤ويروى عن شهر بن حوشب، قال رأيت جرير بن عبد الله توضأ ومسح على خفيه…٩٥للمسافر ثلاثة وللمقيم يوم٩٦. قال أبو عيسى هذا حديث حسن صحيح . وقد روى الحكم بن عتيبة وحماد عن…٩٧حدثنا أبو الوليد الدمشقي، حدثنا الوليد بن مسلم، أخبرني ثور بن يزيد،…٩٨حدثنا علي بن حجر، قال حدثنا عبد الرحمن بن أبي الزناد، عن أبيه، عن…٩٩ومحمود بن غيلان، قالا حدثنا وكيع، عن سفيان، عن أبي قيس، عن هزيل بن…١٠٠حدثنا محمد بن بشار، حدثنا يحيى بن سعيد القطان، عن سليمان التيمي، عن…١٠١أن النبي صلى الله عليه وسلم مسح على الخفين والخمار .١٠٢على العمامة فقال أمس الشعر الماء .١٠٣حدثنا هناد، حدثنا وكيع، عن الأعمش، عن سالم بن أبي الجعد، عن كريب، عن…١٠٤. قال أبو عيسى هذا حديث حسن صحيح . وهو الذي اختاره أهل العلم في الغسل…١٠٥لا إنما يكفيك أن تحثين على رأسك ثلاث حثيات من ماء ثم تفيضين على سائر…١٠٦تحت كل شعرة جنابة فاغسلوا الشعر وأنقوا البشر١٠٧. قال أبو عيسى هذا حديث حسن صحيح . قال أبو عيسى وهذا قول غير واحد من…١٠٨حدثنا أبو موسى، محمد بن المثنى حدثنا الوليد بن مسلم، عن الأوزاعي، عن…١٠٩إذا جاوز الختان الختان وجب الغسل١١٠قال إنما كان الماء من الماء رخصة في أول الإسلام ثم نهي عنها .١١١بهذا الإسناد مثله . قال أبو عيسى هذا حديث حسن صحيح . وإنما كان الماء…١١٢حدثنا علي بن حجر، أخبرنا شريك، عن أبي الجحاف، عن عكرمة، عن ابن عباس،…١١٣حدثنا أحمد بن منيع، حدثنا حماد بن خالد الخياط، عن عبد الله بن عمر، هو…١١٤من المذى الوضوء ومن المني الغسل١١٥إنما يجزئك من ذلك الوضوء١١٦حدثنا هناد، حدثنا أبو معاوية، عن الأعمش، عن إبراهيم، عن همام بن…١١٧حدثنا أحمد بن منيع، قال حدثنا أبو معاوية، عن عمرو بن ميمون بن مهران،…١١٨قالت كان رسول الله صلى الله عليه وسلم ينام وهو جنب ولا يمس ماء .١١٩نحوه . قال أبو عيسى وهذا قول سعيد بن المسيب وغيره . وقد روى غير،…١٢٠حدثنا محمد بن المثنى، حدثنا يحيى بن سعيد، عن عبيد الله بن عمر، عن…١٢١أين كنت أو أين ذهبت١٢٢نعم إذا هي رأت الماء فلتغتسل١٢٣. قال أبو عيسى هذا حديث ليس بإسناده بأس . وهو قول غير واحد من أهل…١٢٤إن الصعيد الطيب طهور المسلم وإن لم يجد الماء عشر سنين فإذا وجد الماء…١٢٥لا إنما ذلك عرق وليست بالحيضة فإذا أقبلت الحيضة فدعي الصلاة وإذا…١٢٦تدع الصلاة أيام أقرائها التي كانت تحيض فيها ثم تغتسل وتتوضأ عند كل…١٢٧حدثنا علي بن حجر، أخبرنا شريك، نحوه بمعناه . قال أبو عيسى هذا حديث قد…١٢٨أنعت لك الكرسف فإنه يذهب الدم١٢٩لا إنما ذلك عرق فاغتسلي ثم صلي١٣٠أن امرأة، سألت عائشة قالت أتقضي إحدانا صلاتها أيام محيضها فقالت…١٣١لا تقرإ الحائض ولا الجنب شيئا من القرآن١٣٢حدثنا بندار، حدثنا عبد الرحمن بن مهدي، عن سفيان، عن منصور، عن…١٣٣ومحمد بن عبد الأعلى، قالا حدثنا عبد الرحمن بن مهدي، حدثنا معاوية بن…١٣٤ناوليني الخمرة من المسجد١٣٥من أتى حائضا أو امرأة في دبرها أو كاهنا فقد كفر بما أنزل على محمد صلى…١٣٦يتصدق بنصف دينار١٣٧إذا كان دما أحمر فدينار وإذا كان دما أصفر فنصف دينار١٣٨حتيه ثم اقرصيه بالماء ثم رشيه وصلي فيه١٣٩حدثنا نصر بن علي الجهضمي، حدثنا شجاع بن الوليد أبو بدر، عن علي بن عبد…١٤٠حدثنا بندار، محمد بن بشار حدثنا أبو أحمد، حدثنا سفيان، عن معمر، عن…١٤١إذا أتى أحدكم أهله ثم أراد أن يعود فليتوضأ بينهما وضوءا١٤٢إذا أقيمت الصلاة ووجد أحدكم الخلاء فليبدأ بالخلاء١٤٣حدثنا أبو رجاء، قتيبة حدثنا مالك بن أنس،
