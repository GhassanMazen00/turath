كتاب فضائل الجهاد عن رسول الله صلى الله عليه وسلم

٥١ حديثًا
١٦١٩حدثنا قتيبة بن سعيد، حدثنا أبو عوانة، عن سهيل بن أبي صالح، عن أبيه،…١٦٢٠يقول الله عز وجل المجاهد في سبيل الله هو على ضامن إن قبضته أورثته…١٦٢١كل ميت يختم على عمله إلا الذي مات مرابطا في سبيل الله فإنه ينمى له…١٦٢٢من صام يوما في سبيل الله زحزحه الله عن النار سبعين خريفا١٦٢٣لا يصوم عبد يوما في سبيل الله إلا باعد ذلك اليوم النار عن وجهه سبعين…١٦٢٤من صام يوما في سبيل الله جعل الله بينه وبين النار خندقا كما بين…١٦٢٥من أنفق نفقة في سبيل الله كتبت له بسبعمائة ضعف١٦٢٦خدمة عبد في سبيل الله أو ظل فسطاط أو طروقة فحل في سبيل الله١٦٢٧أفضل الصدقات ظل فسطاط في سبيل الله ومنيحة خادم في سبيل الله أو طروقة…١٦٢٨من جهز غازيا في سبيل الله فقد غزا ومن خلف غازيا في أهله فقد غزا١٦٢٩من جهز غازيا في سبيل الله أو خلفه في أهله فقد غزا١٦٣٠حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، حدثنا عبد الملك بن أبي…١٦٣١من جهز غازيا في سبيل الله فقد غزا ومن خلف غازيا في أهله فقد غزا١٦٣٢من اغبرت قدماه في سبيل الله فهما حرام على النار١٦٣٣لا يلج النار رجل بكى من خشية الله حتى يعود اللبن في الضرع ولا يجتمع…١٦٣٤من شاب شيبة في الإسلام كانت له نورا يوم القيامة١٦٣٥من شاب شيبة في سبيل الله كانت له نورا يوم القيامة١٦٣٦الخيل معقود في نواصيها الخير إلى يوم القيامة الخيل لثلاثة هي لرجل أجر…١٦٣٧حديث ١٦٣٧١٦٣٨من رمى بسهم في سبيل الله فهو له عدل محرر١٦٣٩عينان لا تمسهما النار عين بكت من خشية الله وعين باتت تحرس في سبيل الله١٦٤٠القتل في سبيل الله يكفر كل خطيئة١٦٤١إن أرواح الشهداء في طير خضر تعلق من ثمرة الجنة أو شجر الجنة١٦٤٢عرض على أول ثلاثة يدخلون الجنة شهيد وعفيف متعفف وعبد أحسن عبادة الله…١٦٤٣ما من عبد يموت له عند الله خير يحب أن يرجع إلى الدنيا وأن له الدنيا…١٦٤٤الشهداء أربعة رجل مؤمن جيد الإيمان لقي العدو فصدق الله حتى قتل فذلك…١٦٤٥ناس من أمتي عرضوا على غزاة في سبيل الله يركبون ثبج هذا البحر ملوك على…١٦٤٦من قاتل لتكون كلمة الله هي العليا فهو في سبيل الله١٦٤٧إنما الأعمال بالنية وإنما لامرئ ما نوى فمن كانت هجرته إلى الله وإلى…١٦٤٨غدوة في سبيل الله خير من الدنيا وما فيها وموضع سوط في الجنة خير من…١٦٤٩غدوة في سبيل الله أو روحة خير من الدنيا وما فيها١٦٥٠لا تفعل فإن مقام أحدكم في سبيل الله أفضل من صلاته في بيته سبعين عاما…١٦٥١لغدوة في سبيل الله أو روحة خير من الدنيا وما فيها ولقاب قوس أحدكم أو…١٦٥٢ألا أخبركم بخير الناس رجل ممسك بعنان فرسه في سبيل الله ألا أخبركم…١٦٥٣من سأل الله الشهادة من قلبه صادقا بلغه الله منازل الشهداء وإن مات على…١٦٥٤من سأل الله القتل في سبيله صادقا من قلبه أعطاه الله أجر الشهيد١٦٥٥ثلاثة حق على الله عونهم المجاهد في سبيل الله والمكاتب الذي يريد…١٦٥٦لا يكلم أحد في سبيل الله والله أعلم بمن يكلم في سبيله إلا جاء يوم…١٦٥٧من قاتل في سبيل الله من رجل مسلم فواق ناقة وجبت له الجنة ومن جرح جرحا…١٦٥٨إيمان بالله ورسوله١٦٥٩إن أبواب الجنة تحت ظلال السيوف١٦٦٠رجل يجاهد في سبيل الله١٦٦١ما من أحد من أهل الجنة يسره أن يرجع إلى الدنيا غير الشهيد فإنه يحب أن…١٦٦٢حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، عن قتادة، عن أنس،…١٦٦٣للشهيد عند الله ست خصال يغفر له في أول دفعة ويرى مقعده من الجنة ويجار…١٦٦٤رباط يوم في سبيل الله خير من الدنيا وما فيها وموضع سوط أحدكم في الجنة…١٦٦٥رباط يوم في سبيل الله أفضل وربما قال خير من صيام شهر وقيامه ومن مات…١٦٦٦من لقي الله بغير أثر من جهاد لقي الله وفيه ثلمة١٦٦٧رباط يوم في سبيل الله خير من ألف يوم فيما سواه من المنازل١٦٦٨ما يجد الشهيد من مس القتل إلا كما يجد أحدكم من مس القرصة١٦٦٩ليس شيء أحب إلى الله من قطرتين وأثرين قطرة من دموع في خشية الله وقطرة…
