٤٢٤ حديثًا•صفحة ١ من ٣
٢٩٥٠من قال في القرآن بغير علم فليتبوأ مقعده من النار٢٩٥١اتقوا الحديث عني إلا ما علمتم فمن كذب على متعمدا فليتبوأ مقعده من…٢٩٥٢من قال في القرآن برأيه فأصاب فقد أخطأ٢٩٥٣من صلى صلاة لم يقرأ فيها بأم القرآن فهي خداج فهي خداج غير تمام٢٩٥٣.٢ما يفرك أن تقول لا إله إلا الله فهل تعلم من إله سوى الله٢٩٥٤اليهود مغضوب عليهم والنصارى ضلال٢٩٥٥إن الله تعالى خلق آدم من قبضة قبضها من جميع الأرض فجاء بنو آدم على…٢٩٥٦دخلوا متزحفين على أوراكهم٢٩٥٧حدثنا محمود بن غيلان، حدثنا وكيع، حدثنا أشعث السمان، عن عاصم بن عبيد…٢٩٥٨حدثنا عبد بن حميد، أخبرنا يزيد بن هارون، أخبرنا عبد الملك بن أبي…٢٩٥٩أن عمر، قال يا رسول الله لو صلينا خلف المقام فنزلت : ( واتخذوا من…٢٩٦٠حدثنا أحمد بن منيع، حدثنا هشيم، أخبرنا حميد الطويل، عن أنس، قال قال…٢٩٦١نحوه .٢٩٦٢حدثنا هناد، حدثنا وكيع، عن إسرائيل، عن أبي إسحاق، عن البراء بن عازب،…٢٩٦٣حدثنا هناد، حدثنا وكيع، عن سفيان، عن عبد الله بن دينار، عن ابن عمر،…٢٩٦٤حدثنا هناد، وأبو عمار قالا حدثنا وكيع، عن إسرائيل، عن سماك، عن عكرمة،…٢٩٦٥حدثنا ابن أبي عمر، حدثنا سفيان، قال سمعت الزهري، يحدث عن عروة، قال…٢٩٦٦والمروة، فقال كانا من شعائر الجاهلية فلما كان الإسلام أمسكنا عنهما…٢٩٦٧نبدأ بما بدأ الله به٢٩٦٨قال كان أصحاب النبي صلى الله عليه وسلم إذا كان الرجل صائما فحضر…٢٩٦٩الدعاء هو العبادة٢٩٧٠إنما ذاك بياض النهار من سواد الليل٢٩٧١إنما هو الليل والنهار٢٩٧٢حدثنا عبد بن حميد، حدثنا الضحاك بن مخلد أبو عاصم النبيل، عن حيوة بن…٢٩٧٣كأن هوام رأسك تؤذيك٢٩٧٤أتؤذيك هوام رأسك٢٩٧٥الحج عرفات الحج عرفات الحج عرفات أيام منى ثلاث : ( فمن تعجل في يومين…٢٩٧٦أبغض الرجال إلى الله الألد الخصم٢٩٧٧حدثنا عبد بن حميد، حدثني سليمان بن حرب، حدثنا حماد بن سلمة، عن ثابت،…٢٩٧٨سمع جابرا، يقول كانت اليهود تقول من أتى امرأته في قبلها من دبرها كان…٢٩٧٩حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، حدثنا سفيان، عن ابن…٢٩٨٠قال جاء عمر إلى رسول الله صلى الله عليه وسلم فقال يا رسول الله هلكت…٢٩٨١حدثنا عبد بن حميد، حدثنا هشام بن القاسم، عن المبارك بن فضالة، عن…٢٩٨٢حدثنا قتيبة، عن مالك بن أنس، ح قال وحدثنا الأنصاري، حدثنا معن، حدثنا…٢٩٨٣صلاة الوسطى صلاة العصر٢٩٨٤اللهم املأ قبورهم وبيوتهم نارا كما شغلونا عن صلاة الوسطى حتى غابت…٢٩٨٥صلاة الوسطى صلاة العصر٢٩٨٦ويزيد بن هارون، ومحمد بن عبيد، عن إسماعيل بن أبي خالد، عن الحارث بن…٢٩٨٧حدثنا عبد الله بن عبد الرحمن، أخبرنا عبيد الله بن موسى، عن إسرائيل،…٢٩٨٨إن للشيطان لمة بابن آدم وللملك لمة فأما لمة الشيطان فإيعاد بالشر…٢٩٨٩يا أيها الناس إن الله طيب ولا يقبل إلا طيبا وإن الله أمر المؤمنين بما…٢٩٩٠سمع عليا، يقول لما نزلت هذه الآية : ( إن تبدوا ما في أنفسكم أو تخفوه…٢٩٩١هذه معاتبة الله العبد فيما يصيبه من الحمى والنكبة حتى البضاعة يضعها…٢٩٩٢قولوا سمعنا وأطعنا٢٩٩٣فإذا رأيتيهم فاعرفيهم٢٩٩٤إذا رأيتم الذين يتبعون ما تشابه منه فأولئك الذين سماهم الله فاحذروهم٢٩٩٥إن لكل نبي ولاة من النبيين وإن وليي أبي وخليل ربي٢٩٩٦من حلف على يمين هو فيها فاجر ليقتطع بها مال امرئ مسلم لقي الله وهو…٢٩٩٧اجعله في قرابتك أو أقربيك٢٩٩٨قال قام رجل إلى النبي صلى الله عليه وسلم فقال من الحاج يا رسول الله…٢٩٩٩اللهم هؤلاء أهلي٣٠٠٠كلاب النار شر قتلى تحت أديم السماء خير قتلى من قتلوه٣٠٠١إنكم تتمون سبعين أمة أنتم خيرها وأكرمها على الله٣٠٠٢كيف يفلح قوم فعلوا هذا بنبيهم وهو يدعوهم إلى الله٣٠٠٣كيف تفلح أمة فعلوا هذا بنبيهم وهو يدعوهم إلى الله٣٠٠٤اللهم العن أبا سفيان اللهم العن الحارث بن هشام اللهم العن صفوان بن…٣٠٠٥حدثنا يحيى بن حبيب بن عربي البصري، حدثنا خالد بن الحارث، عن محمد بن…٣٠٠٦ما من رجل يذنب ذنبا ثم يقوم فيتطهر ثم يصلي ثم يستغفر الله إلا غفر له٣٠٠٧مثله . قال أبو عيسى هذا حديث حسن صحيح .٣٠٠٨أن أبا طلحة، قال غشينا ونحن في مصافنا يوم أحد حدث أنه كان فيمن غشيه…٣٠٠٩حدثنا قتيبة، حدثنا عبد الواحد بن زياد، عن خصيف، حدثنا مقسم، قال قال…٣٠١٠يا جابر ما لي أراك منكسرا٣٠١١حدثنا ابن أبي عمر، حدثنا سفيان، عن الأعمش، عن عبد الله بن مرة، عن…٣٠١١.٢حديث ٣٠١١.٢٣٠١٢ما من رجل لا يؤدي زكاة ماله إلا جعل الله يوم القيامة في عنقه شجاعا٣٠١٣إن موضع سوط في الجنة لخير من الدنيا وما فيها اقرءوا إن شئتم : ( فمن…٣٠١٤أن حميد بن عبد الرحمن بن عوف، أخبره أن مروان بن الحكم قال اذهب يا…٣٠١٥حدثنا عبد بن حميد، حدثنا يحيى بن آدم، حدثنا ابن عيينة، عن محمد بن…٣٠١٥.٢حديث ٣٠١٥.٢٣٠١٦حدثنا عبد بن حميد، أخبرنا حبان بن هلال، حدثنا همام بن يحيى، حدثنا…٣٠١٧حدثنا أحمد بن منيع، حدثنا هشيم، أخبرنا عثمان البتي، عن أبي الخليل، عن…٣٠١٨الشرك بالله وعقوق الوالدين وقتل النفس وقول الزور٣٠١٩ألا أحدثكم بأكبر الكبائر٣٠٢٠إن من أكبر الكبائر الشرك بالله وعقوق الوالدين واليمين الغموس وما حلف…٣٠٢١الكبائر الإشراك بالله وعقوق الوالدين أو قال اليمين الغموس٣٠٢٢حدثنا ابن أبي عمر، حدثنا سفيان، عن ابن أبي نجيح، عن مجاهد، عن أم…٣٠٢٣قالت يا رسول الله لا أسمع الله ذكر النساء في الهجرة . فأنزل الله…٣٠٢٤حدثنا هناد، حدثنا أبو الأحوص، عن الأعمش، عن إبراهيم، عن علقمة، قال…٣٠٢٥حدثنا محمود بن غيلان، حدثنا معاوية بن هشام، حدثنا سفيان الثوري، عن…٣٠٢٦حدثنا سويد بن نصر، أخبرنا ابن المبارك، عن سفيان، عن الأعمش، نحو حديث…٣٠٢٧اسق يا زبير وأرسل الماء إلى جارك٣٠٢٨إنها طيبة وقال إنها تنفي الخبيث كما تنفي النار خبث الحديد٣٠٢٩يجيء المقتول بالقاتل يوم القيامة ناصيته ورأسه بيده وأوداجه تشخب دما…٣٠٣٠حدثنا عبد بن حميد، حدثنا عبد العزيز بن أبي رزمة، عن إسرائيل، عن سماك،…٣٠٣١ائتوني بالكتف والدواة أو اللوح والدواة٣٠٣٢سمع مقسما، مولى عبد الله بن الحارث يحدث عن ابن عباس، أنه قال : ( لا…٣٠٣٣حدثنا عبد بن حميد، حدثنا يعقوب بن إبراهيم بن سعد، عن أبيه، عن صالح بن…٣٠٣٤صدقة تصدق الله بها عليكم فاقبلوا صدقته٣٠٣٥حدثنا محمود بن غيلان، حدثنا عبد الصمد بن عبد الوارث، حدثنا سعيد بن…٣٠٣٦حدثنا الحسن بن أحمد بن أبي شعيب أبو مسلم الحراني، حدثنا محمد بن سلمة…٣٠٣٧قال ما في القرآن آية أحب إلى من هذه الآية : ( إن الله لا يغفر أن يشرك…٣٠٣٨قاربوا وسددوا وفي كل ما يصيب المؤمن كفارة حتى الشوكة يشاكها أو النكبة…٣٠٣٩يا أبا بكر ألا أقرئك آية أنزلت على٣٠٤٠قال خشيت سودة أن يطلقها، النبي صلى الله عليه وسلم فقالت لا تطلقني…٣٠٤١قال آخر آية أنزلت أو آخر شيء نزل : ( يستفتونك قل الله يفتيكم في…٣٠٤٢تجزيك آية الصيف٣٠٤٣وغيره، عن قيس بن مسلم، عن طارق بن شهاب، قال قال رجل من اليهود لعمر بن…٣٠٤٤حدثنا عبد بن حميد، أخبرنا يزيد بن هارون، أخبرنا حماد بن سلمة، عن عمار…٣٠٤٥يمين الرحمن ملأى سحاء لا يغيضها الليل والنهار قال أرأيتم ما أنفق منذ…٣٠٤٦يا أيها الناس انصرفوا فقد عصمني الله٣٠٤٧لما وقعت بنو إسرائيل في المعاصي نهتهم علماؤهم فلم ينتهوا فجالسوهم في…٣٠٤٨إن بني إسرائيل لما وقع فيهم النقص كان الرجل فيهم يرى أخاه على الذنب…٣٠٤٩عمرو بن شرحبيل أن عمر بن الخطاب، قال اللهم بين لنا في الخمر بيان شفاء…٣٠٥٠حدثنا عبد بن حميد، حدثنا عبيد الله بن موسى، عن إسرائيل، عن أبي إسحاق،…٣٠٥١وقد رواه شعبة عن أبي إسحاق، عن البراء، أيضا حدثنا بذلك، محمد بن بشار…٣٠٥٢قال قالوا يا رسول الله أرأيت الذين ماتوا وهم يشربون الخمر لما نزل…٣٠٥٣قال لما نزلت : ( ليس على الذين آمنوا وعملوا الصالحات جناح فيما طعموا…٣٠٥٤أن رجلا، أتى النبي صلى الله عليه وسلم فقال يا رسول الله إني إذا أصبت…٣٠٥٥لا ولو قلت نعم لوجبت٣٠٥٦حدثنا محمد بن معمر أبو عبد الله البصري، حدثنا روح بن عبادة، حدثنا…٣٠٥٧إن الناس إذا رأوا ظالما فلم يأخذوا على يديه أوشك أن يعمهم الله بعقاب…٣٠٥٨بل ائتمروا بالمعروف وتناهوا عن المنكر حتى إذا رأيت شحا مطاعا وهوى…٣٠٥٩حدثنا الحسن بن أحمد بن أبي شعيب الحراني، حدثنا محمد بن سلمة الحراني،…٣٠٦٠حدثنا سفيان بن وكيع، حدثنا يحيى بن آدم، عن ابن أبي زائدة، عن محمد بن…٣٠٦١أنزلت المائدة من السماء خبزا ولحما وأمروا أن لا يخونوا ولا يدخروا لغد…٣٠٦٢حدثنا ابن أبي عمر، حدثنا سفيان بن عيينة، عن عمرو بن دينار، عن طاوس،…٣٠٦٣حدثنا قتيبة، حدثنا عبد الله بن وهب، عن حيى، عن أبي عبد الرحمن الحبلي،…٣٠٦٤أن أبا جهل، قال للنبي صلى الله عليه وسلم إنا لا نكذبك ولكن نكذب بما…٣٠٦٥سمع جابر بن عبد الله، يقول لما نزلت هذه الآية : ( قل هو القادر على أن…٣٠٦٦أما إنها كائنة ولم يأت تأويلها بعد٣٠٦٧ليس ذلك إنما هو الشرك ألم تسمعوا ما قال لقمان لابنه : ( يا بني لا…٣٠٦٨إنما ذاك جبريل ما رأيته في الصورة التي خلق فيها غير هاتين المرتين…٣٠٦٩حدثنا محمد بن موسى البصري الحرشي، حدثنا زياد بن عبد الله البكائي،…٣٠٧٠قال من سره أن ينظر إلى الصحيفة التي عليها خاتم محمد صلى الله عليه…٣٠٧١طلوع الشمس من مغربها٣٠٧٢ثلاث إذا خرجن : لم ينفع نفسا إيمانها لم تكن آمنت من قبل الآية الدجال…٣٠٧٣قال الله عز وجل وقوله الحق إذا هم عبدي بحسنة فاكتبوها له حسنة فإن…٣٠٧٤حدثنا عبد الله بن عبد الرحمن، أخبرنا سليمان بن حرب، حدثنا حماد بن…٣٠٧٥إن الله خلق آدم ثم مسح ظهره بيمينه فاستخرج منه ذرية فقال خلقت هؤلاء…٣٠٧٦لما خلق الله آدم مسح ظهره فسقط من ظهره كل نسمة هو خالقها من ذريته إلى…٣٠٧٧لما حملت حواء طاف بها إبليس وكان لا يعيش لها ولد فقال سميه عبد الحارث…٣٠٧٨قال قال رسول الله صلى الله عليه وسلم " لما خلق آدم " . الحديث .٣٠٧٩هذا ليس لي ولا لك٣٠٨٠قال لما فرغ رسول الله صلى الله عليه وسلم من بدر قيل له عليك العير ليس…٣٠٨١اللهم أنجز لي ما وعدتني اللهم آتني ما وعدتني اللهم إن تهلك هذه…٣٠٨٢أنزل الله على أمانين لأمتي : ( وما كان الله ليعذبهم وأنت فيهم ) ( وما…٣٠٨٣ألا إن القوة الرمى ثلاث مرات ألا إن الله سيفتح لكم الأرض وستكفون…٣٠٨٤ما تقولون في هؤلاء الأسارى٣٠٨٥لم تحل الغنائم لأحد سود الرءوس من قبلكم العوام،…٣٠٩٢وغير، واحد، قالوا حدثنا سفيان بن عيينة، عن أبي إسحاق، عن زيد بن يثيع،…٣٠٩٣إذا رأيتم الرجل يعتاد المسجد فاشهدوا له بالإيمان . قال الله تعالى : (…٣٠٩٤أفضله لسان ذاكر وقلب شاكر وزوجة مؤمنة
