١٣٨ حديثًا
٢٧٣٦للمسلم على المسلم ست بالمعروف يسلم عليه إذا لقيه ويجيبه إذا دعاه…٢٧٣٧للمؤمن على المؤمن ست خصال يعوده إذا مرض ويشهده إذا مات ويجيبه إذا…٢٧٣٨أن رجلا، عطس إلى جنب ابن عمر فقال الحمد لله والسلام على رسول الله .…٢٧٣٩يهديكم الله ويصلح بالكم٢٧٤٠عليك وعلى أمك إذا عطس أحدكم فليقل الحمد لله رب العالمين وليقل له من…٢٧٤١إذا عطس أحدكم فليقل الحمد لله على كل حال وليقل الذي يرد عليه يرحمك…٢٧٤١.٢بهذا الإسناد نحوه . قال هكذا روى شعبة، هذا الحديث عن ابن أبي ليلى، عن…٢٧٤١.٣ومحمد بن يحيى الثقفي المروزي، قالا حدثنا يحيى بن سعيد القطان، عن ابن…٢٧٤٢إنه حمد الله وإنك لم تحمد الله٢٧٤٣حدثنا سويد بن نصر، أخبرنا عبد الله، أخبرنا عكرمة بن عمار، عن إياس بن…٢٧٤٣.٢حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، حدثنا عكرمة بن عمار، عن إياس…٢٧٤٣.٣وقد روى شعبة، عن عكرمة بن عمار، هذا الحديث نحو رواية يحيى بن سعيد…٢٧٤٣.٤وروى عبد الرحمن بن مهدي، عن عكرمة بن عمار، نحو رواية ابن المبارك وقال…٢٧٤٤يشمت العاطس ثلاثا فإن زاد فإن شئت فشمته وإن شئت فلا٢٧٤٥أن النبي صلى الله عليه وسلم كان إذا عطس غطى وجهه بيده أو بثوبه وغض…٢٧٤٦العطاس من الله والتثاؤب من الشيطان فإذا تثاءب أحدكم فليضع يده على فيه…٢٧٤٧إن الله يحب العطاس ويكره التثاؤب فإذا عطس أحدكم فقال الحمد لله فحق…٢٧٤٨العطاس والنعاس والتثاؤب في الصلاة والحيض والقىء والرعاف من الشيطان٢٧٤٩لا يقم أحدكم أخاه من مجلسه ثم يجلس فيه٢٧٥٠لا يقم أحدكم أخاه من مجلسه ثم يجلس فيه٢٧٥١الرجل أحق بمجلسه وإن خرج لحاجته ثم عاد فهو أحق بمجلسه٢٧٥٢لا يحل للرجل أن يفرق بين اثنين إلا بإذنهما٢٧٥٣أن رجلا، قعد وسط حلقة فقال حذيفة ملعون على لسان محمد أو لعن الله على…٢٧٥٤قال لم يكن شخص أحب إليهم من رسول الله صلى الله عليه وسلم قال وكانوا…٢٧٥٥من سره أن يتمثل له الرجال قياما فليتبوأ مقعده من النار٢٧٥٦خمس من الفطرة الاستحداد والختان وقص الشارب ونتف الإبط وتقليم الأظفار٢٧٥٧عشر من الفطرة قص الشارب وإعفاء اللحية والسواك والاستنشاق وقص الأظفار…٢٧٥٨حدثنا إسحاق بن منصور، أخبرنا عبد الصمد بن عبد الوارث، حدثنا صدقة بن…٢٧٥٩حدثنا قتيبة، حدثنا جعفر بن سليمان، عن أبي عمران الجوني، عن أنس بن…٢٧٦٠قال كان النبي صلى الله عليه وسلم يقص أو يأخذ من شاربه وكان إبراهيم…٢٧٦١من لم يأخذ من شاربه فليس منا٢٧٦٢أن النبي صلى الله عليه وسلم نصب المنجنيق على أهل الطائف . قال قتيبة…٢٧٦٣أحفوا الشوارب وأعفوا اللحى٢٧٦٤حدثنا الأنصاري، حدثنا معن، حدثنا مالك، عن أبي بكر بن نافع، عن أبيه،…٢٧٦٥وغير، واحد، قالوا حدثنا سفيان بن عيينة، عن الزهري، عن عباد بن تميم،…٢٧٦٦إذا استلقى أحدكم على ظهره فلا يضع إحدى رجليه على الأخرى٢٧٦٧حدثنا قتيبة، حدثنا الليث، عن أبي الزبير، عن جابر، أن رسول الله صلى…٢٧٦٨إن هذه ضجعة لا يحبها الله٢٧٦٩احفظ عورتك إلا من زوجتك أو مما ملكت يمينك٢٧٧٠حدثنا عباس بن محمد الدوري البغدادي، حدثنا إسحاق بن منصور الكوفي،…٢٧٧١قال رأيت النبي صلى الله عليه وسلم متكئا على وسادة . هذا حديث صحيح .٢٧٧٢لا يؤم الرجل في سلطانه ولا يجلس على تكرمته في بيته إلا بإذنه٢٧٧٣لأنت أحق بصدر دابتك إلا أن تجعله لي٢٧٧٤حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، حدثنا سفيان، عن محمد بن…٢٧٧٥حدثنا عباس العنبري بن عبد العظيم، حدثنا النضر بن محمد، هو الجرشي…٢٧٧٦حدثنا أحمد بن منيع، حدثنا هشيم، أخبرنا يونس بن عبيد، عن عمرو بن سعيد،…٢٧٧٧يا علي لا تتبع النظرة النظرة فإن لك الأولى وليست لك الآخرة٢٧٧٨مولى أم سلمة أنه حدثه أن أم سلمة حدثته أنها، كانت عند رسول الله صلى…٢٧٧٩عمرو بن العاصي أن عمرو بن العاصي، أرسله إلى علي يستأذنه على أسماء بنت…٢٧٨٠ما تركت بعدي في الناس فتنة أضر على الرجال من النساء٢٧٨٠.٢حدثنا ابن أبي عمر، حدثنا سفيان، عن سليمان التيمي، عن أبي عثمان، عن…٢٧٨١إنما هلكت بنو إسرائيل حين اتخذها نساؤهم٢٧٨٢حدثنا أحمد بن منيع، حدثنا عبيدة بن حميد، عن منصور، عن إبراهيم، عن…٢٧٨٣لعن الله الواصلة والمستوصلة والواشمة والمستوشمة٢٧٨٣.٢حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، حدثنا عبيد الله بن عمر، عن…٢٧٨٤وهمام، عن قتادة، عن عكرمة، عن ابن عباس، قال لعن رسول الله صلى الله…٢٧٨٥وأيوب، عن عكرمة، عن ابن عباس، قال لعن رسول الله صلى الله عليه وسلم…٢٧٨٦كل عين زانية والمرأة إذا استعطرت فمرت بالمجلس فهي كذا وكذا يعني زانية٢٧٨٧طيب الرجال ما ظهر ريحه وخفي لونه وطيب النساء ما ظهر لونه وخفي ريحه٢٧٨٧.٢حديث ٢٧٨٧.٢٢٧٨٨إن خير طيب الرجل ما ظهر ريحه وخفي لونه وخير طيب النساء ما ظهر لونه…٢٧٨٩حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، حدثنا عزرة بن ثابت، عن…٢٧٩٠ثلاث لا ترد الوسائد والدهن واللبن٢٧٩١إذا أعطي أحدكم الريحان فلا يرده فإنه خرج من الجنة٢٧٩٢لا تباشر المرأة المرأة حتى تصفها لزوجها كأنما ينظر إليها٢٧٩٣لا ينظر الرجل إلى عورة الرجل ولا تنظر المرأة إلى عورة المرأة ولا يفضي…٢٧٩٤احفظ عورتك إلا من زوجتك أو ما ملكت يمينك٢٧٩٦حدثنا واصل بن عبد الأعلى الكوفي، حدثنا يحيى بن آدم، عن إسرائيل، عن…٢٧٩٧حدثنا واصل بن عبد الأعلى، حدثنا يحيى بن آدم، عن الحسن بن صالح، عن عبد…٢٧٩٨غط فخذك فإنها من العورة٢٧٩٩ويقال ابن إياس، عن صالح بن أبي حسان، قال سمعت سعيد بن المسيب، يقول إن…٢٨٠٠إياكم والتعري فإن معكم من لا يفارقكم إلا عند الغائط وحين يفضي الرجل…٢٨٠١من كان يؤمن بالله واليوم الآخر فلا يدخل الحمام بغير إزار ومن كان يؤمن…٢٨٠٢وكان، قد أدرك النبي صلى الله عليه وسلم عن عائشة أن النبي صلى الله…٢٨٠٣ما من امرأة تضع ثيابها في غير بيت زوجها إلا هتكت الستر بينها وبين ربها٢٨٠٤لا تدخل الملائكة بيتا فيه كلب ولا صورة تماثيل٢٨٠٥أن رافع بن إسحاق، أخبره قال دخلت أنا وعبد الله بن أبي طلحة، على أبي…٢٨٠٦أتاني جبريل فقال إني كنت أتيتك البارحة فلم يمنعني أن أكون دخلت عليك…٢٨٠٧قال مر رجل وعليه ثوبان أحمران فسلم على النبي صلى الله عليه وسلم فلم…٢٨٠٨حدثنا قتيبة، حدثنا أبو الأحوص، عن أبي إسحاق، عن هبيرة بن يريم، قال…٢٨٠٩وعبد الرحمن بن مهدي، قالا حدثنا شعبة، عن الأشعث بن سليم، عن معاوية بن…٢٨١٠البسوا البياض فإنها أطهر وأطيب وكفنوا فيها موتاكم٢٨١١قال رأيت رسول الله صلى الله عليه وسلم في ليلة إضحيان فجعلت أنظر إلى…٢٨١١.٢وروى شعبة، والثوري، عن أبي إسحاق، عن البراء بن عازب، قال رأيت على…٢٨١٢قال رأيت رسول الله صلى الله عليه وسلم وعليه بردان أخضران . قال أبو…٢٨١٣قالت خرج النبي صلى الله عليه وسلم ذات غداة وعليه مرط من شعر أسود .…٢٨١٤وعليك السلام ورحمة الله٢٨١٥حدثنا قتيبة، حدثنا حماد بن زيد، قال ح وحدثنا إسحاق بن منصور، حدثنا…٢٨١٥.٢حدثنا قتيبة، حدثنا حماد بن زيد، قال ح وحدثنا إسحاق بن منصور، حدثنا…٢٨١٦اذهب فاغسله ثم اغسله ثم لا تعد٢٨١٧من لبس الحرير في الدنيا لم يلبسه في الآخرة٢٨١٨حدثنا قتيبة، حدثنا الليث، عن ابن أبي مليكة، عن المسور بن مخرمة، أن…٢٨١٩إن الله يحب أن يرى أثر نعمته على عبده٢٨٢٠أن النجاشي، أهدى إلى النبي صلى الله عليه وسلم خفين أسودين ساذجين…٢٨٢١إنه نور المسلم٢٨٢٢المستشار مؤتمن٢٨٢٣المستشار مؤتمن٢٨٢٤الشؤم في ثلاثة في المرأة والمسكن والدابة٢٨٢٤.٢الشؤم في ثلاثة في المرأة والمسكن والدابة٢٨٢٤.٣الشؤم في ثلاثة في المرأة والمسكن والدابة٢٨٢٤.٤الشؤم في ثلاثة في المرأة والمسكن والدابة٢٨٢٥إذا كنتم ثلاثة فلا يتناجى اثنان دون صاحبهما٢٨٢٦حدثنا واصل بن عبد الأعلى الكوفي، حدثنا محمد بن فضيل، عن إسماعيل بن…٢٨٢٧حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، عن إسماعيل بن أبي خالد، حدثنا…٢٨٢٨حدثنا إبراهيم بن سعيد الجوهري، حدثنا سفيان بن عيينة، عن يحيى بن سعيد،…٢٨٢٩ارم فداك أبي وأمي٢٨٣٠ارم فداك أبي وأمي٢٨٣١حدثنا محمد بن عبد الملك بن أبي الشوارب، حدثنا أبو عوانة، حدثنا أبو…٢٨٣٢أن النبي صلى الله عليه وسلم أمر بتسمية المولود يوم سابعه ووضع الأذى…٢٨٣٣أحب الأسماء إلى الله عز وجل عبد الله وعبد الرحمن٢٨٣٤إن أحب الأسماء إلى الله عبد الله وعبد الرحمن٢٨٣٥لأنهين أن يسمى رافع وبركة ويسار٢٨٣٦لا تسمي غلامك رباح ولا أفلح ولا يسار ولا نجيح يقال أثم هو فيقال لا٢٨٣٧أخنع اسم عند الله يوم القيامة رجل تسمى بملك الأملاك٢٨٣٨حدثنا يعقوب بن إبراهيم الدورقي، وأبو بكر محمد بن بشار وغير واحد قالوا…٢٨٣٩حدثنا أبو بكر بن نافع البصري، حدثنا عمر بن علي المقدمي، عن هشام بن…٢٨٤٠إن لي أسماء أنا محمد وأنا أحمد وأنا الماحي الذي يمحو الله بي الكفر…٢٨٤١لا تكتنوا بكنيتي٢٨٤٢إذا سميتم باسمي فلا تكتنوا بي٢٨٤٣أنه قال يا رسول الله أرأيت إن ولد لي بعدك أسميه محمدا وأكنيه بكنيتك…٢٨٤٤إن من الشعر حكمة٢٨٤٥إن من الشعر حكما٢٨٤٦إن الله يؤيد حسان بروح القدس ما يفاخر أو ينافح عن رسول الله صلى الله…٢٨٤٦.٢وعلي بن حجر، قالا حدثنا ابن أبي الزناد، عن أبيه، عن عروة، عن عائشة،…٢٨٤٧خل عنه يا عمر فلهي أسرع فيهم من نضح النبل٢٨٤٨ويأتيك بالأخبار من لم تزود٢٨٤٩أشعر كلمة تكلمت بها العرب كلمة لبيد ألا كل شيء ما خلا الله البليغ من الرجال الذي يتخلل بلسانه كما تتخلل البقرة٢٨٥٤حدثنا إسحاق بن موسى الأنصاري، حدثنا عبد الله بن وهب، عن عبد الجبار بن…٢٨٥٥حدثنا محمود بن غيلان، حدثنا أبو أحمد، حدثنا سفيان، عن الأعمش، عن أبي…٢٨٥٥.٢حديث ٢٨٥٥.٢٢٨٥٦قال سئلت عائشة وأم سلمة أى العمل كان أحب إلى رسول الله صلى الله عليه…٢٨٥٦.٢وقد روي عن هشام بن عروة، عن أبيه، عن عائشة، قالت كان أحب العمل إلى…٢٨٥٧خمروا الآنية وأوكوا الأسقية وأجيفوا الأبواب وأطفئوا المصابيح فإن…٢٨٥٨إذا سافرتم في الخصب فأعطوا الإبل حظها من الأرض وإذا سافرتم في السنة…
