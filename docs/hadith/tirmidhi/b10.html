كتاب الجنائز عن رسول الله صلى الله عليه وسلم

١١٥ حديثًا
٩٦٥لا يصيب المؤمن شوكة فما فوقها إلا رفعه الله بها درجة وحط عنه بها خطيئة٩٦٦ما من شيء يصيب المؤمن من نصب ولا حزن ولا وصب حتى الهم يهمه إلا يكفر…٩٦٧إن المسلم إذا عاد أخاه المسلم لم يزل في خرفة الجنة٩٦٨حدثنا محمد بن وزير الواسطي، حدثنا يزيد بن هارون، عن عاصم الأحول، عن…٩٦٩ما من مسلم يعود مسلما غدوة إلا صلى عليه سبعون ألف ملك حتى يمسي وإن…٩٧٠لا يتمنين أحدكم الموت لضر نزل به وليقل اللهم أحيني ما كانت الحياة…٩٧١حدثنا بذلك علي بن حجر أخبرنا إسماعيل بن إبراهيم أخبرنا عبد العزيز بن…٩٧٢أن جبريل، أتى النبي صلى الله عليه وسلم فقال يا محمد اشتكيت قال " نعم…٩٧٣اللهم رب الناس مذهب الباس اشف أنت الشافي لا شافي إلا أنت شفاء لا…٩٧٤ما حق امرئ مسلم يبيت ليلتين وله شيء يوصي فيه إلا ووصيته مكتوبة عنده٩٧٥حدثنا قتيبة، حدثنا جرير، عن عطاء بن السائب، عن أبي عبد الرحمن السلمي،…٩٧٦لقنوا موتاكم لا إله إلا الله٩٧٧إذا حضرتم المريض أو الميت فقولوا خيرا فإن الملائكة يؤمنون على ما…٩٧٨اللهم أعني على غمرات الموت٩٧٩حدثنا الحسن بن الصباح البغدادي، حدثنا مبشر بن إسماعيل الحلبي، عن عبد…٩٨٠إن نفس المؤمن تخرج رشحا ولا أحب موتا كموت الحمار٩٨١ما من حافظين رفعا إلى الله ما حفظا من ليل أو نهار فيجد الله في أول…٩٨٢المؤمن يموت بعرق الجبين٩٨٣وهارون بن عبد الله البزاز البغدادي، قالا حدثنا سيار، هو ابن حاتم…٩٨٤إياكم والنعى فإن النعى من عمل الجاهلية٩٨٥حدثنا سعيد بن عبد الرحمن المخزومي، حدثنا عبد الله بن الوليد العدني،…٩٨٦حدثنا أحمد بن منيع، حدثنا عبد القدوس بن بكر بن خنيس، حدثنا حبيب بن…٩٨٧الصبر في الصدمة الأولى٩٨٨الصبر عند الصدمة الأولى٩٨٩حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، حدثنا سفيان، عن عاصم بن…٩٩٠اغسلنها وترا ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن واغسلنها بماء وسدر…٩٩١أطيب الطيب المسك٩٩٢حدثنا سفيان بن وكيع، حدثنا أبي، عن شعبة، عن خليد بن جعفر، عن أبي…٩٩٣من غسله الغسل ومن حمله الوضوء٩٩٤البسوا من ثيابكم البياض فإنها من خير ثيابكم وكفنوا فيها موتاكم٩٩٥إذا ولي أحدكم أخاه فليحسن كفنه٩٩٦قالت كفن النبي صلى الله عليه وسلم في ثلاثة أثواب بيض يمانية ليس فيها…٩٩٧حدثنا ابن أبي عمر، حدثنا بشر بن السري، عن زائدة، عن عبد الله بن محمد…٩٩٨اصنعوا لأهل جعفر طعاما فإنه قد جاءهم ما يشغلهم٩٩٩ليس منا من شق الجيوب وضرب الخدود ودعا بدعوة الجاهلية١٠٠٠من نيح عليه عذب بما نيح عليه١٠٠١أربع في أمتي من أمر الجاهلية لن يدعهن الناس النياحة والطعن في الأحساب…١٠٠٢الميت يعذب ببكاء أهله عليه١٠٠٣ما من ميت يموت فيقوم باكيه فيقول واجبلاه واسيداه أو نحو ذلك إلا وكل…١٠٠٤الميت يعذب ببكاء أهله عليه١٠٠٥لا ولكن نهيت عن صوتين أحمقين فاجرين صوت عند مصيبة خمش وجوه وشق جيوب…١٠٠٦إنهم ليبكون عليها وإنها لتعذب في قبرها١٠٠٧وأحمد بن منيع، وإسحاق بن منصور، ومحمود بن غيلان، قالوا حدثنا سفيان بن…١٠٠٨وبكر الكوفي، وزياد، وسفيان، كلهم يذكر أنه سمعه من الزهري عن سالم بن…١٠٠٩حدثنا عبد بن حميد، حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، قال كان…١٠١٠حدثنا أبو موسى، محمد بن المثنى حدثنا محمد بن بكر، حدثنا يونس بن يزيد،…١٠١١ما دون الخبب فإن كان خيرا عجلتموه وإن كان شرا فلا يبعد إلا أهل النار…١٠١٢ألا تستحيون إن ملائكة الله على أقدامهم وأنتم على ظهور الدواب١٠١٣حدثنا محمود بن غيلان، حدثنا أبو داود، حدثنا شعبة، عن سماك بن حرب، قال…١٠١٤أن النبي صلى الله عليه وسلم اتبع جنازة أبي الدحداح ماشيا ورجع على فرس…١٠١٥أسرعوا بالجنازة فإن يكن خيرا تقدموها إليه وإن يكن شرا تضعوه عن رقابكم١٠١٦لولا أن تجد صفية في نفسها لتركته حتى تأكله العافية حتى يحشر يوم…١٠١٧حدثنا علي بن حجر، أخبرنا علي بن مسهر، عن مسلم الأعور، عن أنس بن مالك،…١٠١٨ما قبض الله نبيا إلا في الموضع الذي يحب أن يدفن فيه١٠١٩اذكروا محاسن موتاكم وكفوا عن مساويهم١٠٢٠قال كان رسول الله صلى الله عليه وسلم إذا اتبع الجنازة لم يقعد حتى…١٠٢١إذا مات ولد العبد قال الله لملائكته قبضتم ولد عبدي . فيقولون نعم .…١٠٢٢حدثنا أحمد بن منيع، حدثنا إسماعيل بن إبراهيم، حدثنا معمر، عن الزهري،…١٠٢٣حدثنا محمد بن المثنى، حدثنا محمد بن جعفر، أخبرنا شعبة، عن عمرو بن…١٠٢٤اللهم اغفر لحينا وميتنا وشاهدنا وغائبنا وصغيرنا وكبيرنا وذكرنا وأنثانا١٠٢٥اللهم اغفر له وارحمه واغسله بالبرد واغسله كما يغسل الثوب١٠٢٦حدثنا أحمد بن منيع، حدثنا زيد بن حباب، حدثنا إبراهيم بن عثمان، عن…١٠٢٧أن ابن عباس، صلى على جنازة فقرأ بفاتحة الكتاب فقلت له فقال إنه من…١٠٢٨من صلى عليه ثلاثة صفوف فقد أوجب١٠٢٩لا يموت أحد من المسلمين فتصلي عليه أمة من المسلمين يبلغون أن يكونوا…١٠٣٠حدثنا هناد، حدثنا وكيع، عن موسى بن على بن رباح، عن أبيه، عن عقبة بن…١٠٣١الراكب خلف الجنازة والماشي حيث شاء منها والطفل يصلى عليه١٠٣٢الطفل لا يصلى عليه ولا يرث ولا يورث حتى يستهل١٠٣٣قالت صلى رسول الله صلى الله عليه وسلم على سهيل ابن بيضاء في المسجد .…١٠٣٤حدثنا عبد الله بن منير، عن سعيد بن عامر، عن همام، عن أبي غالب، قال…١٠٣٥والفضل بن موسى، عن حسين المعلم، عن عبد الله بن بريدة، عن سمرة بن…١٠٣٦أيهما أكثر أخذا للقرآن١٠٣٧حدثنا أحمد بن منيع، حدثنا هشيم، أخبرنا الشيباني، حدثنا الشعبي، أخبرني…١٠٣٨أن أم سعد، ماتت والنبي صلى الله عليه وسلم غائب فلما قدم صلى عليها وقد…١٠٣٩إن أخاكم النجاشي قد مات فقوموا فصلوا عليه١٠٤٠من صلى على جنازة فله قيراط ومن تبعها حتى يقضى دفنها فله قيراطان…١٠٤١من تبع جنازة وحملها ثلاث مرات فقد قضى ما عليه من حقها١٠٤٢إذا رأيتم الجنازة فقوموا لها حتى تخلفكم أو توضع١٠٤٣إذا رأيتم الجنازة فقوموا لها فمن تبعها فلا يقعدن حتى توضع١٠٤٤إذا رأيتم الجنازة فقوموا١٠٤٥اللحد لنا والشق لغيرنا١٠٤٦بسم الله وبالله وعلى ملة رسول الله١٠٤٧حدثنا زيد بن أخزم الطائي البصري، حدثنا عثمان بن فرقد، قال سمعت جعفر…١٠٤٨ويحيى، عن شعبة، عن أبي جمرة، عن ابن عباس، . وهذا أصح . قال أبو عيسى…١٠٤٩أن لا�� تدع قبرا مشرفا إلا سويته ولا تمثالا إلا طمسته١٠٥٠لا تجلسوا على القبور ولا تصلوا إليها١٠٥١حدثنا علي بن حجر، وأبو عمار قالا أخبرنا الوليد بن مسلم، عن عبد الرحمن…١٠٥٢حدثنا عبد الرحمن بن الأسود أبو عمرو البصري، حدثنا محمد بن ربيعة، عن…١٠٥٣السلام عليكم يا أهل القبور يغفر الله لنا ولكم أنتم سلفنا ونحن بالأثر١٠٥٤قد كنت نهيتكم عن زيارة القبور فقد أذن لمحمد في زيارة قبر أمه فزوروها…١٠٥٥حدثنا الحسين بن حريث، حدثنا عيسى بن يونس، عن ابن جريج، عن عبد الله بن…١٠٥٦حدثنا قتيبة، حدثنا أبو عوانة، عن عمر بن أبي سلمة، عن أبيه، عن أبي…١٠٥٧رحمك الله إن كنت لأواها تلاء للقرآن١٠٥٨حدثنا أحمد بن منيع، حدثنا يزيد بن هارون، أخبرنا حميد، عن أنس، قال مر…١٠٥٩ما من مسلم يشهد له ثلاثة إلا وجبت له الجنة١٠٦٠لا يموت لأحد من المسلمين ثلاثة من الولد فتمسه النار إلا تحلة القسم١٠٦١من قدم ثلاثة لم يبلغوا الحلم كانوا له حصنا حصينا من النار١٠٦٢من كان له فرطان من أمتي أدخله الله بهما الجنة١٠٦٣الشهداء خمس المطعون والمبطون والغرق وصاحب الهدم والشهيد في سبيل الله١٠٦٤من قتله بطنه لم يعذب في قبره١٠٦٥بقية رجز - أو عذاب أرسل على طائفة من بني إسرائيل فإذا وقع بأرض وأنتم…١٠٦٦من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه١٠٦٧من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه١٠٦٨وشريك، عن سماك بن حرب، عن جابر بن سمرة، أن رجلا، قتل نفسه فلم يصل…١٠٦٩صلوا على صاحبكم فإن عليه دينا١٠٧٠هل ترك لدينه من قضاء١٠٧١إذا قبر الميت - أو قال أحدكم أتاه ملكان أسودان أزرقان يقال لأحدهما…١٠٧٢إذا مات الميت عرض عليه مقعده بالغداة والعشي فإن كان من أهل الجنة فمن…١٠٧٣من عزى مصابا فله مثل أجره١٠٧٤ما من مسلم يموت يوم الجمعة أو ليلة الجمعة إلا وقاه الله فتنة القبر١٠٧٥يا علي ثلاث لا تؤخرها الصلاة إذا آنت والجنازة إذا حضرت والأيم إذا…١٠٧٦من عزى ثكلى كسي بردا في الجنة١٠٧٧حدثنا القاسم بن دينار الكوفي، حدثنا إسماعيل بن
