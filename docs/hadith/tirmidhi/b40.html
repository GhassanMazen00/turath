كتاب الإيمان عن رسول الله صلى الله عليه وسلم

٣٩ حديثًا
٢٦٠٦أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوها منعوا مني…٢٦٠٧أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله ومن قال لا إله إلا…٢٦٠٨أمرت أن أقاتل الناس حتى يشهدوا أن لا إله إلا الله وأن محمدا عبده…٢٦٠٩بني الإسلام على خمس شهادة أن لا إله إلا الله وأن محمدا رسول الله…٢٦١٠أن تؤمن بالله وملائكته وكتبه ورسله واليوم الآخر والقدر خيره وشره٢٦١١آمركم بأربع الإيمان بالله ثم فسرها لهم شهادة أن لا إله إلا الله وأني…٢٦١٢إن من أكمل المؤمنين إيمانا أحسنهم خلقا وألطفهم بأهله٢٦١٣يا معشر النساء تصدقن فإنكن أكثر أهل النار٢٦١٤الإيمان بضع وسبعون بابا فأدناها إماطة الأذى عن الطريق وأرفعها قول لا…٢٦١٥الحياء من الإيمان٢٦١٦لقد سألتني عن عظيم وإنه ليسير على من يسره الله عليه تعبد الله ولا…٢٦١٧إذا رأيتم الرجل يتعاهد المسجد فاشهدوا له بالإيمان فإن الله تعالى يقول…٢٦١٨بين الكفر والإيمان ترك الصلاة٢٦١٩بين العبد وبين الشرك أو الكفر ترك الصلاة٢٦٢٠بين العبد وبين الكفر ترك الصلاة٢٦٢١العهد الذي بيننا وبينهم الصلاة فمن تركها فقد كفر٢٦٢٢حدثنا قتيبة، حدثنا بشر بن المفضل، عن الجريري، عن عبد الله بن شقيق…٢٦٢٣ذاق طعم الإيمان من رضي بالله ربا وبالإسلام دينا وبمحمد نبيا٢٦٢٤ثلاث من كن فيه وجد بهن طعم الإيمان من كان الله ورسوله أحب إليه مما…٢٦٢٥لا يزني الزاني حين يزني وهو مؤمن ولا يسرق السارق حين يسرق وهو مؤمن…٢٦٢٦من أصاب حدا فعجل عقوبته في الدنيا فالله أعدل من أن يثني على عبده…٢٦٢٧المسلم من سلم المسلمون من لسانه ويده والمؤمن من أمنه الناس على دمائهم…٢٦٢٨من سلم المسلمون من لسانه ويده٢٦٢٩إن الإسلام بدأ غريبا وسيعود غريبا كما بدأ فطوبى للغرباء٢٦٣٠إن الدين ليأرز إلى الحجاز كما تأرز الحية إلى جحرها وليعقلن الدين من…٢٦٣١آية المنافق ثلاث إذا حدث كذب وإذا وعد أخلف وإذا اؤتمن خان٢٦٣٢أربع من كن فيه كان منافقا وإن كانت خصلة منهن فيه كانت فيه خصلة من…٢٦٣٣إذا وعد الرجل وينوي أن يفي به فلم يف به فلا جناح عليه٢٦٣٤قتال المسلم أخاه كفر وسبابه فسوق٢٦٣٥سباب المسلم فسوق وقتاله كفر٢٦٣٦ليس على العبد نذر فيما لا يملك ولاعن المؤمن كقاتله ومن قذف مؤمنا بكفر…٢٦٣٧أيما رجل قال لأخيه كافر فقد باء به أحدهما٢٦٣٨من شهد أن لا إله إلا الله وأن محمدا رسول الله حرم الله عليه النار٢٦٣٩إن الله سيخلص رجلا من أمتي على رءوس الخلائق يوم القيامة فينشر عليه…٢٦٤٠تفرقت اليهود على إحدى وسبعين أو اثنتين وسبعين فرقة والنصارى مثل ذلك…٢٦٤١ليأتين على أمتي ما أتى على بني إسرائيل حذو النعل بالنعل حتى إن كان…٢٦٤٢إن الله عز وجل خلق خلقه في ظلمة فألقى عليهم من نوره فمن أصابه من ذلك…٢٦٤٣أتدري ما حق الله على العباد٢٦٤٤أتاني جبريل فبشرني فأخبرني أنه من مات لا يشرك بالله شيئا دخل الجنة
