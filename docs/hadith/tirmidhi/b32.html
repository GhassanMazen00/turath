كتاب القدر عن رسول الله صلى الله عليه وسلم

٢٦ حديثًا
٢١٣٣أبهذا أمرتم أم بهذا أرسلت إليكم إنما هلك من كان قبلكم حين تنازعوا في…٢١٣٤احتج آدم وموسى فقال موسى يا آدم أنت الذي خلقك الله بيده ونفخ فيك من…٢١٣٥فيما قد فرغ منه يا ابن الخطاب وكل ميسر أما من كان من أهل السعادة فإنه…٢١٣٦ما منكم من أحد إلا قد علم وقال وكيع إلا قد كتب مقعده من النار ومقعده…٢١٣٧إن أحدكم يجمع خلقه في بطن أمه في أربعين يوما ثم يكون علقة مثل ذلك ثم…٢١٣٨كل مولود يولد على الملة فأبواه يهودانه أو ينصرانه أو يشركانه٢١٣٩لا يرد القضاء إلا الدعاء ولا يزيد في العمر إلا البر٢١٤٠يا مقلب القلوب ثبت قلبي على دينك٢١٤١أتدرون ما هذان الكتابان٢١٤٢إذا أراد الله بعبد خيرا استعمله٢١٤٣لا يعدي شيء شيئا٢١٤٤لا يؤمن عبد حتى يؤمن بالقدر خيره وشره حتى يعلم أن ما أصابه لم يكن…٢١٤٥لا يؤمن عبد حتى يؤمن بأربع يشهد أن لا إله إلا الله وأني محمد رسول…٢١٤٦إذا قضى الله لعبد أن يموت بأرض جعل له إليها حاجة٢١٤٧إذا قضى الله لعبد أن يموت بأرض جعل له إليها حاجة أو قال بها حاجة٢١٤٨أن رجلا، أتى النبي صلى الله عليه وسلم فقال يا رسول الله أرأيت رقى…٢١٤٩صنفان من أمتي ليس لهما في الإسلام نصيب المرجئة والقدرية٢١٥٠مثل ابن آدم وإلى جنبه تسع وتسعون منية إن أخطأته المنايا وقع في الهرم…٢١٥١من سعادة ابن آدم رضاه بما قضى الله له ومن شقاوة ابن آدم تركه استخارة…٢١٥٢يكون في هذه الأمة أو في أمتي الشك منه خسف أو مسخ أو قذف في أهل القدر٢١٥٣يكون في أمتي خسف ومسخ وذلك في المكذبين بالقدر٢١٥٤ستة لعنتهم لعنهم الله وكل نبي كان الزائد في كتاب الله والمكذب بقدر…٢١٥٥إن أول ما خلق الله القلم فقال اكتب . فقال ما أكتب قال اكتب القدر ما…٢١٥٦قدر الله المقادير قبل أن يخلق السموات والأرض بخمسين ألف سنة٢١٥٧حدثنا أبو كريب، محمد بن العلاء ومحمد بن بشار قالا حدثنا وكيع، عن…٢٢٩٨لا تجوز شهادة خائن ولا خائنة ولا مجلود حدا ولا مجلودة ولا ذي غمر…
