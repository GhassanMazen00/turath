كتاب الزهد عن رسول الله صلى الله عليه وسلم

١١١ حديثًا
٢٣٠٤نعمتان مغبون فيهما كثير من الناس الصحة والفراغ٢٣٠٥من يأخذ عني هؤلاء الكلمات فيعمل بهن أو يعلم من يعمل بهن٢٣٠٦بادروا بالأعمال سبعا هل تنظرون إلا فقرا منسيا أو غنى مطغيا أو مرضا…٢٣٠٧أكثروا ذكر هاذم اللذات٢٣٠٨إن القبر أول منازل الآخرة فإن نجا منه فما بعده أيسر منه وإن لم ينج…٢٣٠٩من أحب لقاء الله أحب الله لقاءه ومن كره لقاء الله كره الله لقاءه٢٣١٠يا صفية بنت عبد المطلب يا فاطمة بنت محمد يا بني عبد المطلب إني لا…٢٣١١لا يلج النار رجل بكى من خشية الله حتى يعود اللبن في الضرع ولا يجتمع…٢٣١٢إني أرى ما لا ترون وأسمع ما لا تسمعون أطت السماء وحق لها أن تئط ما…٢٣١٣لو تعلمون ما أعلم لضحكتم قليلا ولبكيتم كثيرا٢٣١٤إن الرجل ليتكلم بالكلمة لا يرى بها بأسا يهوي بها سبعين خريفا في النار٢٣١٥ويل للذي يحدث بالحديث ليضحك به القوم فيكذب ويل له ويل له٢٣١٦أولا تدري فلعله تكلم فيما لا يعنيه أو بخل بما لا ينقصه٢٣١٧من حسن إسلام المرء تركه ما لا يعنيه٢٣١٨إن من حسن إسلام المرء تركه ما لا يعنيه٢٣١٩إن أحدكم ليتكلم بالكلمة من رضوان الله ما يظن أن تبلغ ما بلغت فيكتب…٢٣٢٠لو كانت الدنيا تعدل عند الله جناح بعوضة ما سقى كافرا منها شربة ماء٢٣٢١أترون هذه هانت على أهلها حين ألقوها٢٣٢٢ألا إن الدنيا ملعونة ملعون ما فيها إلا ذكر الله وما والاه وعالما أو…٢٣٢٣ما الدنيا في الآخرة إلا مثل ما يجعل أحدكم إصبعه في اليم فلينظر بماذا…٢٣٢٤الدنيا سجن المؤمن وجنة الكافر٢٣٢٥ثلاثة أقسم عليهن وأحدثكم حديثا فاحفظوه٢٣٢٦من نزلت به فاقة فأنزلها بالناس لم تسد فاقته ومن نزلت به فاقة فأنزلها…٢٣٢٧إنما يكفيك من جمع المال خادم ومركب في سبيل الله٢٣٢٨لا تتخذوا الضيعة فترغبوا في الدنيا٢٣٢٩من طال عمره وحسن عمله٢٣٣٠من طال عمره وحسن عمله٢٣٣١عمر أمتي من ستين سنة إلى سبعين سنة٢٣٣٢لا تقوم الساعة حتى يتقارب الزمان فتكون السنة كالشهر والشهر كالجمعة…٢٣٣٣كن في الدنيا كأنك غريب أو عابر سبيل وعد نفسك في أهل القبور٢٣٣٤هذا ابن آدم وهذا أجله٢٣٣٥حدثنا هناد، حدثنا أبو معاوية، عن الأعمش، عن أبي السفر، عن عبد الله بن…٢٣٣٦إن لكل أمة فتنة وفتنة أمتي المال٢٣٣٧لو كان لابن آدم واديا من ذهب لأحب أن يكون له ثانيا ولا يملأ فاه إلا…٢٣٣٨قلب الشيخ شاب على حب اثنتين طول الحياة وكثرة المال٢٣٣٩يهرم ابن آدم ويشب منه اثنتان الحرص على العمر والحرص على المال٢٣٤٠الزهادة في الدنيا ليست بتحريم الحلال ولا إضاعة المال ولكن الزهادة في…٢٣٤١ليس لابن آدم حق في سوى هذه الخصال بيت يسكنه وثوب يواري عورته وجلف…٢٣٤٢يقول ابن آدم مالي مالي وهل لك من مالك إلا ما تصدقت فأمضيت أو أكلت…٢٣٤٣يا ابن آدم إنك إن تبذل الفضل خير لك وإن تمسكه شر لك ولا تلام على كفاف…٢٣٤٤لو أنكم كنتم توكلون على الله حق توكله لرزقتم كما ترزق الطير تغدو…٢٣٤٥قال كان أخوان على عهد النبي صلى الله عليه وسلم فكان أحدهما يأتي النبي…٢٣٤٦حدثنا عمرو بن مالك ومحمود بن خداش البغدادي قالا حدثنا مروان بن معاوية…٢٣٤٧إن أغبط أوليائي عندي لمؤمن خفيف الحاذ ذو حظ من الصلاة أحسن عبادة ربه…٢٣٤٨قد أفلح من أسلم وكان رزقه كفافا وقنعه الله٢٣٤٩طوبى لمن هدي إلى الإسلام وكان عيشه كفافا وقنع٢٣٥٠نحوه بمعناه . قال أبو عيسى هذا حديث حسن غريب . وأبو الوازع الراسبي…٢٣٥١فقراء المهاجرين يدخلون الجنة قبل أغنيائهم بخمسمائة سنة٢٣٥٢اللهم أحيني مسكينا وأمتني مسكينا واحشرني في زمرة المساكين يوم القيامة٢٣٥٣يدخل الفقراء الجنة قبل الأغنياء بخمسمائة عام نصف يوم٢٣٥٤يدخل فقراء المسلمين الجنة قبل أغنيائهم بنصف يوم وهو خمسمائة عام٢٣٥٥يدخل فقراء المسلمين الجنة قبل أغنيائهم بأربعين خريفا٢٣٥٦قال دخلت على عائشة فدعت لي بطعام وقالت ما أشبع من طعام فأشاء أن أبكي…٢٣٥٧حدثنا محمود بن غيلان، حدثنا أبو داود، أنبأنا شعبة، عن أبي إسحاق، قال…٢٣٥٨قال ما شبع رسول الله صلى الله عليه وسلم وأهله ثلاثا تباعا من خبز البر…٢٣٥٩بيت النبي صلى الله عليه وسلم خبز الشعير . قال أبو عيسى هذا حديث حسن…٢٣٦٠قال كان رسول الله صلى الله عليه وسلم يبيت الليالي المتتابعة طاويا…٢٣٦١اللهم اجعل رزق آل محمد قوتا٢٣٦٢حدثنا قتيبة، حدثنا جعفر بن سليمان، عن ثابت، عن أنس، قال كان النبي صلى…٢٣٦٣قال ما أكل رسول الله صلى الله عليه وسلم على خوان ولا أكل خبزا مرققا…٢٣٦٤حدثنا عبد الله بن عبد الرحمن، أخبرنا عبيد الله بن عبد المجيد الحنفي،…٢٣٦٥حدثنا عمر بن إسماعيل بن مجالد بن سعيد، حدثنا أبي، عن بيان، عن قيس بن…٢٣٦٦حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، حدثنا إسماعيل بن أبي خالد،…٢٣٦٧حدثنا قتيبة، حدثنا حماد بن زيد، عن أيوب، عن محمد بن سيرين، قال كنا…٢٣٦٨لو تعلمون ما لكم عند الله لأحببتم أن تزدادوا فاقة وحاجة٢٣٦٩ما جاء بك يا أبا بكر٢٣٧٠حدثنا صالح بن عبد الله، حدثنا أبو عوانة، عن عبد الملك بن عمير، عن أبي…٢٣٧١حدثنا عبد الله بن أبي زياد، حدثنا سيار بن حاتم، عن سهل بن أسلم، عن…٢٣٧٢حدثنا قتيبة، حدثنا أبو الأحوص، عن سماك بن حرب، قال سمعت النعمان بن…٢٣٧٣ليس الغنى عن كثرة العرض ولكن الغنى غنى النفس٢٣٧٤إن هذا المال خضرة حلوة من أصابه بحقه بورك له فيه ورب متخوض فيما شاءت…٢٣٧٥لعن عبد الدينار لعن عبد الدرهم٢٣٧٦ما ذئبان جائعان أرسلا في غنم بأفسد لها من حرص المرء على المال والشرف…٢٣٧٧ما لي وما للدنيا ما أنا في الدنيا إلا كراكب استظل تحت شجرة ثم راح…٢٣٧٨الرجل على دين خليله فلينظر أحدكم من يخالل٢٣٧٩يتبع الميت ثلاث فيرجع اثنان ويبقى واحد يتبعه أهله وماله وعمله فيرجع…٢٣٨٠ما ملأ آدمي وعاء شرا من بطن بحسب ابن آدم أكلات يقمن صلبه فإن كان لا…٢٣٨١من يرائي يرائي الله به ومن يسمع يسمع الله به٢٣٨٢. ثم ضرب رسول الله صلى الله عليه وسلم على ركبتي فقال٢٣٨٣تعوذوا بالله من جب الحزن٢٣٨٤له أجران أجر السر وأجر العلانية٢٣٨٥أين السائل عن قيام الساعة٢٣٨٦المرء مع من أحب وله ما اكتسب٢٣٨٧المرء مع من أحب٢٣٨٨إن الله يقول أنا عند ظن عبدي في وأنا معه إذا دعاني٢٣٨٩البر حسن الخلق والإثم ما حاك في نفسك وكرهت أن يطلع عليه الناس٢٣٩٠قال الله عز وجل المتحابون في جلالي لهم منابر من نور يغبطهم النبيون…٢٣٩١سبعة يظلهم الله في ظله يوم لا ظل إلا ظله إمام عادل وشاب نشأ بعبادة…٢٣٩٢حديث ٢٣٩٢٢٣٩٣حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، حدثنا سفيان، عن حبيب بن…٢٣٩٤حدثنا محمد بن عثمان الكوفي، حدثنا عبيد الله بن موسى، عن سالم الخياط،…٢٣٩٥لا تصاحب إلا مؤمنا ولا يأكل طعامك إلا تقي٢٣٩٦إذا أراد الله بعبده الخير عجل له العقوبة في الدنيا وإذا أراد الله…٢٣٩٧يقول قالت عائشة ما رأيت الوجع على أحد أشد منه على رسول الله صلى الله…٢٣٩٨الأنبياء ثم الأمثل فالأمثل فيبتلى الرجل على حسب دينه فإن كان دينه…٢٣٩٩ما يزال البلاء بالمؤمن والمؤمنة في نفسه وولده وماله حتى يلقى الله وما…٢٤٠٠إن الله يقول إذا أخذت كريمتى عبدي في الدنيا لم يكن له جزاء عندي إلا…٢٤٠١يقول الله عز وجل من أذهبت حبيبتيه فصبر واحتسب لم أرض له ثوابا دون…٢٤٠٢يود أهل العافية يوم القيامة حين يعطى أهل البلاء الثواب لو أن جلودهم…٢٤٠٣ما من أحد يموت إلا ندم٢٤٠٤يخرج في آخر الزمان رجال يختلون الدنيا بالدين يلبسون للناس جلود الضأن…٢٤٠٥إن الله تعالى قال لقد خلقت خلقا ألسنتهم أحلى من العسل وقلوبهم أمر من…٢٤٠٦أمسك عليك لسانك وليسعك بيتك وابك على خطيئتك٢٤٠٧إذا أصبح ابن آدم فإن الأعضاء كلها تكفر اللسان فتقول اتق الله فينا…٢٤٠٨من يتكفل لي ما بين لحييه وما بين رجليه أتكفل له بالجنة٢٤٠٩من وقاه الله شر ما بين لحييه وشر ما بين رجليه دخل الجنة٢٤١٠قل ربي الله ثم استقم٢٤١١لا تكثروا الكلام بغير ذكر الله فإن كثرة الكلام بغير ذكر الله قسوة…٢٤١٢كل كلام ابن آدم عليه لا له إلا أمر بمعروف أو نهى عن منكر أو ذكر الله٢٤١٣حدثنا محمد بن بشار، حدثنا جعفر بن عون، حدثنا أبو العميس، عن عون بن…٢٤١٤من التمس رضاء الله بسخط الناس كفاه الله مؤنة الناس ومن التمس رضاء…
