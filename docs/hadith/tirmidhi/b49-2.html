كتاب المناقب عن رسول الله صلى الله عليه وسلم

٣٥١ حديثًا•صفحة ٢ من ٣
٣٧٥٦ليت رجلا صالحا يحرسني الليلة٣٧٥٧اثبت حراء فإنه ليس عليك إلا نبي أو صديق أو شهيد٣٧٥٨حديث ٣٧٥٨٣٧٥٩حديث ٣٧٥٩٣٧٦٠إن عم الرجل صنو أبيه٣٧٦١العباس عم رسول الله وإن عم الرجل صنو أبيه أو من صنو أبيه٣٧٦٢إذا كان غداة الاثنين فأتني أنت وولدك حتى أدعو لهم بدعوة ينفعك الله…٣٧٦٣رأيت جعفرا يطير في الجنة مع الملائكة٣٧٦٤قال ما احتذى النعال ولا انتعل ولا ركب المطايا ولا ركب الكور بعد رسول…٣٧٦٥أشبهت خلقي وخلقي٣٧٦٦حدثنا أبو سعيد الأشج، قال حدثنا إسماعيل بن إبراهيم أبو يحيى التيمي،…٣٧٦٧حدثنا أبو أحمد حاتم بن سياه المروزي حدثنا عبد الرزاق، أخبرنا معمر، عن…٣٧٦٨الحسن والحسين سيدا شباب أهل الجنة٣٧٦٩هذان ابناى وابنا ابنتي اللهم إني أحبهما فأحبهما وأحب من يحبهما٣٧٧٠إن الحسن والحسين هما ريحانتاى من الدنيا٣٧٧١شهدت قتل الحسين آنفا٣٧٧٢أنه سمع أنس بن مالك، يقول سئل رسول الله صلى الله عليه وسلم أى أهل…٣٧٧٣إن ابني هذا سيد يصلح الله على يديه فئتين عظيمتين٣٧٧٤صدق الله : ( إنما أموالكم وأولادكم فتنة ) نظرت إلى هذين الصبيين…٣٧٧٥حسين مني وأنا من حسين أحب الله من أحب حسينا حسين سبط من الأسباط٣٧٧٦قال لم يكن أحد منهم أشبه برسول الله من الحسن بن علي . هذا حديث حسن…٣٧٧٧حدثنا محمد بن بشار، قال حدثنا يحيى بن سعيد، قال حدثنا إسماعيل بن أبي…٣٧٧٨حدثنا خلاد بن أسلم أبو بكر البغدادي، قال حدثنا النضر بن شميل، قال…٣٧٧٩قال الحسن أشبه برسول الله صلى الله عليه وسلم ما بين الصدر إلى الرأس…٣٧٨٠قال لما جيء برأس عبيد الله بن زياد وأصحابه نضدت في المسجد في الرحبة…٣٧٨١وإسحاق بن منصور، قالا أخبرنا محمد بن يوسف، عن إسرائيل، عن ميسرة بن…٣٧٨٢اللهم إني أحبهما فأحبهما٣٧٨٣اللهم إني أحبه فأحبه٣٧٨٤قال كان رسول الله صلى الله عليه وسلم حامل الحسن بن علي على عاتقه فقال…٣٧٨٥إن كل نبي أعطي سبعة نجباء أو نقباء وأعطيت أنا أربعة عشر٣٧٨٦يا أيها الناس إني قد تركت فيكم ما إن أخذتم به لن تضلوا كتاب الله…٣٧٨٧اللهم هؤلاء أهل بيتي فأذهب عنهم الرجس وطهرهم تطهيرا٣٧٨٨إني تارك فيكم ما إن تمسكتم به لن تضلوا بعدي أحدهما أعظم من الآخر كتاب…٣٧٨٩أحبوا الله لما يغذوكم من نعمه وأحبوني بحب الله وأحبوا أهل بيتي لحبي٣٧٩٠أرحم أمتي بأمتي أبو بكر وأشدهم في أمر الله عمر وأصدقهم حياء عثمان…٣٧٩١أرحم أمتي بأمتي أبو بكر وأشدهم في أمر الله عمر وأصدقهم حياء عثمان…٣٧٩٢إن الله أمرني أن أقرأ عليك : ( لم يكن الذين كفروا )٣٧٩٣إن الله أمرني أن أقرأ عليك٣٧٩٤قال جمع القرآن على عهد رسول الله صلى الله عليه وسلم أربعة كلهم من…٣٧٩٥نعم الرجل أبو بكر نعم الرجل عمر نعم الرجل أبو عبيدة بن الجراح نعم…٣٧٩٦فإني سأبعث معكم أمينا حق أمين٣٧٩٧إن الجنة تشتاق إلى ثلاثة علي وعمار وسلمان٣٧٩٨ائذنوا له مرحبا بالطيب المطيب٣٧٩٩حديث ٣٧٩٩٣٨٠٠أبشر عمار تقتلك الفئة الباغية٣٨٠١ما أظلت الخضراء ولا أقلت الغبراء أصدق من أبي ذر٣٨٠٢ما أظلت الخضراء ولا أقلت الغبراء من ذي لهجة أصدق ولا أوفى من أبي ذر…٣٨٠٣حدثنا علي بن سعيد الكندي، قال حدثنا أبو محياة يحيى بن يعلى بن عطاء عن…٣٨٠٤إنه عاشر عشرة في الجنة٣٨٠٥اقتدوا باللذين من بعدي من أصحابي أبي بكر وعمر واهتدوا بهدى عمار…٣٨٠٦أنه سمع أبا موسى، يقول لقد قدمت أنا وأخي، من اليمن وما نرى حينا إلا…٣٨٠٧حدثنا محمد بن بشار، قال حدثنا عبد الرحمن بن مهدي، قال حدثنا إسرائيل،…٣٨٠٨لو كنت مؤمرا أحدا من غير مشورة منهم لأمرت عليهم ابن أم عبد٣٨٠٩لو كنت مؤمرا أحدا من غير مشورة لأمرت ابن أم عبد٣٨١٠خذوا القرآن من أربعة من ابن مسعود وأبى بن كعب ومعاذ بن جبل وسالم مولى…٣٨١١قال أتيت المدينة فسألت الله أن ييسر لي جليسا صالحا فيسر لي أبا هريرة…٣٨١٢إن استخلفت عليكم فعصيتموه عذبتم ولكن ما حدثكم حذيفة فصدقوه وما أقرأكم…٣٨١٣أنه فرض لأسامة بن زيد في ثلاثة آلاف وخمسمائة وفرض لعبد الله بن عمر في…٣٨١٤حدثنا قتيبة، قال حدثنا يعقوب بن عبد الرحمن، عن موسى بن عقبة، عن سالم…٣٨١٥وغير، واحد، قالوا حدثنا محمد بن عمر بن الرومي، قال حدثنا علي بن مسهر،…٣٨١٦إن تطعنوا في إمارته فقد كنتم تطعنون في إمرة أبيه من قبل وايم الله إن…٣٨١٧قال لما ثقل رسول الله صلى الله عليه وسلم هبطت وهبط الناس المدينة…٣٨١٨يا عائشة أحبيه فإني أحبه٣٨١٩أتدري ما جاء بهما٣٨٢٠قال ما حجبني رسول الله صلى الله عليه وسلم منذ أسلمت ولا رآني إلا ضحك…٣٨٢١قال ما حجبني رسول الله صلى الله عليه وسلم منذ أسلمت ولا رآني إلا تبسم…٣٨٢٢ومحمود بن غيلان، قالا حدثنا أبو أحمد، عن سفيان، عن ليث، عن أبي جهضم،…٣٨٢٣حدثنا محمد بن حاتم المكتب المؤدب، قال حدثنا القاسم بن مالك المزني، عن…٣٨٢٤اللهم علمه الحكمة٣٨٢٥إن أخاك رجل صالح٣٨٢٦يا عائشة ما أرى أسماء إلا قد نفست فلا تسموه حتى أسميه٣٨٢٧حدثنا قتيبة، قال حدثنا جعفر بن سليمان، عن الجعد أبي عثمان، عن أنس بن…٣٨٢٨قال ربما قال لي النبي صلى الله عليه وسلم " يا ذا الأذنين " . قال أبو…٣٨٢٩اللهم أكثر ماله وولده وبارك له فيما أعطيته٣٨٣٠حدثنا زيد بن أخزم الطائي، قال حدثنا أبو داود، عن شعبة، عن جابر، عن…٣٨٣١حدثنا إبراهيم بن يعقوب، قال حدثنا زيد بن حباب، قال حدثنا ميمون أبو…٣٨٣٢حدثنا أبو كريب، قال حدثنا زيد بن حباب، عن ميمون أبي عبد الله، عن…٣٨٣٣قال قلت لأبي العالية سمع أنس، من النبي صلى الله عليه وسلم قال خدمه…٣٨٣٤قال أتيت النبي صلى الله عليه وسلم فبسطت ثوبي عنده ثم أخذه فجمعه على…٣٨٣٥حدثنا أبو موسى، محمد بن المثنى قال حدثنا عثمان بن عمر، قال حدثنا ابن…٣٨٣٦حدثنا أحمد بن منيع، قال حدثنا هشيم، قال أخبرنا يعلى بن عطاء، عن…٣٨٣٧حدثنا عبد الله بن عبد الرحمن، قال أخبرنا أحمد بن أبي شعيب الحراني،…٣٨٣٨قال قال النبي صلى الله عليه وسلم " ممن أنت " . قال قلت من دوس . قال "…٣٨٣٩خذهن واجعلهن في مزودك هذا أو في هذا المزود كلما أردت أن تأخذ منه شيئا…٣٨٤٠قال قلت لأبي هريرة لم كنيت أبا هريرة قال أما تفرق مني قلت بلى والله…٣٨٤١حدثنا قتيبة، قال حدثنا سفيان بن عيينة، عن عمرو بن دينار، عن وهب بن…٣٨٤٢اللهم اجعله هاديا مهديا واهد به٣٨٤٣حدثنا محمد بن يحيى، قال حدثنا عبد الله بن محمد النفيلي، قال حدثنا…٣٨٤٤أسلم الناس وآمن عمرو بن العاصي٣٨٤٥إن عمرو بن العاصي من صالحي قريش٣٨٤٦من هذا يا أبا هريرة٣٨٤٧تعجبون من هذا لمناديل سعد بن معاذ في الجنة أحسن من هذا٣٨٤٨اهتز له عرش الرحمن٣٨٤٩إن الملائكة كانت تحمله٣٨٥٠نحوه ولم يذكر فيه قول الأنصاري .٣٨٥١قال جاءني رسول الله صلى الله عليه وسلم ليس براكب بغل ولا برذون . قال…٣٨٥٢حدثنا ابن أبي عمر، قال حدثنا بشر بن السري، عن حماد بن سلمة، عن أبي…٣٨٥٣غطوا رأسه واجعلوا على رجليه الإذخر٣٨٥٤كم من أشعث أغبر ذي طمرين لا يؤبه له لو أقسم على الله لأبره منهم…٣٨٥٥يا أبا موسى لقد أعطيت مزمارا من مزامير آل داود٣٨٥٦اللهم لا عيش إلا عيش الآخرة فاغفر للأنصار والمهاجرة٣٨٥٧اللهم لا عيش إلا عيش الآخرة فأكرم الأنصار والمهاجرة٣٨٥٨لا تمس النار مسلما رآني أو رأى من رآني٣٨٥٩خير الناس قرني ثم الذين يلونهم ثم الذين يلونهم ثم يأتي قوم من بعد ذلك…٣٨٦٠لا يدخل النار أحد ممن بايع تحت الشجرة٣٨٦١لا تسبوا أصحابي فوالذي نفسي بيده لو أن أحدكم أنفق مثل أحد ذهبا ما…٣٨٦٢الله الله في أصحابي الله الله في أصحابي لا تتخذوهم غرضا بعدي فمن…٣٨٦٣ليدخلن الجنة من بايع تحت الشجرة إلا صاحب الجمل الأحمر٣٨٦٤كذبت لا يدخلها فإنه قد شهد بدرا والحديبية٣٨٦٥ما من أحد من أصحابي يموت بأرض إلا بعث قائدا ونورا لهم يوم القيامة٣٨٦٦إذا رأيتم الذين يسبون أصحابي فقولوا لعنة الله على شركم٣٨٦٧إن بني هشام بن المغيرة استأذنوني في أن ينكحوا ابنتهم علي بن أبي طالب…٣٨٦٨قال كان أحب النساء إلى رسول الله صلى الله عليه وسلم فاطمة ومن الرجال…٣٨٦٩إنما فاطمة بضعة مني يؤذيني ما آذاها وينصبني ما أنصبها٣٨٧٠أنا حرب لمن حاربتم وسلم لمن سالمتم٣٨٧١اللهم هؤلاء أهل بيتي وخاصتي أذهب عنهم الرجس وطهرهم تطهيرا٣٨٧٢حدثنا محمد بن بشار، قال حدثنا عثمان بن عمر، قال أخبرنا إسرائيل، عن…٣٨٧٣أن عبد الله بن وهب، أخبره أن أم سلمة أخبرته أن رسول الله صلى الله…٣٨٧٤حدثنا حسين بن يزيد الكوفي، قال حدثنا عبد السلام بن حرب، عن أبي…٣٨٧٥قالت ما غرت على أحد من أزواج النبي صلى الله عليه وسلم ما غرت على…٣٨٧٦قالت ما حسدت أحدا ما حسدت خديجة وما تزوجني رسول الله صلى الله عليه…٣٨٧٧خير نسائها خديجة بنت خويلد وخير نسائها مريم بنت عمران٣٨٧٨حسبك من نساء العالمين مريم ابنة عمران وخديجة بنت خويلد وفاطمة بنت…٣٨٧٩يا أم سلمة لا تؤذيني في عائشة فإنه ما أنزل على الوحى وأنا في لحاف…٣٨٨٠هذه زوجتك في الدنيا والآخرة٣٨٨١يا عائشة هذا جبريل وهو يقرأ عليك السلام٣٨٨٢إن جبريل يقرأ عليك السلام٣٨٨٣حدثنا حميد بن مسعدة، قال حدثنا زياد بن الربيع، قال حدثنا خالد بن سلمة…٣٨٨٤قال ما رأيت أحدا أفصح من عائشة . هذا حديث حسن صحيح غريب .٣٨٨٥ومحمد بن بشار، واللفظ، لابن يعقوب قالا حدثنا يحيى بن حماد، قال حدثنا…٣٨٨٦حدثنا إبراهيم بن سعيد الجوهري، قال حدثنا يحيى بن سعيد الأموي، عن…٣٨٨٧فضل عائشة على النساء كفضل الثريد على سائر الطعام٣٨٨٨أن رجلا، نال من عائشة عند عمار بن ياسر فقال اغرب مقبوحا منبوحا أتؤذي…٣٨٨٩حدثنا محمد بن بشار، قال حدثنا عبد الرحمن بن مهدي، قال حدثنا أبو بكر…٣٨٩٠حدثنا أحمد بن عبدة الضبي، قال حدثنا المعتمر بن سليمان، عن حميد، عن…٣٨٩١إذا رأيتم آية فاسجدوا٣٨٩٢ألا قلت فكيف تكونان خيرا مني وزوجي محمد وأبي هارون وعمي موسى٣٨٩٣حدثنا محمد بن بشار، قال حدثنا محمد بن خالد بن عثمة، قال حدثني موسى بن…٣٨٩٤وعبد بن حميد، قالا أخبرنا عبد الرزاق، قال أخبرنا معمر، عن ثابت، عن…٣٨٩٥خيركم خيركم لأهله وأنا خيركم لأهلي وإذا مات صاحبكم فدعوه٣٨٩٦لا يبلغني أحد عن أحد من أصحابي شيئا فإني أحب أن أخرج إليهم وأنا سليم…٣٨٩٧لا يبلغني أحد عن أحد شيئا٣٨٩٨إن الله أمرني أن أقرأ عليك القرآن٣٨٩٩لولا الهجرة لكنت امرأ من الأنصار٣٩٠٠لا يحبهم إلا مؤمن ولا يبغضهم إلا منافق من أحبهم فأحبه الله ومن أبغضهم…٣٩٠١هلم هل فيكم أحد من غيركم٣٩٠٢اللهم اغفر للأنصار ولذراري الأنصار ولذراري ذراريهم٣٩٠٣اقرأ قومك السلام فإنهم ما علمت أعفة صبر٣٩٠٤ألا إن عيبتي التي آوي إليها أهل بيتي وإن كرشي الأنصار فاعفوا عن…٣٩٠٥من يرد هوان قريش أهانه الله
