كتاب الأشربة عن رسول الله صلى الله عليه وسلم

٣٧ حديثًا
١٨٦١كل مسكر خمر وكل مسكر حرام ومن شرب الخمر في الدنيا فمات وهو يدمنها لم…١٨٦٢من شرب الخمر لم يقبل الله له صلاة أربعين صباحا فإن تاب تاب الله عليه…١٨٦٣كل شراب أسكر فهو حرام١٨٦٤حدثنا عبيد بن أسباط بن محمد القرشي الكوفي، وأبو سعيد الأشج قالا حدثنا…١٨٦٥ما أسكر كثيره فقليله حرام١٨٦٦كل مسكر حرام ما أسكر الفرق منه فملء الكف منه حرام١٨٦٧ويزيد بن هارون، قالا أخبرنا سليمان التيمي، عن طاوس، أن رجلا، أتى ابن…١٨٦٨حدثنا أبو موسى، محمد بن المثنى حدثنا أبو داود الطيالسي، حدثنا شعبة،…١٨٦٩إني كنت نهيتكم عن الظروف وإن ظرفا لا يحل شيئا ولا يحرمه وكل مسكر حرام١٨٧٠حدثنا محمود بن غيلان، حدثنا أبو داود الحفري، عن سفيان، عن منصور، عن…١٨٧١حدثنا محمد بن المثنى، حدثنا عبد الوهاب الثقفي، عن يونس بن عبيد، عن…١٨٧٢إن من الحنطة خمرا ومن الشعير خمرا ومن التمر خمرا ومن الزبيب خمرا ومن…١٨٧٣نحوه . وروى أبو حيان التيمي، هذا الحديث عن الشعبي، عن ابن عمر، عن…١٨٧٤حدثنا بذلك، أحمد بن منيع حدثنا عبد الله بن إدريس، عن أبي حيان التيمي،…١٨٧٥الخمر من هاتين الشجرتين النخلة والعنبة١٨٧٦أن رسول الله صلى الله عليه وسلم نهى أن ينبذ البسر والرطب جميعا . قال…١٨٧٧حدثنا سفيان بن وكيع، حدثنا جرير، عن سليمان التيمي، عن أبي نضرة، عن…١٨٧٨هي لهم في الدنيا ولكم في الآخرة١٨٧٩أن النبي صلى الله عليه وسلم نهى أن يشرب الرجل قائما . فقيل الأكل قال…١٨٨٠حدثنا أبو السائب، سلم بن جنادة الكوفي حدثنا حفص بن غياث، عن عبيد الله…١٨٨١ضالة المسلم حرق النار١٨٨٢ومغيرة، عن الشعبي، عن ابن عباس، أن النبي صلى الله عليه وسلم شرب من…١٨٨٣قال رأيت رسول الله صلى الله عليه وسلم يشرب قائما وقاعدا . قال أبو…١٨٨٤حديث ١٨٨٤١٨٨٤.٢حديث ١٨٨٤.٢١٨٨٥لا تشربوا واحدا كشرب البعير ولكن اشربوا مثنى وثلاث وسموا إذا أنتم…١٨٨٦حدثنا علي بن خشرم، حدثنا عيسى بن يونس، عن رشدين بن كريب، عن أبيه، عن…١٨٨٧وهو ابن حبيب أنه سمع أبا المثنى الجهني، يذكر عن أبي سعيد الخدري، أن…١٨٨٨أن النبي صلى الله عليه وسلم نهى أن يتنفس في الإناء أو ينفخ فيه . قال…١٨٨٩إذا شرب أحدكم فلا يتنفس في الإناء١٨٩٠وابن، عباس وأبي هريرة . قال أبو عيسى هذا حديث حسن صحيح .١٨٩١حدثنا يحيى بن موسى، حدثنا عبد الرزاق، أخبرنا عبد الله بن عمر، عن عيسى…١٨٩٢كبشة قالت دخل على رسول الله صلى الله عليه وسلم فشرب من في قربة معلقة…١٨٩٣حدثنا الأنصاري، حدثنا معن، حدثنا مالك، قال وحدثنا قتيبة، عن مالك، عن…١٨٩٤ساقي القوم آخرهم شربا١٨٩٥حدثنا ابن أبي عمر، حدثنا سفيان بن عيينة، عن معمر، عن الزهري، عن عروة،…١٨٩٦ويونس، عن الزهري، أن رسول الله صلى الله عليه وسلم سئل أى الشراب أطيب…
