كتاب الأمثال عن رسول الله صلى الله عليه وسلم

١٧ حديثًا
٢٨٥٩إن الله ضرب مثلا صراطا مستقيما على كنفى الصراط سوران لهما أبواب مفتحة…٢٨٦٠إني رأيت في المنام كأن جبريل عند رأسي وميكائيل عند رجلى يقول أحدهما…٢٨٦١لا تبرحن خطك فإنه سينتهي إليك رجال فلا تكلمهم فإنهم لا يكلمونك٢٨٦٢إنما مثلي ومثل الأنبياء قبلي كرجل بنى دارا فأكملها وأحسنها إلا موضع…٢٨٦٣إن الله أمر يحيى بن زكريا بخمس كلمات أن يعمل بها ويأمر بني إسرائيل أن…٢٨٦٤حدثنا محمد بن بشار، قال حدثنا أبو داود الطيالسي، قال حدثنا أبان بن…٢٨٦٥مثل المؤمن الذي يقرأ القرآن كمثل الأترجة ريحها طيب وطعمها طيب ومثل…٢٨٦٦مثل المؤمن كمثل الزرع لا تزال الرياح تفيئه ولا يزال المؤمن يصيبه بلاء…٢٨٦٧إن من الشجر شجرة لا يسقط ورقها وهي مثل المؤمن حدثوني ما هي٢٨٦٨أرأيتم لو أن نهرا بباب أحدكم يغتسل منه كل يوم خمس مرات هل يبقى من…٢٨٦٨.٢نحوه .٢٨٦٩مثل أمتي مثل المطر لا يدرى أوله خير أم آخره٢٨٧٠هل تدرون ما هذه وما هذه٢٨٧١إنما أجلكم فيما خلا من الأمم كما بين صلاة العصر إلى مغارب الشمس وإنما…٢٨٧٢إنما الناس كإبل مائة لا يجد الرجل فيها راحلة٢٨٧٣بهذا الإسناد نحوه وقال لا تجد فيها راحلة أو قال لا تجد فيها إلا راحلة…٢٨٧٤إنما مثلي ومثل أمتي كمثل رجل استوقد نارا فجعلت الذباب والفراش يقعن…
