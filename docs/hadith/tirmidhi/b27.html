كتاب البر والصلة عن رسول الله صلى الله عليه وسلم

١٤١ حديثًا
١٨٩٧حدثنا محمد بن بشار، أخبرنا يحيى بن سعيد، أخبرنا بهز بن حكيم، حدثني…١٨٩٨حدثنا أحمد بن محمد، أخبرنا عبد الله بن المبارك، عن المسعودي، عن…١٨٩٩رضا الرب في رضا الوالد وسخط الرب في سخط الوالد١٩٠٠الوالد أوسط أبواب الجنة فإن شئت فأضع ذلك الباب أو احفظه١٩٠١ألا أحدثكم بأكبر الكبائر١٩٠٢من الكبائر أن يشتم الرجل والديه١٩٠٣إن أبر البر أن يصل الرجل أهل ود أبيه١٩٠٤الخالة بمنزلة الأم١٩٠٤.٢أن رجلا، أتى النبي صلى الله عليه وسلم فقال يا رسول الله إني أصبت ذنبا…١٩٠٤.٣حدثنا ابن أبي عمر، حدثنا سفيان بن عيينة، عن محمد بن سوقة، عن أبي بكر…١٩٠٥ثلاث دعوات مستجابات لا شك فيهن دعوة المظلوم ودعوة المسافر ودعوة…١٩٠٦لا يجزي ولد والدا إلا أن يجده مملوكا فيشتريه فيعتقه١٩٠٧قال الله أنا الله وأنا الرحمن خلقت الرحم وشققت لها من اسمي فمن وصلها…١٩٠٨ليس الواصل بالمكافئ ولكن الواصل الذي إذا انقطعت رحمه وصلها١٩٠٩لا يدخل الجنة قاطع١٩١٠إنكم لتبخلون وتجبنون وتجهلون وإنكم لمن ريحان الله١٩١١إنه من لا يرحم لا يرحم١٩١٢لا يكون لأحدكم ثلاث بنات أو ثلاث أخوات فيحسن إليهن إلا دخل الجنة١٩١٣من ابتلي بشيء من البنات فصبر عليهن كن له حجابا من النار١٩١٤من عال جاريتين دخلت أنا وهو الجنة كهاتين١٩١٥من ابتلي بشيء من هذه البنات كن له سترا من النار١٩١٦من كان له ثلاث بنات أو ثلاث أخوات أو ابنتان أو أختان فأحسن صحبتهن…١٩١٧من قبض يتيما بين المسلمين إلى طعامه وشرابه أدخله الله الجنة البتة إلا…١٩١٨أنا وكافل اليتيم في الجنة كهاتين١٩١٩ليس منا من لم يرحم صغيرنا ويوقر كبيرنا١٩٢٠ليس منا من لم يرحم صغيرنا ويعرف شرف كبيرنا١٩٢١ليس منا من لم يرحم صغيرنا ويوقر كبيرنا ويأمر بالمعروف وينه عن المنكر١٩٢٢من لا يرحم الناس لا يرحمه الله١٩٢٣لا تنزع الرحمة إلا من شقي١٩٢٤الراحمون يرحمهم الرحمن ارحموا من في الأرض يرحمكم من في السماء الرحم…١٩٢٥قال بايعت رسول الله صلى الله عليه وسلم على إقام الصلاة وإيتاء الزكاة…١٩٢٦حدثنا محمد بن بشار، حدثنا صفوان بن عيسى، عن محمد بن عجلان، عن القعقاع…١٩٢٧المسلم أخو المسلم لا يخونه ولا يكذبه ولا يخذله كل المسلم على المسلم…١٩٢٨المؤمن للمؤمن كالبنيان يشد بعضه بعضا١٩٢٩إن أحدكم مرآة أخيه فإن رأى به أذى فليمطه عنه١٩٣٠من نفس عن مسلم كربة من كرب الدنيا نفس الله عنه كربة من كرب يوم…١٩٣١من رد عن عرض أخيه رد الله عن وجهه النار يوم القيامة١٩٣٢لا يحل لمسلم أن يهجر أخاه فوق ثلاث يلتقيان فيصد هذا ويصد هذا وخيرهما…١٩٣٣حدثنا أحمد بن منيع، حدثنا إسماعيل بن إبراهيم، حدثنا حميد، عن أنس، قال…١٩٣٤ذكرك أخاك بما يكره١٩٣٥لا تقاطعوا ولا تدابروا ولا تباغضوا ولا تحاسدوا وكونوا عباد الله…١٩٣٦لا حسد إلا في اثنتين رجل آتاه الله مالا فهو ينفق منه آناء الليل وآناء…١٩٣٧إن الشيطان قد يئس أن يعبده المصلون ولكن في التحريش بينهم١٩٣٨ليس بالكاذب من أصلح بين الناس فقال خيرا أو نمى خيرا١٩٣٩لا يحل الكذب إلا في ثلاث يحدث الرجل امرأته ليرضيها والكذب في الحرب…١٩٤٠من ضار ضار الله به ومن شاق شاق الله عليه١٩٤١ملعون من ضار مؤمنا أو مكر به١٩٤٢ما زال جبريل يوصيني بالجار حتى ظننت أنه سيورثه١٩٤٣ما زال جبريل يوصيني بالجار حتى ظننت أنه سيورثه١٩٤٤خير الأصحاب عند الله خيرهم لصاحبه وخير الجيران عند الله خيرهم لجاره١٩٤٥إخوانكم جعلهم الله فتية تحت أيديكم فمن كان أخوه تحت يده فليطعمه من…١٩٤٦لا يدخل الجنة سيئ الملكة١٩٤٧من قذف مملوكه بريئا مما قال له أقام عليه الحد يوم القيامة إلا أن يكون…١٩٤٨اعلم أبا مسعود اعلم أبا مسعود١٩٤٩كل يوم سبعين مرة١٩٥٠إذا ضرب أحدكم خادمه فذكر الله فارفعوا أيديكم١٩٥١لأن يؤدب الرجل ولده خير من أن يتصدق بصاع١٩٥٢ما نحل والد ولدا من نحل أفضل من أدب حسن١٩٥٣وعلي بن خشرم، قالا حدثنا عيسى بن يونس، عن هشام بن عروة، عن أبيه، عن…١٩٥٤من لا يشكر الناس لا يشكر الله١٩٥٥من لم يشكر الناس لم يشكر الله١٩٥٦تبسمك في وجه أخيك لك صدقة وأمرك بالمعروف ونهيك عن المنكر صدقة وإرشادك…١٩٥٧من منح منيحة لبن أو ورق أو هدى زقاقا كان له مثل عتق رقبة١٩٥٨بينما رجل يمشي في طريق إذ وجد غصن شوك فأخره فشكر الله له فغفر له١٩٥٩إذا حدث الرجل الحديث ثم التفت فهي أمانة١٩٦٠نعم ولا توكي فيوكى عليك١٩٦١السخي قريب من الله قريب من الجنة قريب من الناس بعيد من النار والبخيل…١٩٦٢خصلتان لا تجتمعان في مؤمن البخل وسوء الخلق١٩٦٣لا يدخل الجنة خب ولا منان ولا بخيل١٩٦٤المؤمن غر كريم والفاجر خب لئيم١٩٦٥نفقة الرجل على أهله صدقة١٩٦٦أفضل الدينار دينار ينفقه الرجل على عياله ودينار ينفقه الرجل على دابته…١٩٦٧من كان يؤمن بالله واليوم الآخر فليكرم ضيفه جائزته١٩٦٨الضيافة ثلاثة أيام وجائزته يوم وليلة وما أنفق عليه بعد ذلك فهو صدقة…١٩٦٩الساعي على الأرملة والمسكين كالمجاهد في سبيل الله أو كالذي يصوم…١٩٧٠كل معروف صدقة وإن من المعروف أن تلقى أخاك بوجه طلق وأن تفرغ من دلوك…١٩٧١عليكم بالصدق فإن الصدق يهدي إلى البر وإن البر يهدي إلى الجنة وما يزال…١٩٧٢إذا كذب العبد تباعد عنه الملك ميلا من نتن ما جاء به١٩٧٣قالت ما كان خلق أبغض إلى رسول الله صلى الله عليه وسلم من الكذب ولقد…١٩٧٤ما كان الفحش في شيء إلا شانه وما كان الحياء في شيء إلا زانه١٩٧٥خياركم أحاسنكم أخلاقا١٩٧٦لا تلاعنوا بلعنة الله ولا بغضبه ولا بالنار١٩٧٧ليس المؤمن بالطعان ولا اللعان ولا الفاحش ولا البذيء١٩٧٨لا تلعن الريح فإنها مأمورة وإنه من لعن شيئا ليس له بأهل رجعت اللعنة…١٩٧٩تعلموا من أنسابكم ما تصلون به أرحامكم فإن صلة الرحم محبة في الأهل…١٩٨٠ما دعوة أسرع إجابة من دعوة غائب لغائب١٩٨١المستبان ما قالا فعلى البادي منهما ما لم يعتد المظلوم١٩٨٢لا تسبوا الأموات فتؤذوا الأحياء١٩٨٣سباب المسلم فسوق وقتاله كفر١٩٨٤إن في الجنة غرفا ترى ظهورها من بطونها وبطونها من ظهورها١٩٨٥نعما لأحدهم أن يطيع ربه ويؤدي حق سيده١٩٨٦ثلاثة على كثبان المسك أراه قال يوم القيامة عبد أدى حق الله وحق مواليه…١٩٨٧اتق الله حيثما كنت وأتبع السيئة الحسنة تمحها وخالق الناس بخلق حسن١٩٨٨إياكم والظن فإن الظن أكذب الحديث١٩٨٩يا أبا عمير ما فعل النغير١٩٩٠إني لا أقول إلا حقا١٩٩١إني حاملك على ولد الناقة١٩٩٢أن النبي صلى الله عليه وسلم قال له " يا ذا الأذنين " . قال محمود قال…١٩٩٣من ترك الكذب وهو باطل بني له في ربض الجنة ومن ترك المراء وهو محق بني…١٩٩٤كفى بك إثما أن لا تزال مخاصما١٩٩٥لا تمار أخاك ولا تمازحه ولا تعده موعدة فتخلفه١٩٩٦بئس ابن العشيرة أو أخو العشيرة١٩٩٧أحبب حبيبك هونا ما عسى أن يكون بغيضك يوما ما وأبغض بغيضك هونا ما عسى…١٩٩٨لا يدخل الجنة من كان في قلبه مثقال حبة من خردل من كبر ولا يدخل النار…١٩٩٩لا يدخل الجنة من كان في قلبه مثقال ذرة من كبر ولا يدخل النار يعني من…٢٠٠٠لا يزال الرجل يذهب بنفسه حتى يكتب في الجبارين فيصيبه ما أصابهم٢٠٠١من فعل هذا فليس فيه من الكبر شيء٢٠٠٢ما شيء أثقل في ميزان المؤمن يوم القيامة من خلق حسن وإن الله ليبغض…٢٠٠٣ما من شيء يوضع في الميزان أثقل من حسن الخلق وإن صاحب حسن الخلق ليبلغ…٢٠٠٤تقوى الله وحسن الخلق٢٠٠٥أنه وصف حسن الخلق فقال هو بسط الوجه وبذل المعروف وكف الأذى .٢٠٠٦وأحمد بن منيع، ومحمود بن غيلان، قالوا حدثنا أبو أحمد الزبيري، عن…٢٠٠٧لا تكونوا إمعة تقولون إن أحسن الناس أحسنا وإن ظلموا ظلمنا ولكن وطنوا…٢٠٠٨من عاد مريضا أو زار أخا له في الله ناداه مناد أن طبت وطاب ممشاك…٢٠٠٩الحياء من الإيمان والإيمان في الجنة والبذاء من الجفاء والجفاء في النار٢٠١٠السمت الحسن والتؤدة والاقتصاد جزء من أربعة وعشرين جزءا من النبوة٢٠١١إن فيك خصلتين يحبهما الله الحلم والأناة٢٠١٢الأناة من الله والعجلة من الشيطان٢٠١٣من أعطي حظه من الرفق فقد أعطي حظه من الخير ومن حرم حظه من الرفق فقد…٢٠١٤اتق دعوة المظلوم فإنها ليس بينها وبين الله حجاب٢٠١٥حدثنا قتيبة، حدثنا جعفر بن سليمان الضبعي، عن ثابت، عن أنس، قال خدمت…٢٠١٦رسول الله صلى الله عليه وسلم فقالت لم يكن فاحشا ولا متفحشا ولا صخابا…٢٠١٧قالت ما غرت على أحد من أزواج النبي صلى الله عليه وسلم ما غرت على…٢٠١٨إن من أحبكم إلى وأقربكم مني مجلسا يوم القيامة أحاسنكم أخلاقا وإن…٢٠١٩لا يكون المؤمن لعانا٢٠٢٠حدثنا أبو كريب، حدثنا أبو بكر بن عياش، عن أبي حصين، عن أبي صالح، عن…٢٠٢١من كظم غيظا وهو يستطيع أن ينفذه دعاه الله يوم القيامة على رءوس…٢٠٢٢ما أكرم شاب شيخا لسنه إلا قيض الله له من يكرمه عند سنه٢٠٢٣تفتح أبواب الجنة يوم الاثنين والخميس فيغفر فيهما لمن لا يشرك بالله…٢٠٢٤ما يكون عندي من خير فلن أدخره عنكم ومن يستغن يغنه الله ومن يستعفف…٢٠٢٥إن من شر الناس عند الله يوم القيامة ذا الوجهين٢٠٢٦لا يدخل الجنة قتات٢٠٢٧الحياء والعي شعبتان من الإيمان والبذاء والبيان شعبتان من النفاق٢٠٢٨إن من البيان سحرا٢٠٢٩ما نقصت صدقة من مال وما زاد الله رجلا بعفو إلا عزا وما تواضع أحد لله…٢٠٣٠الظلم ظلمات يوم القيامة٢٠٣١قال ما عاب رسول الله صلى الله عليه وسلم طعاما قط كان إذا اشتهاه أكله…٢٠٣٢يا معشر من قد أسلم بلسانه ولم يفض الإيمان إلى قلبه لا تؤذوا المسلمين…٢٠٣٣لا حليم إلا ذو عثرة ولا حكيم إلا ذو تجربة٢٠٣٤من أعطي عطاء فوجد فليجز به ومن لم يجد فليثن فإن من أثنى فقد شكر ومن…٢٠٣٥من صنع إليه معروف فقال لفاعله جزاك الله خيرا فقد أبلغ في
