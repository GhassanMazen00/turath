كتاب الجهاد عن رسول الله صلى الله عليه وسلم

٥٠ حديثًا
١٦٧٠ائتوني بالكتف أو اللوح١٦٧١وشعبة، عن حبيب بن أبي ثابت، عن أبي العباس، عن عبد الله بن عمرو، قال…١٦٧٢حدثنا محمد بن يحيى النيسابوري، حدثنا الحجاج بن محمد، حدثنا ابن جريج،…١٦٧٣لو أن الناس يعلمون ما أعلم من الوحدة ما سرى راكب بليل١٦٧٤الراكب شيطان والراكبان شيطانان والثلاثة ركب١٦٧٥ونصر بن علي، قالا حدثنا سفيان بن عيينة، عن عمرو بن دينار، سمع جابر بن…١٦٧٦قال كنت إلى جنب زيد بن أرقم فقيل له كم غزا النبي صلى الله عليه وسلم…١٦٧٧حدثنا محمد بن حميد الرازي، حدثنا سلمة بن الفضل، عن محمد بن إسحاق، عن…١٦٧٨اللهم منزل الكتاب سريع الحساب اهزم الأحزاب اللهم اهزمهم وزلزلهم١٦٧٩حدثنا محمد بن عمر بن الوليد الكندي الكوفي، وأبو كريب ومحمد بن رافع…١٦٨٠حدثنا أحمد بن منيع، حدثنا يحيى بن زكريا بن أبي زائدة، حدثنا أبو يعقوب…١٦٨١قال كانت راية رسول الله صلى الله عليه وسلم سوداء ولواؤه أبيض . قال…١٦٨٢إن بيتكم العدو فقولوا : حم لا ينصرون١٦٨٣قال صنعت سيفي على سيف سمرة بن جندب وزعم سمرة أنه صنع سيفه على سيف…١٦٨٤حدثنا أحمد بن محمد بن موسى، أنبأنا عبد الله بن المبارك، أنبأنا سعيد…١٦٨٥ما كان من فزع وإن وجدناه لبحرا١٦٨٦ما رأينا من فزع وإن وجدناه لبحرا١٦٨٧لم تراعوا لم تراعوا١٦٨٨أنا النبي لا كذب أنا ابن عبد المطلب١٦٨٩حدثنا محمد بن عمر بن علي المقدمي البصري، حدثني أبي، عن سفيان بن حسين،…١٦٩٠حدثنا محمد بن صدران أبو جعفر البصري، حدثنا طالب بن حجير، عن هود بن…١٦٩١حدثنا محمد بن بشار، حدثنا وهب بن جرير بن حازم، حدثنا أبي، عن قتادة،…١٦٩٢حدثنا أبو سعيد الأشج، حدثنا يونس بن بكير، عن محمد بن إسحاق، عن يحيى…١٦٩٣حدثنا قتيبة، حدثنا مالك بن أنس، عن ابن شهاب، عن أنس بن مالك، قال دخل…١٦٩٤الخير معقود في نواصي الخيل إلى يوم القيامة الأجر والمغنم١٦٩٥يمن الخيل في الشقر١٦٩٦خير الخيل الأدهم الأقرح الأرثم ثم الأقرح المحجل طلق اليمين فإن لم يكن…١٦٩٧بهذا الإسناد نحوه بمعناه . قال أبو عيسى هذا حديث حسن غريب صحيح .١٦٩٨حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، حدثنا سفيان، قال حدثني سلم بن…١٦٩٩حدثنا محمد بن وزير الواسطي، حدثنا إسحاق بن يوسف الأزرق، عن سفيان، عن…١٧٠٠لا سبق إلا في نصل أو خف أو حافر١٧٠١حدثنا أبو كريب، حدثنا إسماعيل بن إبراهيم، حدثنا أبو جهضم، موسى بن…١٧٠٢ابغوني ضعفاءكم فإنما ترزقون وتنصرون بضعفائكم١٧٠٣لا تصحب الملائكة رفقة فيها كلب ولا جرس١٧٠٤إذا كان القتال فعلي١٧٠٥حديث ١٧٠٥١٧٠٦يا أيها الناس اتقوا الله وإن أمر عليكم عبد حبشي مجدع فاسمعوا له…١٧٠٧السمع والطاعة على المرء المسلم فيما أحب وكره ما لم يؤمر بمعصية فإن…١٧٠٨حدثنا أبو كريب، حدثنا يحيى بن آدم، عن قطبة بن عبد العزيز، عن الأعمش،…١٧٠٩حديث ١٧٠٩١٧١٠حدثنا أحمد بن منيع، حدثنا روح بن عبادة، عن ابن جريج، عن أبي الزبير،…١٧١١حديث ١٧١١١٧١٢نعم إن قتلت في سبيل الله وأنت صابر محتسب مقبل غير مدبر١٧١٣احفروا وأوسعوا وأحسنوا وادفنوا الاثنين والثلاثة في قبر واحد وقدموا…١٧١٤ما تقولون في هؤلاء الأسارى١٧١٥أن المشركين، أرادوا أن يشتروا، جسد رجل من المشركين فأبى النبي صلى…١٧١٦بل أنتم العكارون وأنا فئتكم١٧١٧حدثنا محمود بن غيلان، حدثنا أبو داود، أخبرنا شعبة، عن الأسود بن قيس،…١٧١٨وسعيد بن عبد الرحمن المخزومي، قالا حدثنا سفيان بن عيينة، عن الزهري،…١٧١٩حدثنا ابن أبي عمر، حدثنا سفيان بن عيينة، عن عمرو بن دينار، عن ابن…
