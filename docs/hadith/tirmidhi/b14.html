١١٧ حديثًا
١٢٠٥حديث ١٢٠٥١٢٠٦حدثنا قتيبة، حدثنا أبو عوانة، عن سماك بن حرب، عن عبد الرحمن بن عبد…١٢٠٧الشرك بالله وعقوق الوالدين وقتل النفس وقول الزور١٢٠٨حديث ١٢٠٨١٢٠٩حديث ١٢٠٩١٢١٠أنه خرج مع النبي صلى الله عليه وسلم إلى المصلى فرأى الناس يتبايعون…١٢١١ثلاثة لا ينظر الله إليهم يوم القيامة ولا يزكيهم ولهم عذاب أليم١٢١٢اللهم بارك لأمتي في بكورها١٢١٣كذب قد علم أني من أتقاهم لله وآداهم للأمانة١٢١٤وعثمان بن عمر، عن هشام بن حسان، عن عكرمة، عن ابن عباس، قال توفي النبي…١٢١٥ما أمسى في آل محمد صلى الله عليه وسلم صاع تمر ولا صاع حب١٢١٦هذا ما اشترى العداء بن خالد بن هوذة من محمد رسول الله صلى الله عليه…١٢١٧إنكم قد وليتم أمرين هلكت فيه الأمم السالفة قبلكم١٢١٨من يشتري هذا الحلس والقدح١٢١٩أن رجلا، من الأنصار دبر غلاما له فمات ولم يترك مالا غيره فباعه النبي…١٢٢٠حدثنا هناد، حدثنا ابن المبارك، أخبرنا سليمان التيمي، عن أبي عثمان، عن…١٢٢١حدثنا سلمة بن شبيب، حدثنا عبد الله بن جعفر الرقي، حدثنا عبيد الله بن…١٢٢٢لا يبيع حاضر لباد١٢٢٣لا يبيع حاضر لباد دعوا الناس يرزق الله بعضهم من بعض١٢٢٤حدثنا قتيبة، حدثنا يعقوب بن عبد الرحمن الإسكندراني، عن سهيل بن أبي…١٢٢٥حديث ١٢٢٥١٢٢٦حدثنا أحمد بن منيع، حدثنا إسماعيل بن إبراهيم، عن أيوب، عن نافع، عن…١٢٢٧وبهذا الإسناد أن النبي صلى الله عليه وسلم نهى عن بيع السنبل حتى يبيض…١٢٢٨وعفان، وسليمان بن حرب، قالوا حدثنا حماد بن سلمة، عن حميد، عن أنس، أن…١٢٢٩حدثنا قتيبة، حدثنا حماد بن زيد، عن أيوب، عن نافع، عن ابن عمر، أن…١٢٣٠حدثنا أبو كريب، أنبأنا أبو أسامة، عن عبيد الله بن عمر، عن أبي الزناد،…١٢٣١حدثنا هناد، حدثنا عبدة بن سليمان، عن محمد بن عمرو، عن أبي سلمة، عن…١٢٣٢لا تبع ما ليس عندك١٢٣٣حدثنا قتيبة، حدثنا حماد بن زيد، عن أيوب، عن يوسف بن ماهك، عن حكيم بن…١٢٣٤لا يحل سلف وبيع ولا شرطان في بيع ولا ربح ما لم يضمن ولا بيع ما ليس…١٢٣٥وعبدة بن عبد الله الخزاعي البصري أبو سهل، وغير، واحد، قالوا حدثنا عبد…١٢٣٦وشعبة، عن عبد الله بن دينار، عن ابن عمر، أن رسول الله صلى الله عليه…١٢٣٧حدثنا أبو موسى، محمد بن مثنى حدثنا عبد الرحمن بن مهدي، عن حماد بن…١٢٣٨الحيوان اثنان بواحد لا يصلح نسيئا ولا بأس به يدا بيد١٢٣٩حدثنا قتيبة، أخبرنا الليث، عن أبي الزبير، عن جابر، قال جاء عبد فبايع…١٢٤٠الذهب بالذهب مثلا بمثل والفضة بالفضة مثلا بمثل والتمر بالتمر مثلا…١٢٤١لا تبيعوا الذهب بالذهب إلا مثلا بمثل والفضة بالفضة إلا مثلا بمثل لا…١٢٤٢لا بأس به بالقيمة١٢٤٣الورق بالذهب ربا إلا هاء وهاء والبر بالبر ربا إلا هاء وهاء والشعير…١٢٤٤من ابتاع نخلا بعد أن تؤبر فثمرتها للذي باعها إلا أن يشترط المبتاع ومن…١٢٤٥البيعان بالخيار ما لم يتفرقا أو يختارا١٢٤٦البيعان بالخيار ما لم يتفرقا فإن صدقا وبينا بورك لهما في بيعهما وإن…١٢٤٧البيعان بالخيار ما لم يتفرقا إلا أن تكون صفقة خيار ولا يحل له أن…١٢٤٨لا يتفرقن عن بيع إلا عن تراض١٢٤٩أن النبي صلى الله عليه وسلم خير أعرابيا بعد البيع . وهذا حديث حسن…١٢٥٠إذا بايعت فقل هاء وهاء ولا خلابة١٢٥١من اشترى مصراة فهو بالخيار إذا حلبها إن شاء ردها ورد معها صاعا من تمر١٢٥٢من اشترى مصراة فهو بالخيار ثلاثة أيام فإن ردها رد معها صاعا من طعام…١٢٥٣حدثنا ابن أبي عمر، حدثنا وكيع، عن زكريا، عن الشعبي، عن جابر بن عبد…١٢٥٤الظهر يركب إذا كان مرهونا ولبن الدر يشرب إذا كان مرهونا وعلى الذي…١٢٥٥حديث ١٢٥٥١٢٥٦اشتريها فإنما الولاء لمن أعطى الثمن أو لمن ولي النعمة١٢٥٧ضح بالشاة وتصدق بالدينار١٢٥٨حديث ١٢٥٨١٢٥٩إذا أصاب المكاتب حدا أو ميراثا ورث بحساب ما عتق منه١٢٦٠من كاتب عبده على مائة أوقية فأداها إلا عشر أواق أو قال عشرة دراهم ثم…١٢٦١إذا كان عند مكاتب إحداكن ما يؤدي فلتحتجب منه١٢٦٢أيما امرئ أفلس ووجد رجل سلعته عنده بعينها فهو أولى بها من غيره١٢٦٣حدثنا علي بن خشرم، أخبرنا عيسى بن يونس، عن مجالد، عن أبي الوداك، عن…١٢٦٤أد الأمانة إلى من ائتمنك ولا تخن من خانك١٢٦٥العارية مؤداة والزعيم غارم والدين مقضي١٢٦٦على اليد ما أخذت حتى تؤدي١٢٦٧لا يحتكر إلا خاطئ١٢٦٨لا تستقبلوا السوق ولا تحفلوا ولا ينفق بعضكم لبعض١٢٦٩من حلف على يمين وهو فيها فاجر ليقتطع بها مال امرئ مسلم لقي الله وهو…١٢٧٠إذا اختلف البيعان فالقول قول البائع والمبتاع بالخيار١٢٧١حدثنا قتيبة، حدثنا داود بن عبد الرحمن العطار، عن عمرو بن دينار، عن…١٢٧٢لا يمنع فضل الماء ليمنع به الكلأ١٢٧٣حدثنا أحمد بن منيع، وأبو عمار قالا حدثنا إسماعيل ابن علية، قال أخبرنا…١٢٧٤أن رجلا، من كلاب سأل النبي صلى الله عليه وسلم عن عسب الفحل فنهاه فقال…١٢٧٥كسب الحجام خبيث ومهر البغي خبيث وثمن الكلب خبيث١٢٧٦وغير، واحد، قالوا حدثنا سفيان بن عيينة، عن الزهري، عن أبي بكر بن عبد…١٢٧٧اعلفه ناضحك وأطعمه رقيقك١٢٧٨إن أفضل ما تداويتم به الحجامة١٢٧٩وعلي بن خشرم، قالا أنبأنا عيسى بن يونس، عن الأعمش، عن أبي سفيان، عن…١٢٨٠حدثنا يحيى بن موسى، حدثنا عبد الرزاق، أخبرنا عمر بن زيد الصنعاني، عن…١٢٨١أخبرنا أبو كريب، أخبرنا وكيع، عن حماد بن سلمة، عن أبي المهزم، عن أبي…١٢٨٢لا تبيعوا القينات ولا تشتروهن ولا تعلموهن ولا خير في تجارة فيهن…١٢٨٣من فرق بين الوالدة وولدها فرق الله بينه وبين أحبته يوم القيامة١٢٨٤يا علي ما فعل غلامك١٢٨٥أن رسول الله صلى الله عليه وسلم قضى أن الخراج بالضمان . قال أبو عيسى…١٢٨٦حدثنا أبو سلمة، يحيى بن خلف أخبرنا عمر بن علي المقدمي، عن هشام بن…١٢٨٧من دخل حائطا فليأكل ولا يتخذ خبنة١٢٨٨يا رافع لم ترمي نخلهم١٢٨٩من أصاب منه من ذي حاجة غير متخذ خبنة فلا شىء عليه١٢٩٠حدثنا زياد بن أيوب البغدادي، أخبرنا عباد بن العوام، قال أخبرني سفيان…١٢٩١من ابتاع طعاما فلا يبعه حتى يستوفيه١٢٩٢لا يبيع بعضكم على بيع بعض ولا يخطب بعضكم على خطبة بعض١٢٩٣أهرق الخمر واكسر الدنان١٢٩٤قال سئل النبي صلى الله عليه وسلم أيتخذ الخمر خلا قال " لا " . قال أبو…١٢٩٥حدثنا عبد الله بن منير، قال سمعت أبا عاصم، عن شبيب بن بشر، عن أنس بن…١٢٩٦إذا أتى أحدكم على ماشية فإن كان فيها صاحبها فليستأذنه فإن أذن له…١٢٩٧إن الله ورسوله حرم بيع الخمر والميتة والخنزير والأصنام١٢٩٨ليس لنا مثل السوء العائد في هبته كالكلب يعود في قيئه١٢٩٩لا يحل لأحد أن يعطي عطية فيرجع فيها إلا الوالد فيما يعطي ولده١٣٠٠حدثنا هناد، حدثنا عبدة، عن محمد بن إسحاق، عن نافع، عن ابن عمر، عن زيد…١٣٠١حديث ١٣٠١١٣٠٢حدثنا قتيبة، حدثنا حماد بن زيد، عن أيوب، عن نافع، عن ابن عمر، عن زيد…١٣٠٣مولى بني حارثة أن رافع بن خديج، وسهل بن أبي حثمة، حدثاه أن رسول الله…١٣٠٤وأحمد بن منيع، قالا حدثنا سفيان، عن الزهري، عن سعيد بن المسيب، عن أبي…١٣٠٥ومحمود بن غيلان، قالا حدثنا وكيع، عن سفيان، عن سماك بن حرب، عن سويد…١٣٠٦من أنظر معسرا أو وضع له أظله الله يوم القيامة تحت ظل عرشه يوم لا ظل…١٣٠٧حوسب رجل ممن كان قبلكم فلم يوجد له من الخير شيء إلا أنه كان رجلا…١٣٠٨مطل الغني ظلم وإذا أتبع أحدكم على ملي فليتبع١٣٠٩مطل الغني ظلم وإذا أحلت على مليء فاتبعه ولا تبع بيعتين في بيعة١٣١٠ومحمود بن غيلان، قالا حدثنا وكيع، عن سفيان، عن أبي الزناد، عن الأعرج،…١٣١١من أسلف فليسلف في كيل معلوم ووزن معلوم إلى أجل معلوم١٣١٢من كان له شريك في حائط فلا يبيع نصيبه من ذلك حتى يعرضه على شريكه١٣١٣حدثنا محمد بن بشار، حدثنا عبد الوهاب الثقفي، حدثنا أيوب، عن أبي…١٣١٤إن الله هو المسعر القابض الباسط الرزاق وإني لأرجو أن ألقى ربي وليس…١٣١٥يا صاحب الطعام ما هذا١٣١٦خياركم أحاسنكم قضاء١٣١٧حديث ١٣١٧١٣١٨أعطه إياه فإن خيار الناس أحسنهم قضاء١٣١٩إن الله يحب سمح البيع سمح الشراء سمح القضاء١٣٢٠غفر الله لرجل كان قبلكم كان سهلا إذا باع سهلا إذا اشترى سهلا إذا اقتضى١٣٢١إذا رأيتم من يبيع أو يبتاع في المسجد فقولوا لا أربح الله تجارتك وإذا…
