٣٠٩٧أخر عني يا عمر . إني خيرت فاخترت قد قيل لي : ( استغفر لهم أو لا…٣٠٩٨إذا فرغتم فآذنوني٣٠٩٩حدثنا قتيبة، حدثنا الليث، عن عمران بن أبي أنس، عن عبد الرحمن بن أبي…٣١٠٠نزلت هذه الآية في أهل قباء : ( فيه رجال يحبون أن يتطهروا والله يحب…٣١٠١حدثنا محمود بن غيلان، حدثنا وكيع، حدثنا سفيان، عن أبي إسحاق، عن أبي…٣١٠٢أبشر يا كعب بن مالك بخير يوم أتى عليك منذ ولدتك أمك٣١٠٣أن زيد بن ثابت، حدثه قال بعث إلى أبو بكر الصديق مقتل أهل اليمامة فإذا…٣١٠٤أن حذيفة، قدم على عثمان بن عفان وكان يغازي أهل الشام في فتح أرمينية…٣١٠٥إذا دخل أهل الجنة الجنة نادى مناد إن لكم عند الله موعدا يريد أن…٣١٠٦ما سألني عنها أحد غيرك منذ أنزلت فهي الرؤيا الصالحة يراها المسلم أو…٣١٠٧لما أغرق الله فرعون قال آمنت أنه لا إله إلا الذي آمنت به بنو إسرائيل…٣١٠٨أن جبريل صلى الله عليه وسلم جعل يدس في في فرعون الطين خشية أن يقول لا…٣١٠٩كان في عماء ما تحته هواء وما فوقه هواء وخلق عرشه على الماء٣١١٠إن الله تبارك وتعالى يملي وربما قال يمهل للظالم حتى إذا أخذه لم يفلته٣١١١بل على شيء قد فرغ منه وجرت به الأقلام يا عمر ولكن كل ميسر لما خلق له٣١١٢لا بل للناس كافة٣١١٣بل للمؤمنين عامة٣١١٤لك ولمن عمل بها من أمتي٣١١٥أخلفت غازيا في سبيل الله في أهله بمثل هذا٣١١٦إن الكريم بن الكريم بن الكريم بن الكريم يوسف بن يعقوب بن إسحاق بن…٣١١٧ملك من الملائكة موكل بالسحاب معه مخاريق من نار يسوق بها السحاب حيث…٣١١٨الدقل والفارسي والحلو والحامض٣١١٩مثل كلمة طيبة كشجرة طيبة أصلها ثابت وفرعها في السماء تؤتي أكلها كل…٣١٢٠في القبر إذا قيل له من ربك وما دينك ومن نبيك؟٣١٢١حدثنا ابن أبي عمر، حدثنا سفيان، عن داود بن أبي هند، عن الشعبي، عن…٣١٢٢حدثنا قتيبة، حدثنا نوح بن قيس الحداني، عن عمرو بن مالك، عن أبي…٣١٢٣لجهنم سبعة أبواب باب منها لمن سل السيف على أمتي أو قال على أمة محمد…٣١٢٤الحمد لله أم القرآن وأم الكتاب والسبع المثاني٣١٢٥ما أنزل الله في التوراة ولا في الإنجيل مثل أم القرآن وهي السبع…٣١٢٦عن قول لا إله إلا الله٣١٢٧اتقوا فراسة المؤمن فإنه ينظر بنور الله٣١٢٨أربع قبل الظهر بعد الزوال تحسب بمثلهن في صلاة السحر٣١٢٩كفوا عن القوم إلا أربعة٣١٣٠حين أسري بي لقيت موسى . قال فنعته فإذا رجل حسبته قال مضطرب رجل الرأس…٣١٣١أن النبي صلى الله عليه وسلم أتي بالبراق ليلة أسري به ملجما مسرجا…٣١٣٢لما انتهينا إلى بيت المقدس قال جبريل بإصبعه فخرق به الحجر وشد به…٣١٣٣لما كذبتني قريش قمت في الحجر فجلا الله لي بيت المقدس فطفقت أخبرهم عن…٣١٣٤حدثنا ابن أبي عمر، حدثنا سفيان، عن عمرو بن دينار، عن عكرمة، عن ابن…٣١٣٥تشهده ملائكة الليل وملائكة النهار٣١٣٦يدعى أحدهم فيعطى كتابه بيمينه ويمد له في جسمه ستون ذراعا ويبيض وجهه…٣١٣٧قال قال رسول الله صلى الله عليه وسلم في قوله : ( عسى أن يبعثك ربك…٣١٣٨حدثنا ابن أبي عمر، حدثنا سفيان، عن ابن أبي نجيح، عن مجاهد، عن أبي…٣١٣٩حدثنا أحمد بن منيع، حدثنا جرير، عن قابوس بن أبي ظبيان، عن أبيه، عن…٣١٤٠حدثنا قتيبة، حدثنا يحيى بن زكريا بن أبي زائدة، عن داود بن أبي هند، عن…٣١٤١حدثنا علي بن خشرم، أخبرنا عيسى بن يونس، عن الأعمش، عن إبراهيم، عن…٣١٤٢يحشر الناس يوم القيامة ثلاثة أصناف صنفا مشاة وصنفا ركبانا وصنفا على…٣١٤٣إنكم محشورون رجالا وركبانا ويجرون على وجوهكم٣١٤٤لا تشركوا بالله شيئا ولا تزنوا ولا تقتلوا النفس التي حرم الله إلا…٣١٤٥وهشيم، عن أبي بشر، عن سعيد بن جبير، عن ابن عباس : ( ولا تجهر بصلاتك )…٣١٤٦حدثنا أحمد بن منيع، حدثنا هشيم، حدثنا أبو بشر، عن سعيد بن جبير، عن…٣١٤٧قال قلت لحذيفة بن اليمان أصلى رسول الله صلى الله عليه وسلم في بيت…٣١٤٨أنا سيد ولد آدم يوم القيامة ولا فخر وبيدي لواء الحمد ولا فخر وما من…٣١٤٩قام موسى خطيبا في بني إسرائيل فسئل أى الناس أعلم فقال أنا أعلم . فعتب…٣١٥٠الغلام الذي قتله الخضر طبع يوم طبع كافرا٣١٥١إنما سمي الخضر لأنه جلس على فروة بيضاء فاهتزت تحته خضراء٣١٥٢وغير، واحد، قالوا حدثنا صفوان بن صالح، حدثنا الوليد بن مسلم، عن يزيد…٣١٥٣يحفرونه كل يوم حتى إذا كادوا يخرقونه قال الذي عليهم ارجعوا فستخرقونه…٣١٥٤إذا جمع الله الناس ليوم القيامة ليوم لا ريب فيه نادى مناد من كان أشرك…٣١٥٥ألا أخبرتهم أنهم كانوا يسمون بأنبيائهم والصالحين قبلهم٣١٥٦يؤتى بالموت كأنه كبش أملح حتى يوقف على السور بين الجنة والنار فيقال…٣١٥٧لما عرج بي رأيت إدريس في السماء الرابعة٣١٥٨ما يمنعك أن تزورنا أكثر مما تزورنا٣١٥٩يرد الناس النار ثم يصدرون منها بأعمالهم فأولهم كلمح البرق ثم كالريح…٣١٦٠حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، حدثنا شعبة، عن السدي، عن مرة،…٣١٦١إذا أحب الله عبدا نادى جبريل إني قد أحببت فلانا فأحبه قال فينادي في…٣١٦٢نحوه . قال هذا حديث حسن صحيح .٣١٦٣يا بلال اكلأ لنا الليلة٣١٦٤الويل واد في جهنم يهوي فيه الكافر أربعين خريفا قبل أن يبلغ قعره٣١٦٥يحسب ما خانوك وعصوك وكذبوك وعقابك إياهم فإن كان عقابك إياهم بقدر…٣١٦٦لم يكذب إبراهيم عليه السلام في شيء قط إلا في ثلاث قوله : ( إني سقيم )…٣١٦٧يا أيها الناس إنكم محشورون إلى الله عراة غرلا٣١٦٨أتدرون أى يوم ذلك٣١٦٩هل تدرون أى يوم ذلك٣١٧٠إنما سمي البيت العتيق لأنه لم يظهر عليه جبار٣١٧١حدثنا سفيان بن وكيع، حدثنا أبي وإسحاق بن يوسف الأزرق، عن سفيان…٣١٧٢قال لما أخرج النبي صلى الله عليه وسلم من مكة قال رجل أخرجوا نبيهم…٣١٧٣اللهم زدنا ولا تنقصنا وأكرمنا ولا تهنا وأعطنا ولا تحرمنا وآثرنا ولا…٣١٧٤يا أم حارثة إنها جنان في جنة وإن ابنك أصاب الفردوس الأعلى والفردوس…٣١٧٥لا يا بنت الصديق ولكنهم الذين يصومون ويصلون ويتصدقون وهم يخافون أن لا…٣١٧٦تشويه النار فتقلص شفته العليا حتى تبلغ وسط رأسه وتسترخي شفته السفلى…٣١٧٧يا مرثد الزاني لا ينكح إلا زانية أو مشركة والزانية لا ينكحها إلا زان…٣١٧٨حدثنا هناد، حدثنا عبدة بن سليمان، عن عبد الملك بن أبي سليمان، عن سعيد…٣١٧٩البينة وإلا حد في ظهرك٣١٨٠أما بعد أشيروا على في أناس أبنوا أهلي والله ما علمت على أهلي من سوء…٣١٨١قالت لما نزل عذري قام رسول الله صلى الله عليه وسلم على المنبر فذكر…٣١٨٢أن تجعل لله ندا وهو خلقك٣١٨٣أن تجعل لله ندا وهو خلقك وأن تقتل ولدك من أجل أن يأكل معك أو من طعامك…٣١٨٤يا صفية بنت عبد المطلب يا فاطمة بنت محمد يا بني عبد المطلب إني لا…٣١٨٥يا معشر قريش أنقذوا أنفسكم من النار فإني لا أملك لكم من الله ضرا ولا…٣١٨٦يا بني عبد مناف يا صباحاه٣١٨٧تخرج الدابة معها خاتم سليمان وعصا موسى فتجلو وجه المؤمن وتختم أنف…٣١٨٨قل لا إله إلا الله أشهد لك بها يوم القيامة٣١٨٩ومحمد بن المثنى، قالا حدثنا محمد بن جعفر، حدثنا شعبة، عن سماك بن حرب،…٣١٩٠كانوا يخذفون أهل الأرض ويسخرون منهم٣١٩١ألا احتطت يا أبا بكر فإن البضع ما بين الثلاث إلى التسع٣١٩٢قال لما كان يوم بدر ظهرت الروم على فارس فأعجب ذلك المؤمنين فنزلت : (…٣١٩٣أما إنهم سيغلبون٣١٩٤حدثنا محمد بن إسماعيل، حدثنا إسماعيل بن أبي أويس، حدثني ابن أبي…٣١٩٥لا تبيعوا القينات ولا تشتروهن ولا تعلموهن ولا خير في تجارة فيهن…٣١٩٦أن هذه الآية : ( تتجافى، جنوبهم عن المضاجع، ) نزلت في انتظار هذه…٣١٩٧قال الله تعالى أعددت لعبادي الصالحين ما لا عين رأت ولا أذن سمعت ولا…٣١٩٨إن موسى عليه السلام سأل ربه فقال أى رب أى أهل الجنة أدنى منزلة قال…٣١٩٩أن أباه، حدثه قال قلنا لابن عباس أرأيت قول الله عز وجل : ( ما جعل…٣٢٠٠قال قال عمي أنس بن النضر سميت به لم يشهد بدرا مع رسول الله صلى الله…٣٢٠١أن عمه، غاب عن قتال، بدر فقال غبت عن أول، قتال قاتله رسول الله صلى…٣٢٠٢طلحة ممن قضى نحبه٣٢٠٣أين السائل عمن قضى نحبه٣٢٠٤يا عائشة إني ذاكر لك أمرا فلا عليك أن لا تستعجلي حتى تستأمري أبويك٣٢٠٥اللهم هؤلاء أهل بيتي فأذهب عنهم الرجس وطهرهم تطهيرا٣٢٠٦الصلاة يا أهل البيت : ( إنما يريد الله ليذهب عنكم الرجس أهل البيت…٣٢٠٧حدثنا علي بن حجر، أخبرنا داود بن الزبرقان، عن داود بن أبي هند، عن…٣٢٠٨رضى الله عنها قالت لو كان النبي صلى الله عليه وسلم كاتما شيئا من…٣٢٠٩حدثنا قتيبة، حدثنا يعقوب بن عبد الرحمن، عن موسى بن عقبة، عن سالم، عن…٣٢١٠في قول الله عز وجل : ( ما كان محمد أبا أحد من رجالكم ) قال ما كان…٣٢١١أنها أتت النبي صلى الله عليه وسلم فقالت ما أرى كل شيء إلا للرجال وما…٣٢١٢أمسك عليك زوجك واتق الله٣٢١٣قال نزلت هذه الآية في زينب بنت جحش : ( فلما قضى زيد منها وطرا…٣٢١٤حدثنا عبد بن حميد، حدثنا عبيد الله بن موسى، عن إسرائيل، عن السدي، عن…٣٢١٥حدثنا عبد بن حميد، حدثنا روح، عن عبد الحميد بن بهرام، عن شهر بن حوشب،…٣٢١٦قال قالت عائشة ما مات رسول الله صلى الله عليه وسلم حتى أحل له النساء…٣٢١٧قال كنت عند النبي صلى الله عليه وسلم فأتى باب امرأة عرس بها فإذا…٣٢١٨حدثنا قتيبة بن سعيد، حدثنا جعفر بن سليمان الضبعي، عن الجعد بن عثمان،…٣٢١٩حدثنا عمر بن إسماعيل بن مجالد، حدثني أبي، عن بيان، عن أنس بن مالك،…٣٢٢٠قولوا اللهم صل على محمد وعلى آل محمد كما صليت على آل إبراهيم وبارك…٣٢٢١أن موسى عليه السلام كان رجلا حييا ستيرا ما يرى من جلده شيء استحياء…٣٢٢٢ادع القوم فمن أسلم منهم فاقبل منه ومن لم يسلم فلا تعجل حتى أحدث إليك٣٢٢٣إذا قضى الله في السماء أمرا ضربت الملائكة بأجنحتها خضعانا لقوله كأنها…٣٢٢٤ما كنتم تقولون لمثل هذا في الجاهلية إذا رأيتموه٣٢٢٥هؤلاء كلهم بمنزلة واحدة وكلهم في الجنة٣٢٢٦إن آثاركم تكتب فلا ينتقلوا٣٢٢٧أتدري يا أبا ذر أين تذهب هذه٣٢٢٨ما من داع دعا إلى شيء إلا كان موقوفا يوم القيامة لازما له لا يفارقه…٣٢٢٩حدثنا علي بن حجر، أخبرنا الوليد بن مسلم، عن زهير بن محمد، عن رجل، عن…٣٢٣٠حدثنا محمد بن المثنى، حدثنا محمد بن خالد بن عثمة، حدثنا سعيد بن بشير،…٣٢٣١سام أبو العرب وحام أبو الحبش ويافث أبو الروم٣٢٣٢إني أريد منهم كلمة واحدة تدين لهم بها العرب وتؤدي إليهم العجم الجزية٣٢٣٣أتاني الليلة ربي تبارك وتعالى في أحسن صورة قال أحسبه قال في المنام…٣٢٣٤أتاني ربي في أحسن صورة فقال يا محمد قلت لبيك ربي وسعديك قال فيم يختصم…٣٢٣٥إني نعست فاستثقلت نوما فرأيت ربي في أحسن صورة فقال فيم يختصم الملأ…٣٢٣٦حدثنا ابن أبي عمر، حدثنا سفيان، عن محمد بن عمرو بن علقمة، عن يحيى بن…٣٢٣٧وسليمان بن حرب، وحجاج بن منهال، قالوا حدثنا حماد بن سلمة، عن ثابت، عن…٣٢٣٨وسليمان الأعمش، عن إبراهيم، عن عبيدة، عن عبد الله، قال جاء يهودي إلى…٣٢٣٩قال فضحك النبي صلى الله عليه وسلم تعجبا وتصديقا . قال هذا حديث حسن…٣٢٤٠يا يهودي حدثنا٣٢٤١حدثنا سويد بن نصر، حدثنا عبد الله بن المبارك، عن عنبسة بن سعيد، عن…٣٢٤٢على الصراط يا عائشة٣٢٤٣كيف أنعم
