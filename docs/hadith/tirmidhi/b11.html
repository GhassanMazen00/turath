كتاب النكاح عن رسول الله صلى الله عليه وسلم

٦٧ حديثًا
١٠٨٠أربع من سنن المرسلين الحياء والتعطر والسواك والنكاح١٠٨١يا معشر الشباب عليكم بالباءة فإنه أغض للبصر وأحصن للفرج فمن لم يستطع…١٠٨٢وزيد بن أخزم الطائي، وإسحاق بن إبراهيم البصري، قالوا حدثنا معاذ بن…١٠٨٣وغير، واحد، قالوا أخبرنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن سعيد…١٠٨٤إذا خطب إليكم من ترضون دينه وخلقه فزوجوه إلا تفعلوا تكن فتنة في الأرض…١٠٨٥إذا جاءكم من ترضون دينه وخلقه فأنكحوه إلا تفعلوا تكن فتنة في الأرض…١٠٨٦إن المرأة تنكح على دينها ومالها وجمالها فعليك بذات الدين تربت يداك١٠٨٧انظر إليها فإنه أحرى أن يؤدم بينكما١٠٨٨فصل ما بين الحرام والحلال الدف والصوت١٠٨٩أعلنوا هذا النكاح واجعلوه في المساجد واضربوا عليه بالدفوف١٠٩٠اسكتي عن هذه وقولي الذي كنت تقولين قبلها١٠٩١بارك الله لك وبارك عليك وجمع بينكما في خير١٠٩٢لو أن أحدكم إذا أتى أهله قال بسم الله اللهم جنبنا الشيطان وجنب…١٠٩٣حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، حدثنا سفيان، عن إسماعيل بن…١٠٩٤حدثنا قتيبة، حدثنا حماد بن زيد، عن ثابت، عن أنس بن مالك، أن رسول الله…١٠٩٥أن النبي صلى الله عليه وسلم أولم على صفية بنت حيى بسويق وتمر . قال…١٠٩٦نحو هذا . وقد روى غير، واحد، هذا الحديث عن ابن عيينة، عن الزهري، عن…١٠٩٧طعام أول يوم حق وطعام يوم الثاني سنة وطعام يوم الثالث سمعة ومن سمع…١٠٩٨ائتوا الدعوة إذا دعيتم١٠٩٩إنه اتبعنا رجل لم يكن معنا حين دعوتنا فإن أذنت له دخل١١٠٠أتزوجت يا جابر١١٠١لا نكاح إلا بولي١١٠٢أيما امرأة نكحت بغير إذن وليها فنكاحها باطل فنكاحها باطل فنكاحها باطل…١١٠٣البغايا اللاتي ينكحن أنفسهن بغير بينة١١٠٤حدثنا قتيبة، حدثنا غندر، محمد بن جعفر عن سعيد بن أبي عروبة، نحوه ولم…١١٠٥التشهد في الصلاة التحيات لله والصلوات والطيبات السلام عليك أيها النبي…١١٠٦كل خطبة ليس فيها تشهد فهي كاليد الجذماء١١٠٧لا تنكح الثيب حتى تستأمر ولا تنكح البكر حتى تستأذن وإذنها الصموت١١٠٨الأيم أحق بنفسها من وليها والبكر تستأذن في نفسها وإذنها صماتها١١٠٩اليتيمة تستأمر في نفسها فإن صمتت فهو إذنها وإن أبت فلا جواز عليها١١١٠أيما امرأة زوجها وليان فهي للأول منهما ومن باع بيعا من رجلين فهو…١١١١أيما عبد تزوج بغير إذن سيده فهو عاهر١١١٢أيما عبد تزوج بغير إذن سيده فهو عاهر١١١٣أرضيت من نفسك ومالك بنعلين١١١٤هل عندك من شيء تصدقها١١١٤.٢قال قال عمر بن الخطاب ألا لا تغالوا صدقة النساء فإنها لو كانت مكرمة…١١١٥وعبد العزيز بن صهيب، عن أنس بن مالك، أن رسول الله صلى الله عليه وسلم…١١١٦ثلاثة يؤتون أجرهم مرتين عبد أدى حق الله وحق مواليه فذلك يؤتى أجره…١١١٧أيما رجل نكح امرأة فدخل بها فلا يحل له نكاح ابنتها فإن لم يكن دخل بها…١١١٨أتريدين أن ترجعي إلى رفاعة لا حتى تذوقي عسيلته ويذوق عسيلتك١١١٩حدثنا أبو سعيد الأشج، حدثنا أشعث بن عبد الرحمن بن زبيد الأيامي، حدثنا…١١٢٠حدثنا محمود بن غيلان، حدثنا أبو أحمد الزبيري، حدثنا سفيان، عن أبي…١١٢١والحسن، ابنى محمد بن علي عن أبيهما، عن علي بن أبي طالب، أن النبي صلى…١١٢٢قال إنما كانت المتعة في أول الإسلام كان الرجل يقدم البلدة ليس له بها…١١٢٣لا جلب ولا جنب ولا شغار في الإسلام ومن انتهب نهبة فليس منا١١٢٤حدثنا إسحاق بن موسى الأنصاري، حدثنا معن، حدثنا مالك، عن نافع، عن ابن…١١٢٥حدثنا نصر بن علي الجهضمي، حدثنا عبد الأعلى بن عبد الأعلى، حدثنا سعيد…١١٢٦حدثنا الحسن بن علي الخلال، حدثنا يزيد بن هارون، أنبأنا داود بن أبي…١١٢٧إن أحق الشروط أن يوفى بها ما استحللتم به الفروج١١٢٨أن غيلان بن سلمة الثقفي، أسلم وله عشر نسوة في الجاهلية فأسلمن معه…١١٢٩اختر أيتهما شئت١١٣٠اختر أيتهما شئت١١٣١من كان يؤمن بالله واليوم الآخر فلا يسق ماءه ولد غيره١١٣٢.١١٣٣حدثنا قتيبة، حدثنا الليث، عن ابن شهاب، عن أبي بكر بن عبد الرحمن، عن…١١٣٤لا يبيع الرجل على بيع أخيه ولا يخطب على خطبة أخيه١١٣٥حدثنا محمود بن غيلان، حدثنا أبو داود، قال أنبأنا شعبة، قال أخبرني أبو…١١٣٦كذبت اليهود إن الله إذا أراد أن يخلقه لم يمنعه١١٣٧وابن أبي عمر، قالا حدثنا سفيان بن عيينة، عن عمرو بن دينار، عن عطاء،…١١٣٨لم يفعل ذلك أحدكم١١٣٩قال لو شئت أن أقول، قال رسول الله صلى الله عليه وسلم ولكنه قال السنة…١١٤٠اللهم هذه قسمتي فيما أملك فلا تلمني فيما تملك ولا أملك١١٤١إذا كان عند الرجل امرأتان فلم يعدل بينهما جاء يوم القيامة وشقه ساقط١١٤٢وهناد، قالا حدثنا أبو معاوية، عن الحجاج، عن عمرو بن شعيب، عن أبيه، عن…١١٤٣قال رد النبي صلى الله عليه وسلم ابنته زينب على أبي العاصي بن الربيع…١١٤٤أن رجلا، جاء مسلما على عهد النبي صلى الله عليه وسلم ثم جاءت امرأته…١١٤٥وعبد الرزاق، كلاهما عن سفيان، عن منصور، نحوه . قال أبو عيسى حديث ابن…
