٦٨ حديثًا
١٧٢٠حرم لباس الحرير والذهب على ذكور أمتي وأحل لإناثهم١٧٢١حدثنا محمد بن بشار، حدثنا معاذ بن هشام، حدثنا أبي، عن قتادة، عن…١٧٢٢أن عبد الرحمن بن عوف، والزبير بن العوام، شكيا القمل إلى النبي صلى…١٧٢٣أتعجبون من هذه لمناديل سعد في الجنة خير مما ترون١٧٢٤حدثنا محمود بن غيلان، حدثنا وكيع، حدثنا سفيان، عن أبي إسحاق، عن…١٧٢٥حدثنا قتيبة، حدثنا مالك بن أنس، عن نافع، عن إبراهيم بن عبد الله بن…١٧٢٦الحلال ما أحل الله في كتابه والحرام ما حرم الله في كتابه وما سكت عنه…١٧٢٧ألا نزعتم جلدها ثم دبغتموه فاستمتعتم به١٧٢٨أيما إهاب دبغ فقد طهر١٧٢٩أن لا تنتفعوا من الميتة بإهاب ولا عصب١٧٣٠لا ينظر الله يوم القيامة إلى من جر ثوبه خيلاء١٧٣١من جر ثوبه خيلاء لم ينظر الله إليه يوم القيامة١٧٣٢أن أم سلمة، حدثتهم أن النبي صلى الله عليه وسلم شبر لفاطمة شبرا من…١٧٣٣حدثنا أحمد بن منيع، حدثنا إسماعيل بن إبراهيم، حدثنا أيوب، عن حميد بن…١٧٣٤كان على موسى يوم كلمه ربه كساء صوف وجبة صوف وكمة صوف وسراويل صوف…١٧٣٥حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، عن حماد بن سلمة، عن أبي…١٧٣٦حدثنا هارون بن إسحاق الهمداني، حدثنا يحيى بن محمد المدني، عن عبد…١٧٣٧والحسن بن علي الخلال، وغير، واحد، قالوا حدثنا عبد الرزاق، أخبرنا…١٧٣٨حدثنا يوسف بن حماد المعني البصري، حدثنا عبد الوارث بن سعيد، عن أبي…١٧٣٩وغير، واحد، عن عبد الله بن وهب، عن يونس، عن ابن شهاب، عن أنس، قال كان…١٧٤٠قال كان خاتم رسول الله صلى الله عليه وسلم من فضة فصه منه . قال أبو…١٧٤١إني كنت اتخذت هذا الخاتم في يميني١٧٤٢حدثنا محمد بن حميد الرازي، حدثنا جرير، عن محمد بن إسحاق، عن الصلت بن…١٧٤٣قال كان الحسن والحسين يتختمان في يسارهما . وهذا حديث حسن صحيح .١٧٤٤فقال رأيت عبد الله بن جعفر يتختم في يمينه . وقال عبد الله بن جعفر كان…١٧٤٥أن النبي صلى الله عليه وسلم صنع خاتما من ورق فنقش فيه محمد رسول الله…١٧٤٦والحجاج بن منهال، قالا حدثنا همام، عن ابن جريج، عن الزهري، عن أنس،…١٧٤٧قال كان نقش خاتم النبي صلى الله عليه وسلم محمد سطر ورسول سطر والله…١٧٤٨ومحمد بن يحيى، وغير، واحد، قالوا حدثنا محمد بن عبد الله الأنصاري،…١٧٤٩حدثنا أحمد بن منيع، حدثنا روح بن عبادة، حدثنا ابن جريج، أخبرني أبو…١٧٥٠إلا ما كان رقما في ثوب١٧٥١من صور صورة عذبه الله حتى ينفخ فيها يعني الروح وليس بنافخ فيها ومن…١٧٥٢غيروا الشيب ولا تشبهوا باليهود١٧٥٣إن أحسن ما غير به الشيب الحناء والكتم١٧٥٤حدثنا حميد بن مسعدة، حدثنا عبد الوهاب الثقفي، عن حميد، عن أنس، قال…١٧٥٥حدثنا هناد، حدثنا عبد الرحمن بن أبي الزناد، عن هشام بن عروة، عن أبيه،…١٧٥٦حديث ١٧٥٦١٧٥٧حديث ١٧٥٧١٧٥٨حدثنا قتيبة، حدثنا يعقوب بن عبد الرحمن الإسكندراني، عن سهيل بن أبي…١٧٥٩لعن الله الواصلة والمستوصلة والواشمة والمستوشمة١٧٦٠حدثنا علي بن حجر، أخبرنا علي بن مسهر، حدثنا أبو إسحاق الشيباني، عن…١٧٦١حدثنا علي بن حجر، أخبرنا علي بن مسهر، عن هشام بن عروة، عن أبيه، عن…١٧٦٢والفضل بن موسى، وزيد بن حباب، عن عبد المؤمن بن خالد، عن عبد الله بن…١٧٦٣حدثنا زياد بن أيوب البغدادي، حدثنا أبو تميلة، عن عبد المؤمن بن خالد،…١٧٦٤قالت كان أحب الثياب إلى رسول الله صلى الله عليه وسلم القميص .١٧٦٥قالت كان كم يد رسول الله صلى الله عليه وسلم إلى الرسغ . قال أبو عيسى…١٧٦٦حدثنا نصر بن علي الجهضمي، حدثنا عبد الصمد بن عبد الوارث، حدثنا شعبة،…١٧٦٧حديث ١٧٦٧١٧٦٨أن النبي صلى الله عليه وسلم لبس جبة رومية ضيقة الكمين . قال أبو عيسى…١٧٦٩حدثنا قتيبة، حدثنا ابن أبي زائدة، عن الحسن بن عياش، عن أبي إسحاق، هو…١٧٧٠حديث ١٧٧٠١٧٧١حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، عن يزيد الرشك، عن…١٧٧٢قال قلت لأنس بن مالك كيف كان نعل رسول الله صلى الله عليه وسلم قال…١٧٧٣حدثنا إسحاق بن منصور، أخبرنا حبان بن هلال، حدثنا همام، حدثنا قتادة،…١٧٧٤لا يمشي أحدكم في نعل واحدة لينعلهما جميعا أو ليحفهما جميعا١٧٧٥حدثنا أزهر بن مروان البصري، حدثنا الحارث بن نبهان، عن معمر، عن عمار…١٧٧٦حدثنا أبو جعفر السمناني، حدثنا سليمان بن عبيد الله الرقي، حدثنا عبيد…١٧٧٧قالت ربما مشى النبي صلى الله عليه وسلم في نعل واحدة .١٧٧٨حدثنا أحمد بن منيع، حدثنا سفيان بن عيينة، عن عبد الرحمن بن القاسم، عن…١٧٧٩إذا انتعل أحدكم فليبدأ باليمين وإذا نزع فليبدأ بالشمال فلتكن اليمنى…١٧٨٠إذا أردت اللحوق بي فليكفيك من الدنيا كزاد الراكب وإياك ومجالسة…١٧٨١حديث ١٧٨١١٧٨٢يقول كانت كمام أصحاب رسول الله صلى الله عليه وسلم بطحا . قال أبو عيسى…١٧٨٣هذا موضع الإزار فإن أبيت فأسفل فإن أبيت فلا حق للإزار في الكعبين١٧٨٤إن فرق ما بيننا وبين المشركين العمائم على القلانس١٧٨٥ما لي أرى عليك حلية أهل النار١٧٨٦حدثنا ابن أبي عمر، حدثنا سفيان، عن عاصم بن كليب، عن ابن أبي موسى، قال…١٧٨٧قال كان أحب الثياب إلى رسول الله صلى الله عليه وسلم يلبسها الحبرة .…
