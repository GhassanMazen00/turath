١٠٨ حديثًا
٢٤١٥ما منكم من رجل إلا سيكلمه ربه يوم القيامة وليس بينه وبينه ترجمان…٢٤١٦لا تزول قدما ابن آدم يوم القيامة من عند ربه حتى يسأل عن خمس عن عمره…٢٤١٧لا تزول قدما عبد يوم القيامة حتى يسأل عن عمره فيما أفناه وعن علمه…٢٤١٨أتدرون ما المفلس٢٤١٩رحم الله عبدا كانت لأخيه عنده مظلمة في عرض أو مال فجاءه فاستحله قبل…٢٤٢٠لتؤدن الحقوق إلى أهلها حتى يقاد للشاة الجلحاء من الشاة القرناء٢٤٢١إذا كان يوم القيامة أدنيت الشمس من العباد حتى تكون قيد ميل أو اثنين٢٤٢٢حدثنا أبو زكريا، يحيى بن درست البصري حدثنا حماد بن زيد، عن أيوب، عن…٢٤٢٣يحشر الناس يوم القيامة حفاة عراة غرلا كما خلقوا ثم قرأ ( كما بدأنا…٢٤٢٤إنكم محشورون رجالا وركبانا وتجرون على وجوهكم٢٤٢٥يعرض الناس يوم القيامة ثلاث عرضات فأما عرضتان فجدال ومعاذير وأما…٢٤٢٦من نوقش الحساب هلك٢٤٢٧يجاء بابن آدم يوم القيامة كأنه بذج فيوقف بين يدى الله فيقول الله له…٢٤٢٨يؤتى بالعبد يوم القيامة فيقول الله له ألم أجعل لك سمعا وبصرا ومالا…٢٤٢٩أتدرون ما أخبارها٢٤٣٠حدثنا سويد بن نصر، أخبرنا عبد الله بن المبارك، أخبرنا سليمان التيمي،…٢٤٣١كيف أنعم وصاحب القرن قد التقم القرن واستمع الإذن متى يؤمر بالنفخ فينفخ٢٤٣٢شعار المؤمن على الصراط رب سلم سلم٢٤٣٣حدثنا عبد الله بن الصباح الهاشمي، حدثنا بدل بن المحبر، حدثنا حرب بن…٢٤٣٤أنا سيد الناس يوم القيامة هل تدرون لم ذاك يجمع الله الناس الأولين…٢٤٣٥شفاعتي لأهل الكبائر من أمتي٢٤٣٦شفاعتي لأهل الكبائر من أمتي٢٤٣٧وعدني ربي أن يدخل الجنة من أمتي سبعين ألفا لا حساب عليهم ولا عذاب مع…٢٤٣٨يدخل الجنة بشفاعة رجل من أمتي أكثر من بني تميم٢٤٣٩يشفع عثمان بن عفان يوم القيامة في مثل ربيعة ومضر٢٤٤٠إن من أمتي من يشفع للفئام ومنهم من يشفع للقبيلة ومنهم من يشفع للعصبة…٢٤٤١أتاني آت من عند ربي فخيرني بين أن يدخل نصف أمتي الجنة وبين الشفاعة…٢٤٤٢إن في حوضي من الأباريق بعدد نجوم السماء٢٤٤٣إن لكل نبي حوضا وإنهم يتباهون أيهم أكثر واردة وإني أرجو أن أكون…٢٤٤٤حوضي من عدن إلى عمان البلقاء ماؤه أشد بياضا من اللبن وأحلى من العسل…٢٤٤٥والذي نفسي بيده لآنيته أكثر من عدد نجوم السماء وكواكبها في ليلة مظلمة…٢٤٤٦هم الذين لا يكتوون ولا يسترقون ولا يتطيرون وعلى ربهم يتوكلون٢٤٤٧حدثنا محمد بن عبد الله بن بزيع، حدثنا زياد بن الربيع، حدثنا أبو عمران…٢٤٤٨بئس العبد عبد تخيل واختال ونسي الكبير المتعال بئس العبد عبد تجبر…٢٤٤٩أيما مؤمن أطعم مؤمنا على جوع أطعمه الله يوم القيامة من ثمار الجنة…٢٤٥٠من خاف أدلج ومن أدلج بلغ المنزل ألا إن سلعة الله غالية ألا إن سلعة…٢٤٥١لا يبلغ العبد أن يكون من المتقين حتى يدع ما لا بأس به حذرا لما به…٢٤٥٢لو أنكم تكونون كما تكونون عندي لأظلتكم الملائكة بأجنحتها٢٤٥٣إن لكل شيء شرة ولكل شرة فترة فإن كان صاحبها سدد وقارب فارجوه وإن أشير…٢٤٥٤هذا ابن آدم وهذا أجله محيط به وهذا الذي في الوسط الإنسان وهذه الخطوط…٢٤٥٥يهرم ابن آدم ويشب منه اثنان الحرص على المال والحرص على العمر٢٤٥٦مثل ابن آدم وإلى جنبه تسعة وتسعون منية إن أخطأته المنايا وقع في الهرم٢٤٥٧يا أيها الناس اذكروا الله اذكروا الله جاءت الراجفة تتبعها الرادفة جاء…٢٤٥٨استحيوا من الله حق الحياء٢٤٥٩الكيس من دان نفسه وعمل لما بعد الموت والعاجز من أتبع نفسه هواها وتمنى…٢٤٦٠أما إنكم لو أكثرتم ذكر هاذم اللذات لشغلكم عما أرى فأكثروا من ذكر هاذم…٢٤٦١قال دخلت على رسول الله صلى الله عليه وسلم فإذا هو متكئ على رمل حصير…٢٤٦٢أظنكم سمعتم أن أبا عبيدة قدم بشيء٢٤٦٣يا حكيم إن هذا المال خضرة حلوة فمن أخذه بسخاوة نفس بورك له فيه ومن…٢٤٦٤حدثنا قتيبة، حدثنا أبو صفوان، عن يونس، عن الزهري، عن حميد بن عبد…٢٤٦٥من كانت الآخرة همه جعل الله غناه في قلبه وجمع له شمله وأتته الدنيا…٢٤٦٦إن الله تعالى يقول يا ابن آدم تفرغ لعبادتي أملأ صدرك غنى وأسد فقرك…٢٤٦٧حدثنا هناد، حدثنا أبو معاوية، عن هشام بن عروة، عن أبيه، عن عائشة،…٢٤٦٨انزعيه فإنه يذكرني الدنيا٢٤٦٩قالت كانت وسادة رسول الله صلى الله عليه وسلم التي يضطجع عليها من أدم…٢٤٧٠أنهم ذبحوا شاة فقال النبي صلى الله عليه وسلم " ما بقي منها " . قالت…٢٤٧١حدثنا هارون بن إسحاق الهمداني، حدثنا عبدة، عن هشام بن عروة، عن أبيه،…٢٤٧٢لقد أخفت في الله وما يخاف أحد ولقد أوذيت في الله وما يؤذى أحد ولقد…٢٤٧٣سمع علي بن أبي طالب، يقول خرجت في يوم شات من بيت رسول الله صلى الله…٢٤٧٤أنه أصابهم جوع فأعطاهم رسول الله صلى الله عليه وسلم تمرة تمرة . قال…٢٤٧٥حدثنا هناد، حدثنا عبدة، عن هشام بن عروة، عن وهب بن كيسان، عن جابر بن…٢٤٧٦كيف بكم إذا غدا أحدكم في حلة وراح في حلة ووضعت بين يديه صحفة ورفعت…٢٤٧٧حدثنا هناد، حدثنا يونس بن بكير، حدثني عمر بن ذر، حدثنا مجاهد، عن أبي…٢٤٧٨كف عنا جشاءك فإن أكثرهم شبعا في الدنيا أطولهم جوعا يوم القيامة٢٤٧٩حدثنا قتيبة، حدثنا أبو عوانة، عن قتادة، عن أبي بردة بن أبي موسى، عن…٢٤٨٠قال البناء كله وبال قلت أرأيت ما لا بد منه قال لا أجر ولا وزر .٢٤٨١من ترك اللباس تواضعا لله وهو يقدر عليه دعاه الله يوم القيامة على رءوس…٢٤٨٢النفقة كلها في سبيل الله إلا البناء فلا خير فيه٢٤٨٣حدثنا علي بن حجر، أخبرنا شريك، عن أبي إسحاق، عن حارثة بن مضرب، قال…٢٤٨٤ما من مسلم كسا مسلما ثوبا إلا كان في حفظ الله ما دام منه عليه خرقة٢٤٨٥أيها الناس أفشوا السلام وأطعموا الطعام وصلوا والناس نيام تدخلون الجنة…٢٤٨٦الطاعم الشاكر بمنزلة الصائم الصابر٢٤٨٧لا ما دعوتم الله لهم وأثنيتم عليهم٢٤٨٨ألا أخبركم بمن يحرم على النار أو بمن تحرم عليه النار على كل قريب هين…٢٤٨٩قال قلت لعائشة أى شيء كان النبي صلى الله عليه وسلم يصنع إذا دخل بيته…٢٤٩٠حدثنا سويد بن نصر، أخبرنا عبد الله بن المبارك، عن عمران بن زيد…٢٤٩١خرج رجل ممن كان قبلكم في حلة له يختال فيها فأمر الله الأرض فأخذته فهو…٢٤٩٢يحشر المتكبرون يوم القيامة أمثال الذر في صور الرجال يغشاهم الذل من كل…٢٤٩٣من كظم غيظا وهو يقدر على أن ينفذه دعاه الله على رءوس الخلائق يوم…٢٤٩٤ثلاث من كن فيه نشر الله عليه كنفه وأدخله جنته رفق بالضعيف وشفقة على…٢٤٩٥يقول الله تعالى يا عبادي كلكم ضال إلا من هديته فسلوني الهدى أهدكم…٢٤٩٦كان الكفل من بني إسرائيل لا يتورع من ذنب عمله فأتته امرأة فأعطاها…٢٤٩٧والآخر، عن النبي صلى الله عليه وسلم قال عبد الله إن المؤمن يرى ذنوبه…٢٤٩٨لله أفرح بتوبة أحدكم من رجل بأرض دوية مهلكة معه راحلته�� عليها زاده…٢٤٩٩كل ابن آدم خطاء وخير الخطائين التوابون٢٥٠٠من كان يؤمن بالله واليوم الآخر فليكرم ضيفه ومن كان يؤمن بالله واليوم…٢٥٠١قال قال رسول الله صلى الله عليه وسلم " من صمت نجا " . قال أبو عيسى…٢٥٠٢ما يسرني أني حكيت رجلا وأن لي كذا وكذا٢٥٠٣ما أحب أني حكيت أحدا وأن لي كذا وكذا٢٥٠٤من سلم المسلمون من لسانه ويده٢٥٠٥من عير أخاه بذنب لم يمت حتى يعمله٢٥٠٦لا تظهر الشماتة لأخيك فيرحمه الله ويبتليك٢٥٠٧المسلم إذا كان يخالط الناس ويصبر على أذاهم خير من المسلم الذي لا…٢٥٠٨إياكم وسوء ذات البين فإنها الحالقة٢٥٠٩ألا أخبركم بأفضل من درجة الصيام والصلاة والصدقة٢٥١٠دب إليكم داء الأمم قبلكم الحسد والبغضاء هي الحالقة لا أقول تحلق الشعر…٢٥١١ما من ذنب أجدر أن يعجل الله لصاحبه العقوبة في الدنيا مع ما يدخر له في…٢٥١٢خصلتان من كانتا فيه كتبه الله شاكرا صابرا ومن لم تكونا فيه لم يكتبه…٢٥١٣انظروا إلى من هو أسفل منكم ولا تنظروا إلى من هو فوقكم فإنه أجدر أن لا…٢٥١٤واحد، عن أبي عثمان النهدي، عن حنظلة الأسيدي، وكان، من كتاب النبي صلى…٢٥١٥لا يؤمن أحدكم حتى يحب لأخيه ما يحب لنفسه٢٥١٦يا غلام إني أعلمك كلمات احفظ الله يحفظك احفظ الله تجده تجاهك إذا سألت…٢٥١٧حدثنا أبو حفص، عمرو بن علي حدثنا يحيى بن سعيد القطان، حدثنا المغيرة…٢٥١٨دع ما يريبك إلى ما لا يريبك فإن الصدق طمأنينة وإن الكذب ريبة٢٥١٩قال ذكر رجل عند النبي صلى الله عليه وسلم بعبادة واجتهاد وذكر عنده آخر…٢٥٢٠من أكل طيبا وعمل في سنة وأمن الناس بوائقه دخل الجنة٢٥٢١من أعطى لله ومنع لله وأحب لله وأبغض لله وأنكح لله فقد استكمل إيمانه٢٥٢٢أول زمرة تدخل الجنة على صورة القمر ليلة البدر ، والثانية على لون أحسن…
