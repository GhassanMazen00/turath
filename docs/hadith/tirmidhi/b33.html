كتاب الفتن عن رسول الله صلى الله عليه وسلم

١١٢ حديثًا
٢١٥٨لا يحل دم امرئ مسلم إلا بإحدى ثلاث زنا بعد إحصان أو ارتداد بعد إسلام…٢١٥٩حدثنا هناد، حدثنا أبو الأحوص، عن شبيب بن غرقدة، عن سليمان بن عمرو بن…٢١٦٠لا يأخذ أحدكم عصا أخيه لاعبا أو جادا فمن أخذ عصا أخيه فليردها إليه٢١٦١حدثنا قتيبة حدثنا حاتم بن إسماعيل عن محمد بن يوسف عن السائب بن يزيد…٢١٦٢من أشار على أخيه بحديدة لعنته الملائكة٢١٦٣حدثنا عبد الله بن معاوية الجمحي البصري، حدثنا حماد بن سلمة، عن أبي…٢١٦٤من صلى الصبح فهو في ذمة الله فلا يتبعنكم الله بشيء من ذمته٢١٦٥أوصيكم بأصحابي ثم الذين يلونهم ثم الذين يلونهم ثم يفشو الكذب حتى يحلف…٢١٦٦يد الله مع الجماعة٢١٦٧إن الله لا يجمع أمتي - أو قال أمة محمد صلى الله عليه وسلم - على ضلالة…٢١٦٨إن الناس إذا رأوا الظالم فلم يأخذوا على يديه أوشك أن يعمهم الله بعقاب…٢١٦٩والذي نفسي بيده لتأمرن بالمعروف ولتنهون عن المنكر أو ليوشكن الله أن…٢١٧٠والذي نفسي بيده لا تقوم الساعة حتى تقتلوا إمامكم وتجتلدوا بأسيافكم…٢١٧١إنهم يبعثون على نياتهم٢١٧٢من رأى منكرا فلينكره بيده ومن لم يستطع فبلسانه ومن لم يستطع فبقلبه…٢١٧٣مثل القائم على حدود الله والمدهن فيها كمثل قوم استهموا على سفينة في…٢١٧٤إن من أعظم الجهاد كلمة عدل عند سلطان جائر٢١٧٥أجل إنها صلاة رغبة ورهبة إني سألت الله فيها ثلاثا فأعطاني اثنتين…٢١٧٦إن الله زوى لي الأرض فرأيت مشارقها ومغاربها وإن أمتي سيبلغ ملكها ما…٢١٧٧رجل في ماشيته يؤدي حقها ويعبد ربه ورجل آخذ برأس فرسه يخيف العدو…٢١٧٨تكون فتنة تستنظف العرب قتلاها في النار اللسان فيها أشد من السيف٢١٧٩أن الأمانة نزلت في جذر قلوب الرجال ثم نزل القرآن فعلموا من القرآن…٢١٨٠سبحان الله هذا كما قال قوم موسى : ( اجعل لنا إلها كما لهم آلهة )…٢١٨١والذي نفسي بيده لا تقوم الساعة حتى تكلم السباع الإنس وحتى تكلم الرجل…٢١٨٢حدثنا محمود بن غيلان، حدثنا أبو داود، عن شعبة، عن الأعمش، عن مجاهد،…٢١٨٣لا تقوم الساعة حتى تروا عشر آيات طلوع الشمس من مغربها ويأجوج ومأجوج…٢١٨٤لا ينتهي الناس عن غزو هذا البيت حتى يغزو جيش حتى إذا كانوا بالبيداء…٢١٨٥يكون في آخر هذه الأمة خسف ومسخ وقذف٢١٨٦يا أبا ذر أتدري أين تذهب هذه٢١٨٧لا إله إلا الله يرددها ثلاث مرات ويل للعرب من شر قد اقترب فتح اليوم…٢١٨٨يخرج في آخر الزمان قوم أحداث الأسنان سفهاء الأحلام يقرءون القرآن لا…٢١٨٩إنكم سترون بعدي أثرة فاصبروا حتى تلقوني على الحوض٢١٩٠إنكم سترون بعدي أثرة وأمورا تنكرونها٢١٩١إن الدنيا حلوة خضرة وإن الله مستخلفكم فيها فناظر كيف تعملون ألا…٢١٩٢إذا فسد أهل الشام فلا خير فيكم لا تزال طائفة من أمتي منصورين لا يضرهم…٢١٩٣لا ترجعوا بعدي كفارا يضرب بعضكم رقاب بعض٢١٩٤إنها ستكون فتنة القاعد فيها خير من القائم والقائم خير من الماشي…٢١٩٥بادروا بالأعمال فتنا كقطع الليل المظلم يصبح الرجل مؤمنا ويمسي كافرا…٢١٩٦سبحان الله ماذا أنزل الليلة من الفتنة ماذا أنزل من الخزائن من يوقظ…٢١٩٧تكون بين يدى الساعة فتن كقطع الليل المظلم يصبح الرجل فيها مؤمنا ويمسي…٢١٩٨يصبح الرجل مؤمنا ويمسي كافرا ويمسي مؤمنا ويصبح كافرا٢١٩٩اسمعوا وأطيعوا فإنما عليهم ما حملوا وعليكم ما حملتم٢٢٠٠إن من ورائكم أياما يرفع فيها العلم ويكثر فيها الهرج٢٢٠١العبادة في الهرج كالهجرة إلى٢٢٠٢إذا وضع السيف في أمتي لم يرفع عنها إلى يوم القيامة٢٢٠٣حدثنا علي بن حجر، حدثنا إسماعيل بن إبراهيم، عن عبد الله بن عبيد، عن…٢٢٠٤كسروا فيها قسيكم وقطعوا فيها أوتاركم والزموا فيها أجواف بيوتكم وكونوا…٢٢٠٥إن من أشراط الساعة أن يرفع العلم ويظهر الجهل ويفشو الزنا وتشرب الخمر…٢٢٠٦ما من عام إلا الذي بعده شر منه حتى تلقوا ربكم٢٢٠٧لا تقوم الساعة حتى لا يقال في الأرض الله الله٢٢٠٨تقيء الأرض أفلاذ كبدها أمثال الأسطوان من الذهب والفضة قال فيجيء…٢٢٠٩لا تقوم الساعة حتى يكون أسعد الناس بالدنيا لكع ابن لكع٢٢١٠إذا فعلت أمتي خمس عشرة خصلة حل بها البلاء٢٢١١إذا اتخذ الفىء دولا والأمانة مغنما والزكاة مغرما وتعلم لغير الدين…٢٢١٢في هذه الأمة خسف ومسخ وقذف٢٢١٣بعثت في نفس الساعة فسبقتها كما سبقت هذه هذه٢٢١٤بعثت أنا والساعة كهاتين٢٢١٥لا تقوم الساعة حتى تقاتلوا قوما نعالهم الشعر ولا تقوم الساعة حتى…٢٢١٦إذا هلك كسرى فلا كسرى بعده وإذا هلك قيصر فلا قيصر بعده والذي نفسي…٢٢١٧ستخرج نار من حضرموت أو من نحو حضرموت قبل يوم القيامة تحشر الناس٢٢١٨لا تقوم الساعة حتى ينبعث دجالون كذابون قريب من ثلاثين كلهم يزعم أنه…٢٢١٩لا تقوم الساعة حتى تلحق قبائل من أمتي بالمشركين وحتى يعبدوا الأوثان…٢٢٢٠في ثقيف كذاب ومبير٢٢٢١خير الناس قرني ثم الذين يلونهم ثم الذين يلونهم ثم يأتي من بعدهم قوم…٢٢٢٢خير أمتي القرن الذي بعثت فيهم ثم الذين يلونهم٢٢٢٣يكون من بعدي اثنا عشر أميرا٢٢٢٤من أهان سلطان الله في الأرض أهانه الله٢٢٢٥حدثنا يحيى بن موسى، حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن سالم…٢٢٢٦الخلافة في أمتي ثلاثون سنة ثم ملك بعد ذلك٢٢٢٧قريش ولاة الناس في الخير والشر إلى يوم القيامة٢٢٢٨لا يذهب الليل والنهار حتى يملك رجل من الموالي يقال له جهجاه٢٢٢٩إنما أخاف على أمتي الأئمة المضلين٢٢٣٠لا تذهب الدنيا حتى يملك العرب رجل من أهل بيتي يواطئ اسمه اسمي٢٢٣١يلي رجل من أهل بيتي يواطئ اسمه اسمي٢٢٣٢إن في أمتي المهدي يخرج يعيش خمسا أو سبعا أو تسعا٢٢٣٣والذي نفسي بيده ليوشكن أن ينزل فيكم ابن مريم حكما مقسطا فيكسر الصليب…٢٢٣٤إنه لم يكن نبي بعد نوح إلا قد أنذر الدجال قومه وإني أنذركموه٢٢٣٥إني لأنذركموه وما من نبي إلا وقد أنذر قومه ولقد أنذره نوح قومه ولكني…٢٢٣٦تقاتلكم اليهود فتسلطون عليهم حتى يقول الحجر يا مسلم هذا يهودي ورائي…٢٢٣٧الدجال يخرج من أرض بالمشرق يقال لها خراسان يتبعه أقوام كأن وجوههم…٢٢٣٨الملحمة العظمى وفتح القسطنطينية وخروج الدجال في سبعة أشهر٢٢٣٩قال فتح القسطنطينية مع قيام الساعة . قال هذا حديث غريب . والقسطنطينية…٢٢٤٠وعبد الله بن عبد الرحمن بن يزيد بن جابر، دخل حديث أحدهما في حديث…٢٢٤١ألا إن ربكم ليس بأعور ألا وإنه أعور عينه اليمنى كأنها عنبة طافية٢٢٤٢يأتي الدجال المدينة فيجد الملائكة يحرسونها فلا يدخلها الطاعون ولا…٢٢٤٣الإيمان يمان والكفر من قبل المشرق والسكينة لأهل الغنم والفخر والرياء…٢٢٤٤يقتل ابن مريم الدجال بباب لد٢٢٤٥ما من نبي إلا وقد أنذر أمته الأعور الكذاب ألا إنه أعور وإن ربكم ليس…٢٢٤٦حدثنا سفيان بن وكيع، حدثنا عبد الأعلى بن عبد الأعلى، عن الجريري، عن…٢٢٤٧أتشهد أني رسول الله٢٢٤٨يمكث أبو الدجال وأمه ثلاثين عاما لا يولد لهما ولد ثم يولد لهما غلام…٢٢٤٩أتشهد أني رسول الله٢٢٥٠ما على الأرض نفس منفوسة - يعني اليوم تأتي عليها مائة سنة٢٢٥١أرأيتكم ليلتكم هذه على رأس مائة سنة منها لا يبقى ممن هو على ظهر الأرض…٢٢٥٢لا تسبوا الريح فإذا رأيتم ما تكرهون فقولوا اللهم إنا نسألك من خير هذه…٢٢٥٣إن تميما الداري حدثني بحديث ففرحت فأحببت أن أحدثكم حدثني أن ناسا من…٢٢٥٤لا ينبغي للمؤمن أن يذل نفسه٢٢٥٥انصر أخاك ظالما أو مظلوما٢٢٥٦من سكن البادية جفا ومن اتبع الصيد غفل ومن أتى أبواب السلاطين افتتن٢٢٥٧إنكم منصورون ومصيبون ومفتوح لكم فمن أدرك ذلك منكم فليتق الله وليأمر…٢٢٥٨وحماد، وعاصم بن بهدلة، سمعوا أبا وائل، عن حذيفة، قال قال عمر أيكم…٢٢٥٩اسمعوا هل سمعتم أنه سيكون بعدي أمراء فمن دخل عليهم فصدقهم بكذبهم…٢٢٦٠يأتي على الناس زمان الصابر فيهم على دينه كالقابض على الجمر٢٢٦١إذا مشت أمتي المطيطياء وخدمها أبناء الملوك أبناء فارس والروم سلط…٢٢٦٢رسول الله صلى الله عليه وسلم لما هلك كسرى قال " من استخلفوا " . قالوا…٢٢٦٣ألا أخبركم بخيركم من شركم٢٢٦٤ألا أخبركم بخيار أمرائكم وشرارهم خيارهم الذين تحبونهم ويحبونكم وتدعون…٢٢٦٥إنه سيكون عليكم أئمة تعرفون وتنكرون فمن أنكر فقد برئ ومن كره فقد سلم…٢٢٦٦إذا كان أمراؤكم خياركم وأغنياؤكم سمحاءكم وأموركم شورى بينكم فظهر…٢٢٦٧إنكم في زمان من ترك منكم عشر ما أمر به هلك ثم يأتي زمان من عمل منهم…٢٢٦٨ها هنا أرض الفتن وأشار إلى المشرق حيث يطلع جذل الشيطان٢٢٦٩تخرج من خراسان رايات سود لا يردها شيء حتى تنصب بإيلياء
