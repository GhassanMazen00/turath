أَبْوَابُ السَّفَرِ

٧٣ حديثًا
٥٤٤حدثنا عبد الوهاب بن عبد الحكم الوراق البغدادي، حدثنا يحيى بن سليم، عن…٥٤٥فقال حججت مع رسول الله صلى الله عليه وسلم فصلى ركعتين وحججت مع أبي…٥٤٦وإبراهيم بن ميسرة، سمعا أنس بن مالك، قال صلينا مع النبي صلى الله عليه…٥٤٧أن النبي صلى الله عليه وسلم خرج من المدينة إلى مكة لا يخاف إلا الله…٥٤٨حدثنا أحمد بن منيع، حدثنا هشيم، أخبرنا يحيى بن أبي إسحاق الحضرمي،…٥٤٩حدثنا هناد بن السري، حدثنا أبو معاوية، عن عاصم الأحول، عن عكرمة، عن…٥٥٠حدثنا قتيبة بن سعيد، حدثنا الليث بن سعد، عن صفوان بن سليم، عن أبي…٥٥١حدثنا علي بن حجر، حدثنا حفص بن غياث، عن الحجاج، عن عطية، عن ابن عمر،…٥٥٢ونافع، عن ابن عمر، قال صليت مع النبي صلى الله عليه وسلم في الحضر…٥٥٣حدثنا قتيبة بن سعيد، حدثنا الليث بن سعد، عن يزيد بن أبي حبيب، عن أبي…٥٥٤حدثنا عبد الصمد بن سليمان، حدثنا زكريا اللؤلؤي، حدثنا أبو بكر الأعين،…٥٥٥حدثنا هناد بن السري، حدثنا عبدة بن سليمان، عن عبيد الله بن عمر، عن…٥٥٦حدثنا يحيى بن موسى، حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن عباد…٥٥٧حدثنا قتيبة، حدثنا الليث بن سعد، عن خالد بن يزيد، عن سعيد بن أبي…٥٥٨رسول الله صلى الله عليه وسلم فأتيته فقال إن رسول الله صلى الله عليه…٥٥٩حدثنا محمود بن غيلان، حدثنا وكيع، عن سفيان، عن هشام بن إسحاق بن عبد…٥٦٠حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، عن سفيان، عن حبيب بن أبي…٥٦١سمع الله لمن حمده٥٦٢حدثنا محمود بن غيلان، حدثنا وكيع، حدثنا سفيان، عن الأسود بن قيس، عن…٥٦٣حدثنا أبو بكر، محمد بن أبان حدثنا إبراهيم بن صدقة، عن سفيان بن حسين،…٥٦٤حدثنا محمد بن عبد الملك بن أبي الشوارب، حدثنا يزيد بن زريع، حدثنا…٥٦٥أنه قال في صلاة الخوف قال يقوم الإمام مستقبل القبلة وتقوم طائفة منهم…٥٦٦قال أبو عيسى قال محمد بن بشار سألت يحيى بن سعيد عن هذا الحديث، فحدثني…٥٦٧وروى مالك بن أنس، عن يزيد بن رومان، عن صالح بن خوات، عمن صلى مع النبي…٥٦٨قال سجدت مع رسول الله صلى الله عليه وسلم إحدى عشرة سجدة منها التي في…٥٦٩حدثنا عبد الله بن عبد الرحمن، أخبرنا عبد الله بن صالح، حدثنا الليث بن…٥٧٠ايذنوا للنساء بالليل إلى المساجد٥٧١إذا كنت في الصلاة فلا تبزق عن يمينك ولكن خلفك أو تلقاء شمالك أو تحت…٥٧٢البزاق في المسجد خطيئة وكفارتها دفنها٥٧٣حدثنا قتيبة بن سعيد، حدثنا سفيان بن عيينة، عن أيوب بن موسى، عن عطاء…٥٧٤حدثنا قتيبة، حدثنا سفيان بن عيينة، عن يحيى بن سعيد، عن أبي بكر بن…٥٧٥حدثنا هارون بن عبد الله البزاز البغدادي، حدثنا عبد الصمد بن عبد…٥٧٦حدثنا يحيى بن موسى، حدثنا وكيع، عن ابن أبي ذئب، عن يزيد بن عبد الله…٥٧٧قال رأيت رسول الله صلى الله عليه وسلم يسجد في ص . قال ابن عباس وليست…٥٧٨نعم ومن لم يسجدهما فلا يقرأهما٥٧٩حدثنا قتيبة، حدثنا محمد بن يزيد بن خنيس، حدثنا الحسن بن محمد بن عبيد…٥٨٠سجد وجهي للذي خلقه وشق سمعه وبصره بحوله وقوته٥٨١من نام عن حزبه أو عن شيء منه فقرأه ما بين صلاة الفجر وصلاة الظهر كتب…٥٨٢أما يخشى الذي يرفع رأسه قبل الإمام أن يحول الله رأسه رأس حمار٥٨٣أن معاذ بن جبل، كان يصلي مع رسول الله صلى الله عليه وسلم المغرب ثم…٥٨٤حدثنا أحمد بن محمد، حدثنا عبد الله بن المبارك، أخبرنا خالد بن عبد…٥٨٥قال كان النبي صلى الله عليه وسلم إذا صلى الفجر قعد في مصلاه حتى تطلع…٥٨٦من صلى الغداة في جماعة ثم قعد يذكر الله حتى تطلع الشمس ثم صلى ركعتين…٥٨٧وغير، واحد، قالوا حدثنا الفضل بن موسى، عن عبد الله بن سعيد بن أبي…٥٨٨حدثنا محمود بن غيلان، حدثنا وكيع، عن عبد الله بن سعيد بن أبي هند، عن…٥٨٩يا بنى إياك والالتفات في الصلاة فإن الالتفات في الصلاة هلكة فإن كان…٥٩٠هو اختلاس يختلسه الشيطان من صلاة الرجل٥٩١إذا أتى أحدكم الصلاة والإمام على حال فليصنع كما يصنع الإمام٥٩٢إذا أقيمت الصلاة فلا تقوموا حتى تروني خرجت٥٩٣سل تعطه سل تعطه٥٩٤قالت أمر رسول الله صلى الله عليه وسلم ببناء المساجد في الدور وأن تنظف…٥٩٥ووكيع، عن هشام بن عروة، عن أبيه، أن النبي صلى الله عليه وسلم أمر فذكر…٥٩٦ببناء المساجد في الدور٥٩٧صلاة الليل والنهار مثنى مثنى٥٩٨حدثنا محمود بن غيلان، حدثنا وهب بن جرير، حدثنا شعبة، عن أبي إسحاق، عن…٥٩٩حدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي إسحاق،…٦٠٠حدثنا محمد بن عبد الأعلى، حدثنا خالد بن الحارث، عن أشعث، وهو ابن عبد…٦٠١قالت جئت ورسول الله صلى الله عليه وسلم يصلي في البيت والباب عليه مغلق…٦٠٢حدثنا محمود بن غيلان، حدثنا أبو داود، قال أنبأنا شعبة، عن الأعمش، قال…٦٠٣إذا توضأ الرجل فأحسن الوضوء ثم خرج إلى الصلاة لا يخرجه أو قال لا…٦٠٤عليكم بهذه الصلاة في البيوت٦٠٥حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، حدثنا سفيان، عن الأغر…٦٠٦ستر ما بين أعين الجن وعورات بني آدم إذا دخل أحدهم الخلاء أن يقول بسم…٦٠٧أمتي يوم القيامة غر من السجود محجلون من الوضوء٦٠٨أن رسول الله صلى الله عليه وسلم كان يحب التيمن في طهوره إذا تطهر وفي…٦٠٩يجزئ في الوضوء رطلان من ماء٦١٠ينضح بول الغلام ويغسل بول الجارية٦١١قال رأيت جرير بن عبد الله توضأ ومسح على خفيه قال فقلت له في ذلك فقال…٦١٢حدثنا محمد بن حميد الرازي، قال حدثنا نعيم بن ميسرة النحوي، عن خالد بن…٦١٣أن النبي صلى الله عليه وسلم رخص للجنب إذا أراد أن يأكل أو يشرب أو…٦١٤أعيذك بالله يا كعب بن عجرة من أمراء يكونون من بعدي فمن غشي أبوابهم…٦١٥بهذا .٦١٦اتقوا الله ربكم وصلوا خمسكم وصوموا شهركم وأدوا زكاة أموالكم وأطيعوا…
