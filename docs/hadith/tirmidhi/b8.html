كتاب الصوم عن رسول الله صلى الله عليه وسلم

١٢٧ حديثًا
٦٨٢إذا كان أول ليلة من شهر رمضان صفدت الشياطين ومردة الجن وغلقت أبواب…٦٨٣من صام رمضان وقامه إيمانا واحتسابا غفر له ما تقدم من ذنبه ومن قام…٦٨٤لا تقدموا الشهر بيوم ولا بيومين إلا أن يوافق ذلك صوما كان يصومه أحدكم…٦٨٥لا تقدموا شهر رمضان بصيام قبله بيوم أو يومين إلا أن يكون رجل كان يصوم…٦٨٦حدثنا أبو سعيد عبد الله بن سعيد الأشج، حدثنا أبو خالد الأحمر، عن عمرو…٦٨٧أحصوا هلال شعبان لرمضان٦٨٨لا تصوموا قبل رمضان صوموا لرؤيته وأفطروا لرؤيته فإن حالت دونه غياية…٦٨٩الشهر يكون تسعا وعشرين٦٩٠الشهر تسع وعشرون٦٩١أتشهد أن لا إله إلا الله أتشهد أن محمدا رسول الله٦٩٢شهرا عيد لا ينقصان رمضان وذو الحجة٦٩٣أن أم الفضل بنت الحارث، بعثته إلى معاوية بالشام . قال فقدمت الشام…٦٩٤من وجد تمرا فليفطر عليه ومن لا فليفطر على ماء فإن الماء طهور٦٩٥إذا أفطر أحدكم فليفطر على تمر٦٩٦كان النبي صلى الله عليه وسلم يفطر قبل أن يصلي على رطبات فإن لم تكن…٦٩٧الصوم يوم تصومون والفطر يوم تفطرون والأضحى يوم تضحون٦٩٨إذا أقبل الليل وأدبر النهار وغابت الشمس فقد أفطرت٦٩٩لا يزال الناس بخير ما عجلوا الفطر٧٠٠قال الله عز وجل أحب عبادي إلى أعجلهم فطرا٧٠١وأبو المغيرة، عن الأوزاعي، بهذا الإسناد نحوه . قال أبو عيسى هذا حديث…٧٠٢حدثنا هناد، حدثنا أبو معاوية، عن الأعمش، عن عمارة بن عمير، عن أبي…٧٠٣حدثنا يحيى بن موسى، حدثنا أبو داود الطيالسي، حدثنا هشام الدستوائي، عن…٧٠٤حدثنا هناد، حدثنا وكيع، عن هشام، بنحوه إلا أنه قال قدر قراءة خمسين…٧٠٥كلوا واشربوا ولا يهيدنكم الساطع المصعد وكلوا واشربوا حتى يعترض لكم…٧٠٦لا يمنعنكم من سحوركم أذان بلال ولا الفجر المستطيل ولكن الفجر المستطير…٧٠٧من لم يدع قول الزور والعمل به فليس لله حاجة بأن يدع طعامه وشرابه٧٠٨تسحروا فإن في السحور بركة٧٠٩فصل ما بين صيامنا وصيام أهل الكتاب أكلة السحر٧١٠حدثنا قتيبة، حدثنا عبد العزيز بن محمد، عن جعفر بن محمد، عن أبيه، عن…٧١١إن شئت فصم وإن شئت فأفطر٧١٢حدثنا نصر بن علي الجهضمي، حدثنا بشر بن المفضل، عن سعيد بن يزيد أبي…٧١٣حدثنا نصر بن علي، حدثنا يزيد بن زريع، حدثنا الجريري، ح قال وحدثنا…٧١٤حدثنا قتيبة، حدثنا ابن لهيعة، عن يزيد بن أبي حبيب، عن معمر بن أبي…٧١٥ويوسف بن عيسى، قالا حدثنا وكيع، حدثنا أبو هلال، عن عبد الله بن سوادة،…٧١٦أرأيت لو كان على أختك دين أكنت تقضينه٧١٧. قال محمد وقد روى غير أبي خالد، عن الأعمش، مثل رواية أبي خالد . قال…٧١٨من مات وعليه صيام شهر فليطعم عنه مكان كل يوم مسكينا٧١٩ثلاث لا يفطرن الصائم الحجامة والقىء والاحتلام٧٢٠من ذرعه القىء فليس عليه قضاء ومن استقاء عمدا فليقض٧٢١من أكل أو شرب ناسيا فلا يفطر فإنما هو رزق رزقه الله٧٢٢وخلاس، عن أبي هريرة، عن النبي صلى الله عليه وسلم مثله أو نحوه . قال…٧٢٣من أفطر يوما من رمضان من غير رخصة ولا مرض لم يقض عنه صوم الدهر كله…٧٢٤حدثنا نصر بن علي الجهضمي، وأبو عمار والمعنى واحد واللفظ لفظ أبي عمار…٧٢٥حدثنا محمد بن بشار، حدثنا عبد الرحمن بن مهدي، حدثنا سفيان، عن عاصم بن…٧٢٦حدثنا عبد الأعلى بن واصل الكوفي، حدثنا الحسن بن عطية، حدثنا أبو…٧٢٧وقتيبة، قالا حدثنا أبو الأحوص، عن زياد بن علاقة، عن عمرو بن ميمون، عن…٧٢٨قالت كان رسول الله صلى الله عليه وسلم يباشرني وهو صائم وكان أملككم…٧٢٩والأسود، عن عائشة، قالت كان رسول الله صلى الله عليه وسلم يقبل ويباشر…٧٣٠من لم يجمع الصيام قبل الفجر فلا صيام له٧٣١حدثنا قتيبة، حدثنا أبو الأحوص، عن سماك بن حرب، عن ابن أم هانئ، عن أم…٧٣٢الصائم المتطوع أمين نفسه إن شاء صام وإن شاء أفطر٧٣٣قالت دخل على رسول الله صلى الله عليه وسلم يوما فقال " هل عندكم شيء "…٧٣٤حدثنا محمود بن غيلان، حدثنا بشر بن السري، عن سفيان، عن طلحة بن يحيى،…٧٣٥اقضيا يوما آخر مكانه٧٣٦أنها قالت ما رأيت النبي صلى الله عليه وسلم في شهر أكثر صياما منه في…٧٣٧حدثنا هناد حدثنا عبدة عن محمد بن عمرو حدثنا أبو سلمة عن عائشة عن…٧٣٨إذا بقي نصف من شعبان فلا تصوموا٧٣٩أكنت تخافين أن يحيف الله عليك ورسوله٧٤٠أفضل الصيام بعد صيام شهر رمضان شهر الله المحرم٧٤١إن كنت صائما بعد شهر رمضان فصم المحرم فإنه شهر الله فيه يوم تاب الله…٧٤٢وطلق بن غنام، عن شيبان، عن عاصم، عن زر، عن عبد الله، قال كان رسول…٧٤٣لا يصوم أحدكم يوم الجمعة إلا أن يصوم قبله أو يصوم بعده٧٤٤لا تصوموا يوم السبت إلا فيما افترض الله عليكم فإن لم يجد أحدكم إلا…٧٤٥حدثنا أبو حفص، عمرو بن علي الفلاس حدثنا عبد الله بن داود، عن ثور بن…٧٤٦ومعاوية بن هشام، قالا حدثنا سفيان، عن منصور، عن خيثمة، عن عائشة، قالت…٧٤٧تعرض الأعمال يوم الاثنين والخميس فأحب أن يعرض عملي وأنا صائم٧٤٨إن لأهلك عليك حقا صم رمضان والذي يليه وكل أربعاء وخميس فإذا أنت قد…٧٤٩صيام يوم عرفة إني أحتسب على الله أن يكفر السنة التي قبله والسنة التي…٧٥٠حدثنا أحمد بن منيع، حدثنا إسماعيل ابن علية، حدثنا أيوب، عن عكرمة، عن…٧٥١وعلي بن حجر، قالا حدثنا سفيان بن عيينة، وإسماعيل بن إبراهيم، عن ابن…٧٥٢صيام يوم عاشوراء إني أحتسب على الله أن يكفر السنة التي قبله٧٥٣حدثنا هارون بن إسحاق الهمداني، حدثنا عبدة بن سليمان، عن هشام بن عروة،…٧٥٤حدثنا هناد، وأبو كريب قالا حدثنا وكيع، عن حاجب بن عمر، عن الحكم بن…٧٥٥حدثنا قتيبة، حدثنا عبد الوارث، عن يونس، عن الحسن، عن ابن عباس، قال…٧٥٦حدثنا هناد، حدثنا أبو معاوية، عن الأعمش، عن إبراهيم، عن الأسود، عن…٧٥٧ما من أيام العمل الصالح فيهن أحب إلى الله من هذه الأيام العشر٧٥٨ما من أيام أحب إلى الله أن يتعبد له فيها من عشر ذي الحجة يعدل صيام كل…٧٥٩من صام رمضان ثم أتبعه ستا من شوال فذلك صيام الدهر٧٦٠قال عهد إلى النبي صلى الله عليه وسلم ثلاثة أن لا أنام إلا على وتر…٧٦١يا أبا ذر إذا صمت من الشهر ثلاثة أيام فصم ثلاث عشرة وأربع عشرة وخمس…٧٦٢من صام من كل شهر ثلاثة أيام فذلك صيام الدهر٧٦٣قالت قلت لعائشة أكان رسول الله صلى الله عليه وسلم يصوم ثلاثة أيام من…٧٦٤إن ربكم يقول كل حسنة بعشر أمثالها إلى سبعمائة ضعف والصوم لي وأنا أجزي…٧٦٥إن في الجنة لبابا يدعى الريان يدعى له الصائمون فمن كان من الصائمين…٧٦٦للصائم فرحتان فرحة حين يفطر وفرحة حين يلقى ربه٧٦٧وأحمد بن عبدة الضبي، قالا حدثنا حماد بن زيد، عن غيلان بن جرير، عن عبد…٧٦٨حدثنا قتيبة، حدثنا حماد بن زيد، عن أيوب، عن عبد الله بن شقيق، قال…٧٦٩صلى الله عليه وسلم قال كان يصوم من الشهر حتى نرى أنه لا يريد أن يفطر…٧٧٠أفضل الصوم صوم أخي داود كان يصوم يوما ويفطر يوما ولا يفر إذا لاقى٧٧١حدثنا محمد بن عبد الملك بن أبي الشوارب، حدثنا يزيد بن زريع، حدثنا…٧٧٢حدثنا قتيبة، حدثنا عبد العزيز بن محمد، عن عمرو بن يحيى، عن أبيه، عن…٧٧٣يوم عرفة ويوم النحر وأيام التشريق عيدنا أهل الإسلام وهي أيام أكل وشرب٧٧٤أفطر الحاجم والمحجوم٧٧٥حدثنا بشر بن هلال البصري، حدثنا عبد الوارث بن سعيد، حدثنا أيوب، عن…٧٧٦أن النبي صلى الله عليه وسلم احتجم وهو صائم . هذا حديث حسن غريب من هذا…٧٧٧حدثنا أحمد بن منيع، حدثنا عبد الله بن إدريس، عن يزيد بن أبي زياد، عن…٧٧٨وخالد بن الحارث، عن سعيد بن أبي عروبة، عن قتادة، عن أنس، قال قال رسول…٧٧٩قال أخبرتني عائشة، وأم سلمة زوجا النبي صلى الله عليه وسلم أن النبي…٧٨٠إذا دعي أحدكم إلى طعام فليجب فإن كان صائما فليصل٧٨١إذا دعي أحدكم وهو صائم فليقل إني صائم٧٨٢لا تصوم المرأة وزوجها شاهد يوما من غير شهر رمضان إلا بإذنه٧٨٣حدثنا قتيبة، حدثنا أبو عوانة، عن إسماعيل السدي، عن عبد الله البهي، عن…٧٨٤الصائم إذا أكل عنده المفاطير صلت عليه الملائكة٧٨٥أن النبي صلى الله عليه وسلم دخل عليها فقدمت إليه طعاما فقال " كلي " .…٧٨٦حتى يفرغوا أو يشبعوا٧٨٧حدثنا علي بن حجر، أخبرنا علي بن مسهر، عن عبيدة، عن إبراهيم، عن…٧٨٨أسبغ الوضوء وخلل بين الأصابع وبالغ في الاستنشاق إلا أن تكون صائما٧٨٩من نزل على قوم فلا يصومن تطوعا إلا بإذنهم٧٩٠وعروة، عن عائشة، أن النبي صلى الله عليه وسلم كان يعتكف العشر الأواخر…٧٩١حدثنا هناد، حدثنا أبو معاوية، عن يحيى بن سعيد، عن عمرة، عن عائشة،…٧٩٢تحروا ليلة القدر في العشر الأواخر من رمضان٧٩٣أنها ليلة صبيحتها تطلع الشمس ليس لها شعاع٧٩٤التمسوها في تسع يبقين أو في سبع يبقين أو في خمس يبقين أو في ثلاث أو…٧٩٥أن النبي صلى الله عليه وسلم كان يوقظ أهله في العشر الأواخر من رمضان .…٧٩٦قالت كان رسول الله صلى الله عليه وسلم يجتهد في العشر الأواخر ما لا…٧٩٧الغنيمة الباردة الصوم في الشتاء٧٩٨حدثنا قتيبة، حدثنا بكر بن مضر، عن عمرو بن الحارث، عن بكير بن عبد الله…٧٩٩أنه قال أتيت أنس بن مالك في رمضان وهو يريد سفرا وقد رحلت له راحلته…٨٠٠قال أتيت أنس بن مالك في رمضان فذكر نحوه . قال أبو عيسى هذا حديث حسن .…٨٠١تحفة الصائم الدهن والمجمر٨٠٢الفطر يوم يفطر الناس والأضحى يوم يضحي الناس٨٠٣حدثنا محمد بن بشار، حدثنا ابن أبي عدي، قال أنبأنا حميد الطويل، عن أنس…٨٠٤وعمرة، عن عائشة، أنها قالت كان رسول الله صلى الله عليه وسلم إذا اعتكف…٨٠٥وعمرة، عن عائشة، . والعمل على هذا عند أهل العلم إذا اعتكف الرجل أن لا…٨٠٦إنه من قام مع الإمام حتى ينصرف كتب له قيام ليلة٨٠٧من فطر صائما كان له مثل أجره غير أنه لا ينقص من أجر الصائم شيئا٨٠٨من قام رمضان إيمانا واحتسابا غفر له ما تقدم من ذنبه
