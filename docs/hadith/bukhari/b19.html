٦٣ حديثًا
١١٢٠اللهم لك الحمد أنت قيم السموات والأرض ومن فيهن ولك الحمد، لك ملك…١١٢٢نعم الرجل عبد الله، لو كان يصلي من الليل١١٢٣أن عائشة رضى الله عنها أخبرته أن رسول الله صلى الله عليه وسلم كان…١١٢٤يقول اشتكى النبي صلى الله عليه وسلم فلم يقم ليلة أو ليلتين .١١٢٥حدثنا محمد بن كثير، قال أخبرنا سفيان، عن الأسود بن قيس، عن جندب بن…١١٢٦سبحان الله ماذا أنزل الليلة من الفتنة، ماذا أنزل من الخزائن من يوقظ…١١٢٧أن حسين بن علي، أخبره أن علي بن أبي طالب أخبره أن رسول الله صلى الله…١١٢٨وما سبح رسول الله صلى الله عليه وسلم سبحة الضحى قط، وإني لأسبحها .١١٢٩قد رأيت الذي صنعتم ولم يمنعني من الخروج إليكم إلا أني خشيت أن تفرض…١١٣٠أفلا أكون عبدا شكورا١١٣١أحب الصلاة إلى الله صلاة داود عليه السلام وأحب الصيام إلى الله صيام…١١٣٢قال سألت عائشة رضى الله عنها أى العمل كان أحب إلى النبي صلى الله عليه…١١٣٢.٢حدثنا محمد بن سلام قال أخبرنا أبو الأحوص عن الأشعث قال إذا سمع الصارخ…١١٣٣حدثنا موسى بن إسماعيل، قال حدثنا إبراهيم بن سعد، قال ذكر أبي عن أبي…١١٣٤حدثنا يعقوب بن إبراهيم، قال حدثنا روح، قال حدثنا سعيد، عن قتادة، عن…١١٣٥حدثنا سليمان بن حرب، قال حدثنا شعبة، عن الأعمش، عن أبي وائل، عن عبد…١١٣٦حدثنا حفص بن عمر، قال حدثنا خالد بن عبد الله، عن حصين، عن أبي وائل،…١١٣٧مثنى مثنى، فإذا خفت الصبح فأوتر بواحدة١١٣٨حدثنا مسدد، قال حدثنا يحيى، عن شعبة، قال حدثني أبو جمرة، عن ابن عباس…١١٣٩حدثنا إسحاق، قال حدثنا عبيد الله، قال أخبرنا إسرائيل، عن أبي حصين، عن…١١٤٠حدثنا عبيد الله بن موسى، قال أخبرنا حنظلة، عن القاسم بن محمد، عن…١١٤١أنه سمع أنسا رضى الله عنه يقول كان رسول الله صلى الله عليه وسلم يفطر…١١٤٢يعقد الشيطان على قافية رأس أحدكم إذا هو نام ثلاث عقد، يضرب كل عقدة…١١٤٣أما الذي يثلغ رأسه بالحجر فإنه يأخذ القرآن فيرفضه وينام عن الصلاة…١١٤٤بال الشيطان في أذنه١١٤٥ينزل ربنا تبارك وتعالى كل ليلة إلى السماء الدنيا حين يبقى ثلث الليل…١١٤٦قال سألت عائشة رضى الله عنها كيف صلاة النبي صلى الله عليه وسلم بالليل…١١٤٧يا عائشة، إن عينى تنامان ولا ينام قلبي١١٤٨حتى إذا كبر قرأ جالسا، فإذا بقي عليه من السورة ثلاثون أو أربعون آية…١١٤٩يا بلال حدثني بأرجى عمل عملته في الإسلام، فإني سمعت دف نعليك بين يدى…١١٥٠حلوه، ليصل أحدكم نشاطه، فإذا فتر فليقعد " .١١٥١فإن الله لا يمل حتى تملوا " .١١٥٢يا عبد الله، لا تكن مثل فلان، كان يقوم الليل فترك قيام الليل١١٥٣ألم أخبر أنك تقوم الليل وتصوم النهار١١٥٤من تعار من الليل فقال لا إله إلا الله وحده لا شريك له، له الملك، وله…١١٥٥حدثنا يحيى بن بكير، قال حدثنا الليث، عن يونس، عن ابن شهاب، أخبرني…١١٥٨نعم الرجل عبد الله لو كان يصلي من الليل١١٥٩ولم يكن يدعهما أبدا .١١٦٠حدثنا عبد الله بن يزيد، حدثنا سعيد بن أبي أيوب، قال حدثني أبو الأسود،…١١٦١حدثنا بشر بن الحكم، حدثنا سفيان، قال حدثني سالم أبو النضر، عن أبي…١١٦٢حدثنا علي بن عبد الله، حدثنا سفيان، قال أبو النضر عن أبي سلمة، عن…١١٦٣حدثنا بيان بن عمرو، حدثنا يحيى بن سعيد، حدثنا ابن جريج، عن عطاء، عن…١١٦٤ثم يصلي إذا سمع النداء بالصبح ركعتين خفيفتين .١١٦٥حدثنا محمد بن بشار، قال حدثنا محمد بن جعفر، حدثنا شعبة، عن محمد بن…١١٦٦إذا هم أحدكم بالأمر فليركع ركعتين من غير الفريضة ثم ليقل اللهم إني…١١٦٧إذا دخل أحدكم المسجد فلا يجلس حتى يصلي ركعتين١١٦٨حدثنا عبد الله بن يوسف، قال أخبرنا مالك، عن إسحاق بن عبد الله بن أبي…١١٦٩وركعتين بعد الظهر، وركعتين بعد الجمعة، وركعتين بعد المغرب، وركعتين…١١٧٠إذا جاء أحدكم والإمام يخطب أو قد خرج فليصل ركعتين١١٧١يقول أتي ابن عمر رضى الله عنهما في منزله فقيل له هذا رسول الله صلى…١١٧٣وسجدتين بعد الظهر، وسجدتين بعد المغرب، وسجدتين بعد العشاء، وسجدتين…١١٧٤حدثنا علي بن عبد الله، قال حدثنا سفيان، عن عمرو، قال سمعت أبا…١١٧٥قال قلت لابن عمر رضى الله عنهما أتصلي الضحى قال لا . قلت فعمر . قال…١١٧٦أنه رأى النبي صلى الله عليه وسلم يصلي الضحى غير أم هانئ فإنها قالت إن…١١٧٧وإني لأسبحها .١١٧٨رضى الله عنه قال أوصاني خليلي بثلاث لا أدعهن حتى أموت صوم ثلاثة أيام…١١٧٩قال قال رجل من الأنصار وكان ضخما للنبي صلى الله عليه وسلم إني لا…١١٨١وركعتين بعدها، وركعتين بعد المغرب في بيته، وركعتين بعد العشاء في…١١٨٢حدثنا مسدد، قال حدثنا يحيى، عن شعبة، عن إبراهيم بن محمد بن المنتشر،…١١٨٣صلوا قبل صلاة المغرب١١٨٤حدثنا عبد الله بن يزيد، قال حدثنا سعيد بن أبي أيوب، قال حدثني يزيد بن…١١٨٥أنه عقل رسول الله صلى الله عليه وسلم، وعقل مجة مجها في وجهه من بئر…١١٨٧اجعلوا فى بيوتكم من صلاتكم ولا تتخذوها قبورا
