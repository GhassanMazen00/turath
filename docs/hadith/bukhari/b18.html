٣٩ حديثًا
١٠٨٠وحصين، عن عكرمة، عن ابن عباس رضى الله عنهما قال أقام النبي صلى الله…١٠٨١حدثنا أبو معمر، قال حدثنا عبد الوارث، قال حدثنا يحيى بن أبي إسحاق،…١٠٨٢وأبي بكر وعمر، ومع عثمان صدرا من إمارته ثم أتمها .١٠٨٣حدثنا أبو الوليد، قال حدثنا شعبة، أنبأنا أبو إسحاق، قال سمعت حارثة بن…١٠٨٤فقيل ذلك لعبد الله بن مسعود رضى الله عنه فاسترجع ثم قال صليت مع رسول…١٠٨٥حدثنا موسى بن إسماعيل، قال حدثنا وهيب، قال حدثنا أيوب، عن أبي العالية…١٠٨٦لا تسافر المرأة ثلاثة أيام إلا مع ذي محرم١٠٨٧لا تسافر المرأة ثلاثا إلا مع ذي محرم١٠٨٨لا يحل لامرأة تؤمن بالله واليوم الآخر أن تسافر مسيرة يوم وليلة ليس…١٠٨٩وإبراهيم بن ميسرة، عن أنس رضى الله عنه قال صليت الظهر مع النبي صلى…١٠٩٠وأتمت صلاة الحضر . قال الزهري فقلت لعروة ما بال عائشة تتم قال تأولت…١٠٩٢قال سالم كان ابن عمر رضى الله عنهما يجمع بين المغرب والعشاء بالمزدلفة…١٠٩٣قال رأيت النبي صلى الله عليه وسلم يصلي على راحلته حيث توجهت به .١٠٩٤أن جابر بن عبد الله، أخبره أن النبي صلى الله عليه وسلم كان يصلي…١٠٩٥قال وكان ابن عمر رضى الله عنهما يصلي على راحلته ويوتر عليها، ويخبر أن…١٠٩٦قال كان عبد الله بن عمر رضى الله عنهما يصلي في السفر على راحلته،…١٠٩٧أن عامر بن ربيعة، أخبره قال رأيت رسول الله صلى الله عليه وسلم وهو على…١٠٩٨قال قال سالم كان عبد الله يصلي على دابته من الليل وهو مسافر، ما يبالي…١٠٩٩أن النبي صلى الله عليه وسلم كان يصلي على راحلته نحو المشرق فإذا أراد…١١٠٠فلقيناه بعين التمر، فرأيته يصلي على حمار ووجهه من ذا الجانب، يعني عن…١١٠١أن حفص بن عاصم، حدثه قال سافر ابن عمر رضى الله عنهما فقال صحبت النبي…١١٠٢سمع ابن عمر، يقول صحبت رسول الله صلى الله عليه وسلم فكان لا يزيد في…١١٠٣قال ما أنبأ أحد، أنه رأى النبي صلى الله عليه وسلم صلى الضحى غير أم…١١٠٤أن أباه، أخبره أنه، رأى النبي صلى الله عليه وسلم صلى السبحة بالليل في…١١٠٥يومئ برأسه، وكان ابن عمر يفعله .١١٠٦قال كان النبي صلى الله عليه وسلم يجمع بين المغرب والعشاء إذا جد به…١١٠٧ويجمع بين المغرب والعشاء .١١٠٨وعن حسين، عن يحيى بن أبي كثير، عن حفص بن عبيد الله بن أنس، عن أنس بن…١١٠٩حتى يجمع بينها وبين العشاء . قال سالم وكان عبد الله يفعله إذا أعجله…١١١٠أن أنسا رضى الله عنه حدثه أن رسول الله صلى الله عليه وسلم كان يجمع…١١١١ثم يجمع بينهما، وإذا زاغت صلى الظهر ثم ركب .١١١٢قال كان رسول الله صلى الله عليه وسلم إذا ارتحل قبل أن تزيغ الشمس أخر…١١١٣إنما جعل الإمام ليؤتم به، فإذا ركع فاركعوا، وإذا رفع فارفعوا١١١٤إنما جعل الإمام ليؤتم به، فإذا كبر فكبروا وإذا ركع فاركعوا، وإذا رفع…١١١٥إن صلى قائما فهو أفضل، ومن صلى قاعدا فله نصف أجر القائم، ومن صلى…١١١٦من صلى قائما فهو أفضل، ومن صلى قاعدا فله نصف أجر القائم، ومن صلى…١١١٧صل قائما، فإن لم تستطع فقاعدا، فإن لم تستطع فعلى جنب١١١٨فكان يقرأ قاعدا حتى إذا أراد أن يركع قام، فقرأ نحوا من ثلاثين آية أو…١١١٩وأبي النضر، مولى عمر بن عبيد الله عن أبي سلمة بن عبد الرحمن، عن عائشة…
