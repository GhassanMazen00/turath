٢٢ حديثًا
٢٤٨٣فأمر أبو عبيدة بأزواد ذلك الجيش فجمع ذلك كله فكان مزودى تمر، فكان…٢٤٨٤ناد في الناس فيأتون بفضل أزوادهم٢٤٨٥فتقسم عشر قسم، فنأكل لحما نضيجا قبل أن تغرب الشمس .٢٤٨٦إن الأشعريين إذا أرملوا في الغزو، أو قل طعام عيالهم بالمدينة جمعوا ما…٢٤٨٧وما كان من خليطين فإنهما يتراجعان بينهما بالسوية٢٤٨٨إن لهذه البهائم أوابد كأوابد الوحش فما غلبكم منها فاصنعوا به هكذا٢٤٨٩حتى يستأذن أصحابه .٢٤٩٠إلا أن يستأذن الرجل منكم أخاه .٢٤٩١من أعتق شقصا له من عبد أو شركا أو قال نصيبا وكان له ما يبلغ ثمنه…٢٤٩٢من أعتق شقيصا من مملوكه فعليه خلاصه في ماله، فإن لم يكن له مال قوم…٢٤٩٣مثل القائم على حدود الله والواقع فيها كمثل قوم استهموا على سفينة،…٢٤٩٤فيعجبه مالها وجمالها، فيريد وليها أن يتزوجها بغير أن يقسط في صداقها،…٢٤٩٥فإذا وقعت الحدود وصرفت الطرق فلا شفعة .٢٤٩٦فإذا وقعت الحدود وصرفت الطرق فلا شفعة .٢٤٩٨ما كان يدا بيد فخذوه، وما كان نسيئة فذروه٢٤٩٩حدثنا موسى بن إسماعيل، حدثنا جويرية بن أسماء، عن نافع، عن عبد الله…٢٥٠٠فبقي عتود فذكره لرسول الله صلى الله عليه وسلم فقال " ضح به أنت " .٢٥٠٢أنه كان يخرج به جده عبد الله بن هشام إلى السوق فيشتري الطعام فيلقاه…٢٥٠٣من أعتق شركا له في مملوك وجب عليه أن يعتق كله، إن كان له مال قدر ثمنه…٢٥٠٤من أعتق شقصا له في عبد، أعتق كله إن كان له مال، وإلا يستسع غير مشقوق…٢٥٠٦بلغني أن أقواما يقولون كذا وكذا، والله لأنا أبر وأتقى لله منهم، ولو…٢٥٠٧إن لهذه البهائم أوابد كأوابد الوحش، فما غلبكم منها فاصنعوا به هكذا
