٢١ حديثًا
٧٢٤٦ارجعوا إلى أهليكم، فأقيموا فيهم، وعلموهم، ومروهم وذكر أشياء أحفظها أو…٧٢٤٧لا يمنعن أحدكم أذان بلال من سحوره، فإنه يؤذن أو قال ينادي ليرجع…٧٢٤٨إن بلالا ينادي بليل، فكلوا واشربوا حتى ينادي ابن أم مكتوم٧٢٤٩حدثنا حفص بن عمر، حدثنا شعبة، عن الحكم، عن إبراهيم، عن علقمة، عن عبد…٧٢٥٠أن رسول الله صلى الله عليه وسلم انصرف من اثنتين فقال له ذو اليدين…٧٢٥١وقد أمر أن يستقبل الكعبة فاستقبلوها . وكانت وجوههم إلى الشأم…٧٢٥٢قال لما قدم رسول الله صلى الله عليه وسلم المدينة صلى نحو بيت المقدس…٧٢٥٣حدثني يحيى بن قزعة، حدثني مالك، عن إسحاق بن عبد الله بن أبي طلحة، عن…٧٢٥٤لأبعثن إليكم رجلا أمينا حق أمين٧٢٥٥لكل أمة أمين، وأمين هذه الأمة أبو عبيدة٧٢٥٦حدثنا سليمان بن حرب، حدثنا حماد بن زيد، عن يحيى بن سعيد، عن عبيد بن…٧٢٥٧لو دخلوها لم يزالوا فيها إلى يوم القيامة٧٢٥٩أن عبيد الله بن عبد الله، أخبره أن أبا هريرة وزيد بن خالد أخبراه أن…٧٢٦٠أن أبا هريرة، قال بينما نحن عند رسول الله صلى الله عليه وسلم إذ قام…٧٢٦١لكل نبي حواري وحواري الزبير٧٢٦٢ائذن له، وبشره بالجنة٧٢٦٣سمع ابن عباس، عن عمر رضى الله عنهم قال جئت فإذا رسول الله صلى الله…٧٢٦٤أن عبد الله بن عباس، أخبره أن رسول الله صلى الله عليه وسلم بعث بكتابه…٧٢٦٥أذن في قومك أو في الناس يوم عاشوراء أن من أكل فليتم بقية يومه، ومن لم…٧٢٦٦قال كان ابن عباس يقعدني على سريره فقال إن وفد عبد القيس لما أتوا رسول…٧٢٦٧كلوا أو اطعموا فإنه حلال أو قال لا بأس به . شك فيه ولكنه ليس من طعامي
