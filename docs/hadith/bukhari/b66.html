كتاب فضائل القرآن

٨٧ حديثًا
٤٩٧٩قال أخبرتني عائشة، وابن، عباس رضى الله عنهم قالا لبث النبي صلى الله…٤٩٨٠قال أنبئت أن جبريل، أتى النبي صلى الله عليه وسلم وعنده أم سلمة فجعل…٤٩٨١ما من الأنبياء نبي إلا أعطي ما مثله آمن عليه البشر، وإنما كان الذي…٤٩٨٢تعالى تابع على رسوله صلى الله عليه وسلم قبل وفاته حتى توفاه أكثر ما…٤٩٨٣يقول اشتكى النبي صلى الله عليه وسلم فلم يقم ليلة أو ليلتين فأتته…٤٩٨٤قال فأمر عثمان زيد بن ثابت وسعيد بن العاص وعبد الله بن الزبير وعبد…٤٩٨٥أين الذي يسألني عن العمرة آنفا٤٩٨٦أن زيد بن ثابت رضى الله عنه قال أرسل إلى أبو بكر مقتل أهل اليمامة…٤٩٨٧أن أنس بن مالك، حدثه أن حذيفة بن اليمان قدم على عثمان وكان يغازي أهل…٤٩٨٨سمع زيد بن ثابت، قال فقدت آية من الأحزاب حين نسخنا المصحف قد كنت أسمع…٤٩٨٩أن ابن السباق، قال إن زيد بن ثابت قال أرسل إلى أبو بكر رضى الله عنه…٤٩٩٠ادع لي زيدا وليجئ باللوح والدواة والكتف أو الكتف والدواة ثم قال٤٩٩١أقرأني جبريل على حرف فراجعته، فلم أزل أستزيده ويزيدني حتى انتهى إلى…٤٩٩٢أرسله اقرأ يا هشام٤٩٩٣أن ابن جريج، أخبرهم قال وأخبرني يوسف بن ماهك، قال إني عند عائشة أم…٤٩٩٤يقول في بني إسرائيل والكهف ومريم وطه والأنبياء إنهن من العتاق الأول…٤٩٩٥سمع البراء رضى الله عنه قال تعلمت { سبح اسم ربك } قبل أن يقدم النبي…٤٩٩٦قال قال عبد الله قد علمت النظائر التي كان النبي صلى الله عليه وسلم…٤٩٩٧وأجود ما يكون في شهر رمضان لأن جبريل كان يلقاه في كل ليلة في شهر…٤٩٩٨قال كان يعرض على النبي صلى الله عليه وسلم القرآن كل عام مرة، فعرض…٤٩٩٩خذوا القرآن من أربعة من عبد الله بن مسعود وسالم ومعاذ وأبى بن كعب٥٠٠٠حدثنا عمر بن حفص، حدثنا أبي، حدثنا الأعمش، حدثنا شقيق بن سلمة، قال…٥٠٠١حدثني محمد بن كثير، أخبرنا سفيان، عن الأعمش، عن إبراهيم، عن علقمة،…٥٠٠٢ولو أعلم أحدا أعلم مني بكتاب الله تبلغه الإبل لركبت إليه .٥٠٠٣قال سألت أنس بن مالك رضى الله عنه من جمع القرآن على عهد النبي صلى…٥٠٠٤وثمامة، عن أنس، قال مات النبي صلى الله عليه وسلم ولم يجمع القرآن غير…٥٠٠٥وأبى يقول أخذته من في رسول الله صلى الله عليه وسلم فلا أتركه لشىء قال…٥٠٠٦ألم يقل الله { استجيبوا لله وللرسول إذا دعاكم } ثم قال ألا أعلمك أعظم…٥٠٠٧وما كان يدريه أنها رقية اقسموا واضربوا لي بسهم٥٠٠٨حدثنا محمد بن كثير، أخبرنا شعبة، عن سليمان، عن إبراهيم، عن عبد…٥٠٠٩من قرأ بالآيتين من آخر سورة البقرة في ليلة كفتاه٥٠١٠صدقك وهو كذوب ذاك شيطان٥٠١١تلك السكينة تنزلت بالقرآن٥٠١٢لقد أنزلت على الليلة سورة لهي أحب إلى مما طلعت عليه الشمس٥٠١٣والذي نفسي بيده إنها لتعدل ثلث القرآن٥٠١٤قتادة بن النعمان أن رجلا، قام في زمن النبي صلى الله عليه وسلم يقرأ من…٥٠١٥أيعجز أحدكم أن يقرأ ثلث القرآن في ليلة٥٠١٦رضى الله عنها أن رسول الله صلى الله عليه وسلم كان إذا اشتكى يقرأ على…٥٠١٧أن النبي صلى الله عليه وسلم كان إذا أوى إلى فراشه كل ليلة جمع كفيه ثم…٥٠١٨اقرأ يا ابن حضير اقرأ يا ابن حضير٥٠١٩حدثنا قتيبة بن سعيد، حدثنا سفيان، عن عبد العزيز بن رفيع، قال دخلت أنا…٥٠٢٠مثل الذي يقرأ القرآن كالأترجة طعمها طيب وريحها طيب والذي لا يقرأ…٥٠٢١إنما أجلكم في أجل من خلا من الأمم كما بين صلاة العصر ومغرب الشمس،…٥٠٢٢قال سألت عبد الله بن أبي أوفى أوصى النبي صلى الله عليه وسلم فقال لا .…٥٠٢٣لم يأذن الله لشىء ما أذن للنبي صلى الله عليه وسلم يتغنى بالقرآن٥٠٢٤ما أذن الله لشىء ما أذن للنبي أن يتغنى بالقرآن٥٠٢٥لا حسد إلا على اثنتين، رجل آتاه الله الكتاب وقام به آناء الليل، ورجل…٥٠٢٦لا حسد إلا في اثنتين رجل علمه الله القرآن فهو يتلوه آناء الليل وآناء…٥٠٢٧خيركم من تعلم القرآن وعلمه٥٠٢٨إن أفضلكم من تعلم القرآن وعلمه٥٠٢٩ما لي في النساء من حاجة٥٠٣٠هل عندك من شىء٥٠٣١إنما مثل صاحب القرآن كمثل صاحب الإبل المعقلة إن عاهد عليها أمسكها وإن…٥٠٣٢بئس ما لأحدهم أن يقول نسيت آية كيت وكيت بل نسي، واستذكروا القرآن فإنه…٥٠٣٢.٢حدثنا عثمان، حدثنا جرير، عن منصور، مثله . تابعه بشر عن ابن المبارك،…٥٠٣٣تعاهدوا القرآن فوالذي نفسي بيده لهو أشد تفصيا من الإبل في عقلها٥٠٣٤قال رأيت رسول الله صلى الله عليه وسلم يوم فتح مكة وهو يقرأ على راحلته…٥٠٣٥قال إن الذي تدعونه المفصل هو المحكم، قال وقال ابن عباس توفي رسول الله…٥٠٣٦حدثنا يعقوب بن إبراهيم، حدثنا هشيم، أخبرنا أبو بشر، عن سعيد بن جبير،…٥٠٣٧يرحمه الله لقد أذكرني كذا وكذا آية من سورة كذا٥٠٣٧.٢وقال، أسقطتهن من سورة كذا . تابعه علي بن مسهر وعبدة عن هشام .٥٠٣٨يرحمه الله لقد أذكرني كذا وكذا آية كنت أنسيتها من سورة كذا وكذا٥٠٣٩ما لأحدهم يقول نسيت آية كيت وكيت . بل هو نسي٥٠٤٠الآيتان من آخر سورة البقرة من قرأ بهما في ليلة كفتاه٥٠٤١وعبد الرحمن بن عبد القاري، أنهما سمعا عمر بن الخطاب، يقول سمعت هشام…٥٠٤٢يرحمه الله لقد أذكرني كذا وكذا آية، أسقطتها من سورة كذا وكذا٥٠٤٣حدثنا أبو النعمان، حدثنا مهدي بن ميمون، حدثنا واصل، عن أبي وائل، عن…٥٠٤٤فإذا ذهب قرأه كما وعده الله .٥٠٤٥صلى الله عليه وسلم فقال كان يمد مدا .٥٠٤٦قال سئل أنس كيف كانت قراءة النبي صلى الله عليه وسلم . فقال كانت مدا .…٥٠٤٧قال رأيت النبي صلى الله عليه وسلم يقرأ وهو على ناقته أو جمله وهى تسير…٥٠٤٨يا أبا موسى لقد أوتيت مزمارا من مزامير آل داود٥٠٤٩اقرأ على القرآن٥٠٥٠حدثنا محمد بن يوسف، حدثنا سفيان، عن الأعمش، عن إبراهيم، عن عبيدة، عن…٥٠٥١قال لي ابن شبرمة نظرت كم يكفي الرجل من القرآن فلم أجد سورة أقل من…٥٠٥١.٢أن من قرأ بالآيتين من آخر سورة البقرة في ليلة كفتاه٥٠٥٢حدثنا موسى، حدثنا أبو عوانة، عن مغيرة، عن مجاهد، عن عبد الله بن عمرو،…٥٠٥٣في كم تقرأ القرآن٥٠٥٤اقرإ القرآن في شهر٥٠٥٥حدثنا صدقة، أخبرنا يحيى، عن سفيان، عن سليمان، عن إبراهيم، عن عبيدة،…٥٠٥٦حدثنا قيس بن حفص، حدثنا عبد الواحد، حدثنا الأعمش، عن إبراهيم، عن…٥٠٥٧يأتي في آخر الزمان قوم حدثاء الأسنان، سفهاء الأحلام، يقولون من خير…٥٠٥٨يخرج فيكم قوم تحقرون صلاتكم مع صلاتهم، وصيامكم مع صيامهم، وعملكم مع…٥٠٥٩المؤمن الذي يقرأ القرآن ويعمل به كالأترجة، طعمها طيب وريحها طيب،…٥٠٦٠اقرءوا القرآن ما ائتلفت قلوبكم، فإذا اختلفتم فقوموا عنه٥٠٦١اقرءوا القرآن ما ائتلفت عليه قلوبكم فإذا اختلفتم فقوموا عنه٥٠٦٢كلاكما محسن فاقرآ أكبر علمي قال فإن من كان قبلكم اختلفوا فأهلكهم
