١٥ حديثًا
١٨٠٦من أجل أن رسول الله صلى الله عليه وسلم كان أهل بعمرة عام الحديبية .١٨٠٧إنما شأنهما واحد، أشهدكم أني قد أوجبت حجة مع عمرتي١٨٠٨أن بعض بني عبد الله، قال له لو أقمت . بهذا .١٨٠٩قال قال ابن عباس رضى الله عنهما قد أحصر رسول الله صلى الله عليه وسلم…١٨١٠قال كان ابن عمر رضى الله عنهما يقول أليس حسبكم سنة رسول الله صلى الله…١٨١١وأمر أصحابه بذلك .١٨١٢قال وحدث نافع، أن عبد الله، وسالما، كلما عبد الله بن عمر رضى الله…١٨١٣من أجل أن النبي صلى الله عليه وسلم كان أهل بعمرة عام الحديبية، ثم إن…١٨١٤لعلك آذاك هوامك١٨١٥أن كعب بن عجرة، حدثه قال وقف على رسول الله صلى الله عليه وسلم…١٨١٦ما كنت أرى الوجع بلغ بك ما أرى أو ما كنت أرى الجهد بلغ بك ما أرى، تجد…١٨١٧ولم يتبين لهم أنهم يحلون بها، وهم على طمع أن يدخلوا مكة، فأنزل الله…١٨١٨وقمله يسقط على وجهه . مثله .١٨١٩من حج هذا البيت، فلم يرفث ولم يفسق، رجع كما ولدته أمه١٨٢٠من حج هذا البيت، فلم يرفث، ولم يفسق، رجع كيوم ولدته أمه
