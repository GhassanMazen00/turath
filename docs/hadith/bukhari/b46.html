٤٣ حديثًا
٢٤٤٠إذا خلص المؤمنون من النار حبسوا بقنطرة بين الجنة والنار، فيتقاصون…٢٤٤١إن الله يدني المؤمن فيضع عليه كنفه، ويستره فيقول أتعرف ذنب كذا أتعرف…٢٤٤٢المسلم أخو المسلم، لا يظلمه ولا يسلمه، ومن كان في حاجة أخيه كان الله…٢٤٤٣انصر أخاك ظالما أو مظلوما٢٤٤٤انصر أخاك ظالما أو مظلوما٢٤٤٥واتباع الجنائز، وتشميت العاطس، ورد السلام، ونصر المظلوم، وإجابة…٢٤٤٦المؤمن للمؤمن كالبنيان يشد بعضه بعضا٢٤٤٧الظلم ظلمات يوم القيامة٢٤٤٨اتق دعوة المظلوم، فإنها ليس بينها وبين الله حجاب٢٤٤٩من كانت له مظلمة لأحد من عرضه أو شىء فليتحلله منه اليوم، قبل أن لا…٢٤٥٠ليس بمستكثر منها، يريد أن يفارقها، فتقول أجعلك من شأني في حل . فنزلت…٢٤٥١أتأذن لي أن أعطي هؤلاء٢٤٥٢من ظلم من الأرض شيئا طوقه من سبع أرضين٢٤٥٣من ظلم قيد شبر من الأرض طوقه من سبع أرضين٢٤٥٤من أخذ من الأرض شيئا بغير حقه خسف به يوم القيامة إلى سبع أرضين٢٤٥٥إلا أن يستأذن الرجل منكم أخاه .٢٤٥٦إن هذا قد اتبعنا أتأذن له٢٤٥٧إن أبغض الرجال إلى الله الألد الخصم٢٤٥٨إنما أنا بشر وإنه يأتيني الخصم، فلعل بعضكم أن يكون أبلغ من بعض، فأحسب…٢٤٥٩أربع من كن فيه كان منافقا، أو كانت فيه خصلة من أربعة كانت فيه خصلة من…٢٤٦٠لا حرج عليك أن تطعميهم بالمعروف٢٤٦١إن نزلتم بقوم، فأمر لكم بما ينبغي للضيف فاقبلوا، فإن لم يفعلوا فخذوا…٢٤٦٢أن ابن عباس، أخبره عن عمر رضى الله عنهم قال حين توفى الله نبيه صلى…٢٤٦٣لا يمنع جار جاره أن يغرز خشبه في جداره٢٤٦٤ألا إن الخمر قد حرمت٢٤٦٥إياكم والجلوس على الطرقات٢٤٦٦بينا رجل بطريق، اشتد عليه العطش فوجد بئرا فنزل فيها فشرب، ثم خرج،…٢٤٦٧هل ترون ما أرى إني أرى مواقع الفتن خلال بيوتكم كمواقع القطر٢٤٦٨فتبرز حتى جاء، فسكبت على يديه من الإداوة، فتوضأ فقلت يا أمير المؤمنين…٢٤٦٩لا، ولكني آليت منهن شهرا٢٤٧٠الثمن والجمل لك٢٤٧١أو قال لقد أتى النبي صلى الله عليه وسلم سباطة قوم فبال قائما .٢٤٧٢بينما رجل يمشي بطريق، وجد غصن شوك فأخذه، فشكر الله له، فغفر له٢٤٧٣حدثنا موسى بن إسماعيل، حدثنا جرير بن حازم، عن الزبير بن خريت، عن…٢٤٧٤حدثنا آدم بن أبي إياس، حدثنا شعبة، حدثنا عدي بن ثابت، سمعت عبد الله…٢٤٧٥لا يزني الزاني حين يزني وهو مؤمن، ولا يشرب الخمر حين يشرب وهو مؤمن،…٢٤٧٦لا تقوم الساعة حتى ينزل فيكم ابن مريم حكما مقسطا، فيكسر الصليب، ويقتل…٢٤٧٧على ما توقد هذه النيران٢٤٧٨وحول الكعبة ثلاثمائة وستون نصبا فجعل يطعنها بعود في يده وجعل يقول {…٢٤٧٩فهتكه النبي صلى الله عليه وسلم، فاتخذت منه نمرقتين، فكانتا في البيت…٢٤٨٠من قتل دون ماله فهو شهيد٢٤٨١فأرسلت إحدى أمهات المؤمنين مع خادم بقصعة فيها طعام فضربت بيدها، فكسرت…٢٤٨٢كان رجل في بني إسرائيل، يقال له جريج، يصلي، فجاءته أمه فدعته، فأبى أن…
