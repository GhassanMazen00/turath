كتاب المزارعة

٢٨ حديثًا
٢٣٢٠ما من مسلم يغرس غرسا، أو يزرع زرعا، فيأكل منه طير أو إنسان أو بهيمة،…٢٣٢١لا يدخل هذا بيت قوم إلا أدخله الذل٢٣٢٢من أمسك كلبا فإنه ينقص كل يوم من عمله قيراط، إلا كلب حرث أو ماشية٢٣٢٣من اقتنى كلبا لا يغني عنه زرعا ولا ضرعا، نقص كل يوم من عمله قيراط٢٣٢٤بينما رجل راكب على بقرة التفتت إليه . فقالت لم أخلق لهذا، خلقت…٢٣٢٥حدثنا الحكم بن نافع، أخبرنا شعيب، حدثنا أبو الزناد، عن الأعرج، عن أبي…٢٣٢٦وهى البويرة، ولها يقول حسان وهان على سراة بني لؤى حريق بالبويرة…٢٣٢٧سمع رافع بن خديج، قال كنا أكثر أهل المدينة مزدرعا، كنا نكري الأرض…٢٣٢٨أن عبد الله بن عمر رضى الله عنهما أخبره أن النبي صلى الله عليه وسلم…٢٣٢٩حدثنا مسدد، حدثنا يحيى بن سعيد، عن عبيد الله، حدثني نافع، عن ابن عمر…٢٣٣٠أن يمنح أحدكم أخاه خير له من أن يأخذ عليه خرجا معلوما٢٣٣١ولهم شطر ما خرج منها .٢٣٣٢سمع حنظلة الزرقي، عن رافع رضى الله عنه قال كنا أكثر أهل المدينة حقلا،…٢٣٣٣بينما ثلاثة نفر يمشون أخذهم المطر، فأووا إلى غار في جبل، فانحطت على…٢٣٣٤قال قال عمر رضى الله عنه لولا آخر المسلمين ما فتحت قرية إلا قسمتها…٢٣٣٥من أعمر أرضا ليست لأحد فهو أحق٢٣٣٦يتحرى معرس رسول الله صلى الله عليه وسلم وهو أسفل من المسجد الذي ببطن…٢٣٣٧الليلة أتاني آت من ربي وهو بالعقيق أن صل في هذا الوادي المبارك وقل…٢٣٣٨نقركم بها على ذلك ما شئنا٢٣٣٩ما تصنعون بمحاقلكم٢٣٤١من كانت له أرض فليزرعها أو ليمنحها، فإن لم يفعل فليمسك أرضه٢٣٤٢أن يمنح أحدكم أخاه خير له من أن يأخذ شيئا معلوما٢٣٤٤فذهب ابن عمر إلى رافع فذهبت معه، فسأله فقال نهى النبي صلى الله عليه…٢٣٤٥أن عبد الله بن عمر رضى الله عنهما قال كنت أعلم في عهد رسول الله صلى…٢٣٤٧لما فيه من المخاطرة .٢٣٤٨أن رجلا من أهل الجنة استأذن ربه في الزرع فقال له ألست فيما شئت قال…٢٣٤٩حدثنا قتيبة بن سعيد، حدثنا يعقوب، عن أبي حازم، عن سهل بن سعد رضى الله…٢٣٥٠لن يبسط أحد منكم ثوبه حتى أقضي مقالتي هذه، ثم يجمعه إلى صدره، فينسى…
