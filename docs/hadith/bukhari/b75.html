٣٧ حديثًا
٥٦٤٠ما من مصيبة تصيب المسلم إلا كفر الله بها عنه، حتى الشوكة يشاكها٥٦٤٢ما يصيب المسلم من نصب ولا وصب ولا هم ولا حزن ولا أذى ولا غم حتى…٥٦٤٣مثل المؤمن كالخامة من الزرع تفيئها الريح مرة، وتعدلها مرة، ومثل…٥٦٤٤مثل المؤمن كمثل الخامة من الزرع من حيث أتتها الريح كفأتها، فإذا…٥٦٤٥من يرد الله به خيرا يصب منه٥٦٤٦حدثنا قبيصة، حدثنا سفيان، عن الأعمش، . حدثني بشر بن محمد، أخبرنا عبد…٥٦٤٧أجل ما من مسلم يصيبه أذى، إلا حات الله عنه خطاياه، كما تحات ورق الشجر٥٦٤٨أجل إني أوعك كما يوعك رجلان منكم٥٦٤٩أطعموا الجائع، وعودوا المريض، وفكوا العاني٥٦٥٠ولبس الحرير، والديباج، والإستبرق، وعن القسي، والميثرة، وأمرنا أن نتبع…٥٦٥١سمع جابر بن عبد الله رضى الله عنهما يقول مرضت مرضا، فأتاني النبي صلى…٥٦٥٢إن شئت صبرت ولك الجنة وإن شئت دعوت الله أن يعافيك٥٦٥٣إن الله قال إذا ابتليت عبدي بحبيبتيه فصبر عوضته منهما الجنة٥٦٥٤اللهم حبب إلينا المدينة كحبنا مكة أو أشد، اللهم وصححها، وبارك لنا في…٥٦٥٥إن لله ما أخذ وما أعطى وكل شىء عنده مسمى فلتحتسب ولتصبر٥٦٥٦لا بأس طهور إن شاء الله٥٦٥٧لما حضر أبو طالب جاءه النبي صلى الله عليه وسلم .٥٦٥٨إن الإمام ليؤتم به، فإذا ركع فاركعوا وإذا رفع فارفعوا، وإن صلى جالسا…٥٦٥٩أن أباها، قال تشكيت بمكة شكوا شديدا، فجاءني النبي صلى الله عليه وسلم…٥٦٦٠أجل إني أوعك كما يوعك رجلان منكم٥٦٦١أجل، وما من مسلم يصيبه أذى إلا حاتت عنه خطاياه كما تحات ورق الشجر٥٦٦٢لا بأس طهور إن شاء الله٥٦٦٣أى سعد ألم تسمع ما قال أبو حباب٥٦٦٤حدثنا عمرو بن عباس، حدثنا عبد الرحمن، حدثنا سفيان، عن محمد هو ابن…٥٦٦٥أيؤذيك هوام رأسك٥٦٦٦ذاك لو كان وأنا حى، فأستغفر لك وأدعو لك٥٦٦٧أجل كما يوعك رجلان منكم٥٦٦٨أن تدع ورثتك أغنياء خير من أن تذرهم عالة يتكففون الناس ولن تنفق نفقة…٥٦٦٩هلم أكتب لكم كتابا لا تضلوا بعده٥٦٧٠يقول ذهبت بي خالتي إلى رسول الله صلى الله عليه وسلم فقالت يا رسول…٥٦٧١لا يتمنين أحدكم الموت من ضر أصابه، فإن كان لا بد فاعلا فليقل اللهم…٥٦٧٢ثم أتيناه مرة أخرى وهو يبني حائطا له فقال إن المسلم ليوجر في كل شىء…٥٦٧٣لن يدخل أحدا عمله الجنة٥٦٧٤اللهم اغفر لي وارحمني وألحقني بالرفيق الأعلى٥٦٧٥أذهب الباس رب الناس، اشف وأنت الشافي لا شفاء إلا شفاؤك، شفاء لا يغادر…٥٦٧٦فكيف الميراث فنزلت آية الفرائض .٥٦٧٧اللهم حبب إلينا المدينة كحبنا مكة أو أشد وصححها وبارك لنا في صاعها…
