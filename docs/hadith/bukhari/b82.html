٢٦ حديثًا
٦٥٩٤إن أحدكم يجمع في بطن أمه أربعين يوما، ثم علقة مثل ذلك، ثم يكون مضغة…٦٥٩٥وكل الله بالرحم ملكا فيقول أى رب نطفة، أى رب علقة، أى رب مضغة . فإذا…٦٥٩٦قال قال رجل يا رسول الله أيعرف أهل الجنة من أهل النار قال " نعم " .…٦٥٩٧الله أعلم بما كانوا عاملين٦٥٩٨أنه سمع أبا هريرة، يقول سئل رسول الله صلى الله عليه وسلم عن ذراري…٦٦٠٠ما من مولود إلا يولد على الفطرة، فأبواه يهودانه وينصرانه، كما تنتجون…٦٦٠١لا تسأل المرأة طلاق أختها لتستفرغ صحفتها، ولتنكح، فإن لها ما قدر لها٦٦٠٢لله ما أخذ، ولله ما أعطى، كل بأجل، فلتصبر ولتحتسب٦٦٠٣أوإنكم تفعلون ذلك، لا عليكم أن لا تفعلوا، فإنه ليست نسمة كتب الله أن…٦٦٠٤ما ترك فيها شيئا إلى قيام الساعة إلا ذكره، علمه من علمه، وجهله من…٦٦٠٥ما منكم من أحد إلا قد كتب مقعده من النار أو من الجنة٦٦٠٦هذا من أهل النار٦٦٠٧من أحب أن ينظر إلى الرجل من أهل النار فلينظر إلى هذا٦٦٠٨إنه لا يرد شيئا، وإنما يستخرج به من البخيل٦٦٠٩لا يأتي ابن آدم النذر بشىء لم يكن قد قدرته، ولكن يلقيه القدر وقد…٦٦١٠يا أيها الناس اربعوا على أنفسكم فإنكم لا تدعون أصم ولا غائبا إنما…٦٦١١ما استخلف خليفة إلا له بطانتان بطانة تأمره بالخير وتحضه عليه، وبطانة…٦٦١٢إن الله كتب على ابن آدم حظه من الزنا، أدرك ذلك لا محالة، فزنا العين…٦٦١٣حدثنا الحميدي، حدثنا سفيان، حدثنا عمرو، عن عكرمة، عن ابن عباس رضى…٦٦١٤احتج آدم وموسى، فقال له موسى يا آدم أنت أبونا خيبتنا وأخرجتنا من…٦٦١٥لا إله إلا الله، وحده لا شريك له، اللهم لا مانع لما أعطيت، ولا معطي…٦٦١٦تعوذوا بالله من جهد البلاء، ودرك الشقاء، وسوء القضاء، وشماتة الأعداء٦٦١٧قال كثيرا مما كان النبي صلى الله عليه وسلم يحلف ( لا ومقلب القلوب ) .٦٦١٨وبشر بن محمد، قالا أخبرنا عبد الله، أخبرنا معمر، عن الزهري، عن سالم،…٦٦١٩كان عذابا يبعثه الله على من يشاء، فجعله الله رحمة للمؤمنين، ما من عبد…٦٦٢٠والله لولا الله ما اهتدينا، ولا صمنا ولا صلينا، فأنزلن سكينة علينا،…
