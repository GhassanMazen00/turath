كتاب الغسل

٤٦ حديثًا
٢٤٨زوج النبي صلى الله عليه وسلم أن النبي صلى الله عليه وسلم كان إذا…٢٤٩زوج النبي صلى الله عليه وسلم قالت توضأ رسول الله صلى الله عليه وسلم…٢٥٠صلى الله عليه وسلم من إناء واحد من قدح يقال له الفرق .٢٥١صلى الله عليه وسلم فدعت بإناء نحوا من صاع، فاغتسلت وأفاضت على رأسها،…٢٥٢أنه كان عند جابر بن عبد الله هو وأبوه، وعنده قوم فسألوه عن الغسل، .…٢٥٣والصحيح ما روى أبو نعيم .٢٥٤أما أنا فأفيض على رأسي ثلاثا٢٥٥قال كان النبي صلى الله عليه وسلم يفرغ على رأسه ثلاثا .٢٥٦قال قال لي جابر أتاني ابن عمك يعرض بالحسن بن محمد ابن الحنفية قال كيف…٢٥٧قال قالت ميمونة وضعت للنبي صلى الله عليه وسلم ماء للغسل، فغسل يديه…٢٥٨قالت كان النبي صلى الله عليه وسلم إذا اغتسل من الجنابة دعا بشىء نحو…٢٥٩قالت صببت للنبي صلى الله عليه وسلم غسلا، فأفرغ بيمينه على يساره…٢٦٠أن النبي صلى الله عليه وسلم اغتسل من الجنابة، فغسل فرجه بيده، ثم دلك…٢٦١حدثنا عبد الله بن مسلمة، أخبرنا أفلح، عن القاسم، عن عائشة، قالت كنت…٢٦٢قالت كان رسول الله صلى الله عليه وسلم إذا اغتسل من الجنابة غسل يده .٢٦٣حدثنا أبو الوليد، قال حدثنا شعبة، عن أبي بكر بن حفص، عن عروة، عن…٢٦٤حدثنا أبو الوليد، قال حدثنا شعبة، عن عبد الله بن عبد الله بن جبر، قال…٢٦٥قال قالت ميمونة وضعت لرسول الله صلى الله عليه وسلم ماء يغتسل به،…٢٦٦قالت وضعت لرسول الله صلى الله عليه وسلم غسلا وسترته، فصب على يده،…٢٦٧ويحيى بن سعيد، عن شعبة، عن إبراهيم بن محمد بن المنتشر، عن أبيه، قال…٢٦٨قال كان النبي صلى الله عليه وسلم يدور على نسائه في الساعة الواحدة من…٢٦٩توضأ واغسل ذكرك٢٧٠قال سألت عائشة فذكرت لها قول ابن عمر ما أحب أن أصبح، محرما أنضخ طيبا…٢٧١قالت كأني أنظر إلى وبيص الطيب في مفرق النبي صلى الله عليه وسلم وهو…٢٧٢قالت كان رسول الله صلى الله عليه وسلم إذا اغتسل من الجنابة غسل يديه،…٢٧٣قالت كان رسول الله صلى الله عليه وسلم إذا اغتسل من الجنابة غسل يديه،…٢٧٤قالت وضع رسول الله صلى الله عليه وسلم وضوءا لجنابة فأكفأ بيمينه على…٢٧٥قال أقيمت الصلاة، وعدلت الصفوف قياما، فخرج إلينا رسول الله صلى الله…٢٧٦قال قالت ميمونة وضعت للنبي صلى الله عليه وسلم غسلا، فسترته بثوب، وصب…٢٧٧أخذت بيديها ثلاثا فوق رأسها، ثم تأخذ بيدها على شقها الأيمن، وبيدها…٢٧٨كانت بنو إسرائيل يغتسلون عراة، ينظر بعضهم إلى بعض، وكان موسى يغتسل…٢٧٩بينا أيوب يغتسل عريانا فخر عليه جراد من ذهب، فجعل أيوب يحتثي في ثوبه،…٢٨٠مولى عمر بن عبيد الله أن أبا مرة، مولى أم هانئ بنت أبي طالب أخبره…٢٨١قالت سترت النبي صلى الله عليه وسلم وهو يغتسل من الجنابة، فغسل يديه ثم…٢٨٢نعم إذا رأت الماء٢٨٣أين كنت يا أبا هريرة٢٨٤أن أنس بن مالك، حدثهم أن نبي الله صلى الله عليه وسلم كان يطوف على…٢٨٥أين كنت يا أبا هر٢٨٦وشيبان، عن يحيى، عن أبي سلمة، قال سألت عائشة أكان النبي صلى الله عليه…٢٨٧نعم إذا توضأ أحدكم فليرقد وهو جنب٢٨٨قالت كان النبي صلى الله عليه وسلم إذا أراد أن ينام وهو جنب، غسل فرجه،…٢٨٩إذا توضأ " .٢٩٠توضأ واغسل ذكرك ثم نم٢٩١إذا جلس بين شعبها الأربع ثم جهدها، فقد وجب الغسل٢٩٢أن عطاء بن يسار، أخبره أن زيد بن خالد الجهني أخبره أنه، سأل عثمان بن…٢٩٣يغسل ما مس المرأة منه، ثم يتوضأ ويصلي
