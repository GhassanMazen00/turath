١١٣ حديثًا
١٣٥لا تقبل صلاة من أحدث حتى يتوضأ١٣٦إن أمتي يدعون يوم القيامة غرا محجلين من آثار الوضوء، فمن استطاع منكم…١٣٧لا ينفتل أو لا ينصرف حتى يسمع صوتا أو يجد ريحا١٣٨فقام النبي صلى الله عليه وسلم من الليل، فلما كان في بعض الليل قام…١٣٩أنه سمعه يقول دفع رسول الله صلى الله عليه وسلم من عرفة حتى إذا كان…١٤٠أنه توضأ فغسل وجهه، ثم أخذ غرفة من ماء، فمضمض بها واستنشق، ثم أخذ…١٤١لو أن أحدكم إذا أتى أهله قال بسم الله اللهم جنبنا الشيطان وجنب…١٤٢اللهم إني أعوذ بك من الخبث والخبائث١٤٣أن النبي صلى الله عليه وسلم دخل الخلاء، فوضعت له وضوءا قال " من وضع…١٤٤إذا أتى أحدكم الغائط فلا يستقبل القبلة ولا يولها ظهره، شرقوا أو غربوا١٤٥واسع بن حبان، عن عبد الله بن عمر، أنه كان يقول إن ناسا يقولون إذا…١٤٦أن أزواج النبي، صلى الله عليه وسلم كن يخرجن بالليل إذا تبرزن إلى…١٤٧قد أذن أن تخرجن في حاجتكن١٤٨قال ارتقيت فوق ظهر بيت حفصة لبعض حاجتي، فرأيت رسول الله صلى الله عليه…١٤٩أن عمه، واسع بن حبان، أخبره أن عبد الله بن عمر أخبره قال لقد ظهرت ذات…١٥٠حدثنا أبو الوليد، هشام بن عبد الملك قال حدثنا شعبة، عن أبي معاذ واسمه…١٥١حدثنا سليمان بن حرب، قال حدثنا شعبة، عن أبي معاذ هو عطاء بن أبي…١٥٢سمع أنس بن مالك، يقول كان رسول الله صلى الله عليه وسلم يدخل الخلاء،…١٥٣إذا شرب أحدكم فلا يتنفس في الإناء، وإذا أتى الخلاء فلا يمس ذكره…١٥٤إذا بال أحدكم فلا يأخذن ذكره بيمينه، ولا يستنجي بيمينه، ولا يتنفس في…١٥٥ابغني أحجارا أستنفض بها أو نحوه ولا تأتني بعظم ولا روث١٥٦أنه سمع عبد الله، يقول أتى النبي صلى الله عليه وسلم الغائط، فأمرني أن…١٥٧قال توضأ النبي صلى الله عليه وسلم مرة مرة .١٥٨أن النبي صلى الله عليه وسلم توضأ مرتين مرتين .١٥٩من توضأ نحو وضوئي هذا، ثم صلى ركعتين، لا يحدث فيهما نفسه، غفر له ما…١٦٠لا يتوضأ رجل فيحسن وضوءه، ويصلي الصلاة إلا غفر له ما بينه وبين الصلاة…١٦١من توضأ فليستنثر، ومن استجمر فليوتر١٦٢إذا توضأ أحدكم فليجعل في أنفه ثم لينثر، ومن استجمر فليوتر، وإذا…١٦٣ويل للأعقاب من النار١٦٤من توضأ نحو وضوئي هذا ثم صلى ركعتين، لا يحدث فيهما نفسه، غفر الله له…١٦٥ويل للأعقاب من النار١٦٦أنه قال لعبد الله بن عمر يا أبا عبد الرحمن، رأيتك تصنع أربعا لم أر…١٦٧ابدأن بميامنها ومواضع الوضوء منها١٦٨قالت كان النبي صلى الله عليه وسلم يعجبه التيمن في تنعله وترجله وطهوره…١٦٩أنه قال رأيت رسول الله صلى الله عليه وسلم وحانت صلاة العصر، فالتمس…١٧٠أو من قبل أهل أنس فقال لأن تكون عندي شعرة منه أحب إلى من الدنيا وما…١٧١أن رسول الله صلى الله عليه وسلم لما حلق رأسه كان أبو طلحة أول من أخذ…١٧٢إذا شرب الكلب في إناء أحدكم فليغسله سبعا١٧٣أن رجلا رأى كلبا يأكل الثرى من العطش، فأخذ الرجل خفه فجعل يغرف له به…١٧٤قال كانت الكلاب تبول وتقبل وتدبر في المسجد في زمان رسول الله صلى الله…١٧٥إذا أرسلت كلبك المعلم فقتل فكل، وإذا أكل فلا تأكل، فإنما أمسكه على…١٧٦لا يزال العبد في صلاة ما كان في المسجد ينتظر الصلاة، ما لم يحدث١٧٧لا ينصرف حتى يسمع صوتا أو يجد ريحا١٧٨قال قال علي كنت رجلا مذاء، فاستحييت أن أسأل رسول الله صلى الله عليه…١٧٩أن عطاء بن يسار، أخبره أن زيد بن خالد أخبره أنه، سأل عثمان بن عفان…١٨٠أن رسول الله صلى الله عليه وسلم أرسل إلى رجل من الأنصار فجاء ورأسه…١٨١أن رسول الله صلى الله عليه وسلم لما أفاض من عرفة عدل إلى الشعب، فقضى…١٨٢أن نافع بن جبير بن مطعم، أخبره أنه، سمع عروة بن المغيرة بن شعبة، يحدث…١٨٣مولى ابن عباس أن عبد الله بن عباس، أخبره أنه، بات ليلة عند ميمونة زوج…١٨٤ما من شىء كنت لم أره إلا قد رأيته في مقامي هذا حتى الجنة والنار، ولقد…١٨٥أن رجلا، قال لعبد الله بن زيد وهو جد عمرو بن يحيى أتستطيع أن تريني،…١٨٦صلى الله عليه وسلم فدعا بتور من ماء، فتوضأ لهم وضوء النبي صلى الله…١٨٧فأتي بوضوء فتوضأ، فجعل الناس يأخذون من فضل وضوئه فيتمسحون به، فصلى…١٨٨وقال أبو موسى دعا النبي صلى الله عليه وسلم بقدح فيه ماء، فغسل يديه…١٨٩حدثنا علي بن عبد الله، قال حدثنا يعقوب بن إبراهيم بن سعد، قال حدثنا…١٩٠يقول ذهبت بي خالتي إلى النبي صلى الله عليه وسلم فقالت يا رسول الله،…١٩١أنه أفرغ من الإناء على يديه فغسلهما، ثم غسل أو مضمض، واستنشق من كفة…١٩٢صلى الله عليه وسلم فدعا بتور من ماء، فتوضأ لهم، فكفأ على يديه فغسلهما…١٩٣أنه قال كان الرجال والنساء يتوضئون في زمان رسول الله صلى الله عليه…١٩٤يقول جاء رسول الله صلى الله عليه وسلم يعودني، وأنا مريض لا أعقل،…١٩٥سمع عبد الله بن بكر، قال حدثنا حميد، عن أنس، قال حضرت الصلاة، فقام من…١٩٦أن النبي صلى الله عليه وسلم دعا بقدح فيه ماء، فغسل يديه ووجهه فيه ومج…١٩٧فغسل وجهه ثلاثا ويديه مرتين مرتين، ومسح برأسه فأقبل به وأدبر، وغسل…١٩٨هريقوا على من سبع قرب، لم تحلل أوكيتهن، لعلي أعهد إلى الناس١٩٩قال كان عمي يكثر من الوضوء، قال لعبد الله بن زيد أخبرني كيف رأيت…٢٠٠أن النبي صلى الله عليه وسلم دعا بإناء من ماء، فأتي بقدح رحراح فيه شىء…٢٠١يقول كان النبي صلى الله عليه وسلم يغسل أو كان يغتسل بالصاع إلى خمسة…٢٠٢حدثنا أصبغ بن الفرج المصري، عن ابن وهب، قال حدثني عمرو، حدثني أبو…٢٠٣فصب عليه حين فرغ من حاجته، فتوضأ ومسح على الخفين .٢٠٤أن أباه، أخبره أنه، رأى النبي صلى الله عليه وسلم يمسح على الخفين .…٢٠٥حدثنا عبدان، قال أخبرنا عبد الله، قال أخبرنا الأوزاعي، عن يحيى، عن…٢٠٦دعهما، فإني أدخلتهما طاهرتين٢٠٧. أن رسول الله صلى الله عليه وسلم أكل كتف شاة، ثم صلى ولم يتوضأ .٢٠٨أن أباه، أخبره أنه، رأى رسول الله صلى الله عليه وسلم يحتز من كتف شاة،…٢٠٩مولى بني حارثة أن سويد بن النعمان، أخبره . أنه، خرج مع رسول الله صلى…٢١٠أن النبي صلى الله عليه وسلم أكل عندها كتفا، ثم صلى ولم يتوضأ .٢١١وقتيبة، قالا حدثنا الليث، عن عقيل، عن ابن شهاب، عن عبيد الله بن عبد…٢١٢إذا نعس أحدكم وهو يصلي فليرقد حتى يذهب عنه النوم، فإن أحدكم إذا صلى…٢١٣إذا نعس أحدكم في الصلاة فلينم حتى يعلم ما يقرأ٢١٤حدثنا محمد بن يوسف، قال حدثنا سفيان، عن عمرو بن عامر، قال سمعت أنسا،…٢١٥فلما صلى دعا بالأطعمة، فلم يؤت إلا بالسويق، فأكلنا وشربنا، ثم قام…٢١٦يعذبان، وما يعذبان في كبير٢١٧قال كان النبي صلى الله عليه وسلم إذا تبرز لحاجته أتيته بماء فيغسل به .٢١٨إنهما ليعذبان، وما يعذبان في كبير أما أحدهما فكان لا يستتر من البول،…٢١٩أن النبي صلى الله عليه وسلم رأى أعرابيا يبول في المسجد فقال " دعوه "…٢٢٠دعوه وهريقوا على بوله سجلا من ماء، أو ذنوبا من ماء، فإنما بعثتم…٢٢١حدثنا عبدان، قال أخبرنا عبد الله، قال أخبرنا يحيى بن سعيد، قال سمعت…٢٢٢أنها قالت أتي رسول الله صلى الله عليه وسلم بصبي، فبال على ثوبه، فدعا…٢٢٣أنها أتت بابن لها صغير، لم يأكل الطعام إلى رسول الله صلى الله عليه…٢٢٤قال أتى النبي صلى الله عليه وسلم سباطة قوم فبال قائما، ثم دعا بماء،…٢٢٥صلى الله عليه وسلم نتماشى، فأتى سباطة قوم خلف حائط، فقام كما يقوم…٢٢٦قال كان أبو موسى الأشعري يشدد في البول ويقول إن بني إسرائيل كان إذا…٢٢٧تحته، ثم تقرصه بالماء، وتنضحه وتصلي فيه٢٢٨لا، إنما ذلك عرق، وليس بحيض، فإذا أقبلت حيضتك فدعي الصلاة، وإذا أدبرت…٢٢٩قالت كنت أغسل الجنابة من ثوب النبي صلى الله عليه وسلم، فيخرج إلى…٢٣٠يصيب الثوب فقالت كنت أغسله من ثوب رسول الله صلى الله عليه وسلم، فيخرج…٢٣١قال سألت سليمان بن يسار في الثوب تصيبه الجنابة قال قالت عائشة كنت…٢٣٢أنها كانت تغسل المني من ثوب النبي صلى الله عليه وسلم، ثم أراه فيه…٢٣٣قال قدم أناس من عكل أو عرينة، فاجتووا المدينة، فأمرهم النبي صلى الله…٢٣٤قال كان النبي صلى الله عليه وسلم يصلي قبل أن يبنى المسجد في مرابض…٢٣٥ألقوها وما حولها فاطرحوه . وكلوا سمنكم٢٣٦خذوها وما حولها فاطرحوه٢٣٧كل كلم يكلمه المسلم في سبيل الله يكون يوم القيامة كهيئتها إذ طعنت،…٢٣٨نحن الآخرون السابقون٢٣٩لا يبولن أحدكم في الماء الدائم الذي لا يجري، ثم يغتسل فيه٢٤٠اللهم عليك بقريش٢٤١حدثنا محمد بن يوسف، قال حدثنا سفيان، عن حميد، عن أنس، قال بزق النبي…٢٤٢كل شراب أسكر فهو حرام٢٤٣سمع سهل بن سعد الساعدي،، وسأله الناس، وما بيني وبينه أحد بأى شىء دووي…٢٤٤قال أتيت النبي صلى الله
