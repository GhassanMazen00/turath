كتاب الجهاد والسير

٢٩٤ حديثًا•صفحة ١ من ٢
٢٧٨٢الصلاة على ميقاتها٢٧٨٣لا هجرة بعد الفتح ولكن جهاد ونية، وإذا استنفرتم فانفروا٢٧٨٤لكن أفضل الجهاد حج مبرور٢٧٨٥لا أجده قال هل تستطيع إذا خرج المجاهد أن تدخل مسجدك فتقوم ولا تفتر…٢٧٨٦مؤمن يجاهد في سبيل الله بنفسه وماله٢٧٨٧مثل المجاهد في سبيل الله والله أعلم بمن يجاهد في سبيله كمثل الصائم…٢٧٨٨ناس من أمتي عرضوا على، غزاة في سبيل الله، يركبون ثبج هذا البحر، ملوكا…٢٧٩٠من آمن بالله وبرسوله وأقام الصلاة وصام رمضان، كان حقا على الله أن…٢٧٩١رأيت الليلة رجلين أتياني فصعدا بي الشجرة، فأدخلاني دارا هي أحسن…٢٧٩٢لغدوة في سبيل الله أو روحة خير من الدنيا وما فيها٢٧٩٣لقاب قوس في الجنة خير مما تطلع عليه الشمس وتغرب٢٧٩٤الروحة والغدوة في سبيل الله أفضل من الدنيا وما فيها٢٧٩٥ما من عبد يموت له عند الله خير، يسره أن يرجع إلى الدنيا، وأن له…٢٧٩٦لروحة في سبيل الله أو غدوة خير من الدنيا وما فيها، ولقاب قوس أحدكم من…٢٧٩٧والذي نفسي بيده لولا أن رجالا من المؤمنين لا تطيب أنفسهم أن يتخلفوا…٢٧٩٨أخذ الراية زيد فأصيب، ثم أخذها جعفر فأصيب، ثم أخذها عبد الله بن رواحة…٢٧٩٩أناس من أمتي عرضوا على يركبون هذا البحر الأخضر، كالملوك على الأسرة٢٨٠١فلما قدموا، قال لهم خالي أتقدمكم، فإن أمنوني حتى أبلغهم عن رسول الله…٢٨٠٢هل أنت إلا إصبع دميت، وفي سبيل الله ما لقيت٢٨٠٣والذي نفسي بيده لا يكلم أحد في سبيل الله والله أعلم بمن يكلم في سبيله…٢٨٠٤أن عبد الله بن عباس، أخبره أن أبا سفيان أخبره أن هرقل قال له سألتك…٢٨٠٦اللهم إني أعتذر إليك مما صنع هؤلاء يعني أصحابه وأبرأ إليك مما صنع…٢٨٠٧أن زيد بن ثابت رضى الله عنه قال نسخت الصحف في المصاحف، ففقدت آية من…٢٨٠٨فقتل، فقال رسول الله صلى الله عليه وسلم " عمل قليلا وأجر كثيرا " .٢٨٠٩يا أم حارثة، إنها جنان في الجنة، وإن ابنك أصاب الفردوس الأعلى٢٨١٠من قاتل لتكون كلمة الله هي العليا فهو في سبيل الله٢٨١١ما اغبرت قدما عبد في سبيل الله فتمسه النار٢٨١٢ويح عمار، تقتله الفئة الباغية، عمار يدعوهم إلى الله ويدعونه إلى النار٢٨١٣فأتاه جبريل وقد عصب رأسه الغبار فقال وضعت السلاح، فوالله ما وضعته .…٢٨١٤على رعل وذكوان وعصية عصت الله ورسوله، قال أنس أنزل في الذين قتلوا…٢٨١٥سمع جابر بن عبد الله رضى الله عنهما يقول اصطبح ناس الخمر يوم أحد، ثم…٢٨١٦لم تبكي أو لا تبكي، ما زالت الملائكة تظله بأجنحتها٢٨١٧ما أحد يدخل الجنة يحب أن يرجع إلى الدنيا وله ما على الأرض من شىء، إلا…٢٨١٨واعلموا أن الجنة تحت ظلال السيوف٢٨١٩قال سليمان بن داود عليهما السلام لأطوفن الليلة على مائة امرأة أو تسع…٢٨٢٠رضى الله عنه قال كان النبي صلى الله عليه وسلم أحسن الناس وأشجع الناس…٢٨٢١أعطوني ردائي، لو كان لي عدد هذه العضاه نعما لقسمته بينكم، ثم لا…٢٨٢٢اللهم إني أعوذ بك من الجبن، وأعوذ بك أن أرد إلى أرذل العمر، وأعوذ بك…٢٨٢٣اللهم إني أعوذ بك من العجز والكسل والجبن والهرم، وأعوذ بك من فتنة…٢٨٢٤حدثنا قتيبة بن سعيد، حدثنا حاتم، عن محمد بن يوسف، عن السائب بن يزيد،…٢٨٢٥لا هجرة بعد الفتح ولكن جهاد ونية، وإذا استنفرتم فانفروا٢٨٢٦يضحك الله إلى رجلين يقتل أحدهما الآخر يدخلان الجنة، يقاتل هذا في سبيل…٢٨٢٧حدثنا الحميدي، حدثنا سفيان، حدثنا الزهري، قال أخبرني عنبسة بن سعيد،…٢٨٢٨فلما قبض النبي صلى الله عليه وسلم لم أره مفطرا، إلا يوم فطر أو أضحى .٢٨٢٩الشهداء خمسة المطعون، والمبطون، والغرق وصاحب الهدم، والشهيد في سبيل…٢٨٣٠الطاعون شهادة لكل مسلم٢٨٣١فجاء بكتف فكتبها، وشكا ابن أم مكتوم ضرارته فنزلت { لا يستوي القاعدون…٢٨٣٢أنه قال رأيت مروان بن الحكم جالسا في المسجد، فأقبلت حتى جلست إلى…٢٨٣٣إذا لقيتموهم فاصبروا٢٨٣٤فلم يكن لهم عبيد يعملون ذلك لهم، فلما رأى ما بهم من النصب والجوع قال…٢٨٣٥حدثنا أبو معمر، حدثنا عبد الوارث، حدثنا عبد العزيز، عن أنس رضى الله…٢٨٣٦لولا أنت ما اهتدينا٢٨٣٧حدثنا حفص بن عمر، حدثنا شعبة، عن أبي إسحاق، عن البراء رضى الله عنه…٢٨٣٨أن أنسا، حدثهم قال رجعنا من غزوة تبوك مع النبي صلى الله عليه وسلم .٢٨٣٩إن أقواما بالمدينة خلفنا، ما سلكنا شعبا ولا واديا إلا وهم معنا فيه،…٢٨٤٠من صام يوما في سبيل الله بعد الله وجهه عن النار سبعين خريفا٢٨٤١من أنفق زوجين في سبيل الله دعاه خزنة الجنة، كل خزنة باب أى فل هلم٢٨٤٢إنما أخشى عليكم من بعدي ما يفتح عليكم من بركات الأرض٢٨٤٣من جهز غازيا في سبيل الله فقد غزا، ومن خلف غازيا في سبيل الله بخير…٢٨٤٤إني أرحمها، قتل أخوها معي٢٨٤٥يعني من الحنوط، ثم جاء فجلس، فذكر في الحديث انكشافا من الناس، فقال…٢٨٤٦من يأتيني بخبر القوم يوم الأحزاب٢٨٤٧إن لكل نبي حواريا، وإن حواري الزبير بن العوام٢٨٤٨أذنا وأقيما، وليؤمكما أكبركما٢٨٤٩الخيل في نواصيها الخير إلى يوم القيامة٢٨٥٠الخيل معقود في نواصيها الخير إلى يوم القيامة٢٨٥١البركة في نواصي الخيل٢٨٥٢الخيل معقود في نواصيها الخير إلى يوم القيامة الأجر والمغنم٢٨٥٣من احتبس فرسا في سبيل الله إيمانا بالله وتصديقا بوعده، فإن شبعه وريه…٢٨٥٤هل معكم منه شىء٢٨٥٥حدثنا علي بن عبد الله بن جعفر، حدثنا معن بن عيسى، حدثنا أبى بن عباس…٢٨٥٦يا معاذ، هل تدري حق الله على عباده وما حق العباد على الله٢٨٥٧ما رأينا من فزع، وإن وجدناه لبحرا٢٨٥٨إنما الشؤم في ثلاثة في الفرس والمرأة والدار٢٨٥٩إن كان في شىء ففي المرأة والفرس والمسكن٢٨٦٠الخيل لثلاثة لرجل أجر، ولرجل ستر، وعلى رجل وزر، فأما الذي له أجر فرجل…٢٨٦١من أحب أن يتعجل إلى أهله فليعجل٢٨٦٢ما رأينا من فزع، وإن وجدناه لبحرا٢٨٦٣حدثنا عبيد بن إسماعيل، عن أبي أسامة، عن عبيد الله، عن نافع، عن ابن…٢٨٦٤أنا النبي لا كذب أنا ابن عبد المطلب٢٨٦٥أهل من عند مسجد ذي الحليفة .٢٨٦٦ما عليه سرج، في عنقه سيف .٢٨٦٧وجدنا فرسكم هذا بحرا٢٨٦٨وبين ثنية إلى مسجد بني زريق ميل .٢٨٦٩وكان أمدها من الثنية إلى مسجد بني زريق . وأن عبد الله بن عمر كان سابق…٢٨٧٠وكان أمدها ثنية الوداع . فقلت لموسى فكم كان بين ذلك قال ستة أميال أو…٢٨٧١حدثنا عبد الله بن محمد، حدثنا معاوية، حدثنا أبو إسحاق، عن حميد، قال…٢٨٧٢حق على الله أن لا يرتفع شىء من الدنيا إلا وضعه٢٨٧٣قال ما ترك النبي صلى الله عليه وسلم إلا بغلته البيضاء وسلاحه وأرضا…٢٨٧٤أنا النبي لا كذب أنا ابن عبد المطلب٢٨٧٥حدثنا محمد بن كثير، أخبرنا سفيان، عن معاوية بن إسحاق، عن عائشة بنت…٢٨٧٦نعم الجهاد الحج٢٨٧٧ناس من أمتي يركبون البحر الأخضر في سبيل الله، مثلهم مثل الملوك على…٢٨٧٩وسعيد بن المسيب، وعلقمة بن وقاص، وعبيد الله بن عبد الله، عن حديث،…٢٨٨٠تنقزان القرب وقال غيره تنقلان القرب على متونهما، ثم تفرغانه في أفواه…٢٨٨١قال ثعلبة بن أبي مالك إن عمر بن الخطاب رضى الله عنه قسم مروطا بين…٢٨٨٢ونداوي الجرحى، ونرد القتلى إلى المدينة .٢٨٨٣ونرد الجرحى والقتلى إلى المدينة .٢٨٨٤اللهم اغفر لعبيد أبي عامر٢٨٨٥ليت رجلا من أصحابي صالحا يحرسني الليلة٢٨٨٦تعس عبد الدينار والدرهم والقطيفة والخميصة، إن أعطي رضي، وإن لم يعط لم…٢٨٨٧تعس عبد الدينار وعبد الدرهم وعبد الخميصة، إن أعطي رضي، وإن لم يعط…٢٨٨٨فكان يخدمني . وهو أكبر من أنس قال جرير إني رأيت الأنصار يصنعون شيئا…٢٨٨٩هذا جبل يحبنا ونحبه٢٨٩٠ذهب المفطرون اليوم بالأجر٢٨٩١كل سلامى عليه صدقة كل يوم، يعين الرجل في دابته يحامله عليها أو يرفع…٢٨٩٢رباط يوم في سبيل الله خير من الدنيا وما عليها، وموضع سوط أحدكم من…٢٨٩٣التمس غلاما من غلمانكم يخدمني حتى أخرج إلى خيبر٢٨٩٤عجبت من قوم من أمتي يركبون البحر، كالملوك على الأسرة٢٨٩٦هل تنصرون وترزقون إلا بضعفائكم٢٨٩٧يأتي زمان يغزو فئام من الناس، فيقال فيكم من صحب النبي صلى الله عليه…٢٨٩٨أما إنه من أهل النار٢٨٩٩ارموا بني إسماعيل، فإن أباكم كان راميا ارموا وأنا مع بني فلان٢٩٠٠إذا أكثبوكم فعليكم بالنبل٢٩٠١حدثنا إبراهيم بن موسى، أخبرنا هشام، عن معمر، عن الزهري، عن ابن…٢٩٠٢وكان أبو طلحة حسن الرمى، فكان إذا رمى تشرف النبي صلى الله عليه وسلم…٢٩٠٣قال لما كسرت بيضة النبي صلى الله عليه وسلم على رأسه وأدمي وجهه، وكسرت…٢٩٠٤فكانت لرسول الله صلى الله عليه وسلم خاصة، وكان ينفق على أهله نفقة…٢٩٠٥ارم فداك أبي وأمي٢٩٠٧فاضطجع على الفراش وحول وجهه، فدخل أبو بكر فانتهرني وقال مزمارة…٢٩٠٨لم تراعوا لم تراعوا٢٩٠٩يقول لقد فتح الفتوح قوم ما كانت حلية سيوفهم الذهب ولا الفضة، إنما…٢٩١٠إن هذا اخترط على سيفي وأنا نائم، فاستيقظت وهو في يده صلتا٢٩١١صلى الله عليه وسلم يوم أحد . فقال جرح وجه النبي صلى الله عليه وسلم…٢٩١٢قال ما ترك النبي صلى الله عليه وسلم إلا سلاحه وبغلة بيضاء وأرضا جعلها…٢٩١٣إن هذا اخترط سيفي٢٩١٤إنما هي طعمة أطعمكموها الله٢٩١٥اللهم إني أنشدك عهدك ووعدك، اللهم إن شئت لم تعبد بعد اليوم٢٩١٦حدثنا محمد بن كثير، أخبرنا سفيان، عن الأعمش، عن إبراهيم، عن الأسود،…٢٩١٧مثل البخيل والمتصدق مثل رجلين عليهما جبتان من حديد، قد اضطرت أيديهما…٢٩١٨قال انطلق رسول الله صلى الله عليه وسلم لحاجته ثم أقبل، فلقيته بماء،…٢٩١٩أن أنسا، حدثهم أن النبي صلى الله عليه وسلم رخص لعبد الرحمن بن عوف…٢٩٢٠الرحمن بن عوف والزبير شكوا إلى النبي صلى الله عليه وسلم يعني القمل…٢٩٢١أن أنسا، حدثهم قال رخص النبي صلى الله عليه وسلم لعبد الرحمن بن عوف…٢٩٢٢رخص أو رخص لحكة بهما .٢٩٢٣قال رأيت النبي صلى الله عليه وسلم يأكل من كتف يحتز منها، ثم دعي إلى…٢٩٢٤أول جيش من أمتي يغزون البحر قد أوجبوا٢٩٢٥تقاتلون اليهود حتى يختبي أحدهم وراء الحجر فيقول يا عبد الله هذا يهودي…٢٩٢٦لا تقوم الساعة حتى تقاتلوا اليهود حتى يقول الحجر وراءه اليهودي يا…٢٩٢٧إن من أشراط الساعة أن تقاتلوا قوما ينتعلون نعال الشعر، وإن من أشراط…٢٩٢٨لا تقوم الساعة حتى تقاتلوا الترك صغار الأعين، حمر الوجوه، ذلف الأنوف،…٢٩٢٩لا تقوم الساعة حتى تقاتلوا قوما نعالهم الشعر، ولا تقوم الساعة حتى…٢٩٣٠وسأله، رجل أكنتم فررتم يا أبا عمارة يوم حنين قال لا، والله ما ولى…٢٩٣١ملأ الله بيوتهم وقبورهم نارا، شغلونا عن الصلاة الوسطى حين غابت الشمس٢٩٣٢اللهم أنج سلمة بن هشام، اللهم أنج الوليد بن الوليد، اللهم أنج عياش بن…٢٩٣٣اللهم منزل الكتاب سريع الحساب، اللهم اهزم الأحزاب، اللهم اهزمهم…٢٩٣٤اللهم عليك بقريش، اللهم عليك بقريش، اللهم عليك بقريش٢٩٣٥دخلوا على النبي صلى الله عليه وسلم فقالوا السام عليك . فلعنتهم . فقال…٢٩٣٦فإن توليت فإن عليك إثم الأريسيين٢٩٣٧اللهم اهد دوسا وائت بهم
