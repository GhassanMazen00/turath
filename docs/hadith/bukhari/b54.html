كتاب الشروط

٢٤ حديثًا
٢٧١٢أنه سمع مروان، والمسور بن مخرمة، رضى الله عنهما يخبران عن أصحاب، رسول…٢٧١٣قال عروة فأخبرتني عائشة، أن رسول الله صلى الله عليه وسلم كان يمتحنهن…٢٧١٤حدثنا أبو نعيم، حدثنا سفيان، عن زياد بن علاقة، قال سمعت جريرا رضى…٢٧١٥حدثنا مسدد، حدثنا يحيى، عن إسماعيل، قال حدثني قيس بن أبي حازم، عن…٢٧١٦من باع نخلا قد أبرت فثمرتها للبائع إلا أن يشترط المبتاع٢٧١٧ابتاعي فأعتقي، فإنما الولاء لمن أعتق٢٧١٨فمر النبي صلى الله عليه وسلم فضربه، فدعا له، فسار بسير ليس يسير مثله…٢٧١٩حدثنا أبو اليمان، أخبرنا شعيب، حدثنا أبو الزناد، عن الأعرج، عن أبي…٢٧٢٠ولهم شطر ما يخرج منها .٢٧٢١أحق الشروط أن توفوا به ما استحللتم به الفروج٢٧٢٢فربما أخرجت هذه ولم تخرج ذه، فنهينا عن ذلك، ولم ننه عن الورق .٢٧٢٣لا يبيع حاضر لباد، ولا تناجشوا، ولا يزيدن على بيع أخيه، ولا يخطبن على…٢٧٢٤وزيد بن خالد الجهني، رضى الله عنهم أنهما قالا إن رجلا من الأعراب أتى…٢٧٢٦ما شأن بريرة فقال اشتريها فأعتقيها وليشترطوا ما شاءوا٢٧٢٧وأن يبتاع المهاجر للأعرابي، وأن تشترط المرأة طلاق أختها، وأن يستام…٢٧٢٨أن ابن جريج، أخبره قال أخبرني يعلى بن مسلم، وعمرو بن دينار، عن سعيد…٢٧٢٩خذيها واشترطي لهم الولاء، فإنما الولاء لمن أعتق٢٧٣٠نقركم ما أقركم الله٢٧٣٢إن خالد بن الوليد بالغميم في خيل لقريش طليعة فخذوا ذات اليمين٢٧٣٣قال عروة فأخبرتني عائشة، أن رسول الله صلى الله عليه وسلم كان يمتحنهن،…٢٧٣٤فدفعها إليه إلى أجل مسمى . وقال ابن عمر رضى الله عنهما وعطاء إذا أجله…٢٧٣٥ابتاعيها فأعتقيها، فإنما الولاء لمن أعتق٢٧٣٦إن لله تسعة وتسعين اسما مائة إلا واحدا من أحصاها دخل الجنة٢٧٣٧إن شئت حبست أصلها، وتصدقت بها
