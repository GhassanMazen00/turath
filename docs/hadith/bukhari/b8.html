١٦٧ حديثًا•صفحة ١ من ٢
٣٤٩فرج عن سقف بيتي وأنا بمكة، فنزل جبريل ففرج صدري، ثم غسله بماء زمزم،…٣٥٠فرض الله الصلاة حين فرضها ركعتين ركعتين في الحضر والسفر، فأقرت صلاة…٣٥١لتلبسها صاحبتها من جلبابها٣٥٢قال صلى جابر في إزار قد عقده من قبل قفاه، وثيابه موضوعة على المشجب…٣٥٣قال رأيت جابر بن عبد الله يصلي في ثوب واحد وقال رأيت النبي صلى الله…٣٥٤أن النبي صلى الله عليه وسلم صلى في ثوب واحد قد خالف بين طرفيه .٣٥٥أنه رأى النبي صلى الله عليه وسلم يصلي في ثوب واحد في بيت أم سلمة، قد…٣٥٦أن عمر بن أبي سلمة، أخبره قال رأيت رسول الله صلى الله عليه وسلم يصلي…٣٥٧مولى عمر بن عبيد الله أن أبا مرة، مولى أم هانئ بنت أبي طالب أخبره…٣٥٨أولكلكم ثوبان٣٥٩لا يصلي أحدكم في الثوب الواحد، ليس على عاتقيه شىء٣٦٠من صلى في ثوب واحد، فليخالف بين طرفيه٣٦١في الثوب الواحد فقال خرجت مع النبي صلى الله عليه وسلم في بعض أسفاره،…٣٦٢قال كان رجال يصلون مع النبي صلى الله عليه وسلم عاقدي أزرهم على…٣٦٣يا مغيرة، خذ الإداوة٣٦٤يحدث أن رسول الله صلى الله عليه وسلم كان ينقل معهم الحجارة للكعبة…٣٦٥أوكلكم يجد ثوبين٣٦٦لا يلبس القميص ولا السراويل ولا البرنس ولا ثوبا مسه الزعفران ولا ورس،…٣٦٧ليس على فرجه منه شىء .٣٦٨وأن يشتمل الصماء، وأن يحتبي الرجل في ثوب واحد .٣٦٩أن أبا هريرة، قال بعثني أبو بكر في تلك الحجة في مؤذنين يوم النحر نؤذن…٣٧٠قال دخلت على جابر بن عبد الله وهو يصلي في ثوب ملتحفا به ورداؤه موضوع،…٣٧١الله أكبر، خربت خيبر، إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٣٧٢أن عائشة، قالت لقد كان رسول الله صلى الله عليه وسلم يصلي الفجر، فيشهد…٣٧٣اذهبوا بخميصتي هذه إلى أبي جهم وائتوني بأنبجانية أبي جهم، فإنها…٣٧٤أميطي عنا قرامك هذا، فإنه لا تزال تصاويره تعرض في صلاتي٣٧٥لا ينبغي هذا للمتقين٣٧٦قال رأيت رسول الله صلى الله عليه وسلم في قبة حمراء من أدم، ورأيت…٣٧٧قال سألوا سهل بن سعد من أى شىء المنبر فقال ما بقي بالناس أعلم مني هو…٣٧٨إنما جعل الإمام ليؤتم به، فإذا كبر فكبروا، وإذا ركع فاركعوا، وإذا سجد…٣٧٩حدثنا مسدد، عن خالد، قال حدثنا سليمان الشيباني، عن عبد الله بن شداد،…٣٨٠قوموا فلأصل لكم٣٨١قالت كان النبي صلى الله عليه وسلم يصلي على الخمرة .٣٨٢زوج النبي صلى الله عليه وسلم أنها قالت كنت أنام بين يدى رسول الله صلى…٣٨٣أن عائشة، أخبرته أن رسول الله صلى الله عليه وسلم كان يصلي وهى بينه…٣٨٤أن النبي صلى الله عليه وسلم كان يصلي وعائشة معترضة بينه وبين القبلة…٣٨٥حدثنا أبو الوليد، هشام بن عبد الملك قال حدثنا بشر بن المفضل، قال…٣٨٦سعيد بن يزيد الأزدي قال سألت أنس بن مالك أكان النبي صلى الله عليه…٣٨٧قال رأيت جرير بن عبد الله بال ثم توضأ، ومسح على خفيه، ثم قام فصلى،…٣٨٨قال وضأت النبي صلى الله عليه وسلم فمسح على خفيه وصلى .٣٨٩رأى رجلا لا يتم ركوعه ولا سجوده، فلما قضى صلاته قال له حذيفة ما صليت…٣٩٠أخبرنا يحيى بن بكير، حدثنا بكر بن مضر، عن جعفر، عن ابن هرمز، عن عبد…٣٩١من صلى صلاتنا، واستقبل قبلتنا، وأكل ذبيحتنا، فذلك المسلم الذي له ذمة…٣٩٢أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله . فإذا قالوها وصلوا…٣٩٣ما يحرم دم العبد وماله فقال من شهد أن لا إله إلا الله، واستقبل…٣٩٤إذا أتيتم الغائط فلا تستقبلوا القبلة ولا تستدبروها، ولكن شرقوا أو…٣٩٦طاف بالبيت العمرة، ولم يطف بين الصفا والمروة، أيأتي امرأته فقال قدم…٣٩٧قال أتي ابن عمر فقيل له هذا رسول الله صلى الله عليه وسلم دخل الكعبة .…٣٩٨قال لما دخل النبي صلى الله عليه وسلم البيت دعا في نواحيه كلها، ولم…٣٩٩وكان رسول الله صلى الله عليه وسلم يحب أن يوجه إلى الكعبة، فأنزل الله…٤٠٠قال كان رسول الله صلى الله عليه وسلم يصلي على راحلته حيث توجهت، فإذا…٤٠١قال قال عبد الله صلى النبي صلى الله عليه وسلم قال إبراهيم لا أدري زاد…٤٠٢قال قال عمر وافقت ربي في ثلاث، فقلت يا رسول الله لو اتخذنا من مقام…٤٠٢.٢بهذا .٤٠٣وقد أمر أن يستقبل الكعبة فاستقبلوها، وكانت وجوههم إلى الشأم،…٤٠٤قال صلى النبي صلى الله عليه وسلم الظهر خمسا فقالوا أزيد في الصلاة قال…٤٠٥إن أحدكم إذا قام في صلاته، فإنه يناجي ربه أو إن ربه بينه وبين القبلة…٤٠٦إذا كان أحدكم يصلي، فلا يبصق قبل وجهه، فإن الله قبل وجهه إذا صلى٤٠٧أن رسول الله صلى الله عليه وسلم رأى في جدار القبلة مخاطا أو بصاقا أو…٤٠٩إذا تنخم أحدكم فلا يتنخمن قبل وجهه ولا عن يمينه، وليبصق عن يساره أو…٤١١إذا تنخم أحدكم فلا يتنخم قبل وجهه ولا عن يمينه، وليبصق عن يساره، أو…٤١٢لا يتفلن أحدكم بين يديه ولا عن يمينه، ولكن عن يساره أو تحت رجله٤١٣إن المؤمن إذا كان في الصلاة فإنما يناجي ربه، فلا يبزقن بين يديه ولا…٤١٤. أن النبي صلى الله عليه وسلم أبصر نخامة في قبلة المسجد فحكها بحصاة،…٤١٥البزاق في المسجد خطيئة، وكفارتها دفنها٤١٦إذا قام أحدكم إلى الصلاة فلا يبصق أمامه، فإنما يناجي الله ما دام في…٤١٧إن أحدكم إذا قام في صلاته فإنما يناجي ربه أو ربه بينه وبين قبلته فلا…٤١٨هل ترون قبلتي ها هنا فوالله ما يخفى على خشوعكم ولا ركوعكم، إني لأراكم…٤١٩إني لأراكم من ورائي كما أراكم٤٢٠أن رسول الله صلى الله عليه وسلم سابق بين الخيل التي أضمرت من الحفياء،…٤٢١انثروه في المسجد٤٢٢سمع أنسا، قال وجدت النبي صلى الله عليه وسلم في المسجد معه ناس فقمت،…٤٢٣أن رجلا، قال يا رسول الله، أرأيت رجلا وجد مع امرأته رجلا أيقتله…٤٢٤أين تحب أن أصلي لك من بيتك٤٢٥سأفعل إن شاء الله٤٢٦قالت كان النبي صلى الله عليه وسلم يحب التيمن ما استطاع في شأنه كله في…٤٢٧إن أولئك إذا كان فيهم الرجل الصالح فمات بنوا على قبره مسجدا، وصوروا…٤٢٨يا بني النجار ثامنوني بحائطكم هذا٤٢٩قال كان النبي صلى الله عليه وسلم يصلي في مرابض الغنم، ثم سمعته بعد…٤٣٠قال رأيت ابن عمر يصلي إلى بعيره وقال رأيت النبي صلى الله عليه وسلم…٤٣١أريت النار، فلم أر منظرا كاليوم قط أفظع٤٣٢اجعلوا في بيوتكم من صلاتكم، ولا تتخذوها قبورا٤٣٣لا تدخلوا على هؤلاء المعذبين إلا أن تكونوا باكين، فإن لم تكونوا باكين…٤٣٤أولئك قوم إذا مات فيهم العبد الصالح أو الرجل الصالح بنوا على قبره…٤٣٦لعنة الله على اليهود والنصارى اتخذوا قبور أنبيائهم مساجد٤٣٧قاتل الله اليهود اتخذوا قبور أنبيائهم مساجد٤٣٨أعطيت خمسا لم يعطهن أحد من الأنبياء قبلي، نصرت بالرعب مسيرة شهر،…٤٣٩أن وليدة، كانت سوداء لحى من العرب، فأعتقوها، فكانت معهم قالت فخرجت…٤٤٠أنه كان ينام وهو شاب أعزب لا أهل له في مسجد النبي صلى الله عليه وسلم .٤٤١قال جاء رسول الله صلى الله عليه وسلم بيت فاطمة، فلم يجد عليا في البيت…٤٤٢قال رأيت سبعين من أصحاب الصفة، ما منهم رجل عليه رداء، إما إزار وإما…٤٤٣قال أتيت النبي صلى الله عليه وسلم وهو في المسجد قال مسعر أراه قال ضحى…٤٤٤إذا دخل أحدكم المسجد فليركع ركعتين قبل أن يجلس٤٤٥الملائكة تصلي على أحدكم ما دام في مصلاه الذي صلى فيه، ما لم يحدث،…٤٤٦أن عبد الله، أخبره أن المسجد كان على عهد رسول الله صلى الله عليه وسلم…٤٤٧ويح عمار تقتله الفئة الباغية، يدعوهم إلى الجنة، ويدعونه إلى النار٤٤٨قال بعث رسول الله صلى الله عليه وسلم إلى امرأة أن مري غلامك النجار…٤٤٩أن امرأة، قالت يا رسول الله، ألا أجعل لك شيئا تقعد عليه، فإن لي غلاما…٤٥٠من بنى مسجدا قال بكير حسبت أنه قال يبتغي به وجه الله، بنى الله له…٤٥١فقال له رسول الله صلى الله عليه وسلم " أمسك بنصالها " .٤٥٢من مر في شىء من مساجدنا أو أسواقنا بنبل، فليأخذ على نصالها، لا يعقر…٤٥٣يا حسان، أجب عن رسول الله صلى الله عليه وسلم، اللهم أيده بروح القدس٤٥٥أن عائشة، قالت لقد رأيت رسول الله صلى الله عليه وسلم يوما على باب…٤٥٦ابتاعيها فأعتقيها، فإن الولاء لمن أعتق٤٥٧فارتفعت أصواتهما حتى سمعها رسول الله صلى الله عليه وسلم وهو في بيته،…٤٥٨أفلا كنتم آذنتموني به دلوني على قبره٤٥٩قالت لما أنزل الآيات من سورة البقرة في الربا، خرج النبي صلى الله عليه…٤٦٠أن امرأة أو رجلا كانت تقم المسجد ولا أراه إلا امرأة فذكر حديث النبي…٤٦١إن عفريتا من الجن تفلت على البارحة أو كلمة نحوها ليقطع على الصلاة،…٤٦٢سمع أبا هريرة، قال بعث النبي صلى الله عليه وسلم خيلا قبل نجد، فجاءت…٤٦٣قالت أصيب سعد يوم الخندق في الأكحل، فضرب النبي صلى الله عليه وسلم…٤٦٤طوفي من وراء الناس وأنت راكبة٤٦٥أن رجلين، من أصحاب النبي صلى الله عليه وسلم خرجا من عند النبي صلى…٤٦٦إن الله خير عبدا بين الدنيا وبين ما عنده، فاختار ما عند الله٤٦٧إنه ليس من الناس أحد أمن على في نفسه وماله من أبي بكر بن أبي قحافة،…٤٦٨وقتيبة، قالا حدثنا حماد، عن أيوب، عن نافع، عن ابن عمر، أن النبي صلى…٤٦٩أنه سمع أبا هريرة، يقول بعث رسول الله صلى الله عليه وسلم خيلا قبل…٤٧٠قال كنت قائما في المسجد فحصبني رجل، فنظرت فإذا عمر بن الخطاب فقال…٤٧١يا كعب بن مالك، يا كعب٤٧٢مثنى مثنى، فإذا خشي الصبح صلى واحدة، فأوترت له ما صلى٤٧٣مثنى مثنى، فإذا خشيت الصبح فأوتر بواحدة، توتر لك ما قد صليت٤٧٤ألا أخبركم عن الثلاثة أما أحدهم فأوى إلى الله، فآواه الله، وأما الآخر…٤٧٥أنه رأى رسول الله صلى الله عليه وسلم مستلقيا في المسجد، واضعا إحدى…٤٧٦أن عائشة، زوج النبي صلى الله عليه وسلم قالت لم أعقل أبوى إلا وهما…٤٧٧صلاة الجميع تزيد على صلاته في بيته، وصلاته في سوقه خمسا وعشرين درجة،…٤٧٩أو ابن عمرو شبك النبي صلى الله عليه وسلم أصابعه .٤٨٠يا عبد الله بن عمرو، كيف بك إذا بقيت في حثالة من الناس بهذا٤٨١إن المؤمن للمؤمن كالبنيان، يشد بعضه بعضا٤٨٢لم أنس، ولم تقصر٤٨٣قال رأيت سالم بن عبد الله يتحرى أماكن من الطريق فيصلي فيها، ويحدث أن…٤٨٤أن عبد الله، أخبره أن رسول الله صلى الله عليه وسلم كان ينزل بذي…٤٨٥وأن عبد الله بن عمر حدثه أن النبي صلى الله عليه وسلم صلى حيث المسجد…٤٨٦وأن ابن عمر كان يصلي إلى العرق الذي عند منصرف الروحاء، وذلك العرق…٤٨٧ووجاه الطريق في مكان بطح سهل، حتى يفضي من أكمة دوين بريد الرويثة…٤٨٨وأن عبد الله بن عمر حدثه أن النبي صلى الله عليه وسلم صلى في طرف تلعة…٤٨٩في مسيل دون هرشى، ذلك المسيل لاصق بكراع هرشى، بينه وبين الطريق قريب…٤٩٠وأن عبد الله بن عمر حدثه أن النبي صلى الله عليه وسلم كان ينزل في…٤٩١وأن عبد الله بن عمر حدثه أن النبي صلى الله عليه وسلم كان ينزل بذي طوى…٤٩٢وأن عبد الله حدثه أن النبي صلى الله عليه وسلم استقبل فرضتى الجبل الذي…٤٩٣أنه قال أقبلت راكبا على حمار أتان، وأنا يومئذ قد ناهزت الاحتلام،…٤٩٤أن رسول الله صلى الله عليه وسلم كان إذا خرج يوم العيد أمر بالحربة…٤٩٥والعصر ركعتين، تمر بين يديه المرأة والحمار .٤٩٦قال كان بين مصلى رسول الله صلى الله عليه وسلم وبين الجدار ممر الشاة .٤٩٧قال كان جدار المسجد عند المنبر ما كادت الشاة تجوزها .٤٩٨أن النبي صلى الله عليه وسلم كان يركز له الحربة فيصلي إليها .٤٩٩والمرأة والحمار يمرون من ورائها .٥٠٠فإذا فرغ من حاجته ناولناه الإداوة .٥٠١قال خرج رسول الله صلى الله عليه وسلم بالهاجرة فصلى بالبطحاء الظهر…٥٠٢قال كنت آتي مع سلمة بن الأكوع فيصلي عند الأسطوانة التي عند المصحف .…٥٠٣حدثنا قبيصة، قال حدثنا سفيان، عن عمرو بن عامر، عن أنس، قال لقد رأيت…
