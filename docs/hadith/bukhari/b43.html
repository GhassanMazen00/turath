٢٤ حديثًا
٢٣٨٥كيف ترى بعيرك أتبيعنيه٢٣٨٦ورهنه درعا من حديد .٢٣٨٧من أخذ أموال الناس يريد أداءها أدى الله عنه، ومن أخذ يريد إتلافها…٢٣٨٨ما أحب أنه يحول لي ذهبا يمكث عندي منه دينار فوق ثلاث، إلا دينارا…٢٣٨٩لو كان لي مثل أحد ذهبا، ما يسرني أن لا يمر على ثلاث وعندي منه شىء،…٢٣٩٠دعوه، فإن لصاحب الحق مقالا . واشتروا له بعيرا، فأعطوه إياه٢٣٩١مات رجل، فقيل له قال كنت أبايع الناس، فأتجوز عن الموسر، وأخفف عن…٢٣٩٢أتى النبي صلى الله عليه وسلم يتقاضاه بعيرا، فقال رسول الله صلى الله…٢٣٩٣وفى الله بك . قال النبي صلى الله عليه وسلم " إن خياركم أحسنكم قضاء " .٢٣٩٤حدثنا خلاد، حدثنا مسعر، حدثنا محارب بن دثار، عن جابر بن عبد الله رضى…٢٣٩٥أن جابر بن عبد الله رضى الله عنهما أخبره أن أباه قتل يوم أحد شهيدا،…٢٣٩٦جد له فأوف له الذي له٢٣٩٧اللهم إني أعوذ بك من المأثم والمغرم٢٣٩٨من ترك مالا فلورثته، ومن ترك كلا فإلينا٢٣٩٩ما من مؤمن إلا وأنا أولى به في الدنيا والآخرة اقرءوا إن شئتم { النبي…٢٤٠٠أخي وهب بن منبه أنه سمع أبا هريرة رضى الله عنه يقول قال رسول الله صلى…٢٤٠١دعوه فإن لصاحب الحق مقالا٢٤٠٢من أدرك ماله بعينه عند رجل أو إنسان قد أفلس، فهو أحق به من غيره٢٤٠٣فأخذ ثمنه، فدفعه إليه .٢٤٠٤سأل بعض بني إسرائيل أن يسلفه، فدفعها إليه إلى أجل مسمى . فذكر الحديث .٢٤٠٦صنف تمرك كل شىء منه على حدته، عذق ابن زيد على حدة، واللين على حدة،…٢٤٠٧إذا بايعت فقل لا خلابة٢٤٠٨إن الله حرم عليكم عقوق الأمهات، ووأد البنات، ومنع وهات، وكره لكم قيل…٢٤٠٩كلكم راع ومسئول عن رعيته، فالإمام راع، وهو مسئول عن رعيته، والرجل في…
