١١٢ حديثًا
١٨٩١الصلوات الخمس، إلا أن تطوع شيئا١٨٩٢وأمر بصيامه . فلما فرض رمضان ترك . وكان عبد الله لا يصومه، إلا أن…١٨٩٣من شاء فليصمه، ومن شاء أفطر١٨٩٤الصيام جنة، فلا يرفث ولا يجهل، وإن امرؤ قاتله أو شاتمه فليقل إني صائم…١٨٩٥فتنة الرجل في أهله وماله وجاره تكفرها الصلاة والصيام والصدقة١٨٩٦إن في الجنة بابا يقال له الريان، يدخل منه الصائمون يوم القيامة، لا…١٨٩٧من أنفق زوجين في سبيل الله نودي من أبواب الجنة يا عبد الله، هذا خير .…١٨٩٨إذا جاء رمضان فتحت أبواب الجنة١٨٩٩إذا دخل شهر رمضان فتحت أبواب السماء، وغلقت أبواب جهنم، وسلسلت الشياطين١٩٠٠إذا رأيتموه فصوموا، وإذا رأيتموه فأفطروا، فإن غم عليكم فاقدروا له١٩٠١من قام ليلة القدر إيمانا واحتسابا غفر له ما تقدم من ذنبه، ومن صام…١٩٠٢أن ابن عباس رضى الله عنهما قال كان النبي صلى الله عليه وسلم أجود…١٩٠٣من لم يدع قول الزور والعمل به فليس لله حاجة في أن يدع طعامه وشرابه١٩٠٤قال الله كل عمل ابن آدم له إلا الصيام، فإنه لي، وأنا أجزي به .…١٩٠٥من استطاع الباءة فليتزوج، فإنه أغض للبصر وأحصن للفرج، ومن لم يستطع…١٩٠٦لا تصوموا حتى تروا الهلال، ولا تفطروا حتى تروه، فإن غم عليكم فاقدروا…١٩٠٧الشهر تسع وعشرون ليلة، فلا تصوموا حتى تروه، فإن غم عليكم فأكملوا…١٩٠٨الشهر هكذا وهكذا١٩٠٩صوموا لرؤيته، وأفطروا لرؤيته، فإن غبي عليكم فأكملوا عدة شعبان ثلاثين١٩١٠إن الشهر يكون تسعة وعشرين يوما١٩١١إن الشهر يكون تسعا وعشرين١٩١٢شهران لا ينقصان شهرا عيد رمضان وذو الحجة١٩١٣إنا أمة أمية، لا نكتب ولا نحسب الشهر هكذا وهكذا١٩١٤لا يتقدمن أحدكم رمضان بصوم يوم أو يومين، إلا أن يكون رجل كان يصوم…١٩١٥فحضر الإفطار، فنام قبل أن يفطر لم يأكل ليلته ولا يومه، حتى يمسي، وإن…١٩١٦إنما ذلك سواد الليل وبياض النهار١٩١٧قال أنزلت { وكلوا واشربوا حتى يتبين لكم الخيط الأبيض من الخيط الأسود…١٩١٩كلوا واشربوا حتى يؤذن ابن أم مكتوم، فإنه لا يؤذن حتى يطلع الفجر١٩٢٠ثم تكون سرعتي أن أدرك السجود مع رسول الله صلى الله عليه وسلم .١٩٢١حدثنا مسلم بن إبراهيم، حدثنا هشام، حدثنا قتادة، عن أنس، عن زيد بن…١٩٢٢لست كهيئتكم، إني أظل أطعم وأسقى١٩٢٣تسحروا فإن في السحور بركة١٩٢٤أن من أكل فليتم أو فليصم، ومن لم يأكل فلا يأكل١٩٢٦مولى أبي بكر بن عبد الرحمن بن الحارث بن هشام بن المغيرة أنه سمع أبا…١٩٢٧وهو صائم، وكان أملككم لإربه . وقال قال ابن عباس { مآرب } حاجة . قال…١٩٢٨حدثنا محمد بن المثنى، حدثنا يحيى، عن هشام، قال أخبرني أبي، عن عائشة،…١٩٢٩رسول الله صلى الله عليه وسلم في الخميلة إذ حضت فانسللت، فأخذت ثياب…١٩٣٠وأبي، بكر قالت عائشة رضى الله عنها كان النبي صلى الله عليه وسلم يدركه…١٩٣٢مولى أبي بكر بن عبد الرحمن بن الحارث بن هشام بن المغيرة أنه سمع أبا…١٩٣٣إذا نسي فأكل وشرب فليتم صومه، فإنما أطعمه الله وسقاه١٩٣٤من توضأ وضوئي هذا، ثم يصلي ركعتين، لا يحدث نفسه فيهما بشىء، إلا غفر…١٩٣٥سمع يزيد بن هارون، حدثنا يحيى هو ابن سعيد أن عبد الرحمن بن القاسم،…١٩٣٦أن أبا هريرة رضى الله عنه قال بينما نحن جلوس عند النبي صلى الله عليه…١٩٣٧أتجد ما تحرر رقبة١٩٣٨وهو محرم واحتجم وهو صائم .١٩٣٩حدثنا أبو معمر، حدثنا عبد الوارث، حدثنا أيوب، عن عكرمة، عن ابن عباس…١٩٤٠حدثنا آدم بن أبي إياس، حدثنا شعبة، قال سمعت ثابتا البناني، يسأل أنس…١٩٤١فجدح له، فشرب، ثم رمى بيده ها هنا، ثم قال " إذا رأيتم الليل أقبل من…١٩٤٢أن حمزة بن عمرو الأسلمي، قال يا رسول الله إني أسرد الصوم .١٩٤٣إن شئت فصم، وإن شئت فأفطر١٩٤٤فأفطر الناس . قال أبو عبد الله والكديد ماء بين عسفان وقديد .١٩٤٥أن إسماعيل بن عبيد الله، حدثه عن أم الدرداء، عن أبي الدرداء رضى الله…١٩٤٦فرأى زحاما، ورجلا قد ظلل عليه، فقال " ما هذا " . فقالوا صائم . فقال "…١٩٤٧ولا المفطر على الصائم .١٩٤٨فصام حتى بلغ عسفان، ثم دعا بماء فرفعه إلى يديه ليريه الناس فأفطر، حتى…١٩٤٩حدثنا عياش، حدثنا عبد الأعلى، حدثنا عبيد الله، عن نافع، عن ابن عمر…١٩٥٠فما أستطيع أن أقضي إلا في شعبان . قال يحيى الشغل من النبي أو بالنبي…١٩٥١أليس إذا حاضت لم تصل، ولم تصم فذلك نقصان دينها١٩٥٢من مات وعليه صيام صام عنه وليه١٩٥٣نعم قال فدين الله أحق أن يقضى١٩٥٤إذا أقبل الليل من ها هنا، وأدبر النهار من ها هنا، وغربت الشمس، فقد…١٩٥٥يا فلان قم، فاجدح لنا١٩٥٦فلما غربت الشمس قال " انزل، فاجدح لنا " . قال يا رسول الله، لو أمسيت…١٩٥٧لا يزال الناس بخير ما عجلوا الفطر١٩٥٨فصام حتى أمسى، قال لرجل " انزل، فاجدح لي " . قال لو انتظرت حتى تمسي .…١٩٥٩حدثني عبد الله بن أبي شيبة، حدثنا أبو أسامة، عن هشام بن عروة، عن…١٩٦٠من أصبح مفطرا فليتم بقية يومه، ومن أصبح صائما فليصم١٩٦١إني أطعم وأسقى، أو إني أبيت أطعم وأسقى " .١٩٦٢إني لست مثلكم، إني أطعم وأسقى١٩٦٣لا تواصلوا، فأيكم إذا أراد أن يواصل فليواصل حتى السحر١٩٦٤إني لست كهيئتكم، إني يطعمني ربي ويسقين١٩٦٥وأيكم مثلي إني أبيت يطعمني ربي ويسقين١٩٦٦فاكلفوا من العمل ما تطيقون " .١٩٦٧لا تواصلوا، فأيكم أراد أن يواصل فليواصل حتى السحر١٩٦٨قال آخى النبي صلى الله عليه وسلم بين سلمان، وأبي الدرداء، فزار سلمان…١٩٦٩ويفطر حتى نقول لا يصوم . فما رأيت رسول الله صلى الله عليه وسلم استكمل…١٩٧٠خذوا من العمل ما تطيقون، فإن الله لا يمل حتى تملوا، وأحب الصلاة إلى…١٩٧١ويصوم حتى يقول القائل لا والله لا يفطر، ويفطر حتى يقول القائل لا…١٩٧٢أنه سمع أنسا رضى الله عنه يقول كان رسول الله صلى الله عليه وسلم يفطر…١٩٧٣ولا من الليل قائما إلا رأيته، ولا نائما إلا رأيته، ولا مسست خزة ولا…١٩٧٤إن لزورك عليك حقا، وإن لزوجك عليك حقا١٩٧٥يا عبد الله ألم أخبر أنك تصوم النهار وتقوم الليل١٩٧٦فإنك لا تستطيع ذلك، فصم وأفطر، وقم ونم، وصم من الشهر ثلاثة أيام، فإن…١٩٧٧ألم أخبر أنك تصوم ولا تفطر، وتصلي ولا تنام، فصم وأفطر، وقم ونم، فإن…١٩٧٨صم من الشهر ثلاثة أيام١٩٧٩إنك لتصوم الدهر، وتقوم الليل١٩٨٠أما يكفيك من كل شهر ثلاثة أيام١٩٨١وركعتى الضحى، وأن أوتر قبل أن أنام .١٩٨٢أعيدوا سمنكم في سقائه، وتمركم في وعائه، فإني صائم١٩٨٣يا أبا فلان أما صمت سرر هذا
