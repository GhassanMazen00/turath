١٨٨ حديثًا•صفحة ١ من ٢
٧٣٧١رضى الله عنهما أن النبي صلى الله عليه وسلم بعث معاذا إلى اليمن .٧٣٧٢إنك تقدم على قوم من أهل الكتاب فليكن أول ما تدعوهم إلى أن يوحدوا الله…٧٣٧٣يا معاذ أتدري ما حق الله على العباد٧٣٧٤والذي نفسي بيده إنها لتعدل ثلث القرآن٧٣٧٥سلوه لأى شىء يصنع ذلك٧٣٧٦لا يرحم الله من لا يرحم الناس٧٣٧٧ارجع فأخبرها أن لله ما أخذ، وله ما أعطى، وكل شىء عنده بأجل مسمى،…٧٣٧٨ما أحد أصبر على أذى سمعه من الله، يدعون له الولد، ثم يعافيهم ويرزقهم٧٣٧٩مفاتيح الغيب خمس لا يعلمها إلا الله، لا يعلم ما تغيض الأرحام إلا…٧٣٨٠وهو يقول لا يعلم الغيب إلا الله .٧٣٨١إن الله هو السلام ولكن قولوا التحيات لله والصلوات والطيبات، السلام…٧٣٨٢يقبض الله الأرض يوم القيامة، ويطوي السماء بيمينه ثم يقول أنا الملك…٧٣٨٣أعوذ بعزتك الذي لا إله إلا أنت، الذي لا يموت والجن والإنس يموتون٧٣٨٤ثم تقول قد قد بعزتك وكرمك . ولا تزال الجنة تفضل حتى ينشئ الله لها…٧٣٨٥اللهم لك الحمد أنت رب السموات والأرض، لك الحمد أنت قيم السموات والأرض…٧٣٨٦اربعوا على أنفسكم، فإنكم لا تدعون أصم ولا غائبا، تدعون سميعا بصيرا…٧٣٨٨قل اللهم إني ظلمت نفسي ظلما كثيرا، ولا يغفر الذنوب إلا أنت، فاغفر لي…٧٣٨٩إن جبريل عليه السلام ناداني قال إن الله قد سمع قول قومك وما ردوا عليك٧٣٩٠إذا هم أحدكم بالأمر فليركع ركعتين من غير الفريضة ثم ليقل اللهم إني…٧٣٩١لا ومقلب القلوب٧٣٩٢إن لله تسعة وتسعين اسما مائة إلا واحدا، من أحصاها دخل الجنة٧٣٩٣إذا جاء أحدكم فراشه فلينفضه بصنفة ثوبه ثلاث مرات، وليقل باسمك رب وضعت…٧٣٩٤اللهم باسمك أحيا وأموت٧٣٩٥باسمك نموت ونحيا، فإذا استيقظ قال الحمد لله الذي أحيانا بعد ما أماتنا…٧٣٩٦لو أن أحدكم إذا أراد أن يأتي أهله فقال باسم الله، اللهم جنبنا…٧٣٩٧إذا أرسلت كلابك المعلمة وذكرت اسم الله فأمسكن فكل، وإذا رميت بالمعراض…٧٣٩٨اذكروا أنتم اسم الله وكلوا٧٣٩٩قال ضحى النبي صلى الله عليه وسلم بكبشين، يسمي ويكبر .٧٤٠٠من ذبح قبل أن يصلي فليذبح مكانها أخرى، ومن لم يذبح فليذبح باسم الله٧٤٠١لا تحلفوا بآبائكم، ومن كان حالفا فليحلف بالله٧٤٠٢فلما خرجوا من الحرم ليقتلوه قال خبيب الأنصاري ولست أبالي حين أقتل…٧٤٠٣ما من أحد أغير من الله، من أجل ذلك حرم الفواحش، وما أحد أحب إليه…٧٤٠٤لما خلق الله الخلق كتب في كتابه هو يكتب على نفسه، وهو وضع عنده على…٧٤٠٥يقول الله تعالى أنا عند ظن عبدي بي، وأنا معه إذا ذكرني، فإن ذكرني في…٧٤٠٦قال لما نزلت هذه الآية { قل هو القادر على أن يبعث عليكم عذابا من…٧٤٠٧إن الله لا يخفى عليكم، إن الله ليس بأعور وأشار بيده إلى عينه وإن…٧٤٠٨ما بعث الله من نبي إلا أنذر قومه الأعور الكذاب، إنه أعور، وإن ربكم…٧٤٠٩ما عليكم أن لا تفعلوا، فإن الله قد كتب من هو خالق إلى يوم القيامة٧٤١٠يجمع الله المؤمنين يوم القيامة كذلك فيقولون لو استشفعنا إلى ربنا حتى…٧٤١١يد الله ملأى لا يغيضها نفقة، سحاء الليل والنهار وقال أرأيتم ما أنفق…٧٤١٣إن الله يقبض يوم القيامة الأرض وتكون السموات بيمينه ثم يقول أنا الملك٧٤١٤سمع يحيى بن سعيد، عن سفيان، حدثني منصور، وسليمان، عن إبراهيم، عن…٧٤١٥يقول قال عبد الله جاء رجل إلى النبي صلى الله عليه وسلم من أهل الكتاب…٧٤١٦تعجبون من غيرة سعد، والله لأنا أغير منه، والله أغير مني، ومن أجل غيرة…٧٤١٧أمعك من القرآن شىء٧٤١٨اقبلوا البشرى يا بني تميم٧٤١٩إن يمين الله ملأى لا يغيضها نفقة سحاء الليل والنهار، أرأيتم ما أنفق…٧٤٢٠اتق الله، وأمسك عليك زوجك٧٤٢١حدثنا خلاد بن يحيى، حدثنا عيسى بن طهمان، قال سمعت أنس بن مالك رضى…٧٤٢٢إن الله لما قضى الخلق كتب عنده فوق عرشه إن رحمتي سبقت غضبي٧٤٢٣من آمن بالله ورسوله، وأقام الصلاة، وصام رمضان، كان حقا على الله أن…٧٤٢٤يا أبا ذر هل تدري أين تذهب هذه٧٤٢٥أن زيد بن ثابت، . وقال الليث حدثني عبد الرحمن بن خالد، عن ابن شهاب،…٧٤٢٦لا إله إلا الله العليم الحليم لا إله إلا الله رب العرش العظيم لا إله…٧٤٢٨الناس يصعقون يوم القيامة فإذا أنا بموسى آخذ بقائمة من قوائم العرش٧٤٢٩يتعاقبون فيكم ملائكة بالليل وملائكة بالنهار، ويجتمعون في صلاة العصر…٧٤٣٠من تصدق بعدل تمرة من كسب طيب، ولا يصعد إلى الله إلا الطيب، فإن الله…٧٤٣١لا إله إلا الله العظيم الحليم، لا إله إلا الله رب العرش العظيم، لا…٧٤٣٢فقسمها بين الأقرع بن حابس الحنظلي ثم أحد بني مجاشع، وبين عيينة بن بدر…٧٤٣٣مستقرها تحت العرش٧٤٣٤إنكم سترون ربكم كما ترون هذا القمر لا تضامون في رؤيته، فإن استطعتم أن…٧٤٣٥إنكم سترون ربكم عيانا٧٤٣٦إنكم سترون ربكم يوم القيامة كما ترون هذا، لا تضامون في رؤيته٧٤٣٨هل تضارون في القمر ليلة البدر٧٤٣٩هل تضارون في رؤية الشمس والقمر إذا كانت صحوا٧٤٤٠يحبس المؤمنون يوم القيامة حتى يهموا بذلك فيقولون لو استشفعنا إلى ربنا…٧٤٤١اصبروا حتى تلقوا الله ورسوله، فإني على الحوض٧٤٤٢اللهم ربنا لك الحمد، أنت قيم السموات والأرض، ولك الحمد أنت رب السموات…٧٤٤٣ما منكم من أحد إلا سيكلمه ربه، ليس بينه وبينه ترجمان ولا حجاب يحجبه٧٤٤٤جنتان من فضة آنيتهما وما فيهما، وجنتان من ذهب آنيتهما وما فيهما، وما…٧٤٤٥من اقتطع مال امرئ مسلم بيمين كاذبة، لقي الله وهو عليه غضبان٧٤٤٦ثلاثة لا يكلمهم الله يوم القيامة ولا ينظر إليهم رجل حلف على سلعة لقد…٧٤٤٧الزمان قد استدار كهيئته يوم خلق الله السموات والأرض، السنة اثنا عشر…٧٤٤٨إن لله ما أخذ، وله ما أعطى، وكل إلى أجل مسمى، فلتصبر ولتحتسب٧٤٤٩اختصمت الجنة والنار إلى ربهما فقالت الجنة يا رب ما لها لا يدخلها إلا…٧٤٥٠ليصيبن أقواما سفع من النار بذنوب أصابوها عقوبة، ثم يدخلهم الله الجنة…٧٤٥١{ وما قدروا الله حق قدره }٧٤٥٢قال بت في بيت ميمونة ليلة والنبي صلى الله عليه وسلم عندها لأنظر كيف…٧٤٥٣لما قضى الله الخلق كتب عنده فوق عرشه، إن رحمتي سبقت غضبي٧٤٥٤إن خلق أحدكم يجمع في بطن أمه أربعين يوما وأربعين ليلة، ثم يكون علقة…٧٤٥٥يا جبريل ما يمنعك أن تزورنا أكثر مما تزورنا٧٤٥٦قال كنت أمشي مع رسول الله صلى الله عليه وسلم في حرث بالمدينة وهو متكئ…٧٤٥٧تكفل الله لمن جاهد في سبيله، لا يخرجه إلا الجهاد في سبيله، وتصديق…٧٤٥٨من قاتل لتكون كلمة الله هي العليا، فهو في سبيل الله٧٤٥٩لا يزال من أمتي قوم ظاهرين على الناس، حتى يأتيهم أمر الله٧٤٦٠لا يزال من أمتي أمة قائمة بأمر الله، ما يضرهم من كذبهم، ولا من…٧٤٦١لو سألتني هذه القطعة ما أعطيتكها، ولن تعدو أمر الله فيك، ولئن أدبرت…٧٤٦٢مع النبي صلى الله عليه وسلم في بعض حرث المدينة وهو يتوكأ على عسيب…٧٤٦٣تكفل الله لمن جاهد في سبيله، لا يخرجه من بيته إلا الجهاد في سبيله،…٧٤٦٤إذا دعوتم الله فاعزموا في الدعاء، ولا يقولن أحدكم إن شئت فأعطني، فإن…٧٤٦٥فانصرف رسول الله صلى الله عليه وسلم حين قلت ذلك، ولم يرجع إلى شيئا،…٧٤٦٦مثل المؤمن كمثل خامة الزرع، يفيء ورقه من حيث أتتها الريح تكفئها، فإذا…٧٤٦٧إنما بقاؤكم فيما سلف قبلكم من الأمم، كما بين صلاة العصر إلى غروب…٧٤٦٨أبايعكم على أن لا تشركوا بالله شيئا، ولا تسرقوا، ولا تزنوا، ولا…٧٤٦٩لو كان سليمان استثنى لحملت كل امرأة منهن، فولدت فارسا يقاتل في سبيل…٧٤٧٠لا بأس عليك طهور، إن شاء الله٧٤٧١إن الله قبض أرواحكم حين شاء، وردها حين شاء٧٤٧٢لا تخيروني على موسى، فإن الناس يصعقون يوم القيامة فأكون أول من يفيق،…٧٤٧٣المدينة يأتيها الدجال فيجد الملائكة يحرسونها فلا يقربها الدجال ولا…٧٤٧٤لكل نبي دعوة، فأريد إن شاء الله أن أختبي دعوتي شفاعة لأمتي يوم القيامة٧٤٧٥بينا أنا نائم رأيتني على قليب فنزعت ما شاء الله أن أنزع، ثم أخذها ابن…٧٤٧٦اشفعوا فلتؤجروا، ويقضي الله على لسان رسوله ما شاء٧٤٧٧لا يقل أحدكم اللهم اغفر لي إن شئت، ارحمني إن شئت، ارزقني إن شئت،…٧٤٧٨بينا موسى في ملإ بني إسرائيل إذ جاءه رجل فقال هل تعلم أحدا أعلم منك…٧٤٧٩ننزل غدا إن شاء الله بخيف بني كنانة حيث تقاسموا على الكفر٧٤٨٠إنا قافلون إن شاء الله٧٤٨١إذا قضى الله الأمر في السماء ضربت الملائكة بأجنحتها خضعانا لقوله،…٧٤٨٢ما أذن الله لشىء ما أذن للنبي صلى الله عليه وسلم يتغنى بالقرآن٧٤٨٣يقول الله يا آدم . فيقول لبيك وسعديك . فينادى بصوت إن الله يأمرك أن…٧٤٨٤ولقد أمره ربه أن يبشرها ببيت في الجنة .٧٤٨٥إن الله تبارك وتعالى إذا أحب عبدا نادى جبريل إن الله قد أحب فلانا…٧٤٨٦يتعاقبون فيكم ملائكة بالليل وملائكة بالنهار، ويجتمعون في صلاة العصر…٧٤٨٧أتاني جبريل فبشرني أنه من مات لا يشرك بالله شيئا دخل الجنة٧٤٨٨يا فلان إذا أويت إلى فراشك فقل اللهم أسلمت نفسي إليك، ووجهت وجهي إليك…٧٤٨٩اللهم منزل الكتاب، سريع الحساب، اهزم الأحزاب وزلزل بهم٧٤٩٠فكان إذا رفع صوته سمع المشركون فسبوا القرآن ومن أنزله ومن جاء به .…٧٤٩١قال الله تعالى يؤذيني ابن آدم، يسب الدهر وأنا الدهر، بيدي الأمر، أقلب…٧٤٩٢يقول الله عز وجل الصوم لي وأنا أجزي به يدع شهوته وأكله وشربه من أجلي،…٧٤٩٣بينما أيوب يغتسل عريانا خر عليه رجل جراد من ذهب فجعل يحثي في ثوبه،…٧٤٩٤يتنزل ربنا تبارك وتعالى كل ليلة إلى السماء الدنيا حين يبقى ثلث الليل…٧٤٩٦نحن الآخرون السابقون يوم القيامة٧٤٩٧هذه خديجة أتتك بإناء فيه طعام أو إناء فيه شراب فأقرئها من ربها السلام…٧٤٩٨قال الله أعددت لعبادي الصالحين ما لا عين رأت، ولا أذن سمعت، ولا خطر…٧٤٩٩اللهم لك الحمد أنت نور السموات والأرض، ولك الحمد أنت قيم السموات…٧٥٠٠وسعيد بن المسيب، وعلقمة بن وقاص، وعبيد الله بن عبد الله، عن حديث،…٧٥٠١يقول الله إذا أراد عبدي أن يعمل سيئة فلا تكتبوها عليه حتى يعملها، فإن…٧٥٠٢خلق الله الخلق فلما فرغ منه قامت الرحم فقال مه . قالت هذا مقام العائذ…٧٥٠٣قال الله أصبح من عبادي كافر بي ومؤمن بي٧٥٠٤قال الله إذا أحب عبدي لقائي أحببت لقاءه، وإذا كره لقائي كرهت لقاءه٧٥٠٥قال الله أنا عند ظن عبدي بي٧٥٠٦قال رجل لم يعمل خيرا قط، فإذا مات فحرقوه واذروا نصفه في البر ونصفه في…٧٥٠٧إن عبدا أصاب ذنبا وربما قال أذنب ذنبا فقال رب أذنبت وربما قال أصبت…٧٥٠٨أنه ذكر رجلا فيمن سلف أو فيمن كان قبلكم قال كلمة يعني أعطاه الله مالا…٧٥٠٩إذا كان يوم القيامة شفعت، فقلت يا رب أدخل الجنة من كان في قلبه خردلة…٧٥١٠إذا كان يوم القيامة ماج الناس بعضهم في بعض فيأتون آدم فيقولون اشفع…٧٥١١إن آخر أهل الجنة دخولا الجنة، وآخر أهل النار خروجا من النار رجل يخرج…٧٥١٢ما منكم أحد إلا سيكلمه ربه، ليس بينه وبينه ترجمان، فينظر أيمن منه فلا…٧٥١٣والأرضين على إصبع، والماء والثرى على إصبع، والخلائق على إصبع، ثم…٧٥١٤يدنو أحدكم من ربه حتى يضع كنفه عليه فيقول أعملت كذا وكذا فيقول نعم .…٧٥١٥احتج آدم وموسى، فقال موسى أنت آدم الذي أخرجت ذريتك من الجنة . قال آدم…٧٥١٦يجمع المؤمنون يوم القيامة فيقولون لو استشفعنا إلى ربنا، فيريحنا من…٧٥١٧يقول ليلة أسري برسول الله صلى الله عليه وسلم من مسجد الكعبة أنه جاءه…٧٥١٨إن الله يقول لأهل الجنة يا أهل الجنة . فيقولون لبيك ربنا وسعديك…٧٥١٩أن رجلا من أهل الجنة استأذن ربه في الزرع فقال أو لست فيما شئت . قال…٧٥٢٠أن تجعل لله ندا وهو خلقك٧٥٢١أو قرشيان وثقفي، كثيرة شحم بطونهم قليلة فقه قلوبهم فقال أحدهم أترون…٧٥٢٢تقرءونه محضا لم يشب٧٥٢٣أن عبد الله بن عباس، قال يا معشر المسلمين كيف تسألون أهل الكتاب عن…٧٥٢٤في قوله تعالى { لا تحرك به لسانك } قال كان النبي صلى الله عليه وسلم…٧٥٢٥فكان إذا صلى بأصحابه رفع صوته بالقرآن، فإذا سمعه المشركون سبوا القرآن…
