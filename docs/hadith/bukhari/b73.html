٣٠ حديثًا
٥٥٤٥إن أول ما نبدأ به في يومنا هذا أن نصلي ثم نرجع فننحر، من فعله فقد…٥٥٤٦من ذبح قبل الصلاة فإنما ذبح لنفسه، ومن ذبح بعد الصلاة فقد تم نسكه،…٥٥٤٧قال قسم النبي صلى الله عليه وسلم بين أصحابه ضحايا، فصارت لعقبة جذعة .…٥٥٤٨قبل أن تدخل مكة وهى تبكي فقال " ما لك أنفست " . قالت نعم . قال " إن…٥٥٤٩من كان ذبح قبل الصلاة فليعد٥٥٥٠الزمان قد استدار كهيئته يوم خلق الله السموات والأرض، السنة اثنا عشر…٥٥٥١قال كان عبد الله ينحر في المنحر . قال عبيد الله يعني منحر النبي صلى…٥٥٥٢أن ابن عمر رضى الله عنهما أخبره قال كان رسول الله صلى الله عليه وسلم…٥٥٥٣حدثنا آدم بن أبي إياس، حدثنا شعبة، حدثنا عبد العزيز بن صهيب، قال سمعت…٥٥٥٤حدثنا قتيبة بن سعيد، حدثنا عبد الوهاب، عن أيوب، عن أبي قلابة، عن أنس،…٥٥٥٥فبقي عتود فذكره للنبي صلى الله عليه وسلم فقال " ضح أنت به " .٥٥٥٦ومن ذبح بعد الصلاة فقد تم نسكه، وأصاب سنة المسلمين " . تابعه عبيدة عن…٥٥٥٧قال ذبح أبو بردة قبل الصلاة فقال له النبي صلى الله عليه وسلم " أبدلها…٥٥٥٨قال ضحى النبي صلى الله عليه وسلم بكبشين أملحين، فرأيته واضعا قدمه على…٥٥٥٩حدثنا قتيبة، حدثنا سفيان، عن عبد الرحمن بن القاسم، عن أبيه، عن عائشة…٥٥٦٠إن أول ما نبدأ من يومنا هذا أن نصلي، ثم نرجع فننحر، فمن فعل هذا فقد…٥٥٦١من ذبح قبل الصلاة فليعد٥٥٦٢من ذبح قبل أن يصلي فليعد مكانها أخرى، ومن لم يذبح فليذبح٥٥٦٣من صلى صلاتنا واستقبل قبلتنا، فلا يذبح حتى ينصرف٥٥٦٤ووضع رجله على صفحتهما، ويذبحهما بيده .٥٥٦٥قال ضحى النبي صلى الله عليه وسلم بكبشين أملحين أقرنين، ذبحهما بيده،…٥٥٦٦أنه أتى عائشة، فقال لها يا أم المؤمنين إن رجلا يبعث بالهدى إلى…٥٥٦٧وقال غير مرة لحوم الهدى .٥٥٦٨أن ابن خباب، أخبره أنه، سمع أبا سعيد، يحدث أنه كان غائبا، فقدم فقدم…٥٥٦٩من ضحى منكم فلا يصبحن بعد ثالثة وفي بيته منه شىء٥٥٧٠لا تأكلوا إلا ثلاثة أيام٥٥٧١مولى ابن أزهر أنه شهد العيد يوم الأضحى مع عمر بن الخطاب رضى الله عنه…٥٥٧٢قال أبو عبيد ثم شهدت مع عثمان بن عفان فكان ذلك يوم الجمعة، فصلى قبل…٥٥٧٣قال أبو عبيد ثم شهدته مع علي بن أبي طالب، فصلى قبل الخطبة، ثم خطب…٥٥٧٤كلوا من الأضاحي ثلاثا
