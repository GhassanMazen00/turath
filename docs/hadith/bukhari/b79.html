٧٥ حديثًا
٦٢٢٧خلق الله آدم على صورته، طوله ستون ذراعا، فلما خلقه قال اذهب فسلم على…٦٢٢٨وكان الفضل رجلا وضيئا، فوقف النبي صلى الله عليه وسلم للناس يفتيهم،…٦٢٢٩إياكم والجلوس بالطرقات٦٢٣٠إن الله هو السلام، فإذا جلس أحدكم في الصلاة فليقل التحيات لله،…٦٢٣١يسلم الصغير على الكبير، والمار على القاعد، والقليل على الكثير٦٢٣٢يسلم الراكب على الماشي، والماشي على القاعد، والقليل على الكثير٦٢٣٣يسلم الراكب على الماشي والماشي على القاعد، والقليل على الكثير٦٢٣٤يسلم الصغير على الكبير، والمار على القاعد، والقليل على الكثير٦٢٣٥واتباع الجنائز، وتشميت العاطس، ونصر الضعيف، وعون المظلوم، وإفشاء…٦٢٣٦تطعم الطعام، وتقرأ السلام على من عرفت، وعلى من لم تعرف٦٢٣٧لا يحل لمسلم أن يهجر أخاه فوق ثلاث، يلتقيان فيصد هذا، ويصد هذا،…٦٢٣٨أنه كان ابن عشر سنين مقدم رسول الله صلى الله عليه وسلم المدينة، فخدمت…٦٢٣٩ثم جلسوا يتحدثون فأخذ كأنه يتهيأ للقيام فلم يقوموا، فلما رأى ذلك قام،…٦٢٤٠أن عائشة رضى الله عنها زوج النبي صلى الله عليه وسلم قالت كان عمر بن…٦٢٤١لو أعلم أنك تنظر لطعنت به في عينك، إنما جعل الاستئذان من أجل البصر٦٢٤٢أن رجلا، اطلع من بعض حجر النبي صلى الله عليه وسلم فقام إليه النبي صلى…٦٢٤٣إن الله كتب على ابن آدم حظه من الزنا، أدرك ذلك لا محالة، فزنا العين…٦٢٤٤وإذا تكلم بكلمة أعادها ثلاثا .٦٢٤٥إذا استأذن أحدكم ثلاثا فلم يؤذن له، فليرجع٦٢٤٦أبا هر الحق أهل الصفة فادعهم إلى٦٢٤٧حدثنا علي بن الجعد، أخبرنا شعبة، عن سيار، عن ثابت البناني، عن أنس بن…٦٢٤٨وتكركر حبات من شعير، فإذا صلينا الجمعة انصرفنا ونسلم عليها فتقدمه…٦٢٤٩يا عائشة هذا جبريل يقرأ عليك السلام٦٢٥٠حدثنا أبو الوليد، هشام بن عبد الملك حدثنا شعبة، عن محمد بن المنكدر،…٦٢٥١وعليك السلام ارجع فصل فإنك لم تصل٦٢٥٢ثم ارفع حتى تطمئن جالسا٦٢٥٣إن جبريل يقرئك السلام٦٢٥٤أى سعد ألم تسمع ما قال أبو حباب٦٢٥٥أن عبد الله بن كعب، قال سمعت كعب بن مالك، يحدث حين تخلف عن تبوك،…٦٢٥٦مهلا يا عائشة، فإن الله يحب الرفق في الأمر كله٦٢٥٧إذا سلم عليكم اليهود فإنما يقول أحدهم السام عليك . فقل وعليك٦٢٥٨إذا سلم عليكم أهل الكتاب فقولوا وعليكم٦٢٥٩انطلقوا حتى تأتوا روضة خاخ، فإن بها امرأة من المشركين معها صحيفة من…٦٢٦٠بسم الله الرحمن الرحيم، من محمد عبد الله ورسوله إلى هرقل عظيم الروم،…٦٢٦١نجر خشبة، فجعل المال في جوفها، وكتب إليه صحيفة من فلان إلى فلان٦٢٦٢قوموا إلى سيدكم٦٢٦٣قال قلت لأنس أكانت المصافحة في أصحاب النبي صلى الله عليه وسلم قال نعم…٦٢٦٤زهرة بن معبد سمع جده عبد الله بن هشام، قال كنا مع النبي صلى الله عليه…٦٢٦٥يقول علمني رسول الله صلى الله عليه وسلم وكفي بين كفيه التشهد، كما…٦٢٦٦أن عبد الله بن عباس، أخبره أن عليا يعني ابن أبي طالب خرج من عند النبي…٦٢٦٧بهذا .٦٢٦٨يا أبا ذر ما أحب أن أحدا لي ذهبا يأتي على ليلة أو ثلاث عندي منه…٦٢٦٩لا يقيم الرجل الرجل من مجلسه، ثم يجلس فيه٦٢٧٠ولكن تفسحوا وتوسعوا . وكان ابن عمر يكره أن يقوم الرجل من مجلسه، ثم…٦٢٧١فلما رأى ذلك قام، فلما قام قام من قام معه من الناس، وبقي ثلاثة، وإن…٦٢٧٢حدثنا محمد بن أبي غالب، أخبرنا إبراهيم بن المنذر الحزامي، حدثنا محمد…٦٢٧٣ألا أخبركم بأكبر الكبائر٦٢٧٤مثله، وكان متكئا فجلس فقال " ألا وقول الزور " . فما زال يكررها حتى…٦٢٧٥أن عقبة بن الحارث، حدثه قال صلى النبي صلى الله عليه وسلم العصر،…٦٢٧٦فأكره أن أقوم فأستقبله فأنسل انسلالا .٦٢٧٧أما يكفيك من كل شهر ثلاثة أيام٦٢٧٨قال ذهب علقمة إلى الشأم، فأتى المسجد فصلى ركعتين فقال اللهم ارزقني…٦٢٧٩حدثنا محمد بن كثير، حدثنا سفيان، عن أبي حازم، عن سهل بن سعد، قال كنا…٦٢٨٠قال ما كان لعلي اسم أحب إليه من أبي تراب، وإن كان ليفرح به إذا دعي…٦٢٨١أن أم سليم، كانت تبسط للنبي صلى الله عليه وسلم نطعا فيقيل عندها على…٦٢٨٣ناس من أمتي عرضوا على غزاة في سبيل الله، يركبون ثبج هذا البحر، ملوكا…٦٢٨٤والاحتباء في ثوب واحد، ليس على فرج الإنسان منه شىء، والملامسة،…٦٢٨٦حدثتني عائشة أم المؤمنين، قالت إنا كنا أزواج النبي صلى الله عليه وسلم…٦٢٨٧قال رأيت رسول الله صلى الله عليه وسلم في المسجد مستلقيا، واضعا إحدى…٦٢٨٨إذا كانوا ثلاثة فلا يتناجى اثنان دون الثالث٦٢٨٩أسر إلى النبي صلى الله عليه وسلم سرا فما أخبرت به أحدا بعده، ولقد…٦٢٩٠حتى تختلطوا بالناس، أجل أن يحزنه } .٦٢٩١رحمة الله على موسى، أوذي بأكثر من هذا فصبر٦٢٩٢ثم قام فصلى .٦٢٩٣حدثنا أبو نعيم، حدثنا ابن عيينة، عن الزهري، عن سالم، عن أبيه، عن…٦٢٩٤إن هذه النار إنما هي عدو لكم، فإذا نمتم فأطفئوها عنكم٦٢٩٥خمروا الآنية وأجيفوا الأبواب، وأطفئوا المصابيح، فإن الفويسقة ربما جرت…٦٢٩٦أطفئوا المصابيح بالليل إذا رقدتم، وغلقوا الأبواب، وأوكوا الأسقية،…٦٢٩٧الفطرة خمس الختان، والاستحداد، ونتف الإبط، وقص الشارب، وتقليم الأظفار٦٢٩٨اختتن إبراهيم بعد ثمانين سنة، واختتن بالقدوم٦٢٩٩حدثنا محمد بن عبد الرحيم، أخبرنا عباد بن موسى، حدثنا إسماعيل بن جعفر،…٦٣٠٠وقال ابن إدريس عن أبيه، عن أبي إسحاق، عن سعيد بن جبير، عن ابن عباس،…٦٣٠١من حلف منكم فقال في حلفه باللات والعزى . فليقل لا إله إلا الله . ومن…٦٣٠٢يكنني من المطر، ويظلني من الشمس، ما أعانني عليه أحد من خلق الله .٦٣٠٣قال عمرو قال ابن عمر والله ما وضعت لبنة على لبنة، ولا غرست نخلة، منذ…
