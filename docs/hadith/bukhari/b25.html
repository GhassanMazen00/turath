١٥١٣فجعل الفضل ينظر إليها وتنظر إليه، وجعل النبي صلى الله عليه وسلم يصرف…١٥١٤أن سالم بن عبد الله، أخبره أن ابن عمر رضى الله عنهما قال رأيت رسول…١٥١٥سمع عطاء، يحدث عن جابر بن عبد الله رضى الله عنهما أن إهلال رسول الله…١٥١٦رضى الله عنها أن النبي صلى الله عليه وسلم بعث معها أخاها عبد الرحمن،…١٥١٧قال حج أنس على رحل، ولم يكن شحيحا، وحدث أن رسول الله صلى الله عليه…١٥١٨يا عبد الرحمن اذهب بأختك فأعمرها من التنعيم١٥١٩إيمان بالله ورسوله١٥٢٠لا، لكن أفضل الجهاد حج مبرور١٥٢١من حج لله فلم يرفث ولم يفسق رجع كيوم ولدته أمه١٥٢٢أنه أتى عبد الله بن عمر رضى الله عنهما في منزله وله فسطاط وسرادق،…١٥٢٣فإذا قدموا مكة سألوا الناس، فأنزل الله تعالى { وتزودوا فإن خير الزاد…١٥٢٤قال إن النبي صلى الله عليه وسلم وقت لأهل المدينة ذا الحليفة، ولأهل…١٥٢٥يهل أهل المدينة من ذي الحليفة، وأهل الشأم من الجحفة، وأهل نجد من قرن١٥٢٦ولأهل الشأم الجحفة، ولأهل نجد قرن المنازل، ولأهل اليمن يلملم، فهن لهن…١٥٢٧وقت النبي صلى الله عليه وسلم .١٥٢٨مهل أهل المدينة ذو الحليفة، ومهل أهل الشأم مهيعة وهي الجحفة، وأهل نجد…١٥٢٩ولأهل الشأم الجحفة، ولأهل اليمن يلملم، ولأهل نجد قرنا، فهن لهن ولمن…١٥٣٠رضى الله عنهما أن النبي صلى الله عليه وسلم وقت لأهل المدينة ذا…١٥٣١إن رسول الله صلى الله عليه وسلم حد لأهل نجد قرنا، وهو جور عن طريقنا،…١٥٣٢حدثنا عبد الله بن يوسف، أخبرنا مالك، عن نافع، عن عبد الله بن عمر رضى…١٥٣٣ويدخل من طريق المعرس، وأن رسول الله صلى الله عليه وسلم كان إذا خرج…١٥٣٤أتاني الليلة آت من ربي فقال صل في هذا الوادي المبارك وقل عمرة في حجة١٥٣٥يتوخى بالمناخ الذي كان عبد الله ينيخ، يتحرى معرس رسول الله صلى الله…١٥٣٦أين الذي سأل عن العمرة١٥٣٨حدثنا محمد بن يوسف، حدثنا سفيان، عن منصور، عن سعيد بن جبير، قال كان…١٥٣٩ولحله قبل أن يطوف بالبيت .١٥٤٠حدثنا أصبغ، أخبرنا ابن وهب، عن يونس، عن ابن شهاب، عن سالم، عن أبيه…١٥٤١أنه سمع أباه، يقول ما أهل رسول الله صلى الله عليه وسلم إلا من عند…١٥٤٢لا يلبس القمص ولا العمائم ولا السراويلات ولا البرانس ولا الخفاف، إلا…١٥٤٤ثم أردف الفضل من المزدلفة إلى منى . قال فكلاهما قال لم يزل النبي صلى…١٥٤٥بعد ما ترجل وادهن ولبس إزاره ورداءه، هو وأصحابه، فلم ينه عن شىء من…١٥٤٦وبذي الحليفة ركعتين، ثم بات حتى أصبح بذي الحليفة، فلما ركب راحلته…١٥٤٧وصلى العصر بذي الحليفة ركعتين، قال وأحسبه بات بها حتى أصبح .١٥٤٨والعصر بذي الحليفة ركعتين، وسمعتهم يصرخون بهما جميعا .١٥٤٩رسول الله صلى الله عليه وسلم لبيك اللهم لبيك، لبيك لا شريك لك لبيك،…١٥٥٠لبيك لا شريك لك لبيك، إن الحمد والنعمة لك . تابعه أبو معاوية عن…١٥٥١والعصر بذي الحليفة ركعتين، ثم بات بها حتى أصبح، ثم ركب حتى استوت به…١٥٥٢حدثنا أبو عاصم، أخبرنا ابن جريج، قال أخبرني صالح بن كيسان، عن نافع،…١٥٥٣قال كان ابن عمر رضى الله عنهما إذا صلى بالغداة بذي الحليفة أمر…١٥٥٤قال كان ابن عمر رضى الله عنهما إذا أراد الخروج إلى مكة ادهن بدهن ليس…١٥٥٥مكتوب بين عينيه كافر١٥٥٦من كان معه هدى فليهل بالحج مع العمرة، ثم لا يحل حتى يحل منهما جميعا١٥٥٧قال عطاء قال جابر رضى الله عنه أمر النبي صلى الله عليه وسلم عليا رضى…١٥٥٨حدثنا الحسن بن علي الخلال الهذلي، حدثنا عبد الصمد، حدثنا سليم بن…١٥٥٩أو غسلت رأسي، فقدم عمر رضى الله عنه فقال إن نأخذ بكتاب الله فإنه…١٥٦٠من لم يكن منكم معه هدى فأحب أن يجعلها عمرة فليفعل، ومن كان معه الهدى…١٥٦١وما طفت ليالي قدمنا مكة١٥٦٢فأما من أهل بالحج أو جمع الحج والعمرة لم يحلوا حتى كان يوم النحر .١٥٦٣أهل بهما لبيك بعمرة وحجة قال ما كنت لأدع سنة النبي صلى الله عليه وسلم…١٥٦٤ويجعلون المحرم صفرا ويقولون إذا برأ الدبر، وعفا الأثر، وانسلخ صفر،…١٥٦٥حدثنا محمد بن المثنى، حدثنا غندر، حدثنا شعبة، عن قيس بن مسلم، عن طارق…١٥٦٦إني لبدت رأسي، وقلدت هديي فلا أحل حتى أنحر١٥٦٧نصر بن عمران الضبعي قال تمتعت فنهاني ناس، فسألت ابن عباس رضى الله…١٥٦٨أحلوا من إحرامكم بطواف البيت وبين الصفا والمروة، وقصروا ثم أقيموا…١٥٦٩قال اختلف علي وعثمان رضى الله عنهما وهما بعسفان في المتعة، فقال علي…١٥٧٠حدثنا مسدد، حدثنا حماد بن زيد، عن أيوب، قال سمعت مجاهدا، يقول حدثنا…١٥٧١حدثنا موسى بن إسماعيل، حدثنا همام، عن قتادة، قال حدثني مطرف، عن عمران…١٥٧٢اجعلوا إهلالكم بالحج عمرة إلا من قلد الهدى١٥٧٣ثم يبيت بذي طوى، ثم يصلي به الصبح ويغتسل، ويحدث أن نبي الله صلى الله…١٥٧٤حدثنا مسدد، حدثنا يحيى، عن عبيد الله، قال حدثني نافع، عن ابن عمر رضى…١٥٧٥ويخرج من الثنية السفلى .١٥٧٦وما أبالي كتبي كانت عندي أو عند مسدد .١٥٧٧ومحمد بن المثنى، قالا حدثنا سفيان بن عيينة، عن هشام بن عروة، عن أبيه،…١٥٧٨وخرج من كدا من أعلى مكة .١٥٧٩وأكثر ما يدخل من كداء، وكانت أقربهما إلى منزله .١٥٨٠دخل النبي صلى الله عليه وسلم عام الفتح من كداء من أعلى مكة . وكان…١٥٨١دخل النبي صلى الله عليه وسلم عام الفتح من كداء . وكان عروة يدخل منهما…١٥٨٢وطمحت عيناه إلى السماء فقال " أرني إزاري " . فشده عليه .١٥٨٣ألم ترى أن قومك لما بنوا الكعبة اقتصروا عن قواعد إبراهيم١٥٨٤ولولا أن قومك حديث عهدهم بالجاهلية فأخاف أن تنكر قلوبهم أن أدخل الجدر…١٥٨٥لولا حداثة قومك بالكفر لنقضت البيت ثم لبنيته على أساس إبراهيم عليه…١٥٨٦يا عائشة لولا أن قومك حديث عهد بجاهلية لأمرت بالبيت فهدم، فأدخلت فيه…١٥٨٧إن هذا البلد حرمه الله، لا يعضد شوكه، ولا ينفر صيده، ولا يلتقط لقطته…١٥٨٨وهل ترك عقيل من رباع أو دور١٥٨٩منزلنا غدا إن شاء الله بخيف بني كنانة حيث تقاسموا على الكفر١٥٩٠نحن نازلون غدا بخيف بني كنانة حيث تقاسموا على الكفر١٥٩١يخرب الكعبة ذو السويقتين من الحبشة١٥٩٢من شاء أن يصومه فليصمه، ومن شاء أن يتركه فليتركه١٥٩٣ليحجن البيت وليعتمرن بعد خروج يأجوج ومأجوج١٥٩٤قال جلست مع شيبة على الكرسي في الكعبة فقال لقد جلس هذا المجلس عمر رضى…١٥٩٥كأني به أسود أفحج، يقلعها حجرا حجرا١٥٩٦يخرب الكعبة ذو السويقتين من الحبشة١٥٩٧فقال إني أعلم أنك حجر لا تضر ولا تنفع، ولولا أني رأيت النبي صلى الله…١٥٩٨أنه قال دخل رسول الله صلى الله عليه وسلم البيت هو وأسامة بن زيد،…١٥٩٩ويجعل الباب قبل الظهر، يمشي حتى يكون بينه وبين الجدار الذي قبل وجهه…١٦٠٠قال اعتمر رسول الله صلى الله عليه وسلم فطاف بالبيت وصلى خلف المقام…١٦٠١قاتلهم الله أما والله قد علموا أنهما لم يستقسما بها قط١٦٠٢وقد وهنهم حمى يثرب . فأمرهم النبي صلى الله عليه وسلم أن يرملوا…١٦٠٣إذا استلم الركن الأسود أول ما يطوف يخب ثلاثة أطواف من السبع .١٦٠٤حدثني محمد، حدثنا سريج بن النعمان، حدثنا فليح، عن نافع، عن ابن عمر…١٦٠٥أن عمر بن الخطاب رضى الله عنه قال للركن أما والله إني لأعلم أنك حجر…١٦٠٦منذ رأيت النبي صلى الله عليه وسلم يستلمهما . قلت لنافع أكان ابن عمر…١٦٠٧ويحيى بن سليمان، قالا حدثنا ابن وهب، قال أخبرني يونس، عن ابن شهاب، عن…١٦٠٨أنه قال ومن يتقي شيئا من البيت، وكان معاوية يستلم الأركان، فقال له…١٦٠٩حدثنا أبو الوليد، حدثنا ليث، عن ابن شهاب، عن سالم بن عبد الله، عن…١٦١٠قال رأيت عمر بن الخطاب رضى الله عنه قبل الحجر وقال لولا أني رأيت رسول…١٦١١. فقال رأيت رسول الله صلى الله عليه وسلم يستلمه ويقبله . قال قلت…١٦١٢كلما أتى على الركن أشار إليه .١٦١٣حدثنا مسدد، حدثنا خالد بن عبد الله، حدثنا خالد الحذاء، عن عكرمة، عن…١٦١٥، ذكرت لعروة، قال فأخبرتني عائشة رضى الله عنها أن أول، شىء بدأ به حين…١٦١٦ومشى أربعة، ثم سجد سجدتين، ثم يطوف بين الصفا والمروة .١٦١٧ويمشي أربعة، وأنه كان يسعى بطن المسيل إذا طاف بين الصفا والمروة .١٦١٨إذ منع ابن هشام النساء الطواف مع الرجال قال كيف يمنعهن، وقد طاف نساء…١٦١٩طوفي من وراء الناس، وأنت راكبة١٦٢٠أن ابن جريج، أخبرهم قال أخبرني سليمان الأحول، أن طاوسا، أخبره عن ابن…١٦٢١حدثنا أبو عاصم، عن ابن جريج، عن سليمان الأحول، عن طاوس، عن ابن عباس…١٦٢٢ألا لا يحج بعد العام مشرك، ولا يطوف بالبيت عريان١٦٢٤ثم صلى خلف المقام ركعتين، وطاف بين الصفا والمروة، وقال { لقد كان لكم…١٦٢٥فطاف وسعى بين الصفا والمروة، ولم يقرب الكعبة بعد طوافه بها حتى رجع من…١٦٢٦إذا أقيمت صلاة الصبح فطوفي على بعيرك، والناس يصلون١٦٢٧ثم خرج إلى الصفا، وقد قال الله تعالى { لقد كان لكم في رسول الله أسوة…١٦٢٨طافوا بالبيت بعد صلاة الصبح، ثم قعدوا إلى المذكر، حتى إذا طلعت الشمس…١٦٢٩حدثنا إبراهيم بن المنذر، حدثنا أبو ضمرة، حدثنا موسى بن عقبة، عن نافع،…١٦٣١قال رأيت عبد الله بن الزبير رضى الله عنهما يطوف بعد الفجر، ويصلي…١٦٣٢وهو على بعير، كلما أتى على الركن أشار إليه بشىء في يده وكبر .١٦٣٣طوفي من وراء الناس وأنت راكبة١٦٣٤فأذن له .١٦٣٥فاستسقى، فقال العباس يا فضل اذهب إلى أمك، فأت رسول الله صلى الله عليه…١٦٣٦فرج سقفي وأنا بمكة، فنزل جبريل عليه السلام ففرج صدري، ثم غسله بماء…١٦٣٧أن ابن عباس رضى الله عنهما حدثه قال سقيت رسول الله صلى الله عليه وسلم…١٦٣٨من كان معه هدى فليهل بالحج والعمرة ثم لا يحل حتى يحل منهما١٦٣٩أن ابن عمر رضى الله عنهما دخل ابنه عبد الله بن عبد الله، وظهره في…١٦٤٠أن ابن عمر رضى الله عنهما أراد الحج عام نزل الحجاج بابن الزبير . فقيل…١٦٤٢أنه سأل عروة بن الزبير فقال قد حج النبي صلى الله عليه وسلم فأخبرتني…١٦٤٣قال عروة سألت عائشة رضى الله عنها فقلت لها أرأيت قول الله تعالى { إن…١٦٤٤وكان يسعى بطن المسيل إذا طاف بين الصفا والمروة . فقلت لنافع أكان عبد…١٦٤٦طاف بالبيت في عمرة، ولم يطف بين الصفا والمروة أيأتي امرأته فقال قدم…١٦٤٧فطاف بالبيت، ثم صلى ركعتين، ثم سعى بين الصفا والمروة، ثم تلا { لقد…١٦٤٨قال قلت لأنس بن مالك رضى الله عنه أكنتم تكرهون السعى بين الصفا…١٦٤٩مثله .١٦٥٠افعلي كما يفعل الحاج غير أن لا تطوفي بالبيت حتى تطهري١٦٥١لو استقبلت من أمري ما استدبرت ما أهديت، ولولا أن معي الهدى لأحللت١٦٥٢لتلبسها صاحبتها من جلبابها، ولتشهد الخير، ودعوة المؤمنين١٦٥٣حدثني عبد الله بن محمد، حدثنا إسحاق الأزرق، حدثنا سفيان، عن عبد…١٦٥٤سمع أبا بكر بن عياش، حدثنا عبد العزيز، لقيت أنسا . وحدثني إسماعيل بن…١٦٥٥قال صلى رسول الله صلى الله عليه وسلم بمنى ركعتين، وأبو بكر وعمر…١٦٥٦حدثنا آدم، حدثنا شعبة، عن أبي إسحاق الهمداني، عن حارثة بن وهب الخزاعي…١٦٥٧ومع أبي بكر رضى الله عنه ركعتين ومع عمر رضى الله عنه ركعتين، ثم تفرقت…١٦٥٨شك الناس يوم عرفة في صوم النبي صلى الله عليه وسلم فبعثت إلى النبي صلى…١٦٥٩حدثنا عبد الله بن يوسف، أخبرنا مالك، عن محمد بن أبي بكر الثقفي، أنه…١٦٦٠قال كتب عبد الملك إلى الحجاج أن لا يخالف ابن عمر في الحج، فجاء ابن…١٦٦١أن ناسا، اختلفوا عندها يوم عرفة في صوم النبي صلى الله عليه وسلم فقال…١٦٦٢أن الحجاج بن يوسف، عام نزل بابن الزبير رضى الله عنهما سأل عبد الله…١٦٦٣أن عبد الملك بن مروان، كتب إلى الحجاج أن يأتم، بعبد الله بن عمر في…١٦٦٤سمع محمد بن جبير، عن أبيه، جبير بن مطعم قال أضللت بعيرا لي، فذهبت…١٦٦٥قال عروة كان الناس يطوفون في الجاهلية عراة إلا الحمس، والحمس قريش وما…١٦٦٦كيف كان رسول الله صلى الله عليه وسلم يسير في حجة الوداع حين دفع قال…١٦٦٧حدثنا مسدد، حدثنا حماد بن زيد، عن يحيى بن سعيد، عن موسى بن
