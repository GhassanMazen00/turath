٨٣ حديثًا
٧٠٤٨أنا على، حوضي أنتظر من يرد على، فيؤخذ بناس من دوني فأقول أمتي . فيقول…٧٠٤٩أنا فرطكم على الحوض، ليرفعن إلى رجال منكم حتى إذا أهويت لأناولهم…٧٠٥٠أنا فرطكم، على الحوض، من ورده شرب منه، ومن شرب منه لم يظمأ بعده أبدا،…٧٠٥١إنهم مني . فيقال إنك لا تدري ما بدلوا بعدك فأقول سحقا سحقا لمن بدل…٧٠٥٢إنكم سترون بعدي أثرة وأمورا تنكرونها٧٠٥٣من كره من أميره شيئا فليصبر، فإنه من خرج من السلطان شبرا مات ميتة…٧٠٥٤من رأى من أميره شيئا يكرهه فليصبر عليه، فإنه من فارق الجماعة شبرا…٧٠٥٦وعسرنا، ويسرنا، وأثرة علينا، وأن لا ننازع الأمر أهله، إلا أن تروا…٧٠٥٧إنكم سترون بعدي أثرة، فاصبروا حتى تلقوني٧٠٥٨هلكة أمتي على يدى غلمة من قريش٧٠٥٩لا إله إلا الله، ويل للعرب من شر قد اقترب، فتح اليوم من ردم ياجوج…٧٠٦٠حدثنا أبو نعيم، حدثنا ابن عيينة، عن الزهري، . وحدثني محمود، أخبرنا…٧٠٦١يتقارب الزمان، وينقص العمل، ويلقى الشح، وتظهر الفتن، ويكثر الهرج٧٠٦٣إن بين يدى الساعة لأياما ينزل فيها الجهل، ويرفع فيها العلم، ويكثر…٧٠٦٤إن بين يدى الساعة أياما يرفع فيها العلم، وينزل فيها الجهل، ويكثر فيها…٧٠٦٥والهرج بلسان الحبشة القتل .٧٠٦٦بين يدى الساعة أيام الهرج، يزول العلم، ويظهر فيها الجهل٧٠٦٧من شرار الناس من تدركهم الساعة وهم أحياء٧٠٦٨اصبروا، فإنه لا يأتي عليكم زمان إلا الذي بعده شر منه، حتى تلقوا ربكم٧٠٦٩سبحان الله ماذا أنزل الله من الخزائن وماذا أنزل من الفتن، من يوقظ…٧٠٧٠من حمل علينا السلاح فليس منا٧٠٧١من حمل علينا السلاح فليس منا٧٠٧٢لا يشير أحدكم على أخيه بالسلاح، فإنه لا يدري لعل الشيطان ينزع في يده،…٧٠٧٣يقول مر رجل بسهام في المسجد فقال له رسول الله صلى الله عليه وسلم "…٧٠٧٤أن رجلا، مر في المسجد بأسهم قد أبدى نصولها، فأمر أن يأخذ بنصولها، لا…٧٠٧٥إذا مر أحدكم في مسجدنا أو في سوقنا ومعه نبل فليمسك على نصالها أو قال…٧٠٧٦سباب المسلم فسوق، وقتاله كفر٧٠٧٧لا ترجعوا بعدي كفارا، يضرب بعضكم رقاب بعض٧٠٧٨ألا تدرون أى يوم هذا٧٠٧٩لا ترتدوا بعدي كفارا، يضرب بعضكم رقاب بعض٧٠٨٠جرير قال قال لي رسول الله صلى الله عليه وسلم في حجة الوداع " استنصت…٧٠٨١ستكون فتن القاعد فيها خير من القائم، والقائم فيها خير من الماشي،…٧٠٨٢ستكون فتن القاعد فيها خير من القائم، والقائم خير من الماشي، والماشي…٧٠٨٣إذا تواجه المسلمان بسيفيهما فكلاهما من أهل النار٧٠٨٤أنه سمع أبا إدريس الخولاني، أنه سمع حذيفة بن اليمان، يقول كان الناس…٧٠٨٥وغيره، قالا حدثنا أبو الأسود، . وقال الليث عن أبي الأسود، قال قطع على…٧٠٨٦أن الأمانة نزلت في جذر قلوب الرجال، ثم علموا من القرآن، ثم علموا من…٧٠٨٧وتزوج هناك امرأة وولدت له أولادا، فلم يزل بها حتى قبل أن يموت بليال،…٧٠٨٨يوشك أن يكون خير مال المسلم غنم، يتبع بها شعف الجبال ومواقع القطر،…٧٠٨٩لا تسألوني عن شىء إلا بينت لكم٧٠٩٠أن أنسا، حدثهم أن نبي الله صلى الله عليه وسلم بهذا وقال كل رجل لافا…٧٠٩١ومعتمر، عن أبيه، عن قتادة، أن أنسا، حدثهم عن النبي صلى الله عليه وسلم…٧٠٩٢الفتنة ها هنا الفتنة ها هنا من حيث يطلع قرن الشيطان٧٠٩٣ألا إن الفتنة ها هنا من حيث يطلع قرن الشيطان٧٠٩٤اللهم بارك لنا في شأمنا، اللهم بارك لنا في يمننا٧٠٩٥إنما كان محمد صلى الله عليه وسلم يقاتل المشركين، وكان الدخول في دينهم…٧٠٩٦فتنة الرجل في أهله وماله وولده وجاره، تكفرها الصلاة والصدقة والأمر…٧٠٩٧ائذن له، وبشره بالجنة٧٠٩٨يجاء برجل فيطرح في النار، فيطحن فيها كطحن الحمار برحاه، فيطيف به أهل…٧٠٩٩لن يفلح قوم ولوا أمرهم امرأة٧١٠٠قال لما سار طلحة والزبير وعائشة إلى البصرة بعث علي عمار بن ياسر وحسن…٧١٠١قام عمار على منبر الكوفة، فذكر عائشة وذكر مسيرها وقال إنها زوجة نبيكم…٧١٠٤ثم راحوا إلى المسجد .٧١٠٧كنت جالسا مع أبي مسعود وأبي موسى وعمار فقال أبو مسعود ما من أصحابك…٧١٠٨إذا أنزل الله بقوم عذابا، أصاب العذاب من كان فيهم، ثم بعثوا على…٧١٠٩ابني هذا سيد ولعل الله أن يصلح به بين فئتين من المسلمين٧١١٠أن حرملة، مولى أسامة أخبره قال عمرو وقد رأيت حرملة قال أرسلني أسامة…٧١١١ينصب لكل غادر لواء يوم القيامة٧١١٢قال لما كان ابن زياد ومروان بالشأم، ووثب ابن الزبير بمكة، ووثب القراء…٧١١٣قال إن المنافقين اليوم شر منهم على عهد النبي صلى الله عليه وسلم كانوا…٧١١٤قال إنما كان النفاق على عهد النبي صلى الله عليه وسلم فأما اليوم فإنما…٧١١٥لا تقوم الساعة حتى يمر الرجل بقبر الرجل فيقول يا ليتني مكانه٧١١٦لا تقوم الساعة حتى تضطرب أليات نساء دوس على ذي الخلصة٧١١٧لا تقوم الساعة حتى يخرج رجل من قحطان يسوق الناس بعصاه٧١١٨لا تقوم الساعة حتى تخرج نار من أرض الحجاز، تضيء أعناق الإبل ببصرى٧١١٩يوشك الفرات أن يحسر عن كنز من ذهب، فمن حضره فلا يأخذ منه شيئا٧١٢٠تصدقوا، فسيأتي على الناس زمان يمشي الرجل بصدقته، فلا يجد من يقبلها٧١٢١لا تقوم الساعة حتى تقتتل فئتان عظيمتان، يكون بينهما مقتلة عظيمة،…٧١٢٢حدثنا مسدد، حدثنا يحيى، حدثنا إسماعيل، حدثني قيس، قال قال لي المغيرة…٧١٢٣أعور عين اليمنى، كأنها عنبة طافية٧١٢٤يجيء الدجال حتى ينزل في ناحية المدينة، ثم ترجف المدينة ثلاث رجفات،…٧١٢٥لا يدخل المدينة رعب المسيح الدجال، ولها يومئذ سبعة أبواب، على كل باب…٧١٢٦لا يدخل المدينة رعب المسيح، لها يومئذ سبعة أبواب، على كل باب ملكان٧١٢٧إني لأنذركموه، وما من نبي إلا وقد أنذره قومه، ولكني سأقول لكم فيه…٧١٢٨بينا أنا نائم أطوف بالكعبة، فإذا رجل آدم سبط الشعر ينطف أو يهراق رأسه…٧١٢٩حدثنا عبد العزيز بن عبد الله، حدثنا إبراهيم بن سعد، عن صالح، عن ابن…٧١٣٠إن معه ماء ونارا، فناره ماء بارد، وماؤه نار٧١٣١ما بعث نبي إلا أنذر أمته الأعور الكذاب، ألا إنه أعور، وإن ربكم ليس…٧١٣٢يأتي الدجال وهو محرم عليه أن يدخل نقاب المدينة، فينزل بعض السباخ التي…٧١٣٣على أنقاب المدينة ملائكة، لا يدخلها الطاعون ولا الدجال٧١٣٤المدينة يأتيها الدجال، فيجد الملائكة يحرسونها، فلا يقربها الدجال قال…٧١٣٥لا إله إلا الله، ويل للعرب من شر قد اقترب، فتح اليوم من ردم يأجوج…٧١٣٦يفتح الردم ردم يأجوج ومأجوج مثل هذه
