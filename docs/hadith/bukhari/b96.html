كتاب الاعتصام بالكتاب والسنة

٩٧ حديثًا
٧٢٦٨وغيره، عن قيس بن مسلم، عن طارق بن شهاب، قال قال رجل من اليهود لعمر يا…٧٢٦٩أنه سمع عمر الغد، حين بايع المسلمون أبا بكر، واستوى على منبر رسول…٧٢٧٠اللهم علمه الكتاب٧٢٧١أن أبا المنهال، حدثه أنه، سمع أبا برزة، قال إن الله يغنيكم أو نغشكم…٧٢٧٢أن عبد الله بن عمر، كتب إلى عبد الملك بن مروان يبايعه، وأقر لك بالسمع…٧٢٧٣بعثت بجوامع الكلم، ونصرت بالرعب، وبينا أنا نائم رأيتني أتيت بمفاتيح…٧٢٧٤ما من الأنبياء نبي إلا أعطي من الآيات ما مثله أومن أو آمن عليه البشر،…٧٢٧٥قال جلست إلى شيبة في هذا المسجد قال جلس إلى عمر في مجلسك هذا فقال…٧٢٧٦أن الأمانة نزلت من السماء في جذر قلوب الرجال، ونزل القرآن فقرءوا…٧٢٧٧يقول قال عبد الله إن أحسن الحديث كتاب الله، وأحسن الهدى هدى محمد صلى…٧٢٧٩لأقضين بينكما بكتاب الله٧٢٨٠كل أمتي يدخلون الجنة، إلا من أبى٧٢٨١يقول جاءت ملائكة إلى النبي صلى الله عليه وسلم وهو نائم فقال بعضهم إنه…٧٢٨٢لقد ضللتم ضلالا بعيدا .٧٢٨٣إنما مثلي ومثل ما بعثني الله به كمثل رجل أتى قوما فقال يا قوم إني…٧٢٨٥أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله . فمن قال لا إله إلا…٧٢٨٦أن عبد الله بن عباس رضى الله عنهما قال قدم عيينة بن حصن بن حذيفة بن…٧٢٨٧ما من شىء لم أره إلا وقد رأيته في مقامي، حتى الجنة والنار، وأوحي إلى…٧٢٨٨دعوني ما تركتكم، إنما هلك من كان قبلكم بسؤالهم واختلافهم على…٧٢٨٩إن أعظم المسلمين جرما من سأل عن شىء لم يحرم، فحرم من أجل مسألته٧٢٩٠ما زال بكم الذي رأيت من صنيعكم، حتى خشيت أن يكتب عليكم، ولو كتب عليكم…٧٢٩١حدثنا يوسف بن موسى، حدثنا أبو أسامة، عن بريد بن أبي بردة، عن أبي…٧٢٩٢لا إله إلا الله، وحده لا شريك له، له الملك وله الحمد، وهو على كل شىء…٧٢٩٣حدثنا سليمان بن حرب، حدثنا حماد بن زيد، عن ثابت، عن أنس، قال كنا عند…٧٢٩٤من أحب أن يسأل عن شىء فليسأل عنه، فوالله لا تسألوني عن شىء إلا…٧٢٩٥حدثنا محمد بن عبد الرحيم، أخبرنا روح بن عبادة، حدثنا شعبة، أخبرني…٧٢٩٦لن يبرح الناس يتساءلون حتى يقولوا هذا الله خالق كل شىء فمن خلق الله٧٢٩٧وهو يتوكأ على عسيب، فمر بنفر من اليهود فقال بعضهم سلوه عن الروح .…٧٢٩٨إني اتخذت خاتما من ذهب٧٢٩٩قال قال النبي صلى الله عليه وسلم " لا تواصلوا " . قالوا إنك تواصل .…٧٣٠٠المدينة حرم من عير إلى كذا، فمن أحدث فيها حدثا فعليه لعنة الله…٧٣٠١ما بال أقوام يتنزهون عن الشىء أصنعه، فوالله إني أعلمهم بالله، وأشدهم…٧٣٠٢قال كاد الخيران أن يهلكا أبو بكر، وعمر، لما قدم على النبي صلى الله…٧٣٠٣مروا أبا بكر يصلي بالناس٧٣٠٤قد أنزل الله فيكم قرآنا٧٣٠٥لا نورث ما تركنا صدقة٧٣٠٦قال قلت لأنس أحرم رسول الله صلى الله عليه وسلم المدينة . قال نعم ما…٧٣٠٧إن الله لا ينزع العلم بعد أن أعطاهموه انتزاعا، ولكن ينتزعه منهم مع…٧٣٠٨قال قال سهل بن حنيف يا أيها الناس اتهموا رأيكم على دينكم، لقد رأيتني…٧٣٠٩يقول مرضت فجاءني رسول الله صلى الله عليه وسلم يعودني وأبو بكر وهما…٧٣١٠اجتمعن في يوم كذا وكذا في مكان كذا وكذا٧٣١١لا يزال طائفة من أمتي ظاهرين حتى يأتيهم أمر الله وهم ظاهرون٧٣١٢من يرد الله به خيرا يفقهه في الدين، وإنما أنا قاسم ويعطي الله، ولن…٧٣١٣حدثنا علي بن عبد الله، حدثنا سفيان، قال عمرو سمعت جابر بن عبد الله…٧٣١٤أن أعرابيا، أتى رسول الله صلى الله عليه وسلم فقال إن امرأتي ولدت…٧٣١٥نعم حجي عنها، أرأيت لو كان على أمك دين أكنت قاضيته٧٣١٦لا حسد إلا في اثنتين رجل آتاه الله مالا فسلط على هلكته في الحق، وآخر…٧٣١٨فيه غرة عبد أو أمة٧٣١٩لا تقوم الساعة حتى تأخذ أمتي بأخذ القرون قبلها، شبرا بشبر وذراعا بذراع٧٣٢٠لتتبعن سنن من كان قبلكم شبرا شبرا وذراعا بذراع، حتى لو دخلوا جحر ضب…٧٣٢١ليس من نفس تقتل ظلما إلا كان على ابن آدم الأول كفل منها وربما قال…٧٣٢٢إنما المدينة كالكير، تنفي خبثها، وينصع طيبها٧٣٢٣فلما كان آخر حجة حجها عمر فقال عبد الرحمن بمنى، لو شهدت أمير المؤمنين…٧٣٢٤لقد رأيتني وإني لأخر فيما بين منبر رسول الله صلى الله عليه وسلم إلى…٧٣٢٥قال سئل ابن عباس أشهدت العيد مع النبي صلى الله عليه وسلم قال نعم…٧٣٢٦حدثنا أبو نعيم، حدثنا سفيان، عن عبد الله بن دينار، عن ابن عمر رضى…٧٣٢٨قالت لعبد الله بن الزبير ادفني مع صواحبي ولا تدفني مع النبي صلى الله…٧٣٢٩وبعد العوالي أربعة أميال أو ثلاثة .٧٣٣٠يقول كان الصاع على عهد النبي صلى الله عليه وسلم مدا وثلثا بمدكم…٧٣٣١اللهم بارك لهم في مكيالهم، وبارك لهم في صاعهم ومدهم٧٣٣٢أن اليهود، جاءوا إلى النبي صلى الله عليه وسلم برجل وامرأة زنيا، فأمر…٧٣٣٣هذا جبل يحبنا ونحبه، اللهم إن إبراهيم حرم مكة، وإني أحرم ما بين…٧٣٣٤أنه كان بين جدار المسجد مما يلي القبلة وبين المنبر ممر الشاة .٧٣٣٥ما بين بيتي ومنبري روضة من رياض الجنة، ومنبري على حوضي٧٣٣٦قال سابق النبي صلى الله عليه وسلم بين الخيل، فأرسلت التي ضمرت منها…٧٣٣٧وابن، إدريس وابن أبي غنية عن أبي حيان، عن الشعبي، عن ابن عمر رضى الله…٧٣٣٨سمع عثمان بن عفان، خطبنا على منبر النبي صلى الله عليه وسلم .٧٣٣٩أن هشام بن عروة، حدثه عن أبيه، أن عائشة، قالت كان يوضع لي ولرسول الله…٧٣٤١قال حالف النبي صلى الله عليه وسلم بين الأنصار وقريش في داري التي…٧٣٤٢قال قدمت المدينة فلقيني عبد الله بن سلام فقال لي انطلق إلى المنزل…٧٣٤٣أتاني الليلة آت من ربي وهو بالعقيق أن صل في هذا الوادي المبارك وقل…٧٣٤٤ولأهل اليمن يلملم٧٣٤٥حدثنا عبد الرحمن بن المبارك، حدثنا الفضيل، حدثنا موسى بن عقبة، حدثني…٧٣٤٦اللهم ربنا ولك الحمد في الأخيرة٧٣٤٧فانصرف رسول الله صلى الله عليه وسلم حين قال له ذلك ولم يرجع إليه…٧٣٤٨انطلقوا إلى يهود٧٣٤٩يجاء بنوح يوم القيامة فيقال له هل بلغت فيقول نعم يا رب . فتسأل أمته…٧٣٥١أكل تمر خيبر هكذا٧٣٥٢إذا حكم الحاكم فاجتهد ثم أصاب فله أجران، وإذا حكم فاجتهد ثم أخطأ فله…٧٣٥٣قال استأذن أبو موسى على عمر فكأنه وجده مشغولا فرجع، فقال عمر ألم أسمع…٧٣٥٤من يبسط رداءه حتى أقضي مقالتي ثم يقبضه، فلن ينسى شيئا سمعه مني٧٣٥٥قال رأيت جابر بن عبد الله يحلف بالله أن ابن الصائد الدجال، قلت تحلف…٧٣٥٦الخيل لثلاثة لرجل أجر، ولرجل ستر، وعلى رجل وزر، فأما الذي له أجر فرجل…٧٣٥٧تأخذين فرصة ممسكة فتوضئين بها٧٣٥٨أن أم حفيد بنت الحارث بن حزن، أهدت إلى النبي صلى الله عليه وسلم سمنا…٧٣٥٩من أكل ثوما أو بصلا، فليعتزلنا أو ليعتزل مسجدنا، وليقعد في بيته٧٣٦٠إن لم تجديني فأتي أبا بكر٧٣٦١سمع معاوية، يحدث رهطا من قريش بالمدينة، وذكر كعب الأحبار فقال إن كان…٧٣٦٢لا تصدقوا أهل الكتاب، ولا تكذبوهم وقولوا { آمنا بالله وما أنزل إلينا…٧٣٦٣وكتابكم الذي أنزل على رسول الله صلى الله عليه وسلم أحدث، تقرءونه محضا…٧٣٦٤اقرءوا القرآن ما ائتلفت قلوبكم فإذا اختلفتم فقوموا عنه٧٣٦٥اقرءوا القرآن ما ائتلفت عليه قلوبكم، فإذا اختلفتم فقوموا عنه٧٣٦٦هلم أكتب لكم كتابا لن تضلوا
