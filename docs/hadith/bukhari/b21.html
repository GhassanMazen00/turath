كتاب العمل فى الصلاة

٢٧ حديثًا
١١٩٨واضطجع رسول الله صلى الله عليه وسلم وأهله في طولها، فنام رسول الله…١١٩٩إن في الصلاة شغلا١١٩٩.٢حدثنا ابن نمير، حدثنا إسحاق بن منصور، حدثنا هريم بن سفيان، عن الأعمش،…١٢٠٠حدثنا إبراهيم بن موسى، أخبرنا عيسى هو ابن يونس عن إسماعيل، عن الحارث…١٢٠١وحانت الصلاة، فجاء بلال أبا بكر رضى الله عنهما فقال حبس النبي صلى…١٢٠٢قولوا التحيات لله والصلوات والطيبات، السلام عليك أيها النبي ورحمة…١٢٠٣التسبيح للرجال والتصفيق للنساء١٢٠٤التسبيح للرجال والتصفيح للنساء١٢٠٥أن المسلمين، بينا هم في الفجر يوم الاثنين، وأبو بكر رضى الله عنه يصلي…١٢٠٦نادت امرأة ابنها، وهو في صومعة قالت يا جريج . قال اللهم أمي وصلاتي .…١٢٠٧إن كنت فاعلا فواحدة١٢٠٨حدثنا مسدد، حدثنا بشر، حدثنا غالب، عن بكر بن عبد الله، عن أنس بن مالك…١٢٠٩فإذا سجد غمزني فرفعتها، فإذا قام مددتها .١٢١٠إن الشيطان عرض لي، فشد على ليقطع الصلاة على، فأمكنني الله منه، فذعته،…١٢١١وإذا لجام دابته بيده فجعلت الدابة تنازعه، وجعل يتبعها قال شعبة هو أبو…١٢١٢إنهما آيتان من آيات الله، فإذا رأيتم ذلك فصلوا حتى يفرج عنكم، لقد…١٢١٣إن الله قبل أحدكم، فإذا كان في صلاته، فلا يبزقن أو قال لا يتنخمن١٢١٤إذا كان في الصلاة فإنه يناجي ربه، فلا يبزقن بين يديه ولا عن يمينه،…١٢١٥لا ترفعن رءوسكن حتى يستوي الرجال جلوسا١٢١٦إن في الصلاة شغلا١٢١٧إنما منعني أن أرد عليك أني كنت أصلي١٢١٨يا أيها الناس ما لكم حين نابكم شىء في الصلاة أخذتم بالتصفيح إنما…١٢١٩حدثنا أبو النعمان، حدثنا حماد، عن أيوب، عن محمد، عن أبي هريرة رضى…١٢٢٠رضى الله عنه قال نهي أن يصلي الرجل مختصرا .١٢٢١ذكرت وأنا في الصلاة تبرا عندنا، فكرهت أن يمسي أو يبيت عندنا فأمرت…١٢٢٢إذا أذن بالصلاة أدبر الشيطان له ضراط حتى لا يسمع التأذين، فإذا سكت…١٢٢٣قال قال أبو هريرة رضى الله عنه يقول الناس أكثر أبو هريرة، فلقيت رجلا…
