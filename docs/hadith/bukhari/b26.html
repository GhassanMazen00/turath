كتاب العمرة

٣٠ حديثًا
١٧٧٣العمرة إلى العمرة كفارة لما بينهما، والحج المبرور ليس له جزاء إلا…١٧٧٤أن عكرمة بن خالد، سأل ابن عمر رضى الله عنهما عن العمرة، قبل الحج فقال…١٧٧٦، فإذا عبد الله بن عمر رضى الله عنهما جالس إلى حجرة عائشة، وإذا ناس…١٧٧٧قال سألت عائشة رضى الله عنها قالت ما اعتمر رسول الله صلى الله عليه…١٧٧٨سألت أنسا رضى الله عنه كم اعتمر النبي صلى الله عليه وسلم قال أربع…١٧٧٩قال سألت أنسا رضى الله عنه فقال اعتمر النبي صلى الله عليه وسلم حيث…١٧٨٠وقال، اعتمر أربع عمر في ذي القعدة إلا التي اعتمر مع حجته عمرته من…١٧٨١قبل أن يحج مرتين .١٧٨٢ما منعك أن تحجي معنا١٧٨٣من أحب منكم أن يهل بالحج فليهل ومن أحب أن يهل بعمرة فليهل بعمرة،…١٧٨٤سمع عمرو بن أوس، أن عبد الرحمن بن أبي بكر رضى الله عنهما أخبره أن…١٧٨٥لو استقبلت من أمري ما استدبرت ما أهديت، ولولا أن معي الهدى لأحللت١٧٨٦من أحب أن يهل بعمرة فليهل، ومن أحب أن يهل بحجة فليهل، ولولا أني أهديت…١٧٨٧انتظري، فإذا طهرت فاخرجي إلى التنعيم، فأهلي ثم ائتينا بمكان كذا،…١٧٨٨من لم يكن معه هدى، فأحب أن يجعلها عمرة، فليفعل ومن كان معه هدى فلا١٧٨٩أين السائل عن العمرة اخلع عنك الجبة واغسل أثر الخلوق عنك، وأنق…١٧٩٠لو كانت كما تقول كانت فلا جناح عليه أن لا يطوف بهما . إنما أنزلت هذه…١٧٩٢بشروا خديجة ببيت في الجنة من قصب لا صخب فيه ولا نصب١٧٩٤طاف بالبيت في عمرة، ولم يطف بين الصفا والمروة، أيأتي امرأته فقال قدم…١٧٩٥وبالصفا والمروة، ثم أتيت امرأة من قيس، ففلت رأسي، ثم أهللت بالحج .…١٧٩٦أن عبد الله، مولى أسماء بنت أبي بكر حدثه أنه، كان يسمع أسماء تقول…١٧٩٧لا إله إلا الله وحده لا شريك له، له الملك، وله الحمد، وهو على كل شىء…١٧٩٨فحمل واحدا بين يديه وآخر خلفه .١٧٩٩وإذا رجع صلى بذي الحليفة ببطن الوادي وبات حتى يصبح .١٨٠٠كان لا يدخل إلا غدوة أو عشية .١٨٠١حدثنا مسلم بن إبراهيم، حدثنا شعبة، عن محارب، عن جابر رضى الله عنه قال…١٨٠٢أنه سمع أنسا رضى الله عنه يقول كان رسول الله صلى الله عليه وسلم إذا…١٨٠٣كانت الأنصار إذا حجوا فجاءوا لم يدخلوا من قبل أبواب بيوتهم، ولكن من…١٨٠٤السفر قطعة من العذاب، يمنع أحدكم طعامه وشرابه ونومه، فإذا قضى نهمته…١٨٠٥قال كنت مع عبد الله بن عمر رضى الله عنهما بطريق مكة، فبلغه عن صفية…
