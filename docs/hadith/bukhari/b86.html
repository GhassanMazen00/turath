٨١ حديثًا
٦٧٧٢لا يزني الزاني حين يزني وهو مؤمن، ولا يشرب الخمر حين يشرب وهو مؤمن،…٦٧٧٣وجلد أبو بكر أربعين .٦٧٧٤قال جيء بالنعيمان أو بابن النعيمان شاربا، فأمر النبي صلى الله عليه…٦٧٧٥أن النبي صلى الله عليه وسلم أتي بنعيمان أو بابن نعيمان وهو سكران فشق…٦٧٧٦قال جلد النبي صلى الله عليه وسلم في الخمر بالجريد والنعال، وجلد أبو…٦٧٧٧والضارب بنعله، والضارب بثوبه، فلما انصرف قال بعض القوم أخزاك الله .…٦٧٧٨فأجد في نفسي، إلا صاحب الخمر، فإنه لو مات وديته، وذلك أن رسول الله…٦٧٧٩حتى كان آخر إمرة عمر، فجلد أربعين، حتى إذا عتوا وفسقوا جلد ثمانين .٦٧٨٠لا تلعنوه، فوالله ما علمت أنه يحب الله ورسوله٦٧٨١لا تكونوا عون الشيطان على أخيكم٦٧٨٢لا يزني الزاني حين يزني وهو مؤمن، ولا يسرق حين يسرق وهو مؤمن٦٧٨٣لعن الله السارق، يسرق البيضة فتقطع يده، ويسرق الحبل فتقطع يده٦٧٨٤بايعوني على أن لا تشركوا بالله شيئا ولا تسرقوا، ولا تزنوا٦٧٨٥ألا أى شهر تعلمونه أعظم حرمة٦٧٨٦ما لم يأثم، فإذا كان الإثم كان أبعدهما منه، والله ما انتقم لنفسه في…٦٧٨٧إنما هلك من كان قبلكم أنهم كانوا يقيمون الحد على الوضيع، ويتركون…٦٧٨٨أتشفع في حد من حدود الله٦٧٨٩تقطع اليد في ربع دينار فصاعدا٦٧٩٠تقطع يد السارق في ربع دينار٦٧٩١يقطع في ربع دينار٦٧٩٢قال أخبرتني عائشة، أن يد السارق، لم تقطع على عهد النبي صلى الله عليه…٦٧٩٣قالت لم تكن تقطع يد السارق في أدنى من حجفة أو ترس، كل واحد منهما ذو…٦٧٩٤ترس أو حجفة، وكان كل واحد منهما ذا ثمن .٦٧٩٥حدثنا إسماعيل، حدثني مالك بن أنس، عن نافع، مولى عبد الله بن عمر عن…٦٧٩٦قال قطع النبي صلى الله عليه وسلم في مجن ثمنه ثلاثة دراهم .٦٧٩٧قال قطع النبي صلى الله عليه وسلم في مجن ثمنه ثلاثة دراهم .٦٧٩٨أن عبد الله بن عمر رضى الله عنهما قال قطع النبي صلى الله عليه وسلم يد…٦٧٩٩لعن الله السارق، يسرق البيضة فتقطع يده، ويسرق الحبل فتقطع يده٦٨٠٠أن النبي صلى الله عليه وسلم قطع يد امرأة . قالت عائشة وكانت تأتي بعد…٦٨٠١أبايعكم على أن لا تشركوا بالله شيئا، ولا تسرقوا، ولا تقتلوا أولادكم،…٦٨٠٢فأسلموا فاجتووا المدينة، فأمرهم أن يأتوا إبل الصدقة، فيشربوا من…٦٨٠٣أن النبي صلى الله عليه وسلم قطع العرنيين ولم يحسمهم حتى ماتوا .٦٨٠٤ما أجد لكم إلا أن تلحقوا بإبل رسول الله صلى الله عليه وسلم٦٨٠٥أن رهطا، من عكل أو قال عرينة ولا أعلمه إلا قال من عكل قدموا المدينة،…٦٨٠٦سبعة يظلهم الله يوم القيامة في ظله، يوم لا ظل إلا ظله إمام عادل، وشاب…٦٨٠٧من توكل لي ما بين رجليه وما بين لحييه، توكلت له بالجنة٦٨٠٨لا تقوم الساعة وإما قال من أشراط الساعة أن يرفع العلم ويظهر الجهل،…٦٨٠٩لا يزني العبد حين يزني وهو مؤمن، ولا يسرق حين يسرق وهو مؤمن، ولا يشرب…٦٨١٠لا يزني الزاني حين يزني وهو مؤمن، ولا يسرق حين يسرق وهو مؤمن، ولا…٦٨١١أن تجعل لله ندا وهو خلقك٦٨١٢رضى الله عنه حين رجم المرأة يوم الجمعة وقال قد رجمتها بسنة رسول الله…٦٨١٣سألت عبد الله بن أبي أوفى هل رجم رسول الله صلى الله عليه وسلم قال نعم…٦٨١٤أن رجلا، من أسلم أتى رسول الله صلى الله عليه وسلم فحدثه أنه قد زنى،…٦٨١٦وسعيد بن المسيب، عن أبي هريرة رضى الله عنه قال أتى رجل رسول الله صلى…٦٨١٧هو لك يا عبد بن زمعة، الولد للفراش، واحتجبي منه يا سودة٦٨١٨الولد للفراش، وللعاهر الحجر٦٨١٩ما تجدون في كتابكم٦٨٢٠أن رجلا، من أسلم جاء النبي صلى الله عليه وسلم فاعترف بالزنا فأعرض عنه…٦٨٢١وقع بامرأته في رمضان، فاستفتى رسول الله صلى الله عليه وسلم فقال " هل…٦٨٢٢وقال الليث عن عمرو بن الحارث، عن عبد الرحمن بن القاسم، عن محمد بن…٦٨٢٣أليس قد صليت معنا٦٨٢٤لعلك قبلت أو غمزت أو نظرت٦٨٢٥وأبي، سلمة أن أبا هريرة، قال أتى رسول الله صلى الله عليه وسلم رجل من…٦٨٢٨فزنى بامرأته، فافتديت منه بمائة شاة وخادم، ثم سألت رجالا من أهل…٦٨٢٩ألا وإن الرجم حق على من زنى، وقد أحصن، إذا قامت البينة، أو كان الحمل…٦٨٣٠لا تطروني كما أطري عيسى ابن مريم وقولوا عبد الله ورسوله٦٨٣٢أن عمر بن الخطاب، غرب، ثم لم تزل تلك السنة .٦٨٣٣حدثنا يحيى بن بكير، حدثنا الليث، عن عقيل، عن ابن شهاب، عن سعيد بن…٦٨٣٤أخرجوهم من بيوتكم٦٨٣٦والذي نفسي بيده لأقضين بينكما بكتاب الله، أما الغنم والوليدة فرد…٦٨٣٨إذا زنت فاجلدوها، ثم إن زنت فاجلدوها، ثم إن زنت فاجلدوها، ثم بيعوها…٦٨٣٩إذا زنت الأمة فتبين زناها فليجلدها ولا يثرب، ثم إن زنت فليجلدها ولا…٦٨٤٠حدثنا موسى بن إسماعيل، حدثنا عبد الواحد، حدثنا الشيباني، سألت عبد…٦٨٤١ما تجدون في التوراة في شأن الرجم٦٨٤٣وزيد بن خالد، أنهما أخبراه أن رجلين اختصما إلى رسول الله صلى الله…٦٨٤٤قالت جاء أبو بكر رضى الله عنه ورسول الله صلى الله عليه وسلم واضع رأسه…٦٨٤٥أن عبد الرحمن بن القاسم، حدثه عن أبيه، عن عائشة، قالت أقبل أبو بكر…٦٨٤٦أتعجبون من غيرة سعد، لأنا أغير منه، والله أغير مني٦٨٤٧حدثنا إسماعيل، حدثني مالك، عن ابن شهاب، عن سعيد بن المسيب، عن أبي…٦٨٤٨لا يجلد فوق عشر جلدات إلا في حد من حدود الله٦٨٤٩لا عقوبة فوق عشر ضربات إلا في حد من حدود الله٦٨٥٠لا تجلدوا فوق عشرة أسواط إلا في حد من حدود الله٦٨٥١أيكم مثلي إني أبيت يطعمني ربي ويسقين٦٨٥٢أنهم كانوا يضربون على عهد رسول الله صلى الله عليه وسلم إذا اشتروا…٦٨٥٣حدثنا عبدان، أخبرنا عبد الله، أخبرنا يونس، عن الزهري، أخبرني عروة، عن…٦٨٥٤إن جاءت به كذا وكذا فهو، وإن جاءت به كذا وكذا كأنه وحرة فهو٦٨٥٥لو كنت راجما امرأة عن غير بينة٦٨٥٦ثم انصرف وأتاه رجل من قومه يشكو أنه وجد مع أهله فقال عاصم ما ابتليت…٦٨٥٧اجتنبوا السبع الموبقات٦٨٥٨من قذف مملوكه وهو بريء مما قال، جلد يوم القيامة، إلا أن يكون كما قال٦٨٦٠وزيد بن خالد الجهني، قالا جاء رجل إلى النبي صلى الله عليه وسلم فقال…
