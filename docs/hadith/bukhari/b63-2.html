كتاب مناقب الأنصار

١٧٢ حديثًا•صفحة ٢ من ٢
٣٩٢٦اللهم حبب إلينا المدينة كحبنا مكة أو أشد، وصححها وبارك لنا في صاعها…٣٩٢٧أن عبيد الله بن عدي، أخبره دخلت، على عثمان . وقال بشر بن شعيب حدثني…٣٩٢٨أن ابن عباس، أخبره أن عبد الرحمن بن عوف رجع إلى أهله وهو بمنى، في آخر…٣٩٢٩وما يدريك أن الله أكرمه٣٩٣٠فقدم رسول الله صلى الله عليه وسلم المدينة وقد افترق ملؤهم، وقتلت…٣٩٣١دعهما يا أبا بكر، إن لكل قوم عيدا، وإن عيدنا هذا اليوم٣٩٣٢يا بني النجار، ثامنوني حائطكم هذا٣٩٣٣ثلاث للمهاجر بعد الصدر٣٩٣٤قال ما عدوا من مبعث النبي صلى الله عليه وسلم ولا من وفاته، ما عدوا…٣٩٣٥ثم هاجر النبي صلى الله عليه وسلم ففرضت أربعا، وتركت صلاة السفر على…٣٩٣٦قال عادني النبي صلى الله عليه وسلم عام حجة الوداع من مرض أشفيت منه…٣٩٣٧مهيم يا عبد الرحمن٣٩٣٨أخبرني به جبريل آنفا٣٩٤٠ما كان يدا بيد فليس به بأس، وما كان نسيئة فلا يصلح٣٩٤١لو آمن بي عشرة من اليهود لآمن بي اليهود٣٩٤٢نحن أحق بصومه٣٩٤٣نحن أولى بموسى منكم٣٩٤٤وكان المشركون يفرقون رءوسهم، وكان أهل الكتاب يسدلون رءوسهم، وكان…٣٩٤٥جزءوه أجزاء، فآمنوا ببعضه وكفروا ببعضه . { يعني قول الله تعالى {…٣٩٤٦أنه تداوله بضعة عشر من رب إلى رب .٣٩٤٧رام هرمز .٣٩٤٨قال فترة بين عيسى ومحمد صلى الله عليه وسلم ستمائة سنة .
