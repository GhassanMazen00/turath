كتاب مواقيت الصلاة

٧٧ حديثًا
٥٢٢أن عمر بن عبد العزيز، أخر الصلاة يوما، فدخل عليه عروة بن الزبير،…٥٢٣آمركم بأربع، وأنهاكم عن أربع الإيمان بالله ثم فسرها لهم شهادة أن لا…٥٢٤قال بايعت رسول الله صلى الله عليه وسلم على إقام الصلاة، وإيتاء…٥٢٥فتنة الرجل في أهله وماله وولده وجاره تكفرها الصلاة والصوم والصدقة…٥٢٦لجميع أمتي كلهم٥٢٧الصلاة على وقتها٥٢٨أرأيتم لو أن نهرا بباب أحدكم، يغتسل فيه كل يوم خمسا، ما تقول ذلك يبقي…٥٢٩قال ما أعرف شيئا مما كان على عهد النبي صلى الله عليه وسلم . قيل…٥٣٠يقول دخلت على أنس بن مالك بدمشق وهو يبكي فقلت ما يبكيك فقال لا أعرف…٥٣١إن أحدكم إذا صلى يناجي ربه فلا يتفلن عن يمينه، ولكن تحت قدمه اليسرى٥٣٢اعتدلوا في السجود، ولا يبسط ذراعيه كالكلب، وإذا بزق فلا يبزقن بين…٥٣٤إذا اشتد الحر فأبردوا عن الصلاة، فإن شدة الحر من فيح جهنم٥٣٥أبرد أبرد أو قال انتظر انتظر٥٣٧إذا اشتد الحر فأبردوا بالصلاة، فإن شدة الحر من فيح جهنم٥٣٨أبردوا بالظهر، فإن شدة الحر من فيح جهنم٥٣٩فقال النبي صلى الله عليه وسلم " إن شدة الحر من فيح جهنم، فإذا اشتد…٥٤٠من أحب أن يسأل عن شىء فليسأل، فلا تسألوني عن شىء إلا أخبرتكم ما دمت…٥٤١ويقرأ فيها ما بين الستين إلى المائة، ويصلي الظهر إذا زالت الشمس،…٥٤٢حدثنا محمد يعني ابن مقاتل قال أخبرنا عبد الله، قال أخبرنا خالد بن عبد…٥٤٣أن النبي صلى الله عليه وسلم صلى بالمدينة سبعا وثمانيا الظهر والعصر،…٥٤٤أن عائشة، قالت كان رسول الله صلى الله عليه وسلم يصلي العصر والشمس لم…٥٤٥أن رسول الله صلى الله عليه وسلم صلى العصر والشمس في حجرتها، لم يظهر…٥٤٦قالت كان النبي صلى الله عليه وسلم يصلي صلاة العصر والشمس طالعة في…٥٤٧على أبي برزة الأسلمي، فقال له أبي كيف كان رسول الله صلى الله عليه…٥٤٨حدثنا عبد الله بن مسلمة، عن مالك، عن إسحاق بن عبد الله بن أبي طلحة،…٥٤٩ما هذه الصلاة التي صليت قال العصر، وهذه صلاة رسول الله صلى الله عليه…٥٥٠قال كان رسول الله صلى الله عليه وسلم يصلي العصر والشمس مرتفعة حية،…٥٥١فيأتيهم والشمس مرتفعة .٥٥٢الذي تفوته صلاة العصر كأنما وتر أهله وماله٥٥٣من ترك صلاة العصر فقد حبط عمله٥٥٤إنكم سترون ربكم كما ترون هذا القمر لا تضامون في رؤيته، فإن استطعتم أن…٥٥٥يتعاقبون فيكم ملائكة بالليل وملائكة بالنهار، ويجتمعون في صلاة الفجر…٥٥٦إذا أدرك أحدكم سجدة من صلاة العصر قبل أن تغرب الشمس فليتم صلاته، وإذا…٥٥٧إنما بقاؤكم فيما سلف قبلكم من الأمم كما بين صلاة العصر إلى غروب…٥٥٨مثل المسلمين واليهود والنصارى كمثل رجل استأجر قوما يعملون له عملا إلى…٥٥٩حدثنا محمد بن مهران، قال حدثنا الوليد، قال حدثنا الأوزاعي، قال حدثنا…٥٦٠والعصر والشمس نقية، والمغرب إذا وجبت، والعشاء أحيانا وأحيانا، إذا…٥٦١حدثنا المكي بن إبراهيم، قال حدثنا يزيد بن أبي عبيد، عن سلمة، قال كنا…٥٦٢قال صلى النبي صلى الله عليه وسلم سبعا جميعا وثمانيا جميعا .٥٦٣لا تغلبنكم الأعراب على اسم صلاتكم المغرب٥٦٤أرأيتم ليلتكم هذه فإن رأس مائة سنة منها لا يبقى ممن هو على ظهر الأرض…٥٦٥صلى الله عليه وسلم فقال كان يصلي الظهر بالهاجرة، والعصر والشمس حية،…٥٦٦ما ينتظرها أحد من أهل الأرض غيركم٥٦٧على رسلكم، أبشروا إن من نعمة الله عليكم أنه ليس أحد من الناس يصلي هذه…٥٦٨أن رسول الله صلى الله عليه وسلم كان يكره النوم قبل العشاء والحديث…٥٦٩ما ينتظرها أحد من أهل الأرض غيركم٥٧١ليس أحد من أهل الأرض ينتظر الصلاة غيركم٥٧٢قد صلى الناس وناموا، أما إنكم في صلاة ما انتظرتموها٥٧٣أما إنكم سترون ربكم كما ترون هذا، لا تضامون أو لا تضاهون في رؤيته،…٥٧٤من صلى البردين دخل الجنة٥٧٥أن زيد بن ثابت، حدثه أنهم، تسحروا مع النبي صلى الله عليه وسلم ثم…٥٧٦سمع روحا، حدثنا سعيد، عن قتادة، عن أنس بن مالك، أن نبي الله صلى الله…٥٧٧أنه سمع سهل بن سعد، يقول كنت أتسحر في أهلي ثم يكون سرعة بي أن أدرك…٥٧٨أن عائشة، أخبرته قالت، كن نساء المؤمنات يشهدن مع رسول الله صلى الله…٥٧٩من أدرك من الصبح ركعة قبل أن تطلع الشمس فقد أدرك الصبح، ومن أدرك ركعة…٥٨٠من أدرك ركعة من الصلاة فقد أدرك الصلاة٥٨١بهذا .٥٨٣لا تحروا بصلاتكم طلوع الشمس ولا غروبها٥٨٤وبعد العصر حتى تغرب الشمس، وعن اشتمال الصماء وعن الاحتباء في ثوب واحد…٥٨٥لا يتحرى أحدكم فيصلي عند طلوع الشمس ولا عند غروبها٥٨٦لا صلاة بعد الصبح حتى ترتفع الشمس، ولا صلاة بعد العصر حتى تغيب الشمس٥٨٧قال إنكم لتصلون صلاة، لقد صحبنا رسول الله صلى الله عليه وسلم فما…٥٨٨وبعد العصر حتى تغرب الشمس .٥٨٩قال أصلي كما رأيت أصحابي يصلون، لا أنهى أحدا يصلي بليل ولا نهار ما…٥٩٠سمع عائشة، قالت والذي ذهب به ما تركهما حتى لقي الله، وما لقي الله…٥٩١عائشة ابن أختي ما ترك النبي صلى الله عليه وسلم السجدتين بعد العصر…٥٩٢قالت ركعتان لم يكن رسول الله صلى الله عليه وسلم يدعهما سرا ولا علانية…٥٩٣قال رأيت الأسود ومسروقا شهدا على عائشة قالت ما كان النبي صلى الله…٥٩٤من ترك صلاة العصر حبط عمله٥٩٥أخاف أن تناموا عن الصلاة٥٩٦والله ما صليتها٥٩٧من نسي صلاة فليصل إذا ذكرها، لا كفارة لها إلا ذلك٥٩٨فصلى بعد ما غربت الشمس، ثم صلى المغرب .٥٩٩كان رسول الله صلى الله عليه وسلم يصلي المكتوبة قال كان يصلي الهجير…٦٠٠ألا إن الناس قد صلوا ثم رقدوا، وإنكم لم تزالوا في صلاة ما انتظرتم…٦٠١أرأيتكم ليلتكم هذه فإن رأس مائة لا يبقى ممن هو اليوم على ظهر الأرض أحد٦٠٢من كان عنده طعام اثنين فليذهب بثالث، وإن أربع فخامس أو سادس
