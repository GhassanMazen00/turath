٤١ حديثًا
٢٥١٧أيما رجل أعتق امرأ مسلما استنقذ الله بكل عضو منه عضوا منه من النار٢٥١٨إيمان بالله، وجهاد في سبيله٢٥١٩حدثنا موسى بن مسعود، حدثنا زائدة بن قدامة، عن هشام بن عروة، عن فاطمة…٢٥٢٠حدثنا محمد بن أبي بكر، حدثنا عثام، حدثنا هشام، عن فاطمة بنت المنذر،…٢٥٢١من أعتق عبدا بين اثنين، فإن كان موسرا قوم عليه ثم يعتق٢٥٢٢من أعتق شركا له في عبد، فكان له مال يبلغ ثمن العبد قوم العبد قيمة…٢٥٢٣من أعتق شركا له في مملوك فعليه عتقه كله، إن كان له مال يبلغ ثمنه، فإن…٢٥٢٤من أعتق نصيبا له في مملوك أو شركا له في عبد، وكان له من المال ما يبلغ…٢٥٢٥فيعتق أحدهم نصيبه منه، يقول قد وجب عليه عتقه كله، إذا كان للذي أعتق…٢٥٢٦حدثنا أحمد بن أبي رجاء، حدثنا يحيى بن آدم، حدثنا جرير بن حازم، سمعت…٢٥٢٧من أعتق نصيبا أو شقيصا في مملوك، فخلاصه عليه في ماله إن كان له مال،…٢٥٢٨إن الله تجاوز لي عن أمتي ما وسوست به صدورها، ما لم تعمل أو تكلم٢٥٢٩الأعمال بالنية، ولامرئ ما نوى، فمن كانت هجرته إلى الله ورسوله، فهجرته…٢٥٣٠يا أبا هريرة، هذا غلامك قد أتاك٢٥٣١يا أبا هريرة، هذا غلامك٢٥٣٢قال لما أقبل أبو هريرة رضى الله عنه ومعه غلامه وهو يطلب الإسلام، فأضل…٢٥٣٣هو لك يا عبد بن زمعة٢٥٣٤فدعا النبي صلى الله عليه وسلم به فباعه . قال جابر مات الغلام عام أول .٢٥٣٥حدثنا أبو الوليد، حدثنا شعبة، قال أخبرني عبد الله بن دينار، سمعت ابن…٢٥٣٦أعتقيها، فإن الولاء لمن أعطى الورق٢٥٣٧لا تدعون منه درهما٢٥٣٨أسلمت على ما سلف لك من خير٢٥٤٠إن معي من ترون، وأحب الحديث إلى أصدقه، فاختاروا إحدى الطائفتين إما…٢٥٤١قال كتبت إلى نافع فكتب إلى أن النبي صلى الله عليه وسلم أغار على بني…٢٥٤٢ما عليكم أن لا تفعلوا، ما من نسمة كائنة إلى يوم القيامة إلا وهى كائنة٢٥٤٣هم أشد أمتي على الدجال٢٥٤٤من كانت له جارية فعالها، فأحسن إليها ثم أعتقها وتزوجها، كان له أجران٢٥٤٥فقال لي النبي صلى الله عليه وسلم " أعيرته بأمه " . ثم قال " إن…٢٥٤٦العبد إذا نصح سيده وأحسن عبادة ربه كان له أجره مرتين٢٥٤٧أيما رجل كانت له جارية فأدبها فأحسن تأديبها، وأعتقها وتزوجها، فله…٢٥٤٨للعبد المملوك الصالح أجران، والذي نفسي بيده لولا الجهاد في سبيل الله…٢٥٤٩نعم ما لأحدهم يحسن عبادة ربه وينصح لسيده٢٥٥٠إذا نصح العبد سيده، وأحسن عبادة ربه، كان له أجره مرتين٢٥٥١المملوك الذي يحسن عبادة ربه، ويؤدي إلى سيده الذي له عليه من الحق…٢٥٥٢لا يقل أحدكم أطعم ربك، وضئ ربك، اسق ربك . وليقل سيدي مولاى . ولا يقل…٢٥٥٣من أعتق نصيبا له من العبد، فكان له من المال ما يبلغ قيمته، يقوم عليه…٢٥٥٤كلكم راع فمسئول عن رعيته، فالأمير الذي على الناس راع وهو مسئول عنهم،…٢٥٥٥إذا زنت الأمة فاجلدوها، ثم إذا زنت فاجلدوها، ثم إذا زنت فاجلدوها، في…٢٥٥٧إذا أتى أحدكم خادمه بطعامه، فإن لم يجلسه معه، فليناوله لقمة أو لقمتين…٢٥٥٨كلكم راع ومسئول عن رعيته، فالإمام راع ومسئول عن رعيته، والرجل في أهله…٢٥٥٩إذا قاتل أحدكم فليجتنب الوجه
