كتاب جزاء الصيد

٤٦ حديثًا
١٨٢١قال انطلق أبي عام الحديبية فأحرم أصحابه، ولم يحرم، وحدث النبي صلى…١٨٢٢أن أباه، حدثه قال انطلقنا مع النبي صلى الله عليه وسلم عام الحديبية…١٨٢٣فرأيت أصحابي يتراءون شيئا فنظرت، فإذا حمار وحش يعني وقع سوطه فقالوا…١٨٢٤منكم أحد أمره أن يحمل عليها، أو أشار إليها١٨٢٥إنا لم نرده عليك إلا أنا حرم١٨٢٦خمس من الدواب ليس على المحرم في قتلهن جناح١٨٢٧حدثنا مسدد، حدثنا أبو عوانة، عن زيد بن جبير، قال سمعت ابن عمر رضى…١٨٢٨خمس من الدواب لا حرج على من قتلهن الغراب والحدأة والفأرة والعقرب…١٨٢٩خمس من الدواب كلهن فاسق، يقتلهن في الحرم الغراب والحدأة والعقرب…١٨٣٠إذ نزل عليه { والمرسلات } وإنه ليتلوها، وإني لأتلقاها من فيه، وإن فاه…١٨٣١حدثنا إسماعيل، قال حدثني مالك، عن ابن شهاب، عن عروة بن الزبير، عن…١٨٣٢إن مكة حرمها الله ولم يحرمها الناس، فلا يحل لامرئ يؤمن بالله واليوم…١٨٣٣إن الله حرم مكة، فلم تحل لأحد قبلي، ولا تحل لأحد بعدي، وإنما أحلت لي…١٨٣٤لا هجرة ولكن جهاد ونية، وإذا استنفرتم فانفروا، فإن هذا بلد حرم الله…١٨٣٥حدثنا علي بن عبد الله، حدثنا سفيان، قال قال عمرو أول شىء سمعت عطاء،…١٨٣٦حدثنا خالد بن مخلد، حدثنا سليمان بن بلال، عن علقمة بن أبي علقمة، عن…١٨٣٧حدثنا أبو المغيرة عبد القدوس بن الحجاج، حدثنا الأوزاعي، حدثني عطاء بن…١٨٣٨لا تلبسوا القميص ولا السراويلات ولا العمائم، ولا البرانس إلا أن يكون…١٨٣٩اغسلوه، وكفنوه، ولا تغطوا رأسه، ولا تقربوه طيبا، فإنه يبعث يهل١٨٤٠أن عبد الله بن العباس، والمسور بن مخرمة، اختلفا بالأبواء، فقال عبد…١٨٤١من لم يجد النعلين فليلبس الخفين، ومن لم يجد إزارا فليلبس سراويل١٨٤٢لا يلبس القميص، ولا العمائم، ولا السراويلات، ولا البرنس، ولا ثوبا مسه…١٨٤٣من لم يجد الإزار فليلبس السراويل، ومن لم يجد النعلين فليلبس الخفين١٨٤٤فأبى أهل مكة أن يدعوه يدخل مكة، حتى قاضاهم لا يدخل مكة سلاحا إلا في…١٨٤٥ولأهل نجد قرن المنازل، ولأهل اليمن يلملم، هن لهن ولكل آت أتى عليهن من…١٨٤٦وعلى رأسه المغفر، فلما نزعه جاء رجل، فقال إن ابن خطل متعلق بأستار…١٨٤٧اصنع في عمرتك ما تصنع في حجك١٨٤٨وعض رجل يد رجل يعني فانتزع ثنيته فأبطله النبي صلى الله عليه وسلم .١٨٤٩اغسلوه بماء وسدر، وكفنوه في ثوبين أو قال ثوبيه ولا تحنطوه، ولا تخمروا…١٨٥٠اغسلوه بماء وسدر، وكفنوه في ثوبين، ولا تمسوه طيبا، ولا تخمروا رأسه،…١٨٥١اغسلوه بماء وسدر، وكفنوه في ثوبيه، ولا تمسوه بطيب، ولا تخمروا رأسه،…١٨٥٢نعم . حجي عنها، أرأيت لو كان على أمك دين أكنت قاضية اقضوا الله، فالله…١٨٥٣ح .١٨٥٤عام حجة الوداع، قالت يا رسول الله إن فريضة الله على عباده في الحج…١٨٥٥فجعل الفضل ينظر إليها، وتنظر إليه فجعل النبي صلى الله عليه وسلم يصرف…١٨٥٦حدثنا أبو النعمان، حدثنا حماد بن زيد، عن عبيد الله بن أبي يزيد، قال…١٨٥٧أن عبد الله بن عباس رضى الله عنهما قال أقبلت وقد ناهزت الحلم، أسير…١٨٥٨حدثنا عبد الرحمن بن يونس، حدثنا حاتم بن إسماعيل، عن محمد بن يوسف، عن…١٨٥٩يقول للسائب بن يزيد، وكان قد حج به في ثقل النبي صلى الله عليه وسلم .١٨٦٠أذن عمر رضى الله عنه لأزواج النبي صلى الله عليه وسلم في آخر حجة حجها،…١٨٦١لكن أحسن الجهاد وأجمله الحج، حج مبرور١٨٦٢لا تسافر المرأة إلا مع ذي محرم، ولا يدخل عليها رجل إلا ومعها محرم١٨٦٣ما منعك من الحج١٨٦٤أن لا تسافر امرأة مسيرة يومين ليس معها زوجها أو ذو محرم، ولا صوم…١٨٦٥حدثنا ابن سلام، أخبرنا الفزاري، عن حميد الطويل، قال حدثني ثابت، عن…١٨٦٦أن ابن جريج، أخبرهم قال أخبرني سعيد بن أبي أيوب، أن يزيد بن أبي حبيب،…
