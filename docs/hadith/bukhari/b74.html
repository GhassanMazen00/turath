٦٥ حديثًا
٥٥٧٥من شرب الخمر في الدنيا، ثم لم يتب منها، حرمها في الآخرة٥٥٧٦أنه سمع أبا هريرة رضى الله عنه أن رسول الله صلى الله عليه وسلم أتي…٥٥٧٧من أشراط الساعة أن يظهر الجهل، ويقل العلم، ويظهر الزنا، وتشرب الخمر،…٥٥٧٨لا يزني الزاني حين يزني وهو مؤمن، ولا يشرب الخمر حين يشربها وهو مؤمن،…٥٥٧٩وما بالمدينة منها شىء .٥٥٨٠حدثنا أحمد بن يونس، حدثنا أبو شهاب عبد ربه بن نافع، عن يونس، عن ثابت…٥٥٨١والخمر ما خامر العقل .٥٥٨٢حدثنا إسماعيل بن عبد الله، قال حدثني مالك بن أنس، عن إسحاق بن عبد…٥٥٨٣حدثنا مسدد، حدثنا معتمر، عن أبيه، قال سمعت أنسا، قال كنت قائما على…٥٥٨٤أن أنس بن مالك، حدثهم أن الخمر حرمت، والخمر يومئذ البسر والتمر .٥٥٨٥كل شراب أسكر فهو حرام٥٥٨٦كل شراب أسكر فهو حرام٥٥٨٧لا تنتبذوا في الدباء، ولا في المزفت٥٥٨٨وهى من خمسة أشياء العنب والتمر والحنطة والشعير والعسل، والخمر ما خامر…٥٥٨٩قال الخمر يصنع من خمسة من الزبيب والتمر والحنطة والشعير والعسل .٥٥٩٠ليكونن من أمتي أقوام يستحلون الحر والحرير والخمر والمعازف، ولينزلن…٥٥٩١يقول أتى أبو أسيد الساعدي فدعا رسول الله صلى الله عليه وسلم في عرسه،…٥٥٩٢حدثنا يوسف بن موسى، حدثنا محمد بن عبد الله أبو أحمد الزبيري، حدثنا…٥٥٩٣حدثنا علي بن عبد الله، حدثنا سفيان، عن سليمان بن أبي مسلم الأحول، عن…٥٥٩٤حدثنا مسدد، حدثنا يحيى، عن سفيان، حدثني سليمان، عن إبراهيم التيمي، عن…٥٥٩٥أفأحدث ما لم أسمع٥٥٩٦حدثنا موسى بن إسماعيل، حدثنا عبد الواحد، حدثنا الشيباني، قال سمعت عبد…٥٥٩٧أن أبا أسيد الساعدي، دعا النبي صلى الله عليه وسلم لعرسه، فكانت امرأته…٥٥٩٨. فقال سبق محمد صلى الله عليه وسلم الباذق، فما أسكر فهو حرام . قال…٥٥٩٩حدثنا عبد الله بن أبي شيبة، حدثنا أبو أسامة، حدثنا هشام بن عروة، عن…٥٦٠٠حدثنا مسلم، حدثنا هشام، حدثنا قتادة، عن أنس رضى الله عنه قال إني…٥٦٠١حدثنا أبو عاصم، عن ابن جريج، أخبرني عطاء، أنه سمع جابرا رضى الله عنه…٥٦٠٢قال نهى النبي صلى الله عليه وسلم أن يجمع بين التمر والزهو، والتمر…٥٦٠٣حدثنا عبدان، أخبرنا عبد الله، أخبرنا يونس، عن الزهري، عن سعيد بن…٥٦٠٤سمع سفيان، أخبرنا سالم أبو النضر، أنه سمع عميرا، مولى أم الفضل يحدث…٥٦٠٥ألا خمرته ولو أن تعرض عليه عودا٥٦٠٦ألا خمرته، ولو أن تعرض عليه عودا٥٦٠٧فشرب حتى رضيت، وأتانا سراقة بن جعشم على فرس فدعا عليه، فطلب إليه…٥٦٠٨نعم الصدقة اللقحة الصفي منحة، والشاة الصفي منحة، تغدو بإناء، وتروح…٥٦٠٩حدثنا أبو عاصم، عن الأوزاعي، عن ابن شهاب، عن عبيد الله بن عبد الله،…٥٦١٠رفعت إلى السدرة فإذا أربعة أنهار، نهران ظاهران، ونهران باطنان، فأما…٥٦١١بخ ذلك مال رابح أو رايح شك عبد الله وقد سمعت ما قلت وإني أرى أن…٥٦١٢وأتى داره فحلبت شاة فشبت لرسول الله صلى الله عليه وسلم من البئر،…٥٦١٣إن كان عندك ماء بات هذه الليلة في شنة، وإلا كرعنا٥٦١٤حدثنا علي بن عبد الله، حدثنا أبو أسامة، قال أخبرني هشام، عن أبيه، عن…٥٦١٥قال أتى علي رضى الله عنه على باب الرحبة، فشرب قائما فقال إن ناسا يكره…٥٦١٦ثم أتي بماء فشرب وغسل وجهه ويديه وذكر رأسه ورجليه، ثم قام فشرب فضله…٥٦١٧قال شرب النبي صلى الله عليه وسلم قائما من زمزم .٥٦١٨أنها أرسلت إلى النبي صلى الله عليه وسلم بقدح لبن، وهو واقف عشية عرفة،…٥٦١٩فشرب، ثم أعطى الأعرابي، وقال " الأيمن الأيمن " .٥٦٢٠أتأذن لي أن أعطي هؤلاء٥٦٢١إن كان عندك ماء بات في شنة وإلا كرعنا٥٦٢٢حدثنا مسدد، حدثنا معتمر، عن أبيه، قال سمعت أنسا رضى الله عنه قال كنت…٥٦٢٣إذا كان جنح الليل أو أمسيتم فكفوا صبيانكم، فإن الشياطين تنتشر حينئذ،…٥٦٢٤أطفئوا المصابيح إذا رقدتم، وغلقوا الأبواب، وأوكوا الأسقية، وخمروا…٥٦٢٥حدثنا آدم، حدثنا ابن أبي ذئب، عن الزهري، عن عبيد الله بن عبد الله بن…٥٦٢٦أنه سمع أبا سعيد الخدري، يقول سمعت رسول الله صلى الله عليه وسلم ينهى…٥٦٢٧وأن يمنع جاره أن يغرز خشبه في داره .٥٦٢٨حدثنا مسدد، حدثنا إسماعيل، أخبرنا أيوب، عن عكرمة، عن أبي هريرة رضى…٥٦٢٩حدثنا مسدد، حدثنا يزيد بن زريع، حدثنا خالد، عن عكرمة، عن ابن عباس رضى…٥٦٣٠إذا شرب أحدكم فلا يتنفس في الإناء، وإذا بال أحدكم فلا يمسح ذكره…٥٦٣١قال كان أنس يتنفس في الإناء مرتين أو ثلاثا، وزعم أن النبي صلى الله…٥٦٣٢هن لهم في الدنيا وهى لكم في الآخرة٥٦٣٣لا تشربوا في آنية الذهب والفضة، ولا تلبسوا الحرير والديباج، فإنها لهم…٥٦٣٤الذي يشرب في إناء الفضة إنما يجرجر في بطنه نار جهنم٥٦٣٥واتباع الجنازة، وتشميت العاطس، وإجابة الداعي، وإفشاء السلام، ونصر…٥٦٣٦أنهم شكوا في صوم النبي صلى الله عليه وسلم يوم عرفة، فبعث إليه بقدح من…٥٦٣٧فأمر أبا أسيد الساعدي أن يرسل إليها فأرسل إليها، فقدمت فنزلت في أجم…٥٦٣٨قال رأيت قدح النبي صلى الله عليه وسلم عند أنس بن مالك، وكان قد انصدع…٥٦٣٩حى على أهل الوضوء، البركة من الله
