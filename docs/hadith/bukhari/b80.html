١٠٦ حديثًا
٦٣٠٤لكل نبي دعوة مستجابة يدعو بها، وأريد أن أختبئ دعوتي شفاعة لأمتي في…٦٣٠٥كل نبي سأل سؤلا أو قال لكل نبي دعوة قد دعا بها فاستجيب، فجعلت دعوتي…٦٣٠٦سيد الاستغفار أن تقول اللهم أنت ربي، لا إله إلا أنت، خلقتني وأنا…٦٣٠٧والله إني لأستغفر الله وأتوب إليه في اليوم أكثر من سبعين مرة٦٣٠٨إن المؤمن يرى ذنوبه كأنه قاعد تحت جبل يخاف أن يقع عليه، وإن الفاجر…٦٣٠٩الله أفرح بتوبة عبده من أحدكم سقط على بعيره، وقد أضله في أرض فلاة٦٣١٠فإذا طلع الفجر صلى ركعتين خفيفتين، ثم اضطجع على شقه الأيمن، حتى يجيء…٦٣١١إذا أتيت مضجعك فتوضأ وضوءك للصلاة، ثم اضطجع على شقك الأيمن، وقل اللهم…٦٣١٢باسمك أموت وأحيا٦٣١٣إذا أردت مضجعك فقل اللهم أسلمت نفسي إليك، وفوضت أمري إليك، ووجهت وجهي…٦٣١٤اللهم باسمك أموت وأحيا٦٣١٥اللهم أسلمت نفسي إليك، ووجهت وجهي إليك، وفوضت أمري إليك، وألجأت ظهري…٦٣١٦اللهم اجعل في قلبي نورا، وفي بصري نورا، وفي سمعي نورا، وعن يميني…٦٣١٧اللهم لك الحمد، أنت نور السموات والأرض ومن فيهن، ولك الحمد أنت قيم…٦٣١٨أن فاطمة عليهما السلام شكت ما تلقى في يدها من الرحى، فأتت النبي صلى…٦٣١٩وقرأ بالمعوذات، ومسح بهما جسده .٦٣٢٠إذا أوى أحدكم إلى فراشه فلينفض فراشه بداخلة إزاره، فإنه لا يدري ما…٦٣٢١يتنزل ربنا تبارك وتعالى كل ليلة إلى السماء الدنيا حين يبقى ثلث الليل…٦٣٢٢اللهم إني أعوذ بك من الخبث والخبائث٦٣٢٣سيد الاستغفار اللهم أنت ربي لا إله إلا أنت، خلقتني وأنا عبدك، وأنا…٦٣٢٤باسمك اللهم أموت وأحيا٦٣٢٥اللهم باسمك أموت وأحيا٦٣٢٦قل اللهم إني ظلمت نفسي ظلما كثيرا، ولا يغفر الذنوب إلا أنت، فاغفر لي…٦٣٢٧{ ولا تجهر بصلاتك ولا تخافت بها } أنزلت في الدعاء .٦٣٢٨إن الله هو السلام، فإذا قعد أحدكم في الصلاة فليقل التحيات لله إلى…٦٣٢٩. قالوا يا رسول الله ذهب أهل الدثور بالدرجات والنعيم المقيم . قال "…٦٣٣٠لا إله إلا الله، وحده لا شريك له، له الملك، وله الحمد، وهو على كل شىء…٦٣٣١فلما صاف القوم قاتلوهم، فأصيب عامر بقائمة سيف نفسه فمات، فلما أمسوا…٦٣٣٢اللهم صل على آل فلان٦٣٣٣ألا تريحني من ذي الخلصة٦٣٣٤اللهم أكثر ماله وولده، وبارك له فيما أعطيته٦٣٣٥رحمه الله، لقد أذكرني كذا وكذا آية أسقطتها في سورة كذا وكذا٦٣٣٦يرحم الله موسى، لقد أوذي بأكثر من هذا فصبر٦٣٣٧قال حدث الناس، كل جمعة مرة، فإن أبيت فمرتين، فإن أكثرت فثلاث مرار ولا…٦٣٣٨إذا دعا أحدكم فليعزم المسألة، ولا يقولن اللهم إن شئت فأعطني . فإنه لا…٦٣٣٩لا يقولن أحدكم اللهم اغفر لي، اللهم ارحمني، إن شئت . ليعزم المسألة،…٦٣٤٠يستجاب لأحدكم ما لم يعجل يقول دعوت فلم يستجب لي٦٣٤١وشريك، سمعا أنسا، عن النبي صلى الله عليه وسلم رفع يديه حتى رأيت بياض…٦٣٤٢اللهم حوالينا ولا علينا٦٣٤٣قال خرج النبي صلى الله عليه وسلم إلى هذا المصلى يستسقي، فدعا واستسقى…٦٣٤٤اللهم أكثر ماله وولده، وبارك له فيما أعطيته٦٣٤٥لا إله إلا الله العظيم الحليم، لا إله إلا الله رب السموات والأرض، رب…٦٣٤٦لا إله إلا الله العظيم الحليم، لا إله إلا الله رب العرش العظيم، لا…٦٣٤٧كان رسول الله صلى الله عليه وسلم يتعوذ من جهد البلاء، ودرك الشقاء،…٦٣٤٨لن يقبض نبي قط حتى يرى مقعده من الجنة ثم يخير٦٣٤٩حدثنا مسدد، حدثنا يحيى، عن إسماعيل، عن قيس، قال أتيت خبابا وقد اكتوى…٦٣٥٠حدثنا محمد بن المثنى، حدثنا يحيى، عن إسماعيل، قال حدثني قيس، قال أتيت…٦٣٥١لا يتمنين أحد منكم الموت لضر نزل به، فإن كان لا بد متمنيا للموت فليقل…٦٣٥٢يقول ذهبت بي خالتي إلى رسول الله صلى الله عليه وسلم فقالت يا رسول…٦٣٥٣أنه كان يخرج به جده عبد الله بن هشام من السوق أو إلى السوق فيشتري…٦٣٥٤وهو الذي مج رسول الله صلى الله عليه وسلم في وجهه وهو غلام من بئرهم .٦٣٥٥فأتي بصبي فبال على ثوبه، فدعا بماء فأتبعه إياه، ولم يغسله .٦٣٥٦حدثنا أبو اليمان، أخبرنا شعيب، عن الزهري، قال أخبرني عبد الله بن…٦٣٥٧فقولوا اللهم صل على محمد، وعلى آل محمد، كما صليت على آل إبراهيم، إنك…٦٣٥٨قولوا اللهم صل على محمد عبدك ورسولك، كما صليت على إبراهيم، وبارك على…٦٣٥٩اللهم صل عليه٦٣٦٠قولوا اللهم صل على محمد وأزواجه وذريته، كما صليت على آل إبراهيم،…٦٣٦١اللهم فأيما مؤمن سببته فاجعل ذلك له قربة إليك يوم القيامة٦٣٦٢لا تسألوني اليوم عن شىء إلا بينته لكم٦٣٦٣التمس لنا غلاما من غلمانكم يخدمني٦٣٦٤حدثنا الحميدي، حدثنا سفيان، حدثنا موسى بن عقبة، قال سمعت أم خالد بنت…٦٣٦٥اللهم إني أعوذ بك من البخل، وأعوذ بك من الجبن، وأعوذ بك أن أرد إلى…٦٣٦٦صدقتا، إنهم يعذبون عذابا تسمعه البهائم كلها٦٣٦٧اللهم إني أعوذ بك من العجز والكسل، والجبن والهرم، وأعوذ بك من عذاب…٦٣٦٨اللهم إني أعوذ بك من الكسل والهرم، والمأثم والمغرم، ومن فتنة القبر…٦٣٦٩اللهم إني أعوذ بك من الهم والحزن، والعجز والكسل، والجبن والبخل، وضلع…٦٣٧٠اللهم إني أعوذ بك من البخل، وأعوذ بك من الجبن، وأعوذ بك أن أرد إلى…٦٣٧١اللهم إني أعوذ بك من الكسل، وأعوذ بك من الجبن، وأعوذ بك من الهرم،…٦٣٧٢اللهم حبب إلينا المدينة، كما حببت إلينا مكة أو أشد، وانقل حماها إلى…٦٣٧٣أن أباه، قال عادني رسول الله صلى الله عليه وسلم في حجة الوداع من…٦٣٧٤اللهم إني أعوذ بك من الجبن، وأعوذ بك من البخل، وأعوذ بك من أن أرد إلى…٦٣٧٥اللهم إني أعوذ بك من الكسل والهرم والمغرم والمأثم، اللهم إني أعوذ بك…٦٣٧٦اللهم إني أعوذ بك من فتنة النار ومن عذاب النار، وأعوذ بك من فتنة…٦٣٧٧اللهم إني أعوذ بك من فتنة النار وعذاب النار، وفتنة القبر وعذاب القبر،…٦٣٧٩اللهم أكثر ماله وولده، وبارك له فيما أعطيته٦٣٨١اللهم أكثر ماله وولده، وبارك له فيما أعطيته٦٣٨٢إذا هم بالأمر فليركع ركعتين، ثم يقول اللهم إني أستخيرك بعلمك،…٦٣٨٣اللهم اغفر لعبيد أبي عامر٦٣٨٤أيها الناس اربعوا على أنفسكم، فإنكم لا تدعون أصم ولا غائبا، ولكن…٦٣٨٥لا إله إلا الله، وحده لا شريك له، له الملك وله الحمد، وهو على كل شىء…٦٣٨٦حدثنا مسدد، حدثنا حماد بن زيد، عن ثابت، عن أنس رضى الله عنه قال رأى…٦٣٨٧فتزوجت امرأة فقال النبي صلى الله عليه وسلم " تزوجت يا جابر " . قلت…٦٣٨٨لو أن أحدهم إذا أراد أن يأتي أهله قال باسم الله، اللهم جنبنا الشيطان،…٦٣٨٩اللهم ربنا آتنا في الدنيا حسنة، وفي الآخرة حسنة، وقنا عذاب النار٦٣٩٠اللهم إني أعوذ بك من البخل، وأعوذ بك من الجبن، وأعوذ بك من أن نرد إلى…٦٣٩١أشعرت أن الله قد أفتاني فيما استفتيته فيه٦٣٩٢اللهم منزل الكتاب، سريع الحساب، اهزم الأحزاب، اهزمهم وزلزلهم٦٣٩٣سمع الله لمن حمده٦٣٩٤إن عصية عصوا الله ورسوله٦٣٩٥مهلا يا عائشة، إن الله يحب الرفق في الأمر كله٦٣٩٦ملأ الله قبورهم وبيوتهم نارا، كما شغلونا عن صلاة الوسطى حتى غابت الشمس٦٣٩٧اللهم اهد دوسا وأت بهم٦٣٩٨رب اغفر لي خطيئتي وجهلي وإسرافي في أمري كله، وما أنت أعلم به مني،…٦٣٩٩اللهم اغفر لي خطيئتي وجهلي وإسرافي في أمري، وما أنت أعلم به مني،…٦٤٠٠في الجمعة ساعة لا يوافقها مسلم وهو قائم يصلي يسأل خيرا إلا أعطاه٦٤٠١أتوا النبي صلى الله عليه وسلم فقالوا السام عليك . قال " وعليكم " .…٦٤٠٢إذا أمن القارئ فأمنوا، فإن الملائكة تؤمن، فمن وافق تأمينه تأمين…٦٤٠٣من قال لا إله إلا الله، وحده لا شريك له، له الملك وله الحمد، وهو على…٦٤٠٤حدثنا عبد الله بن محمد، حدثنا عبد الملك بن عمرو، حدثنا عمر بن أبي…٦٤٠٥من قال سبحان الله وبحمده . في يوم مائة مرة حطت خطاياه، وإن كانت مثل…٦٤٠٦كلمتان خفيفتان على اللسان، ثقيلتان في الميزان، حبيبتان إلى الرحمن،…٦٤٠٧مثل الذي يذكر ربه والذي لا يذكر مثل الحى والميت٦٤٠٨إن لله ملائكة يطوفون في الطرق، يلتمسون أهل الذكر، فإذا وجدوا قوما…٦٤٠٩فإنكم لا تدعون أصم ولا غائبا٦٤١٠لله تسعة وتسعون اسما، مائة إلا واحدا، لا يحفظها أحد إلا دخل الجنة،…٦٤١١حدثنا عمر بن حفص، حدثنا أبي، حدثنا الأعمش، قال حدثني شقيق، قال كنا…
