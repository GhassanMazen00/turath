كتاب الذبائح والصيد

٦٩ حديثًا
٥٤٧٥ما أصاب بحده فكله، وما أصاب بعرضه فهو وقيذ٥٤٧٦إذا أصبت بحده فكل، فإذا أصاب بعرضه فقتل فإنه وقيذ، فلا تأكل٥٤٧٧كل ما أمسكن عليك٥٤٧٨أما ما ذكرت من أهل الكتاب فإن وجدتم غيرها فلا تأكلوا فيها، وإن لم…٥٤٧٩إنه لا يصاد به صيد ولا ينكى به عدو، ولكنها قد تكسر السن وتفقأ العين٥٤٨٠من اقتنى كلبا ليس بكلب ماشية أو ضارية، نقص كل يوم من عمله قيراطان٥٤٨١من اقتنى كلبا إلا كلب ضار لصيد أو كلب ماشية، فإنه ينقص من أجره كل يوم…٥٤٨٢من اقتنى كلبا إلا كلب ماشية أو ضار، نقص من عمله كل يوم قيراطان٥٤٨٣إذا أرسلت كلابك المعلمة وذكرت اسم الله، فكل مما أمسكن عليكم، وإن قتلن…٥٤٨٤إذا أرسلت كلبك وسميت فأمسك وقتل، فكل، وإن أكل فلا تأكل، فإنما أمسك…٥٤٨٥أنه قال للنبي صلى الله عليه وسلم، يرمي الصيد فيقتفر أثره اليومين…٥٤٨٦إذا أرسلت كلبك وسميت، فأخذ فقتل فأكل فلا تأكل، فإنما أمسك على نفسه٥٤٨٧إذا أرسلت كلابك المعلمة وذكرت اسم الله، فكل مما أمسكن عليك، إلا أن…٥٤٨٨أما ما ذكرت أنك بأرض قوم أهل الكتاب، تأكل في آنيتهم، فإن وجدتم غير…٥٤٨٩فسعوا عليها حتى لغبوا، فسعيت عليها حتى أخذتها، فجئت بها إلى أبي طلحة،…٥٤٩٠إنما هي طعمة أطعمكموها الله٥٤٩١هل معكم من لحمه شىء٥٤٩٢أبقي معكم شىء منه٥٤٩٣حدثنا مسدد، حدثنا يحيى، عن ابن جريج، قال أخبرني عمرو، أنه سمع جابرا…٥٤٩٤ثم ثلاث جزائر، ثم نهاه أبو عبيدة .٥٤٩٥حدثنا أبو الوليد، حدثنا شعبة، عن أبي يعفور، قال سمعت ابن أبي أوفى رضى…٥٤٩٦أما ما ذكرت أنك بأرض أهل كتاب فلا تأكلوا في آنيتهم، إلا أن لا تجدوا…٥٤٩٧على ما أوقدتم هذه النيران٥٤٩٨إن لهذه البهائم أوابد كأوابد الوحش، فما ند عليكم فاصنعوا به هكذا٥٤٩٩أنه سمع عبد الله، يحدث عن رسول الله صلى الله عليه وسلم أنه لقي زيد بن…٥٥٠٠من ذبح قبل الصلاة فليذبح مكانها أخرى، ومن كان لم يذبح حتى صلينا…٥٥٠١سمع ابن كعب بن مالك، يخبر ابن عمر أن أباه، أخبره أن جارية لهم كانت…٥٥٠٢من بني سلمة أخبر عبد الله، أن جارية، لكعب بن مالك ترعى غنما له…٥٥٠٣ما أنهر الدم وذكر اسم الله فكل، ليس الظفر والسن، أما الظفر فمدى…٥٥٠٤أن امرأة، ذبحت شاة بحجر، فسئل النبي صلى الله عليه وسلم عن ذلك، فأمر…٥٥٠٥فأصيبت شاة منها، فأدركتها فذبحتها بحجر، فسئل النبي صلى الله عليه وسلم…٥٥٠٦كل يعني ما أنهر الدم إلا السن والظفر٥٥٠٧سموا عليه أنتم وكلوه٥٥٠٨فرمى إنسان بجراب فيه شحم، فنزوت لآخذه، فالتفت فإذا النبي صلى الله…٥٥٠٩اعجل أو أرن ما أنهر الدم وذكر اسم الله فكل، ليس السن والظفر، وسأحدثك،…٥٥١٠قال أخبرتني فاطمة بنت المنذر، امرأتي عن أسماء بنت أبي بكر رضى الله…٥٥١١سمع عبدة، عن هشام، عن فاطمة، عن أسماء، قالت ذبحنا على عهد رسول الله…٥٥١٢أن أسماء بنت أبي بكر، قالت نحرنا على عهد رسول الله صلى الله عليه وسلم…٥٥١٣قال دخلت مع أنس على الحكم بن أيوب، فرأى غلمانا أو فتيانا نصبوا دجاجة…٥٥١٤أنه سمعه يحدث، عن ابن عمر رضى الله عنهما أنه دخل على يحيى بن سعيد…٥٥١٥قال كنت عند ابن عمر فمروا بفتية أو بنفر نصبوا دجاجة يرمونها، فلما…٥٥١٦حدثنا حجاج بن منهال، حدثنا شعبة، قال أخبرني عدي بن ثابت، قال سمعت عبد…٥٥١٧حدثنا يحيى، حدثنا وكيع، عن سفيان، عن أيوب، عن أبي قلابة، عن زهدم…٥٥١٨ما عندي ما أحملكم عليه٥٥١٩حدثنا الحميدي، حدثنا سفيان، حدثنا هشام، عن فاطمة، عن أسماء، قالت…٥٥٢٠ورخص في لحوم الخيل .٥٥٢١ونافع، عن ابن عمر رضى الله عنهما نهى النبي صلى الله عليه وسلم عن لحوم…٥٥٢٢حدثنا مسدد، حدثنا يحيى، عن عبيد الله، حدثني نافع، عن عبد الله، قال…٥٥٢٣والحسن، ابنى محمد بن علي عن أبيهما، عن علي رضى الله عنهم قال نهى رسول…٥٥٢٤ورخص في لحوم الخيل .٥٥٢٦وابن أبي أوفى، رضى الله عنهم قالا نهى النبي صلى الله عليه وسلم عن…٥٥٢٧أن أبا إدريس، أخبره أن أبا ثعلبة قال حرم رسول الله صلى الله عليه وسلم…٥٥٢٨فإنها رجس . فأكفئت القدور وإنها لتفور باللحم .٥٥٢٩ولكن أبى ذاك البحر ابن عباس وقرأ { قل لا أجد فيما أوحي إلى محرما }٥٥٣٠حدثنا عبد الله بن يوسف، أخبرنا مالك، عن ابن شهاب، عن أبي إدريس…٥٥٣١هلا استمتعتم بإهابها٥٥٣٢ما على أهلها لو انتفعوا بإهابها٥٥٣٣ما من مكلوم يكلم في الله إلا جاء يوم القيامة وكلمه يدمى، اللون لون دم…٥٥٣٤مثل الجليس الصالح والسوء كحامل المسك ونافخ الكير، فحامل المسك إما أن…٥٥٣٥فسعى القوم فلغبوا، فأخذتها فجئت بها إلى أبي طلحة فذبحها، فبعث بوركيها…٥٥٣٦الضب لست آكله ولا أحرمه٥٥٣٧لا، ولكن لم يكن بأرض قومي فأجدني أعافه٥٥٣٨ألقوها وما حولها وكلوه٥٥٣٩تموت في الزيت والسمن وهو جامد أو غير جامد، الفأرة أو غيرها قال بلغنا…٥٥٤٠ألقوها وما حولها وكلوه٥٥٤١أنه كره أن تعلم الصورة، . وقال ابن عمر نهى النبي صلى الله عليه وسلم…٥٥٤٢قال دخلت على النبي صلى الله عليه وسلم بأخ لي يحنكه، وهو في مربد له،…٥٥٤٣ما أنهر الدم وذكر اسم الله فكلوا، ما لم يكن سن ولا ظفر، وسأحدثكم عن…٥٥٤٤إن لها أوابد كأوابد الوحش فما غلبكم منها فاصنعوا به هكذا
