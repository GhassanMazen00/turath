كتاب الهبة وفضلها والتحريض عليها

٦٨ حديثًا
٢٥٦٦يا نساء المسلمات لا تحقرن جارة لجارتها، ولو فرسن شاة٢٥٦٧ثلاثة أهلة في شهرين، وما أوقدت في أبيات رسول الله صلى الله عليه وسلم…٢٥٦٨لو دعيت إلى ذراع أو كراع لأجبت، ولو أهدي إلى ذراع أو كراع لقبلت٢٥٦٩مري عبدك فليعمل لنا أعواد المنبر٢٥٧٠فأبصروا حمارا وحشيا، وأنا مشغول أخصف نعلي، فلم يؤذنوني به، وأحبوا لو…٢٥٧١الأيمنون، الأيمنون، ألا فيمنوا٢٥٧٢فسعى القوم فلغبوا، فأدركتها فأخذتها، فأتيت بها أبا طلحة فذبحها، وبعث…٢٥٧٣أما إنا لم نرده عليك إلا أنا حرم٢٥٧٤كانوا يتحرون بهداياهم يوم عائشة، يبتغون بها أو يبتغون بذلك مرضاة رسول…٢٥٧٥فأكل النبي صلى الله عليه وسلم من الأقط والسمن، وترك الضب تقذرا . قال…٢٥٧٦إذا أتي بطعام سأل عنه أهدية أم صدقة فإن قيل صدقة . قال لأصحابه كلوا .…٢٥٧٧هو لها صدقة، ولنا هدية٢٥٧٨اشتريها فأعتقيها، فإنما الولاء لمن أعتق٢٥٧٩قالت دخل النبي صلى الله عليه وسلم على عائشة رضى الله عنها فقال "…٢٥٨٠فأعرض عنها .٢٥٨١لا تؤذيني في عائشة، فإن الوحى لم يأتني، وأنا في ثوب امرأة إلا عائشة٢٥٨٢قال دخلت عليه فناولني طيبا، قال كان أنس رضى الله عنه لا يرد الطيب .…٢٥٨٣أما بعد، فإن إخوانكم جاءونا تائبين، وإني رأيت أن أرد إليهم سبيهم، فمن…٢٥٨٥حدثنا مسدد، حدثنا عيسى بن يونس، عن هشام، عن أبيه، عن عائشة رضى الله…٢٥٨٦أكل ولدك نحلت مثله٢٥٨٧أعطيت سائر ولدك مثل هذا٢٥٨٨قالت عائشة رضى الله عنها لما ثقل النبي صلى الله عليه وسلم فاشتد وجعه…٢٥٨٩العائد في هبته كالكلب يقيء، ثم يعود في قيئه٢٥٩٠تصدقي، ولا توعي فيوعى عليك٢٥٩١أنفقي ولا تحصي فيحصي الله عليك، ولا توعي فيوعي الله عليك٢٥٩٢مولى ابن عباس أن ميمونة بنت الحارث رضى الله عنها أخبرته أنها، أعتقت…٢٥٩٣فأيتهن خرج سهمها خرج بها معه، وكان يقسم لكل امرأة منهن يومها وليلتها،…٢٥٩٤ولو وصلت بعض أخوالك كان أعظم لأجرك٢٥٩٥إلى أقربهما منك بابا٢٥٩٦ليس بنا رد عليك، ولكنا حرم٢٥٩٧فهلا جلس في بيت أبيه أو بيت أمه، فينظر يهدى له أم لا والذي نفسي بيده…٢٥٩٨لو جاء مال البحرين أعطيتك هكذا ثلاثا٢٥٩٩ولم يعط مخرمة منها شيئا، فقال مخرمة يا بنى انطلق بنا إلى رسول الله…٢٦٠٠حدثنا محمد بن محبوب، حدثنا عبد الواحد، حدثنا معمر، عن الزهري، عن حميد…٢٦٠١أن جابر بن عبد الله رضى الله عنهما أخبره أن أباه قتل يوم أحد شهيدا،…٢٦٠٢إن أذنت لي أعطيت هؤلاء٢٦٠٣حدثنا ثابت حدثنا مسعر عن محارب عن جابر رضي الله عنه أتيت النبي صلى…٢٦٠٤ائت المسجد فصل ركعتين٢٦٠٥أتأذن لي أن أعطي هؤلاء٢٦٠٦دعوه فإن لصاحب الحق مقالا . وقال اشتروا له سنا فأعطوها إياه٢٦٠٧معي من ترون، وأحب الحديث إلى أصدقه، فاختاروا إحدى الطائفتين إما السبى…٢٦٠٩إن لصاحب الحق مقالا٢٦١٠فكان يتقدم النبي صلى الله عليه وسلم فيقول أبوه يا عبد الله لا يتقدم…٢٦١١وكنت على بكر صعب فقال النبي صلى الله عليه وسلم لعمر " بعنيه " .…٢٦١٢إنما يلبسها من لا خلاق له في الآخرة٢٦١٣إني رأيت على بابها سترا موشيا٢٦١٤فرأيت الغضب في وجهه، فشققتها بين نسائي .٢٦١٦والذي نفس محمد بيده لمناديل سعد بن معاذ في الجنة أحسن من هذا٢٦١٧أتت النبي صلى الله عليه وسلم بشاة مسمومة، فأكل منها فجيء بها فقيل ألا…٢٦١٨هل مع أحد منكم طعام٢٦١٩إنما يلبس هذا من لا خلاق له في الآخرة٢٦٢٠في عهد رسول الله صلى الله عليه وسلم، فاستفتيت رسول الله صلى الله عليه…٢٦٢١العائد في هبته كالعائد في قيئه٢٦٢٢ليس لنا مثل السوء، الذي يعود في هبته كالكلب يرجع في قيئه٢٦٢٣لا تشتره، وإن أعطاكه بدرهم واحد، فإن العائد في صدقته كالكلب يعود في…٢٦٢٤أن ابن جريج، أخبرهم قال أخبرني عبد الله بن عبيد الله بن أبي مليكة، أن…٢٦٢٥حدثنا أبو نعيم، حدثنا شيبان، عن يحيى، عن أبي سلمة، عن جابر رضى الله…٢٦٢٦حدثنا حفص بن عمر، حدثنا همام، حدثنا قتادة، قال حدثني النضر بن أنس، عن…٢٦٢٧ما رأينا من شىء، وإن وجدناه لبحرا٢٦٢٨دخلت على عائشة رضى الله عنها وعليها درع قطر ثمن خمسة دراهم، فقالت…٢٦٢٩نعم المنيحة اللقحة الصفي منحة، والشاة الصفي تغدو بإناء وتروح بإناء٢٦٣٠فقاسمهم الأنصار على أن يعطوهم ثمار أموالهم كل عام ويكفوهم العمل…٢٦٣١أربعون خصلة أعلاهن منيحة العنز، ما من عامل يعمل بخصلة منها رجاء…٢٦٣٢من كانت له أرض فليزرعها أو ليمنحها أخاه، فإن أبى فليمسك أرضه٢٦٣٣ويحك إن الهجرة شأنها شديد فهل لك من إبل٢٦٣٤بذاك يعني ابن عباس رضى الله عنهما أن النبي صلى الله عليه وسلم خرج إلى…٢٦٣٥هاجر إبراهيم بسارة، فأعطوها آجر، فرجعت فقالت أشعرت أن الله كبت الكافر…٢٦٣٦لا تشتر، ولا تعد في صدقتك
