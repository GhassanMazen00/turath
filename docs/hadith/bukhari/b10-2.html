٧٥٧ارجع فصل، فإنك لم تصل٧٥٨قال قال سعد كنت أصلي بهم صلاة رسول الله صلى الله عليه وسلم صلاتى…٧٥٩قال كان النبي صلى الله عليه وسلم يقرأ في الركعتين الأوليين من صلاة…٧٦٠حدثنا عمر بن حفص، قال حدثنا أبي قال، حدثنا الأعمش، حدثني عمارة، عن…٧٦١قال قلت لخباب بن الأرت أكان النبي صلى الله عليه وسلم يقرأ في الظهر…٧٦٢قال كان النبي صلى الله عليه وسلم يقرأ في الركعتين من الظهر والعصر…٧٦٣حدثنا عبد الله بن يوسف، قال أخبرنا مالك، عن ابن شهاب، عن عبيد الله بن…٧٦٤قال قال لي زيد بن ثابت ما لك تقرأ في المغرب بقصار، وقد سمعت النبي صلى…٧٦٥حدثنا عبد الله بن يوسف، قال أخبرنا مالك، عن ابن شهاب، عن محمد بن جبير…٧٦٦قال صليت مع أبي هريرة العتمة فقرأ { إذا السماء انشقت } فسجد فقلت له…٧٦٧أن النبي صلى الله عليه وسلم كان في سفر فقرأ في العشاء في إحدى…٧٦٨قال صليت مع أبي هريرة العتمة فقرأ { إذا السماء انشقت } فسجد فقلت ما…٧٦٩سمع البراء، رضى الله عنه قال سمعت النبي صلى الله عليه وسلم يقرأ {…٧٧٠وأحذف في الأخريين، ولا آلو ما اقتديت به من صلاة رسول الله صلى الله…٧٧١فقال كان النبي صلى الله عليه وسلم يصلي الظهر حين تزول الشمس، والعصر…٧٧٢أنه سمع أبا هريرة رضى الله عنه يقول في كل صلاة يقرأ، فما أسمعنا رسول…٧٧٣وقد حيل بين الشياطين وبين خبر السماء، وأرسلت عليهم الشهب، فرجعت…٧٧٤قال قرأ النبي صلى الله عليه وسلم فيما أمر، وسكت فيما أمر { وما كان…٧٧٤.٢يا فلان ما يمنعك أن تفعل ما يأمرك به أصحابك وما يحملك على لزوم هذه…٧٧٥قال جاء رجل إلى ابن مسعود فقال قرأت المفصل الليلة في ركعة . فقال هذا…٧٧٦أن النبي صلى الله عليه وسلم كان يقرأ في الظهر في الأوليين بأم الكتاب…٧٧٧حدثنا قتيبة بن سعيد، قال حدثنا جرير، عن الأعمش، عن عمارة بن عمير، عن…٧٧٨أن النبي صلى الله عليه وسلم كان يقرأ بأم الكتاب وسورة معها في…٧٧٩أن النبي صلى الله عليه وسلم كان يطول في الركعة الأولى من صلاة الظهر،…٧٨٠إذا أمن الإمام فأمنوا فإنه من وافق تأمينه تأمين الملائكة غفر له ما…٧٨١إذا قال أحدكم آمين . وقالت الملائكة في السماء آمين . فوافقت إحداهما…٧٨٢إذا قال الإمام { غير المغضوب عليهم ولا الضالين } فقولوا آمين . فإنه…٧٨٣زادك الله حرصا ولا تعد٧٨٤حدثنا إسحاق الواسطي، قال حدثنا خالد، عن الجريري، عن أبي العلاء، عن…٧٨٥أنه كان يصلي بهم، فيكبر كلما خفض ورفع، فإذا انصرف قال إني لأشبهكم…٧٨٦، فكان إذا سجد كبر، وإذا رفع رأسه كبر، وإذا نهض من الركعتين كبر، فلما…٧٨٧قال رأيت رجلا عند المقام يكبر في كل خفض ورفع وإذا قام وإذا وضع،…٧٨٨قال صليت خلف شيخ بمكة فكبر ثنتين وعشرين تكبيرة، فقلت لابن عباس إنه…٧٨٩أنه سمع أبا هريرة، يقول كان رسول الله صلى الله عليه وسلم إذا قام إلى…٧٩٠يقول صليت إلى جنب أبي فطبقت بين كفى ثم وضعتهما بين فخذى، فنهاني أبي…٧٩١قال رأى حذيفة رجلا لا يتم الركوع والسجود قال ما صليت، ولو مت مت على…٧٩٢قال كان ركوع النبي صلى الله عليه وسلم وسجوده وبين السجدتين وإذا رفع…٧٩٣ارجع فصل فإنك لم تصل٧٩٤سبحانك اللهم ربنا وبحمدك، اللهم اغفر لي٧٩٥سمع الله لمن حمده٧٩٦إذا قال الإمام سمع الله لمن حمده . فقولوا اللهم ربنا لك الحمد . فإنه…٧٩٧قال لأقربن صلاة النبي صلى الله عليه وسلم . فكان أبو هريرة رضى الله…٧٩٨حدثنا عبد الله بن أبي الأسود، قال حدثنا إسماعيل، عن خالد الحذاء، عن…٧٩٩سمع الله لمن حمده٨٠٠حدثنا أبو الوليد، قال حدثنا شعبة، عن ثابت، قال كان أنس ينعت لنا صلاة…٨٠١حدثنا أبو الوليد، قال حدثنا شعبة، عن الحكم، عن ابن أبي ليلى، عن…٨٠٢فقام فأمكن القيام، ثم ركع فأمكن الركوع، ثم رفع رأسه فأنصت هنية، قال…٨٠٣وأبو سلمة بن عبد الرحمن أن أبا هريرة، كان يكبر في كل صلاة من المكتوبة…٨٠٤اللهم أنج الوليد بن الوليد وسلمة بن هشام وعياش بن أبي ربيعة،…٨٠٥إنما جعل الإمام ليؤتم به، فإذا كبر فكبروا وإذا ركع فاركعوا، وإذا رفع…٨٠٦هل تمارون في القمر ليلة البدر ليس دونه سحاب٨٠٧حدثنا يحيى بن بكير، قال حدثني بكر بن مضر، عن جعفر، عن ابن هرمز، عن…٨٠٨رأى رجلا لا يتم ركوعه ولا سجوده، فلما قضى صلاته قال له حذيفة ما صليت…٨٠٩أمر النبي صلى الله عليه وسلم أن يسجد على سبعة أعضاء، ولا يكف شعرا ولا…٨١٠أمرنا أن نسجد على سبعة أعظم ولا نكف ثوبا ولا شعرا٨١١حدثنا آدم، حدثنا إسرائيل، عن أبي إسحاق، عن عبد الله بن يزيد الخطمي،…٨١٢أمرت أن أسجد على سبعة أعظم على الجبهة وأشار بيده على أنفه واليدين،…٨١٣من كان اعتكف مع النبي صلى الله عليه وسلم فليرجع، فإني أريت ليلة…٨١٤حدثنا محمد بن كثير، قال أخبرنا سفيان، عن أبي حازم، عن سهل بن سعد، قال…٨١٥قال أمر النبي صلى الله عليه وسلم أن يسجد على سبعة أعظم، ولا يكف ثوبه…٨١٦أمرت أن أسجد على سبعة، لا أكف شعرا ولا ثوبا٨١٧سبحانك اللهم ربنا وبحمدك، اللهم اغفر لي٨١٩لو رجعتم إلى أهليكم صلوا صلاة كذا في حين كذا، صلوا صلاة كذا في حين…٨٢٠قال كان سجود النبي صلى الله عليه وسلم وركوعه، وقعوده بين السجدتين…٨٢١كان إذا رفع رأسه من الركوع قام حتى يقول القائل قد نسي . وبين السجدتين…٨٢٢اعتدلوا في السجود، ولا يبسط أحدكم ذراعيه انبساط الكلب٨٢٣أنه رأى النبي صلى الله عليه وسلم يصلي، فإذا كان في وتر من صلاته لم…٨٢٤وما أريد الصلاة، ولكن أريد أن أريكم كيف رأيت النبي صلى الله عليه وسلم…٨٢٥وحين سجد، وحين رفع، وحين قام من الركعتين وقال هكذا رأيت النبي صلى…٨٢٦صلاة خلف علي بن أبي طالب رضى الله عنه فكان إذا سجد كبر، وإذا رفع كبر،…٨٢٧أنه أخبره أنه، كان يرى عبد الله بن عمر رضى الله عنهما يتربع في الصلاة…٨٢٨ويزيد بن محمد، عن محمد بن عمرو بن حلحلة، عن محمد بن عمرو بن عطاء، أنه…٨٢٩مولى بني عبد المطلب وقال مرة مولى ربيعة بن الحارث أن عبد الله ابن…٨٣٠فلما كان في آخر صلاته سجد سجدتين وهو جالس .٨٣١إن الله هو السلام، فإذا صلى أحدكم فليقل التحيات لله، والصلوات…٨٣٣اللهم إني أعوذ بك من عذاب القبر وأعوذ بك من فتنة المسيح الدجال، الله عليه وسلم يسجد…٨٣٧أن أم سلمة رضى الله عنها قالت كان رسول الله صلى الله عليه وسلم إذا…٨٣٨حدثنا حبان بن موسى، قال أخبرنا عبد الله، قال أخبرنا معمر، عن الزهري،…٨٤٠أفعل إن شاء الله٨٤١أن أبا معبد، مولى ابن عباس أخبره أن ابن عباس رضى الله عنهما أخبره أن…٨٤٢حدثنا علي بن عبد الله، قال حدثنا سفيان، { عن عمرو، } قال أخبرني أبو…٨٤٣ألا أحدثكم بأمر إن أخذتم به أدركتم من سبقكم ولم يدرككم أحد بعدكم،…٨٤٤لا إله إلا الله وحده لا شريك له، له الملك، وله الحمد، وهو على كل شىء…٨٤٥حدثنا موسى بن إسماعيل، قال حدثنا جرير بن حازم، قال حدثنا أبو رجاء، عن…٨٤٦هل تدرون ماذا قال ربكم٨٤٧إن الناس قد صلوا ورقدوا، وإنكم لن تزالوا في صلاة ما انتظرتم الصلاة٨٤٨وقال لنا آدم حدثنا شعبة، عن أيوب، عن نافع، قال كان ابن عمر يصلي في…٨٥٠أن ابن شهاب، كتب إليه قال حدثتني هند بنت الحارث الفراسية، عن أم سلمة،…٨٥١ذكرت شيئا من تبر عندنا فكرهت أن يحبسني، فأمرت بقسمته٨٥٢قال قال عبد الله لا يجعل أحدكم للشيطان شيئا من صلاته، يرى أن حقا عليه…٨٥٣من أكل من هذه الشجرة يعني الثوم فلا يقربن مسجدنا٨٥٤من أكل من هذه الشجرة يريد الثوم فلا يغشانا في مساجدنا٨٥٥من أكل ثوما أو بصلا فليعتزلنا أو قال فليعتزل مسجدنا، وليقعد في بيته٨٥٦من أكل من هذه الشجرة فلا يقربنا، أو لا يصلين معنا٨٥٧مر مع النبي صلى الله عليه وسلم على قبر منبوذ، فأمهم وصفوا عليه . فقلت…٨٥٨الغسل يوم الجمعة واجب على كل محتلم٨٥٩فنام النبي صلى الله عليه وسلم فلما كان في بعض الليل قام رسول الله صلى…٨٦٠قوموا فلأصلي بكم٨٦١فمررت بين يدى بعض الصف، فنزلت وأرسلت الأتان ترتع ودخلت في الصف، فلم…٨٦٢إنه ليس أحد من أهل الأرض يصلي هذه الصلاة غيركم٨٦٣ولولا مكاني منه ما شهدته يعني من صغره أتى العلم الذي عند دار كثير بن…٨٦٤ما ينتظرها أحد غيركم من أهل الأرض٨٦٥إذا استأذنكم نساؤكم بالليل إلى المسجد فأذنوا لهن٨٦٦قال حدثتني هند بنت الحارث، أن أم سلمة، زوج النبي صلى الله عليه وسلم…٨٦٧قالت إن كان رسول الله صلى الله عليه وسلم ليصلي الصبح، فينصرف النساء…٨٦٨إني لأقوم إلى الصلاة وأنا أريد أن أطول فيها، فأسمع بكاء الصبي، فأتجوز…٨٦٩حدثنا عبد الله بن يوسف، قال أخبرنا مالك، عن يحيى بن سعيد، عن عمرة، عن…٨٧٠ويمكث هو في مقامه يسيرا قبل أن يقوم . قال نرى والله أعلم أن ذلك كان…٨٧١فقمت ويتيم خلفه، وأم سليم خلفنا .٨٧٢رضى الله عنها أن رسول الله صلى الله عليه وسلم كان يصلي الصبح بغلس…٨٧٣إذا استأذنت امرأة أحدكم فلا يمنعها٨٧٤قال صلى النبي صلى الله عليه وسلم في بيت أم سليم، فقمت ويتيم خلفه، وأم…٨٧٥قالت كان رسول الله صلى الله عليه وسلم إذا سلم قام النساء حين يقضي…
