٣٤٨٩رضى الله عنهما . { وجعلناكم شعوبا وقبائل } قال الشعوب القبائل العظام،…٣٤٩٠حدثنا محمد بن بشار، حدثنا يحيى بن سعيد، عن عبيد الله، قال حدثني سعيد…٣٤٩١قال حدثتني ربيبة النبي، صلى الله عليه وسلم زينب ابنة أبي سلمة قال قلت…٣٤٩٢حدثتني ربيبة النبي، صلى الله عليه وسلم وأظنها زينب قالت نهى رسول الله…٣٤٩٤تجدون الناس معادن، خيارهم في الجاهلية خيارهم في الإسلام إذا فقهوا،…٣٤٩٦الناس تبع لقريش في هذا الشأن، مسلمهم تبع لمسلمهم، وكافرهم تبع لكافرهم٣٤٩٧فنزلت عليه إلا أن تصلوا قرابة بيني وبينكم .٣٤٩٨من ها هنا جاءت الفتن نحو المشرق، والجفاء وغلظ القلوب في الفدادين أهل…٣٤٩٩الفخر والخيلاء في الفدادين أهل الوبر، والسكينة في أهل الغنم، والإيمان…٣٥٠٠إن هذا الأمر في قريش، لا يعاديهم أحد إلا كبه الله على وجهه، ما أقاموا…٣٥٠١لا يزال هذا الأمر في قريش، ما بقي منهم اثنان٣٥٠٢إنما بنو هاشم وبنو المطلب شىء واحد٣٥٠٣قال ذهب عبد الله بن الزبير مع أناس من بني زهرة إلى عائشة، وكانت أرق…٣٥٠٤قريش والأنصار وجهينة ومزينة وأسلم وأشجع وغفار موالي ليس لهم مولى، دون…٣٥٠٥قال كان عبد الله بن الزبير أحب البشر إلى عائشة بعد النبي صلى الله…٣٥٠٦أن عثمان، دعا زيد بن ثابت وعبد الله بن الزبير وسعيد بن العاص وعبد…٣٥٠٧ارموا بني إسماعيل، فإن أباكم كان راميا، وأنا مع بني فلان٣٥٠٨ليس من رجل ادعى لغير أبيه وهو يعلمه إلا كفر، ومن ادعى قوما ليس له…٣٥٠٩إن من أعظم الفرى أن يدعي الرجل إلى غير أبيه، أو يري عينه ما لم تر، أو…٣٥١٠آمركم بأربع، وأنهاكم عن أربع، الإيمان بالله شهادة أن لا إله إلا الله،…٣٥١١ألا إن الفتنة ها هنا يشير إلى المشرق من حيث يطلع قرن الشيطان٣٥١٢قريش والأنصار وجهينة ومزينة وأسلم وغفار وأشجع موالي، ليس لهم مولى دون…٣٥١٣غفار غفر الله لها، وأسلم سالمها الله، وعصية عصت الله ورسوله٣٥١٤أسلم سالمها الله، وغفار غفر الله لها٣٥١٥أرأيتم إن كان جهينة ومزينة وأسلم وغفار خيرا من بني تميم وبني أسد، ومن…٣٥١٦أرأيت إن كان أسلم وغفار ومزينة وأحسبه وجهينة خيرا من بني تميم وبني…٣٥١٧لا تقوم الساعة حتى يخرج رجل من قحطان يسوق الناس بعصاه٣٥١٨ما بال دعوى أهل الجاهلية٣٥١٩ليس منا من ضرب الخدود، وشق الجيوب، ودعا بدعوى الجاهلية٣٥٢٠عمرو بن لحى بن قمعة بن خندف أبو خزاعة٣٥٢١رأيت عمرو بن عامر بن لحى الخزاعي يجر قصبه في النار، وكان أول من سيب…٣٥٢٢يا أبا ذر اكتم هذا الأمر، وارجع إلى بلدك، فإذا بلغك ظهورنا فأقبل٣٥٢٣حدثنا سليمان بن حرب، حدثنا حماد، عن أيوب، عن محمد، عن أبي هريرة رضى…٣٥٢٤حدثنا أبو النعمان، حدثنا أبو عوانة، عن أبي بشر، عن سعيد بن جبير، عن…٣٥٢٥يا بني فهر، يا بني عدي لبطون قريش٣٥٢٦قال لما نزلت { وأنذر عشيرتك الأقربين } جعل النبي صلى الله عليه وسلم…٣٥٢٧يا بني عبد مناف، اشتروا أنفسكم من الله، يا بني عبد المطلب اشتروا…٣٥٢٨هل فيكم أحد من غيركم٣٥٣٠دعهما يا أبا بكر، فإنها أيام عيد، وتلك الأيام أيام منى٣٥٣١حدثني عثمان بن أبي شيبة، حدثنا عبدة، عن هشام، عن أبيه، عن عائشة رضى…٣٥٣٢لي خمسة أسماء أنا محمد، وأحمد، وأنا الماحي الذي يمحو الله بي الكفر،…٣٥٣٣ألا تعجبون كيف يصرف الله عني شتم قريش ولعنهم يشتمون مذمما ويلعنون…٣٥٣٤مثلي ومثل الأنبياء كرجل بنى دارا فأكملها وأحسنها، إلا موضع لبنة، فجعل…٣٥٣٥إن مثلي ومثل الأنبياء من قبلي كمثل رجل بنى بيتا فأحسنه وأجمله، إلا…٣٥٣٦حدثنا عبد الله بن يوسف، حدثنا الليث، عن عقيل، عن ابن شهاب، عن عروة بن…٣٥٣٧سموا باسمي، ولا تكتنوا بكنيتي٣٥٣٨تسموا باسمي، ولا تكتنوا بكنيتي٣٥٣٩سموا باسمي، ولا تكتنوا بكنيتي٣٥٤٠رأيت السائب بن يزيد ابن أربع وتسعين جلدا معتدلا فقال قد علمت ما متعت…٣٥٤١قال ذهبت بي خالتي إلى رسول الله صلى الله عليه وسلم فقالت يا رسول الله…٣٥٤٢قال صلى أبو بكر رضى الله عنه العصر، ثم خرج يمشي فرأى الحسن يلعب مع…٣٥٤٣حدثنا أحمد بن يونس، حدثنا زهير، حدثنا إسماعيل، عن أبي جحيفة رضى الله…٣٥٤٤حدثني عمرو بن علي، حدثنا ابن فضيل، حدثنا إسماعيل بن أبي خالد، قال…٣٥٤٥قال رأيت النبي صلى الله عليه وسلم ورأيت بياضا من تحت شفته السفلى…٣٥٤٦أنه سأل عبد الله بن بسر صاحب النبي صلى الله عليه وسلم قال أرأيت النبي…٣٥٤٧يصف النبي صلى الله عليه وسلم قال كان ربعة من القوم، ليس بالطويل ولا…٣٥٤٨ولا بالأبيض الأمهق، وليس بالآدم وليس بالجعد القطط ولا بالسبط، بعثه…٣٥٤٩يقول كان رسول الله صلى الله عليه وسلم أحسن الناس وجها وأحسنه خلقا،…٣٥٥٠قال سألت أنسا هل خضب النبي صلى الله عليه وسلم قال لا، إنما كان شىء في…٣٥٥١بعيد ما بين المنكبين، له شعر يبلغ شحمة أذنه، رأيته في حلة حمراء، لم…٣٥٥٢قال سئل البراء أكان وجه النبي صلى الله عليه وسلم مثل السيف قال لا بل…٣٥٥٣قال خرج رسول الله صلى الله عليه وسلم بالهاجرة إلى البطحاء فتوضأ ثم…٣٥٥٤وأجود ما يكون في رمضان، حين يلقاه جبريل، وكان جبريل عليه السلام يلقاه…٣٥٥٥ألم تسمعي ما قال المدلجي لزيد وأسامة ورأى أقدامهما إن بعض هذه الأقدام…٣٥٥٦أن عبد الله بن كعب، قال سمعت كعب بن مالك، يحدث حين تخلف عن تبوك، قال…٣٥٥٧بعثت من خير قرون بني آدم قرنا فقرنا، حتى كنت من القرن الذي كنت فيه٣٥٥٨وكان المشركون يفرقون رءوسهم فكان أهل الكتاب يسدلون رءوسهم، وكان رسول…٣٥٥٩إن من خياركم أحسنكم أخلاقا٣٥٦٠ما لم يكن إثما، فإن كان إثما كان أبعد الناس منه، وما انتقم رسول الله…٣٥٦١ولا شممت ريحا قط أو عرفا قط أطيب من ريح أو عرف النبي صلى الله عليه…٣٥٦٢حدثنا مسدد، حدثنا يحيى، عن شعبة، عن قتادة، عن عبد الله بن أبي عتبة،…٣٥٦٢.٢وابن، مهدي قالا حدثنا شعبة، مثله وإذا كره شيئا عرف في وجهه .٣٥٦٣إن اشتهاه أكله، وإلا تركه .٣٥٦٤حدثنا قتيبة بن سعيد، حدثنا بكر بن مضر، عن جعفر بن ربيعة، عن الأعرج،…٣٥٦٥أن أنسا رضى الله عنه حدثهم أن رسول الله صلى الله عليه وسلم كان لا…٣٥٦٦قال دفعت إلى النبي صلى الله عليه وسلم وهو بالأبطح في قبة كان…٣٥٦٨يسمعني ذلك وكنت أسبح فقام قبل أن أقضي سبحتي، ولو أدركته لرددت عليه،…٣٥٦٩تنام عيني ولا ينام قلبي٣٥٧٠أسري بالنبي صلى الله عليه وسلم من مسجد الكعبة جاء ثلاثة نفر قبل أن…٣٥٧١يا فلان ما يمنعك أن تصلي معنا٣٥٧٢فوضع يده في الإناء، فجعل الماء ينبع من بين أصابعه، فتوضأ القوم . قال…٣٥٧٣فالتمس الوضوء فلم يجدوه فأتي رسول الله صلى الله عليه وسلم بوضوء، فوضع…٣٥٧٤فانطلقوا يسيرون، فحضرت الصلاة فلم يجدوا ماء يتوضئون، فانطلق رجل من…٣٥٧٥سمع يزيد، أخبرنا حميد، عن أنس رضى الله عنه قال حضرت الصلاة فقام من…٣٥٧٦والنبي صلى الله عليه وسلم بين يديه ركوة فتوضأ فجهش الناس نحوه، فقال "…٣٥٧٧والحديبية بئر فنزحناها حتى لم نترك فيها قطرة، فجلس النبي صلى الله…٣٥٧٨أنه سمع أنس بن مالك، يقول قال أبو طلحة لأم سليم لقد سمعت صوت، رسول…٣٥٧٩اطلبوا فضلة من ماء٣٥٨٠توفي وعليه دين، فأتيت النبي صلى الله عليه وسلم فقلت إن أبي ترك عليه…٣٥٨١من كان عنده طعام اثنين فليذهب بثالث، ومن كان عنده طعام أربعة فليذهب…٣٥٨٢حوالينا ولا علينا٣٥٨٣حدثنا محمد بن المثنى، حدثنا يحيى بن كثير أبو غسان، حدثنا أبو حفص…٣٥٨٤فقالت امرأة من الأنصار أو رجل يا رسول الله ألا نجعل لك منبرا قال " إن…٣٥٨٥أنه سمع جابر بن عبد الله رضى الله عنهما يقول كان المسجد مسقوفا على…٣٥٨٦فتنة الرجل في أهله وماله وجاره تكفرها الصلاة والصدقة والأمر بالمعروف…٣٥٨٩لا تقوم الساعة حتى تقاتلوا قوما نعالهم الشعر، وحتى تقاتلوا الترك،…٣٥٩٠لا تقوم الساعة حتى تقاتلوا خوزا وكرمان من الأعاجم، حمر الوجوه، فطس…٣٥٩١بين يدى الساعة تقاتلون قوما نعالهم الشعر، وهو هذا البارز٣٥٩٢بين يدى الساعة تقاتلون قوما ينتعلون الشعر، وتقاتلون قوما كأن وجوههم…٣٥٩٣تقاتلكم اليهود فتسلطون عليهم ثم يقول الحجر يا مسلم، هذا يهودي ورائي…٣٥٩٤يأتي على الناس زمان يغزون، فيقال فيكم من صحب الرسول صلى الله عليه…٣٥٩٥يا عدي هل رأيت الحيرة٣٥٩٥.٢كنت عند النبي صلى الله عليه وسلم .٣٥٩٦إني فرطكم، وأنا شهيد عليكم، إني والله لأنظر إلى حوضي الآن، وإني قد…٣٥٩٧هل ترون ما أرى إني أرى الفتن تقع خلال بيوتكم مواقع القطر٣٥٩٨لا إله إلا الله، ويل للعرب من شر قد اقترب، فتح اليوم من ردم يأجوج…٣٥٩٩سبحان الله، ماذا أنزل من الخزائن وماذا أنزل من الفتن٣٦٠٠يأتي على الناس زمان تكون الغنم فيه خير مال المسلم، يتبع بها شعف…٣٦٠٢ستكون فتن، القاعد فيها خير من القائم، والقائم فيها خير من الماشي،…٣٦٠٣ستكون أثرة وأمور تنكرونها٣٦٠٤يهلك الناس هذا الحى من قريش٣٦٠٥هلاك أمتي على يدى غلمة من قريش٣٦٠٦أنه سمع حذيفة بن اليمان، يقول كان الناس يسألون رسول الله صلى الله…٣٦٠٧حدثني محمد بن المثنى، قال حدثني يحيى بن سعيد، عن إسماعيل، حدثني قيس،…٣٦٠٨لا تقوم الساعة حتى يقتتل فئتان دعواهما واحدة٣٦٠٩لا تقوم الساعة حتى يقتتل فئتان، فيكون بينهما مقتلة عظيمة، دعواهما…٣٦١٠ويلك، ومن يعدل إذا لم أعدل قد خبت وخسرت إن لم أكن أعدل٣٦١١يأتي في آخر الزمان قوم حدثاء الأسنان، سفهاء الأحلام، يقولون من خير…٣٦١٢كان الرجل فيمن قبلكم يحفر له في الأرض فيجعل فيه، فيجاء بالمنشار،…٣٦١٣اذهب إليه فقل له إنك لست من أهل النار، ولكن من أهل الجنة٣٦١٤اقرأ فلان، فإنها السكينة نزلت للقرآن، أو تنزلت للقرآن٣٦١٥ألم يأن للرحيل٣٦١٦لا بأس طهور إن شاء الله٣٦١٧فكان يكتب للنبي صلى الله عليه وسلم، فعاد نصرانيا فكان يقول ما يدري…٣٦١٨إذا هلك كسرى فلا كسرى بعده، وإذا هلك قيصر فلا قيصر بعده، والذي نفس…٣٦١٩إذا هلك كسرى فلا كسرى بعده وذكر وقال لتنفقن كنوزهما في سبيل الله٣٦٢١لو سألتني هذه القطعة ما أعطيتكها، ولن تعدو أمر الله فيك، ولئن أدبرت…٣٦٢٢رأيت في المنام أني أهاجر من مكة إلى أرض بها نخل، فذهب وهلي إلى أنها…٣٦٢٤ثم أسر إليها حديثا، فبكت فقلت لها لم تبكين ثم أسر إليها حديثا فضحكت…٣٦٢٦فسارها بشىء فبكت، ثم دعاها، فسارها فضحكت، قالت فسألتها عن ذلك . فقالت…٣٦٢٧قال كان عمر بن الخطاب رضى الله عنه يدني ابن عباس، فقال له عبد الرحمن…٣٦٢٨أما بعد فإن الناس يكثرون ويقل الأنصار، حتى يكونوا في الناس بمنزلة…٣٦٢٩ابني هذا سيد، ولعل الله أن يصلح به بين فئتين من المسلمين٣٦٣٠وعيناه تذرفان .٣٦٣١هل لكم من أنماط٣٦٣٢وكان أمية إذا انطلق إلى الشأم فمر بالمدينة نزل على سعد، فقال أمية…٣٦٣٣رأيت الناس مجتمعين في صعيد، فقام أبو بكر فنزع ذنوبا أو ذنوبين، وفي…٣٦٣٤قال أنبئت أن جبريل عليه السلام أتى النبي صلى الله عليه وسلم وعنده أم…٣٦٣٥ما تجدون في التوراة في شأن الرجم٣٦٣٦حدثنا صدقة بن الفضل، أخبرنا ابن عيينة، عن ابن أبي نجيح، عن مجاهد، عن…٣٦٣٧فأراهم انشقاق القمر .٣٦٣٨انشق في زمان النبي صلى الله عليه وسلم٣٦٣٩من أصحاب النبي صلى الله عليه وسلم خرجا من عند النبي صلى الله عليه…٣٦٤٠لا يزال ناس من أمتي ظاهرين حتى يأتيهم أمر الله وهم ظاهرون٣٦٤١لا يزال من أمتي أمة قائمة بأمر الله، لا يضرهم من خذلهم ولا من خالفهم…٣٦٤٣الخير معقود بنواصي الخيل إلى يوم القيامة٣٦٤٤الخيل في نواصيها الخير إلى يوم القيامة٣٦٤٥الخيل معقود في نواصيها الخير٣٦٤٦الخيل لثلاثة لرجل أجر، ولرجل ستر وعلى رجل وزر . فأما الذي له أجر،…٣٦٤٧الله أكبر، خربت خيبر، إنا إذا نزلنا بساحة قوم
