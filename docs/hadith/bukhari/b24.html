١١٢ حديثًا
١٣٩٥ادعهم إلى شهادة أن لا إله إلا الله، وأني رسول الله، فإن هم أطاعوا…١٣٩٦أرب ماله، تعبد الله، ولا تشرك به شيئا، وتقيم الصلاة، وتؤتي الزكاة،…١٣٩٧تعبد الله لا تشرك به شيئا، وتقيم الصلاة المكتوبة، وتؤدي الزكاة…١٣٩٨آمركم بأربع، وأنهاكم عن أربع الإيمان بالله وشهادة أن لا إله إلا الله…١٤٠٠أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله . فمن قالها فقد عصم…١٤٠١قال قال جرير بن عبد الله بايعت النبي صلى الله عليه وسلم على إقام…١٤٠٢تأتي الإبل على صاحبها، على خير ما كانت، إذا هو لم يعط فيها حقها، تطؤه…١٤٠٣من آتاه الله مالا، فلم يؤد زكاته مثل له يوم القيامة شجاعا أقرع، له…١٤٠٤{ والذين يكنزون الذهب والفضة ولا ينفقونها في سبيل الله } قال ابن عمر…١٤٠٥ليس فيما دون خمس أواق صدقة، وليس فيما دون خمس ذود صدقة، وليس فيما دون…١٤٠٦سمع هشيما، أخبرنا حصين، عن زيد بن وهب، قال مررت بالربذة فإذا أنا…١٤٠٨يا أبا ذر أتبصر أحدا١٤٠٩لا حسد إلا في اثنتين رجل آتاه الله مالا فسلطه على هلكته في الحق، ورجل…١٤١٠من تصدق بعدل تمرة من كسب طيب ولا يقبل الله إلا الطيب وإن الله يتقبلها…١٤١١تصدقوا فإنه يأتي عليكم زمان يمشي الرجل بصدقته، فلا يجد من يقبلها يقول…١٤١٢لا تقوم الساعة حتى يكثر فيكم المال فيفيض، حتى يهم رب المال من يقبل…١٤١٣أما قطع السبيل فإنه لا يأتي عليك إلا قليل حتى تخرج العير إلى مكة بغير…١٤١٤ليأتين على الناس زمان يطوف الرجل فيه بالصدقة من الذهب ثم لا يجد أحدا…١٤١٥حدثنا عبيد الله بن سعيد، حدثنا أبو النعمان الحكم هو ابن عبد الله…١٤١٦وإن لبعضهم اليوم لمائة ألف .١٤١٧اتقوا النار ولو بشق تمرة١٤١٨من ابتلي من هذه البنات بشىء كن له سترا من النار١٤١٩أن تصدق وأنت صحيح شحيح، تخشى الفقر وتأمل الغنى، ولا تمهل حتى إذا بلغت…١٤٢٠رضى الله عنها أن بعض، أزواج النبي صلى الله عليه وسلم قلن للنبي صلى…١٤٢١قال رجل لأتصدقن بصدقة . فخرج بصدقته فوضعها في يد سارق فأصبحوا يتحدثون…١٤٢٢لك ما نويت يا يزيد، ولك ما أخذت يا معن١٤٢٣سبعة يظلهم الله تعالى في ظله يوم لا ظل إلا ظله إمام عدل، وشاب نشأ في…١٤٢٤تصدقوا، فسيأتي عليكم زمان يمشي الرجل بصدقته فيقول الرجل لو جئت بها…١٤٢٥إذا أنفقت المرأة من طعام بيتها غير مفسدة كان لها أجرها بما أنفقت…١٤٢٦خير الصدقة ما كان عن ظهر غنى، وابدأ بمن تعول١٤٢٨اليد العليا خير من اليد السفلى، وابدأ بمن تعول، وخير الصدقة عن ظهر…١٤٢٩اليد العليا خير من اليد السفلى، فاليد العليا هي المنفقة، والسفلى هي…١٤٣٠كنت خلفت في البيت تبرا من الصدقة، فكرهت أن أبيته فقسمته١٤٣١ثم مال على النساء ومعه بلال، فوعظهن وأمرهن أن يتصدقن، فجعلت المرأة…١٤٣٢اشفعوا تؤجروا، ويقضي الله على لسان نبيه صلى الله عليه وسلم ما شاء١٤٣٣لا توكي فيوكى عليك١٤٣٤لا توعي فيوعي الله عليك، ارضخي ما استطعت١٤٣٥الصلاة والصدقة، والأمر بالمعروف والنهى عن المنكر١٤٣٦أسلمت على ما سلف من خير١٤٣٧إذا تصدقت المرأة من طعام زوجها غير مفسدة كان لها أجرها، ولزوجها بما…١٤٣٨الخازن المسلم الأمين الذي ينفذ وربما قال يعطي ما أمر به كاملا موفرا…١٤٣٩والأعمش، عن أبي وائل، عن مسروق، عن عائشة رضى الله عنها عن النبي صلى…١٤٤٠إذا أطعمت المرأة من بيت زوجها غير مفسدة، لها أجرها، وله مثله، وللخازن…١٤٤١إذا أنفقت المرأة من طعام بيتها غير مفسدة فلها أجرها، وللزوج بما…١٤٤٢ما من يوم يصبح العباد فيه إلا ملكان ينزلان فيقول أحدهما اللهم أعط…١٤٤٣مثل البخيل والمتصدق كمثل رجلين، عليهما جبتان من حديد١٤٤٤وقال حنظلة عن طاوس، " جنتان " . وقال الليث حدثني جعفر، عن ابن هرمز،…١٤٤٥على كل مسلم صدقة١٤٤٦حدثنا أحمد بن يونس، حدثنا أبو شهاب، عن خالد الحذاء، عن حفصة بنت…١٤٤٧ليس فيما دون خمس ذود صدقة من الإبل، وليس فيما دون خمس أواق صدقة، وليس…١٤٤٨ومن بلغت صدقته بنت مخاض وليست عنده وعنده بنت لبون فإنها تقبل منه،…١٤٤٩قال قال ابن عباس رضى الله عنهما أشهد على رسول الله صلى الله عليه وسلم…١٤٥٠ولا يجمع بين متفرق، ولا يفرق بين مجتمع، خشية الصدقة١٤٥١وما كان من خليطين فإنهما يتراجعان بينهما بالسوية١٤٥٢ويحك، إن شأنها شديد، فهل لك من إبل تؤدي صدقتها١٤٥٣من بلغت عنده من الإبل صدقة الجذعة، وليست عنده جذعة وعنده حقة، فإنها…١٤٥٤هذه فريضة الصدقة التي فرض رسول الله صلى الله عليه وسلم على المسلمين،…١٤٥٥ولا يخرج في الصدقة هرمة، ولا ذات عوار، ولا تيس، إلا ما شاء المصدق١٤٥٧أن أبا هريرة رضى الله عنه قال قال أبو بكر رضى الله عنه والله لو…١٤٥٨إنك تقدم على قوم أهل كتاب، فليكن أول ما تدعوهم إليه عبادة الله، فإذا…١٤٥٩ليس فيما دون خمسة أوسق من التمر صدقة، وليس فيما دون خمس أواق من الورق…١٤٦٠والذي نفسي بيده أو والذي لا إله غيره، أو كما حلف ما من رجل تكون له…١٤٦١بخ، ذلك مال رابح، ذلك مال رابح، وقد سمعت ما قلت وإني أرى أن تجعلها في…١٤٦٢أيها الناس تصدقوا١٤٦٣ليس على المسلم في فرسه وغلامه صدقة١٤٦٤ليس على المسلم صدقة في عبده ولا في فرسه١٤٦٥إني مما أخاف عليكم من بعدي ما يفتح عليكم من زهرة الدنيا وزينتها١٤٦٦تصدقن ولو من حليكن١٤٦٧أنفقي عليهم، فلك أجر ما أنفقت عليهم١٤٦٨ما ينقم ابن جميل إلا أنه كان فقيرا فأغناه الله ورسوله، وأما خالد…١٤٦٩ما يكون عندي من خير فلن أدخره عنكم، ومن يستعفف يعفه الله، ومن يستغن…١٤٧٠والذي نفسي بيده لأن يأخذ أحدكم حبله فيحتطب على ظهره خير له من أن يأتي…١٤٧١لأن يأخذ أحدكم حبله فيأتي بحزمة الحطب على ظهره فيبيعها فيكف الله بها…١٤٧٢يا حكيم إن هذا المال خضرة حلوة، فمن أخذه بسخاوة نفس بورك له فيه، ومن…١٤٧٣خذه، إذا جاءك من هذا المال شىء، وأنت غير مشرف ولا سائل، فخذه، وما لا…١٤٧٥ما يزال الرجل يسأل الناس حتى يأتي يوم القيامة ليس في وجهه مزعة لحم١٤٧٦ليس المسكين الذي ترده الأكلة والأكلتان، ولكن المسكين الذي ليس له غنى…١٤٧٧إن الله كره لكم ثلاثا قيل وقال، وإضاعة المال، وكثرة السؤال١٤٧٨وهو أعجبهم إلى، فقمت إلى رسول الله صلى الله عليه وسلم فساررته فقلت ما…١٤٧٩ليس المسكين الذي يطوف على الناس ترده اللقمة واللقمتان والتمرة…١٤٨٠لأن يأخذ أحدكم حبله، ثم يغدو أحسبه قال إلى الجبل فيحتطب، فيبيع فيأكل…١٤٨٢ومن كان معه بعير فليعقله " . فعقلناها وهبت ريح شديدة فقام رجل فألقته…١٤٨٣فيما سقت السماء والعيون أو كان عثريا العشر، وما سقي بالنضح نصف العشر١٤٨٤ليس فيما أقل من خمسة أوسق صدقة، ولا في أقل من خمسة من الإبل الذود…١٤٨٥أما علمت أن آل محمد صلى الله عليه وسلم لا يأكلون الصدقة١٤٨٦حدثنا حجاج، حدثنا شعبة، أخبرني عبد الله بن دينار، سمعت ابن عمر رضى…١٤٨٧حدثنا عبد الله بن يوسف، حدثني الليث، حدثني خالد بن يزيد، عن عطاء بن…١٤٨٨قال حتى تحمار .١٤٨٩لا تعد في صدقتك١٤٩٠لا تشتر ولا تعد في صدقتك، وإن أعطاكه بدرهم، فإن العائد في صدقته…١٤٩١كخ كخ ليطرحها ثم قال أما شعرت أنا لا نأكل الصدقة١٤٩٢هلا انتفعتم بجلدها١٤٩٣اشتريها، فإنما الولاء لمن أعتق١٤٩٤حدثنا علي بن عبد الله، حدثنا يزيد بن زريع، حدثنا خالد، عن حفصة بنت…١٤٩٥هو عليها صدقة، وهو لنا هدية١٤٩٦إنك ستأتي قوما أهل كتاب، فإذا جئتهم فادعهم إلى أن يشهدوا أن لا إله…١٤٩٧اللهم صل على آل فلان١٤٩٨أن رجلا من بني إسرائيل سأل بعض بني إسرائيل بأن يسلفه ألف دينار،…١٤٩٩العجماء جبار، والبئر جبار، والمعدن جبار، وفي الركاز الخمس١٥٠٠فلما جاء حاسبه .١٥٠١من عرينة اجتووا المدينة، فرخص لهم رسول الله صلى الله عليه وسلم أن…١٥٠٢فوافيته في يده الميسم يسم إبل الصدقة .١٥٠٣أو صاعا من شعير على العبد والحر، والذكر والأنثى، والصغير والكبير من…١٥٠٤أو صاعا من شعير، على كل حر أو عبد، ذكر أو أنثى، من المسلمين .١٥٠٥حدثنا قبيصة، حدثنا سفيان، عن زيد بن أسلم، عن عياض بن عبد الله، عن أبي…١٥٠٦أو صاعا من شعير، أو صاعا من تمر، أو صاعا من أقط، أو صاعا من زبيب .١٥٠٧أن عبد الله، قال أمر النبي صلى الله عليه وسلم بزكاة الفطر، صاعا من…١٥٠٨سمع يزيد العدني، حدثنا سفيان، عن زيد بن أسلم، قال حدثني عياض بن عبد…١٥٠٩حدثنا آدم، حدثنا حفص بن ميسرة، حدثنا موسى بن عقبة، عن نافع، عن ابن…١٥١٠حدثنا معاذ بن فضالة، حدثنا أبو عمر، عن زيد، عن عياض بن عبد الله بن…١٥١١والحر والمملوك، صاعا من تمر أو صاعا من شعير، فعدل الناس به نصف صاع من…١٥١٢أو صاعا من تمر على الصغير والكبير والحر والمملوك .
