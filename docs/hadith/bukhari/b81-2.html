١٨١ حديثًا•صفحة ٢ من ٢
٦٥٦٢إن أهون أهل النار عذابا يوم القيامة رجل على أخمص قدميه جمرتان يغلي…٦٥٦٣اتقوا النار ولو بشق تمرة، فمن لم يجد فبكلمة طيبة٦٥٦٤لعله تنفعه شفاعتي يوم القيامة فيجعل في ضحضاح من النار، يبلغ كعبيه،…٦٥٦٥يجمع الله الناس يوم القيامة فيقولون لو استشفعنا على ربنا حتى يريحنا…٦٥٦٦يخرج قوم من النار بشفاعة محمد صلى الله عليه وسلم فيدخلون الجنة، يسمون…٦٥٦٨هبلت، أجنة واحدة هي إنها جنان كثيرة، وإنه في الفردوس الأعلى٦٥٦٩لا يدخل أحد الجنة إلا أري مقعده من النار، لو أساء، ليزداد شكرا، ولا…٦٥٧٠لقد ظننت يا أبا هريرة أن لا يسألني عن هذا الحديث أحد أول منك، لما…٦٥٧١إني لأعلم آخر أهل النار خروجا منها، وآخر أهل الجنة دخولا رجل يخرج من…٦٥٧٢حدثنا مسدد، حدثنا أبو عوانة، عن عبد الملك، عن عبد الله بن الحارث بن…٦٥٧٣هل تضارون في الشمس، ليس دونها سحاب٦٥٧٤هذا لك ومثله معه٦٥٧٥أنا فرطكم، على الحوض٦٥٧٦أنا فرطكم، على الحوض، وليرفعن رجال منكم ثم ليختلجن دوني فأقول يا رب…٦٥٧٧أمامكم حوض كما بين جرباء وأذرح٦٥٧٨وعطاء بن السائب، عن سعيد بن جبير، عن ابن عباس رضى الله عنه قال الكوثر…٦٥٧٩حوضي مسيرة شهر، ماؤه أبيض من اللبن، وريحه أطيب من المسك، وكيزانه…٦٥٨٠وإن فيه من الأباريق كعدد نجوم السماء ) .٦٥٨١بينما أنا أسير في الجنة إذا أنا بنهر حافتاه قباب الدر المجوف قلت ما…٦٥٨٢ليردن على ناس من أصحابي الحوض، حتى عرفتهم اختلجوا دوني، فأقول أصحابي…٦٥٨٣إني فرطكم على الحوض، من مر على شرب، ومن شرب لم يظمأ أبدا، ليردن على…٦٥٨٤فأقول إنهم مني . فيقال إنك لا تدري ما أحدثوا بعدك . فأقول سحقا سحقا…٦٥٨٥يرد على يوم القيامة رهط من أصحابي فيحلئون عن الحوض فأقول يا رب أصحابي…٦٥٨٦يرد على الحوض رجال من أصحابي فيحلئون عنه فأقول يا رب أصحابي . فيقول…٦٥٨٧بينا أنا قائم إذا زمرة، حتى إذا عرفتهم خرج رجل من بيني وبينهم فقال…٦٥٨٨ما بين بيتي ومنبري روضة من رياض الجنة، ومنبري على حوضي٦٥٨٩أنا فرطكم، على الحوض٦٥٩٠إني فرط لكم، وأنا شهيد عليكم، وإني والله لأنظر إلى حوضي الآن، وإني…٦٥٩١كما بين المدينة وصنعاء٦٥٩٢سمع النبي صلى الله عليه وسلم قوله حوضه ما بين صنعاء والمدينة . فقال…٦٥٩٣إني على الحوض حتى أنظر من يرد على منكم، وسيؤخذ ناس دوني فأقول يا رب…
