كتاب فرض الخمس

٦٣ حديثًا
٣٠٩١أن حسين بن علي، عليهما السلام أخبره أن عليا قال كانت لي شارف من نصيبي…٣٠٩٣لا نورث ما تركنا صدقة٣٠٩٤لا نورث ما تركنا صدقة٣٠٩٥آمركم بأربع، وأنهاكم عن أربع، الإيمان بالله شهادة أن لا إله إلا الله…٣٠٩٦لا يقتسم ورثتي دينارا، ما تركت بعد نفقة نسائي ومئونة عاملي فهو صدقة٣٠٩٧قالت توفي رسول الله صلى الله عليه وسلم وما في بيتي من شىء يأكله ذو…٣٠٩٨قال ما ترك النبي صلى الله عليه وسلم إلا سلاحه وبغلته البيضاء، وأرضا…٣٠٩٩ومحمد، قالا أخبرنا عبد الله، أخبرنا معمر، ويونس، عن الزهري، قال…٣١٠٠قال قالت عائشة رضى الله عنها توفي النبي صلى الله عليه وسلم في بيتي،…٣١٠١أن صفية، زوج النبي صلى الله عليه وسلم أخبرته أنها جاءت رسول الله صلى…٣١٠٢فرأيت النبي صلى الله عليه وسلم يقضي حاجته، مستدبر القبلة، مستقبل…٣١٠٣أن عائشة رضى الله عنها قالت كان رسول الله صلى الله عليه وسلم يصلي…٣١٠٤هنا الفتنة ثلاثا من حيث يطلع قرن الشيطان٣١٠٥أراه فلانا، لعم حفصة من الرضاعة، الرضاعة تحرم ما تحرم الولادة٣١٠٦أن أبا بكر رضى الله عنه لما استخلف بعثه إلى البحرين، وكتب له هذا…٣١٠٧حدثني عبد الله بن محمد، حدثنا محمد بن عبد الله الأسدي، حدثنا عيسى بن…٣١٠٨وكساء من هذه التي يدعونها الملبدة .٣١٠٩النبي صلى الله عليه وسلم انكسر، فاتخذ مكان الشعب سلسلة من فضة . قال…٣١١٠إن فاطمة مني، وأنا أتخوف أن تفتن في دينها٣١١١قال لو كان علي رضى الله عنه ذاكرا عثمان رضى الله عنه ذكره يوم جاءه…٣١١٢قال أرسلني أبي، خذ هذا الكتاب فاذهب به إلى عثمان، فإن فيه أمر النبي…٣١١٣أن فاطمة عليها السلام اشتكت ما تلقى من الرحى مما تطحن، فبلغها أن رسول…٣١١٤سموا باسمي، ولا تكنوا بكنيتي، فإني إنما جعلت قاسما أقسم بينكم٣١١٥أحسنت الأنصار، سموا باسمي، ولا تكنوا بكنيتي، فإنما أنا قاسم٣١١٦من يرد الله به خيرا يفقهه في الدين، والله المعطي وأنا القاسم، ولا…٣١١٧ما أعطيكم ولا أمنعكم، أنا قاسم أضع حيث أمرت٣١١٨إن رجالا يتخوضون في مال الله بغير حق، فلهم النار يوم القيامة٣١١٩الخيل معقود في نواصيها الخير الأجر والمغنم إلى يوم القيامة٣١٢٠إذا هلك كسرى فلا كسرى بعده، وإذا هلك قيصر فلا قيصر بعده، والذي نفسي…٣١٢١إذا هلك كسرى فلا كسرى بعده، وإذا هلك قيصر فلا قيصر بعده، والذي نفسي…٣١٢٢أحلت لي الغنائم٣١٢٣تكفل الله لمن جاهد في سبيله، لا يخرجه إلا الجهاد في سبيله وتصديق…٣١٢٤غزا نبي من الأنبياء فقال لقومه لا يتبعني رجل ملك بضع امرأة وهو يريد…٣١٢٥قال قال عمر رضى الله عنه لولا آخر المسلمين ما فتحت قرية إلا قسمتها…٣١٢٦من قاتل لتكون كلمة الله هي العليا فهو في سبيل الله٣١٢٧يا أبا المسور، خبأت هذا لك، يا أبا المسور، خبأت هذا لك٣١٢٨فكان بعد ذلك يرد عليهم .٣١٢٩فقمت إلى جنبه فقال يا بنى، إنه لا يقتل اليوم إلا ظالم أو مظلوم، وإني…٣١٣٠إن لك أجر رجل ممن شهد بدرا وسهمه٣١٣٢أحب الحديث إلى أصدقه، فاختاروا إحدى الطائفتين إما السبى وإما المال،…٣١٣٣والله لا أحملكم، وما عندي ما أحملكم٣١٣٤فغنموا إبلا كثيرا، فكانت سهامهم اثنى عشر بعيرا أو أحد عشر بعيرا،…٣١٣٥حدثنا يحيى بن بكير، أخبرنا الليث، عن عقيل، عن ابن شهاب، عن سالم، عن…٣١٣٦أحدهما أبو بردة والآخر أبو رهم، إما قال في بضع، وإما قال في ثلاثة…٣١٣٧لو قد جاءني مال البحرين لقد أعطيتك هكذا وهكذا وهكذا٣١٣٨شقيت إن لم أعدل٣١٣٩لو كان المطعم بن عدي حيا، ثم كلمني في هؤلاء النتنى، لتركتهم له٣١٤٠إنما بنو المطلب وبنو هاشم شىء واحد٣١٤١وشمالي، فإذا أنا بغلامين، من الأنصار حديثة أسنانهما، تمنيت أن أكون…٣١٤٢من قتل قتيلا له عليه بينة فله سلبه٣١٤٣يا حكيم، إن هذا المال خضر حلو، فمن أخذه بسخاوة نفس بورك له فيه، ومن…٣١٤٤أن عمر بن الخطاب رضى الله عنه قال يا رسول الله إنه كان على اعتكاف يوم…٣١٤٥إني أعطي قوما أخاف ظلعهم وجزعهم، وأكل أقواما إلى ما جعل الله في…٣١٤٦إني أعطي قريشا أتألفهم، لأنهم حديث عهد بجاهلية٣١٤٧ما كان حديث بلغني عنكم٣١٤٨أعطوني ردائي، فلو كان عدد هذه العضاه نعما لقسمته بينكم، ثم لا تجدوني…٣١٤٩فأدركه أعرابي فجذبه جذبة شديدة، حتى نظرت إلى صفحة عاتق النبي صلى الله…٣١٥٠فمن يعدل إذا لم يعدل الله ورسوله رحم الله موسى قد أوذي بأكثر من هذا…٣١٥١حدثنا محمود بن غيلان، حدثنا أبو أسامة، حدثنا هشام، قال أخبرني أبي، عن…٣١٥٢نقركم على ذلك ما شئنا٣١٥٣فرمى إنسان بجراب فيه شحم، فنزوت لآخذه، فالتفت فإذا النبي صلى الله…٣١٥٤حدثنا مسدد، حدثنا حماد بن زيد، عن أيوب، عن نافع، عن ابن عمر رضى الله…٣١٥٥فانتحرناها فلما غلت القدور، نادى منادي رسول الله صلى الله عليه وسلم…
