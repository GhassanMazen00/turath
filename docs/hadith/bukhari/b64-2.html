٤١٠٧أن ابن عمر رضى الله عنهما قال أول يوم شهدته يوم الخندق .٤١٠٨قال دخلت على حفصة ونسواتها تنطف، قلت قد كان من أمر الناس ما ترين، فلم…٤١٠٩نغزوهم ولا يغزوننا٤١١٠الآن نغزوهم ولا يغزوننا، نحن نسير إليهم٤١١١ملأ الله عليهم بيوتهم وقبورهم نارا كما شغلونا عن صلاة الوسطى حتى غابت…٤١١٢والله ما صليتها٤١١٣من يأتينا بخبر القوم٤١١٤لا إله إلا الله وحده، أعز جنده، ونصر عبده وغلب الأحزاب وحده، فلا شىء…٤١١٥اللهم منزل الكتاب، سريع الحساب، اهزم الأحزاب، اللهم اهزمهم وزلزلهم٤١١٦لا إله إلا الله وحده لا شريك له، له الملك وله الحمد، وهو على كل شىء…٤١١٧أتاه جبريل عليه السلام فقال قد وضعت السلاح والله ما وضعناه، فاخرج…٤١١٨حدثنا موسى، حدثنا جرير بن حازم، عن حميد بن هلال، عن أنس رضى الله عنه…٤١١٩لا يصلين أحد العصر إلا في بني قريظة٤١٢٠وإن أهلي أمروني أن آتي النبي صلى الله عليه وسلم فأسأله الذين كانوا…٤١٢١قوموا إلى سيدكم أو خيركم٤١٢٢رماه رجل من قريش يقال له حبان ابن العرقة، رماه في الأكحل، فضرب النبي…٤١٢٣اهجهم أو هاجهم وجبريل معك٤١٢٤اهج المشركين، فإن جبريل معك٤١٢٥وقال عبد الله بن رجاء أخبرنا عمران القطان، عن يحيى بن أبي كثير، عن…٤١٢٦أن جابرا، حدثهم صلى النبي، صلى الله عليه وسلم بهم يوم محارب وثعلبة .٤١٢٧خرج النبي صلى الله عليه وسلم إلى ذات الرقاع من نخل فلقي جمعا من…٤١٢٨فسميت غزوة ذات الرقاع، لما كنا نعصب من الخرق على أرجلنا، وحدث أبو…٤١٢٩عمن شهد رسول الله صلى الله عليه وسلم يوم ذات الرقاع صلى صلاة الخوف أن…٤١٣٠وقال معاذ حدثنا هشام، عن أبي الزبير، عن جابر، قال كنا مع النبي صلى…٤١٣١قال يقوم الإمام مستقبل القبلة، وطائفة منهم معه وطائفة من قبل العدو،…٤١٣٢أن ابن عمر رضى الله عنهما قال غزوت مع رسول الله صلى الله عليه وسلم…٤١٣٣أن رسول الله صلى الله عليه وسلم صلى بإحدى الطائفتين، والطائفة الأخرى…٤١٣٤وأبو سلمة أن جابرا، أخبر أنه، غزا مع رسول الله صلى الله عليه وسلم قبل…٤١٣٥إن هذا اخترط سيفي، وأنا نائم فاستيقظت، وهو في يده صلتا، فقال لي من…٤١٣٦فجاء رجل من المشركين وسيف النبي صلى الله عليه وسلم معلق بالشجرة…٤١٣٧وقال أبو الزبير عن جابر، كنا مع النبي صلى الله عليه وسلم بنخل فصلى…٤١٣٨ما عليكم أن لا تفعلوا، ما من نسمة كائنة إلى يوم القيامة إلا وهى كائنة٤١٣٩إن هذا أتاني وأنا نائم، فاخترط سيفي فاستيقظت، وهو قائم على رأسي،…٤١٤٠قال رأيت النبي صلى الله عليه وسلم في غزوة أنمار يصلي على راحلته،…٤١٤١وسعيد بن المسيب، وعلقمة بن وقاص، وعبيد الله بن عبد الله بن عتبة بن…٤١٤٢قال قال لي الوليد بن عبد الملك أبلغك أن عليا، كان فيمن قذف عائشة قلت…٤١٤٣فما أفاقت إلا وعليها حمى بنافض، فطرحت عليها ثيابها فغطيتها . فجاء…٤١٤٤حدثني يحيى، حدثنا وكيع، عن نافع بن عمر، عن ابن أبي مليكة، عن عائشة…٤١٤٥قال ذهبت أسب حسان عند عائشة فقالت لا تسبه، فإنه كان ينافح عن رسول…٤١٤٦حدثني بشر بن خالد، أخبرنا محمد بن جعفر، عن شعبة، عن سليمان، عن أبي…٤١٤٧أتدرون ماذا قال ربكم٤١٤٨أن أنسا رضى الله عنه أخبره قال اعتمر رسول الله صلى الله عليه وسلم…٤١٤٩أن أباه، حدثه قال انطلقنا مع النبي صلى الله عليه وسلم عام الحديبية…٤١٥٠وقد كان فتح مكة فتحا، ونحن نعد الفتح بيعة الرضوان يوم الحديبية . كنا…٤١٥١ائتوني بدلو من مائها٤١٥٢فتوضأ منها، ثم أقبل الناس نحوه فقال رسول الله صلى الله عليه وسلم " ما…٤١٥٣قلت لسعيد بن المسيب بلغني أن جابر بن عبد الله، كان يقول كانوا أربع…٤١٥٤أنتم خير أهل الأرض٤١٥٥وقال عبيد الله بن معاذ حدثنا أبي، حدثنا شعبة، عن عمرو بن مرة، حدثني…٤١٥٦أنه سمع مرداسا الأسلمي، يقول وكان من أصحاب الشجرة يقبض الصالحون الأول…٤١٥٨والمسور بن مخرمة، قالا خرج النبي صلى الله عليه وسلم عام الحديبية في…٤١٥٩ورقاء، عن ابن أبي نجيح، عن مجاهد، قال حدثني عبد الرحمن بن أبي ليلى،…٤١٦١قال خرجت مع عمر بن الخطاب رضى الله عنه إلى السوق، فلحقت عمر امرأة…٤١٦٢قال لقد رأيت الشجرة، ثم أتيتها بعد فلم أعرفها . قال محمود ثم أنسيتها…٤١٦٣قال انطلقت حاجا فمررت بقوم يصلون قلت ما هذا المسجد قالوا هذه الشجرة،…٤١٦٤أنه كان ممن بايع تحت الشجرة، فرجعنا إليها العام المقبل فعميت علينا .٤١٦٥شهدها .٤١٦٦اللهم صل عليهم٤١٦٧قال لما كان يوم الحرة والناس يبايعون لعبد الله بن حنظلة فقال ابن زيد…٤١٦٨وليس للحيطان ظل نستظل فيه .٤١٦٩قال قلت لسلمة بن الأكوع على أى شىء بايعتم رسول الله صلى الله عليه…٤١٧٠حدثني أحمد بن إشكاب، حدثنا محمد بن فضيل، عن العلاء بن المسيب، عن…٤١٧١أن ثابت بن الضحاك، أخبره أنه، بايع النبي صلى الله عليه وسلم تحت…٤١٧٢حدثني أحمد بن إسحاق، حدثنا عثمان بن عمر، أخبرنا شعبة، عن قتادة، عن…٤١٧٣حدثنا عبد الله بن محمد، حدثنا أبو عامر، حدثنا إسرائيل، عن مجزأة بن…٤١٧٤منهم من أصحاب الشجرة اسمه أهبان بن أوس وكان اشتكى ركبته، وكان إذا سجد…٤١٧٥حدثني محمد بن بشار، حدثنا ابن أبي عدي، عن شعبة، عن يحيى بن سعيد، عن…٤١٧٦قال سألت عائذ بن عمرو رضى الله عنه وكان من أصحاب النبي صلى الله عليه…٤١٧٧لقد أنزلت على الليلة سورة لهي أحب إلى مما طلعت عليه الشمس، ثم قرأ {…٤١٧٩أشيروا أيها الناس على، أترون أن أميل إلى عيالهم وذراري هؤلاء الذين…٤١٨١أنه سمع مروان بن الحكم، والمسور بن مخرمة، يخبران خبرا من خبر رسول…٤١٨٢قال ابن شهاب وأخبرني عروة بن الزبير، أن عائشة رضى الله عنها زوج النبي…٤١٨٣حدثنا قتيبة، عن مالك، عن نافع، أن عبد الله بن عمر رضى الله عنهما خرج…٤١٨٤أنه أهل وقال إن حيل بيني وبينه لفعلت كما فعل النبي صلى الله عليه وسلم…٤١٨٥أشهدكم أني أوجبت عمرة٤١٨٦سمع النضر بن محمد، حدثنا صخر، عن نافع، قال إن الناس يتحدثون أن ابن…٤١٨٧كانوا مع النبي صلى الله عليه وسلم يوم الحديبية، تفرقوا في ظلال الشجر،…٤١٨٨وسعى بين الصفا والمروة، فكنا نستره من أهل مكة، لا يصيبه أحد بشىء .٤١٨٩قال قال أبو وائل لما قدم سهل بن حنيف من صفين أتيناه نستخبره فقال…٤١٩٠فاحلق، وصم ثلاثة أيام، أو أطعم ستة مساكين، أو انسك نسيكة٤١٩١أيؤذيك هوام رأسك٤١٩٢ولم نكن أهل ريف . واستوخموا المدينة، فأمرهم رسول الله صلى الله عليه…٤١٩٣والحجاج الصواف، قال حدثني أبو رجاء، مولى أبي قلابة وكان معه بالشأم أن…٤١٩٤يا ابن الأكوع، ملكت فأسجح٤١٩٥أن سويد بن النعمان، أخبره أنه، خرج مع النبي صلى الله عليه وسلم عام…٤١٩٦ثم إن الله تعالى فتحها عليهم، فلما أمسى الناس مساء اليوم الذي فتحت…٤١٩٧خربت خيبر، إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٤١٩٨الله أكبر خربت خيبر، إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٤١٩٩ثم أتاه الثانية فقال أكلت الحمر . فسكت، ثم الثالثة فقال أفنيت الحمر .…٤٢٠٠الله أكبر خربت خيبر، إنا إذا نزلنا بساحة قوم، فساء صباح المنذرين٤٢٠١فأعتقها وتزوجها . فقال ثابت لأنس ما أصدقها قال أصدقها نفسها فأعتقها .٤٢٠٢أما إنه من أهل النار٤٢٠٣هذا من أهل النار٤٢٠٤وعبد الرحمن بن عبد الله بن كعب، أن أبا هريرة، قال شهدنا مع النبي صلى…٤٢٠٥اربعوا على أنفسكم، إنكم لا تدعون أصم ولا غائبا، إنكم تدعون سميعا…٤٢٠٦قال رأيت أثر ضربة في ساق سلمة، فقلت يا أبا مسلم، ما هذه الضربة قال…٤٢٠٧إنه من أهل النار٤٢٠٨قال نظر أنس إلى الناس يوم الجمعة، فرأى طيالسة فقال كأنهم الساعة يهود…٤٢٠٩لأعطين الراية غدا أو ليأخذن الراية غدا رجل يحبه الله ورسوله، يفتح عليه٤٢١٠لأعطين هذه الراية غدا رجلا، يفتح الله على يديه، يحب الله ورسوله،…٤٢١١فلما فتح الله عليه الحصن ذكر له جمال صفية بنت حيى بن أخطب، وقد قتل…٤٢١٢سمع أنس بن مالك رضى الله عنه أن النبي صلى الله عليه وسلم أقام على…٤٢١٣أنه سمع أنسا رضى الله عنه يقول أقام النبي صلى الله عليه وسلم بين خيبر…٤٢١٤فنزوت لآخذه، فالتفت فإذا النبي صلى الله عليه وسلم�� فاستحييت .٤٢١٥وسالم، عن ابن عمر رضى الله عنهما أن رسول الله صلى الله عليه وسلم نهى…٤٢١٦والحسن، ابنى محمد بن علي عن أبيهما، عن علي بن أبي طالب رضى الله عنه…٤٢١٧حدثنا محمد بن مقاتل، أخبرنا عبد الله، حدثنا عبيد الله بن عمر، عن…٤٢١٨وسالم، عن ابن عمر رضى الله عنهما قال نهى النبي صلى الله عليه وسلم عن…٤٢١٩ورخص في الخيل .٤٢٢٠لأنها كانت تأكل العذرة .٤٢٢٢وعبد الله بن أبي أوفى، رضى الله عنهم أنهم كانوا مع النبي صلى الله…٤٢٢٤وابن أبي أوفى، يحدثان عن النبي صلى الله عليه وسلم أنه قال يوم خيبر…٤٢٢٥حدثنا مسلم، حدثنا شعبة، عن عدي بن ثابت، عن البراء، قال غزونا مع النبي…٤٢٢٦حدثني إبراهيم بن موسى، أخبرنا ابن أبي زائدة، أخبرنا عاصم، عن عامر، عن…٤٢٢٧فكره أن تذهب حمولتهم، أو حرمه في يوم خيبر، لحم الحمر الأهلية .٤٢٢٨وللراجل سهما . قال فسره نافع فقال إذا كان مع الرجل فرس فله ثلاثة…٤٢٢٩إنما بنو هاشم وبنو المطلب شىء واحد٤٢٣٠أحدهما أبو بردة، والآخر أبو رهم إما قال بضع وإما قال في ثلاثة وخمسين…٤٢٣١فلما جاء النبي صلى الله عليه وسلم قالت يا نبي الله إن عمر قال كذا…٤٢٣٢إني لأعرف أصوات رفقة الأشعريين بالقرآن، حين يدخلون بالليل، وأعرف…٤٢٣٣سمع حفص بن غياث، حدثنا بريد بن عبد الله، عن أبي بردة، عن أبي موسى،…٤٢٣٤بلى والذي نفسي بيده، إن الشملة التي أصابها يوم خيبر من المغانم لم…٤٢٣٥أنه سمع عمر بن الخطاب رضى الله عنه يقول أما والذي نفسي بيده، لولا أن…٤٢٣٦كما قسم النبي صلى الله عليه وسلم خيبر .٤٢٣٧وسأله، إسماعيل بن أمية قال أخبرني عنبسة بن سعيد، أن أبا هريرة رضى…٤٢٣٨أنه سمع أبا هريرة، يخبر سعيد بن العاصي قال بعث رسول الله صلى الله…٤٢٣٩أن أبان بن سعيد، أقبل إلى النبي صلى الله عليه وسلم فسلم عليه، فقال…٤٢٤١لا نورث، ما تركنا صدقة، إنما يأكل آل محمد صلى الله عليه وسلم في هذا…٤٢٤٢حدثني محمد بن بشار، حدثنا حرمي، حدثنا شعبة، قال أخبرني عمارة، عن…٤٢٤٣حدثنا الحسن، حدثنا قرة بن حبيب، حدثنا عبد الرحمن بن عبد الله بن…٤٢٤٥كل تمر خيبر هكذا٤٢٤٧أن أبا سعيد، وأبا، هريرة حدثاه أن النبي صلى الله عليه وسلم بعث أخا…٤٢٤٨ولهم شطر ما يخرج منها .٤٢٤٩حدثنا عبد الله بن يوسف، حدثنا الليث، حدثني سعيد، عن أبي هريرة رضى…٤٢٥٠إن تطعنوا في إمارته، فقد طعنتم في إمارة أبيه من قبله، وايم الله لقد…٤٢٥١أنا رسول الله، وأنا محمد بن عبد الله٤٢٥٢فحال كفار قريش بينه وبين البيت، فنحر هديه، وحلق رأسه بالحديبية،…٤٢٥٤عائشة قال عروة يا أم المؤمنين ألا تسمعين ما يقول أبو عبد الرحمن إن…٤٢٥٥سمع ابن أبي أوفى، يقول لما اعتمر رسول الله صلى الله عليه وسلم سترناه…٤٢٥٦وأن يمشوا ما بين الركنين، ولم يمنعه أن يأمرهم أن يرملوا الأشواط كلها…٤٢٥٧حدثني محمد، عن سفيان بن عيينة، عن عمرو، عن عطاء، عن ابن عباس رضى الله…٤٢٥٨قال تزوج النبي صلى الله عليه وسلم ميمونة وهو محرم، وبنى بها وهو حلال…٤٢٥٩وأبان بن صالح، عن عطاء، ومجاهد، عن ابن عباس، قال تزوج النبي صلى الله…٤٢٦٠أن ابن عمر، أخبره أنه، وقف على جعفر يومئذ وهو قتيل، فعددت به خمسين…٤٢٦١إن قتل زيد فجعفر، وإن قتل جعفر فعبد الله بن رواحة٤٢٦٢أخذ الراية زيد فأصيب، ثم أخذ جعفر فأصيب، ثم أخذ ابن رواحة فأصيب…٤٢٦٣فاحث في أفواههن من التراب٤٢٦٤قال كان ابن عمر إذا حيا ابن جعفر قال السلام عليك يا ابن ذي الجناحين .٤٢٦٥يقول لقد انقطعت في يدي يوم موتة تسعة أسياف، فما بقي في يدي إلا صفيحة…٤٢٦٦يقول لقد دق في يدي يوم موتة تسعة أسياف، وصبرت في يدي صفيحة لي يمانية .
