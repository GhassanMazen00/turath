٦٤١٢نعمتان مغبون فيهما كثير من الناس، الصحة والفراغ٦٤١٣اللهم لا عيش إلا عيش الآخرة، فأصلح الأنصار والمهاجرة٦٤١٤اللهم لا عيش إلا عيش الآخره، فاغفر للأنصار والمهاجره٦٤١٥موضع سوط في الجنة خير من الدنيا وما فيها، ولغدوة في سبيل الله أو روحة…٦٤١٦كن في الدنيا كأنك غريب، أو عابر سبيل٦٤١٧هذا الإنسان، وهذا أجله محيط به أو قد أحاط به وهذا الذي هو خارج أمله،…٦٤١٨هذا الأمل وهذا أجله، فبينما هو كذلك إذ جاءه الخط الأقرب٦٤١٩أعذر الله إلى امرئ أخر أجله حتى بلغه ستين سنة٦٤٢٠لا يزال قلب الكبير شابا في اثنتين في حب الدنيا، وطول الأمل٦٤٢١يكبر ابن آدم ويكبر معه اثنان حب المال، وطول العمر٦٤٢٢وزعم، محمود أنه عقل رسول الله صلى الله عليه وسلم وقال وعقل مجة مجها…٦٤٢٣لن يوافي عبد يوم القيامة يقول لا إله إلا الله . يبتغي به وجه الله،…٦٤٢٤يقول الله تعالى ما لعبدي المؤمن عندي جزاء، إذا قبضت صفيه من أهل…٦٤٢٥أظنكم سمعتم بقدوم أبي عبيدة، وأنه جاء بشىء٦٤٢٦إني فرطكم وأنا شهيد عليكم، وإني والله لأنظر إلى حوضي الآن، وإني قد…٦٤٢٧إن أكثر ما أخاف عليكم ما يخرج الله لكم من بركات الأرض٦٤٢٨خيركم قرني، ثم الذين يلونهم، ثم الذين يلونهم٦٤٢٩خير الناس قرني، ثم الذين يلونهم، ثم الذين يلونهم، ثم يجيء من بعدهم…٦٤٣٠إن أصحاب محمد صلى الله عليه وسلم مضوا ولم تنقصهم الدنيا بشىء، وإنا…٦٤٣١لا نجد له موضعا إلا التراب .٦٤٣٢حدثنا محمد بن كثير، عن سفيان، عن الأعمش، عن أبي وائل، عن خباب رضى…٦٤٣٣من توضأ مثل هذا الوضوء، ثم أتى المسجد فركع ركعتين، ثم جلس، غفر له ما…٦٤٣٤يذهب الصالحون الأول فالأول، ويبقى حفالة كحفالة الشعير أو التمر، لا…٦٤٣٥تعس عبد الدينار والدرهم والقطيفة والخميصة، إن أعطي رضي، وإن لم يعط لم…٦٤٣٦لو كان لابن آدم واديان من مال لابتغى ثالثا، ولا يملأ جوف ابن آدم إلا…٦٤٣٧لو أن لابن آدم مثل واد مالا لأحب أن له إليه مثله، ولا يملأ عين ابن…٦٤٣٨لو أن ابن آدم أعطي واديا ملأ من ذهب أحب إليه ثانيا، ولو أعطي ثانيا…٦٤٣٩لو أن لابن آدم واديا من ذهب أحب أن يكون له واديان، ولن يملأ فاه إلا…٦٤٤٠وقال لنا أبو الوليد حدثنا حماد بن سلمة، عن ثابت، عن أنس، عن أبى، قال…٦٤٤١هذا المال وربما قال سفيان قال لي يا حكيم إن هذا المال خضرة حلوة، فمن…٦٤٤٢أيكم مال وارثه أحب إليه من ماله٦٤٤٣وليس معه إنسان قال فظننت أنه يكره أن يمشي معه أحد قال فجعلت أمشي في…٦٤٤٤تمضي على ثالثة وعندي منه دينار، إلا شيئا أرصده لدين، إلا أن أقول به…٦٤٤٥لو كان لي مثل أحد ذهبا لسرني أن لا تمر على ثلاث ليال وعندي منه شىء،…٦٤٤٦ليس الغنى عن كثرة العرض، ولكن الغنى غنى النفس٦٤٤٧أنه قال مر رجل على رسول الله صلى الله عليه وسلم فقال لرجل عنده جالس "…٦٤٤٨منهم مصعب بن عمير قتل يوم أحد، وترك نمرة فإذا غطينا رأسه بدت رجلاه،…٦٤٤٩اطلعت في الجنة فرأيت أكثر أهلها الفقراء، واطلعت في النار فرأيت أكثر…٦٤٥٠وما أكل خبزا مرققا حتى مات .٦٤٥١إلا شطر شعير في رف لي، فأكلت منه حتى طال على، فكلته، ففني .٦٤٥٢أن أبا هريرة، كان يقول آلله الذي لا إله إلا هو إن كنت لأعتمد بكبدي…٦٤٥٣يقول إني لأول العرب رمى بسهم في سبيل الله، ورأيتنا نغزو، وما لنا طعام…٦٤٥٤قالت ما شبع آل محمد صلى الله عليه وسلم منذ قدم المدينة من طعام بر…٦٤٥٥إلا إحداهما تمر .٦٤٥٦قالت كان فراش رسول الله صلى الله عليه وسلم من أدم، وحشوه من ليف .٦٤٥٧حتى لحق بالله، ولا رأى شاة سميطا بعينه قط .٦٤٥٨إنما هو التمر والماء، إلا أن نؤتى باللحيم .٦٤٥٩وما أوقدت في أبيات رسول الله صلى الله عليه وسلم نار . فقلت ما كان…٦٤٦٠اللهم ارزق آل محمد قوتا٦٤٦١قال سألت عائشة رضى الله عنها أى العمل كان أحب إلى النبي صلى الله عليه…٦٤٦٢أنها قالت كان أحب العمل إلى رسول الله صلى الله عليه وسلم الذي يدوم…٦٤٦٣لن ينجي أحدا منكم عمله٦٤٦٤سددوا وقاربوا، واعلموا أن لن يدخل أحدكم عمله الجنة، وأن أحب الأعمال…٦٤٦٥حدثني محمد بن عرعرة، حدثنا شعبة، عن سعد بن إبراهيم، عن أبي سلمة، عن…٦٤٦٦قال سألت أم المؤمنين عائشة قلت يا أم المؤمنين كيف كان عمل النبي صلى…٦٤٦٧سددوا وقاربوا، وأبشروا، فإنه لا يدخل أحدا الجنة عمله٦٤٦٨قد أريت الآن منذ صليت لكم الصلاة الجنة والنار ممثلتين في قبل هذا…٦٤٦٩إن الله خلق الرحمة يوم خلقها مائة رحمة، فأمسك عنده تسعا وتسعين رحمة،…٦٤٧٠ما يكن عندي من خير لا أدخره عنكم، وإنه من يستعف يعفه الله، ومن يتصبر…٦٤٧١أفلا أكون عبدا شكورا٦٤٧٢يدخل الجنة من أمتي سبعون ألفا بغير حساب، هم الذين لا يسترقون، ولا…٦٤٧٣لا إله إلا الله، وحده لا شريك له، له الملك، وله الحمد، وهو على كل شىء…٦٤٧٤من يضمن لي ما بين لحييه وما بين رجليه أضمن له الجنة٦٤٧٥من كان يؤمن بالله واليوم الآخر فليقل خيرا، أو ليصمت، ومن كان يؤمن…٦٤٧٦الضيافة ثلاثة أيام جائزته٦٤٧٧إن العبد ليتكلم بالكلمة ما يتبين فيها، يزل بها في النار أبعد مما بين…٦٤٧٨إن العبد ليتكلم بالكلمة من رضوان الله لا يلقي لها بالا، يرفع الله بها…٦٤٧٩سبعة يظلهم الله، رجل ذكر الله ففاضت عيناه٦٤٨٠كان رجل ممن كان قبلكم يسيء الظن بعمله، فقال لأهله إذا أنا مت فخذوني…٦٤٨١ذكر رجلا فيمن كان سلف أو قبلكم آتاه الله مالا وولدا يعني أعطاه قال…٦٤٨٢مثلي ومثل ما بعثني الله كمثل رجل أتى قوما فقال رأيت الجيش بعينى، وإني…٦٤٨٣إنما مثلي ومثل الناس كمثل رجل استوقد نارا، فلما أضاءت ما حوله جعل…٦٤٨٤المسلم من سلم المسلمون من لسانه ويده، والمهاجر من هجر ما نهى الله عنه٦٤٨٥لو تعلمون ما أعلم لضحكتم قليلا، ولبكيتم كثيرا٦٤٨٦لو تعلمون ما أعلم لضحكتم قليلا ولبكيتم كثيرا٦٤٨٧حجبت النار بالشهوات، وحجبت الجنة بالمكاره٦٤٨٨الجنة أقرب إلى أحدكم من شراك نعله، والنار مثل ذلك٦٤٨٩أصدق بيت قاله الشاعر ألا كل شىء ما خلا الله باطل٦٤٩٠إذا نظر أحدكم إلى من فضل عليه في المال والخلق، فلينظر إلى من هو أسفل…٦٤٩١إن الله كتب الحسنات والسيئات، ثم بين ذلك فمن هم بحسنة فلم يعملها…٦٤٩٢حدثنا أبو الوليد، حدثنا مهدي، عن غيلان، عن أنس رضى الله عنه قال إنكم…٦٤٩٣من أحب أن ينظر إلى رجل من أهل النار فلينظر إلى هذا٦٤٩٤رجل جاهد بنفسه وماله، ورجل في شعب من الشعاب يعبد ربه، ويدع الناس من…٦٤٩٥يأتي على الناس زمان خير مال الرجل المسلم الغنم، يتبع بها شعف الجبال…٦٤٩٦إذا ضيعت الأمانة فانتظر الساعة٦٤٩٧أن الأمانة نزلت في جذر قلوب الرجال، ثم علموا من القرآن، ثم علموا من…٦٤٩٨إنما الناس كالإبل المائة لا تكاد تجد فيها راحلة٦٤٩٩من سمع سمع الله به، ومن يرائي يرائي الله به٦٥٠٠صلى الله عليه وسلم ليس بيني وبينه إلا آخرة الرحل فقال " يا معاذ " .…٦٥٠١إن حقا على الله أن لا يرفع شيئا من الدنيا إلا وضعه٦٥٠٢إن الله قال من عادى لي وليا فقد آذنته بالحرب، وما تقرب إلى عبدي بشىء…٦٥٠٣بعثت أنا والساعة هكذا٦٥٠٤بعثت أنا والساعة كهاتين٦٥٠٥بعثت أنا والساعة كهاتين٦٥٠٦لا تقوم الساعة حتى تطلع الشمس من مغربها، فإذا طلعت فرآها الناس آمنوا…٦٥٠٧من أحب لقاء الله أحب الله لقاءه، ومن كره لقاء الله كره الله لقاءه٦٥٠٨من أحب لقاء الله أحب الله لقاءه، ومن كره لقاء الله كره الله لقاءه٦٥٠٩إنه لم يقبض نبي قط حتى يرى مقعده من الجنة ثم يخير٦٥١٠لا إله إلا الله، إن للموت سكرات٦٥١١إن يعش هذا لا يدركه الهرم حتى تقوم عليكم ساعتكم٦٥١٢مستريح، ومستراح منه٦٥١٣مستريح، ومستراح منه، المؤمن يستريح٦٥١٤يتبع الميت ثلاثة، فيرجع اثنان ويبقى معه واحد، يتبعه أهله وماله وعمله،…٦٥١٥إذا مات أحدكم عرض عليه مقعده غدوة وعشيا، إما النار وإما الجنة، فيقال…٦٥١٦لا تسبوا الأموات، فإنهم قد أفضوا إلى ما قدموا٦٥١٧لا تخيروني على موسى، فإن الناس يصعقون يوم القيامة، فأكون في أول من…٦٥١٨يصعق الناس حين يصعقون، فأكون أول من قام، فإذا موسى آخذ بالعرش، فما…٦٥١٩يقبض الله الأرض، ويطوي السماء بيمينه، ثم يقول أنا الملك أين ملوك الأرض٦٥٢٠تكون الأرض يوم القيامة خبزة واحدة، يتكفؤها الجبار بيده، كما يكفأ…٦٥٢١يحشر الناس يوم القيامة على أرض بيضاء عفراء كقرصة نقي٦٥٢٢يحشر الناس على ثلاث طرائق، راغبين راهبين واثنان على بعير، وثلاثة على…٦٥٢٣أليس الذي أمشاه على الرجلين في الدنيا قادرا على أن يمشيه على وجهه يوم…٦٥٢٤إنكم ملاقو الله حفاة عراة مشاة غرلا٦٥٢٥إنكم ملاقو الله حفاة عراة غرلا٦٥٢٦إنكم محشورون حفاة عراة { كما بدأنا أول خلق نعيده } الآية، وإن أول…٦٥٢٧تحشرون حفاة عراة غرلا٦٥٢٨أترضون أن تكونوا ربع أهل الجنة٦٥٢٩أول من يدعى يوم القيامة آدم، فتراءى ذريته فيقال هذا أبوكم آدم . فيقول…٦٥٣٠يقول الله يا آدم . فيقول لبيك وسعديك والخير في يديك . قال يقول أخرج…٦٥٣١يقوم أحدهم في رشحه إلى أنصاف أذنيه٦٥٣٢يعرق الناس يوم القيامة حتى يذهب عرقهم في الأرض سبعين ذراعا، ويلجمهم…٦٥٣٣أول ما يقضى بين الناس بالدماء٦٥٣٤من كانت عنده مظلمة لأخيه فليتحلله منها، فإنه ليس ثم دينار ولا درهم
