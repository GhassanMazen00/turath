٤٢٦٧فجعلت أخته عمرة تبكي واجبلاه واكذا واكذا . تعدد عليه فقال حين أفاق ما…٤٢٦٨قال أغمي على عبد الله بن رواحة بهذا، فلما مات لم تبك عليه .٤٢٦٩يا أسامة أقتلته بعد ما قال لا إله إلا الله٤٢٧٠يقول غزوت مع النبي صلى الله عليه وسلم سبع غزوات، وخرجت فيما يبعث من…٤٢٧١يقول غزوت مع النبي صلى الله عليه وسلم سبع غزوات، وخرجت فيما يبعث من…٤٢٧٢حدثنا أبو عاصم الضحاك بن مخلد، حدثنا يزيد، عن سلمة بن الأكوع رضى الله…٤٢٧٣قال غزوت مع النبي صلى الله عليه وسلم سبع غزوات . فذكر خيبر والحديبية…٤٢٧٤انطلقوا حتى تأتوا روضة خاخ، فإن بها ظعينة معها كتاب، فخذوا منها٤٢٧٥أن ابن عباس، أخبره أن رسول الله صلى الله عليه وسلم غزا غزوة الفتح في…٤٢٧٦ومعه عشرة آلاف، وذلك على رأس ثمان سنين ونصف من مقدمه المدينة، فسار هو…٤٢٧٧قال خرج النبي صلى الله عليه وسلم في رمضان إلى حنين، والناس مختلفون…٤٢٧٨وقال عبد الرزاق أخبرنا معمر، عن أيوب، عن عكرمة، عن ابن عباس رضى الله…٤٢٧٩قال سافر رسول الله صلى الله عليه وسلم في رمضان، فصام حتى بلغ عسفان،…٤٢٨٠احبس أبا سفيان عند حطم الخيل حتى ينظر إلى المسلمين٤٢٨١يقول رأيت رسول الله صلى الله عليه وسلم يوم فتح مكة على ناقته، وهو…٤٢٨٣وهل ترك لنا عقيل من منزل٤٢٨٤منزلنا إن شاء الله، إذا فتح الله الخيف، حيث تقاسموا على الكفر٤٢٨٥منزلنا غدا إن شاء الله بخيف بني كنانة، حيث تقاسموا على الكفر٤٢٨٦فلما نزعه جاء رجل فقال ابن خطل متعلق بأستار الكعبة . فقال " اقتله "…٤٢٨٧جاء الحق وزهق الباطل، جاء الحق، وما يبدئ الباطل وما يعيد٤٢٨٨قاتلهم الله لقد علموا ما استقسما بها قط٤٢٨٩مردفا أسامة بن زيد ومعه بلال ومعه عثمان بن طلحة، من الحجبة حتى أناخ…٤٢٩٠أن عائشة رضى الله عنها أخبرته أن النبي صلى الله عليه وسلم دخل عام…٤٢٩١دخل النبي صلى الله عليه وسلم عام الفتح من أعلى مكة من كداء .٤٢٩٢أنه رأى النبي صلى الله عليه وسلم يصلي الضحى غير أم هانئ، فإنها ذكرت…٤٢٩٣سبحانك اللهم، ربنا وبحمدك، اللهم اغفر لي٤٢٩٤فقال بعضهم لم تدخل هذا الفتى معنا، ولنا أبناء مثله فقال إنه ممن قد…٤٢٩٥إن مكة حرمها الله ولم يحرمها الناس، لا يحل لامرئ يؤمن بالله واليوم…٤٢٩٦إن الله ورسوله حرم بيع الخمر٤٢٩٧حدثنا أبو نعيم، حدثنا سفيان، . حدثنا قبيصة، حدثنا سفيان، عن يحيى بن…٤٢٩٨رضى الله عنهما قال أقام النبي صلى الله عليه وسلم بمكة تسعة عشر يوما…٤٢٩٩حدثنا أحمد بن يونس، حدثنا أبو شهاب، عن عاصم، عن عكرمة، عن ابن عباس،…٤٣٠٠وكان النبي، صلى الله عليه وسلم قد مسح وجهه عام الفتح .٤٣٠١مع ابن المسيب قال وزعم أبو جميلة أنه أدرك النبي صلى الله عليه وسلم،…٤٣٠٢صلوا صلاة كذا في حين كذا، وصلوا كذا في حين كذا، فإذا حضرت الصلاة،…٤٣٠٣هو لك، هو أخوك يا عبد بن زمعة٤٣٠٤أتكلمني في حد من حدود الله٤٣٠٦ذهب أهل الهجرة بما فيها٤٣٠٨مضت الهجرة لأهلها، أبايعه على الإسلام والجهاد.٤٣٠٩قلت لابن عمر رضى الله عنهما إني أريد أن أهاجر إلى الشأم . قال لا هجرة…٤٣١٠قلت لابن عمر فقال لا هجرة اليوم، أو بعد رسول الله صلى الله عليه وسلم…٤٣١١أن عبد الله بن عمر رضى الله عنهما كان يقول لا هجرة بعد الفتح .٤٣١٢فقالت لا هجرة اليوم، كان المؤمن يفر أحدهم بدينه إلى الله وإلى رسوله…٤٣١٣إن الله حرم مكة يوم خلق السموات والأرض، فهى حرام بحرام الله إلى يوم…٤٣١٤رأيت بيد ابن أبي أوفى ضربة، قال ضربتها مع النبي صلى الله عليه وسلم…٤٣١٥ولكن عجل سرعان القوم، فرشقتهم هوازن، وأبو سفيان بن الحارث آخذ برأس…٤٣١٦أنا النبي لا كذب أنا ابن عبد المطلب٤٣١٧كانت هوازن رماة، وإنا لما حملنا عليهم انكشفوا، فأكببنا على الغنائم،…٤٣١٩معي من ترون، وأحب الحديث إلى أصدقه، فاختاروا إحدى الطائفتين إما…٤٣٢٠أن عمر، قال يا رسول الله . حدثني محمد بن مقاتل أخبرنا عبد الله أخبرنا…٤٣٢١من قتل قتيلا له عليه بينة فله سلبه٤٣٢٢من أقام بينة على قتيل قتله فله سلبه٤٣٢٣اللهم اغفر لعبيد أبي عامر٤٣٢٤لا يدخلن هؤلاء عليكن٤٣٢٥إنا قافلون إن شاء الله٤٣٢٧من ادعى إلى غير أبيه وهو يعلم فالجنة عليه حرام٤٣٢٨ومج فيه، ثم قال " اشربا منه، وأفرغا على وجوهكما ونحوركما، وأبشرا " .…٤٣٢٩أين الذي يسألني عن العمرة آنفا٤٣٣٠يا معشر الأنصار ألم أجدكم ضلالا فهداكم الله بي، وكنتم متفرقين فألفكم…٤٣٣١ما حديث بلغني عنكم٤٣٣٢أما ترضون أن يذهب الناس بالدنيا، وتذهبون برسول الله صلى الله عليه وسلم٤٣٣٣لبيك نحن بين يديك، فنزل النبي صلى الله عليه وسلم فقال " أنا عبد الله…٤٣٣٤إن قريشا حديث عهد بجاهلية ومصيبة، وإني أردت أن أجبرهم وأتألفهم أما…٤٣٣٥رحمة الله على موسى، لقد أوذي بأكثر من هذا فصبر٤٣٣٦رحم الله موسى . قد أوذي بأكثر من هذا فصبر٤٣٣٧ومع النبي صلى الله عليه وسلم عشرة آلاف ومن الطلقاء، فأدبروا عنه حتى…٤٣٣٨فكنت فيها، فبلغت سهامنا اثنى عشر بعيرا، ونفلنا بعيرا بعيرا، فرجعنا…٤٣٣٩اللهم إني أبرأ إليك مما صنع خالد٤٣٤٠لو دخلوها ما خرجوا منها إلى يوم القيامة، الطاعة في المعروف٤٣٤٢يسرا ولا تعسرا، وبشرا ولا تنفرا٤٣٤٣فقال " وما هي " . قال البتع والمزر . فقلت لأبي بردة ما البتع قال نبيذ…٤٣٤٥يسرا ولا تعسرا، وبشرا ولا تنفرا، وتطاوعا٤٣٤٦أحججت يا عبد الله بن قيس٤٣٤٧إنك ستأتي قوما من أهل الكتاب، فإذا جئتهم فادعهم إلى أن يشهدوا أن لا…٤٣٤٨فقرأ معاذ في صلاة الصبح سورة النساء فلما قال { واتخذ الله إبراهيم…٤٣٤٩قال ثم بعث عليا بعد ذلك مكانه فقال مر أصحاب خالد، من شاء منهم أن يعقب…٤٣٥٠يا بريدة أتبغض عليا٤٣٥١ألا تأمنوني وأنا أمين من في السماء، يأتيني خبر السماء صباحا ومساء٤٣٥٢بم أهللت يا علي٤٣٥٤من لم يكن معه هدى فليجعلها عمرة٤٣٥٥ألا تريحني من ذي الخلصة٤٣٥٦ألا تريحني من ذي الخلصة٤٣٥٧ألا تريحني من ذي الخلصة٤٣٥٨أن رسول الله صلى الله عليه وسلم بعث عمرو بن العاص على جيش ذات السلاسل…٤٣٥٩قال كنت بالبحر فلقيت رجلين من أهل اليمن ذا كلاع وذا عمرو، فجعلت…٤٣٦٠فجمع فكان مزودى تمر، فكان يقوتنا كل يوم قليل قليل حتى فني، فلم يكن…٤٣٦١فسمي ذلك الجيش جيش الخبط، فألقى لنا البحر دابة يقال لها العنبر،…٤٣٦٢كلوا رزقا أخرجه الله، أطعمونا إن كان معكم٤٣٦٣أن أبا بكر الصديق رضى الله عنه بعثه في الحجة التي أمره النبي صلى الله…٤٣٦٤وآخر سورة نزلت خاتمة سورة النساء { يستفتونك قل الله يفتيكم في الكلالة…٤٣٦٥اقبلوا البشرى يا بني تميم٤٣٦٦هم أشد أمتي على الدجال٤٣٦٧أن ابن جريج، أخبرهم عن ابن أبي مليكة، أن عبد الله بن الزبير، أخبرهم…٤٣٦٨مرحبا بالقوم غير خزايا ولا الندامى٤٣٦٩آمركم بأربع وأنهاكم عن أربع، الإيمان بالله شهادة أن لا إله إلا الله…٤٣٧٠يا بنت أبي أمية، سألت عن الركعتين بعد العصر، إنه أتاني أناس من عبد…٤٣٧١رضى الله عنهما قال أول جمعة جمعت بعد جمعة جمعت في مسجد رسول الله صلى…٤٣٧٢ما عندك يا ثمامة٤٣٧٤لو سألتني هذه القطعة ما أعطيتكها ولن تعدو أمر الله فيك، ولئن أدبرت…٤٣٧٥بينا أنا نائم أتيت بخزائن الأرض، فوضع في كفي سواران من ذهب، فكبرا على…٤٣٧٧يقول كنت يوم بعث النبي صلى الله عليه وسلم غلاما أرعى الإبل على أهلي،…٤٣٧٩لو سألتني هذا القضيب ما أعطيتكه وإني لأراك الذي أريت فيه ما أريت،…٤٣٨٠لأبعثن معكم رجلا أمينا حق أمين٤٣٨١لأبعثن إليكم رجلا أمينا حق أمين٤٣٨٢لكل أمة أمين، وأمين هذه الأمة أبو عبيدة بن الجراح٤٣٨٣لو قد جاء مال البحرين لقد أعطيتك هكذا وهكذا ثلاثا٤٣٨٤وإسحاق بن نصر، قالا حدثنا يحيى بن آدم، حدثنا ابن أبي زائدة، عن أبيه،…٤٣٨٥أجل، ولكن لا أحلف على يمين فأرى غيرها خيرا منها إلا أتيت الذي هو خير…٤٣٨٦أبشروا يا بني تميم٤٣٨٧حدثني عبد الله بن محمد الجعفي، حدثنا وهب بن جرير، حدثنا شعبة، عن…٤٣٨٨أتاكم أهل اليمن، هم أرق أفئدة وألين قلوبا، الإيمان يمان والحكمة…٤٣٨٩الإيمان يمان، والفتنة ها هنا، ها هنا يطلع قرن الشيطان٤٣٩٠أتاكم أهل اليمن، أضعف قلوبا وأرق أفئدة، الفقه يمان، والحكمة يمانية٤٣٩١فجاء خباب فقال يا أبا عبد الرحمن، أيستطيع هؤلاء الشباب أن يقرءوا كما…٤٣٩٢اللهم اهد دوسا وأت بهم٤٣٩٣يا أبا هريرة هذا غلامك٤٣٩٤فجعل يدعو رجلا رجلا ويسميهم فقلت أما تعرفني يا أمير المؤمنين قال بلى،…٤٣٩٥من كان معه هدى فليهلل بالحج مع العمرة، ثم لا يحل حتى يحل منهما جميعا٤٣٩٦إذا طاف بالبيت فقد حل . فقلت من أين قال هذا ابن عباس قال من قول الله…٤٣٩٧وأتيت امرأة من قيس ففلت رأسي .٤٣٩٨لبدت رأسي وقلدت هديي، فلست أحل حتى أنحر هديي٤٣٩٩وقال، محمد بن يوسف حدثنا الأوزاعي، قال أخبرني ابن شهاب، عن سليمان بن…٤٤٠٠فجاءه بالمفتاح ففتح له الباب، فدخل النبي صلى الله عليه وسلم وأسامة…٤٤٠١وأبو سلمة بن عبد الرحمن أن عائشة، زوج النبي صلى الله عليه وسلم…٤٤٠٣ما بعث الله من نبي إلا أنذر أمته، أنذره نوح والنبيون من بعده، وإنه…٤٤٠٤أن النبي صلى الله عليه وسلم غزا تسع عشرة غزوة، وأنه حج بعد ما هاجر…٤٤٠٥أن النبي صلى الله عليه وسلم قال في حجة الوداع لجرير " استنصت الناس "…٤٤٠٦الزمان قد استدار كهيئة يوم خلق السموات والأرض، السنة اثنا عشر شهرا…٤٤٠٧أن أناسا، من اليهود قالوا لو نزلت هذه الآية فينا لاتخذنا ذلك اليوم…٤٤٠٨وأهل رسول الله صلى الله عليه وسلم بالحج، فأما من أهل بالحج أو جمع…٤٤٠٩قال عادني النبي صلى الله عليه وسلم في حجة الوداع من وجع، أشفيت منه…٤٤١٠أن ابن عمر رضى الله عنهما أخبرهم أن رسول الله صلى الله عليه وسلم حلق…٤٤١١أخبره ابن عمر، أن النبي صلى الله عليه وسلم حلق في حجة الوداع وأناس من…٤٤١٢أن عبد الله بن عباس رضى الله عنهما أخبره أنه، أقبل يسير على حمار،…٤٤١٣صلى الله عليه وسلم في حجته . فقال العنق، فإذا وجد فجوة نص .٤٤١٤أن باكين، أن يصيبكم مثل ما…٤٤٢١قال ذهب النبي صلى الله عليه وسلم لبعض حاجته، فقمت أسكب عليه الماء لا…٤٤٢٢هذه طابة، وهذا أحد، جبل يحبنا ونحبه٤٤٢٣إن بالمدينة أقواما ما سرتم مسيرا ولا قطعتم واديا إلا كانوا معكم٤٤٢٤أن ابن عباس، أخبره أن رسول الله صلى الله عليه وسلم بعث بكتابه إلى…٤٤٢٥لن يفلح قوم ولوا أمرهم امرأة٤٤٢٦يقول أذكر أني خرجت مع الغلمان إلى ثنية الوداع نتلقى رسول الله صلى…٤٤٢٧أذكر أني خرجت مع الصبيان نتلقى النبي صلى الله عليه وسلم إلى ثنية…٤٤٢٨يا عائشة ما أزال أجد ألم الطعام الذي أكلت بخيبر، فهذا أوان وجدت…
