كتاب الأطعمة

٩٥ حديثًا
٥٣٧٣أطعموا الجائع، وعودوا المريض، وفكوا العاني٥٣٧٤قال ما شبع آل محمد صلى الله عليه وسلم من طعام ثلاثة أيام حتى قبض .٥٣٧٥أصابني جهد شديد فلقيت عمر بن الخطاب، فاستقرأته آية من كتاب الله، فدخل…٥٣٧٦يا غلام سم الله، وكل بيمينك وكل مما يليك٥٣٧٧حدثني عبد العزيز بن عبد الله، قال حدثني محمد بن جعفر، عن محمد بن عمرو…٥٣٧٨سم الله، وكل مما يليك٥٣٧٩أنه سمع أنس بن مالك، يقول إن خياطا دعا رسول الله صلى الله عليه وسلم…٥٣٨٠حدثنا عبدان، أخبرنا عبد الله، أخبرنا شعبة، عن أشعث، عن أبيه، عن…٥٣٨١أرسلك أبو طلحة٥٣٨٢هل مع أحد منكم طعام٥٣٨٣حدثنا مسلم، حدثنا وهيب، حدثنا منصور، عن أمه، عن عائشة رضى الله عنها…٥٣٨٤فما أتي إلا بسويق، فلكناه فأكلنا منه، ثم دعا بماء فمضمض ومضمضنا، فصلى…٥٣٨٥حدثنا محمد بن سنان، حدثنا همام، عن قتادة، قال كنا عند أنس وعنده خباز…٥٣٨٦رضى الله عنه قال ما علمت النبي صلى الله عليه وسلم أكل على سكرجة قط،…٥٣٨٧أنه سمع أنسا، يقول قام النبي صلى الله عليه وسلم يبني بصفية فدعوت…٥٣٨٨قال كان أهل الشأم يعيرون ابن الزبير يقولون يا ابن ذات النطاقين .…٥٣٨٩فدعا بهن فأكلن على مائدته، وتركهن النبي صلى الله عليه وسلم كالمستقذر…٥٣٩٠أنه أخبره أنهم، كانوا مع النبي صلى الله عليه وسلم بالصهباء وهى على…٥٣٩١لا ولكن لم يكن بأرض قومي فأجدني أعافه٥٣٩٢طعام الاثنين كافي الثلاثة، وطعام الثلاثة كافي الأربعة٥٣٩٣المؤمن يأكل في معى واحد والكافر يأكل في سبعة أمعاء٥٣٩٤إن المؤمن يأكل في معى واحد، وإن الكافر أو المنافق فلا أدري أيهما قال…٥٣٩٥إن الكافر يأكل في سبعة أمعاء٥٣٩٦يأكل المسلم في معى واحد، والكافر يأكل في سبعة أمعاء٥٣٩٧إن المؤمن يأكل في معى واحد، والكافر يأكل في سبعة أمعاء٥٣٩٨يقول قال رسول الله صلى الله عليه وسلم " لا آكل متكئا " .٥٣٩٩لا آكل وأنا متكئ٥٤٠٠لا، ولكنه لا يكون بأرض قومي، فأجدني أعافه٥٤٠١سأفعل إن شاء الله٥٤٠٢فوضع الضب على مائدته، فلو كان حراما لم يوضع وشرب اللبن، وأكل الأقط .٥٤٠٣فتجعله في قدر لها، فتجعل فيه حبات من شعير، إذا صلينا زرناها فقربته…٥٤٠٤ثم قام فصلى، ولم يتوضأ .٥٤٠٥وعاصم، عن عكرمة، عن ابن عباس، قال انتشل النبي صلى الله عليه وسلم عرقا…٥٤٠٦حدثني محمد بن المثنى، قال حدثني عثمان بن عمر، حدثنا فليح، حدثنا أبو…٥٤٠٧أنه قال كنت يوما جالسا مع رجال من أصحاب النبي صلى الله عليه وسلم في…٥٤٠٨أن أباه، عمرو بن أمية أخبره أنه، رأى النبي صلى الله عليه وسلم يحتز من…٥٤٠٩قال ما عاب النبي صلى الله عليه وسلم طعاما قط، إن اشتهاه أكله، وإن…٥٤١٠حدثنا سعيد بن أبي مريم، حدثنا أبو غسان، قال حدثني أبو حازم، أنه سأل…٥٤١١قال قسم النبي صلى الله عليه وسلم يوما بين أصحابه تمرا، فأعطى كل إنسان…٥٤١٢ثم أصبحت بنو أسد تعزرني على الإسلام، خسرت إذا وضل سعيي .٥٤١٣فيطير ما طار وما بقي ثريناه فأكلناه .٥٤١٤فدعوه فأبى أن يأكل قال خرج رسول الله صلى الله عليه وسلم من الدنيا ولم…٥٤١٥قال ما أكل النبي صلى الله عليه وسلم على خوان، ولا في سكرجة، ولا خبز…٥٤١٦حتى قبض .٥٤١٧التلبينة مجمة لفؤاد المريض، تذهب ببعض الحزن٥٤١٨ولم يكمل من النساء إلا مريم بنت عمران وآسية امرأة فرعون، وفضل عائشة…٥٤١٩حدثنا عمرو بن عون، حدثنا خالد بن عبد الله، عن أبي طوالة، عن أنس، عن…٥٤٢٠سمع أبا حاتم الأشهل بن حاتم، حدثنا ابن عون، عن ثمامة بن أنس، عن أنس…٥٤٢١ولا رأى شاة سميطا بعينه قط .٥٤٢٢قال رأيت رسول الله صلى الله عليه وسلم يحتز من كتف شاة، فأكل منها،…٥٤٢٣قال قلت لعائشة أنهى النبي صلى الله عليه وسلم أن تؤكل لحوم الأضاحي فوق…٥٤٢٤حدثني عبد الله بن محمد، حدثنا سفيان، عن عمرو، عن عطاء، عن جابر، قال…٥٤٢٥التمس غلاما من غلمانكم يخدمني٥٤٢٦لا تلبسوا الحرير ولا الديباج ولا تشربوا في آنية الذهب والفضة، ولا…٥٤٢٧مثل المؤمن الذي يقرأ القرآن كمثل الأترجة، ريحها طيب وطعمها طيب، ومثل…٥٤٢٨فضل عائشة على النساء كفضل الثريد على سائر الطعام٥٤٢٩السفر قطعة من العذاب، يمنع أحدكم نومه وطعامه، فإذا قضى نهمته من وجهه…٥٤٣٠لو شئت شرطتيه لهم، فإنما الولاء لمن أعتق٥٤٣١حدثني إسحاق بن إبراهيم الحنظلي، عن أبي أسامة، عن هشام، قال أخبرني…٥٤٣٢قال كنت ألزم النبي صلى الله عليه وسلم لشبع بطني حين لا آكل الخمير،…٥٤٣٣أن رسول الله صلى الله عليه وسلم أتى مولى له خياطا، فأتي بدباء، فجعل…٥٤٣٤إنك دعوتنا خامس خمسة وهذا رجل قد تبعنا، فإن شئت أذنت له، وإن شئت تركته٥٤٣٥سمع النضر، أخبرنا ابن عون، قال أخبرني ثمامة بن عبد الله بن أنس، عن…٥٤٣٦أنه سمع أنس بن مالك، أن خياطا، دعا النبي صلى الله عليه وسلم لطعام…٥٤٣٧فرأيته يتتبع الدباء يأكلها .٥٤٣٨أراد أن يطعم الغني الفقير، وإن كنا لنرفع الكراع بعد خمس عشرة، وما شبع…٥٤٣٩أنه سمع أنس بن مالك، يقول إن خياطا دعا رسول الله صلى الله عليه وسلم…٥٤٤٠حدثنا عبد العزيز بن عبد الله، قال حدثني إبراهيم بن سعد، عن أبيه، عن…٥٤٤١قال تضيفت أبا هريرة سبعا، فكان هو وامرأته وخادمه يعتقبون الليل…٥٤٤١.٢ثم رأيت الحشفة هي أشدهن لضرسي .٥٤٤٢حدثتني أمي، عن عائشة رضى الله عنها قالت توفي رسول الله صلى الله عليه…٥٤٤٣امشوا نستنظر لجابر من اليهودي٥٤٤٤إن من الشجر لما بركته كبركة المسلم٥٤٤٥من تصبح كل يوم سبع تمرات عجوة لم يضره في ذلك اليوم سم ولا سحر٥٤٤٦حدثنا آدم، حدثنا شعبة، حدثنا جبلة بن سحيم، قال أصابنا عام سنة مع ابن…٥٤٤٧قال رأيت النبي صلى الله عليه وسلم يأكل الرطب بالقثاء .٥٤٤٨من الشجر شجرة تكون مثل المسلم، وهى النخلة٥٤٤٩حدثنا ابن مقاتل، أخبرنا عبد الله، أخبرنا إبراهيم بن سعد، عن أبيه، عن…٥٤٥٠أن أم سليم، أمه عمدت إلى مد من شعير، جشته وجعلت منه خطيفة، وعصرت عكة…٥٤٥١من أكل فلا يقربن مسجدنا٥٤٥٢من أكل ثوما أو بصلا فليعتزلنا، أو ليعتزل مسجدنا٥٤٥٣عليكم بالأسود منه، فإنه أيطب٥٤٥٤حدثنا علي، حدثنا سفيان، سمعت يحيى بن سعيد، عن بشير بن يسار، عن سويد…٥٤٥٥قال يحيى سمعت بشيرا، يقول حدثنا سويد، خرجنا مع رسول الله صلى الله…٥٤٥٦إذا أكل أحدكم فلا يمسح يده حتى يلعقها أو يلعقها٥٤٥٧ثم نصلي ولا نتوضأ .٥٤٥٨الحمد لله كثيرا طيبا مباركا فيه، غير مكفي، ولا مودع ولا مستغنى عنه،…٥٤٥٩الحمد لله الذي كفانا وأروانا، غير مكفي، ولا مكفور وقال مرة الحمد لله…٥٤٦٠إذا أتى أحدكم خادمه بطعامه، فإن لم يجلسه معه فليناوله أكلة أو أكلتين،…٥٤٦١يا أبا شعيب إن رجلا تبعنا فإن شئت أذنت له، وإن شئت تركته٥٤٦٢أن أباه، عمرو بن أمية أخبره أنه، رأى رسول الله صلى الله عليه وسلم…٥٤٦٣إذا وضع العشاء وأقيمت الصلاة فابدءوا بالعشاء٥٤٦٤أنه تعشى مرة وهو يسمع قراءة الإمام .٥٤٦٥إذا أقيمت الصلاة وحضر العشاء فابدءوا بالعشاء٥٤٦٦أن أنسا، قال أنا أعلم الناس، بالحجاب كان أبى بن كعب يسألني عنه، أصبح…
