كتاب مناقب الأنصار

١٧٢ حديثًا•صفحة ١ من ٢
٣٧٧٦قال قلت لأنس أرأيت اسم الأنصار كنتم تسمون به، أم سماكم الله قال بل…٣٧٧٧وقتلت سرواتهم، وجرحوا، فقدمه الله لرسوله صلى الله عليه وسلم في دخولهم…٣٧٧٨ما الذي بلغني عنكم٣٧٧٩لو أن الأنصار سلكوا واديا أو شعبا، لسلكت في وادي الأنصار، ولولا…٣٧٨٠قال لما قدموا المدينة آخى رسول الله صلى الله عليه وسلم بين عبد الرحمن…٣٧٨١وآخى رسول الله صلى الله عليه وسلم بينه وبين سعد بن الربيع، وكان كثير…٣٧٨٢حدثنا الصلت بن محمد أبو همام، قال سمعت المغيرة بن عبد الرحمن، حدثنا…٣٧٨٣الأنصار لا يحبهم إلا مؤمن، ولا يبغضهم إلا منافق، فمن أحبهم أحبه الله،…٣٧٨٤آية الإيمان حب الأنصار، وآية النفاق بغض الأنصار٣٧٨٥اللهم أنتم من أحب الناس إلى٣٧٨٦والذي نفسي بيده، إنكم أحب الناس إلى٣٧٨٧قالت الأنصار : يا رسول الله لكل نبي أتباع، وإنا قد اتبعناك، فادع الله…٣٧٨٨اللهم اجعل أتباعهم منهم٣٧٨٩خير دور الأنصار بنو النجار، ثم بنو عبد الأشهل، ثم بنو الحارث بن خزرج،…٣٧٩٠خير الأنصار أو قال خير دور الأنصار بنو النجار وبنو عبد الأشهل وبنو…٣٧٩١إن خير دور الأنصار دار بني النجار، ثم عبد الأشهل، ثم دار بني الحارث،…٣٧٩٢ستلقون بعدي أثرة فاصبروا حتى تلقوني على الحوض٣٧٩٣إنكم ستلقون بعدي أثرة فاصبروا حتى تلقوني، وموعدكم الحوض٣٧٩٤إما لا، فاصبروا حتى تلقوني، فإنه سيصيبكم بعدي أثرة٣٧٩٥لا عيش إلا عيش الآخرة، فأصلح الأنصار والمهاجرة٣٧٩٦حدثنا آدم، حدثنا شعبة، عن حميد الطويل، سمعت أنس بن مالك رضى الله عنه…٣٧٩٧اللهم لا عيش إلا عيش الآخره فاغفر للمهاجرين والأنصار٣٧٩٨من يضم، أو يضيف هذا٣٧٩٩أوصيكم بالأنصار، فإنهم كرشي وعيبتي، وقد قضوا الذي عليهم، وبقي الذي…٣٨٠٠أما بعد، أيها الناس، فإن الناس يكثرون وتقل الأنصار، حتى يكونوا كالملح…٣٨٠١الأنصار كرشي وعيبتي، والناس سيكثرون ويقلون، فاقبلوا من محسنهم،…٣٨٠٢أتعجبون من لين هذه لمناديل سعد بن معاذ خير منها٣٨٠٣اهتز العرش لموت سعد بن معاذ٣٨٠٤قوموا إلى خيركم أو سيدكم٣٨٠٥خرجا من عند النبي صلى الله عليه وسلم في ليلة مظلمة، وإذا نور بين…٣٨٠٦استقرئوا القرآن من أربعة من ابن مسعود وسالم مولى أبي حذيفة، وأبى،…٣٨٠٧خير دور الأنصار بنو النجار، ثم بنو عبد الأشهل، ثم بنو الحارث بن…٣٨٠٨خذوا القرآن من أربعة من عبد الله بن مسعود فبدأ به وسالم مولى أبي…٣٨٠٩إن الله أمرني أن أقرأ عليك { لم يكن الذين كفروا }٣٨١٠كلهم من الأنصار أبى، ومعاذ بن جبل، وأبو زيد، وزيد بن ثابت . قلت لأنس…٣٨١١وكان أبو طلحة رجلا راميا شديد القد، يكسر يومئذ قوسين أو ثلاثا، وكان…٣٨١٢حدثنا عبد الله بن يوسف، قال سمعت مالكا، يحدث عن أبي النضر، مولى عمر…٣٨١٣تلك الروضة الإسلام، وذلك العمود عمود الإسلام، وتلك العروة عروة…٣٨١٤أتيت المدينة فلقيت عبد الله بن سلام رضى الله عنه فقال ألا تجيء فأطعمك…٣٨١٥خير نسائها مريم، وخير نسائها خديجة٣٨١٦هلكت قبل أن يتزوجني، لما كنت أسمعه يذكرها، وأمره الله أن يبشرها ببيت…٣٨١٧من كثرة ذكر رسول الله صلى الله عليه وسلم إياها . قالت وتزوجني بعدها…٣٨١٨وما رأيتها، ولكن كان النبي صلى الله عليه وسلم يكثر ذكرها، وربما ذبح…٣٨١٩قال قلت لعبد الله بن أبي أوفى رضى الله عنهما بشر النبي صلى الله عليه…٣٨٢٠فإذا هي أتتك فاقرأ عليها السلام من ربها ومني، وبشرها ببيت في الجنة من…٣٨٢١فعرف استئذان خديجة فارتاع لذلك، فقال " اللهم هالة " . قالت فغرت فقلت…٣٨٢٢ولا رآني إلا ضحك .٣٨٢٣هل أنت مريحي من ذي الخلصة٣٨٢٤رضى الله عنها قالت لما كان يوم أحد هزم المشركون هزيمة بينة، فصاح…٣٨٢٥لا أراه إلا بالمعروف٣٨٢٦رضى الله عنهما أن النبي صلى الله عليه وسلم لقي زيد بن عمرو بن نفيل…٣٨٢٧فقال إني لعلي أن أدين دينكم، فأخبرني . فقال لا تكون على ديننا حتى…٣٨٢٨والله ما منكم على دين إبراهيم غيري، وكان يحيي الموءودة، يقول للرجل…٣٨٢٩سمع جابر بن عبد الله رضى الله عنهما قال لما بنيت الكعبة ذهب النبي صلى…٣٨٣٠وعبيد الله بن أبي يزيد، قالا لم يكن على عهد النبي صلى الله عليه وسلم…٣٨٣١وكان النبي صلى الله عليه وسلم يصومه، فلما قدم المدينة صامه وأمر…٣٨٣٢وكانوا يسمون المحرم صفرا ويقولون إذا برا الدبر، وعفا الأثر، حلت…٣٨٣٣قال جاء سيل في الجاهلية فكسا ما بين الجبلين . قال سفيان ويقول إن هذا…٣٨٣٤قال دخل أبو بكر على امرأة من أحمس يقال لها زينب، فرآها لا تكلم، فقال…٣٨٣٥وعليها وشاح من أدم فسقط منها، فانحطت عليه الحديا وهى تحسبه لحما،…٣٨٣٦ألا من كان حالفا فلا يحلف إلا بالله٣٨٣٧أن عبد الرحمن بن القاسم، حدثه أن القاسم كان يمشي بين يدى الجنازة ولا…٣٨٣٨قال قال عمر رضى الله عنه إن المشركين كانوا لا يفيضون من جمع حتى تشرق…٣٨٣٩قال قلت لأبي أسامة حدثكم يحيى بن المهلب، حدثنا حصين، عن عكرمة، {…٣٨٤٠قال وقال ابن عباس سمعت أبي يقول، في الجاهلية اسقنا كأسا دهاقا .٣٨٤١أصدق كلمة قالها الشاعر كلمة لبيد ألا كل شىء ما خلا الله باطل وكاد…٣٨٤٢وكان أبو بكر يأكل من خراجه، فجاء يوما بشىء فأكل منه أبو بكر فقال له…٣٨٤٣قال وحبل الحبلة أن تنتج الناقة ما في بطنها، ثم تحمل التي نتجت، فنهاهم…٣٨٤٤، وكان، يقول لي فعل قومك كذا وكذا يوم كذا وكذا، وفعل قومك كذا وكذا…٣٨٤٥كان رجل من بني هاشم استأجره رجل من قريش من فخذ أخرى، فانطلق معه في…٣٨٤٦فقدم رسول الله صلى الله عليه وسلم وقد افترق ملؤهم، وقتلت سرواتهم…٣٨٤٧أن كريبا، مولى ابن عباس حدثه أن ابن عباس رضى الله عنهما قال ليس السعى…٣٨٤٨اسمعوا مني ما أقول لكم، وأسمعوني ما تقولون، ولا تذهبوا فتقولوا قال…٣٨٤٩قال رأيت في الجاهلية قردة اجتمع عليها قردة قد زنت، فرجموها فرجمتها…٣٨٥٠ونسي الثالثة، قال سفيان ويقولون إنها الاستسقاء بالأنواء .٣٨٥١فمكث ثلاث عشرة سنة، ثم أمر بالهجرة، فهاجر إلى المدينة، فمكث بها عشر…٣٨٥٢لقد كان من قبلكم ليمشط بمشاط الحديد ما دون عظامه من لحم أو عصب ما…٣٨٥٣فسجد فما بقي أحد إلا سجد، إلا رجل رأيته أخذ كفا من حصا فرفعه فسجد…٣٨٥٤اللهم عليك الملأ من قريش أبا جهل بن هشام، وعتبة بن ربيعة، وشيبة بن…٣٨٥٥ثم قتل فجزاؤه جهنم . فذكرته لمجاهد فقال إلا من ندم .٣٨٥٦فوضع ثوبه في عنقه فخنقه خنقا شديدا، فأقبل أبو بكر حتى أخذ بمنكبه…٣٨٥٧قال قال عمار بن ياسر رأيت رسول الله صلى الله عليه وسلم وما معه إلا…٣٨٥٨سعد بن أبي وقاص يقول ما أسلم أحد إلا في اليوم الذي أسلمت فيه، ولقد…٣٨٥٩حدثني عبيد الله بن سعيد، حدثنا أبو أسامة، حدثنا مسعر، عن معن بن عبد…٣٨٦٠رضى الله عنه أنه كان يحمل مع النبي صلى الله عليه وسلم إداوة لوضوئه…٣٨٦١ارجع إلى قومك، فأخبرهم حتى يأتيك أمري٣٨٦٢في مسجد الكوفة يقول والله لقد رأيتني وإن عمر لموثقي على الإسلام قبل…٣٨٦٣حدثني محمد بن كثير، أخبرنا سفيان، عن إسماعيل بن أبي خالد، عن قيس بن…٣٨٦٤قال بينما هو في الدار خائفا، إذ جاءه العاص بن وائل السهمي أبو عمرو،…٣٨٦٥حدثنا علي بن عبد الله، حدثنا سفيان، قال عمرو بن دينار سمعته قال قال…٣٨٦٦أن سالما، حدثه عن عبد الله بن عمر، قال ما سمعت عمر، لشىء قط يقول إني…٣٨٦٧ولو أن أحدا انقض لما صنعتم، بعثمان لكان محقوقا أن ينقض .٣٨٦٨مكة سألوا رسول الله صلى الله عليه وسلم أن يريهم آية، فأراهم القمر…٣٨٦٩حدثنا عبدان، عن أبي حمزة، عن الأعمش، عن إبراهيم، عن أبي معمر، عن عبد…٣٨٧٠انشق على زمان رسول الله صلى الله عليه وسلم .٣٨٧١حدثنا عمر بن حفص، حدثنا أبي، حدثنا الأعمش، حدثنا إبراهيم، عن أبي…٣٨٧٢أن عبيد الله بن عدي بن الخيار، أخبره أن المسور بن مخرمة وعبد الرحمن…٣٨٧٣إن أولئك إذا كان فيهم الرجل الصالح فمات بنوا على قبره مسجدا، وصوروا…٣٨٧٤فكساني رسول الله صلى الله عليه وسلم خميصة لها أعلام، فجعل رسول الله…٣٨٧٥إن في الصلاة شغلا٣٨٧٦لكم أنتم يا أهل السفينة هجرتان٣٨٧٧مات اليوم رجل صالح، فقوموا فصلوا على أخيكم أصحمة٣٨٧٨أن عطاء، حدثهم عن جابر بن عبد الله الأنصاري رضى الله عنهما أن نبي…٣٨٧٩فكبر عليه أربعا . تابعه عبد الصمد .٣٨٨٠وابن المسيب، أن أبا هريرة رضى الله عنه أخبرهما أن رسول الله صلى الله…٣٨٨١أن أبا هريرة رضى الله عنه أخبرهم أن رسول الله صلى الله عليه وسلم صف…٣٨٨٢منزلنا غدا إن شاء الله بخيف بني كنانة، حيث تقاسموا على الكفر٣٨٨٣هو في ضحضاح من نار، ولولا أنا لكان في الدرك الأسفل من النار٣٨٨٤أى عم، قل لا إله إلا الله . كلمة أحاج لك بها عند الله٣٨٨٥لعله تنفعه شفاعتي يوم القيامة، فيجعل في ضحضاح من النار، يبلغ كعبيه،…٣٨٨٦لما كذبني قريش قمت في الحجر، فجلا الله لي بيت المقدس، فطفقت أخبرهم عن…٣٨٨٧بينما أنا في الحطيم وربما قال في الحجر مضطجعا، إذ أتاني آت فقد قال…٣٨٨٨أريها رسول الله صلى الله عليه وسلم ليلة أسري به إلى بيت المقدس . قال…٣٨٨٩وما أحب أن لي بها مشهد بدر وإن كانت بدر، أذكر في الناس منها .٣٨٩٠حدثنا علي بن عبد الله، حدثنا سفيان، قال كان عمرو يقول سمعت جابر بن…٣٨٩١أن ابن جريج، أخبرهم قال عطاء قال جابر أنا وأبي، وخالي، من أصحاب…٣٨٩٢تعالوا بايعوني على أن لا تشركوا بالله شيئا، ولا تسرقوا، ولا تزنوا،…٣٨٩٣ولا نسرق، ولا نزني، ولا نقتل النفس التي حرم الله، ولا ننتهب، ولا نعصي…٣٨٩٤فوعكت فتمرق شعري فوفى جميمة، فأتتني أمي أم رومان وإني لفي أرجوحة ومعي…٣٨٩٥أريتك في المنام مرتين، أرى أنك في سرقة من حرير ويقول هذه امرأتك فاكشف…٣٨٩٦قال توفيت خديجة قبل مخرج النبي صلى الله عليه وسلم إلى المدينة بثلاث…٣٨٩٧لم يأخذ من أجره شيئا، منهم مصعب بن عمير قتل يوم أحد، وترك نمرة، فكنا…٣٨٩٨الأعمال بالنية، فمن كانت هجرته إلى دنيا يصيبها أو امرأة يتزوجها،…٣٨٩٩أن عبد الله بن عمر رضى الله عنهما كان يقول لا هجرة بعد الفتح .٣٩٠٠فقالت لا هجرة اليوم، كان المؤمنون يفر أحدهم بدينه إلى الله تعالى وإلى…٣٩٠١قال اللهم إنك تعلم أنه ليس أحد أحب إلى أن أجاهدهم فيك من قوم كذبوا…٣٩٠٢فمكث بمكة ثلاث عشرة سنة يوحى إليه، ثم أمر بالهجرة فهاجر عشر سنين،…٣٩٠٣قال مكث رسول الله صلى الله عليه وسلم بمكة ثلاث عشرة، وتوفي وهو ابن…٣٩٠٤إن عبدا خيره الله بين أن يؤتيه من زهرة الدنيا ما شاء، وبين ما عنده،…٣٩٠٥إني أريت دار هجرتكم ذات نخل بين لابتين٣٩٠٦هذا إن شاء الله المنزل٣٩٠٧وفاطمة، عن أسماء، رضى الله عنها صنعت سفرة للنبي صلى الله عليه وسلم…٣٩٠٨فدعا عليه النبي صلى الله عليه وسلم فساخت به فرسه . قال ادع الله لي…٣٩٠٩فأتيت المدينة، فنزلت بقباء، فولدته بقباء، ثم أتيت به النبي صلى الله…٣٩١٠أتوا به النبي صلى الله عليه وسلم فأخذ النبي صلى الله عليه وسلم تمرة…٣٩١١وأبو بكر شيخ يعرف، ونبي الله صلى الله عليه وسلم شاب لا يعرف، قال…٣٩١٢رضى الله عنه قال كان فرض للمهاجرين الأولين أربعة آلاف في أربعة، وفرض…٣٩١٣حدثنا محمد بن كثير، أخبرنا سفيان، عن الأعمش، عن أبي وائل، عن خباب،…٣٩١٤منهم مصعب بن عمير، قتل يوم أحد فلم نجد شيئا نكفنه فيه، إلا نمرة كنا…٣٩١٥قال قال لي عبد الله بن عمر هل تدري ما قال أبي لأبيك قال قلت لا . قال…٣٩١٦فأرسلني عمر وقال اذهب فانظر هل استيقظ فأتيته، فدخلت عليه فبايعته، ثم…٣٩١٧فأتيناها ولها شىء من ظل قال ففرشت لرسول الله صلى الله عليه وسلم فروة…٣٩١٨قال البراء فدخلت مع أبي بكر على أهله، فإذا عائشة ابنته مضطجعة، قد…٣٩١٩أن عقبة بن وساج، حدثه عن أنس، خادم النبي صلى الله عليه وسلم قال قدم…٣٩٢٠فكان أسن أصحابه أبو بكر، فغلفها بالحناء والكتم حتى قنأ لونها .٣٩٢١أن أبا بكر رضى الله عنه تزوج امرأة من كلب يقال لها أم بكر، فلما هاجر…٣٩٢٢اسكت يا أبا بكر، اثنان الله ثالثهما٣٩٢٣ويحك إن الهجرة شأنها شديد، فهل لك من إبل٣٩٢٤حدثنا أبو الوليد، حدثنا شعبة، قال أنبأنا أبو إسحاق، سمع البراء رضى…٣٩٢٥فقدم بلال وسعد وعمار بن ياسر، ثم قدم عمر بن الخطاب في عشرين من أصحاب…
