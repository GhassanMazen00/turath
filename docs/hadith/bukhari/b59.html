كتاب بدء الخلق

١٣١ حديثًا
٣١٩٠يا بني تميم، أبشروا٣١٩١اقبلوا البشرى يا بني تميم٣١٩٢وروى عيسى، عن رقبة، عن قيس بن مسلم، عن طارق بن شهاب، قال سمعت عمر رضى…٣١٩٣يقول الله شتمني ابن آدم وما ينبغي له أن يشتمني، وتكذبني وما ينبغي له،…٣١٩٤لما قضى الله الخلق كتب في كتابه، فهو عنده فوق العرش إن رحمتي غلبت غضبي٣١٩٥من ظلم قيد شبر طوقه من سبع أرضين٣١٩٦من أخذ شيئا من الأرض بغير حقه خسف به يوم القيامة إلى سبع أرضين٣١٩٧الزمان قد استدار كهيئته يوم خلق السموات والأرض، السنة اثنا عشر شهرا،…٣١٩٨من أخذ شبرا من الأرض ظلما، فإنه يطوقه يوم القيامة من سبع أرضين٣١٩٩فتستأذن فيؤذن لها، ويوشك أن تسجد فلا يقبل منها، وتستأذن فلا يؤذن لها،…٣٢٠٠الشمس والقمر مكوران يوم القيامة٣٢٠١إن الشمس والقمر لا يخسفان لموت أحد ولا لحياته، ولكنهما آيتان من آيات…٣٢٠٢إن الشمس والقمر آيتان من آيات الله، لا يخسفان لموت أحد ولا لحياته،…٣٢٠٣سمع الله لمن حمده٣٢٠٤الشمس والقمر لا ينكسفان لموت أحد ولا لحياته، ولكنهما آيتان من آيات…٣٢٠٥نصرت بالصبا، وأهلكت عاد بالدبور٣٢٠٦ما أدري لعله كما قال قوم { فلما رأوه عارضا مستقبل أوديتهم }٣٢٠٧بينا أنا عند البيت بين النائم واليقظان وذكر بين الرجلين فأتيت بطست من…٣٢٠٨إن أحدكم يجمع خلقه في بطن أمه أربعين يوما، ثم يكون علقة مثل ذلك، ثم…٣٢٠٩إذا أحب الله العبد نادى جبريل إن الله يحب فلانا فأحببه . فيحبه جبريل،…٣٢١٠إن الملائكة تنزل في العنان وهو السحاب فتذكر الأمر قضي في السماء،…٣٢١١إذا كان يوم الجمعة كان على كل باب من أبواب المسجد الملائكة، يكتبون…٣٢١٢أجب عني، اللهم أيده بروح القدس٣٢١٣اهجهم أو هاجهم وجبريل معك٣٢١٤وحدثنا إسحاق، أخبرنا وهب بن جرير، حدثنا أبي قال، سمعت حميد بن هلال،…٣٢١٥كل ذاك يأتي الملك أحيانا في مثل صلصلة الجرس، فيفصم عني وقد وعيت ما…٣٢١٦من أنفق زوجين في سبيل الله دعته خزنة الجنة أى فل هلم٣٢١٧يا عائشة، هذا جبريل يقرأ عليك السلام٣٢١٨ألا تزورنا أكثر مما تزورنا٣٢١٩أقرأني جبريل على حرف، فلم أزل أستزيده حتى انتهى إلى سبعة أحرف٣٢٢٠وكان أجود ما يكون في رمضان حين يلقاه جبريل، وكان جبريل يلقاه في كل…٣٢٢١نزل جبريل فأمني، فصليت معه، ثم صليت معه، ثم صليت معه، ثم صليت معه، ثم…٣٢٢٢قال لي جبريل من مات من أمتك لا يشرك بالله شيئا دخل الجنة، أو لم يدخل…٣٢٢٣الملائكة يتعاقبون، ملائكة بالليل وملائكة بالنهار، ويجتمعون في صلاة…٣٢٢٤ما بال هذه الوسادة٣٢٢٥لا تدخل الملائكة بيتا فيه كلب ولا صورة تماثيل٣٢٢٦لا تدخل الملائكة بيتا فيه صورة٣٢٢٧حدثنا يحيى بن سليمان، قال حدثني ابن وهب، قال حدثني عمر، عن سالم، عن…٣٢٢٨إذا قال الإمام سمع الله لمن حمده . فقولوا اللهم ربنا لك الحمد، فإنه…٣٢٢٩إن أحدكم في صلاة ما دامت الصلاة تحبسه، والملائكة تقول اللهم اغفر له…٣٢٣٠حدثنا علي بن عبد الله، حدثنا سفيان، عن عمرو، عن عطاء، عن صفوان بن…٣٢٣١لقد لقيت من قومك ما لقيت، وكان أشد ما لقيت منهم يوم العقبة، إذ عرضت…٣٢٣٢حدثنا قتيبة، حدثنا أبو عوانة، حدثنا أبو إسحاق الشيباني، قال سألت زر…٣٢٣٣حدثنا حفص بن عمر، حدثنا شعبة، عن الأعمش، عن إبراهيم، عن علقمة، عن عبد…٣٢٣٤ولكن قد رأى جبريل في صورته، وخلقه ساد ما بين الأفق .٣٢٣٥وإنه أتاه هذه المرة في صورته التي هي صورته، فسد الأفق .٣٢٣٦رأيت الليلة رجلين أتياني قالا الذي يوقد النار مالك خازن النار، وأنا…٣٢٣٧إذا دعا الرجل امرأته إلى فراشه فأبت، فبات غضبان عليها، لعنتها…٣٢٣٨ثم فتر عني الوحى فترة، فبينا أنا أمشي سمعت صوتا من السماء، فرفعت بصري…٣٢٣٩رأيت ليلة أسري بي موسى رجلا آدم طوالا جعدا، كأنه من رجال شنوءة، ورأيت…٣٢٤٠إذا مات أحدكم فإنه يعرض عليه مقعده بالغداة والعشي، فإن كان من أهل…٣٢٤١اطلعت في الجنة فرأيت أكثر أهلها الفقراء، واطلعت في النار فرأيت أكثر…٣٢٤٢بينا أنا نائم رأيتني في الجنة، فإذا امرأة تتوضأ إلى جانب قصر، فقلت…٣٢٤٣الخيمة درة مجوفة، طولها في السماء ثلاثون ميلا، في كل زاوية منها…٣٢٤٤قال الله أعددت لعبادي الصالحين ما لا عين رأت، ولا أذن سمعت، ولا خطر…٣٢٤٥أول زمرة تلج الجنة صورتهم على صورة القمر ليلة البدر، لا يبصقون فيها…٣٢٤٦أول زمرة تدخل الجنة على صورة القمر ليلة البدر، والذين على إثرهم كأشد…٣٢٤٧ليدخلن من أمتي سبعون ألفا أو سبعمائة ألف لا يدخل أولهم حتى يدخل…٣٢٤٨والذي نفس محمد بيده، لمناديل سعد بن معاذ في الجنة أحسن من هذا٣٢٤٩لمناديل سعد بن معاذ في الجنة أفضل من هذا٣٢٥٠موضع سوط في الجنة خير من الدنيا وما فيها٣٢٥١إن في الجنة لشجرة يسير الراكب في ظلها مائة عام لا يقطعها٣٢٥٣إن في الجنة لشجرة يسير الراكب في ظلها مائة سنة، واقرءوا إن شئتم { وظل…٣٢٥٤أول زمرة تدخل الجنة على صورة القمر ليلة البدر، والذين على آثارهم…٣٢٥٥إن له مرضعا في الجنة٣٢٥٦إن أهل الجنة يتراءيون أهل الغرف من فوقهم كما يتراءيون الكوكب الدري…٣٢٥٧في الجنة ثمانية أبواب، فيها باب يسمى الريان لا يدخله إلا الصائمون٣٢٥٨يعني للتلول، ثم قال " أبردوا بالصلاة، فإن شدة الحر من فيح جهنم " .٣٢٥٩أبردوا بالصلاة فإن شدة الحر من فيح جهنم٣٢٦٠اشتكت النار إلى ربها، فقالت رب أكل بعضي بعضا، فأذن لها بنفسين نفس في…٣٢٦١الحمى من فيح جهنم فأبردوها بالماء٣٢٦٢الحمى من فور جهنم، فأبردوها عنكم بالماء٣٢٦٣الحمى من فيح جهنم، فأبردوها بالماء٣٢٦٤الحمى من فيح جهنم فأبردوها بالماء٣٢٦٥ناركم جزء من سبعين جزءا من نار جهنم٣٢٦٦سمع عطاء، يخبر عن صفوان بن يعلى، عن أبيه، أنه سمع النبي صلى الله عليه…٣٢٦٧يجاء بالرجل يوم القيامة فيلقى في النار، فتندلق أقتابه في النار، فيدور…٣٢٦٨أشعرت أن الله أفتاني فيما فيه شفائي أتاني رجلان، فقعد أحدهما عند رأسي…٣٢٦٩يعقد الشيطان على قافية رأس أحدكم إذا هو نام ثلاث عقد، يضرب كل عقدة…٣٢٧٠ذاك رجل بال الشيطان في أذنيه أو قال في أذنه٣٢٧١أما إن أحدكم إذا أتى أهله وقال بسم الله اللهم جنبنا الشيطان، وجنب…٣٢٧٣إذا طلع حاجب الشمس فدعوا الصلاة حتى تبرز، وإذا غاب حاجب الشمس فدعوا…٣٢٧٤إذا مر بين يدى أحدكم شىء وهو يصلي فليمنعه، فإن أبى فليمنعه، فإن أبى…٣٢٧٥صدقك وهو كذوب، ذاك شيطان٣٢٧٦يأتي الشيطان أحدكم فيقول من خلق كذا من خلق كذا حتى يقول من خلق ربك…٣٢٧٧إذا دخل رمضان فتحت أبواب الجنة، وغلقت أبواب جهنم، وسلسلت الشياطين٣٢٧٨إن موسى قال لفتاه آتنا غداءنا، قال أرأيت إذ أوينا إلى الصخرة، فإني…٣٢٧٩ها إن الفتنة ها هنا إن الفتنة ها هنا من حيث يطلع قرن الشيطان٣٢٨٠إذا استجنح { الليل } أو كان جنح الليل فكفوا صبيانكم، فإن الشياطين…٣٢٨١على رسلكما إنها صفية بنت حيى٣٢٨٢إني لأعلم كلمة لو قالها ذهب عنه ما يجد، لو قال أعوذ بالله من الشيطان…٣٢٨٣لو أن أحدكم إذا أتى أهله قال { اللهم } جنبني الشيطان، وجنب الشيطان ما…٣٢٨٤إن الشيطان عرض لي، فشد على يقطع الصلاة على، فأمكنني الله منه٣٢٨٥إذا نودي بالصلاة أدبر الشيطان وله ضراط، فإذا قضي أقبل، فإذا ثوب بها…٣٢٨٦كل بني آدم يطعن الشيطان في جنبيه بإصبعه حين يولد، غير عيسى بن مريم،…٣٢٨٧حدثنا مالك بن إسماعيل، حدثنا إسرائيل، عن المغيرة، عن إبراهيم، عن…٣٢٨٨الملائكة تتحدث في العنان والعنان الغمام بالأمر يكون في الأرض، فتسمع…٣٢٨٩التثاؤب من الشيطان، فإذا تثاءب أحدكم فليرده ما استطاع، فإن أحدكم إذا…٣٢٩٠رضى الله عنها قالت لما كان يوم أحد هزم المشركون فصاح إبليس أى عباد…٣٢٩١هو اختلاس يختلس الشيطان من صلاة أحدكم٣٢٩٢الرؤيا الصالحة من الله، والحلم من الشيطان فإذا حلم أحدكم حلما يخافه…٣٢٩٣من قال لا إله إلا الله وحده لا شريك له، له الملك، وله الحمد، وهو على…٣٢٩٤عجبت من هؤلاء اللاتي كن عندي، فلما سمعن صوتك ابتدرن الحجاب٣٢٩٥إذا استيقظ أراه أحدكم من منامه فتوضأ فليستنثر ثلاثا، فإن الشيطان يبيت…٣٢٩٦إني أراك تحب الغنم والبادية، فإذا كنت في غنمك وباديتك فأذنت بالصلاة،…٣٢٩٨اقتلوا الحيات، واقتلوا ذا الطفيتين والأبتر، فإنهما يطمسان البصر،…٣٢٩٩رآني أبو لبابة وزيد بن الخطاب .٣٣٠٠يوشك أن يكون خير مال الرجل غنم يتبع بها شعف الجبال ومواقع القطر، يفر…٣٣٠١رأس الكفر نحو المشرق، والفخر والخيلاء في أهل الخيل والإبل، والفدادين…٣٣٠٢الإيمان يمان ها هنا، ألا إن القسوة وغلظ القلوب في الفدادين عند أصول…٣٣٠٣إذا سمعتم صياح الديكة فاسألوا الله من فضله، فإنها رأت ملكا، وإذا…٣٣٠٤إذا كان جنح الليل أو أمسيتم فكفوا صبيانكم، فإن الشياطين تنتشر حينئذ،…٣٣٠٥فقدت أمة من بني إسرائيل لا يدرى ما فعلت، وإني لا أراها إلا الفار إذا…٣٣٠٦حدثنا سعيد بن عفير، عن ابن وهب، قال حدثني يونس، عن ابن شهاب، عن عروة،…٣٣٠٧أن أم شريك، أخبرته أن النبي صلى الله عليه وسلم أمرها بقتل الأوزاغ .٣٣٠٨اقتلوا ذا الطفيتين، فإنه يلتمس البصر، ويصيب الحبل٣٣٠٩إنه يصيب البصر، ويذهب الحبل٣٣١١أن ابن عمر، كان يقتل الحيات ثم نهى قال إن النبي صلى الله عليه وسلم…٣٣١٣أنه كان يقتل الحيات . فحدثه أبو لبابة، أن النبي صلى الله عليه وسلم…٣٣١٤خمس فواسق يقتلن في الحرم الفأرة، والعقرب، والحديا، والغراب، والكلب…٣٣١٥خمس من الدواب من قتلهن وهو محرم فلا جناح عليه العقرب، والفأرة، والكلب…٣٣١٦خمروا الآنية، وأوكوا الأسقية، وأجيفوا الأبواب، واكفتوا صبيانكم عند…٣٣١٧وقيت شركم، كما وقيتم شرها٣٣١٨دخلت امرأة النار في هرة ربطتها، فلم تطعمها، ولم تدعها تأكل من خشاش…٣٣١٩نزل نبي من الأنبياء تحت شجرة فلدغته نملة، فأمر بجهازه فأخرج من تحتها،…٣٣٢٠إذا وقع الذباب في شراب أحدكم فليغمسه، ثم لينزعه، فإن في إحدى جناحيه…٣٣٢١غفر لامرأة مومسة مرت بكلب على رأس ركي يلهث، قال كاد يقتله العطش،…٣٣٢٢لا تدخل الملائكة بيتا فيه كلب ولا صورة٣٣٢٣حدثنا عبد الله بن يوسف، أخبرنا مالك، عن نافع، عن عبد الله بن عمر رضى…٣٣٢٤من أمسك كلبا ينقص من عمله كل يوم قيراط، إلا كلب حرث أو كلب ماشية٣٣٢٥من اقتنى كلبا لا يغني عنه زرعا ولا ضرعا، نقص من عمله كل يوم قيراط
