١٨٥ حديثًا•صفحة ١ من ٢
٥٧٨٣لا ينظر الله إلى من جر ثوبه خيلاء٥٧٨٤من جر ثوبه خيلاء لم ينظر الله إليه يوم القيامة٥٧٨٥إن الشمس والقمر آيتان من آيات الله، فإذا رأيتم منها شيئا فصلوا وادعوا…٥٧٨٦قال فرأيت بلالا جاء بعنزة فركزها، ثم أقام الصلاة، فرأيت رسول الله صلى…٥٧٨٧ما أسفل من الكعبين من الإزار ففي النار٥٧٨٨لا ينظر الله يوم القيامة إلى من جر إزاره بطرا٥٧٨٩بينما رجل يمشي في حلة، تعجبه نفسه مرجل جمته، إذ خسف الله به، فهو…٥٧٩٠بينا رجل يجر إزاره، خسف به، فهو يتجلل في الأرض إلى يوم القيامة٥٧٩١من جر ثوبه مخيلة، لم ينظر الله إليه يوم القيامة٥٧٩٢لعلك تريدين أن ترجعي إلى رفاعة، لا، حتى يذوق عسيلتك وتذوقي عسيلته٥٧٩٣أن حسين بن علي، أخبره أن عليا رضى الله عنه قال فدعا النبي صلى الله…٥٧٩٤لا يلبس المحرم القميص، ولا السراويل، ولا البرنس، ولا الخفين، إلا أن…٥٧٩٥سمع جابر بن عبد الله رضى الله عنهما قال أتى النبي صلى الله عليه وسلم…٥٧٩٦قال لما توفي عبد الله بن أبى جاء ابنه إلى رسول الله صلى الله عليه…٥٧٩٧قال ضرب رسول الله صلى الله عليه وسلم مثل البخيل والمتصدق، كمثل رجلين…٥٧٩٨قال انطلق النبي صلى الله عليه وسلم لحاجته ثم أقبل، فتلقيته بماء،…٥٧٩٩فمشى حتى توارى عني في سواد الليل، ثم جاء فأفرغت عليه الإداوة، فغسل…٥٨٠٠قال قسم رسول الله صلى الله عليه وسلم أقبية، ولم يعط مخرمة شيئا فقال…٥٨٠١لا ينبغي هذا للمتقين٥٨٠٢رأيت على أنس برنسا أصفر من خز .٥٨٠٣لا تلبسوا القمص، ولا العمائم، ولا السراويلات، ولا البرانس، ولا…٥٨٠٤من لم يجد إزارا فليلبس سراويل، ومن لم يجد نعلين فليلبس خفين٥٨٠٥لا تلبسوا القميص، والسراويل، والعمائم والبرانس، والخفاف، إلا أن يكون…٥٨٠٦لا يلبس المحرم القميص، ولا العمامة، ولا السراويل، ولا البرنس، ولا…٥٨٠٧على رسلك، فإني أرجو أن يؤذن لي٥٨٠٨حدثنا أبو الوليد، حدثنا مالك، عن الزهري، عن أنس رضى الله عنه أن النبي…٥٨٠٩قال كنت أمشي مع رسول الله صلى الله عليه وسلم وعليه برد نجراني غليظ…٥٨١٠قال جاءت امرأة ببردة قال سهل هل تدري ما البردة قال نعم هي الشملة،…٥٨١١يدخل الجنة من أمتي زمرة هي سبعون ألفا، تضيء وجوههم إضاءة القمر٥٨١٢قال قلت له أى الثياب كان أحب إلى النبي صلى الله عليه وسلم قال الحبرة .٥٨١٣رضى الله عنه قال كان أحب الثياب إلى النبي صلى الله عليه وسلم أن…٥٨١٤أن عائشة رضى الله عنها زوج النبي صلى الله عليه وسلم أخبرته أن رسول…٥٨١٦لعنة الله على اليهود والنصارى اتخذوا قبور أنبيائهم مساجد٥٨١٧اذهبوا بخميصتي هذه إلى أبي جهم، فإنها ألهتني آنفا عن صلاتي، وائتوني…٥٨١٨حدثنا مسدد، حدثنا إسماعيل، حدثنا أيوب، عن حميد بن هلال، عن أبي بردة،…٥٨١٩والمنابذة، وعن صلاتين بعد الفجر حتى ترتفع الشمس، وبعد العصر حتى تغيب،…٥٨٢٠أن أبا سعيد الخدري، قال نهى رسول الله صلى الله عليه وسلم عن لبستين…٥٨٢١وأن يشتمل بالثوب الواحد، ليس على أحد شقيه، وعن الملامسة والمنابذة .٥٨٢٢وأن يحتبي الرجل في ثوب واحد، ليس على فرجه منه شىء .٥٨٢٣من ترون نكسو هذه٥٨٢٤فإذا هو في حائط وعليه خميصة حريثية، وهو يسم الظهر الذي قدم عليه في…٥٨٢٥فإن كان ذلك لم تحلي له أو لم تصلحي له حتى يذوق من عسيلتك٥٨٢٦قال رأيت بشمال النبي صلى الله عليه وسلم وبيمينه رجلين عليهما ثياب بيض…٥٨٢٧ما من عبد قال لا إله إلا الله . ثم مات على ذلك، إلا دخل الجنة٥٨٢٨إلا هكذا، وأشار بإصبعيه اللتين تليان الإبهام قال فيما علمنا أنه يعني…٥٨٢٩حدثنا أحمد بن يونس، حدثنا زهير، حدثنا عاصم، عن أبي عثمان، قال كتب…٥٨٣٠لا يلبس الحرير في الدنيا، إلا لم يلبس في الآخرة منه٥٨٣١الذهب والفضة والحرير والديباج هي لهم في الدنيا، ولكم في الآخرة٥٨٣٢من لبس الحرير في الدنيا فلن يلبسه في الآخرة٥٨٣٣من لبس الحرير في الدنيا لم يلبسه في الآخرة٥٨٣٤من لبس الحرير في الدنيا لم يلبسه في الآخرة٥٨٣٥إنما يلبس الحرير في الدنيا من لا خلاق له في الآخرة٥٨٣٦ونتعجب منه، فقال النبي صلى الله عليه وسلم " أتعجبون من هذا " . قلنا…٥٨٣٧وأن نأكل فيها، وعن لبس الحرير والديباج، وأن نجلس عليه .٥٨٣٨حدثنا محمد بن مقاتل، أخبرنا عبد الله، أخبرنا سفيان، عن أشعث بن أبي…٥٨٣٩قال رخص النبي صلى الله عليه وسلم للزبير وعبد الرحمن في لبس الحرير…٥٨٤٠فخرجت فيها، فرأيت الغضب في وجهه، فشققتها بين نسائي .٥٨٤١إنما يلبس هذه من لا خلاق له٥٨٤٢أنه رأى على أم كلثوم عليها السلام بنت رسول الله صلى الله عليه وسلم…٥٨٤٣فنزل يوما منزلا فدخل الأراك، فلما خرج سألته فقال عائشة وحفصة ثم قال…٥٨٤٤لا إله إلا الله، ماذا أنزل الليلة من الفتنة، ماذا أنزل من الخزائن، من…٥٨٤٥من ترون نكسوها هذه الخميصة٥٨٤٦قال نهى النبي صلى الله عليه وسلم أن يتزعفر الرجل .٥٨٤٧حدثنا أبو نعيم، حدثنا سفيان، عن عبد الله بن دينار، عن ابن عمر رضى…٥٨٤٨سمع البراء رضى الله عنه يقول كان النبي صلى الله عليه وسلم مربوعا، وقد…٥٨٤٩واتباع الجنائز، وتشميت العاطس، ونهانا عن لبس الحرير، والديباج،…٥٨٥٠قال سألت أنسا أكان النبي صلى الله عليه وسلم يصلي في نعليه قال نعم .٥٨٥١أنه قال لعبد الله بن عمر رضى الله عنهما رأيتك تصنع أربعا لم أر أحدا…٥٨٥٢من لم يجد نعلين فليلبس خفين، وليقطعهما أسفل من الكعبين٥٨٥٣من لم يكن له إزار فليلبس السراويل، ومن لم يكن له نعلان فليلبس خفين٥٨٥٤حدثنا حجاج بن منهال، حدثنا شعبة، قال أخبرني أشعث بن سليم، سمعت أبي…٥٨٥٥إذا انتعل أحدكم فليبدأ باليمين وإذا نزع فليبدأ بالشمال، لتكن اليمنى…٥٨٥٦لا يمشي أحدكم في نعل واحدة ليحفهما جميعا، أو لينعلهما جميعا٥٨٥٧النبي صلى الله عليه وسلم كان لها قبالان .٥٨٥٨فقال ثابت البناني هذه نعل النبي صلى الله عليه وسلم .٥٨٥٩قال أتيت النبي صلى الله عليه وسلم وهو في قبة حمراء من أدم، ورأيت…٥٨٦٠وجمعهم في قبة من أدم .٥٨٦١يا أيها الناس خذوا من الأعمال ما تطيقون، فإن الله لا يمل حتى تملوا،…٥٨٦٢يا مخرمة هذا خبأناه لك٥٨٦٣والإستبرق، والديباج، والميثرة الحمراء، والقسي، وآنية الفضة، وأمرنا…٥٨٦٤حدثني محمد بن بشار، حدثنا غندر، حدثنا شعبة، عن قتادة، عن النضر بن…٥٨٦٥وجعل فصه مما يلي كفه، فاتخذه الناس، فرمى به، واتخذ خاتما من ورق أو…٥٨٦٦وجعل فصه مما يلي كفه، ونقش فيه محمد رسول الله . فاتخذ الناس مثله،…٥٨٦٧حدثنا عبد الله بن مسلمة، عن مالك، عن عبد الله بن دينار، عن عبد الله…٥٨٦٨ثم إن الناس اصطنعوا الخواتيم من ورق ولبسوها، فطرح رسول الله صلى الله…٥٨٦٩إن الناس قد صلوا وناموا، وإنكم لم تزالوا في صلاة ما انتظرتموها٥٨٧٠حدثنا إسحاق، أخبرنا معتمر، قال سمعت حميدا، يحدث عن أنس رضى الله عنه…٥٨٧١عندك شىء تصدقها٥٨٧٢فقيل له إنهم لا يقبلون كتابا إلا عليه خاتم، فاتخذ النبي صلى الله عليه…٥٨٧٣وكان في يده، ثم كان بعد في يد أبي بكر، ثم كان بعد في يد عمر، ثم كان…٥٨٧٤إنا اتخذنا خاتما، ونقشنا فيه نقشا، فلا ينقش عليه أحد٥٨٧٥ونقشه محمد رسول الله . فكأنما أنظر إلى بياضه في يده .٥٨٧٦إني كنت اصطنعته، وإني لا ألبسه٥٨٧٧إني اتخذت خاتما من ورق، ونقشت فيه، محمد رسول الله . فلا ينقشن أحد على…٥٨٧٨أن أبا بكر رضى الله عنه لما استخلف كتب له، وكان نقش الخاتم ثلاثة أسطر…٥٨٧٩قال كان خاتم النبي صلى الله عليه وسلم في يده، وفي يد أبي بكر بعده،…٥٨٨٠حدثنا أبو عاصم، أخبرنا ابن جريج، أخبرنا الحسن بن مسلم، عن طاوس، عن…٥٨٨١لم يصل قبل ولا بعد، ثم أتى النساء فأمرهن بالصدقة، فجعلت المرأة تصدق…٥٨٨٢فبعث النبي صلى الله عليه وسلم في طلبها رجالا، فحضرت الصلاة وليسوا على…٥٨٨٣لم يصل قبلها ولا بعدها ثم أتى النساء ومعه بلال فأمرهن بالصدقة، فجعلت…٥٨٨٤أين لكع ثلاثا ادع الحسن بن علي٥٨٨٥حدثنا محمد بن بشار، حدثنا غندر، حدثنا شعبة، عن قتادة، عن عكرمة، عن…٥٨٨٦أخرجوهم من بيوتكم٥٨٨٧لا يدخلن هؤلاء عليكن٥٨٨٨من الفطرة قص الشارب٥٨٨٩الفطرة خمس أو خمس من الفطرة الختان، والاستحداد، ونتف الإبط، وتقليم…٥٨٩٠من الفطرة حلق العانة، وتقليم الأظفار، وقص الشارب٥٨٩١الفطرة خمس الختان، والاستحداد، وقص الشارب، وتقليم الأظفار، ونتف الآباط٥٨٩٢خالفوا المشركين، وفروا اللحى، وأحفوا الشوارب٥٨٩٣انهكوا الشوارب، وأعفوا اللحى٥٨٩٤قال سألت أنسا أخضب النبي صلى الله عليه وسلم قال لم يبلغ الشيب إلا…٥٨٩٥صلى الله عليه وسلم فقال إنه لم يبلغ ما يخضب، لو شئت أن أعد شمطاته في…٥٨٩٦قال أرسلني أهلي إلى أم سلمة بقدح من ماء وقبض إسرائيل ثلاث أصابع من…٥٨٩٧حدثنا موسى بن إسماعيل، حدثنا سلام، عن عثمان بن عبد الله بن موهب، قال…٥٨٩٨أن أم سلمة، أرته شعر النبي صلى الله عليه وسلم أحمر .٥٨٩٩إن اليهود والنصارى لا يصبغون فخالفوهم٥٩٠٠ولا بالقصير، وليس بالأبيض الأمهق، وليس بالآدم، وليس بالجعد القطط، ولا…٥٩٠١ما حدث به قط إلا ضحك . تابعه شعبة شعره يبلغ شحمة أذنيه .٥٩٠٢أراني الليلة عند الكعبة، فرأيت رجلا آدم كأحسن ما أنت راء من أدم…٥٩٠٣أن النبي صلى الله عليه وسلم كان يضرب شعره منكبيه .٥٩٠٤كان يضرب شعر النبي صلى الله عليه وسلم منكبيه .٥٩٠٥رسول الله صلى الله عليه وسلم فقال كان شعر رسول الله صلى الله عليه…٥٩٠٦قال كان النبي صلى الله عليه وسلم ضخم اليدين، لم أر بعده مثله، وكان…٥٩٠٧لم أر بعده ولا قبله مثله، وكان بسط الكفين .٥٩٠٩قال كان النبي صلى الله عليه وسلم ضخم القدمين، حسن الوجه، لم أر بعده…٥٩١٠كان النبي صلى الله عليه وسلم شثن القدمين والكفين .٥٩١٢. أو جابر بن عبد الله كان النبي صلى الله عليه وسلم ضخم الكفين…٥٩١٣أما إبراهيم فانظروا إلى صاحبكم، وأما موسى فرجل آدم جعد، على جمل أحمر…٥٩١٤أن عبد الله بن عمر، قال سمعت عمر رضى الله عنه يقول من ضفر فليحلق، ولا…٥٩١٥لبيك اللهم لبيك، لبيك لا شريك لك لبيك، إن الحمد والنعمة لك والملك لا…٥٩١٦إني لبدت رأسي، وقلدت هديي، فلا أحل حتى أنحر٥٩١٧وكان أهل الكتاب يسدلون أشعارهم، وكان المشركون يفرقون رءوسهم، فسدل…٥٩١٨وعبد الله بن رجاء، قالا حدثنا شعبة، عن الحكم، عن إبراهيم، عن الأسود،…٥٩١٩وكان رسول الله صلى الله عليه وسلم عندها في ليلتها قال فقام رسول الله…٥٩٢٠أن عمر بن نافع، أخبره عن نافع، مولى عبد الله أنه سمع ابن عمر رضى الله…٥٩٢١حدثنا مسلم بن إبراهيم، حدثنا عبد الله بن المثنى بن عبد الله بن أنس بن…٥٩٢٢قالت طيبت النبي صلى الله عليه وسلم بيدي لحرمه، وطيبته بمنى قبل أن…٥٩٢٣قالت كنت أطيب النبي صلى الله عليه وسلم بأطيب ما يجد، حتى أجد وبيص…٥٩٢٤لو علمت أنك تنظر لطعنت بها في عينك، إنما جعل الإذن من قبل الأبصار٥٩٢٥مثله .٥٩٢٦حدثنا أبو الوليد، حدثنا شعبة، عن أشعث بن سليم، عن أبيه، عن مسروق، عن…٥٩٢٧كل عمل ابن آدم له، إلا الصوم فإنه لي، وأنا أجزي به، ولخلوف فم الصائم…٥٩٢٨حدثنا موسى، حدثنا وهيب، حدثنا هشام، عن عثمان بن عروة، عن أبيه، عن…٥٩٢٩وزعم أن النبي صلى الله عليه وسلم كان لا يرد الطيب .٥٩٣٠سمع عروة، والقاسم، يخبران عن عائشة، قالت طيبت رسول الله صلى الله عليه…٥٩٣١والمستوشمات، والمتنمصات والمتفلجات للحسن، المغيرات خلق الله تعالى،…٥٩٣٢إنما هلكت بنو إسرائيل حين اتخذ هذه نساؤهم٥٩٣٣لعن الله الواصلة والمستوصلة، والواشمة والمستوشمة٥٩٣٤لعن الله الواصلة والمستوصلة٥٩٣٥قال حدثتني أمي، عن أسماء بنت أبي بكر رضى الله عنهما أن امرأة، جاءت…
