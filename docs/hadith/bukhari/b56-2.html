كتاب الجهاد والسير

٢٩٤ حديثًا•صفحة ٢ من ٢
٢٩٣٨قيل له إنهم لا يقرءون كتابا إلا أن يكون مختوما . فاتخذ خاتما من فضة،…٢٩٣٩أن عبد الله بن عباس، أخبره أن رسول الله صلى الله عليه وسلم بعث بكتابه…٢٩٤١بسم الله الرحمن الرحيم من محمد عبد الله ورسوله، إلى هرقل عظيم الروم،…٢٩٤٢لأعطين الراية رجلا يفتح الله على يديه٢٩٤٣حدثنا عبد الله بن محمد، حدثنا معاوية بن عمرو، حدثنا أبو إسحاق، عن…٢٩٤٤حدثنا قتيبة، حدثنا إسماعيل بن جعفر، عن حميد، عن أنس، أن النبي صلى…٢٩٤٥الله أكبر، خربت خيبر، إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٢٩٤٦أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله . فمن قال لا إله إلا…٢٩٤٧حدثنا يحيى بن بكير، حدثنا الليث، عن عقيل، عن ابن شهاب، قال أخبرني عبد…٢٩٤٨حتى كانت غزوة تبوك، فغزاها رسول الله صلى الله عليه وسلم في حر شديد،…٢٩٤٩أن كعب بن مالك رضى الله عنه كان يقول لقلما كان رسول الله صلى الله…٢٩٥٠وكان يحب أن يخرج يوم الخميس .٢٩٥١والعصر بذي الحليفة ركعتين، وسمعتهم يصرخون بهما جميعا .٢٩٥٢ولا نرى إلا الحج، فلما دنونا من مكة أمر رسول الله صلى الله عليه وسلم…٢٩٥٣حدثنا علي بن عبد الله، حدثنا سفيان، قال حدثني الزهري، عن عبيد الله،…٢٩٥٤إن لقيتم فلانا وفلانا٢٩٥٥السمع والطاعة حق، ما لم يؤمر بالمعصية، فإذا أمر بمعصية فلا سمع ولا…٢٩٥٧نحن الآخرون السابقون٢٩٥٨كانت رحمة من الله . فسألت نافعا على أى شىء بايعهم على الموت قال لا،…٢٩٥٩حدثنا موسى بن إسماعيل، حدثنا وهيب، حدثنا عمرو بن يحيى، عن عباد بن…٢٩٦٠يا ابن الأكوع، ألا تبايع٢٩٦١حدثنا حفص بن عمر، حدثنا شعبة، عن حميد، قال سمعت أنسا رضى الله عنه…٢٩٦٣مضت الهجرة لأهلها٢٩٦٤فقال أرأيت رجلا مؤديا نشيطا، يخرج مع أمرائنا في المغازي، فيعزم علينا…٢٩٦٦أيها الناس، لا تتمنوا لقاء العدو، وسلوا الله العافية، فإذا لقيتموهم…٢٩٦٧فما زال بين يدى الإبل قدامها يسير . فقال لي " كيف ترى بعيرك " . قال…٢٩٦٨ما رأينا من شىء، وإن وجدناه لبحرا٢٩٦٩لم تراعوا، إنه لبحر٢٩٧٠لا تشتره، ولا تعد في صدقتك٢٩٧١لا تبتعه، ولا تعد في صدقتك٢٩٧٢لولا أن أشق على أمتي ما تخلفت عن سرية، ولكن لا أجد حمولة، ولا أجد ما…٢٩٧٣أيدفع يده إليك فتقضمها كما يقضم الفحل٢٩٧٤أن قيس بن سعد الأنصاري رضى الله عنه وكان صاحب لواء رسول الله صلى الله…٢٩٧٥لأعطين الراية أو قال ليأخذن غدا رجل يحبه الله ورسوله أو قال يحب الله…٢٩٧٦حدثنا محمد بن العلاء، حدثنا أبو أسامة، عن هشام بن عروة، عن أبيه، عن…٢٩٧٧بعثت بجوامع الكلم، ونصرت بالرعب، فبينا أنا نائم أتيت بمفاتيح خزائن…٢٩٧٨أن ابن عباس رضى الله عنهما أخبره أن أبا سفيان أخبره أن هرقل أرسل إليه…٢٩٧٩حدثتني أيضا، فاطمة عن أسماء رضى الله عنها قالت صنعت سفرة رسول الله…٢٩٨٠حدثنا علي بن عبد الله، أخبرنا سفيان، عن عمرو، قال أخبرني عطاء، سمع…٢٩٨١أن سويد بن النعمان رضى الله عنه أخبره أنه، خرج مع النبي صلى الله عليه…٢٩٨٢ناد في الناس يأتون بفضل أزوادهم٢٩٨٣ففني زادنا، حتى كان الرجل منا يأكل في كل يوم تمرة . قال رجل يا أبا…٢٩٨٤اذهبي وليردفك عبد الرحمن٢٩٨٥حدثني عبد الله، حدثنا ابن عيينة، عن عمرو بن دينار، عن عمرو بن أوس، عن…٢٩٨٦وإنهم ليصرخون بهما جميعا الحج والعمرة .٢٩٨٧على إكاف عليه قطيفة، وأردف أسامة وراءه .٢٩٨٨مردفا أسامة بن زيد ومعه بلال ومعه عثمان بن طلحة من الحجبة، حتى أناخ…٢٩٨٩كل سلامى من الناس عليه صدقة كل يوم تطلع فيه الشمس، يعدل بين الاثنين…٢٩٩٠حدثنا عبد الله بن مسلمة، عن مالك، عن نافع، عن عبد الله بن عمر رضى…٢٩٩١الله أكبر، خربت خيبر، إنا إذا نزلنا بساحة قوم فساء صباح المنذرين٢٩٩٢يا أيها الناس، اربعوا على أنفسكم، فإنكم لا تدعون أصم ولا غائبا، إنه…٢٩٩٣حدثنا محمد بن يوسف، حدثنا سفيان، عن حصين بن عبد الرحمن، عن سالم بن…٢٩٩٤حدثنا محمد بن بشار، حدثنا ابن أبي عدي، عن شعبة، عن حصين، عن سالم، عن…٢٩٩٥لا إله إلا الله، وحده لا شريك له، له الملك، وله الحمد، وهو على كل شىء…٢٩٩٦إذا مرض العبد أو سافر، كتب له مثل ما كان يعمل مقيما صحيحا٢٩٩٧إن لكل نبي حواريا، وحواري الزبير٢٩٩٨لو يعلم الناس ما في الوحدة ما أعلم ما سار راكب بليل وحده٢٩٩٩قال فكان يسير العنق، فإذا وجد فجوة نص . والنص فوق العنق .٣٠٠٠قال كنت مع عبد الله بن عمر رضى الله عنهما بطريق مكة، فبلغه عن صفية…٣٠٠١السفر قطعة من العذاب، يمنع أحدكم نومه وطعامه وشرابه، فإذا قضى أحدكم…٣٠٠٢لا تبتعه، ولا تعد في صدقتك٣٠٠٣لا تشتره وإن بدرهم، فإن العائد في هبته كالكلب يعود في قيئه٣٠٠٤حدثنا آدم، حدثنا شعبة، حدثنا حبيب بن أبي ثابت، قال سمعت أبا العباس…٣٠٠٥أن أبا بشير الأنصاري رضى الله عنه أخبره أنه، كان مع رسول الله صلى…٣٠٠٦لا يخلون رجل بامرأة، ولا تسافرن امرأة إلا ومعها محرم٣٠٠٧انطلقوا حتى تأتوا روضة خاخ، فإن بها ظعينة ومعها كتاب، فخذوه منها٣٠٠٨سمع جابر بن عبد الله رضى الله عنهما قال لما كان يوم بدر أتي بأسارى،…٣٠٠٩لأعطين الراية غدا رجلا يفتح على يديه، يحب الله ورسوله، ويحبه الله…٣٠١٠عجب الله من قوم يدخلون الجنة في السلاسل٣٠١١ثلاثة يؤتون أجرهم مرتين الرجل تكون له الأمة فيعلمها فيحسن تعليمها،…٣٠١٢حدثنا علي بن عبد الله، حدثنا سفيان، حدثنا الزهري، عن عبيد الله، عن…٣٠١٣أنه سمع عبيد الله، عن ابن عباس، حدثنا الصعب، في الذراري كان عمرو…٣٠١٤أن عبد الله رضى الله عنه أخبره أن امرأة وجدت في بعض مغازي النبي صلى…٣٠١٥قال قلت لأبي أسامة حدثكم عبيد الله، عن نافع، عن ابن عمر رضى الله…٣٠١٦إن وجدتم فلانا وفلانا فأحرقوهما بالنار٣٠١٧لا تعذبوا بعذاب الله٣٠١٨ما أجد لكم إلا أن تلحقوا بالذود٣٠١٩قرصت نملة نبيا من الأنبياء، فأمر بقرية النمل فأحرقت، فأوحى الله إليه…٣٠٢٠ألا تريحني من ذي الخلصة٣٠٢١حدثنا محمد بن كثير، أخبرنا سفيان، عن موسى بن عقبة، عن نافع، عن ابن…٣٠٢٢فانطلق رجل منهم فدخل حصنهم قال فدخلت في مربط دواب لهم، قال وأغلقوا…٣٠٢٣فقتله وهو نائم .٣٠٢٥أيها الناس لا تمنوا لقاء العدو وسلوا الله العافية، فإذا لقيتموهم…٣٠٢٦لا تمنوا لقاء العدو، فإذا لقيتموهم فاصبروا٣٠٢٨هلك كسرى ثم لا يكون كسرى بعده، وقيصر ليهلكن ثم لا يكون قيصر بعده،…٣٠٢٩حدثنا أبو بكر بن أصرم، أخبرنا عبد الله، أخبرنا معمر، عن همام بن منبه،…٣٠٣٠سمع جابر بن عبد الله رضى الله عنهما قال قال النبي صلى الله عليه وسلم…٣٠٣١من لكعب بن الأشرف، فإنه قد آذى الله ورسوله٣٠٣٢من لكعب بن الأشرف٣٠٣٣فحدث به في نخل، فلما دخل عليه رسول الله صلى الله عليه وسلم النخل، طفق…٣٠٣٤حدثنا مسدد، حدثنا أبو الأحوص، حدثنا أبو إسحاق، عن البراء رضى الله عنه…٣٠٣٦اللهم ثبته واجعله هاديا مهديا٣٠٣٧قال سألوا سهل بن سعد الساعدي رضى الله عنه بأى شىء دووي جرح النبي صلى…٣٠٣٨يسرا ولا تعسرا، وبشرا ولا تنفرا، وتطاوعا ولا تختلفا٣٠٣٩إن رأيتمونا تخطفنا الطير، فلا تبرحوا مكانكم هذا حتى أرسل إليكم، وإن…٣٠٤٠لم تراعوا، لم تراعوا٣٠٤١يا ابن الأكوع، ملكت فأسجح . إن القوم يقرون في قومهم٣٠٤٢قال سأل رجل البراء رضى الله عنه فقال يا أبا عمارة، أوليتم يوم حنين…٣٠٤٣قوموا إلى سيدكم٣٠٤٤فلما نزعه جاء رجل فقال إن ابن خطل متعلق بأستار الكعبة، فقال " اقتلوه…٣٠٤٥وأمر عليهم عاصم بن ثابت الأنصاري جد عاصم بن عمر، فانطلقوا حتى إذا…٣٠٤٦فكوا العاني يعني الأسير وأطعموا الجائع وعودوا المريض٣٠٤٧أن عامرا، حدثهم عن أبي جحيفة رضى الله عنه قال قلت لعلي رضى الله عنه…٣٠٤٨لا تدعون منها درهما٣٠٤٩قال أتي النبي صلى الله عليه وسلم بمال من البحرين، فجاءه العباس فقال…٣٠٥٠حدثني محمود، حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن محمد بن…٣٠٥١قال أتى النبي صلى الله عليه وسلم عين من المشركين وهو في سفر، فجلس عند…٣٠٥٢وأن يقاتل من ورائهم، ولا يكلفوا إلا طاقتهم .٣٠٥٣ائتوني بكتاب أكتب لكم كتابا لن تضلوا بعده أبدا٣٠٥٤إنما هذه لباس من لا خلاق له، أو إنما يلبس هذه من لا خلاق له٣٠٥٥أتشهد أني رسول الله صلى الله عليه وسلم٣٠٥٧قال ابن عمر انطلق النبي صلى الله عليه وسلم وأبى بن كعب يأتيان النخل…٣٠٥٨وهل ترك لنا عقيل منزلا٣٠٥٩أن عمر بن الخطاب رضى الله عنه استعمل مولى له يدعى هنيا على الحمى فقال…٣٠٦٠اكتبوا لي من تلفظ بالإسلام من الناس٣٠٦١ارجع فحج مع امرأتك٣٠٦٢هذا من أهل النار٣٠٦٣أخذ الراية زيد فأصيب، ثم أخذها جعفر فأصيب، ثم أخذها عبد الله بن رواحة…٣٠٦٤وسهل بن يوسف، عن سعيد، عن قتادة، عن أنس رضى الله عنه أن النبي صلى…٣٠٦٥حدثنا محمد بن عبد الرحيم، حدثنا روح بن عبادة، حدثنا سعيد، عن قتادة،…٣٠٦٦أن أنسا، أخبره قال اعتمر النبي صلى الله عليه وسلم من الجعرانة، حيث…٣٠٦٧فأخذه العدو، فظهر عليه المسلمون فرد عليه في زمن رسول الله صلى الله…٣٠٦٨أن عبدا، لابن عمر أبق فلحق بالروم، فظهر عليه خالد بن الوليد، فرده على…٣٠٦٩وأمير المسلمين يومئذ خالد بن الوليد، بعثه أبو بكر، فأخذه العدو، فلما…٣٠٧٠يا أهل الخندق، إن جابرا قد صنع سؤرا، فحى هلا بكم٣٠٧١قالت أتيت رسول الله صلى الله عليه وسلم مع أبي وعلى قميص أصفر، قال…٣٠٧٢كخ كخ، أما تعرف أنا لا نأكل الصدقة٣٠٧٣لا ألفين أحدكم يوم القيامة على رقبته شاة لها ثغاء على رقبته فرس له…٣٠٧٤قال كان على ثقل النبي صلى الله عليه وسلم رجل يقال له كركرة فمات، فقال…٣٠٧٥هذه البهائم لها أوابد كأوابد الوحش، فما ند عليكم فاصنعوا به هكذا٣٠٧٦ألا تريحني من ذي الخلصة٣٠٧٧لا هجرة ولكن جهاد ونية، وإذا استنفرتم فانفروا٣٠٧٩لا هجرة بعد فتح مكة، ولكن أبايعه على الإسلام٣٠٨٠حدثنا علي بن عبد الله، حدثنا سفيان، قال عمرو وابن جريج سمعت عطاء،…٣٠٨١ائتوا روضة كذا، وتجدون بها امرأة أعطاها حاطب كتابا٣٠٨٢وحميد بن الأسود، عن حبيب بن الشهيد، عن ابن أبي مليكة، قال ابن الزبير…٣٠٨٣حدثنا مالك بن إسماعيل، حدثنا ابن عيينة، عن الزهري، قال قال السائب بن…٣٠٨٤آيبون إن شاء الله تائبون عابدون حامدون لربنا ساجدون، صدق الله وعده،…٣٠٨٥ورسول الله صلى الله عليه وسلم على راحلته، وقد أردف صفية بنت حيى،…٣٠٨٦لا، ولكن عليك بالمرأة٣٠٨٧ادخل المسجد فصل ركعتين٣٠٨٨وعمه، عبيد الله بن كعب عن كعب رضى الله عنه أن النبي صلى الله عليه…٣٠٨٩فلما قدم صرارا أمر ببقرة فذبحت فأكلوا منها، فلما قدم المدينة أمرني أن…٣٠٩٠قال قدمت من سفر فقال النبي صلى الله عليه وسلم " صل ركعتين " . صرار…
