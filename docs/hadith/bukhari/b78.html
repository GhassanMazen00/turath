٢٥٠ حديثًا•صفحة ١ من ٢
٥٩٧٠الصلاة على وقتها٥٩٧١حدثنا قتيبة بن سعيد، حدثنا جرير، عن عمارة بن القعقاع بن شبرمة، عن أبي…٥٩٧٢وشعبة، قالا حدثنا حبيب، ح قال وحدثنا محمد بن كثير، أخبرنا سفيان، عن…٥٩٧٣إن من أكبر الكبائر أن يلعن الرجل والديه٥٩٧٤بينما ثلاثة نفر يتماشون أخذهم المطر، فمالوا إلى غار في الجبل، فانحطت…٥٩٧٥إن الله حرم عليكم عقوق الأمهات، ومنع وهات، ووأد البنات، وكره لكم قيل…٥٩٧٦ألا أنبئكم بأكبر الكبائر٥٩٧٧الشرك بالله، وقتل النفس، وعقوق الوالدين٥٩٧٨حدثنا الحميدي، حدثنا سفيان، حدثنا هشام بن عروة، أخبرني أبي، أخبرتني…٥٩٧٩قالت قدمت أمي وهى مشركة في عهد قريش ومدتهم، إذ عاهدوا النبي صلى الله…٥٩٨٠أن عبد الله بن عباس، أخبره أن أبا سفيان أخبره أن هرقل أرسل إليه فقال…٥٩٨١إنما يلبس هذه من لا خلاق له٥٩٨٢قال قيل يا رسول الله، أخبرني بعمل، يدخلني الجنة .٥٩٨٣وأبوه، عثمان بن عبد الله أنهما سمعا موسى بن طلحة، عن أبي أيوب…٥٩٨٤لا يدخل الجنة قاطع٥٩٨٥من سره أن يبسط له في رزقه، وأن ينسأ له في أثره، فليصل رحمه٥٩٨٦من أحب أن يبسط له في رزقه، وينسأ له في أثره، فليصل رحمه٥٩٨٧إن الله خلق الخلق حتى إذا فرغ من خلقه، قالت الرحم هذا مقام العائذ بك…٥٩٨٨إن الرحم سجنة من الرحمن، فقال الله من وصلك وصلته، ومن قطعك قطعته٥٩٨٩الرحم شجنة، فمن وصلها وصلته، ومن قطعها قطعته٥٩٩٠أن عمرو بن العاص، قال سمعت النبي صلى الله عليه وسلم جهارا غير سر يقول…٥٩٩١ليس الواصل بالمكافئ، ولكن الواصل الذي إذا قطعت رحمه وصلها٥٩٩٢أسلمت على ما سلف من خير٥٩٩٣قالت أتيت رسول الله صلى الله عليه وسلم مع أبي وعلى قميص أصفر، قال…٥٩٩٤هما ريحانتاى من الدنيا٥٩٩٥من يلي من هذه البنات شيئا فأحسن إليهن كن له سترا من النار٥٩٩٦فصلى فإذا ركع وضعها، وإذا رفع رفعها .٥٩٩٧من لا يرحم لا يرحم٥٩٩٨أو أملك لك أن نزع الله من قلبك الرحمة٥٩٩٩أترون هذه طارحة ولدها في النار٦٠٠٠جعل الله الرحمة مائة جزء، فأمسك عنده تسعة وتسعين جزءا، وأنزل في الأرض…٦٠٠١أن تجعل لله ندا وهو خلقك٦٠٠٢أن النبي صلى الله عليه وسلم وضع صبيا في حجره يحنكه، فبال عليه، فدعا…٦٠٠٣اللهم ارحمهما فإني أرحمهما٦٠٠٤ولقد هلكت قبل أن يتزوجني بثلاث سنين، لما كنت أسمعه يذكرها، ولقد أمره…٦٠٠٥أنا وكافل اليتيم، في الجنة هكذا٦٠٠٦الساعي على الأرملة والمسكين كالمجاهد في سبيل الله، أو كالذي يصوم…٦٠٠٧الساعي على الأرملة والمسكين كالمجاهد في سبيل الله وأحسبه قال، يشك…٦٠٠٨ارجعوا إلى أهليكم فعلموهم ومروهم، وصلوا كما رأيتموني أصلي، وإذا حضرت…٦٠٠٩بينما رجل يمشي بطريق اشتد عليه العطش، فوجد بئرا فنزل فيها فشرب ثم…٦٠١٠لقد حجرت واسعا٦٠١١ترى المؤمنين في تراحمهم وتوادهم وتعاطفهم كمثل الجسد إذا اشتكى عضوا…٦٠١٢ما من مسلم غرس غرسا فأكل منه إنسان أو دابة إلا كان له صدقة٦٠١٣من لا يرحم لا يرحم٦٠١٤ما زال يوصيني جبريل بالجار حتى ظننت أنه سيورثه٦٠١٥ما زال جبريل يوصيني بالجار حتى ظننت أنه سيورثه٦٠١٦والله لا يؤمن، والله لا يؤمن، والله لا يؤمن٦٠١٧يا نساء المسلمات لا تحقرن جارة لجارتها ولو فرسن شاة٦٠١٨من كان يؤمن بالله واليوم الآخر فلا يؤذ جاره، ومن كان يؤمن بالله…٦٠١٩من كان يؤمن بالله واليوم الآخر فليكرم جاره، ومن كان يؤمن بالله واليوم…٦٠٢٠إلى أقربهما منك بابا٦٠٢١حدثنا علي بن عياش، حدثنا أبو غسان، قال حدثني محمد بن المنكدر، عن جابر…٦٠٢٢على كل مسلم صدقة٦٠٢٣اتقوا النار ولو بشق تمرة، فإن لم تجد فبكلمة طيبة٦٠٢٤مهلا يا عائشة، إن الله يحب الرفق في الأمر كله٦٠٢٥أن أعرابيا، بال في المسجد، فقاموا إليه، فقال رسول الله صلى الله عليه…٦٠٢٧المؤمن للمؤمن كالبنيان، يشد بعضه بعضا٦٠٢٨اشفعوا فلتؤجروا، وليقض الله على لسان رسوله ما شاء٦٠٢٩إن من أخيركم أحسنكم خلقا٦٠٣٠مهلا يا عائشة، عليك بالرفق، وإياك والعنف والفحش٦٠٣١ما له، ترب جبينه٦٠٣٢بئس أخو العشيرة، وبئس ابن العشيرة٦٠٣٣لن تراعوا، لن تراعوا٦٠٣٤حدثنا محمد بن كثير، أخبرنا سفيان، عن ابن المنكدر، قال سمعت جابرا رضى…٦٠٣٥إن خياركم أحاسنكم أخلاقا٦٠٣٦قال جاءت امرأة إلى النبي صلى الله عليه وسلم ببردة . فقال سهل للقوم…٦٠٣٧يتقارب الزمان وينقص العمل، ويلقى الشح ويكثر الهرج٦٠٣٨سمع سلام بن مسكين، قال سمعت ثابتا، يقول حدثنا أنس رضى الله عنه قال…٦٠٣٩قال سألت عائشة ما كان النبي صلى الله عليه وسلم يصنع في أهله قالت كان…٦٠٤٠إذا أحب الله عبدا نادى جبريل إن الله يحب فلانا، فأحبه . فيحبه جبريل،…٦٠٤١لا يجد أحد حلاوة الإيمان حتى يحب المرء، لا يحبه إلا لله، وحتى أن يقذف…٦٠٤٢بم يضرب أحدكم امرأته ضرب الفحل، ثم لعله يعانقها٦٠٤٣أتدرون أى يوم هذا٦٠٤٤سباب المسلم فسوق، وقتاله كفر٦٠٤٥لا يرمي رجل رجلا بالفسوق، ولا يرميه بالكفر، إلا ارتدت عليه، إن لم يكن…٦٠٤٦ما له، ترب جبينه٦٠٤٧من حلف على ملة غير الإسلام فهو كما قال، وليس على ابن آدم نذر فيما لا…٦٠٤٨إني لأعلم كلمة لو قالها لذهب عنه الذي يجد٦٠٤٩خرجت لأخبركم، فتلاحى فلان وفلان وإنها رفعت، وعسى أن يكون خيرا لكم،…٦٠٥٠قال رأيت عليه بردا وعلى غلامه بردا فقلت لو أخذت هذا فلبسته كانت حلة،…٦٠٥١لم أنس ولم تقصر٦٠٥٢إنهما ليعذبان، وما يعذبان في كبير، أما هذا فكان لا يستتر من بوله،…٦٠٥٣خير دور الأنصار بنو النجار٦٠٥٤ائذنوا له بئس، أخو العشيرة أو ابن العشيرة٦٠٥٥يعذبان، وما يعذبان في كبيرة، وإنه لكبير، كان أحدهما لا يستتر من…٦٠٥٦لا يدخل الجنة قتات٦٠٥٧من لم يدع قول الزور والعمل به والجهل فليس لله حاجة أن يدع طعامه وشرابه٦٠٥٨تجد من شر الناس يوم القيامة عند الله ذا الوجهين، الذي يأتي هؤلاء بوجه…٦٠٥٩رحم الله موسى، لقد أوذي بأكثر من هذا فصبر٦٠٦٠أهلكتم أو قطعتم ظهر الرجل٦٠٦١ويحك قطعت عنق صاحبك يقوله مرارا إن كان أحدكم مادحا لا محالة فليقل…٦٠٦٢أن رسول الله صلى الله عليه وسلم حين ذكر في الإزار ما ذكر، قال أبو بكر…٦٠٦٣يا عائشة إن الله أفتاني في أمر استفتيته فيه، أتاني رجلان، فجلس أحدهما…٦٠٦٤إياكم والظن، فإن الظن أكذب الحديث، ولا تحسسوا، ولا تجسسوا، ولا…٦٠٦٥لا تباغضوا، ولا تحاسدوا، ولا تدابروا، وكونوا عباد الله إخوانا، ولا…٦٠٦٦إياكم والظن، فإن الظن أكذب الحديث، ولا تحسسوا، ولا تجسسوا، ولا…٦٠٦٧ما أظن فلانا وفلانا يعرفان من ديننا شيئا٦٠٦٨يا عائشة ما أظن فلانا وفلانا يعرفان ديننا الذي نحن عليه٦٠٦٩كل أمتي معافى إلا المجاهرين، وإن من المجانة أن يعمل الرجل بالليل…٦٠٧٠يدنو أحدكم من ربه حتى يضع كنفه عليه فيقول عملت كذا وكذا . فيقول نعم .…٦٠٧١ألا أخبركم بأهل الجنة، كل ضعيف متضاعف، لو أقسم على الله لأبره، ألا…٦٠٧٢قال كانت الأمة من إماء أهل المدينة لتأخذ بيد رسول الله صلى الله عليه…٦٠٧٥حديث ٦٠٧٥٦٠٧٦لا تباغضوا، ولا تحاسدوا، ولا تدابروا، وكونوا عباد الله إخوانا، ولا…٦٠٧٧لا يحل لرجل أن يهجر أخاه فوق ثلاث ليال، يلتقيان فيعرض هذا ويعرض هذا،…٦٠٧٨إني لأعرف غضبك ورضاك٦٠٧٩إني قد أذن لي بالخروج٦٠٨٠فلما أراد أن يخرج أمر بمكان من البيت، فنضح له على بساط، فصلى عليه،…٦٠٨١إنما يلبس الحرير من لا خلاق له٦٠٨٢حدثنا مسدد، حدثنا يحيى، عن حميد، عن أنس، قال لما قدم علينا عبد الرحمن…٦٠٨٣لا حلف في الإسلام٦٠٨٤لعلك تريدين أن ترجعي إلى رفاعة، لا، حتى تذوقي عسيلته، ويذوق عسيلتك٦٠٨٥عجبت من هؤلاء اللاتي كن عندي، لما سمعن صوتك تبادرن الحجاب٦٠٨٦إنا قافلون غدا إن شاء الله٦٠٨٧حدثنا موسى، حدثنا إبراهيم، أخبرنا ابن شهاب، عن حميد بن عبد الرحمن، أن…٦٠٨٨قال كنت أمشي مع رسول الله صلى الله عليه وسلم وعليه برد نجراني غليظ…٦٠٩٠اللهم ثبته واجعله هاديا مهديا٦٠٩١نعم إذا رأت الماء٦٠٩٢أن أبا النضر، حدثه عن سليمان بن يسار، عن عائشة رضى الله عنها قالت ما…٦٠٩٣اللهم حوالينا ولا علينا٦٠٩٤إن الصدق يهدي إلى البر، وإن البر يهدي إلى الجنة، وإن الرجل ليصدق حتى…٦٠٩٥آية المنافق ثلاث إذا حدث كذب، وإذا وعد أخلف، وإذا اؤتمن خان٦٠٩٦رأيت رجلين أتياني قالا الذي رأيته يشق شدقه فكذاب يكذب بالكذبة تحمل…٦٠٩٧قال قلت لأبي أسامة حدثكم الأعمش، سمعت شقيقا، قال سمعت حذيفة، يقول إن…٦٠٩٨قال قال عبد الله إن أحسن الحديث كتاب الله، وأحسن الهدى هدى محمد صلى…٦٠٩٩ليس أحد أو ليس شىء أصبر على أذى سمعه من الله، إنهم ليدعون له ولدا،…٦١٠٠قد أوذي موسى بأكثر من ذلك فصبر٦١٠١ما بال أقوام يتنزهون عن الشىء أصنعه، فوالله إني لأعلمهم بالله وأشدهم…٦١٠٢قال كان النبي صلى الله عليه وسلم أشد حياء من العذراء في خدرها، فإذا…٦١٠٣إذا قال الرجل لأخيه يا كافر فقد باء به أحدهما٦١٠٤أيما رجل قال لأخيه يا كافر . فقد باء بها أحدهما٦١٠٥من حلف بملة غير الإسلام كاذبا فهو كما قال، ومن قتل نفسه بشىء عذب به…٦١٠٦يا معاذ أفتان أنت ثلاثا اقرأ { والشمس وضحاها } و { سبح اسم ربك الأعلى…٦١٠٧من حلف منكم فقال في حلفه باللات والعزى . فليقل لا إله إلا الله . ومن…٦١٠٨ألا إن الله ينهاكم أن تحلفوا بآبائكم، فمن كان حالفا فليحلف بالله،…٦١٠٩من أشد الناس عذابا يوم القيامة الذين يصورون هذه الصور٦١١٠يا أيها الناس إن منكم منفرين، فأيكم ما صلى بالناس فليتجوز، فإن فيهم…٦١١١إن أحدكم إذا كان في الصلاة فإن الله حيال وجهه، فلا يتنخمن حيال وجهه…٦١١٢عرفها سنة، ثم اعرف وكاءها وعفاصها، ثم استنفق بها، فإن جاء ربها فأدها…٦١١٣ما زال بكم صنيعكم حتى ظننت أنه سيكتب عليكم، فعليكم بالصلاة في بيوتكم،…٦١١٤ليس الشديد بالصرعة، إنما الشديد الذي يملك نفسه عند الغضب٦١١٥إني لأعلم كلمة لو قالها لذهب عنه ما يجد لو قال أعوذ بالله من الشيطان…٦١١٦قال للنبي صلى الله عليه وسلم أوصني . قال " لا تغضب " . فردد مرارا،…٦١١٧الحياء لا يأتي إلا بخير٦١١٨دعه فإن الحياء من الإيمان٦١١٩يقول كان النبي صلى الله عليه وسلم أشد حياء من العذراء في خدرها .٦١٢٠إن مما أدرك الناس من كلام النبوة الأولى إذا لم تستحي فاصنع ما شئت٦١٢١نعم إذا رأت الماء٦١٢٢مثل المؤمن كمثل شجرة خضراء، لا يسقط ورقها، ولا يتحات٦١٢٣أنه سمع أنسا رضى الله عنه يقول جاءت امرأة إلى النبي صلى الله عليه…
