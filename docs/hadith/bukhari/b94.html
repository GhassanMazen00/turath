٢٠ حديثًا
٧٢٢٦والذي نفسي بيده لولا أن رجالا يكرهون أن يتخلفوا بعدي ولا أجد ما…٧٢٢٧والذي نفسي بيده وددت أني لأقاتل في سبيل الله فأقتل ثم أحيا ثم أقتل،…٧٢٢٨لو كان عندي أحد ذهبا، لأحببت أن لا يأتي ثلاث وعندي منه دينار، ليس شىء…٧٢٢٩لو استقبلت من أمري ما استدبرت ما سقت الهدى، ولحللت مع الناس حين حلوا٧٢٣٠إني لو استقبلت من أمري ما استدبرت ما أهديت، ولولا أن معي الهدى لحللت٧٢٣١ليت رجلا صالحا من أصحابي يحرسني الليلة٧٢٣٢لا تحاسد إلا في اثنتين رجل آتاه الله القرآن، فهو يتلوه آناء الليل…٧٢٣٣لا تتمنوا الموت٧٢٣٤حدثنا محمد، حدثنا عبدة، عن ابن أبي خالد، عن قيس، قال أتينا خباب بن…٧٢٣٥لا يتمنى أحدكم الموت إما محسنا فلعله يزداد، وإما مسيئا فلعله يستعتب٧٢٣٦لولا أنت ما اهتدينا نحن، ولا تصدقنا ولا صلينا، فأنزلن سكينة علينا، إن…٧٢٣٧لا تتمنوا لقاء العدو، وسلوا الله العافية٧٢٣٨لو كنت راجما امرأة من غير بينة٧٢٣٩لولا أن أشق على أمتي أو على الناس، وقال سفيان أيضا، على أمتي لأمرتهم…٧٢٤٠لولا أن أشق على أمتي لأمرتهم بالسواك٧٢٤١لو مد بي الشهر لواصلت وصالا يدع المتعمقون تعمقهم، إني لست مثلكم، إني…٧٢٤٢أيكم مثلي، إني أبيت يطعمني ربي ويسقين٧٢٤٣ليدخلوا من شاءوا، ويمنعوا من شاءوا، لولا أن قومك حديث عهدهم…٧٢٤٤لولا الهجرة لكنت امرأ من الأنصار، ولو سلك الناس واديا وسلكت الأنصار…٧٢٤٥لولا الهجرة لكنت امرأ من الأنصار، ولو سلك الناس واديا أو شعبا، لسلكت…
