٢٥٠ حديثًا•صفحة ٢ من ٢
٦١٢٤يسرا ولا تعسرا، وبشرا ولا تنفرا، وتطاوعا٦١٢٥يسروا ولا تعسروا، وسكنوا ولا تنفروا٦١٢٦ما لم يكن إثما، فإن كان إثما كان أبعد الناس منه، وما انتقم رسول الله…٦١٢٧فجاء أبو برزة الأسلمي على فرس، فصلى وخلى فرسه، فانطلقت الفرس، فترك…٦١٢٨دعوه، وأهريقوا على بوله ذنوبا من ماء أو سجلا من ماء فإنما بعثتم…٦١٢٩يا أبا عمير ما فعل النغير٦١٣٠فيسربهن إلى فيلعبن معي .٦١٣١ائذنوا له فبئس ابن العشيرة٦١٣٢أن النبي صلى الله عليه وسلم أهديت له أقبية من ديباج مزررة بالذهب،…٦١٣٣لا يلدغ المؤمن من جحر واحد مرتين٦١٣٤ألم أخبر أنك تقوم الليل وتصوم النهار٦١٣٥من كان يؤمن بالله واليوم الآخر فليكرم ضيفه، جائزته يوم وليلة،…٦١٣٦من كان يؤمن بالله واليوم الآخر فلا يؤذ جاره، ومن كان يؤمن بالله…٦١٣٧إن نزلتم بقوم فأمروا لكم بما ينبغي للضيف فاقبلوا، فإن لم يفعلوا فخذوا…٦١٣٨من كان يؤمن بالله واليوم الآخر فليكرم ضيفه، ومن كان يؤمن بالله واليوم…٦١٣٩فلما كان الليل ذهب أبو الدرداء يقوم فقال نم . فنام، ثم ذهب يقوم فقال…٦١٤٠فإنه إن جاء ولم تطعموا لنلقين منه . فأبوا فعرفت أنه يجد على، فلما جاء…٦١٤١قال عبد الرحمن بن أبي بكر رضى الله عنهما جاء أبو بكر بضيف له أو…٦١٤٣وسهل بن أبي حثمة، أنهما حدثاه أن عبد الله بن سهل ومحيصة بن مسعود أتيا…٦١٤٤أخبروني بشجرة مثلها مثل المسلم، تؤتي أكلها كل حين بإذن ربها، ولا تحت…٦١٤٥إن من الشعر حكمة٦١٤٦هل أنت إلا إصبع دميت وفي سبيل الله ما لقيت٦١٤٧أصدق كلمة قالها الشاعر كلمة لبيد ألا كل شىء ما خلا الله باطل٦١٤٨قال وكان عامر رجلا شاعرا، فنزل يحدو بالقوم يقول اللهم لولا أنت ما…٦١٤٩ويحك يا أنجشة، رويدك سوقا بالقوارير٦١٥٠حدثنا محمد، حدثنا عبدة، أخبرنا هشام بن عروة، عن أبيه، عن عائشة رضى…٦١٥١إن أخا لكم لا يقول الرفث٦١٥٢يا حسان أجب عن رسول الله، اللهم أيده بروح القدس٦١٥٣اهجهم أو قال هاجهم وجبريل معك٦١٥٤لأن يمتلئ جوف أحدكم قيحا خير له من أن يمتلئ شعرا٦١٥٥لأن يمتلئ جوف رجل قيحا يريه خير من أن يمتلئ شعرا٦١٥٦ائذني له، فإنه عمك، تربت يمينك٦١٥٧عقرى حلقى لغة قريش إنك لحابستنا٦١٥٨مولى عمر بن عبيد الله أن أبا مرة، مولى أم هانئ بنت أبي طالب أخبره…٦١٥٩حدثنا موسى بن إسماعيل، حدثنا همام، عن قتادة، عن أنس رضى الله عنه أن…٦١٦٠حدثنا قتيبة بن سعيد، عن مالك، عن أبي الزناد، عن الأعرج، عن أبي هريرة…٦١٦١ويحك يا أنجشة رويدك بالقوارير٦١٦٢ويلك قطعت عنق أخيك ثلاثا من كان منكم مادحا لا محالة فليقل أحسب فلانا…٦١٦٣ويلك من يعدل إذا لم أعدل٦١٦٤حدثنا محمد بن مقاتل أبو الحسن، أخبرنا عبد الله، أخبرنا الأوزاعي، قال…٦١٦٥ويحك إن شأن الهجرة شديد، فهل لك من إبل٦١٦٦ويلكم أو ويحكم قال شعبة شك هو لا ترجعوا بعدي كفارا، يضرب بعضكم رقاب…٦١٦٧ويلك وما أعددت لها٦١٦٨المرء مع من أحب٦١٦٩المرء مع من أحب٦١٧٠المرء مع من أحب٦١٧١أن رجلا، سأل النبي صلى الله عليه وسلم متى الساعة يا رسول الله قال "…٦١٧٢قد خبأت لك خبيئا فما هو٦١٧٥حديث ٦١٧٥٦١٧٦مرحبا بالوفد الذين جاءوا غير خزايا ولا ندامى٦١٧٧الغادر يرفع له لواء يوم القيامة، يقال هذه غدرة فلان بن فلان٦١٧٨إن الغادر ينصب له لواء يوم القيامة، فيقال هذه غدرة فلان بن فلان٦١٧٩لا يقولن أحدكم خبثت نفسي . ولكن ليقل لقست نفسي٦١٨٠لا يقولن أحدكم خبثت نفسي، ولكن ليقل لقست نفسي٦١٨١قال الله يسب بنو آدم الدهر، وأنا الدهر، بيدي الليل والنهار٦١٨٢لا تسموا العنب الكرم، ولا تقولوا خيبة الدهر . فإن الله هو الدهر٦١٨٣ويقولون الكرم، إنما الكرم قلب المؤمن٦١٨٤ارم فداك أبي وأمي٦١٨٥لا ولكن عليك بالمرأة٦١٨٦سم ابنك عبد الرحمن٦١٨٧سموا باسمي، ولا تكتنوا بكنيتي٦١٨٨سموا باسمي، ولا تكتنوا بكنيتي٦١٨٩أسم ابنك عبد الرحمن٦١٩٠أن أباه، جاء إلى النبي صلى الله عليه وسلم فقال " ما اسمك " . قال حزن…٦١٩١قال أتي بالمنذر بن أبي أسيد إلى النبي صلى الله عليه وسلم حين ولد،…٦١٩٢أن زينب، كان اسمها برة، فقيل تزكي نفسها . فسماها رسول الله صلى الله…٦١٩٣أن ابن جريج، أخبرهم قال أخبرني عبد الحميد بن جبير بن شيبة، قال جلست…٦١٩٤قلت لابن أبي أوفى رأيت إبراهيم ابن النبي صلى الله عليه وسلم قال مات…٦١٩٥إن له مرضعا في الجنة٦١٩٦سموا باسمي، ولا تكتنوا بكنيتي، فإنما أنا قاسم أقسم بينكم٦١٩٧سموا باسمي ولا تكتنوا بكنيتي، ومن رآني في المنام فقد رآني، فإن…٦١٩٨قال ولد لي غلام، فأتيت به النبي صلى الله عليه وسلم فسماه إبراهيم،…٦١٩٩حدثنا أبو الوليد، حدثنا زائدة، حدثنا زياد بن علاقة، سمعت المغيرة بن…٦٢٠٠اللهم أنج الوليد بن الوليد، وسلمة بن هشام، وعياش بن أبي ربيعة،…٦٢٠١يا عائش هذا جبريل يقرئك السلام٦٢٠٢يا أنجش، رويدك، سوقك بالقوارير٦٢٠٣يا أبا عمير ما فعل النغير٦٢٠٤اجلس يا أبا تراب٦٢٠٥أخنى الأسماء يوم القيامة عند الله رجل تسمى ملك الأملاك٦٢٠٦أخنع اسم عند الله وقال سفيان غير مرة أخنع الأسماء عند الله رجل تسمى…٦٢٠٧أى سعد ألم تسمع ما قال أبو حباب يريد عبد الله بن أبى قال كذا وكذا٦٢٠٨نعم هو في ضحضاح من نار، لولا أنا لكان في الدرك الأسفل من النار٦٢٠٩ارفق يا أنجشة، ويحك، بالقوارير٦٢١٠رويدك يا أنجشة، سوقك بالقوارير٦٢١١رويدك يا أنجشة، لا تكسر القوارير٦٢١٢ما رأينا من شىء، وإن وجدناه لبحرا٦٢١٣أنه سمع عروة، يقول قالت عائشة سأل أناس رسول الله صلى الله عليه وسلم…٦٢١٤ثم فتر عني الوحى، فبينا أنا أمشي سمعت صوتا من السماء، فرفعت بصري إلى…٦٢١٥فلما كان ثلث الليل الآخر أو بعضه قعد فنظر إلى السماء فقرأ { إن في خلق…٦٢١٦افتح له وبشره بالجنة٦٢١٧ليس منكم من أحد إلا وقد فرغ من مقعده من الجنة والنار٦٢١٨سبحان الله ماذا أنزل من الخزائن، وماذا أنزل من الفتن، من يوقظ صواحب…٦٢١٩على رسلكما، إنما هي صفية بنت حيى٦٢٢٠إنه لا يقتل الصيد، ولا ينكأ العدو، وإنه يفقأ العين، ويكسر السن٦٢٢١هذا حمد الله، وهذا لم يحمد الله٦٢٢٢واتباع الجنازة، وتشميت العاطس، وإجابة الداعي، ورد السلام، ونصر…٦٢٢٣إن الله يحب العطاس، ويكره التثاؤب، فإذا عطس فحمد الله، فحق على كل…٦٢٢٤إذا عطس أحدكم فليقل الحمد لله . وليقل له أخوه أو صاحبه يرحمك الله .…٦٢٢٥إن هذا حمد الله، ولم تحمد الله٦٢٢٦إن الله يحب العطاس ويكره التثاؤب، فإذا عطس أحدكم وحمد الله كان حقا…
