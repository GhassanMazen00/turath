كتاب فضائل أصحاب النبى صلى الله عليه وسلم

١٢٠ حديثًا
٣٦٤٩يأتي على الناس زمان فيغزو فئام من الناس، فيقولون فيكم من صاحب رسول…٣٦٥٠خير أمتي قرني ثم الذين يلونهم ثم الذين يلونهم٣٦٥١خير الناس قرني، ثم الذين يلونهم، ثم الذين يلونهم، ثم يجيء قوم تسبق…٣٦٥٢فرميت ببصري هل أرى من ظل فآوي إليه، فإذا صخرة أتيتها فنظرت بقية ظل…٣٦٥٣ما ظنك يا أبا بكر باثنين الله ثالثهما٣٦٥٤إن الله خير عبدا بين الدنيا وبين ما عنده فاختار ذلك العبد ما عند الله٣٦٥٥ثم عمر بن الخطاب، ثم عثمان بن عفان رضى الله عنهم .٣٦٥٦ولو كنت متخذا من أمتي خليلا لاتخذت، أبا بكر ولكن أخي وصاحبي٣٦٥٧لو كنت متخذا خليلا لاتخذته خليلا، ولكن أخوة الإسلام أفضل٣٦٥٨لو كنت متخذا من هذه الأمة خليلا لاتخذته٣٦٥٩إن لم تجديني فأتي أبا بكر٣٦٦٠يقول رأيت رسول الله صلى الله عليه وسلم وما معه إلا خمسة أعبد وامرأتان…٣٦٦١أما صاحبكم فقد غامر٣٦٦٢فأتيته فقلت أى الناس أحب إليك قال " عائشة " . فقلت من الرجال فقال "…٣٦٦٣بينما راع في غنمه عدا عليه الذئب، فأخذ منها شاة، فطلبه الراعي، فالتفت…٣٦٦٤بينا أنا نائم رأيتني على قليب عليها دلو، فنزعت منها ما شاء الله، ثم…٣٦٦٥من جر ثوبه خيلاء لم ينظر الله إليه يوم القيامة٣٦٦٦من أنفق زوجين من شىء من الأشياء في سبيل الله دعي من أبواب يعني الجنة…٣٦٦٨والذي نفسي بيده لا يذيقك الله الموتتين أبدا . ثم خرج فقال أيها الحالف…٣٦٧٠أن عائشة رضى الله عنها قالت شخص بصر النبي صلى الله عليه وسلم ثم قال "…٣٦٧١حدثنا محمد بن كثير، أخبرنا سفيان، حدثنا جامع بن أبي راشد، حدثنا أبو…٣٦٧٢فأقام رسول الله صلى الله عليه وسلم على التماسه، وأقام الناس معه،…٣٦٧٣لا تسبوا أصحابي، فلو أن أحدكم أنفق مثل أحد ذهبا ما بلغ مد أحدهم ولا…٣٦٧٤ائذن له وبشره بالجنة٣٦٧٥اثبت أحد فإنما عليك نبي وصديق وشهيدان٣٦٧٦بينما أنا على بئر أنزع منها جاءني أبو بكر وعمر، فأخذ أبو بكر الدلو،…٣٦٧٧فدعوا الله لعمر بن الخطاب وقد وضع على سريره، إذا رجل من خلفي قد وضع…٣٦٧٨ما صنع المشركون برسول الله صلى الله عليه وسلم قال رأيت عقبة بن أبي…٣٦٧٩رأيتني دخلت الجنة، فإذا أنا بالرميصاء امرأة أبي طلحة وسمعت خشفة، فقلت…٣٦٨٠بينا أنا نائم رأيتني في الجنة، فإذا امرأة تتوضأ إلى جانب قصر، فقلت…٣٦٨١بينا أنا نائم شربت يعني اللبن حتى أنظر إلى الري يجري في ظفري أو في…٣٦٨٢أريت في المنام أني أنزع بدلو بكرة على قليب، فجاء أبو بكر فنزع ذنوبا…٣٦٨٣عجبت من هؤلاء اللاتي كن عندي فلما سمعن صوتك ابتدرن الحجاب٣٦٨٤حدثنا محمد بن المثنى، حدثنا يحيى، عن إسماعيل، حدثنا قيس، قال قال عبد…٣٦٨٥أنه سمع ابن عباس، يقول وضع عمر على سريره، فتكنفه الناس يدعون ويصلون…٣٦٨٦اثبت أحد فما عليك إلا نبي أو صديق أو شهيدان٣٦٨٧هو ابن محمد أن زيد بن أسلم، حدثه عن أبيه، قال سألني ابن عمر عن بعض،…٣٦٨٨وماذا أعددت لها٣٦٨٩لقد كان فيما قبلكم من الأمم محدثون، فإن يك في أمتي أحد فإنه عمر٣٦٩٠بينما راع في غنمه عدا الذئب فأخذ منها شاة، فطلبها حتى استنقذها،…٣٦٩١بينا أنا نائم رأيت الناس عرضوا على وعليهم قمص، فمنها ما يبلغ الثدى،…٣٦٩٢فقال له ابن عباس وكأنه يجزعه يا أمير المؤمنين، ولئن كان ذاك لقد صحبت…٣٦٩٣افتح له وبشره بالجنة٣٦٩٤زهرة بن معبد أنه سمع جده عبد الله بن هشام، قال كنا مع النبي صلى الله…٣٦٩٥ائذن له وبشره بالجنة٣٦٩٦أن عبيد الله بن عدي بن الخيار، أخبره أن المسور بن مخرمة وعبد الرحمن…٣٦٩٧حدثني محمد بن حاتم بن بزيغ، حدثنا شاذان، حدثنا عبد العزيز بن أبي سلمة…٣٦٩٨إن لك أجر رجل ممن شهد بدرا وسهمه٣٦٩٩اسكن أحد أظنه ضربه برجله فليس عليك إلا نبي وصديق وشهيدان٣٧٠٠قال رأيت عمر بن الخطاب رضى الله عنه قبل أن يصاب بأيام بالمدينة وقف…٣٧٠١لأعطين الراية غدا رجلا يفتح الله على يديه٣٧٠٢لأعطين الراية أو ليأخذن الراية غدا رجلا يحبه الله ورسوله أو قال يحب…٣٧٠٣أن رجلا، جاء إلى سهل بن سعد فقال هذا فلان لأمير المدينة يدعو عليا عند…٣٧٠٤قال جاء رجل إلى ابن عمر، فسأله عن عثمان،، فذكر عن محاسن، عمله، قال…٣٧٠٥أن فاطمة، عليها السلام شكت ما تلقى من أثر الرحا، فأتى النبي صلى الله…٣٧٠٦أما ترضى أن تكون مني بمنزلة هارون من موسى٣٧٠٧فإني أكره الاختلاف حتى يكون للناس جماعة، أو أموت كما مات أصحابي .…٣٧٠٨كانوا يقولون أكثر أبو هريرة . وإني كنت ألزم رسول الله صلى الله عليه…٣٧٠٩أن ابن عمر رضى الله عنهما كان إذا سلم على ابن جعفر قال السلام عليك يا…٣٧١٠كان إذا قحطوا استسقى بالعباس بن عبد المطلب، فقال اللهم إنا كنا نتوسل…٣٧١٢لا نورث، ما تركنا فهو صدقة، إنما يأكل آل محمد من هذا المال يعني مال…٣٧١٣أخبرني عبد الله بن عبد الوهاب، حدثنا خالد، حدثنا شعبة، عن واقد، قال…٣٧١٤فاطمة بضعة مني، فمن أغضبها أغضبني٣٧١٦فسارها بشىء فبكت، ثم دعاها فسارها فضحكت، قالت فسألتها عن ذلك . فقالت…٣٧١٧قال أصاب عثمان بن عفان رعاف شديد سنة الرعاف، حتى حبسه عن الحج وأوصى،…٣٧١٨كنت عند عثمان، أتاه رجل فقال استخلف . قال وقيل ذاك قال نعم، الزبير .…٣٧١٩إن لكل نبي حواريا، وإن حواري الزبير بن العوام٣٧٢٠من يأت بني قريظة فيأتيني بخبرهم٣٧٢١أن أصحاب النبي، صلى الله عليه وسلم قالوا للزبير يوم اليرموك ألا تشد…٣٧٢٢حدثني محمد بن أبي بكر المقدمي، حدثنا معتمر، عن أبيه، عن أبي عثمان،…٣٧٢٤، قال رأيت يد طلحة التي وقى بها النبي صلى الله عليه وسلم قد شلت .٣٧٢٥يقول جمع لي النبي صلى الله عليه وسلم أبويه يوم أحد .٣٧٢٦.٣٧٢٧يقول ما أسلم أحد إلا في اليوم الذي أسلمت فيه، ولقد مكثت سبعة أيام…٣٧٢٨ما له خلط، ثم أصبحت بنو أسد تعزرني على الإسلام، لقد خبت إذا وضل عملي…٣٧٢٩أما بعد أنكحت أبا العاص بن الربيع، فحدثني وصدقني، وإن فاطمة بضعة مني،…٣٧٣٠إن تطعنوا في إمارته فقد كنتم تطعنون في إمارة أبيه من قبل، وايم الله،…٣٧٣١وأسامة بن زيد وزيد بن حارثة مضطجعان، فقال إن هذه الأقدام بعضها من بعض…٣٧٣٢رضى الله عنها أن قريشا، أهمهم شأن المخزومية، فقالوا من يجترئ عليه إلا…٣٧٣٣إن بني إسرائيل كان إذا سرق فيهم الشريف تركوه، وإذا سرق الضعيف قطعوه،…٣٧٣٤قال نظر ابن عمر يوما وهو في المسجد إلى رجل يسحب ثيابه في ناحية من…٣٧٣٥اللهم أحبهما فإني أحبهما٣٧٣٦لأسامة بن زيد . أن الحجاج بن أيمن ابن أم أيمن،، وكان، أيمن ابن أم…٣٧٣٧مولى أسامة بن زيد أنه بينما هو مع عبد الله بن عمر إذ دخل الحجاج بن…٣٧٣٩نعم الرجل عبد الله، لو كان يصلي بالليل٣٧٤٠إن عبد الله رجل صالح٣٧٤٢، قال قدمت الشأم فصليت ركعتين، ثم قلت اللهم يسر لي جليسا صالحا، فأتيت…٣٧٤٣قال ذهب علقمة إلى الشأم، فلما دخل المسجد قال اللهم يسر لي جليسا صالحا…٣٧٤٤إن لكل أمة أمينا، وإن أميننا أيتها الأمة أبو عبيدة بن الجراح٣٧٤٥لأبعثن يعني عليكم يعني أمينا حق أمين٣٧٤٦ابني هذا سيد، ولعل الله أن يصلح به بين فئتين من المسلمين٣٧٤٧اللهم إني أحبهما فأحبهما٣٧٤٨فجعل ينكت، وقال في حسنه شيئا . فقال أنس كان أشبههم برسول الله صلى…٣٧٤٩اللهم إني أحبه فأحبه٣٧٥٠قال رأيت أبا بكر رضى الله عنه وحمل الحسن وهو يقول بأبي شبيه بالنبي،…٣٧٥١وصدقة، قالا أخبرنا محمد بن جعفر، عن شعبة، عن واقد بن محمد، عن أبيه،…٣٧٥٢قال لم يكن أحد أشبه بالنبي صلى الله عليه وسلم من الحسن بن علي .٣٧٥٣هما ريحانتاى من الدنيا٣٧٥٤حدثنا أبو نعيم، حدثنا عبد العزيز بن أبي سلمة، عن محمد بن المنكدر،…٣٧٥٥أن بلالا، قال لأبي بكر إن كنت إنما اشتريتني لنفسك فأمسكني، وإن كنت…٣٧٥٦اللهم علمه الحكمة٣٧٥٦.٢وقال، " علمه الكتاب " . حدثنا موسى، حدثنا وهيب، عن خالد، مثله .…٣٧٥٧أخذ الراية زيد فأصيب، ثم أخذ جعفر فأصيب، ثم أخذ ابن رواحة فأصيب…٣٧٥٨استقرئوا القرآن من أربعة من عبد الله بن مسعود، فبدأ به، وسالم مولى…٣٧٦٠إن من أحبكم إلى أحسنكم أخلاقا٣٧٦١دخلت الشأم فصليت ركعتين، فقلت اللهم يسر لي جليسا . فرأيت شيخا مقبلا،…٣٧٦٢، قال سألنا حذيفة عن رجل، قريب السمت والهدى من النبي صلى الله عليه…٣٧٦٣لما نرى من دخوله ودخول أمه على النبي صلى الله عليه وسلم .٣٧٦٤قال أوتر معاوية بعد العشاء بركعة وعنده مولى لابن عباس، فأتى ابن عباس…٣٧٦٥قيل لابن عباس هل لك في أمير المؤمنين معاوية، فإنه ما أوتر إلا بواحدة…٣٧٦٦ولقد نهى عنهما، يعني الركعتين بعد العصر .٣٧٦٧فاطمة بضعة مني، فمن أغضبها أغضبني٣٧٦٨يا عائش، هذا جبريل يقرئك السلام٣٧٦٩كمل من الرجال كثير، ولم يكمل من النساء إلا مريم بنت عمران، وآسية…٣٧٧٠فضل عائشة على النساء كفضل الثريد على الطعام٣٧٧١أن عائشة، اشتكت، فجاء ابن عباس فقال يا أم المؤمنين، تقدمين على فرط…٣٧٧٢قال لما بعث علي عمارا والحسن إلى الكوفة ليستنفرهم خطب عمار فقال إني…٣٧٧٣فأرسل رسول الله صلى الله عليه وسلم ناسا من أصحابه في طلبها، فأدركتهم…٣٧٧٤أين أنا غدا أين أنا غدا٣٧٧٥يا أم سلمة لا تؤذيني في عائشة، فإنه والله ما نزل على الوحى وأنا في…
