٢٦٦ حديثًا•صفحة ١ من ٢
٦٠٣قال ذكروا النار والناقوس، فذكروا اليهود والنصارى، فأمر بلال أن يشفع…٦٠٤يا بلال قم فناد بالصلاة٦٠٥قال أمر بلال أن يشفع، الأذان وأن يوتر الإقامة إلا الإقامة .٦٠٦قال لما كثر الناس قال ذكروا أن يعلموا وقت الصلاة بشىء يعرفونه، فذكروا…٦٠٧قال أمر بلال أن يشفع، الأذان، وأن يوتر الإقامة . قال إسماعيل فذكرت…٦٠٨إذا نودي للصلاة أدبر الشيطان وله ضراط حتى لا يسمع التأذين، فإذا قضى…٦٠٩إني أراك تحب الغنم والبادية، فإذا كنت في غنمك أو باديتك فأذنت بالصلاة…٦١٠الله أكبر، الله أكبر، خربت خيبر، إنا إذا نزلنا بساحة قوم فساء صباح…٦١١إذا سمعتم النداء فقولوا مثل ما يقول المؤذن٦١٢وأشهد أن محمدا رسول الله٦١٣حدثنا إسحاق بن راهويه، قال حدثنا وهب بن جرير، قال حدثنا هشام، عن…٦١٤من قال حين يسمع النداء اللهم رب هذه الدعوة التامة والصلاة القائمة آت…٦١٥لو يعلم الناس ما في النداء والصف الأول، ثم لم يجدوا إلا أن يستهموا…٦١٦وعبد الحميد، صاحب الزيادي وعاصم الأحول عن عبد الله بن الحارث قال…٦١٧إن بلالا يؤذن بليل، فكلوا واشربوا حتى ينادي ابن أم مكتوم٦١٨قال أخبرتني حفصة، أن رسول الله صلى الله عليه وسلم كان إذا اعتكف…٦١٩كان النبي صلى الله عليه وسلم يصلي ركعتين خفيفتين بين النداء والإقامة…٦٢٠إن بلالا ينادي بليل، فكلوا واشربوا حتى ينادي ابن أم مكتوم٦٢١لا يمنعن أحدكم أو أحدا منكم أذان بلال من سحوره، فإنه يؤذن أو ينادي…٦٢٣إن بلالا يؤذن بليل، فكلوا واشربوا حتى يؤذن ابن أم مكتوم٦٢٤بين كل أذانين صلاة ثلاثا لمن شاء٦٢٥قال كان المؤذن إذا أذن قام ناس من أصحاب النبي صلى الله عليه وسلم…٦٢٦أن عائشة، قالت كان رسول الله صلى الله عليه وسلم إذا سكت المؤذن…٦٢٧بين كل أذانين صلاة بين كل أذانين صلاة ثم قال في الثالثة لمن شاء٦٢٨ارجعوا فكونوا فيهم وعلموهم وصلوا، فإذا حضرت الصلاة فليؤذن لكم أحدكم…٦٢٩حدثنا مسلم بن إبراهيم، قال حدثنا شعبة، عن المهاجر أبي الحسن، عن زيد…٦٣٠إذا أنتما خرجتما فأذنا ثم أقيما ثم ليؤمكما أكبركما٦٣١ارجعوا إلى أهليكم فأقيموا فيهم وعلموهم ومروهم وذكر أشياء أحفظها أو لا…٦٣٢قال أذن ابن عمر في ليلة باردة بضجنان ثم قال صلوا في رحالكم، فأخبرنا…٦٣٣قال رأيت رسول الله صلى الله عليه وسلم بالأبطح فجاءه بلال فآذنه…٦٣٤حدثنا محمد بن يوسف، قال حدثنا سفيان، عن عون بن أبي جحيفة، عن أبيه،…٦٣٥إذا أتيتم الصلاة فعليكم بالسكينة، فما أدركتم فصلوا وما فاتكم فأتموا "…٦٣٦إذا سمعتم الإقامة فامشوا إلى الصلاة، وعليكم بالسكينة والوقار ولا…٦٣٧إذا أقيمت الصلاة فلا تقوموا حتى تروني٦٣٨إذا أقيمت الصلاة فلا تقوموا حتى تروني وعليكم بالسكينة٦٣٩أن رسول الله صلى الله عليه وسلم خرج وقد أقيمت الصلاة وعدلت الصفوف،…٦٤٠قال أقيمت الصلاة فسوى الناس صفوفهم، فخرج رسول الله صلى الله عليه وسلم…٦٤١والله ما صليتها٦٤٢قال أقيمت الصلاة والنبي صلى الله عليه وسلم يناجي رجلا في جانب المسجد،…٦٤٣قال أقيمت الصلاة فعرض للنبي صلى الله عليه وسلم رجل فحبسه بعد ما أقيمت…٦٤٤والذي نفسي بيده لقد هممت أن آمر بحطب فيحطب، ثم آمر بالصلاة فيؤذن لها،…٦٤٥صلاة الجماعة تفضل صلاة الفذ بسبع وعشرين درجة٦٤٦صلاة الجماعة تفضل صلاة الفذ بخمس وعشرين درجة٦٤٧صلاة الرجل في الجماعة تضعف على صلاته في بيته وفي سوقه خمسا وعشرين…٦٤٨تفضل صلاة الجميع صلاة أحدكم وحده بخمس وعشرين جزءا، وتجتمع ملائكة…٦٤٩قال تفضلها بسبع وعشرين درجة .٦٥٠تقول دخل على أبو الدرداء وهو مغضب فقلت ما أغضبك فقال والله ما أعرف من…٦٥١أعظم الناس أجرا في الصلاة أبعدهم فأبعدهم ممشى، والذي ينتظر الصلاة حتى…٦٥٤بينما رجل يمشي بطريق وجد غصن شوك على الطريق فأخره، فشكر الله له، فغفر…٦٥٦يا بني سلمة ألا تحتسبون آثاركم٦٥٧ليس صلاة أثقل على المنافقين من الفجر والعشاء، ولو يعلمون ما فيهما…٦٥٨إذا حضرت الصلاة فأذنا وأقيما، ثم ليؤمكما أكبركما٦٥٩الملائكة تصلي على أحدكم ما دام في مصلاه ما لم يحدث اللهم اغفر له،…٦٦٠سبعة يظلهم الله في ظله يوم لا ظل إلا ظله الإمام العادل، وشاب نشأ في…٦٦١صلى الناس ورقدوا ولم تزالوا في صلاة منذ انتظرتموها٦٦٢من غدا إلى المسجد وراح أعد الله له نزله من الجنة كلما غدا أو راح٦٦٣الصبح أربعا، الصبح أربعا٦٦٤مروا أبا بكر فليصل بالناس٦٦٥قال قالت عائشة لما ثقل النبي صلى الله عليه وسلم واشتد وجعه استأذن…٦٦٦أن ابن عمر، أذن بالصلاة في ليلة ذات برد وريح ثم قال ألا صلوا في…٦٦٧أين تحب أن أصلي٦٦٨فأمر المؤذن لما بلغ حى على الصلاة . قال قل الصلاة في الرحال، فنظر…٦٦٩قال سألت أبا سعيد الخدري فقال جاءت سحابة فمطرت حتى سال السقف، وكان من…٦٧٠يقول قال رجل من الأنصار إني لا أستطيع الصلاة معك . وكان رجلا ضخما،…٦٧١إذا وضع العشاء وأقيمت الصلاة فابدءوا بالعشاء٦٧٢إذا قدم العشاء فابدءوا به قبل أن تصلوا صلاة المغرب، ولا تعجلوا عن…٦٧٣إذا وضع عشاء أحدكم وأقيمت الصلاة فابدءوا بالعشاء، ولا يعجل حتى يفرغ…٦٧٤إذا كان أحدكم على الطعام فلا يعجل حتى يقضي حاجته منه، وإن أقيمت الصلاة٦٧٥أن أباه، قال رأيت رسول الله صلى الله عليه وسلم يأكل ذراعا يحتز منها،…٦٧٦قال سألت عائشة ما كان النبي صلى الله عليه وسلم يصنع في بيته قالت كان…٦٧٧وما أريد الصلاة، أصلي كيف رأيت النبي صلى الله عليه وسلم يصلي . فقلت…٦٧٨مروا أبا بكر فليصل بالناس٦٧٩مروا أبا بكر يصلي بالناس٦٨٠حتى إذا كان يوم الاثنين وهم صفوف في الصلاة، فكشف النبي صلى الله عليه…٦٨١قال لم يخرج النبي صلى الله عليه وسلم ثلاثا، فأقيمت الصلاة، فذهب أبو…٦٨٢مروا أبا بكر فليصل بالناس٦٨٣قالت أمر رسول الله صلى الله عليه وسلم أبا بكر أن يصلي بالناس في مرضه،…٦٨٤يا أبا بكر ما منعك أن تثبت إذ أمرتك٦٨٥لو رجعتم إلى بلادكم فعلمتموهم، مروهم فليصلوا صلاة كذا في حين كذا،…٦٨٦أين تحب أن أصلي من بيتك٦٨٧هم ينتظرونك يا رسول الله . قال " ضعوا لي ماء في المخضب " . قالت فقعد…٦٨٨إنما جعل الإمام ليؤتم به، فإذا ركع فاركعوا، وإذا رفع فارفعوا، وإذا…٦٨٩إنما جعل الإمام ليؤتم به، فإذا صلى قائما فصلوا قياما، فإذا ركع…٦٩٠ثم نقع سجودا بعده .٦٩٠.٢نحوه بهذا .٦٩١أما يخشى أحدكم أو لا يخشى أحدكم إذا رفع رأسه قبل الإمام أن يجعل الله…٦٩٢قال لما قدم المهاجرون الأولون العصبة موضع بقباء قبل مقدم رسول الله…٦٩٣اسمعوا وأطيعوا، وإن استعمل حبشي كأن رأسه زبيبة٦٩٤يصلون لكم، فإن أصابوا فلكم، وإن أخطئوا فلكم وعليهم٦٩٥أنه دخل على عثمان بن عفان رضى الله عنه وهو محصور فقال إنك إمام عامة،…٦٩٦اسمع وأطع، ولو لحبشي كأن رأسه زبيبة٦٩٧ثم جاء فصلى أربع ركعات ثم نام، ثم قام فجئت فقمت عن يساره، فجعلني عن…٦٩٨فتوضأ ثم قام يصلي، فقمت على يساره، فأخذني فجعلني عن يمينه، فصلى ثلاث…٦٩٩قال بت عند خالتي فقام النبي صلى الله عليه وسلم يصلي من الليل، فقمت…٧٠٠أن معاذ بن جبل، كان يصلي مع النبي صلى الله عليه وسلم ثم يرجع فيؤم…٧٠١فتان فتان فتان٧٠٢إن منكم منفرين، فأيكم ما صلى بالناس فليتجوز، فإن فيهم الضعيف والكبير…٧٠٣إذا صلى أحدكم للناس فليخفف، فإن منهم الضعيف والسقيم والكبير، وإذا صلى…٧٠٤يا أيها الناس إن منكم منفرين، فمن أم الناس فليتجوز، فإن خلفه الضعيف…٧٠٥يا معاذ أفتان أنت أو فاتن ثلاث مرار فلولا صليت بسبح اسم ربك، والشمس…٧٠٦قال كان النبي صلى الله عليه وسلم يوجز الصلاة ويكملها .٧٠٧إني لأقوم في الصلاة أريد أن أطول فيها، فأسمع بكاء الصبي، فأتجوز في…٧٠٨يقول ما صليت وراء إمام قط أخف صلاة ولا أتم من النبي صلى الله عليه…٧٠٩إني لأدخل في الصلاة وأنا أريد إطالتها، فأسمع بكاء الصبي، فأتجوز في…٧١٠إني لأدخل في الصلاة فأريد إطالتها، فأسمع بكاء الصبي، فأتجوز مما أعلم…٧١١وأبو النعمان، قالا حدثنا حماد بن زيد، عن أيوب، عن عمرو بن دينار، عن…٧١٢مروا أبا بكر فليصل٧١٣مروا أبا بكر أن يصلي بالناس٧١٤أن رسول الله صلى الله عليه وسلم انصرف من اثنتين، فقال له ذو اليدين…٧١٥قال صلى النبي صلى الله عليه وسلم الظهر ركعتين، فقيل صليت ركعتين .…٧١٦مروا أبا بكر يصلي بالناس٧١٧لتسون صفوفكم أو ليخالفن الله بين وجوهكم٧١٨أقيموا الصفوف فإني أراكم خلف ظهري٧١٩أقيموا صفوفكم وتراصوا، فإني أراكم من وراء ظهري٧٢١الشهداء الغرق والمطعون والمبطون والهدم٧٢٢إنما جعل الإمام ليؤتم به فلا تختلفوا عليه، فإذا ركع فاركعوا، وإذا قال…٧٢٣سووا صفوفكم فإن تسوية الصفوف من إقامة الصلاة٧٢٤حدثنا معاذ بن أسد، قال أخبرنا الفضل بن موسى، قال أخبرنا سعيد بن عبيد…٧٢٥أقيموا صفوفكم فإني أراكم من وراء ظهري٧٢٦فأخذ رسول الله صلى الله عليه وسلم برأسي من ورائي، فجعلني عن يمينه،…٧٢٧حدثنا عبد الله بن محمد، قال حدثنا سفيان، عن إسحاق، عن أنس بن مالك،…٧٢٨وقال بيده من ورائي .٧٢٩إني خشيت أن تكتب عليكم صلاة الليل٧٣٠ويحتجره بالليل، فثاب إليه ناس، فصلوا وراءه .٧٣١قد عرفت الذي رأيت من صنيعكم، فصلوا أيها الناس في بيوتكم، فإن أفضل…٧٣٢إنما جعل الإمام ليؤتم به، فإذا صلى قائما فصلوا قياما، وإذا ركع…٧٣٣إنما الإمام أو إنما جعل الإمام ليؤتم به، فإذا كبر فكبروا، وإذا ركع…٧٣٤إنما جعل الإمام ليؤتم به، فإذا كبر فكبروا، وإذا ركع فاركعوا، وإذا قال…٧٣٥سمع الله لمن حمده، ربنا ولك الحمد٧٣٦سمع الله لمن حمده٧٣٧أنه رأى مالك بن الحويرث إذا صلى كبر ورفع يديه، وإذا أراد أن يركع رفع…٧٣٨ربنا ولك الحمد٧٣٩أن ابن عمر، كان إذا دخل في الصلاة كبر ورفع يديه، وإذا ركع رفع يديه،…٧٤٠قال كان الناس يؤمرون أن يضع الرجل اليد اليمنى على ذراعه اليسرى في…٧٤١هل ترون قبلتي ها هنا والله ما يخفى على ركوعكم ولا خشوعكم، وإني لأراكم…٧٤٢أقيموا الركوع والسجود، فوالله إني لأراكم من بعدي وربما قال من بعد…٧٤٣أن النبي صلى الله عليه وسلم وأبا بكر وعمر رضى الله عنهما كانوا…٧٤٤أقول اللهم باعد بيني وبين خطاياى كما باعدت بين المشرق والمغرب، اللهم…٧٤٥قد دنت مني الجنة حتى لو اجترأت عليها لجئتكم بقطاف من قطافها، ودنت مني…٧٤٦حدثنا موسى، قال حدثنا عبد الواحد، قال حدثنا الأعمش، عن عمارة بن عمير،…٧٤٧وكان، غير كذوب أنهم كانوا إذا صلوا مع النبي صلى الله عليه وسلم فرفع…٧٤٨إني أريت الجنة، فتناولت منها عنقودا، ولو أخذته لأكلتم منه ما بقيت…٧٤٩لقد رأيت الآن منذ صليت لكم الصلاة الجنة والنار ممثلتين في قبلة هذا…٧٥٠ما بال أقوام يرفعون أبصارهم إلى السماء في صلاتهم٧٥١هو اختلاس يختلسه الشيطان من صلاة العبد٧٥٢شغلتني أعلام هذه، اذهبوا بها إلى أبي جهم وأتوني بأنبجانية٧٥٣إن أحدكم إذا كان في الصلاة فإن الله قبل وجهه، فلا يتنخمن أحد قبل وجهه…٧٥٤قال بينما المسلمون في صلاة الفجر لم يفجأهم إلا رسول الله صلى الله…٧٥٥قال شكا أهل الكوفة سعدا إلى عمر رضى الله عنه فعزله واستعمل عليهم…٧٥٦لا صلاة لمن لم يقرأ بفاتحة الكتاب
