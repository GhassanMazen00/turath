كتاب الأيمان والنذور

٨٤ حديثًا
٦٦٢١أن أبا بكر رضى الله عنه لم يكن يحنث في يمين قط، حتى أنزل الله كفارة…٦٦٢٢يا عبد الرحمن بن سمرة لا تسأل الإمارة، فإنك إن أوتيتها عن مسألة وكلت…٦٦٢٣والله لا أحملكم، وما عندي ما أحملكم عليه٦٦٢٤نحن الآخرون السابقون يوم القيامة٦٦٢٥والله لأن يلج أحدكم بيمينه في أهله آثم له عند الله من أن يعطي كفارته…٦٦٢٦من استلج في أهله بيمين فهو أعظم إثما، ليبر٦٦٢٧إن كنتم تطعنون في إمرته فقد كنتم تطعنون، في إمرة أبيه من قبل، وايم…٦٦٢٨لا ومقلب القلوب٦٦٢٩إذا هلك قيصر فلا قيصر بعده، وإذا هلك كسرى فلا كسرى بعده، والذي نفسي…٦٦٣٠إذا هلك كسرى فلا كسرى بعده، وإذا هلك قيصر فلا قيصر بعده، والذي نفس…٦٦٣١يا أمة محمد والله لو تعلمون ما أعلم لبكيتم كثيرا، ولضحكتم قليلا٦٦٣٢لا والذي نفسي بيده حتى أكون أحب إليك من نفسك٦٦٣٤وزيد بن خالد، أنهما أخبراه أن رجلين اختصما إلى رسول الله صلى الله…٦٦٣٥أرأيتم إن كان أسلم وغفار ومزينة وجهينة خيرا من تميم وعامر بن صعصعة…٦٦٣٦أفلا قعدت في بيت أبيك وأمك فنظرت أيهدى لك أم لا٦٦٣٧والذي نفس محمد بيده لو تعلمون ما أعلم لبكيتم كثيرا، ولضحكتم قليلا٦٦٣٨هم الأخسرون ورب الكعبة، هم الأخسرون ورب الكعبة٦٦٣٩قال سليمان لأطوفن الليلة على تسعين امرأة، كلهن تأتي بفارس يجاهد في…٦٦٤٠قال أهدي إلى النبي صلى الله عليه وسلم سرقة من حرير، فجعل الناس…٦٦٤١وأيضا والذي نفس محمد بيده٦٦٤٢أترضون أن تكونوا ربع أهل الجنة٦٦٤٣والذي نفسي بيده إنها لتعدل ثلث القرآن٦٦٤٤أتموا الركوع والسجود، فوالذي نفسي بيده إني لأراكم من بعد ظهري إذا ما…٦٦٤٥والذي نفسي بيده إنكم لأحب الناس إلى٦٦٤٦ألا إن الله ينهاكم أن تحلفوا بآبائكم، من كان حالفا فليحلف بالله، أو…٦٦٤٧إن الله ينهاكم أن تحلفوا بآبائكم٦٦٤٨لا تحلفوا بآبائكم٦٦٤٩والله لا أحملكم، وما عندي ما أحملكم٦٦٥٠من حلف فقال في حلفه باللات والعزى . فليقل لا إله إلا الله . ومن قال…٦٦٥١إني كنت ألبس هذا الخاتم وأجعل فصه من داخل٦٦٥٢من حلف بغير ملة الإسلام فهو كما قال قال ومن قتل نفسه بشىء عذب به في…٦٦٥٣إن ثلاثة في بني إسرائيل أراد الله أن يبتليهم، فبعث ملكا فأتى الأبرص…٦٦٥٤حدثنا قبيصة، حدثنا سفيان، عن أشعث، عن معاوية بن سويد بن مقرن، عن…٦٦٥٥إن لله ما أخذ وما أعطى وكل شىء عنده مسمى فلتصبر وتحتسب٦٦٥٦لا يموت لأحد من المسلمين ثلاثة من الولد، تمسه النار، إلا تحلة القسم٦٦٥٧ألا أدلكم على أهل الجنة، كل ضعيف متضعف، لو أقسم على الله لأبره، وأهل…٦٦٥٨قرني، ثم الذين يلونهم، ثم الذين يلونهم، ثم يجيء قوم تسبق شهادة أحدهم…٦٦٦٠من حلف على يمين كاذبة، ليقتطع بها مال رجل مسلم أو قال أخيه لقي الله…٦٦٦١لا تزال جهنم تقول هل من مزيد حتى يضع رب العزة فيها قدمه فتقول قط قط…٦٦٦٢وسعيد بن المسيب، وعلقمة بن وقاص، وعبيد الله بن عبد الله، عن حديث،…٦٦٦٣والله بلى والله .٦٦٦٤إن الله تجاوز لأمتي عما وسوست أو حدثت به أنفسها، ما لم تعمل به أو تكلم٦٦٦٥أن عبد الله بن عمرو بن العاص، حدثه أن النبي صلى الله عليه وسلم بينما…٦٦٦٦حدثنا أحمد بن يونس، حدثنا أبو بكر، عن عبد العزيز بن رفيع، عن عطاء، عن…٦٦٦٧ارجع فصل، فإنك لم تصل٦٦٦٨فصرخ إبليس أى عباد الله أخراكم، فرجعت أولاهم فاجتلدت هي وأخراهم، فنظر…٦٦٦٩من أكل ناسيا وهو صائم فليتم صومه، فإنما أطعمه الله وسقاه٦٦٧٠فمضى في صلاته، فلما قضى صلاته انتظر الناس تسليمه، وسجد قبل أن يسلم،…٦٦٧١سمع عبد العزيز بن عبد الصمد، حدثنا منصور، عن إبراهيم، عن علقمة، عن…٦٦٧٢كانت الأولى من موسى نسيانا٦٦٧٣قال قال البراء بن عازب وكان عندهم ضيف لهم فأمر أهله أن يذبحوا قبل أن…٦٦٧٤من ذبح فليبدل مكانها، ومن لم يكن ذبح فليذبح باسم الله٦٦٧٥الكبائر الإشراك بالله، وعقوق الوالدين، وقتل النفس، واليمين الغموس٦٦٧٧من حلف على يمين صبر، يقتطع بها مال امرئ مسلم، لقي الله وهو عليه غضبان٦٦٧٨والله لا أحملكم على شىء٦٦٧٩وسعيد بن المسيب، وعلقمة بن وقاص، وعبيد الله بن عبد الله بن عتبة، عن…٦٦٨٠والله إن شاء الله لا أحلف على يمين فأرى غيرها خيرا منها، إلا أتيت…٦٦٨١قل لا إله إلا الله كلمة . أحاج لك بها عند الله٦٦٨٢كلمتان خفيفتان على اللسان، ثقيلتان في الميزان، حبيبتان إلى الرحمن…٦٦٨٣من مات يجعل لله ندا أدخل النار٦٦٨٤قال آلى رسول الله صلى الله عليه وسلم من نسائه، وكانت انفكت رجله،…٦٦٨٥سمع عبد العزيز بن أبي حازم، أخبرني أبي، عن سهل بن سعد، أن أبا أسيد،…٦٦٨٦حدثنا محمد بن مقاتل، أخبرنا عبد الله، أخبرنا إسماعيل بن أبي خالد، عن…٦٦٨٧أنه قال لعائشة بهذا .٦٦٨٨أرسلك أبو طلحة٦٦٨٩إنما الأعمال بالنية، وإنما لامرئ ما نوى، فمن كانت هجرته إلى الله…٦٦٩٠أمسك عليك بعض مالك فهو خير لك٦٦٩١لا بل شربت عسلا عند زينب بنت جحش، ولن أعود له٦٦٩٢إن النذر لا يقدم شيئا، ولا يؤخر، وإنما يستخرج بالنذر من البخيل٦٦٩٣إنه لا يرد شيئا، ولكنه يستخرج به من البخيل٦٦٩٤لا يأتي ابن آدم النذر بشىء لم يكن قدر له، ولكن يلقيه النذر إلى القدر…٦٦٩٥خيركم قرني، ثم الذين يلونهم، ثم الذين يلونهم قال عمران لا أدري ذكر…٦٦٩٦من نذر أن يطيع الله فليطعه، ومن نذر أن يعصيه فلا يعصه٦٦٩٧أن عمر، قال يا رسول الله إني نذرت في الجاهلية أن أعتكف ليلة في المسجد…٦٦٩٨أن عبد الله بن عباس، أخبره أن سعد بن عبادة الأنصاري استفتى النبي صلى…٦٦٩٩لو كان عليها دين أكنت قاضيه٦٧٠٠من نذر أن يطيع الله فليطعه، ومن نذر أن يعصيه فلا يعصه٦٧٠١إن الله لغني عن تعذيب هذا نفسه٦٧٠٢أن النبي صلى الله عليه وسلم رأى رجلا يطوف بالكعبة بزمام أو غيره،…٦٧٠٣أن ابن جريج، أخبرهم قال أخبرني سليمان الأحول، أن طاوسا، أخبره عن ابن…٦٧٠٤مره فليتكلم وليستظل وليقعد وليتم صومه٦٧٠٥نذر أن لا، يأتي عليه يوم إلا صام، فوافق يوم أضحى أو فطر . فقال لقد…٦٧٠٦قال كنت مع ابن عمر فسأله رجل فقال نذرت أن أصوم كل يوم ثلاثاء أو…٦٧٠٧كلا والذي نفسي بيده إن الشملة التي أخذها يوم خيبر من المغانم، لم…
