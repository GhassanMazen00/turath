كتاب الفرائض

٤٧ حديثًا
٦٧٢٣سمع جابر بن عبد الله رضى الله عنهما يقول مرضت فعادني رسول الله صلى…٦٧٢٤إياكم والظن فإن الظن أكذب الحديث، ولا تحسسوا، ولا تجسسوا، ولا…٦٧٢٦لا نورث، ما تركنا صدقة، إنما يأكل آل محمد من هذا المال٦٧٢٧لا نورث ما تركنا صدقة٦٧٢٨لا نورث ما تركنا صدقة٦٧٢٩لا يقتسم ورثتي دينارا، ما تركت بعد نفقة نسائي ومؤنة عاملي فهو صدقة٦٧٣٠لا نورث ما تركنا صدقة٦٧٣١أنا أولى، بالمؤمنين من أنفسهم، فمن مات وعليه دين، ولم يترك وفاء،…٦٧٣٢ألحقوا الفرائض بأهلها، فما بقي فهو لأولى رجل ذكر٦٧٣٣قال مرضت بمكة مرضا، فأشفيت منه على الموت، فأتاني النبي صلى الله عليه…٦٧٣٤توفي وترك ابنته وأخته، فأعطى الابنة النصف والأخت النصف .٦٧٣٥ألحقوا الفرائض بأهلها فما بقي فهو لأولى رجل ذكر٦٧٣٦للابنة النصف، ولابنة ابن السدس تكملة الثلثين، وما بقي فللأخت٦٧٣٧ألحقوا الفرائض بأهلها فما بقي فلأولى رجل ذكر٦٧٣٨لو كنت متخذا من هذه الأمة خليلا لاتخذته، ولكن خلة الإسلام أفضل٦٧٣٩وكانت الوصية للوالدين، فنسخ الله من ذلك ما أحب، فجعل للذكر مثل حظ…٦٧٤٠بأن ميراثها لبنيها وزوجها، وأن العقل على عصبتها٦٧٤١حدثنا بشر بن خالد، حدثنا محمد بن جعفر، عن شعبة، عن سليمان، عن…٦٧٤٢قال قال عبد الله لأقضين فيها بقضاء النبي صلى الله عليه وسلم للابنة…٦٧٤٣فدعا بوضوء فتوضأ، ثم نضح على من وضوئه فأفقت فقلت يا رسول الله إنما لي…٦٧٤٤حدثنا عبيد الله بن موسى، عن إسرائيل، عن أبي إسحاق، عن البراء رضى الله…٦٧٤٥أنا أولى بالمؤمنين من أنفسهم، فمن مات وترك مالا فماله لموالي العصبة،…٦٧٤٦ألحقوا الفرائض بأهلها، فما تركت الفرائض فلأولى رجل ذكر٦٧٤٧قال قلت لأبي أسامة حدثكم إدريس، حدثنا طلحة، عن سعيد بن جبير، عن ابن…٦٧٤٨وألحق الولد بالمرأة .٦٧٤٩هو لك يا عبد بن زمعة، الولد للفراش وللعاهر الحجر٦٧٥٠الولد لصاحب الفراش٦٧٥١اشتريها، فإن الولاء لمن أعتق٦٧٥٢إنما الولاء لمن أعتق٦٧٥٣قال إن أهل الإسلام لا يسيبون، وإن أهل الجاهلية كانوا يسيبون .٦٧٥٤أعتقيها فإنما الولاء لمن أعتق٦٧٥٥غير هذه الصحيفة . قال فأخرجها فإذا فيها أشياء من الجراحات وأسنان…٦٧٥٦حدثنا أبو نعيم، حدثنا سفيان، عن عبد الله بن دينار، عن ابن عمر رضى…٦٧٥٧لا يمنعك ذلك، فإنما الولاء لمن أعتق٦٧٥٨أعتقيها فإن الولاء لمن أعطى الورق٦٧٥٩اشتريها، فإنما الولاء لمن أعتق٦٧٦٠الولاء لمن أعطى الورق، وولي النعمة٦٧٦١مولى القوم من أنفسهم٦٧٦٢ابن أخت القوم منهم٦٧٦٣من ترك مالا فلورثته، ومن ترك كلا فإلينا٦٧٦٤لا يرث المسلم الكافر، ولا الكافر المسلم٦٧٦٥هو لك يا عبد، الولد للفراش وللعاهر الحجر، واحتجبي منه يا سودة بنت زمعة٦٧٦٧من ادعى إلى غير أبيه، وهو يعلم أنه غير أبيه، فالجنة عليه حرام٦٧٦٨لا ترغبوا عن آبائكم، فمن رغب عن أبيه فهو كفر٦٧٦٩كانت امرأتان معهما ابناهما، جاء الذئب فذهب بابن إحداهما فقالت…٦٧٧٠ألم ترى أن مجززا نظر آنفا إلى زيد بن حارثة وأسامة بن زيد، فقال إن هذه…٦٧٧١يا عائشة ألم ترى أن مجززا المدلجي دخل فرأى أسامة وزيدا وعليهما قطيفة،…
