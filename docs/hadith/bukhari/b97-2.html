١٨٨ حديثًا•صفحة ٢ من ٢
٧٥٢٦حدثنا عبيد بن إسماعيل، حدثنا أبو أسامة، عن هشام، عن أبيه، عن عائشة…٧٥٢٧ليس منا من لم يتغن بالقرآن٧٥٢٨لا تحاسد إلا في اثنتين رجل آتاه الله القرآن فهو يتلوه آناء الليل…٧٥٢٩لا حسد إلا في اثنتين رجل آتاه الله القرآن فهو يتلوه آناء الليل وآناء…٧٥٣٠أنه من قتل منا صار إلى الجنة٧٥٣١فلا تصدقه، إن الله تعالى يقول { يا أيها الرسول بلغ ما أنزل إليك من…٧٥٣٢أن تدعو لله ندا، وهو خلقك٧٥٣٣إنما بقاؤكم فيمن سلف من الأمم كما بين صلاة العصر إلى غروب الشمس، أوتي…٧٥٣٤الصلاة لوقتها، وبر الوالدين، ثم الجهاد في سبيل الله٧٥٣٥إني أعطي الرجل وأدع الرجل، والذي أدع أحب إلى من الذي أعطي، أعطي…٧٥٣٦إذا تقرب العبد إلى شبرا تقربت إليه ذراعا، وإذا تقرب مني ذراعا تقربت…٧٥٣٧إذا تقرب العبد مني شبرا تقربت منه ذراعا وإذا تقرب مني ذراعا تقربت منه…٧٥٣٨لكل عمل كفارة، والصوم لي وأنا أجزي به، ولخلوف فم الصائم أطيب عند الله…٧٥٣٩لا ينبغي لعبد أن يقول إنه خير من يونس بن متى٧٥٤٠لولا أن يجتمع الناس عليكم لرجعت كما رجع ابن مغفل٧٥٤١بسم الله الرحمن الرحيم من محمد عبد الله ورسوله إلى هرقل، و { يا أهل…٧٥٤٢لا تصدقوا أهل الكتاب، ولا تكذبوهم و { قولوا آمنا بالله وما أنزل }…٧٥٤٣فرأيته يجانئ عليها الحجارة .٧٥٤٤ما أذن الله لشىء ما أذن لنبي حسن الصوت بالقرآن يجهر به٧٥٤٥وسعيد بن المسيب، وعلقمة بن وقاص، وعبيد الله بن عبد الله، عن حديث،…٧٥٤٦حدثنا أبو نعيم، حدثنا مسعر، عن عدي بن ثابت، أراه عن البراء، قال سمعت…٧٥٤٧وكان يرفع صوته، فإذا سمع المشركون سبوا القرآن ومن جاء به، فقال الله…٧٥٤٨إني أراك تحب الغنم والبادية، فإذا كنت في غنمك أو باديتك فأذنت للصلاة…٧٥٤٩حدثنا قبيصة، حدثنا سفيان، عن منصور، عن أمه، عن عائشة، قالت كان النبي…٧٥٥٠أرسله، اقرأ يا هشام٧٥٥١كل ميسر لما خلق له٧٥٥٢ما منكم من أحد إلا كتب مقعده من النار أو من الجنة٧٥٥٣لما قضى الله الخلق كتب كتابا عنده غلبت أو قال سبقت رحمتي غضبي . فهو…٧٥٥٤إن الله كتب كتابا قبل أن يخلق الخلق إن رحمتي سبقت غضبي . فهو مكتوب…٧٥٥٥والله لا أحملكم وما عندي ما أحملكم٧٥٥٦آمركم بأربع وأنهاكم عن أربع، آمركم بالإيمان بالله، وهل تدرون ما…٧٥٥٧إن أصحاب هذه الصور يعذبون يوم القيامة، ويقال لهم أحيوا ما خلقتم٧٥٥٨إن أصحاب هذه الصور يعذبون يوم القيامة، ويقال لهم أحيوا ما خلقتم٧٥٥٩قال الله عز وجل ومن أظلم ممن ذهب يخلق كخلقي، فليخلقوا ذرة، أو ليخلقوا…٧٥٦٠مثل المؤمن الذي يقرأ القرآن كالأترجة، طعمها طيب وريحها طيب، والذي لا…٧٥٦١إنهم ليسوا بشىء٧٥٦٢يخرج ناس من قبل المشرق ويقرءون القرآن لا يجاوز تراقيهم، يمرقون من…٧٥٦٣كلمتان حبيبتان إلى الرحمن، خفيفتان على اللسان، ثقيلتان في الميزان…
