كتاب الحج

٢٤٧ حديثًا•صفحة ٢ من ٢
١٦٧١أيها الناس عليكم بالسكينة، فإن البر ليس بالإيضاع١٦٧٢فنزل الشعب، فبال ثم توضأ، ولم يسبغ الوضوء . فقلت له الصلاة . فقال "…١٦٧٣كل واحدة منهما بإقامة، ولم يسبح بينهما ولا على إثر كل واحدة منهما .١٦٧٤أن رسول الله صلى الله عليه وسلم جمع في حجة الوداع المغرب والعشاء…١٦٧٥أو قريبا من ذلك، فأمر رجلا فأذن وأقام، ثم صلى المغرب، وصلى بعدها…١٦٧٦قال سالم وكان عبد الله بن عمر رضى الله عنهما يقدم ضعفة أهله، فيقفون…١٦٧٧حدثنا سليمان بن حرب، حدثنا حماد بن زيد، عن أيوب، عن عكرمة، عن ابن…١٦٧٨قدم النبي صلى الله عليه وسلم ليلة المزدلفة في ضعفة أهله .١٦٧٩أنها نزلت ليلة جمع عند المزدلفة، فقامت تصلي، فصلت ساعة، ثم قالت يا…١٦٨٠حدثنا محمد بن كثير، أخبرنا سفيان، حدثنا عبد الرحمن هو ابن القاسم عن…١٦٨١وكانت امرأة بطيئة، فأذن لها، فدفعت قبل حطمة الناس، وأقمنا حتى أصبحنا…١٦٨٢وصلى الفجر قبل ميقاتها .١٦٨٣إن هاتين الصلاتين حولتا عن وقتهما في هذا المكان المغرب والعشاء، فلا…١٦٨٤يقول شهدت عمر رضى الله عنه صلى بجمع الصبح، ثم وقف فقال إن المشركين…١٦٨٥فأخبر الفضل أنه لم يزل يلبي حتى رمى الجمرة .١٦٨٦ثم أردف الفضل من المزدلفة إلى منى قال فكلاهما قالا لم يزل النبي صلى…١٦٨٨فأمرني بها، وسألته عن الهدى، فقال فيها جزور أو بقرة أو شاة أو شرك في…١٦٨٩ويلك " . في الثالثة أو في الثانية .١٦٩٠وشعبة، قالا حدثنا قتادة، عن أنس رضى الله عنه أن النبي صلى الله عليه…١٦٩١من كان منكم أهدى فإنه لا يحل لشىء حرم منه حتى يقضي حجه، ومن لم يكن…١٦٩٢وعن عروة، أن عائشة رضى الله عنها أخبرته عن النبي صلى الله عليه وسلم…١٦٩٣قال قال عبد الله بن عبد الله بن عمر رضى الله عنهم لأبيه أقم، فإني لا…١٦٩٤ومروان، قالا خرج النبي صلى الله عليه وسلم من المدينة زمن الحديبية في…١٦٩٦ثم قلدها وأشعرها وأهداها، فما حرم عليه شىء كان أحل له .١٦٩٧إني لبدت رأسي، وقلدت هديي، فلا أحل حتى أحل من الحج١٦٩٨أن عائشة رضى الله عنها قالت كان رسول الله صلى الله عليه وسلم يهدي من…١٦٩٩وأقام بالمدينة، فما حرم عليه شىء كان له حل .١٧٠٠أنها أخبرته أن زياد بن أبي سفيان كتب إلى عائشة رضى الله عنها إن عبد…١٧٠١حدثنا أبو نعيم، حدثنا الأعمش، عن إبراهيم، عن الأسود، عن عائشة رضى…١٧٠٢ويقيم في أهله حلالا .١٧٠٣ثم يمكث حلالا .١٧٠٤حدثنا أبو نعيم، حدثنا زكرياء، عن عامر، عن مسروق، عن عائشة رضى الله…١٧٠٥حدثنا عمرو بن علي، حدثنا معاذ بن معاذ، حدثنا ابن عون، عن القاسم، عن…١٧٠٦حدثنا محمد، أخبرنا عبد الأعلى بن عبد الأعلى، عن معمر، عن يحيى بن أبي…١٧٠٧حدثنا قبيصة، حدثنا سفيان، عن ابن أبي نجيح، عن مجاهد، عن عبد الرحمن بن…١٧٠٨قال أراد ابن عمر رضى الله عنهما الحج عام حجة الحرورية في عهد ابن…١٧٠٩لا نرى إلا الحج، فلما دنونا من مكة أمر رسول الله صلى الله عليه وسلم…١٧١٠سمع خالد بن الحارث، حدثنا عبيد الله بن عمر، عن نافع، أن عبد الله رضى…١٧١١أن ابن عمر رضى الله عنهما كان يبعث بهديه من جمع من آخر الليل، حتى…١٧١٢وضحى بالمدينة كبشين أملحين أقرنين . مختصرا .١٧١٣قال رأيت ابن عمر رضى الله عنهما أتى على رجل، قد أناخ بدنته ينحرها،…١٧١٤والعصر بذي الحليفة ركعتين، فبات بها، فلما أصبح ركب راحلته، فجعل يهلل…١٧١٥فصلى الصبح، ثم ركب راحلته حتى إذا استوت به البيداء أهل بعمرة وحجة .١٧١٦فأمرني فقسمت لحومها، ثم أمرني فقسمت جلالها وجلودها . قال سفيان وحدثني…١٧١٧وعبد الكريم الجزري، أن مجاهدا، أخبرهما أن عبد الرحمن بن أبي ليلى…١٧١٨أن عليا رضى الله عنه حدثه قال أهدى النبي صلى الله عليه وسلم مائة…١٧١٩حدثنا مسدد، حدثنا يحيى، عن ابن جريج، حدثنا عطاء، سمع جابر بن عبد الله…١٧٢٠قال حدثتني عمرة، قالت سمعت عائشة رضى الله عنها تقول خرجنا مع رسول…١٧٢١لا حرج " .١٧٢٢حدثنا أحمد بن يونس، أخبرنا أبو بكر، عن عبد العزيز بن رفيع، عن عطاء،…١٧٢٣حدثنا محمد بن المثنى، حدثنا عبد الأعلى، حدثنا خالد، عن عكرمة، عن ابن…١٧٢٤انطلق فطف بالبيت وبالصفا والمروة " . ثم أتيت امرأة من نساء بني قيس،…١٧٢٥إني لبدت رأسي، وقلدت هديي، فلا أحل حتى أنحر١٧٢٦قال نافع كان ابن عمر رضى الله عنهما يقول حلق رسول الله صلى الله عليه…١٧٢٧اللهم ارحم المحلقين١٧٢٨اللهم اغفر للمحلقين١٧٢٩أن عبد الله، قال حلق النبي صلى الله عليه وسلم وطائفة من أصحابه، وقصر…١٧٣٠حدثنا أبو عاصم، عن ابن جريج، عن الحسن بن مسلم، عن طاوس، عن ابن عباس،…١٧٣١وبالصفا والمروة، ثم يحلوا، ويحلقوا أو يقصروا .١٧٣٢وقال لنا أبو نعيم حدثنا سفيان، عن عبيد الله، عن نافع، عن ابن عمر رضى…١٧٣٣فحاضت صفية، فأراد النبي صلى الله عليه وسلم منها ما يريد الرجل من أهله…١٧٣٤حدثنا موسى بن إسماعيل، حدثنا وهيب، حدثنا ابن طاوس، عن أبيه، عن ابن…١٧٣٥فيقول " لا حرج " . فسأله رجل، فقال حلقت قبل أن أذبح . قال " اذبح، ولا…١٧٣٦أن رسول الله صلى الله عليه وسلم وقف في حجة الوداع، فجعلوا يسألونه،…١٧٣٧شهد النبي صلى الله عليه وسلم يخطب يوم النحر، فقام إليه رجل، فقال كنت…١٧٣٨حدثنا إسحاق، قال أخبرنا يعقوب بن إبراهيم، حدثنا أبي، عن صالح، عن ابن…١٧٣٩يا أيها الناس . أى يوم هذا١٧٤٠حدثنا حفص بن عمر، حدثنا شعبة، قال أخبرني عمرو، قال سمعت جابر بن زيد،…١٧٤١أتدرون أى يوم هذا١٧٤٢أتدرون أى يوم هذا١٧٤٣حدثنا محمد بن عبيد بن ميمون، حدثنا عيسى بن يونس، عن عبيد الله، عن…١٧٤٤حدثنا يحيى بن موسى، حدثنا محمد بن بكر، أخبرنا ابن جريج، أخبرني عبيد…١٧٤٥من أجل سقايته، فأذن له . تابعه أبو أسامة وعقبة بن خالد وأبو ضمرة .١٧٤٦قال سألت ابن عمر رضى الله عنهما متى أرمي الجمار قال إذا رمى إمامك…١٧٤٧قال رمى عبد الله من بطن الوادي، فقلت يا أبا عبد الرحمن، إن ناسا…١٧٤٨ورمى بسبع، وقال هكذا رمى الذي أنزلت عليه سورة البقرة صلى الله عليه…١٧٤٩أنه حج مع ابن مسعود رضى الله عنه فرآه يرمي الجمرة الكبرى بسبع حصيات،…١٧٥٠يقول على المنبر السورة التي يذكر فيها البقرة، والسورة التي يذكر فيها…١٧٥١يكبر على إثر كل حصاة، ثم يتقدم حتى يسهل فيقوم مستقبل القبلة فيقوم…١٧٥٢أن عبد الله بن عمر رضى الله عنهما كان يرمي الجمرة الدنيا بسبع حصيات،…١٧٥٣أن رسول الله صلى الله عليه وسلم كان إذا رمى الجمرة التي تلي مسجد منى…١٧٥٤ولحله حين أحل، قبل أن يطوف . وبسطت يديها .١٧٥٥حدثنا مسدد، حدثنا سفيان، عن ابن طاوس، عن أبيه، عن ابن عباس رضى الله…١٧٥٦أن أنس بن مالك رضى الله عنه حدثه أن النبي صلى الله عليه وسلم صلى…١٧٥٧زوج النبي صلى الله عليه وسلم حاضت، فذكرت ذلك لرسول الله صلى الله عليه…١٧٥٩أن أهل المدينة، سألوا ابن عباس رضى الله عنهما عن امرأة، طافت ثم حاضت،…١٧٦١حدثنا مسلم، حدثنا وهيب، حدثنا ابن طاوس، عن أبيه، عن ابن عباس رضى الله…١٧٦٢ما كنت تطوفي بالبيت ليالي قدمنا١٧٦٣حدثنا محمد بن المثنى، حدثنا إسحاق بن يوسف، حدثنا سفيان الثوري، عن عبد…١٧٦٤أن قتادة، حدثه عن أنس بن مالك رضى الله عنه حدثه عن النبي صلى الله…١٧٦٥حدثنا أبو نعيم، حدثنا سفيان، عن هشام، عن أبيه، عن عائشة رضى الله عنها…١٧٦٦إنما هو منزل نزله رسول الله صلى الله عليه وسلم .١٧٦٧أن ابن عمر رضى الله عنهما كان يبيت بذي طوى بين الثنيتين، ثم يدخل من…١٧٦٨، قال نزل بها رسول الله صلى الله عليه وسلم وعمر وابن عمر . وعن نافع…١٧٦٩حتى إذا أصبح دخل، وإذا نفر مر بذي طوى وبات بها حتى يصبح، وكان يذكر أن…١٧٧٠قال عمرو بن دينار قال ابن عباس رضى الله عنهما كان ذو المجاز وعكاظ…١٧٧٢عقرى حلقى أطافت يوم النحر
