١٨٤ حديثًا•صفحة ١ من ٢
٢٠٤٧إنه لن يبسط أحد ثوبه حتى أقضي مقالتي هذه، ثم يجمع إليه ثوبه إلا وعى…٢٠٤٨فأقسم لك نصف مالي، وانظر أى زوجتى هويت نزلت لك عنها، فإذا حلت تزوجتها…٢٠٤٩وكان سعد ذا غنى، فقال لعبد الرحمن أقاسمك مالي نصفين، وأزوجك . قال…٢٠٥٠فلما كان الإسلام فكأنهم تأثموا فيه فنزلت { ليس عليكم جناح أن تبتغوا…٢٠٥١الحلال بين، والحرام بين وبينهما أمور مشتبهة، فمن ترك ما شبه عليه من…٢٠٥٢سوداء جاءت، فزعمت أنها أرضعتهما، فذكر للنبي فأعرض عنه، وتبسم النبي…٢٠٥٣هو لك يا عبد بن زمعة٢٠٥٤إذا أصاب بحده فكل، وإذا أصاب بعرضه فلا تأكل، فإنه وقيذ٢٠٥٥لولا أن تكون صدقة لأكلتها٢٠٥٦لا، حتى يسمع صوتا أو يجد ريحا٢٠٥٧سموا الله عليه وكلوه٢٠٥٨تحمل طعاما، فالتفتوا إليها، حتى ما بقي مع النبي صلى الله عليه وسلم…٢٠٥٩يأتي على الناس زمان، لا يبالي المرء ما أخذ منه أمن الحلال أم من الحرام٢٠٦١إن كان يدا بيد فلا بأس، وإن كان نساء فلا يصلح٢٠٦٢أن أبا موسى الأشعري، استأذن على عمر بن الخطاب رضى الله عنه فلم يؤذن…٢٠٦٣وقال الليث حدثني جعفر بن ربيعة، عن عبد الرحمن بن هرمز، عن أبي هريرة…٢٠٦٤ونحن نصلي مع النبي صلى الله عليه وسلم الجمعة، فانفض الناس إلا اثنى…٢٠٦٥إذا أنفقت المرأة من طعام بيتها، غير مفسدة، كان لها أجرها بما أنفقت،…٢٠٦٦إذا أنفقت المرأة من كسب زوجها عن غير أمره، فله نصف أجره٢٠٦٧من سره أن يبسط له رزقه أو ينسأ له في أثره فليصل رحمه٢٠٦٨ورهنه درعا من حديد .٢٠٦٩ما أمسى عند آل محمد صلى الله عليه وسلم صاع بر ولا صاع حب، وإن عنده…٢٠٧٠وشغلت بأمر المسلمين، فسيأكل آل أبي بكر من هذا المال ويحترف للمسلمين…٢٠٧١قال قالت عائشة رضى الله عنها كان أصحاب رسول الله صلى الله عليه وسلم…٢٠٧٢ما أكل أحد طعاما قط خيرا من أن يأكل من عمل يده، وإن نبي الله داود…٢٠٧٣أن داود عليه السلام كان لا يأكل إلا من عمل يده٢٠٧٤لأن يحتطب أحدكم حزمة على ظهره خير من أن يسأل أحدا، فيعطيه أو يمنعه٢٠٧٥لأن يأخذ أحدكم أحبله خير له من أن يسأل الناس٢٠٧٦رحم الله رجلا سمحا إذا باع، وإذا اشترى، وإذا اقتضى٢٠٧٧تلقت الملائكة روح رجل ممن كان قبلكم قالوا أعملت من الخير شيئا قال كنت…٢٠٧٨كان تاجر يداين الناس، فإذا رأى معسرا قال لفتيانه تجاوزوا عنه، لعل…٢٠٧٩البيعان بالخيار ما لم يتفرقا أو قال حتى يتفرقا فإن صدقا وبينا بورك…٢٠٨٠لا صاعين بصاع، ولا درهمين بدرهم٢٠٨١إن هذا قد تبعنا، فإن شئت أن تأذن له فأذن له، وإن شئت أن يرجع رجع٢٠٨٢البيعان بالخيار ما لم يتفرقا أو قال حتى يتفرقا فإن صدقا وبينا بورك…٢٠٨٣ليأتين على الناس زمان لا يبالي المرء بما أخذ المال، أمن حلال أم من…٢٠٨٤ثم حرم التجارة في الخمر .٢٠٨٥رأيت الليلة رجلين أتياني، فأخرجاني إلى أرض مقدسة، فانطلقنا حتى أتينا…٢٠٨٦قال رأيت أبي اشترى عبدا حجاما، فسألته فقال نهى النبي صلى الله عليه…٢٠٨٧الحلف منفقة للسلعة ممحقة للبركة٢٠٨٨أقام سلعة، وهو في السوق، فحلف بالله لقد أعطى بها ما لم يعط، ليوقع…٢٠٨٩أن حسين بن علي رضى الله عنهما أخبره أن عليا عليه السلام قال كانت لي…٢٠٩٠إن الله حرم مكة، ولم تحل لأحد قبلي، ولا لأحد بعدي، وإنما حلت لي ساعة…٢٠٩١وكان لي على العاص بن وائل دين، فأتيته أتقاضاه قال لا أعطيك حتى تكفر…٢٠٩٢أنه سمع أنس بن مالك رضى الله عنه يقول إن خياطا دعا رسول الله صلى الله…٢٠٩٣هي الشملة، منسوج في حاشيتها قالت يا رسول الله، إني نسجت هذه بيدي…٢٠٩٤أن مري غلامك النجار، يعمل لي أعوادا أجلس عليهن إذا كلمت الناس٢٠٩٥ألا أجعل لك شيئا تقعد عليه فإن لي غلاما نجارا . قال " إن شئت " . قال…٢٠٩٦ورهنه درعه .٢٠٩٧فأبطأ بي جملي وأعيا، فأتى على النبي صلى الله عليه وسلم فقال " جابر "…٢٠٩٨فلما كان الإسلام تأثموا من التجارة فيها، فأنزل الله { ليس عليكم جناح…٢٠٩٩وكانت عنده إبل هيم، فذهب ابن عمر رضى الله عنهما فاشترى تلك الإبل من…٢١٠٠فأعطاه يعني درعا فبعت الدرع، فابتعت به مخرفا في بني سلمة، فإنه لأول…٢١٠١مثل الجليس الصالح والجليس السوء كمثل صاحب المسك، وكير الحداد، لا…٢١٠٢وأمر أهله أن يخففوا من خراجه .٢١٠٣ولو كان حراما لم يعطه .٢١٠٤إني لم أرسل بها إليك لتلبسها، إنما يلبسها من لا خلاق له، إنما بعثت…٢١٠٥ما بال هذه النمرقة٢١٠٦يا بني النجار ثامنوني بحائطكم٢١٠٧إن المتبايعين بالخيار في بيعهما، ما لم يتفرقا، أو يكون البيع خيارا٢١٠٨البيعان بالخيار ما لم يفترقا٢١٠٩البيعان بالخيار ما لم يتفرقا، أو يقول أحدهما لصاحبه اختر٢١١٠البيعان بالخيار ما لم يتفرقا، فإن صدقا وبينا بورك لهما في بيعهما، وإن…٢١١١المتبايعان كل واحد منهما بالخيار على صاحبه ما لم يتفرقا، إلا بيع…٢١١٢إذا تبايع الرجلان فكل واحد منهما بالخيار، ما لم يتفرقا، وكانا جميعا،…٢١١٣كل بيعين لا بيع بينهما حتى يتفرقا، إلا بيع الخيار٢١١٤البيعان بالخيار ما لم يتفرقا٢١١٥فكان يغلبني فيتقدم أمام القوم، فيزجره عمر ويرده، ثم يتقدم فيزجره عمر…٢١١٦خشية أن يرادني البيع، وكانت السنة أن المتبايعين بالخيار حتى يتفرقا،…٢١١٧إذا بايعت فقل لا خلابة٢١١٨يغزو جيش الكعبة، فإذا كانوا ببيداء من الأرض يخسف بأولهم وآخرهم٢١١٩صلاة أحدكم في جماعة تزيد على صلاته في سوقه وبيته بضعا وعشرين درجة،…٢١٢٠سموا باسمي، ولا تكنوا بكنيتي٢١٢١سموا باسمي، ولا تكتنوا بكنيتي٢١٢٢أثم لكع أثم لكع٢١٢٤أنهم كانوا يشترون الطعام من الركبان على عهد النبي صلى الله عليه وسلم…٢١٢٥رسول الله صلى الله عليه وسلم في التوراة . قال أجل، والله إنه لموصوف…٢١٢٦من ابتاع طعاما فلا يبيعه حتى يستوفيه٢١٢٧اذهب فصنف تمرك أصنافا، العجوة على حدة، وعذق زيد على حدة، ثم أرسل إلى٢١٢٨كيلوا طعامكم يبارك لكم٢١٢٩أن إبراهيم حرم مكة، ودعا لها، وحرمت المدينة كما حرم إبراهيم مكة،…٢١٣٠اللهم بارك لهم في مكيالهم، وبارك لهم في صاعهم ومدهم٢١٣١حدثنا إسحاق بن إبراهيم، أخبرنا الوليد بن مسلم، عن الأوزاعي، عن…٢١٣٢حدثنا موسى بن إسماعيل، حدثنا وهيب، عن ابن طاوس، عن أبيه، عن ابن عباس…٢١٣٣من ابتاع طعاما فلا يبعه حتى يقبضه٢١٣٤الذهب بالذهب ربا إلا هاء وهاء، والبر بالبر ربا إلا هاء وهاء، والتمر…٢١٣٥قال الذي حفظناه من عمرو بن دينار سمع طاوسا، يقول سمعت ابن عباس رضى…٢١٣٦من ابتاع طعاما فلا يبعه حتى يستوفيه٢١٣٧أن ابن عمر رضى الله عنهما قال لقد رأيت الناس في عهد رسول الله صلى…٢١٣٨إلا لأمر حدث، فلما دخل عليه قال لأبي بكر " أخرج من عندك " . قال يا…٢١٣٩لا يبيع بعضكم على بيع أخيه٢١٤٠ولا تناجشوا، ولا يبيع الرجل على بيع أخيه ولا يخطب على خطبة أخيه، ولا…٢١٤١فاحتاج فأخذه النبي صلى الله عليه وسلم فقال " من يشتريه مني " فاشتراه…٢١٤٢حدثنا عبد الله بن مسلمة، حدثنا مالك، عن نافع، عن ابن عمر رضى الله…٢١٤٣وكان بيعا يتبايعه أهل الجاهلية، كان الرجل يبتاع الجزور إلى أن تنتج…٢١٤٤وهى طرح الرجل ثوبه بالبيع إلى الرجل، قبل أن يقلبه، أو ينظر إليه، ونهى…٢١٤٥أن يحتبي الرجل، في الثوب الواحد، ثم يرفعه على منكبه، وعن بيعتين…٢١٤٦حدثنا إسماعيل، قال حدثني مالك، عن محمد بن يحيى بن حبان، وعن أبي…٢١٤٧حدثنا عياش بن الوليد، حدثنا عبد الأعلى، حدثنا معمر، عن الزهري، عن…٢١٤٨لا تصروا الإبل والغنم، فمن ابتاعها بعد فإنه بخير النظرين بعد أن…٢١٤٩فردها فليرد معها صاعا . ونهى النبي صلى الله عليه وسلم أن تلقى البيوع .٢١٥٠لا تلقوا الركبان، ولا يبيع بعضكم على بيع بعض ولا تناجشوا ولا يبيع…٢١٥١من اشترى غنما مصراة فاحتلبها، فإن رضيها أمسكها، وإن سخطها ففي حلبتها…٢١٥٢إذا زنت الأمة فتبين زناها فليجلدها، ولا يثرب، ثم إن زنت فليجلدها، ولا…٢١٥٤إن زنت فاجلدوها، ثم إن زنت فاجلدوها، ثم إن زنت فبيعوها ولو بضفير٢١٥٥اشتري وأعتقي، فإن الولاء لمن أعتق٢١٥٦إنما الولاء لمن أعتق٢١٥٧وإقام الصلاة، وإيتاء الزكاة، والسمع والطاعة، والنصح لكل مسلم .٢١٥٨لا تلقوا الركبان ولا يبيع حاضر لباد٢١٥٩حدثني عبد الله بن صباح، حدثنا أبو علي الحنفي، عن عبد الرحمن بن عبد…٢١٦٠لا يبتاع المرء على بيع أخيه، ولا تناجشوا، ولا يبع حاضر لباد٢١٦١حدثنا محمد بن المثنى، حدثنا معاذ، حدثنا ابن عون، عن محمد، قال أنس بن…٢١٦٢وأن يبيع حاضر لباد .٢١٦٣لا يبيعن حاضر لباد٢١٦٤حدثنا مسدد، حدثنا يزيد بن زريع، قال حدثني التيمي، عن أبي عثمان، عن…٢١٦٥لا يبيع بعضكم على بيع بعض، ولا تلقوا السلع حتى يهبط بها إلى السوق٢١٦٦يبينه حديث عبيد الله .٢١٦٧فنهاهم رسول الله صلى الله عليه وسلم أن يبيعوه في مكانه حتى ينقلوه .٢١٦٨خذيها واشترطي لهم الولاء، فإنما الولاء لمن أعتق٢١٦٩لا يمنعك ذلك، فإنما الولاء لمن أعتق٢١٧٠البر بالبر ربا إلا هاء وهاء، والشعير بالشعير ربا إلا هاء وهاء، والتمر…٢١٧١والمزابنة بيع الثمر بالتمر كيلا، وبيع الزبيب بالكرم كيلا .٢١٧٣أن النبي صلى الله عليه وسلم رخص في العرايا بخرصها .٢١٧٤الذهب بالذهب ربا إلا هاء وهاء، والبر بالبر ربا إلا هاء وهاء، والشعير…٢١٧٥لا تبيعوا الذهب بالذهب إلا سواء بسواء، والفضة بالفضة إلا سواء بسواء،…٢١٧٦الذهب بالذهب مثلا بمثل والورق بالورق مثلا بمثل٢١٧٧لا تبيعوا الذهب بالذهب إلا مثلا بمثل، ولا تشفوا بعضها على بعض، ولا…٢١٧٩لا ربا إلا في النسيئة٢١٨١، فكل واحد منهما يقول هذا خير مني . فكلاهما يقول نهى رسول الله صلى…٢١٨٢إلا سواء بسواء، وأمرنا أن نبتاع الذهب بالفضة كيف شئنا، والفضة بالذهب…٢١٨٤لا تبيعوا الثمر حتى يبدو صلاحه، ولا تبيعوا الثمر بالتمر٢١٨٥وبيع الكرم بالزبيب كيلا .٢١٨٦حدثنا عبد الله بن يوسف، أخبرنا مالك، عن داود بن الحصين، عن أبي سفيان،…٢١٨٧حدثنا مسدد، حدثنا أبو معاوية، عن الشيباني، عن عكرمة، عن ابن عباس رضى…٢١٨٨حدثنا عبد الله بن مسلمة، حدثنا مالك، عن نافع، عن ابن عمر، عن زيد بن…٢١٨٩وأبي الزبير، عن جابر رضى الله عنه قال نهى النبي صلى الله عليه وسلم عن…٢١٩٠وسأله، عبيد الله بن الربيع أحدثك داود عن أبي سفيان، عن أبي هريرة رضى…٢١٩١ورخص في العرية أن تباع بخرصها يأكلها أهلها رطبا . وقال سفيان مرة أخرى…٢١٩٢حدثنا محمد، أخبرنا عبد الله، أخبرنا موسى بن عقبة، عن نافع، عن ابن…٢١٩٣فإما لا فلا يتبايعوا حتى يبدو صلاح الثمر٢١٩٤نهى البائع والمبتاع .٢١٩٥حدثنا ابن مقاتل، أخبرنا عبد الله، أخبرنا حميد الطويل، عن أنس رضى الله…٢١٩٦حدثنا مسدد، حدثنا يحيى بن سعيد، عن سليم بن حيان، حدثنا سعيد بن مينا،…٢١٩٧حدثني علي بن الهيثم، حدثنا معلى، حدثنا هشيم، أخبرنا حميد، حدثنا أنس…٢١٩٨أرأيت إذا منع الله الثمرة، بم يأخذ أحدكم مال أخيه٢١٩٩لا تتبايعوا الثمر حتى يبدو صلاحها، ولا تبيعوا الثمر بالتمر٢٢٠٠فرهنه درعه .٢٢٠٢أكل تمر خيبر هكذا٢٢٠٣مولى ابن عمر أن أيما، نخل بيعت قد أبرت لم يذكر الثمر، فالثمر للذي…٢٢٠٤من باع نخلا قد أبرت فثمرها للبائع، إلا أن يشترط المبتاع
