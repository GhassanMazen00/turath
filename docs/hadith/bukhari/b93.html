٨٤ حديثًا
٧١٣٧من أطاعني فقد أطاع الله، ومن عصاني فقد عصى الله، ومن أطاع أميري فقد…٧١٣٨ألا كلكم راع، وكلكم مسئول عن رعيته، فالإمام الذي على الناس راع وهو…٧١٣٩إن هذا الأمر في قريش، لا يعاديهم أحد إلا كبه الله على وجهه ما أقاموا…٧١٤٠لا يزال الأمر في قريش ما بقي منهم اثنان٧١٤١لا حسد إلا في اثنتين، رجل آتاه الله مالا فسلطه على هلكته في الحق،…٧١٤٢اسمعوا وأطيعوا وإن استعمل عليكم عبد حبشي كأن رأسه زبيبة٧١٤٣من رأى من أميره شيئا فكرهه فليصبر، فإنه ليس أحد يفارق الجماعة شبرا…٧١٤٤السمع والطاعة على المرء المسلم، فيما أحب وكره، ما لم يؤمر بمعصية،…٧١٤٥لو دخلوها ما خرجوا منها أبدا، إنما الطاعة في المعروف٧١٤٦يا عبد الرحمن لا تسأل الإمارة، فإنك إن أعطيتها عن مسألة وكلت إليها،…٧١٤٧يا عبد الرحمن بن سمرة، لا تسأل الإمارة، فإن أعطيتها عن مسألة وكلت…٧١٤٨إنكم ستحرصون على الإمارة، وستكون ندامة يوم القيامة، فنعم المرضعة…٧١٤٩إنا لا نولي هذا من سأله، ولا من حرص عليه٧١٥٠ما من عبد استرعاه الله رعية، فلم يحطها بنصيحة، إلا لم يجد رائحة الجنة٧١٥١ما من وال يلي رعية من المسلمين، فيموت وهو غاش لهم، إلا حرم الله عليه…٧١٥٢من سمع سمع الله به يوم القيامة قال ومن يشاقق يشقق الله عليه يوم…٧١٥٣ولكني أحب الله ورسوله . قال " أنت مع من أحببت " .٧١٥٤اتقي الله واصبري٧١٥٥أن قيس بن سعد، كان يكون بين يدى النبي صلى الله عليه وسلم بمنزلة صاحب…٧١٥٦أن النبي صلى الله عليه وسلم بعثه وأتبعه بمعاذ .٧١٥٧أن رجلا، أسلم ثم تهود، فأتى معاذ بن جبل وهو عند أبي موسى فقال ما هذا…٧١٥٨لا يقضين حكم بين اثنين وهو غضبان٧١٥٩يا أيها الناس، إن منكم منفرين، فأيكم ما صلى بالناس فليوجز، فإن فيهم…٧١٦٠ليراجعها، ثم ليمسكها حتى تطهر، ثم تحيض فتطهر، فإن بدا له أن يطلقها…٧١٦١لا حرج عليك أن تطعميهم من معروف٧١٦٢قال لما أراد النبي صلى الله عليه وسلم أن يكتب إلى الروم قالوا إنهم لا…٧١٦٣خذه فتموله وتصدق به، فما جاءك من هذا المال وأنت غير مشرف ولا سائل…٧١٦٤خذه فتموله وتصدق به، فما جاءك من هذا المال وأنت غير مشرف ولا سائل…٧١٦٥عشرة فرق بينهما .٧١٦٦أخي بني ساعدة أن رجلا، من الأنصار جاء إلى النبي صلى الله عليه وسلم…٧١٦٧وسعيد بن المسيب، عن أبي هريرة، قال أتى رجل رسول الله صلى الله عليه…٧١٦٨سمع جابر بن عبد الله، قال كنت فيمن رجمه بالمصلى . رواه يونس ومعمر…٧١٦٩إنما أنا بشر، وإنكم تختصمون إلى، ولعل بعضكم أن يكون ألحن بحجته من بعض…٧١٧٠من له بينة على قتيل قتله، فله سلبه٧١٧١أن النبي صلى الله عليه وسلم أتته صفية بنت حيى فلما رجعت انطلق معها،…٧١٧٢يسرا ولا تعسرا، وبشرا ولا تنفرا، وتطاوعا٧١٧٣فكوا العاني وأجيبوا الداعي٧١٧٤ما بال العامل نبعثه، فيأتي يقول هذا لك وهذا لي . فهلا جلس في بيت أبيه…٧١٧٥أن نافعا، أخبره أن ابن عمر رضى الله عنهما أخبره قال كان سالم مولى أبي…٧١٧٧إني لا أدري من أذن منكم ممن لم يأذن، فارجعوا حتى يرفع إلينا عرفاؤكم…٧١٧٨حدثنا أبو نعيم، حدثنا عاصم بن محمد بن زيد بن عبد الله بن عمر، عن…٧١٧٩إن شر الناس ذو الوجهين، الذي يأتي هؤلاء بوجه وهؤلاء بوجه٧١٨٠خذي ما يكفيك وولدك بالمعروف٧١٨١إنما أنا بشر، وإنه يأتيني الخصم، فلعل بعضكم أن يكون أبلغ من بعض،…٧١٨٢هو لك يا عبد بن زمعة٧١٨٤لا يحلف على يمين صبر، يقتطع مالا وهو فيها فاجر، إلا لقي الله وهو عليه…٧١٨٥إنما أنا بشر، وإنه يأتيني الخصم، فلعل بعضا أن يكون أبلغ من بعض، أقضي…٧١٨٦لم يكن له مال غيره، فباعه بثمانمائة درهم، ثم أرسل بثمنه إليه .٧١٨٧إن تطعنوا في إمارته فقد كنتم تطعنون في إمارة أبيه من قبله، وايم الله…٧١٨٨أبغض الرجال إلى الله الألد الخصم٧١٨٩اللهم إني أبرأ إليك مما صنع خالد بن الوليد٧١٩٠يا أبا بكر ما منعك إذ أومأت إليك أن لا تكون مضيت٧١٩١قال بعث إلى أبو بكر لمقتل أهل اليمامة وعنده عمر فقال أبو بكر إن عمر…٧١٩٢أنه أخبره هو، ورجال، من كبراء قومه أن عبد الله بن سهل ومحيصة خرجا إلى…٧١٩٤لأقضين بينكما بكتاب الله، أما الوليدة والغنم فرد عليك، وعلى ابنك جلد…٧١٩٥أن النبي صلى الله عليه وسلم أمره أن يتعلم كتاب اليهود، حتى كتبت للنبي…٧١٩٦أن عبد الله بن عباس، أخبره أن أبا سفيان بن حرب أخبره أن هرقل أرسل…٧١٩٧فهلا جلست في بيت أبيك وبيت أمك حتى تأتيك هديتك، إن كنت صادقا٧١٩٨ما بعث الله من نبي ولا استخلف من خليفة، إلا كانت له بطانتان، بطانة…٧٢٠٠بايعنا رسول الله صلى الله عليه وسلم على السمع والطاعة في المنشط…٧٢٠١اللهم إن الخير خير الآخره فاغفر للأنصار والمهاجره٧٢٠٢حدثنا عبد الله بن يوسف، أخبرنا مالك، عن عبد الله بن دينار، عن عبد…٧٢٠٣قال شهدت ابن عمر حيث اجتمع الناس على عبد الملك قال كتب إني أقر بالسمع…٧٢٠٤قال بايعت النبي صلى الله عليه وسلم على السمع والطاعة، فلقنني، فيما…٧٢٠٥قال لما بايع الناس عبد الملك كتب إليه عبد الله بن عمر إلى عبد الله…٧٢٠٦قال قلت لسلمة على أى شىء بايعتم النبي صلى الله عليه وسلم يوم الحديبية…٧٢٠٧أن حميد بن عبد الرحمن، أخبره أن المسور بن مخرمة أخبره . أن الرهط…٧٢٠٨يا سلمة ألا تبايع٧٢٠٩المدينة كالكير، تنفي خبثها، وينصع طيبها٧٢١٠وكان، قد أدرك النبي صلى الله عليه وسلم وذهبت به أمه زينب ابنة حميد…٧٢١١إنما المدينة كالكير تنفي خبثها وينصع طيبها٧٢١٢ثلاثة لا يكلمهم الله يوم القيامة، ولا يزكيهم، ولهم عذاب أليم رجل على…٧٢١٣تبايعوني على أن لا تشركوا بالله شيئا، ولا تسرقوا، ولا تزنوا، ولا…٧٢١٤إلا امرأة يملكها .٧٢١٥فلم يقل شيئا، ثم رجعت، فما وفت امرأة إلا أم سليم وأم العلاء، وابنة…٧٢١٦المدينة كالكير، تنفي خبثها، وينصع طيبها٧٢١٧ذاك لو كان وأنا حى فأستغفر لك وأدعو لك٧٢١٨وإن أترك فقد ترك من هو خير مني رسول الله صلى الله عليه وسلم فأثنوا…٧٢١٩عمر الآخرة حين جلس على المنبر، وذلك الغد من يوم توفي النبي صلى الله…٧٢٢٠إن لم تجديني فأتي أبا بكر٧٢٢١حدثنا مسدد، حدثنا يحيى، عن سفيان، حدثني قيس بن مسلم، عن طارق بن شهاب،…٧٢٢٣يكون اثنا عشر أميرا فقال كلمة لم أسمعها فقال أبي إنه قال كلهم من قريش٧٢٢٤والذي نفسي بيده لقد هممت أن آمر بحطب يحتطب، ثم آمر بالصلاة فيؤذن لها،…٧٢٢٥حدثني يحيى بن بكير، حدثنا الليث، عن عقيل، عن ابن شهاب، عن عبد الرحمن…
