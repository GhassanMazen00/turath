١٤٨ حديثًا
١٢٣٧أتاني آت من ربي فأخبرني أو قال بشرني أنه من مات من أمتي لا يشرك بالله…١٢٣٨من مات يشرك بالله شيئا دخل النار١٢٣٩وعيادة المريض، وإجابة الداعي، ونصر المظلوم، وإبرار القسم، ورد السلام،…١٢٤٠حق المسلم على المسلم خمس رد السلام، وعيادة المريض، واتباع الجنائز،…١٢٤٢ويونس، عن الزهري، قال أخبرني أبو سلمة، أن عائشة رضى الله عنها زوج…١٢٤٣وما يدريك أن الله قد أكرمه١٢٤٤تبكين أو لا تبكين، ما زالت الملائكة تظله بأجنحتها حتى رفعتموه١٢٤٥خرج إلى المصلى، فصف بهم وكبر أربعا .١٢٤٦أخذ الراية زيد فأصيب، ثم أخذها جعفر فأصيب، ثم أخذها عبد الله بن رواحة…١٢٤٧ما منعكم أن تعلموني١٢٤٨ما من الناس من مسلم يتوفى له ثلاث لم يبلغوا الحنث، إلا أدخله الله…١٢٥٠أيما امرأة مات لها ثلاثة من الولد كانوا حجابا من النار١٢٥١لا يموت لمسلم ثلاثة من الولد، فيلج النار إلا تحلة القسم١٢٥٢اتقي الله واصبري١٢٥٣اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك بماء وسدر، واجعلن…١٢٥٤اغسلنها ثلاثا أو خمسا أو أكثر من ذلك بماء وسدر، واجعلن في الآخرة…١٢٥٥ابدأن بميامنها ومواضع الوضوء منها١٢٥٦ابدأن بميامنها ومواضع الوضوء١٢٥٧اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن، فإذا فرغتن فآذنني١٢٥٩اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن بماء وسدر، واجعلن في…١٢٦٠حدثنا أحمد، حدثنا عبد الله بن وهب، أخبرنا ابن جريج، قال أيوب وسمعت…١٢٦١اغسلنها ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك بماء وسدر، واجعلن…١٢٦٢حدثنا قبيصة، حدثنا سفيان، عن هشام، عن أم الهذيل، عن أم عطية رضى الله…١٢٦٣اغسلنها بالسدر وترا ثلاثا أو خمسا أو أكثر من ذلك إن رأيتن ذلك، واجعلن…١٢٦٤ليس فيهن قميص ولا عمامة .١٢٦٥اغسلوه بماء وسدر، وكفنوه في ثوبين، ولا تحنطوه ولا تخمروا رأسه، فإنه…١٢٦٦اغسلوه بماء وسدر، وكفنوه في ثوبين، ولا تحنطوه ولا تخمروا رأسه، فإن…١٢٦٧اغسلوه بماء وسدر، وكفنوه في ثوبين، ولا تمسوه طيبا، ولا تخمروا رأسه،…١٢٦٨اغسلوه بماء وسدر، وكفنوه في ثوبين، ولا تحنطوه ولا تخمروا رأسه، فإنه…١٢٦٩آذني أصلي عليه١٢٧٠سمع جابرا رضى الله عنه قال أتى النبي صلى الله عليه وسلم عبد الله بن…١٢٧١ليس فيها قميص ولا عمامة .١٢٧٢ليس فيها قميص ولا عمامة .١٢٧٣ليس فيها قميص ولا عمامة .١٢٧٤قال أتي عبد الرحمن بن عوف رضى الله عنه يوما بطعامه فقال قتل مصعب بن…١٢٧٥إبراهيم أن عبد الرحمن بن عوف رضى الله عنه أتي بطعام وكان صائما فقال…١٢٧٦وأن نجعل على رجليه من الإذخر .١٢٧٧جاءت النبي صلى الله عليه وسلم ببردة منسوجة فيها حاشيتها أتدرون ما…١٢٧٨حدثنا قبيصة بن عقبة، حدثنا سفيان، عن خالد الحذاء، عن أم الهذيل، عن أم…١٢٧٩قال توفي ابن لأم عطية رضى الله عنها فلما كان اليوم الثالث دعت بصفرة،…١٢٨٠لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد على ميت فوق ثلاث، إلا…١٢٨٢لا يحل لامرأة تؤمن بالله واليوم الآخر تحد على ميت فوق ثلاث، إلا على…١٢٨٣اتقي الله واصبري١٢٨٤إن لله ما أخذ وله ما أعطى وكل عنده بأجل مسمى، فلتصبر ولتحتسب١٢٨٥هل منكم رجل لم يقارف الليلة١٢٨٨إن الميت ليعذب ببكاء أهله عليه١٢٨٩إنهم ليبكون عليها، وإنها لتعذب في قبرها١٢٩٠إن الميت ليعذب ببكاء الحى١٢٩١إن كذبا على ليس ككذب على أحد، من كذب على متعمدا فليتبوأ مقعده من النار١٢٩٢الميت يعذب في قبره بما نيح عليه١٢٩٣قد مثل به حتى وضع بين يدى رسول الله صلى الله عليه وسلم وقد سجي ثوبا…١٢٩٤ليس منا من لطم الخدود، وشق الجيوب، ودعا بدعوى الجاهلية١٢٩٥ولا يرثني إلا ابنة، أفأتصدق بثلثى مالي قال " لا " . فقلت بالشطر فقال…١٢٩٦أن القاسم بن مخيمرة، حدثه قال حدثني أبو بردة بن أبي موسى رضى الله عنه…١٢٩٧ليس منا من ضرب الخدود، وشق الجيوب، ودعا بدعوى الجاهلية١٢٩٨ليس منا من ضرب الخدود، وشق الجيوب، ودعا بدعوى الجاهلية١٢٩٩فاحث في أفواههن التراب١٣٠٠حدثنا عمرو بن علي، حدثنا محمد بن فضيل، حدثنا عاصم الأحول، عن أنس رضى…١٣٠١لعل الله أن يبارك لكما في ليلتكما١٣٠٢الصبر عند الصدمة الأولى١٣٠٣يا ابن عوف إنها رحمة١٣٠٤ولا بحزن القلب، ولكن يعذب بهذا وأشار إلى لسانه أو يرحم وإن الميت يعذب…١٣٠٥فاحث في أفواههن التراب١٣٠٦حدثنا عبد الله بن عبد الوهاب، حدثنا حماد بن زيد، حدثنا أيوب، عن محمد،…١٣٠٧إذا رأيتم الجنازة فقوموا حتى تخلفكم١٣٠٨إذا رأى أحدكم جنازة فإن لم يكن ماشيا معها فليقم حتى يخلفها، أو تخلفه…١٣٠٩حدثنا أحمد بن يونس، حدثنا ابن أبي ذئب، عن سعيد المقبري، عن أبيه، قال…١٣١٠إذا رأيتم الجنازة فقوموا، فمن تبعها فلا يقعد حتى توضع١٣١١إذا رأيتم الجنازة فقوموا١٣١٣قال كان سهل بن حنيف وقيس بن سعد قاعدين بالقادسية، فمروا عليهما بجنازة…١٣١٤إذا وضعت الجنازة واحتملها الرجال على أعناقهم، فإن كانت صالحة قالت…١٣١٥أسرعوا بالجنازة، فإن تك صالحة فخير تقدمونها { إليه } ، وإن يك سوى ذلك…١٣١٦إذا وضعت الجنازة فاحتملها الرجال على أعناقهم، فإن كانت صالحة قالت…١٣١٧فكنت في الصف الثاني أو الثالث .١٣١٨ثم تقدم فصفوا خلفه فكبر أربعا .١٣١٩شهد النبي صلى الله عليه وسلم أنه أتى على قبر منبوذ فصفهم وكبر أربعا .…١٣٢٠قد توفي اليوم رجل صالح من الحبش فهلم فصلوا عليه١٣٢١حدثنا موسى بن إسماعيل، حدثنا عبد الواحد، حدثنا الشيباني، عن عامر، عن…١٣٢٢حدثنا سليمان بن حرب، حدثنا شعبة، عن الشيباني، عن الشعبي، قال أخبرني…١٣٢٤يقول حدث ابن عمر، أن أبا هريرة رضى الله عنهم يقول من تبع جنازة فله…١٣٢٥من شهد الجنازة حتى يصلي عليها فله قيراط، ومن شهد حتى تدفن كان له…١٣٢٦فقالوا هذا دفن، أو دفنت البارحة . قال ابن عباس رضى الله عنهما فصفنا…١٣٢٨وأبي، سلمة أنهما حدثاه عن أبي هريرة رضى الله عنه قال نعى لنا رسول…١٣٢٩جاءوا إلى النبي صلى الله عليه وسلم برجل منهم وامرأة زنيا، فأمر بهما…١٣٣٠لعن الله اليهود والنصارى، اتخذوا قبور أنبيائهم مسجدا١٣٣١فقام عليها وسطها .١٣٣٢حدثنا عمران بن ميسرة، حدثنا عبد الوارث، حدثنا حسين، عن ابن بريدة،…١٣٣٣وخرج بهم إلى المصلى فصف بهم، وكبر عليه أربع تكبيرات .١٣٣٤حدثنا محمد بن سنان، حدثنا سليم بن حيان، حدثنا سعيد بن ميناء، عن جابر…١٣٣٥قال صليت خلف ابن عباس رضى الله عنهما على جنازة فقرأ بفاتحة الكتاب قال…١٣٣٦مر مع النبي صلى الله عليه وسلم على قبر منبوذ فأمهم وصلوا خلفه . قلت…١٣٣٧ما فعل ذلك الإنسان١٣٣٨العبد إذا وضع في قبره، وتولي وذهب أصحابه حتى إنه ليسمع قرع نعالهم،…١٣٣٩أرسل ملك الموت إلى موسى عليهما السلام فلما جاءه صكه فرجع إلى ربه فقال…١٣٤٠وكان سأل عنه فقال " من هذا " . فقالوا فلان، دفن البارحة . فصلوا عليه .١٣٤١أولئك إذا مات منهم الرجل الصالح بنوا على قبره مسجدا، ثم صوروا فيه تلك…١٣٤٢هل فيكم من أحد لم يقارف الليلة١٣٤٣أيهم أكثر أخذا للقرآن١٣٤٤إني فرط لكم، وأنا شهيد عليكم، وإني والله لأنظر إلى حوضي الآن، وإني…١٣٤٥أن جابر بن عبد الله رضى الله عنهما أخبره أن النبي صلى الله عليه وسلم…١٣٤٦ادفنوهم في دمائهم١٣٤٨أيهم أكثر أخذا للقرآن١٣٤٩حرم الله مكة، فلم تحل لأحد قبلي ولا لأحد بعدي، أحلت لي ساعة من نهار،…١٣٥٠فوضعه على ركبتيه، ونفث عليه من ريقه، وألبسه قميصه، فالله أعلم، وكان…١٣٥١وإني لا أترك بعدي أعز على منك، غير نفس رسول الله صلى الله عليه وسلم،…١٣٥٢حدثنا علي بن عبد الله، حدثنا سعيد بن عامر، عن شعبة، عن ابن أبي نجيح،…١٣٥٣أيهم أكثر أخذا للقرآن١٣٥٥تشهد أني رسول الله١٣٥٦فأتاه النبي صلى الله عليه وسلم يعوده، فقعد عند رأسه فقال له " أسلم "…١٣٥٧وأمي، من النساء .١٣٥٨ما من مولود إلا يولد على الفطرة، فأبواه يهودانه أو ينصرانه أو…١٣٥٩ما من مولود إلا يولد على الفطرة، فأبواه يهودانه أو ينصرانه أو…١٣٦٠يا عم، قل لا إله إلا الله، كلمة أشهد لك بها عند الله١٣٦١إنهما ليعذبان وما يعذبان في كبير أما أحدهما فكان لا يستتر من البول،…١٣٦٢ما منكم من أحد، ما من نفس منفوسة إلا كتب مكانها من الجنة والنار، وإلا…١٣٦٤من حلف بملة غير الإسلام كاذبا متعمدا فهو كما قال، ومن قتل نفسه بحديدة…١٣٦٥الذي يخنق نفسه يخنقها في النار، والذي يطعنها يطعنها في النار١٣٦٦أخر عني يا عمر١٣٦٧فقال النبي صلى الله عليه وسلم " وجبت " . ثم مروا بأخرى فأثنوا عليها…١٣٦٨أيما مسلم شهد له أربعة بخير أدخله الله الجنة١٣٦٩إذا أقعد المؤمن في قبره أتي، ثم شهد أن لا إله إلا الله، وأن محمدا…١٣٧٠وجدتم ما وعد ربكم حقا١٣٧١إنهم ليعلمون الآن أن ما كنت أقول حق وقد قال الله تعالى { إنك لا تسمع…١٣٧٢دخلت عليها، فذكرت عذاب القبر، فقالت لها أعاذك الله من عذاب القبر .…١٣٧٣أنه سمع أسماء بنت أبي بكر رضى الله عنهما تقول قام رسول الله صلى الله…١٣٧٤إن العبد إذا وضع في قبره، وتولى عنه أصحابه، وإنه ليسمع قرع نعالهم،…١٣٧٥يهود تعذب في قبورها١٣٧٦قال حدثتني ابنة خالد بن سعيد بن العاص، أنها سمعت النبي صلى الله عليه…١٣٧٧اللهم إني أعوذ بك من عذاب القبر، ومن عذاب النار، ومن فتنة المحيا…١٣٧٨إنهما ليعذبان، وما يعذبان من كبير ثم قال بلى أما أحدهما فكان يسعى…١٣٧٩إن أحدكم إذا مات عرض عليه مقعده بالغداة والعشي، إن كان من أهل الجنة…١٣٨٠إذا وضعت الجنازة فاحتملها الرجال على أعناقهم، فإن كانت صالحة قالت…١٣٨١ما من الناس مسلم يموت له ثلاثة من الولد لم يبلغوا الحنث إلا أدخله…١٣٨٢إن له مرضعا في الجنة١٣٨٣الله إذ خلقهم أعلم بما كانوا عاملين١٣٨٤الله أعلم بما كانوا عاملين١٣٨٥كل مولود يولد على الفطرة، فأبواه يهودانه أو ينصرانه أو يمجسانه، كمثل…١٣٨٦من رأى منكم الليلة رؤيا١٣٨٧ليس فيها قميص ولا عمامة . وقال لها في أى يوم توفي رسول الله صلى الله…١٣٨٨قال للنبي صلى الله عليه وسلم إن أمي افتلتت نفسها، وأظنها لو تكلمت…١٣٨٩أين أنا اليوم أين أنا غدا١٣٩٠لعن الله اليهود والنصارى، اتخذوا قبور أنبيائهم مساجد١٣٩٠.٢أنه حدثه أنه، رأى قبر النبي صلى الله عليه وسلم مسنما .١٣٩٠.٣لما سقط عليهم الحائط في زمان الوليد بن عبد الملك أخذوا في بنائه، فبدت…١٣٩١لا أزكى به أبدا .١٣٩٢قال رأيت عمر بن الخطاب رضى الله عنه قال يا عبد الله بن عمر، اذهب إلى…١٣٩٣لا تسبوا الأموات فإنهم قد أفضوا إلى ما قدموا١٣٩٤حدثنا عمر بن حفص، حدثنا أبي، حدثنا الأعمش، حدثني عمرو بن مرة، عن سعيد…
