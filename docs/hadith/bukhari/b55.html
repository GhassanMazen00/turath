٤٤ حديثًا
٢٧٣٨ما حق امرئ مسلم له شىء، يوصي فيه يبيت ليلتين، إلا ووصيته مكتوبة عنده٢٧٣٩ختن رسول الله صلى الله عليه وسلم أخي جويرية بنت الحارث قال ما ترك…٢٧٤٠قال سألت عبد الله بن أبي أوفى رضى الله عنهما هل كان النبي صلى الله…٢٧٤١قال ذكروا عند عائشة أن عليا رضى الله عنهما كان وصيا . فقالت متى أوصى…٢٧٤٢يرحم الله ابن عفراء٢٧٤٣الثلث، والثلث كثير أو كبير٢٧٤٤لعل الله يرفعك وينفع بك ناسا٢٧٤٥هو لك يا عبد ابن زمعة، الولد للفراش، وللعاهر الحجر٢٧٤٦رض رأس جارية بين حجرين، فقيل لها من فعل بك، أفلان أو فلان حتى سمي…٢٧٤٧وكانت الوصية للوالدين، فنسخ الله من ذلك ما أحب، فجعل للذكر مثل حظ…٢٧٤٨أن تصدق وأنت صحيح حريص . تأمل الغنى، وتخشى الفقر، ولا تمهل حتى إذا…٢٧٤٩آية المنافق ثلاث، إذا حدث كذب، وإذا اؤتمن خان، وإذا وعد أخلف٢٧٥٠يا حكيم، إن هذا المال خضر حلو، فمن أخذه بسخاوة نفس بورك له فيه، ومن…٢٧٥١كلكم راع ومسئول عن رعيته، والإمام راع ومسئول عن رعيته، والرجل راع في…٢٧٥٢أرى أن تجعلها في الأقربين٢٧٥٣يا معشر قريش أو كلمة نحوها اشتروا أنفسكم، لا أغني عنكم من الله شيئا،…٢٧٥٤فقال له " اركبها " . فقال يا رسول الله إنها بدنة . قال في الثالثة أو…٢٧٥٥فقال " اركبها " . قال يا رسول الله إنها بدنة . قال " اركبها، ويلك " .…٢٧٥٦أنه سمع عكرمة، يقول أنبأنا ابن عباس رضى الله عنهما أن سعد بن عبادة…٢٧٥٧أمسك عليك بعض مالك فهو خير لك٢٧٥٨بخ يا أبا طلحة، ذلك مال رابح، قبلناه منك ورددناه عليك، فاجعله في…٢٧٥٩ولا والله ما نسخت، ولكنها مما تهاون الناس، هما واليان وال يرث، وذاك…٢٧٦٠نعم، تصدق عنها٢٧٦١حدثنا عبد الله بن يوسف، أخبرنا مالك، عن ابن شهاب، عن عبيد الله بن عبد…٢٧٦٢أن ابن جريج، أخبرهم قال أخبرني يعلى، أنه سمع عكرمة، مولى ابن عباس…٢٧٦٣قال كان عروة بن الزبير يحدث أنه سأل عائشة رضى الله عنها – { وإن خفتم…٢٧٦٤تصدق بأصله، لا يباع ولا يوهب ولا يورث، ولكن ينفق ثمره٢٧٦٥حدثنا عبيد بن إسماعيل، حدثنا أبو أسامة، عن هشام، عن أبيه، عن عائشة…٢٧٦٦اجتنبوا السبع الموبقات٢٧٦٧وقال لنا سليمان حدثنا حماد، عن أيوب، عن نافع، قال ما رد ابن عمر على…٢٧٦٨فأخذ أبو طلحة بيدي، فانطلق بي إلى رسول الله صلى الله عليه وسلم فقال…٢٧٦٩بخ، ذلك مال رابح أو رايح شك ابن مسلمة وقد سمعت ما قلت، وإني أرى أن…٢٧٧٠قال لرسول الله صلى الله عليه وسلم إن أمه توفيت أينفعها إن تصدقت عنها…٢٧٧١يا بني النجار ثامنوني بحائطكم هذا٢٧٧٢إن شئت حبست أصلها، وتصدقت بها٢٧٧٣إن شئت تصدقت بها٢٧٧٤يا بني النجار ثامنوني بحائطكم هذا٢٧٧٥لا تبتعها، ولا ترجعن في صدقتك٢٧٧٦لا يقتسم ورثتي دينارا، ما تركت بعد نفقة نسائي ومئونة عاملي فهو صدقة٢٧٧٧اشترط في وقفه أن يأكل من وليه ويوكل صديقه غير متمول مالا .٢٧٧٨من حفر رومة فله الجنة٢٧٧٩يا بني النجار ثامنوني بحائطكم٢٧٨٠فلما قدما بتركته فقدوا جاما من فضة مخوصا من ذهب، فأحلفهما رسول الله…٢٧٨١اذهب فبيدر كل تمر على ناحيته
