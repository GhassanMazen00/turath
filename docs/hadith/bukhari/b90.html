كتاب الحيل

٢٨ حديثًا
٦٩٥٣يا أيها الناس إنما الأعمال بالنية وإنما لامرئ ما نوى، فمن كانت هجرته…٦٩٥٤لا يقبل الله صلاة أحدكم إذا أحدث حتى يتوضأ٦٩٥٥ولا يجمع بين متفرق ولا يفرق بين مجتمع خشية الصدقة٦٩٥٦الصلوات الخمس، إلا أن تطوع شيئا٦٩٥٨يكون كنز أحدكم يوم القيامة شجاعا أقرع، يفر منه صاحبه فيطلبه ويقول أنا…٦٩٥٩أنه قال استفتى سعد بن عبادة الأنصاري رسول الله صلى الله عليه وسلم في…٦٩٦٠وينكح أخت الرجل وينكحه أخته بغير صداق . وقال بعض الناس إن احتال حتى…٦٩٦١وعبد الله، ابنى محمد بن علي عن أبيهما، أن عليا رضى الله عنه قيل له إن…٦٩٦٢لا يمنع فضل الماء ليمنع به فضل الكلإ٦٩٦٣حدثنا قتيبة بن سعيد، عن مالك، عن نافع، عن ابن عمر، أن رسول الله صلى…٦٩٦٤إذا بايعت فقل لا خلابة٦٩٦٥قال كان عروة يحدث أنه سأل عائشة { وإن خفتم أن لا تقسطوا في اليتامى…٦٩٦٦لكل غادر لواء يوم القيامة يعرف به٦٩٦٧إنما أنا بشر، وإنكم تختصمون، ولعل بعضكم أن يكون ألحن بحجته من بعض،…٦٩٦٨لا تنكح البكر حتى تستأذن، ولا الثيب حتى تستأمر٦٩٦٩أن امرأة، من ولد جعفر تخوفت أن يزوجها وليها وهى كارهة فأرسلت إلى…٦٩٧٠لا تنكح الأيم حتى تستأمر، ولا تنكح البكر حتى تستأذن٦٩٧١فأبت فاحتال فجاء بشاهدى زور على أنه تزوجها، فأدركت فرضيت اليتيمة،…٦٩٧٢قالت كان رسول الله صلى الله عليه وسلم يحب الحلواء، ويحب العسل، وكان…٦٩٧٣إذا سمعتم بأرض فلا تقدموا عليه، وإذا وقع بأرض وأنتم بها فلا تخرجوا…٦٩٧٤رجز أو عذاب عذب به بعض الأمم، ثم بقي منه بقية، فيذهب المرة ويأتي…٦٩٧٥العائد في هبته كالكلب يعود في قيئه، ليس لنا مثل السوء٦٩٧٦قال إنما جعل النبي صلى الله عليه وسلم الشفعة في كل ما لم يقسم، فإذا…٦٩٧٧الجار أحق بصقبه٦٩٧٨الجار أحق بصقبه٦٩٧٩فهلا جلست في بيت أبيك وأمك، حتى تأتيك هديتك إن كنت صادقا٦٩٨٠الجار أحق بصقبه٦٩٨١الجار أحق بصقبه
