كتاب الإجارة

٢٥ حديثًا
٢٢٦٠الخازن الأمين الذي يؤدي ما أمر به طيبة نفسه أحد المتصدقين٢٢٦١لن أو لا نستعمل على عملنا من أراده٢٢٦٢ما بعث الله نبيا إلا رعى الغنم٢٢٦٣وهو على دين كفار قريش، فأمناه فدفعا إليه راحلتيهما، ووعداه غار ثور…٢٢٦٤أن عائشة رضى الله عنها زوج النبي صلى الله عليه وسلم قالت واستأجر رسول…٢٢٦٦أفيدع إصبعه في فيك تقضمها قال أحسبه قال كما يقضم الفحل٢٢٦٧فانطلقا فوجدا جدارا يريد أن ينقض٢٢٦٨مثلكم ومثل أهل الكتابين كمثل رجل استأجر أجراء فقال من يعمل لي من غدوة…٢٢٦٩إنما مثلكم واليهود والنصارى كرجل استعمل عمالا فقال من يعمل لي إلى نصف…٢٢٧٠قال الله تعالى ثلاثة أنا خصمهم يوم القيامة رجل أعطى بي ثم غدر، ورجل…٢٢٧١مثل المسلمين واليهود والنصارى كمثل رجل استأجر قوما يعملون له عملا…٢٢٧٢انطلق ثلاثة رهط ممن كان قبلكم حتى أووا المبيت إلى غار فدخلوه، فانحدرت…٢٢٧٣وإن لبعضهم لمائة ألف، قال ما نراه إلا نفسه .٢٢٧٤ولا يبيع حاضر لباد . قلت يا ابن عباس ما قوله لا يبيع حاضر لباد قال لا…٢٢٧٥حدثنا عمر بن حفص، حدثنا أبي، حدثنا الأعمش، عن مسلم، عن مسروق، حدثنا…٢٢٧٦وما يدريك أنها رقية ثم قال قد أصبتم اقسموا واضربوا لي معكم سهما٢٢٧٧فأمر له بصاع أو صاعين من طعام، وكلم مواليه فخفف عن غلته أو ضريبته .٢٢٧٨وأعطى الحجام أجره .٢٢٧٩ولو علم كراهية لم يعطه .٢٢٨٠ولم يكن يظلم أحدا أجره .٢٢٨١وأمر له بصاع أو صاعين، أو مد أو مدين، وكلم فيه فخفف من ضريبته .٢٢٨٢حدثنا قتيبة بن سعيد، عن مالك، عن ابن شهاب، عن أبي بكر بن عبد الرحمن…٢٢٨٣حدثنا مسلم بن إبراهيم، حدثنا شعبة، عن محمد بن جحادة، عن أبي حازم، عن…٢٢٨٤وإسماعيل بن إبراهيم، عن علي بن الحكم، عن نافع، عن ابن عمر رضى الله…٢٢٨٦حدثنا موسى بن إسماعيل، حدثنا جويرية بن أسماء، عن نافع، عن عبد الله…
