٤٨٨ حديثًا•صفحة ١ من ٤
٣٩٤٩كنت إلى جنب زيد بن أرقم، فقيل له كم غزا النبي صلى الله عليه وسلم من…٣٩٥٠أنه قال كان صديقا لأمية بن خلف، وكان أمية إذا مر بالمدينة نزل على…٣٩٥١أن عبد الله بن كعب، قال سمعت كعب بن مالك رضى الله عنه يقول لم أتخلف…٣٩٥٢يقول شهدت من المقداد بن الأسود مشهدا، لأن أكون صاحبه أحب إلى مما عدل…٣٩٥٣اللهم أنشدك عهدك ووعدك، اللهم إن شئت لم تعبد٣٩٥٤أن ابن جريج، أخبرهم قال أخبرني عبد الكريم، أنه سمع مقسما، مولى عبد…٣٩٥٥عمر .٣٩٥٦وكان المهاجرون يوم بدر نيفا على ستين، والأنصار نيفا وأربعين ومائتين .٣٩٥٧محمد صلى الله عليه وسلم ممن شهد بدرا أنهم كانوا عدة أصحاب طالوت الذين…٣٩٥٨ولم يجاوز معه إلا مؤمن، بضعة عشر وثلاثمائة .٣٩٥٩بعدة أصحاب طالوت الذين جاوزوا معه النهر، وما جاوز معه إلا مؤمن .٣٩٦٠على شيبة بن ربيعة، وعتبة بن ربيعة والوليد بن عتبة، وأبي جهل بن هشام .…٣٩٦١فقال أبو جهل هل أعمد من رجل قتلتموه٣٩٦٢من ينظر ما صنع أبو جهل٣٩٦٣من ينظر ما فعل أبو جهل٣٩٦٣.٢نحوه .٣٩٦٤في بدر . يعني حديث ابنى عفراء .٣٩٦٥من يجثو بين يدى الرحمن للخصومة يوم القيامة . وقال قيس بن عباد وفيهم…٣٩٦٦حدثنا قبيصة، حدثنا سفيان، عن أبي هاشم، عن أبي مجلز، عن قيس بن عباد،…٣٩٦٧حدثنا إسحاق بن إبراهيم الصواف، حدثنا يوسف بن يعقوب كان ينزل في بني…٣٩٦٨حدثنا يحيى بن جعفر، أخبرنا وكيع، عن سفيان، عن أبي هاشم، عن أبي مجلز،…٣٩٦٩يقسم قسما إن هذه الآية { هذان خصمان اختصموا في ربهم } نزلت في الذين…٣٩٧٠قال أشهد علي بدرا قال بارز وظاهر .٣٩٧١قال كاتبت أمية بن خلف، فلما كان يوم بدر، فذكر قتله وقتل ابنه، فقال…٣٩٧٢وسجد من معه، غير أن شيخا أخذ كفا من تراب فرفعه إلى جبهته فقال يكفيني…٣٩٧٣قال كان في الزبير ثلاث ضربات بالسيف، إحداهن في عاتقه، قال إن كنت…٣٩٧٤قال كان سيف الزبير محلى بفضة . قال هشام وكان سيف عروة محلى بفضة .٣٩٧٥أن أصحاب، رسول الله صلى الله عليه وسلم قالوا للزبير يوم اليرموك ألا…٣٩٧٦يا فلان بن فلان، ويا فلان بن فلان، أيسركم أنكم أطعتم الله ورسوله فإنا…٣٩٧٧حدثنا الحميدي، حدثنا سفيان، حدثنا عمرو، عن عطاء، عن ابن عباس رضى الله…٣٩٧٩إن الميت يعذب في قبره ببكاء أهله٣٩٨١إنهم الآن ليعلمون أن الذي كنت أقول لهم هو الحق٣٩٨٢ويحك أوهبلت أوجنة واحدة هي إنها جنان كثيرة، وإنه في جنة الفردوس٣٩٨٣انطلقوا حتى تأتوا روضة خاخ، فإن بها امرأة من المشركين، معها كتاب من…٣٩٨٤إذا أكثبوكم فارموهم واستبقوا نبلكم٣٩٨٥إذا أكثبوكم يعني كثروكم فارموهم، واستبقوا نبلكم٣٩٨٦وكان النبي صلى الله عليه وسلم وأصحابه أصابوا من المشركين يوم بدر…٣٩٨٧وإذا الخير ما جاء الله به من الخير بعد، وثواب الصدق الذي آتانا بعد…٣٩٨٨قال قال عبد الرحمن بن عوف إني لفي الصف يوم بدر إذ التفت، فإذا عن…٣٩٨٩وأمر عليهم عاصم بن ثابت الأنصاري، جد عاصم بن عمر بن الخطاب، حتى إذا…٣٩٩٠أن ابن عمر رضى الله عنهما ذكر له أن سعيد بن زيد بن عمرو بن نفيل وكان…٣٩٩١أن أباه، كتب إلى عمر بن عبد الله بن الأرقم الزهري، يأمره أن يدخل، على…٣٩٩٢ما تعدون أهل بدر فيكم قال من أفضل المسلمين أو كلمة نحوها قال وكذلك من…٣٩٩٣، وكان، رفاعة من أهل بدر، وكان رافع من أهل العقبة، فكان يقول لابنه ما…٣٩٩٤سمع معاذ بن رفاعة، أن ملكا، سأل النبي صلى الله عليه وسلم . وعن يحيى،…٣٩٩٥هذا جبريل آخذ برأس فرسه عليه أداة الحرب٣٩٩٦وكان بدريا .٣٩٩٧أن أبا سعيد بن مالك الخدري رضى الله عنه قدم من سفر، فقدم إليه أهله…٣٩٩٨قال قال الزبير لقيت يوم بدر عبيدة بن سعيد بن العاص وهو مدجج لا يرى…٣٩٩٩عائذ الله بن عبد الله أن عبادة بن الصامت، وكان، شهد بدرا أن رسول الله…٤٠٠٠وأنكحه بنت أخيه هند بنت الوليد بن عتبة وهو مولى لامرأة من الأنصار كما…٤٠٠١لا تقولي هكذا، وقولي ما كنت تقولين٤٠٠٢لا تدخل الملائكة بيتا فيه كلب ولا صورة٤٠٠٣أن حسين بن علي عليهم السلام أخبره أن عليا قال كانت لي شارف من نصيبي…٤٠٠٤أن عليا رضى الله عنه كبر على سهل بن حنيف فقال إنه شهد بدرا .٤٠٠٥أنه سمع عبد الله بن عمر رضى الله عنهما يحدث أن عمر بن الخطاب حين…٤٠٠٦نفقة الرجل على أهله صدقة٤٠٠٧يحدث عمر بن عبد العزيز في إمارته أخر المغيرة بن شعبة العصر وهو أمير…٤٠٠٨الآيتان من آخر سورة البقرة من قرأهما في ليلة كفتاه٤٠٠٩أن عتبان بن مالك، وكان، من أصحاب النبي صلى الله عليه وسلم ممن شهد…٤٠١٠فصدقة .٤٠١١وكان، من أكبر بني عدي وكان أبوه شهد بدرا مع النبي صلى الله عليه وسلم…٤٠١٣أن سالم بن عبد الله، أخبره قال أخبر رافع بن خديج عبد الله بن عمر، أن…٤٠١٤قال رأيت رفاعة بن رافع الأنصاري، وكان شهد بدرا .٤٠١٥أظنكم سمعتم أن أبا عبيدة قدم بشىء٤٠١٧أن ابن عمر رضى الله عنهما كان يقتل الحيات كلها . حتى حدثه أبو لبابة…٤٠١٨والله لا تذرون منه درهما٤٠١٩ثم الجندعي أن عبيد، الله بن عدي بن الخيار أخبره أن المقداد بن عمرو…٤٠٢٠من ينظر ما صنع أبو جهل٤٠٢١حدثنا موسى، حدثنا عبد الواحد، حدثنا معمر، عن الزهري، عن عبيد الله بن…٤٠٢٢سمع محمد بن فضيل، عن إسماعيل، عن قيس، كان عطاء البدريين خمسة آلاف…٤٠٢٣وذلك أول ما وقر الإيمان في قلبي .٤٠٢٤لو كان المطعم بن عدي حيا ثم كلمني في هؤلاء النتنى لتركتهم له٤٠٢٥وسعيد بن المسيب، وعلقمة بن وقاص، وعبيد الله بن عبد الله، عن حديث،…٤٠٢٦هل وجدتم ما وعدكم ربكم حقا٤٠٢٧قال ضربت يوم بدر للمهاجرين بمائة سهم .٤٠٢٨فأجلى بني النضير، وأقر قريظة ومن عليهم، حتى حاربت قريظة فقتل رجالهم…٤٠٢٩حدثني الحسن بن مدرك، حدثنا يحيى بن حماد، أخبرنا أبو عوانة، عن أبي…٤٠٣٠فكان بعد ذلك يرد عليهم .٤٠٣١حدثنا آدم، حدثنا الليث، عن نافع، عن ابن عمر رضى الله عنهما قال حرق…٤٠٣٢حدثني إسحاق، أخبرنا حبان، أخبرنا جويرية بن أسماء، عن نافع، عن ابن عمر…٤٠٣٤لا نورث، ما تركنا صدقة٤٠٣٦لا نورث، ما تركنا صدقة، إنما يأكل آل محمد في هذا المال٤٠٣٧من لكعب بن الأشرف فإنه قد آذى الله ورسوله٤٠٣٨حدثني إسحاق بن نصر، حدثنا يحيى بن آدم، حدثنا ابن أبي زائدة، عن أبيه،…٤٠٣٩قال بعث رسول الله صلى الله عليه وسلم إلى أبي رافع اليهودي رجالا من…٤٠٤٠فانطلقوا حتى دنوا من الحصن، فقال لهم عبد الله بن عتيك امكثوا أنتم حتى…٤٠٤١هذا جبريل آخذ برأس فرسه عليه أداة الحرب٤٠٤٢إني بين أيديكم فرط، وأنا عليكم شهيد، وإن موعدكم الحوض، وإني لأنظر…٤٠٤٤لا تبرحوا، إن رأيتمونا ظهرنا عليهم فلا تبرحوا وإن رأيتموهم ظهروا…٤٠٤٥إبراهيم أن عبد الرحمن بن عوف، أتي بطعام، وكان صائما فقال قتل مصعب بن…٤٠٤٦ثم قاتل حتى قتل .٤٠٤٧غطوا بها رأسه، واجعلوا على رجله الإذخر أو قال ألقوا على رجله من الإذخر٤٠٤٨لئن أشهدني الله مع النبي صلى الله عليه وسلم ليرين الله ما أجد . فلقي…٤٠٤٩كنت أسمع رسول الله صلى الله عليه وسلم يقرأ بها، فالتمسناها فوجدناها…٤٠٥٠إنها طيبة تنفي الذنوب كما تنفي النار خبث الفضة٤٠٥١وما أحب أنها لم تنزل، والله يقول { والله وليهما }٤٠٥٢هل نكحت يا جابر٤٠٥٣اذهب فبيدر كل تمر على ناحية٤٠٥٤ومعه رجلان يقاتلان عنه، عليهما ثياب بيض، كأشد القتال، ما رأيتهما قبل…٤٠٥٥ارم فداك أبي وأمي٤٠٥٦يقول جمع لي النبي صلى الله عليه وسلم أبويه يوم أحد .٤٠٥٧أنه قال قال سعد بن أبي وقاص رضى الله عنه لقد جمع لي رسول الله صلى…٤٠٥٨حدثنا أبو نعيم، حدثنا مسعر، عن سعد، عن ابن شداد، قال سمعت عليا رضى…٤٠٥٩يا سعد ارم، فداك أبي وأمي٤٠٦١حدثنا موسى بن إسماعيل، عن معتمر، عن أبيه، قال زعم أبو عثمان أنه لم…٤٠٦٢حدثنا عبد الله بن أبي الأسود، حدثنا حاتم بن إسماعيل، عن محمد بن يوسف،…٤٠٦٣قال رأيت يد طلحة شلاء، وقى بها النبي صلى الله عليه وسلم يوم أحد .٤٠٦٤وكان أبو طلحة رجلا راميا شديد النزع، كسر يومئذ قوسين أو ثلاثا، وكان…٤٠٦٥فصرخ إبليس لعنة الله عليه أى عباد الله أخراكم . فرجعت أولاهم فاجتلدت…٤٠٦٦إن لك أجر رجل ممن شهد بدرا وسهمه٤٠٦٧وأقبلوا منهزمين، فذاك إذ يدعوهم الرسول في أخراهم .٤٠٦٨حتى سقط سيفي من يدي مرارا، يسقط وآخذه، ويسقط فآخذه .٤٠٦٩اللهم العن فلانا وفلانا وفلانا٤٠٧٠يقول كان رسول الله صلى الله عليه وسلم يدعو على صفوان بن أمية وسهيل بن…٤٠٧١وقال ثعلبة بن أبي مالك إن عمر بن الخطاب رضى الله عنه قسم مروطا بين…٤٠٧٢قال خرجت مع عبيد الله بن عدي بن الخيار، فلما قدمنا حمص قال لي عبيد…٤٠٧٣اشتد غضب الله على قوم فعلوا بنبيه يشير إلى رباعيته اشتد غضب الله على…٤٠٧٤اشتد غضب الله على قوم دموا وجه نبي الله صلى الله عليه وسلم .٤٠٧٥أنه سمع سهل بن سعد، وهو يسأل عن جرح رسول الله صلى الله عليه وسلم فقال…٤٠٧٦قال اشتد غضب الله على من قتله نبي، اشتد غضب الله على من دمى وجه رسول…٤٠٧٧من يذهب في إثرهم٤٠٧٨ويوم بئر معونة سبعون، ويوم اليمامة سبعون، قال وكان بئر معونة على عهد…٤٠٨٠أيهم أكثر أخذا للقرآن٤٠٨١رأيت في رؤياى أني هززت سيفا فانقطع صدره، فإذا هو ما أصيب من المؤمنين…٤٠٨٢غطوا بها رأسه، واجعلوا على رجليه الإذخر٤٠٨٣هذا جبل يحبنا ونحبه٤٠٨٤هذا جبل يحبنا ونحبه، اللهم إن إبراهيم حرم مكة، وإني حرمت ما بين…٤٠٨٥إني فرط لكم، وأنا شهيد عليكم، وإني لأنظر إلى حوضي الآن، وإني أعطيت…٤٠٨٦وأمر عليهم عاصم بن ثابت وهو جد عاصم بن عمر بن الخطاب فانطلقوا حتى إذا…٤٠٨٧سمع جابرا، يقول الذي قتل خبيبا هو أبو سروعة .٤٠٨٨فعرض لهم حيان من بني سليم رعل وذكوان، عند بئر يقال لها بئر معونة،…٤٠٨٩قال قنت رسول الله صلى الله عليه وسلم شهرا بعد الركوع يدعو على أحياء…٤٠٩٠وذكوان وعصية وبني لحيان استمدوا رسول الله صلى الله عليه وسلم على عدو،…٤٠٩١أن النبي صلى الله عليه وسلم بعث خاله أخ لأم سليم في سبعين راكبا، وكان…٤٠٩٢فنضحه على وجهه ورأسه، ثم قال فزت ورب الكعبة .٤٠٩٣فقال له " أقم " . فقال يا رسول الله أتطمع أن يؤذن لك، فكان رسول الله…٤٠٩٤عصية عصت الله ورسوله٤٠٩٥حدثنا يحيى بن بكير، حدثنا مالك، عن إسحاق بن عبد الله بن أبي طلحة، عن…٤٠٩٦قال كذب إنما قنت رسول الله صلى الله عليه وسلم بعد الركوع شهرا، أنه…٤٠٩٧وعرضه يوم الخندق وهو ابن خمس عشرة فأجازه .٤٠٩٨اللهم لا عيش إلا عيش الآخره، فاغفر للمهاجرين والأنصار٤٠٩٩اللهم إن العيش عيش الآخره فاغفر للأنصار والمهاجره٤١٠٠اللهم إنه لا خير إلا خير الآخره، فبارك في الأنصار والمهاجره٤١٠١فجاءوا النبي صلى الله عليه وسلم فقالوا هذه كدية عرضت في الخندق، فقال…٤١٠٢يا أهل الخندق، إن جابرا قد صنع سورا فحى هلا بكم٤١٠٣رضى الله عنها { إذ جاءوكم من فوقكم ومن أسفل منكم وإذ زاغت الأبصار }…٤١٠٤حدثنا مسلم بن إبراهيم، حدثنا شعبة، عن أبي إسحاق، عن البراء رضى الله…٤١٠٥نصرت بالصبا وأهلكت عاد بالدبور٤١٠٦يحدث قال لما كان يوم الأحزاب، وخندق رسول الله صلى الله عليه وسلم…
