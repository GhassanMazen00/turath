٣٤ حديثًا
١٠٠٥قال خرج النبي صلى الله عليه وسلم يستسقي وحول رداءه .١٠٠٦اللهم أنج عياش بن أبي ربيعة، اللهم أنج سلمة بن هشام، اللهم أنج الوليد…١٠٠٧اللهم سبع كسبع يوسف١٠٠٩إلى وجه النبي صلى الله عليه وسلم يستسقي، فما ينزل حتى يجيش كل ميزاب .…١٠١٠حدثنا الحسن بن محمد، قال حدثنا محمد بن عبد الله الأنصاري، قال حدثني…١٠١١أن النبي صلى الله عليه وسلم استسقى فقلب رداءه .١٠١٢قال عبد الله بن أبي بكر أنه سمع عباد بن تميم، يحدث أباه عن عمه عبد…١٠١٣اللهم اسقنا، اللهم اسقنا، اللهم اسقنا١٠١٤اللهم أغثنا، اللهم أغثنا، اللهم أغثنا١٠١٥اللهم حوالينا ولا علينا١٠١٦اللهم على الآكام والظراب والأودية ومنابت الشجر١٠١٧اللهم على رءوس الجبال والآكام وبطون الأودية ومنابت الشجر١٠١٨أن رجلا، شكا إلى النبي صلى الله عليه وسلم هلاك المال وجهد العيال،…١٠١٩اللهم على ظهور الجبال والآكام وبطون الأودية ومنابت الشجر١٠٢٠اللهم حوالينا ولا علينا١٠٢١قال كان النبي صلى الله عليه وسلم يخطب يوم جمعة، فقام الناس فصاحوا…١٠٢٢خرج عبد الله بن يزيد الأنصاري وخرج معه البراء بن عازب وزيد بن أرقم…١٠٢٣أن عمه وكان من أصحاب النبي صلى الله عليه وسلم أخبره أن النبي صلى الله…١٠٢٤قال خرج النبي صلى الله عليه وسلم يستسقي فتوجه إلى القبلة يدعو، وحول…١٠٢٥قال رأيت النبي صلى الله عليه وسلم يوم خرج يستسقي قال فحول إلى الناس…١٠٢٦أن النبي صلى الله عليه وسلم استسقى فصلى ركعتين، وقلب رداءه .١٠٢٧سمع عباد بن تميم، عن عمه، قال خرج النبي صلى الله عليه وسلم إلى المصلى…١٠٢٨أن عباد بن تميم، أخبره أن عبد الله بن زيد الأنصاري أخبره أن النبي صلى…١٠٢٩قال أتى رجل أعرابي من أهل البدو إلى رسول الله صلى الله عليه وسلم يوم…١٠٣٠وشريك، سمعا أنسا، عن النبي صلى الله عليه وسلم أنه رفع يديه حتى رأيت…١٠٣١وابن أبي عدي، عن سعيد، عن قتادة، عن أنس بن مالك، قال كان النبي صلى…١٠٣٢حدثنا محمد هو ابن مقاتل أبو الحسن المروزي قال أخبرنا عبد الله، قال…١٠٣٣اللهم حوالينا ولا علينا١٠٣٤أنه سمع أنسا، يقول كانت الريح الشديدة إذا هبت عرف ذلك في وجه النبي…١٠٣٥نصرت بالصبا، وأهلكت عاد بالدبور١٠٣٦لا تقوم الساعة حتى يقبض العلم، وتكثر الزلازل، ويتقارب الزمان، وتظهر…١٠٣٧وبها يطلع قرن الشيطان .١٠٣٨هل تدرون ماذا قال ربكم١٠٣٩مفتاح الغيب خمس لا يعلمها إلا الله لا يعلم أحد ما يكون في غد، ولا…
