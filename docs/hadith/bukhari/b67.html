١٨٣ حديثًا•صفحة ١ من ٢
٥٠٦٣أنتم الذين قلتم كذا وكذا أما والله إني لأخشاكم لله وأتقاكم له، لكني…٥٠٦٤سمع حسان بن إبراهيم، عن يونس بن يزيد، عن الزهري، قال أخبرني عروة، أنه…٥٠٦٥يا معشر الشباب من استطاع منكم الباءة فليتزوج، ومن لم يستطع فعليه…٥٠٦٦يا معشر الشباب من استطاع الباءة فليتزوج، فإنه أغض للبصر، وأحصن للفرج،…٥٠٦٧أن ابن جريج، أخبرهم قال أخبرني عطاء، قال حضرنا مع ابن عباس جنازة…٥٠٦٨أن أنسا، حدثهم عن النبي صلى الله عليه وسلم .٥٠٦٩قال قال لي ابن عباس هل تزوجت قلت لا . قال فتزوج فإن خير هذه الأمة…٥٠٧٠العمل بالنية، وإنما لامرئ ما نوى، فمن كانت هجرته إلى الله ورسوله…٥٠٧١حدثنا محمد بن المثنى، حدثنا يحيى، حدثنا إسماعيل، قال حدثني قيس، عن…٥٠٧٢مهيم يا عبد الرحمن٥٠٧٣سمع سعيد بن المسيب، يقول سمعت سعد بن أبي وقاص، يقول رد رسول الله صلى…٥٠٧٤أنه سمع سعد بن أبي وقاص، يقول لقد رد ذلك يعني النبي صلى الله عليه…٥٠٧٥حدثنا قتيبة بن سعيد، حدثنا جرير، عن إسماعيل، عن قيس، قال قال عبد الله…٥٠٧٦يا أبا هريرة جف القلم بما أنت لاق، فاختص على ذلك أو ذر٥٠٧٧في الذي لم يرتع منها٥٠٧٨أريتك في المنام مرتين، إذا رجل يحملك في سرقة حرير فيقول هذه امرأتك،…٥٠٧٩فلحقني راكب من خلفي، فنخس بعيري بعنزة كانت معه، فانطلق بعيري كأجود ما…٥٠٨٠حدثنا آدم، حدثنا شعبة، حدثنا محارب، قال سمعت جابر بن عبد الله، رضى…٥٠٨١أنت أخي في دين الله وكتابه وهى لي حلال٥٠٨٢خير نساء ركبن الإبل صالحو نساء قريش، أحناه على ولد في صغره وأرعاه على…٥٠٨٣أيما رجل كانت عنده وليدة فعلمها فأحسن تعليمها، وأدبها فأحسن تأديبها،…٥٠٨٤لم يكذب إبراهيم إلا ثلاث كذبات بينما إبراهيم مر بجبار ومعه سارة فذكر…٥٠٨٥أمر بالأنطاع فألقى فيها من التمر والأقط والسمن فكانت وليمته، فقال…٥٠٨٦وشعيب بن الحبحاب، عن أنس بن مالك، أن رسول الله صلى الله عليه وسلم…٥٠٨٧وهل عندك من شىء٥٠٨٨، وكان، ممن شهد بدرا مع النبي صلى الله عليه وسلم تبنى سالما، وأنكحه…٥٠٨٩لعلك أردت الحج٥٠٩٠تنكح المرأة لأربع لمالها ولحسبها وجمالها ولدينها، فاظفر بذات الدين…٥٠٩١قال مر رجل على رسول الله صلى الله عليه وسلم فقال " ما تقولون في هذا "…٥٠٩٢أنه سأل عائشة رضى الله عنها { وإن خفتم أن لا تقسطوا في اليتامى } قالت…٥٠٩٣الشؤم في المرأة والدار والفرس٥٠٩٤إن كان الشؤم في شىء ففي الدار والمرأة والفرس٥٠٩٥إن كان في شىء ففي الفرس والمرأة والمسكن٥٠٩٦ما تركت بعدي فتنة أضر على الرجال من النساء٥٠٩٧الولاء لمن أعتق٥٠٩٨{ وإن خفتم أن لا، تقسطوا في اليتامى } . قالت اليتيمة تكون عند الرجل…٥٠٩٩أن عائشة، زوج النبي صلى الله عليه وسلم أخبرتها أن رسول الله صلى الله…٥١٠٠إنها ابنة أخي من الرضاعة٥١٠١أن زينب ابنة أبي سلمة، أخبرته أن أم حبيبة بنت أبي سفيان أخبرتها أنها،…٥١٠٢انظرن ما إخوانكن، فإنما الرضاعة من المجاعة٥١٠٣أن أفلح، أخا أبي القعيس جاء يستأذن عليها وهو عمها من الرضاعة بعد أن…٥١٠٤كيف بها وقد زعمت أنها قد أرضعتكما، دعها عنك٥١٠٥حرم من النسب سبع، ومن الصهر سبع . ثم قرأ { حرمت عليكم أمهاتكم } الآية…٥١٠٦قالت قلت يا رسول الله هل لك في بنت أبي سفيان قال " فأفعل ماذا " . قلت…٥١٠٧أن عروة بن الزبير، أخبره أن زينب ابنة أبي سلمة أخبرته أن أم حبيبة…٥١٠٨سمع جابرا، رضى الله عنه قال نهى رسول الله صلى الله عليه وسلم أن تنكح…٥١٠٩لا يجمع بين المرأة وعمتها، ولا بين المرأة وخالتها٥١١١أنه سمع أبا هريرة، يقول نهى النبي صلى الله عليه وسلم أن تنكح المرأة…٥١١٢والشغار أن يزوج الرجل ابنته على أن يزوجه الآخر ابنته، ليس بينهما صداق…٥١١٣حدثنا محمد بن سلام، حدثنا ابن فضيل، حدثنا هشام، عن أبيه، قال كانت…٥١١٤حدثنا مالك بن إسماعيل، أخبرنا ابن عيينة، أخبرنا عمرو، حدثنا جابر بن…٥١١٥أنه سمع الزهري، يقول أخبرني الحسن بن محمد بن علي، وأخوه عبد الله، عن…٥١١٦فرخص فقال له مولى له إنما ذلك في الحال الشديد وفي النساء قلة أو نحوه…٥١١٨إنه قد أذن لكم أن تستمتعوا فاستمتعوا٥١١٩أيما رجل وامرأة توافقا فعشرة ما بينهما ثلاث ليال فإن أحبا أن يتزايدا…٥١٢٠قال كنت عند أنس وعنده ابنة له، قال أنس جاءت امرأة إلى رسول الله صلى…٥١٢١أن امرأة، عرضت نفسها على النبي صلى الله عليه وسلم فقال له رجل يا رسول…٥١٢٢أنه سمع عبد الله بن عمر رضى الله عنهما يحدث أن عمر بن الخطاب حين…٥١٢٣أعلى أم سلمة لو لم أنكح أم سلمة ما حلت لي، إن أباها أخي من الرضاعة٥١٢٤{ فيما عرضتم } يقول إني أريد التزويج، ولوددت أنه تيسر لي امرأة صالحة…٥١٢٥رأيتك في المنام يجيء بك الملك في سرقة من حرير فقال لي هذه امرأتك .…٥١٢٦هل عندك من شىء٥١٢٧أن عائشة، زوج النبي صلى الله عليه وسلم أخبرته أن النكاح في الجاهلية…٥١٢٨{ وما يتلى عليكم في الكتاب في يتامى النساء اللاتي لا تؤتونهن ما كتب…٥١٢٩أن ابن عمر، أخبره أن عمر حين تأيمت حفصة بنت عمر من ابن حذافة السهمي…٥١٣٠أنها نزلت فيه قال زوجت أختا لي من رجل فطلقها، حتى إذا انقضت عدتها جاء…٥١٣١قالت هي اليتيمة تكون في حجر الرجل، قد شركته في ماله، فيرغب عنها أن…٥١٣٢فقال رجل من أصحابه زوجنيها يا رسول الله . قال " أعندك من شىء " . قال…٥١٣٣وأدخلت عليه وهى بنت تسع، ومكثت عنده تسعا .٥١٣٤أن النبي صلى الله عليه وسلم تزوجها وهى بنت ست سنين، وبنى بها وهى بنت…٥١٣٥هل عندك من شىء تصدقها٥١٣٦لا تنكح الأيم حتى تستأمر ولا تنكح البكر حتى تستأذن٥١٣٧أنها قالت يا رسول الله إن البكر تستحي . قال " رضاها صمتها " .٥١٣٨ومجمع، ابنى يزيد بن جارية عن خنساء بنت خذام الأنصارية، أن أباها،…٥١٣٩أن القاسم بن محمد، حدثه أن عبد الرحمن بن يزيد ومجمع بن يزيد حدثاه أن…٥١٤٠أنه سأل عائشة رضى الله عنها قال لها يا أمتاه { وإن خفتم أن لا تقسطوا…٥١٤١ما لي اليوم في النساء من حاجة٥١٤٢يحدث أن ابن عمر رضى الله عنهما كان يقول نهى النبي صلى الله عليه وسلم…٥١٤٤إياكم والظن، فإن الظن أكذب الحديث، ولا تجسسوا، ولا تحسسوا، ولا…٥١٤٥حدثنا أبو اليمان، أخبرنا شعيب، عن الزهري، قال أخبرني سالم بن عبد…٥١٤٦إن من البيان لسحر٥١٤٧دعي هذه، وقولي بالذي كنت تقولين٥١٤٨أن عبد الرحمن بن عوف، تزوج امرأة على وزن نواة، فرأى النبي صلى الله…٥١٤٩هل عندك من شىء٥١٥٠تزوج ولو بخاتم من حديد٥١٥١أحق ما أوفيتم من الشروط أن توفوا به ما استحللتم به الفروج٥١٥٢لا يحل لامرأة تسأل طلاق أختها لتستفرغ صحفتها، فإنما لها ما قدر لها٥١٥٣رضى الله عنه أن عبد، الرحمن بن عوف جاء إلى رسول الله صلى الله عليه…٥١٥٤قال أولم النبي صلى الله عليه وسلم بزينب فأوسع المسلمين خيرا فخرج كما…٥١٥٥أولم ولو بشاة " .٥١٥٦فإذا نسوة من الأنصار في البيت فقلن على الخير والبركة، وعلى خير طائر .٥١٥٧غزا نبي من الأنبياء فقال لقومه لا يتبعني رجل ملك بضع امرأة وهو يريد…٥١٥٨تزوج النبي صلى الله عليه وسلم عائشة وهى ابنة ست وبنى بها وهى ابنة تسع…٥١٥٩قال أقام النبي صلى الله عليه وسلم بين خيبر والمدينة ثلاثا يبنى عليه…٥١٦٠فلم يرعني إلا رسول الله صلى الله عليه وسلم ضحى .٥١٦١هل اتخذتم أنماطا٥١٦٢يا عائشة ما كان معكم لهو فإن الأنصار يعجبهم اللهو٥١٦٣فاتخذت حيسة في برمة، فأرسلت بها معي إليه، فانطلقت بها إليه فقال لي "…٥١٦٤فهلكت، فأرسل رسول الله صلى الله عليه وسلم ناسا من أصحابه في طلبها،…٥١٦٥أما لو أن أحدهم يقول حين يأتي أهله باسم الله، اللهم جنبني الشيطان،…٥١٦٦فكان أمهاتي يواظبنني على خدمة النبي صلى الله عليه وسلم فخدمته عشر…٥١٦٧حدثنا علي، حدثنا سفيان، قال حدثني حميد، أنه سمع أنسا رضى الله عنه قال…٥١٦٨قال ما أولم النبي صلى الله عليه وسلم على شىء من نسائه، ما أولم على…٥١٦٩أن رسول الله صلى الله عليه وسلم أعتق صفية، وتزوجها وجعل عتقها صداقها،…٥١٧٠يقول بنى النبي صلى الله عليه وسلم بامرأة فأرسلني فدعوت رجالا إلى…٥١٧١قال ذكر تزويج زينب ابنة جحش عند أنس فقال ما رأيت النبي صلى الله عليه…٥١٧٢صفية بنت شيبة قالت أولم النبي صلى الله عليه وسلم على بعض نسائه بمدين…٥١٧٣إذا دعي أحدكم إلى الوليمة فليأتها٥١٧٤فكوا العاني، وأجيبوا الداعي، وعودوا المريض٥١٧٥واتباع الجنازة، وتشميت العاطس، وإبرار القسم، ونصر المظلوم، وإفشاء…٥١٧٦قال دعا أبو أسيد الساعدي رسول الله صلى الله عليه وسلم في عرسه، وكانت…٥١٧٧رضى الله عنه أنه كان يقول شر الطعام طعام الوليمة يدعى لها الأغنياء،…٥١٧٨لو دعيت إلى كراع لأجبت، ولو أهدي إلى ذراع لقبلت٥١٧٩أجيبوا هذه الدعوة إذا دعيتم لها٥١٨٠اللهم أنتم من أحب الناس إلى٥١٨١ما بال هذه النمرقة٥١٨٢قال لما عرس أبو أسيد الساعدي دعا النبي صلى الله عليه وسلم وأصحابه،…٥١٨٣أن أبا أسيد الساعدي، دعا النبي صلى الله عليه وسلم لعرسه، فكانت امرأته…٥١٨٤المرأة كالضلع، إن أقمتها كسرتها، وإن استمتعت بها استمتعت بها وفيها عوج٥١٨٦من كان يؤمن بالله واليوم الآخر فلا يؤذي جاره٥١٨٧حدثنا أبو نعيم، حدثنا سفيان، عن عبد الله بن دينار، عن ابن عمر رضى…٥١٨٨كلكم راع وكلكم مسئول، فالإمام راع وهو مسئول والرجل راع على أهله وهو…٥١٨٩كنت لك كأبي زرع لأم زرع٥١٩٠قالت كان الحبش يلعبون بحرابهم، فسترني رسول الله صلى الله عليه وسلم…٥١٩١وعدل وعدلت معه بإداوة، فتبرز، ثم جاء فسكبت على يديه منها فتوضأ فقلت…٥١٩٢لا تصوم المرأة وبعلها شاهد إلا بإذنه٥١٩٣إذا دعا الرجل امرأته إلى فراشه فأبت أن تجيء لعنتها الملائكة حتى تصبح٥١٩٤إذا باتت المرأة مهاجرة فراش زوجها لعنتها الملائكة حتى ترجع٥١٩٥لا يحل للمرأة أن تصوم وزوجها شاهد إلا بإذنه، ولا تأذن في بيته إلا…٥١٩٦قمت على باب الجنة فكان عامة من دخلها المساكين، وأصحاب الجد محبوسون،…٥١٩٧إن الشمس والقمر آيتان من آيات الله لا يخسفان لموت أحد ولا لحياته،…٥١٩٨اطلعت في الجنة فرأيت أكثر أهلها الفقراء، واطلعت في النار، فرأيت أكثر…٥١٩٩يا عبد الله ألم أخبر أنك تصوم النهار وتقوم الليل٥٢٠٠كلكم راع، وكلكم مسئول عن رعيته، والأمير راع، والرجل راع على أهل بيته،…٥٢٠١إن الشهر تسع وعشرون٥٢٠٢إن الشهر يكون تسعة وعشرين يوما٥٢٠٣لا ولكن آليت منهن شهرا٥٢٠٤لا يجلد أحدكم امرأته جلد العبد، ثم يجامعها في آخر اليوم٥٢٠٥لا إنه قد لعن الموصلات٥٢٠٦لا يستكثر منها فيريد طلاقها، ويتزوج غيرها، تقول له أمسكني ولا تطلقني،…٥٢٠٧حدثنا مسدد، حدثنا يحيى بن سعيد، عن ابن جريج، عن عطاء، عن جابر، قال…٥٢٠٩سمع جابرا، رضى الله عنه قال كنا نعزل والقرآن ينزل . وعن عمرو، عن…٥٢١٠أوإنكم لتفعلون قالها ثلاثا ما من نسمة كائنة إلى يوم القيامة إلا هي…٥٢١١أن النبي صلى الله عليه وسلم كان إذا خرج أقرع بين نسائه، فطارت القرعة…٥٢١٢أن سودة بنت زمعة، وهبت، يومها لعائشة، وكان النبي صلى الله عليه وسلم…٥٢١٣وإذا تزوج الثيب أقام عندها ثلاثا .٥٢١٤وخالد، عن أبي قلابة، عن أنس، قال من السنة إذا تزوج الرجل البكر على…٥٢١٥أن أنس بن مالك، حدثهم أن نبي الله صلى الله عليه وسلم كان يطوف على…٥٢١٦فيدنو من إحداهن، فدخل على حفصة، فاحتبس أكثر ما كان يحتبس .٥٢١٧أين أنا غدا أين أنا غدا
