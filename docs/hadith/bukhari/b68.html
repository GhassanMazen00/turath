٩٥ حديثًا
٥٢٥١مره فليراجعها، ثم ليمسكها حتى تطهر ثم تحيض، ثم تطهر، ثم إن شاء أمسك…٥٢٥٢قال طلق ابن عمر امرأته وهى حائض، فذكر عمر للنبي صلى الله عليه وسلم…٥٢٥٣قال حسبت على بتطليقة .٥٢٥٤لقد عذت بعظيم، الحقي بأهلك٥٢٥٥فأنزلت في بيت في نخل في بيت أميمة بنت النعمان بن شراحيل ومعها دايتها…٥٢٥٦وأبي، أسيد قالا تزوج النبي صلى الله عليه وسلم أميمة بنت شراحيل، فلما…٥٢٥٧بهذا .٥٢٥٨يونس بن جبير قال قلت لابن عمر رجل طلق امرأته وهى حائض . فقال تعرف ابن…٥٢٥٩قد أنزل الله فيك وفي صاحبتك فاذهب فأت بها٥٢٦٠لعلك تريدين أن ترجعي إلى رفاعة، لا، حتى يذوق عسيلتك وتذوقي عسيلته٥٢٦١لا، حتى يذوق عسيلتها كما ذاق الأول٥٢٦٢حدثنا عمر بن حفص، حدثنا أبي، حدثنا الأعمش، حدثنا مسلم، عن مسروق، عن…٥٢٦٣، فقالت خيرنا النبي صلى الله عليه وسلم أفكان طلاقا قال مسروق لا أبالي…٥٢٦٤كان ابن عمر إذا سئل عمن طلق ثلاثا قال لو طلقت مرة أو مرتين فإن النبي…٥٢٦٥لا تحلين لزوجك الأول حتى يذوق الآخر عسيلتك، وتذوقي عسيلته٥٢٦٦سمع الربيع بن نافع، حدثنا معاوية، عن يحيى بن أبي كثير، عن يعلى بن…٥٢٦٧لا بل شربت عسلا عند زينب ابنة جحش ولن أعود له٥٢٦٨وكان إذا انصرف من العصر دخل على نسائه، فيدنو من إحداهن، فدخل على حفصة…٥٢٦٩إن الله تجاوز عن أمتي ما حدثت به أنفسها، ما لم تعمل أو تتكلم٥٢٧٠هل بك جنون هل أحصنت٥٢٧٢وسعيد بن المسيب، أن أبا هريرة، قال أتى رجل من أسلم رسول الله صلى الله…٥٢٧٣أتردين عليه حديقته٥٢٧٤أن أخت عبد الله بن أبى، بهذا، وقال " تردين حديقته " . قالت نعم .…٥٢٧٥فتردين عليه حديقته٥٢٧٦فتردين عليه حديقته٥٢٧٧أن جميلة، فذكر الحديث .٥٢٧٨إن بني المغيرة استأذنوا في أن ينكح علي ابنتهم، فلا آذن٥٢٧٩الولاء لمن أعتق٥٢٨٠وهمام، عن قتادة، عن عكرمة، عن ابن عباس، قال رأيته عبدا يعني زوج بريرة…٥٢٨١قال ذاك مغيث عبد بني فلان يعني زوج بريرة كأني أنظر إليه يتبعها في سكك…٥٢٨٢عبدا لبني فلان، كأني أنظر إليه يطوف وراءها في سكك المدينة .٥٢٨٣يا عباس ألا تعجب من حب مغيث بريرة، ومن بغض بريرة مغيثا٥٢٨٤اشتريها وأعتقيها، فإنما الولاء لمن أعتق٥٢٨٥أن ابن عمر، كان إذا سئل عن نكاح النصرانية، واليهودية، قال إن الله حرم…٥٢٨٦وقال، عطاء عن ابن عباس، كان المشركون على منزلتين من النبي صلى الله…٥٢٨٧كانت قريبة بنت أبي أمية عند عمر بن الخطاب فطلقها، فتزوجها معاوية بن…٥٢٨٨انطلقن فقد بايعتكن٥٢٨٩الشهر تسع وعشرون٥٢٩٠أن ابن عمر، رضى الله عنهما كان يقول في الإيلاء الذي سمى الله لا يحل…٥٢٩١إذا مضت أربعة أشهر يوقف حتى يطلق، ولا يقع عليه الطلاق حتى يطلق .…٥٢٩٢خذها، فإنما هي لك أو لأخيك أو للذئب٥٢٩٣فتح من ردم يأجوج ومأجوج مثل هذه٥٢٩٤في الجمعة ساعة لا يوافقها مسلم قائم يصلي، فسأل الله خيرا، إلا أعطاه٥٢٩٥قال عدا يهودي في عهد رسول الله صلى الله عليه وسلم على جارية، فأخذ…٥٢٩٦الفتنة من ها هنا٥٢٩٧حدثنا علي بن عبد الله، حدثنا جرير بن عبد الحميد، عن أبي إسحاق…٥٢٩٨لا يمنعن أحدا منكم نداء بلال أو قال أذانه من سحوره، فإنما ينادي أو…٥٢٩٩مثل البخيل والمنفق كمثل رجلين عليهما جبتان من حديد، من لدن ثدييهما…٥٣٠٠ألا أخبركم بخير دور الأنصار٥٣٠١بعثت أنا والساعة كهذه من هذه أو كهاتين٥٣٠٢الشهر هكذا وهكذا وهكذا٥٣٠٣الإيمان ها هنا مرتين ألا وإن القسوة وغلظ القلوب في الفدادين حيث يطلع…٥٣٠٤أنا وكافل اليتيم في الجنة هكذا٥٣٠٥أن رجلا، أتى النبي صلى الله عليه وسلم فقال يا رسول الله ولد لي غلام…٥٣٠٦حدثنا موسى بن إسماعيل، حدثنا جويرية، عن نافع، عن عبد الله رضى الله…٥٣٠٧إن الله يعلم أن أحدكما كاذب، فهل منكما تائب٥٣٠٨قد أنزل فيك وفي صاحبتك فاذهب فأت بها٥٣٠٩قد قضى الله فيك وفي امرأتك٥٣١٠ثم انصرف، فأتاه رجل من قومه يشكو إليه أنه وجد مع امرأته رجلا، فقال…٥٣١١الله يعلم أن أحدكما كاذب، فهل منكما تائب٥٣١٢حسابكما على الله أحدكما كاذب، لا سبيل لك عليها٥٣١٣أن ابن عمر رضى الله عنهما أخبره أن رسول الله صلى الله عليه وسلم فرق…٥٣١٤وفرق بينهما .٥٣١٥فانتفى من ولدها ففرق بينهما، وألحق الولد بالمرأة .٥٣١٦أنه قال ذكر المتلاعنان عند رسول الله صلى الله عليه وسلم فقال عاصم بن…٥٣١٧لا حتى تذوقي عسيلته، ويذوق عسيلتك٥٣١٨أن زينب ابنة أبي سلمة، أخبرته عن أمها أم سلمة، زوج النبي صلى الله…٥٣١٩أن ابن شهاب، كتب إليه أن عبيد الله بن عبد الله، أخبره عن أبيه، أنه…٥٣٢٠أن سبيعة الأسلمية، نفست بعد وفاة زوجها، بليال فجاءت النبي صلى الله…٥٣٢٢أنه سمعه يذكر، أن يحيى بن سعيد بن العاص، طلق بنت عبد الرحمن بن الحكم،…٥٣٢٤أنها قالت ما لفاطمة ألا تتقي الله، يعني في قولها لا سكنى ولا نفقة٥٣٢٦عابت عائشة أشد العيب وقالت إن فاطمة كانت في مكان وحش فخيف على…٥٣٢٨أن عائشة، أنكرت ذلك على فاطمة .٥٣٢٩عقرى أو حلقى إنك لحابستنا أكنت أفضت يوم النحر٥٣٣٠قال زوج معقل أخته فطلقها تطليقة .٥٣٣١أن معقل بن يسار، كانت أخته تحت رجل فطلقها، ثم خلى عنها حتى انقضت…٥٣٣٢أن ابن عمر بن الخطاب رضى الله عنهما طلق امرأة له وهى حائض تطليقة…٥٣٣٣سألت ابن عمر فقال طلق ابن عمر امرأته وهى حائض، فسأل عمر النبي صلى…٥٣٣٤لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد على ميت فوق ثلاث ليال،…٥٣٣٥لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد على ميت فوق ثلاث ليال…٥٣٣٦تقول جاءت امرأة إلى رسول الله صلى الله عليه وسلم فقالت يا رسول الله…٥٣٣٧قال حميد فقلت لزينب وما ترمي بالبعرة على رأس الحول فقالت زينب كانت…٥٣٣٨لا تكحل قد كانت إحداكن تمكث في شر أحلاسها أو شر بيتها، فإذا كان حول…٥٣٣٩لا يحل لامرأة مسلمة تؤمن بالله واليوم الآخر أن تحد فوق ثلاثة أيام،…٥٣٤٠حدثنا مسدد، حدثنا بشر، حدثنا سلمة بن علقمة، عن محمد بن سيرين، قالت أم…٥٣٤١إلا على زوج أربعة أشهر وعشرا، ولا نكتحل، ولا نطيب، ولا نلبس ثوبا…٥٣٤٢لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد فوق ثلاث، إلا على زوج،…٥٣٤٣ولا تمس طيبا إلا أدنى طهرها إذا طهرت، نبذة من قسط وأظفار٥٣٤٤{ والذين يتوفون منكم ويذرون أزواجا } قال كانت هذه العدة تعتد عند أهل…٥٣٤٥لا يحل لامرأة تؤمن بالله واليوم الآخر تحد على ميت فوق ثلاث، إلا على…٥٣٤٦وحلوان الكاهن، ومهر البغي .٥٣٤٧والمستوشمة، وآكل الربا وموكله، ونهى عن ثمن الكلب، وكسب البغي، ولعن…٥٣٤٨حدثنا علي بن الجعد، أخبرنا شعبة، عن محمد بن جحادة، عن أبي حازم، عن…٥٣٤٩الله يعلم أن أحدكما كاذب، فهل منكما تائب٥٣٥٠حسابكما على الله، أحدكما كاذب، لا سبيل لك عليها
