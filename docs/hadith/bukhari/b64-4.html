٤٨٨ حديثًا•صفحة ٤ من ٤
٤٤٢٩حدثنا يحيى بن بكير، حدثنا الليث، عن عقيل، عن ابن شهاب، عن عبيد الله…٤٤٣٠فقال ما أعلم منها إلا ما تعلم .٤٤٣١ائتوني أكتب لكم كتابا لن تضلوا بعده أبدا٤٤٣٢هلموا أكتب لكم كتابا لا تضلوا بعده٤٤٣٤فسارها بشىء، فبكت، ثم دعاها فسارها بشىء فضحكت فسألنا عن ذلك . فقالت…٤٤٣٥قالت كنت أسمع أنه لا يموت نبي حتى يخير بين الدنيا والآخرة، فسمعت…٤٤٣٦قالت لما مرض النبي صلى الله عليه وسلم المرض الذي مات فيه جعل يقول "…٤٤٣٧إنه لم يقبض نبي قط حتى يرى مقعده من الجنة ثم يحيا أو يخير٤٤٣٨ومع عبد الرحمن سواك رطب يستن به، فأبده رسول الله صلى الله عليه وسلم…٤٤٣٩أن عائشة رضى الله عنها أخبرته أن رسول الله صلى الله عليه وسلم كان إذا…٤٤٤٠اللهم اغفر لي وارحمني، وألحقني بالرفيق٤٤٤١لعن الله اليهود، اتخذوا قبور أنبيائهم مساجد٤٤٤٢هريقوا على من سبع قرب لم تحلل أوكيتهن لعلي أعهد إلى الناس٤٤٤٤لعنة الله على اليهود والنصارى، اتخذوا قبور أنبيائهم مساجد٤٤٤٥أن عائشة، قالت لقد راجعت رسول الله صلى الله عليه وسلم في ذلك، وما…٤٤٤٦قالت مات النبي صلى الله عليه وسلم وإنه لبين حاقنتي وذاقنتي، فلا أكره…٤٤٤٧فقال الناس يا أبا حسن، كيف أصبح رسول الله صلى الله عليه وسلم فقال…٤٤٤٨فنظر إليهم وهم في صفوف الصلاة . ثم تبسم يضحك، فنكص أبو بكر على عقبيه…٤٤٤٩لا إله إلا الله، إن للموت سكرات٤٤٥٠أين أنا غدا أين أنا غدا٤٤٥١في الرفيق الأعلى في الرفيق الأعلى٤٤٥٣أن عائشة، أخبرته أن أبا بكر رضى الله عنه أقبل على فرس من مسكنه بالسنح…٤٤٥٧وابن، عباس أن أبا بكر رضى الله عنه قبل النبي صلى الله عليه وسلم بعد…٤٤٥٨ألم أنهكم أن تلدوني٤٤٥٩قال ذكر عند عائشة أن النبي صلى الله عليه وسلم أوصى إلى علي، فقالت من…٤٤٦٠قال سألت عبد الله بن أبي أوفى رضى الله عنهما أوصى النبي صلى الله عليه…٤٤٦١قال ما ترك رسول الله صلى الله عليه وسلم دينارا ولا درهما ولا عبدا ولا…٤٤٦٢ليس على أبيك كرب بعد اليوم٤٤٦٣إنه لم يقبض نبي حتى يرى مقعده من الجنة، ثم يخير٤٤٦٥وابن، عباس رضى الله عنهم أن النبي صلى الله عليه وسلم لبث بمكة عشر…٤٤٦٦حدثنا عبد الله بن يوسف، حدثنا الليث، عن عقيل، عن ابن شهاب، عن عروة بن…٤٤٦٧حدثنا قبيصة، حدثنا سفيان، عن الأعمش، عن إبراهيم، عن الأسود، عن عائشة…٤٤٦٨قد بلغني أنكم قلتم في أسامة، وإنه أحب الناس إلى٤٤٦٩إن تطعنوا في إمارته فقد كنتم تطعنون في إمارة أبيه من قبل، وايم الله…٤٤٧٠حدثنا أصبغ، قال أخبرني ابن وهب، قال أخبرني عمرو، عن ابن أبي حبيب، عن…٤٤٧١قال سألت زيد بن أرقم رضى الله عنه كم غزوت مع رسول الله صلى الله عليه…٤٤٧٢حدثنا عبد الله بن رجاء، حدثنا إسرائيل، عن أبي إسحاق، حدثنا البراء رضى…٤٤٧٣قال غزا مع رسول الله صلى الله عليه وسلم ست عشرة غزوة .
