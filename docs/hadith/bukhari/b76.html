٩٣ حديثًا
٥٦٧٨ما أنزل الله داء إلا أنزل له شفاء٥٦٧٩ونخدمهم، ونرد القتلى والجرحى إلى المدينة .٥٦٨٠الشفاء في ثلاثة شربة عسل، وشرطة محجم، وكية نار، وأنهى أمتي عن الكى٥٦٨١الشفاء في ثلاثة شرطة محجم، أو شربة عسل، أو كية بنار، وأنهى أمتي عن…٥٦٨٢حدثنا علي بن عبد الله، حدثنا أبو أسامة، قال أخبرني هشام، عن أبيه، عن…٥٦٨٣إن كان في شىء من أدويتكم أو يكون في شىء من أدويتكم خير ففي شرطة محجم،…٥٦٨٤أن رجلا، أتى النبي صلى الله عليه وسلم فقال أخي يشتكي بطنه . فقال "…٥٦٨٥أن ناسا، كان بهم سقم قالوا يا رسول الله آونا وأطعمنا فلما صحوا قالوا…٥٦٨٦اجتووا في المدينة فأمرهم النبي صلى الله عليه وسلم أن يلحقوا براعيه…٥٦٨٧إن هذه الحبة السوداء شفاء من كل داء إلا من السام٥٦٨٨في الحبة السوداء شفاء من كل داء إلا السام٥٦٨٩إن التلبينة تجم فؤاد المريض، وتذهب ببعض الحزن٥٦٩٠أنها كانت تأمر بالتلبينة وتقول هو البغيض النافع .٥٦٩١حدثنا معلى بن أسد، حدثنا وهيب، عن ابن طاوس، عن أبيه، عن ابن عباس رضى…٥٦٩٣عليكم بهذا العود الهندي، فإن فيه سبعة أشفية . يستعط به من العذرة،…٥٦٩٤قال احتجم النبي صلى الله عليه وسلم وهو صائم .٥٦٩٥وعطاء، عن ابن عباس، قال احتجم النبي صلى الله عليه وسلم وهو محرم .٥٦٩٦إن أمثل ما تداويتم به الحجامة والقسط البحري٥٦٩٧وغيره، أن بكيرا، حدثه أن عاصم بن عمر بن قتادة حدثه أن جابر بن عبد…٥٦٩٩أنه سمع عبد الرحمن الأعرج، أنه سمع عبد الله ابن بحينة، يحدث أن رسول…٥٧٠١أن رسول الله صلى الله عليه وسلم احتجم وهو محرم في رأسه من شقيقة كانت…٥٧٠٢إن كان في شىء من أدويتكم خير ففي شربة عسل أو شرطة محجم أو لذعة من…٥٧٠٣هو ابن عجرة قال أتى على النبي صلى الله عليه وسلم زمن الحديبية، وأنا…٥٧٠٤إن كان في شىء من أدويتكم شفاء ففي شرطة محجم أو لذعة بنار، وما أحب أن…٥٧٠٥عرضت على الأمم، فجعل النبي والنبيان يمرون معهم الرهط، والنبي ليس معه…٥٧٠٦لقد كانت إحداكن تمكث في بيتها في شر أحلاسها أو في أحلاسها في شر بيتها…٥٧٠٧لا عدوى ولا طيرة ولا هامة ولا صفر، وفر من المجذوم كما تفر من الأسد٥٧٠٨الكمأة من المن، وماؤها شفاء للعين٥٧١٢حديث ٥٧١٢٥٧١٣على ما تدغرن أولادكن بهذا العلاق عليكن بهذا العود الهندي، فإن فيه…٥٧١٤هريقوا على من سبع قرب لم تحلل أوكيتهن، لعلي أعهد إلى الناس٥٧١٥على ما تدغرن أولادكن بهذا العلاق عليكم بهذا العود الهندي، فإن فيه…٥٧١٦حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، عن قتادة، عن أبي…٥٧١٧لا عدوى ولا صفر ولا هامة٥٧١٨اتقوا الله، على ما تدغرون أولادكم بهذه الأعلاق عليكم بهذا العود…٥٧٢١قال قرئ على أيوب من كتب أبي قلابة، منه ما حدث به ومنه ما قرئ عليه،…٥٧٢٢قال لما كسرت على رأس رسول الله صلى الله عليه وسلم البيضة، وأدمي وجهه،…٥٧٢٣الحمى من فيح جهنم فأطفئوها بالماء٥٧٢٤أن أسماء بنت أبي بكر رضى الله عنهما كانت إذا أتيت بالمرأة قد حمت تدعو…٥٧٢٥الحمى من فيح جهنم فابردوها بالماء٥٧٢٦الحمى من فوح جهنم، فابردوها بالماء٥٧٢٧أن أنس بن مالك، حدثهم أن ناسا أو رجالا من عكل وعرينة قدموا على رسول…٥٧٢٨إذا سمعتم بالطاعون بأرض فلا تدخلوها، وإذا وقع بأرض وأنتم بها فلا…٥٧٢٩إذا سمعتم به بأرض فلا تقدموا عليه، وإذا وقع بأرض وأنتم بها فلا تخرجوا…٥٧٣٠إذا سمعتم به بأرض فلا تقدموا عليه وإذا وقع بأرض وأنتم بها فلا تخرجوا…٥٧٣١لا يدخل المدينة المسيح ولا الطاعون٥٧٣٢الطاعون شهادة لكل مسلم٥٧٣٣المبطون شهيد، والمطعون شهيد٥٧٣٤أنه كان عذابا يبعثه الله على من يشاء، فجعله الله رحمة للمؤمنين، فليس…٥٧٣٥فلما ثقل كنت أنفث عليه بهن، وأمسح بيد نفسه لبركتها . فسألت الزهري كيف…٥٧٣٦وما أدراك أنها رقية، خذوها، واضربوا لي بسهم٥٧٣٧إن أحق ما أخذتم عليه أجرا كتاب الله٥٧٣٨حدثنا محمد بن كثير، أخبرنا سفيان، قال حدثني معبد بن خالد، قال سمعت…٥٧٣٩استرقوا لها، فإن بها النظرة٥٧٤٠حدثنا إسحاق بن نصر، حدثنا عبد الرزاق، عن معمر، عن همام، عن أبي هريرة…٥٧٤١من الحمة فقالت رخص النبي صلى الله عليه وسلم الرقية من كل ذي حمة .٥٧٤٢اللهم رب الناس مذهب الباس اشف أنت الشافي لا شافي إلا أنت، شفاء لا…٥٧٤٣اللهم رب الناس أذهب الباس، اشفه وأنت الشافي، لا شفاء إلا شفاؤك، شفاء…٥٧٤٤امسح الباس رب الناس، بيدك الشفاء، لا كاشف له إلا أنت٥٧٤٥بسم الله، تربة أرضنا . بريقة بعضنا، يشفى سقيمنا بإذن ربنا٥٧٤٦تربة أرضنا، وريقة بعضنا، يشفى سقيمنا، بإذن ربنا٥٧٤٧الرؤيا من الله، والحلم من الشيطان، فإذا رأى أحدكم شيئا يكرهه فلينفث…٥٧٤٨ثم يمسح بهما وجهه، وما بلغت يداه من جسده . قالت عائشة فلما اشتكى كان…٥٧٤٩وما يدريك أنها رقية أصبتم اقسموا واضربوا لي معكم بسهم٥٧٥٠أذهب الباس رب الناس، واشف أنت الشافي، لا شفاء إلا شفاؤك، شفاء لا…٥٧٥١فأمسح بيد نفسه لبركتها . فسألت ابن شهاب كيف كان ينفث قال ينفث على…٥٧٥٢عرضت على الأمم فجعل يمر النبي معه الرجل والنبي معه الرجلان، والنبي…٥٧٥٣لا عدوى ولا طيرة، والشؤم في ثلاث في المرأة، والدار، والدابة٥٧٥٤لا طيرة، وخيرها الفأل٥٧٥٥لا طيرة، وخيرها الفأل٥٧٥٦لا عدوى ولا طيرة، ويعجبني الفأل الصالح، الكلمة الحسنة٥٧٥٧لا عدوى، ولا طيرة، ولا هامة، ولا صفر٥٧٥٨إنما هذا من إخوان الكهان٥٧٦٠إنما هذا من إخوان الكهان٥٧٦١ومهر البغي، وحلوان الكاهن .٥٧٦٢يخطفها من الجني، فيقرها في أذن وليه، فيخلطون معها مائة كذبة " . قال…٥٧٦٣يا عائشة، أشعرت أن الله أفتاني فيما استفتيته فيه، أتاني رجلان فقعد…٥٧٦٤اجتنبوا الموبقات الشرك بالله، والسحر٥٧٦٥يا عائشة أعلمت أن الله قد أفتاني فيما استفتيته فيه، أتاني رجلان فقعد…٥٧٦٦أشعرت يا عائشة أن الله قد أفتاني فيما استفتيته فيه٥٧٦٧إن من البيان لسحرا أو إن بعض البيان لسحر٥٧٦٨من اصطبح كل يوم تمرات عجوة، لم يضره سم ولا سحر ذلك اليوم إلى الليل٥٧٦٩من تصبح سبع تمرات عجوة، لم يضره ذلك اليوم سم ولا سحر٥٧٧٠لا عدوى، ولا صفر، ولا هامة٥٧٧١لا يوردن ممرض على مصح٥٧٧٢لا عدوى، ولا طيرة، إنما الشؤم في ثلاث في الفرس، والمرأة، والدار٥٧٧٥حديث ٥٧٧٥٥٧٧٦لا عدوى، ولا طيرة، ويعجبني الفأل٥٧٧٧اجمعوا لي من كان ها هنا من اليهود٥٧٧٨من تردى من جبل فقتل نفسه، فهو في نار جهنم، يتردى فيه خالدا مخلدا فيها…٥٧٧٩من اصطبح بسبع تمرات عجوة لم يضره ذلك اليوم سم ولا سحر٥٧٨١قال وسألته هل نتوضأ أو نشرب ألبان الأتن أو مرارة السبع أو أبوال الإبل…٥٧٨٢إذا وقع الذباب في إناء أحدكم، فليغمسه كله، ثم ليطرحه، فإن في أحد…
