سنن أبي داود

كتاب الحدود

١٤٣ حديثًا
٤٣٥١لا تعذبوا بعذاب الله٤٣٥٢لا يحل دم رجل مسلم يشهد أن لا إله إلا الله وأني رسول الله إلا بإحدى…٤٣٥٣لا يحل دم امرئ مسلم يشهد أن لا إله إلا الله وأن محمدا رسول الله إلا…٤٣٥٤ما تقول يا أبا موسى٤٣٥٥وبريد بن عبد الله بن أبي بردة، عن أبي بردة، عن أبي موسى، قال قدم على…٤٣٥٦حدثنا محمد بن العلاء، حدثنا حفص، حدثنا الشيباني، عن أبي بردة، بهذه…٤٣٥٧بهذه القصة قال فلم ينزل حتى ضرب عنقه وما استتابه .٤٣٥٨قال كان عبد الله بن سعد بن أبي سرح يكتب لرسول الله صلى الله عليه وسلم…٤٣٥٩أما كان فيكم رجل رشيد يقوم إلى هذا حيث رآني كففت يدي عن بيعته فيقتله٤٣٦٠إذا أبق العبد إلى الشرك فقد حل دمه٤٣٦١أنشد الله رجلا فعل ما فعل لي عليه حق إلا قام٤٣٦٢وعبد الله بن الجراح، عن جرير، عن مغيرة، عن الشعبي، عن علي، رضي الله…٤٣٦٣ونصير بن الفرج، قالا حدثنا أبو أسامة، عن يزيد بن زريع، عن يونس بن…٤٣٦٤أن قوما، من عكل - أو قال من عرينة - قدموا على رسول الله صلى الله عليه…٤٣٦٥بإسناده بهذا الحديث قال فيه فأمر بمسامير فأحميت فكحلهم وقطع أيديهم…٤٣٦٦بهذا الحديث قال فيه فبعث رسول الله صلى الله عليه وسلم في طلبهم قافة…٤٣٦٧وقتادة، وحميد، عن أنس بن مالك، ذكر هذا الحديث قال أنس فلقد رأيت أحدهم…٤٣٦٨حدثنا محمد بن بشار، حدثنا ابن أبي عدي، عن هشام، عن قتادة، عن أنس بن…٤٣٦٩حدثنا أحمد بن صالح، حدثنا عبد الله بن وهب، أخبرني عمرو، عن سعيد بن…٤٣٧٠أن رسول الله صلى الله عليه وسلم لما قطع الذين سرقوا لقاحه وسمل أعينهم…٤٣٧١قال كان هذا قبل أن تنزل الحدود يعني حديث أنس .٤٣٧٢قال { إنما جزاء الذين يحاربون الله ورسوله ويسعون في الأرض فسادا أن…٤٣٧٣يا أسامة أتشفع في حد من حدود الله٤٣٧٤ومحمد بن يحيى، قالا حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن…٤٣٧٥أقيلوا ذوي الهيئات عثراتهم إلا الحدود٤٣٧٦تعافوا الحدود فيما بينكم فما بلغني من حد فقد وجب٤٣٧٧لو سترته بثوبك كان خيرا لك٤٣٧٨أن هزالا، أمر ماعزا أن يأتي النبي صلى الله عليه وسلم فيخبره .٤٣٧٩اذهبي فقد غفر الله لك٤٣٨٠حدثنا موسى بن إسماعيل، حدثنا حماد، عن إسحاق بن عبد الله بن أبي طلحة،…٤٣٨١توضأت حين أقبلت٤٣٨٢أن قوما، من الكلاعيين سرق لهم متاع فاتهموا أناسا من الحاكة فأتوا…٤٣٨٣رضى الله عنها أن النبي صلى الله عليه وسلم كان يقطع في ربع دينار…٤٣٨٤تقطع يد السارق في ربع دينار فصاعدا٤٣٨٥أن رسول الله صلى الله عليه وسلم قطع في مجن ثمنه ثلاثة دراهم .٤٣٨٦أن نافعا، مولى عبد الله بن عمر حدثه أن عبد الله بن عمر حدثهم أن النبي…٤٣٨٧ومحمد بن أبي السري العسقلاني، - وهذا لفظه وهو أتم - قالا حدثنا ابن…٤٣٨٨لا قطع في ثمر ولا كثر٤٣٨٩بهذا الحديث قال فجلده مروان جلدات وخلى سبيله .٤٣٩٠من أصاب بفيه من ذي حاجة غير متخذ خبنة فلا شىء عليه ومن خرج بشىء منه…٤٣٩١ليس على المنتهب قطع ومن انتهب نهبة مشهورة فليس منا٤٣٩٢ليس على الخائن قطع٤٣٩٣ولا على المختلس قطع٤٣٩٤فهلا كان هذا قبل أن تأتيني به٤٣٩٥هل من امرأة تائبة إلى الله عز وجل ورسوله٤٣٩٦قال كان عروة يحدث أن عائشة رضى الله عنها قالت استعارت امرأة - تعني -…٤٣٩٧ومحمد بن يحيى، قالا حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن…٤٣٩٨رفع القلم عن ثلاثة عن النائم حتى يستيقظ وعن المبتلى حتى يبرأ وعن…٤٣٩٩حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن الأعمش، عن أبي ظبيان، عن ابن…٤٤٠٠حدثنا يوسف بن موسى، حدثنا وكيع، عن الأعمش، نحوه وقال أيضا حتى يعقل .…٤٤٠١رفع القلم عن ثلاثة عن المجنون المغلوب على عقله حتى يفيق وعن النائم…٤٤٠٢رفع القلم عن ثلاثة عن الصبي حتى يبلغ وعن النائم حتى يستيقظ وعن…٤٤٠٣رفع القلم عن ثلاثة عن النائم حتى يستيقظ وعن الصبي حتى يحتلم وعن…٤٤٠٤قال كنت من سبى بني قريظة فكانوا ينظرون فمن أنبت الشعر قتل ومن لم ينبت…٤٤٠٥بهذا الحديث قال فكشفوا عانتي فوجدوها لم تنبت فجعلوني في السبى .٤٤٠٦أن النبي صلى الله عليه وسلم عرضه يوم أحد وهو ابن أربع عشرة سنة فلم…٤٤٠٧قال قال نافع حدثت بهذا الحديث، عمر بن عبد العزيز فقال إن هذا الحد بين…٤٤٠٨لا تقطع الأيدي في السفر٤٤٠٩قال قال لي رسول الله صلى الله عليه وسلم " يا أبا ذر " . قلت لبيك يا…٤٤١٠حدثنا محمد بن عبد الله بن عبيد بن عقيل الهلالي، حدثنا جدي، عن مصعب بن…٤٤١١في العنق للسارق أمن السنة هو قال أتي رسول الله صلى الله عليه وسلم…٤٤١٢إذا سرق المملوك فبعه ولو بنش٤٤١٣قال { واللاتي يأتين الفاحشة من نسائكم فاستشهدوا عليهن أربعة منكم فإن…٤٤١٤قال السبيل الحد قال سفيان { فآذوهما } البكران { فأمسكوهن في البيوت }…٤٤١٥خذوا عني خذوا عني قد جعل الله لهن سبيلا الثيب بالثيب جلد مائة ورمى…٤٤١٦جلد مائة والرجم٤٤١٧كفى بالسيف شاهدا٤٤١٨أن عمر، - يعني ابن الخطاب - رضى الله عنه خطب فقال إن الله بعث محمدا…٤٤١٩إنك قد قلتها أربع مرات فبمن٤٤٢٠حدثنا عبيد الله بن عمر بن ميسرة، حدثنا يزيد بن زريع، عن محمد بن…٤٤٢١أن ماعز بن مالك، أتى النبي صلى الله عليه وسلم فقال إنه زنى . فأعرض…٤٤٢٢حدثنا مسدد، حدثنا أبو عوانة، عن سماك، عن جابر بن سمرة، قال رأيت ماعز…٤٤٢٣بهذا الحديث والأول أتم قال فرده مرتين . قال سماك فحدثت به سعيد بن…٤٤٢٤حدثنا عبد الغني بن أبي عقيل المصري، حدثنا خالد، - يعني ابن عبد الرحمن…٤٤٢٥أحق ما بلغني عنك٤٤٢٦شهدت على نفسك أربع مرات اذهبوا به فارجموه٤٤٢٧لعلك قبلت أو غمزت أو نظرت٤٤٢٨أن عبد الرحمن بن الصامت ابن عم أبي هريرة، أخبره أنه، سمع أبا هريرة،…٤٤٢٩بنحوه زاد واختلفوا فقال بعضهم ربط إلى شجرة وقال بعضهم وقف .٤٤٣٠والحسن بن علي، قالا حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن أبي…٤٤٣١حدثنا أبو كامل، حدثنا يزيد، - يعني ابن زريع - ح وحدثنا أحمد بن منيع،…٤٤٣٢هو رجل أصاب ذنبا حسيبه الله٤٤٣٣أن النبي صلى الله عليه وسلم استنكه ماعزا .٤٤٣٤حدثنا أحمد بن إسحاق الأهوازي، حدثنا أبو أحمد، حدثنا بشير بن المهاجر،…٤٤٣٥ومحمد بن داود بن صبيح، قال عبدة أخبرنا حرمي بن حفص، قال حدثنا محمد بن…٤٤٣٦حدثنا هشام بن عمار، حدثنا صدقة بن خالد، ح وحدثنا نصر بن عاصم…٤٤٣٧حدثنا عثمان بن أبي شيبة، حدثنا طلق بن غنام، حدثنا عبد السلام بن حفص،…٤٤٣٨أن رجلا، زنى بامرأة فأمر به النبي صلى الله عليه وسلم فجلد الحد ثم…٤٤٣٩أن رجلا، زنى بامرأة فلم يعلم بإحصانه فجلد ثم علم بإحصانه فرجم .٤٤٤٠أحسن إليها فإذا وضعت فجئ بها٤٤٤١قال فشكت عليها ثيابها . يعني فشدت .٤٤٤٢أن امرأة، - يعني من غامد - أتت النبي صلى الله عليه وسلم فقالت إني قد…٤٤٤٣حدثنا عثمان بن أبي شيبة، حدثنا وكيع بن الجراح، عن زكريا أبي عمران،…٤٤٤٤ارموا واتقوا الوجه٤٤٤٥وزيد بن خالد الجهني، أنهما أخبراه أن رجلين اختصما إلى رسول الله صلى…٤٤٤٦ما تجدون في التوراة في شأن الزنا٤٤٤٧ما حد الزاني في كتابكم٤٤٤٨هكذا تجدون حد الزاني٤٤٤٩أن زيد بن أسلم، حدثه عن ابن عمر، قال أتى نفر من يهود فدعوا رسول الله…٤٤٥٠أنشدكم بالله الذي أنزل التوراة على موسى ما تجدون في التوراة على من…٤٤٥١حدثنا عبد العزيز بن يحيى أبو الأصبغ الحراني، قال حدثني محمد، - يعني…٤٤٥٢كيف تجدان أمر هذين في التوراة٤٤٥٣والشعبي، عن النبي صلى الله عليه وسلم نحوه لم يذكر فدعا بالشهود فشهدوا…٤٤٥٤بنحو منه .٤٤٥٥أنه سمع أبا الزبير، سمع جابر بن عبد الله، يقول رجم النبي صلى الله…٤٤٥٦على إبل لي ضلت إذ أقبل ركب أو فوارس معهم لواء فجعل الأعراب يطيفون بي…٤٤٥٧قال لقيت عمي ومعه راية فقلت له أين تريد قال بعثني رسول الله صلى الله…٤٤٥٨أن رجلا، يقال له عبد الرحمن بن حنين وقع على جارية امرأته فرفع إلى…٤٤٥٩إن كانت أحلتها له جلد مائة وإن لم تكن أحلتها له رجمته٤٤٦٠حدثنا أحمد بن صالح، حدثنا عبد الرزاق، أخبرنا معمر، عن قتادة، عن…٤٤٦١حدثنا علي بن الحسن الدرهمي، حدثنا عبد الأعلى، عن سعيد، عن قتادة، عن…٤٤٦٢من وجدتموه يعمل عمل قوم لوط فاقتلوا الفاعل والمفعول به٤٤٦٣ومجاهدا، يحدثان عن ابن عباس، في البكر يوجد على اللوطية قال يرجم . قال…٤٤٦٤من أتى بهيمة فاقتلوه واقتلوها معه٤٤٦٥أن شريكا، وأبا الأحوص، وأبا، بكر بن عياش حدثوهم عن عاصم، عن أبي رزين،…٤٤٦٦حدثنا عثمان بن أبي شيبة، حدثنا طلق بن غنام، حدثنا عبد السلام بن حفص،…٤٤٦٧أن رجلا، من بكر بن ليث أتى النبي صلى الله عليه وسلم فأقر أنه زنى…٤٤٦٨والأسود، قالا قال عبد الله جاء رجل إلى النبي صلى الله عليه وسلم فقال…٤٤٦٩إن زنت فاجلدوها ثم إن زنت فاجلدوها ثم إن زنت فاجلدوها ثم إن زنت…٤٤٧٠إذا زنت أمة أحدكم فليحدها ولا يعيرها ثلاث مرار فإن عادت في الرابعة…٤٤٧١فليضربها كتاب الله ولا يثرب عليها٤٤٧٢أنه أخبره بعض، أصحاب رسول الله صلى الله عليه وسلم من الأنصار أنه…٤٤٧٣يا علي انطلق فأقم عليها الحد٤٤٧٤ومالك بن عبد الواحد المسمعي، - وهذا حديثه - أن ابن أبي عدي، حدثهم عن…٤٤٧٥بهذا الحديث لم يذكر عائشة قال فأمر برجلين وامرأة ممن تكلم بالفاحشة…٤٤٧٦ومحمد بن المثنى، - وهذا حديثه - قالا حدثنا أبو عاصم، عن ابن جريج، عن…٤٤٧٧حدثنا قتيبة بن سعيد، حدثنا أبو ضمرة، عن يزيد بن الهاد، عن محمد بن…٤٤٧٨وحيوة بن شريح، وابن، لهيعة عن ابن الهاد، بإسناده ومعناه قال فيه بعد…٤٤٧٩حدثنا مسلم بن إبراهيم، حدثنا هشام، ح وحدثنا مسدد، حدثنا يحيى، عن…٤٤٨٠وموسى بن إسماعيل، - المعنى - قالا حدثنا عبد العزيز بن المختار، حدثنا…٤٤٨١رضى الله عنه قال جلد رسول الله صلى الله عليه وسلم في الخمر وأبو بكر…٤٤٨٢إذا شربوا الخمر فاجلدوهم ثم إن شربوا فاجلدوهم ثم إن شربوا فاجلدوهم ثم…٤٤٨٣إن شربها فاقتلوه٤٤٨٤إذا سكر فاجلدوه ثم إن سكر فاجلدوه ثم إن سكر فاجلدوه فإن عاد الرابعة…٤٤٨٥من شرب الخمر فاجلدوه فإن عاد فاجلدوه فإن عاد فاجلدوه فإن عاد في…٤٤٨٦رضى الله عنه قال لا أدي - أو ما كنت لأدي - من أقمت عليه حدا إلا شارب…٤٤٨٧أن ابن شهاب، حدثه عن عبد الرحمن بن أزهر، قال كأني أنظر إلى رسول الله…٤٤٨٨حدثنا ابن السرح، قال وجدت في كتاب خالي عبد الرحمن بن عبد الحميد عن…٤٤٨٩حدثنا الحسن بن علي، حدثنا عثمان بن عمر، حدثنا أسامة بن زيد، عن…٤٤٩٠أنه قال نهى رسول الله صلى الله عليه وسلم أن يستقاد في المسجد وأن تنشد…٤٤٩١لا يجلد فوق عشر جلدات إلا في حد من حدود الله عز وجل٤٤٩٢أن بكير بن الأشج، حدثه عن سليمان بن يسار، قال حدثني عبد الرحمن بن…٤٤٩٣إذا ضرب أحدكم فليتق الوجه
