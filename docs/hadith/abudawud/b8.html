كتاب الوتر

١٤٠ حديثًا
١٤١٦يا أهل القرآن أوتروا فإن الله وتر يحب الوتر١٤١٧ليس لك ولا لأصحابك١٤١٨إن الله عز وجل قد أمدكم بصلاة وهي خير لكم من حمر النعم وهي الوتر…١٤١٩الوتر حق فمن لم يوتر فليس منا الوتر حق فمن لم يوتر فليس منا الوتر حق…١٤٢٠خمس صلوات كتبهن الله على العباد فمن جاء بهن لم يضيع منهن شيئا…١٤٢١أن رجلا، من أهل البادية سأل النبي صلى الله عليه وسلم عن صلاة الليل…١٤٢٢الوتر حق على كل مسلم فمن أحب أن يوتر بخمس فليفعل ومن أحب أن يوتر…١٤٢٣وزبيد، عن سعيد بن عبد الرحمن بن أبزى، عن أبيه، عن أبى بن كعب، قال كان…١٤٢٤قال سألت عائشة أم المؤمنين بأى شىء كان يوتر رسول الله صلى الله عليه…١٤٢٥اللهم اهدني فيمن هديت وعافني فيمن عافيت وتولني فيمن توليت وبارك لي…١٤٢٦بإسناده ومعناه قال في آخره قال هذا يقول في الوتر في القنوت ولم يذكر…١٤٢٧اللهم إني أعوذ برضاك من سخطك وبمعافاتك من عقوبتك وأعوذ بك منك لا أحصي…١٤٢٨أصحابه أن أبى بن كعب، أمهم - يعني في رمضان - وكان يقنت في النصف الآخر…١٤٢٩أن عمر بن الخطاب، جمع الناس على أبى بن كعب فكان يصلي لهم عشرين ليلة…١٤٣٠سبحان الملك القدوس١٤٣١من نام عن وتره أو نسيه فليصله إذا ذكره١٤٣٢قال أوصاني خليلي صلى الله عليه وسلم بثلاث لا أدعهن في سفر ولا حضر…١٤٣٣أوصاني خليلي صلى الله عليه وسلم بثلاث لا أدعهن لشىء أوصاني بصيام…١٤٣٤أن النبي صلى الله عليه وسلم قال لأبي بكر " متى توتر " قال أوتر من أول…١٤٣٥قال قلت لعائشة متى كان يوتر رسول الله صلى الله عليه وسلم قالت كل ذلك…١٤٣٦بادروا الصبح بالوتر١٤٣٧رسول الله صلى الله عليه وسلم قالت ربما أوتر أول الليل وربما أوتر من…١٤٣٨اجعلوا آخر صلاتكم بالليل وترا١٤٣٩لا وتران في ليلة١٤٤٠حدثنا داود بن أمية، حدثنا معاذ، - يعني ابن هشام - حدثني أبي، عن يحيى…١٤٤١ومسلم بن إبراهيم، وحفص بن عمر، ح وحدثنا ابن معاذ، حدثني أبي قالوا،…١٤٤٢اللهم نج الوليد بن الوليد اللهم نج سلمة بن هشام اللهم نج المستضعفين…١٤٤٣سمع الله لمن حمده١٤٤٤ومسدد، قالا حدثنا حماد، عن أيوب، عن محمد، عن أنس بن مالك، أنه سئل هل…١٤٤٥أن النبي صلى الله عليه وسلم قنت شهرا ثم تركه .١٤٤٦صلى مع النبي صلى الله عليه وسلم صلاة الغداة فلما رفع رأسه من الركعة…١٤٤٧يا أيها الناس ما زال بكم صنيعكم حتى ظننت أن ستكتب عليكم فعليكم…١٤٤٨اجعلوا في بيوتكم من صلاتكم ولا تتخذوها قبورا١٤٤٩أن النبي صلى الله عليه وسلم سئل أى الأعمال أفضل قال " طول القيام " .…١٤٥٠رحم الله رجلا قام من الليل فصلى وأيقظ امرأته فصلت فإن أبت نضح في…١٤٥١من استيقظ من الليل وأيقظ امرأته فصليا ركعتين جميعا كتبا من الذاكرين…١٤٥٢خيركم من تعلم القرآن وعلمه١٤٥٣من قرأ القرآن وعمل بما فيه ألبس والداه تاجا يوم القيامة ضوؤه أحسن من…١٤٥٤الذي يقرأ القرآن وهو ماهر به مع السفرة الكرام البررة والذي يقرؤه وهو…١٤٥٥ما اجتمع قوم في بيت من بيوت الله تعالى يتلون كتاب الله ويتدارسونه…١٤٥٦أيكم يحب أن يغدو إلى بطحان أو العقيق فيأخذ ناقتين كوماوين زهراوين…١٤٥٧{ الحمد لله رب العالمين } أم القرآن وأم الكتاب والسبع المثاني١٤٥٨ما منعك أن تجيبني١٤٥٩قال أوتي رسول الله صلى الله عليه وسلم سبعا من المثاني الطول وأوتي…١٤٦٠أبا المنذر أى آية معك من كتاب الله أعظم١٤٦١والذي نفسي بيده إنها لتعدل ثلث القرآن١٤٦٢يا عقبة ألا أعلمك خير سورتين قرئتا١٤٦٣يا عقبة تعوذ بهما فما تعوذ متعوذ بمثلهما١٤٦٤يقال لصاحب القرآن اقرأ وارتق ورتل كما كنت ترتل في الدنيا فإن منزلك…١٤٦٥صلى الله عليه وسلم فقال كان يمد مدا .١٤٦٦رسول الله صلى الله عليه وسلم وصلاته فقالت وما لكم وصلاته كان يصلي…١٤٦٧قال رأيت رسول الله صلى الله عليه وسلم يوم فتح مكة وهو على ناقة يقرأ…١٤٦٨زينوا القرآن بأصواتكم١٤٦٩ليس منا من لم يتغن بالقرآن١٤٧٠قال قال رسول الله صلى الله عليه وسلم مثله .١٤٧١ليس منا من لم يتغن بالقرآن١٤٧٢قال قال وكيع وابن عيينة يعني يستغني به .١٤٧٣ما أذن الله لشىء ما أذن لنبي حسن الصوت يتغنى بالقرآن يجهر به١٤٧٤ما من امرئ يقرأ القرآن ينساه إلا لقي الله عز وجل يوم القيامة أجذم١٤٧٥حدثنا القعنبي، عن مالك، عن ابن شهاب، عن عروة بن الزبير، عن عبد الرحمن…١٤٧٦قال قال الزهري إنما هذه الأحرف في الأمر الواحد ليس تختلف في حلال ولا…١٤٧٧يا أبى إني أقرئت القرآن فقيل لي على حرف أو حرفين فقال الملك الذي معي…١٤٧٨أسأل الله معافاته ومغفرته إن أمتي لا تطيق ذلك١٤٧٩الدعاء هو العبادة { قال ربكم ادعوني أستجب لكم }١٤٨٠سيكون قوم يعتدون في الدعاء١٤٨١حميد بن هانئ أن أبا علي، عمرو بن مالك حدثه أنه، سمع فضالة بن عبيد،…١٤٨٢رضى الله عنها قالت كان رسول الله صلى الله عليه وسلم يستحب الجوامع من…١٤٨٣لا يقولن أحدكم اللهم اغفر لي إن شئت اللهم ارحمني إن شئت ليعزم المسألة…١٤٨٤يستجاب لأحدكم ما لم يعجل فيقول قد دعوت فلم يستجب لي١٤٨٥لا تستروا الجدر من نظر في كتاب أخيه بغير إذنه فإنما ينظر في النار…١٤٨٦إذا سألتم الله فاسألوه ببطون أكفكم ولا تسألوه بظهورها١٤٨٧قال رأيت رسول الله صلى الله عليه وسلم يدعو هكذا بباطن كفيه وظاهرهما .١٤٨٨إن ربكم تبارك وتعالى حيي كريم يستحيي من عبده إذا رفع يديه إليه أن…١٤٨٩قال المسألة أن ترفع، يديك حذو منكبيك أو نحوهما والاستغفار أن تشير…١٤٩٠بهذا الحديث قال فيه والابتهال هكذا ورفع يديه وجعل ظهورهما مما يلي…١٤٩١أن رسول الله صلى الله عليه وسلم قال فذكر نحوه .١٤٩٢أن النبي صلى الله عليه وسلم كان إذا دعا فرفع يديه مسح وجهه بيديه .١٤٩٣لقد سألت الله بالاسم الذي إذا سئل به أعطى وإذا دعي به أجاب١٤٩٤لقد سألت الله عز وجل باسمه الأعظم١٤٩٥لقد دعا الله باسمه العظيم الذي إذا دعي به أجاب وإذا سئل به أعطى١٤٩٦اسم الله الأعظم في هاتين الآيتين { وإلهكم إله واحد لا إله إلا هو…١٤٩٧قالت سرقت ملحفة لها فجعلت تدعو على من سرقها فجعل النبي صلى الله عليه…١٤٩٨لا تنسنا يا أخى من دعائك١٤٩٩حدثنا زهير بن حرب، حدثنا أبو معاوية، حدثنا الأعمش، عن أبي صالح، عن…١٥٠٠أخبرك بما هو أيسر عليك من هذا أو أفضل١٥٠١أخبرتها أن النبي صلى الله عليه وسلم أمرهن أن يراعين بالتكبير والتقديس…١٥٠٢ومحمد بن قدامة، - في آخرين - قالوا حدثنا عثام، عن الأعمش، عن عطاء بن…١٥٠٣لم تزالي في مصلاك هذا١٥٠٤يا أبا ذر ألا أعلمك كلمات تدرك بهن من سبقك ولا يلحقك من خلفك إلا من…١٥٠٥لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…١٥٠٦لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…١٥٠٧ولا حول ولا قوة إلا بالله لا إله إلا الله لا نعبد إلا إياه له النعمة١٥٠٨اللهم ربنا ورب كل شىء أنا شهيد أنك أنت الرب وحدك لا شريك لك اللهم…١٥٠٩اللهم اغفر لي ما قدمت وما أخرت وما أسررت وما أعلنت وما أسرفت وما أنت…١٥١٠رب أعني ولا تعن على وانصرني ولا تنصر على وامكر لي ولا تمكر على واهدني…١٥١١ويسر الهدى إلى١٥١٢اللهم أنت السلام ومنك السلام تباركت يا ذا الجلال والإكرام١٥١٣مولى رسول الله صلى الله عليه وسلم أن النبي صلى الله عليه وسلم كان إذا…١٥١٤ما أصر من استغفر وإن عاد في اليوم سبعين مرة١٥١٥إنه ليغان على قلبي وإني لأستغفر الله في كل يوم مائة مرة١٥١٦رب اغفر لي وتب على إنك أنت التواب الرحيم١٥١٧من قال أستغفر الله الذي لا إله إلا هو الحى القيوم وأتوب إليه غفر له…١٥١٨من لزم الاستغفار جعل الله له من كل ضيق مخرجا ومن كل هم فرجا ورزقه من…١٥١٩اللهم ربنا آتنا في الدنيا حسنة وفي الآخرة حسنة وقنا عذاب النار١٥٢٠من سأل الله الشهادة صادقا بلغه الله منازل الشهداء وإن مات على فراشه١٥٢١ما من عبد يذنب ذنبا فيحسن الطهور ثم يقوم فيصلي ركعتين ثم يستغفر الله…١٥٢٢يا معاذ والله إني لأحبك والله إني لأحبك١٥٢٣أن حنين بن أبي حكيم، حدثه عن على بن رباح اللخمي، عن عقبة بن عامر، قال…١٥٢٤أن رسول الله صلى الله عليه وسلم كان يعجبه أن يدعو ثلاثا ويستغفر ثلاثا…١٥٢٥ألا أعلمك كلمات تقولينهن عند الكرب أو في الكرب الله الله ربي لا أشرك…١٥٢٦يا أيها الناس إنكم لا تدعون أصم ولا غائبا إن الذي تدعونه بينكم وبين…١٥٢٧إنكم لا تنادون أصم ولا غائبا١٥٢٨يا أيها الناس اربعوا على أنفسكم١٥٢٩من قال رضيت بالله ربا وبالإسلام دينا وبمحمد رسولا وجبت له الجنة١٥٣٠من صلى على واحدة صلى الله عليه عشرا١٥٣١إن من أفضل أيامكم يوم الجمعة فأكثروا على من الصلاة فيه فإن صلاتكم…١٥٣٢لا تدعوا على أنفسكم ولا تدعوا على أولادكم ولا تدعوا على خدمكم ولا…١٥٣٣صلى الله عليك وعلى زوجك١٥٣٤إذا دعا الرجل لأخيه بظهر الغيب قالت الملائكة آمين ولك بمثل١٥٣٥إن أسرع الدعاء إجابة دعوة غائب لغائب١٥٣٦ثلاث دعوات مستجابات لا شك فيهن دعوة الوالد ودعوة المسافر ودعوة المظلوم١٥٣٧اللهم إنا نجعلك في نحورهم ونعوذ بك من شرورهم١٥٣٨إذا هم أحدكم بالأمر فليركع ركعتين من غير الفريضة وليقل اللهم إني…١٥٣٩قال كان النبي صلى الله عليه وسلم يتعوذ من خمس من الجبن والبخل وسوء…١٥٤٠اللهم إني أعوذ بك من العجز والكسل والجبن والبخل والهرم وأعوذ بك من…١٥٤١اللهم إني أعوذ بك من الهم والحزن وضلع الدين وغلبة الرجال١٥٤٢اللهم إني أعوذ بك من عذاب جهنم وأعوذ بك من عذاب القبر وأعوذ بك من…١٥٤٣اللهم إني أعوذ بك من فتنة النار وعذاب النار ومن شر الغنى والفقر١٥٤٤اللهم إني أعوذ بك من الفقر والقلة والذلة وأعوذ بك من أن أظلم أو أظلم١٥٤٥اللهم إني أعوذ بك من زوال نعمتك وتحويل عافيتك وفجاءة نقمتك وجميع سخطك١٥٤٦اللهم إني أعوذ بك من الشقاق والنفاق وسوء الأخلاق١٥٤٧اللهم إني أعوذ بك من الجوع فإنه بئس الضجيع وأعوذ بك من الخيانة فإنها…١٥٤٨اللهم إني أعوذ بك من الأربع من علم لا ينفع ومن قلب لا يخشع ومن نفس لا…١٥٤٩اللهم إني أعوذ بك من صلاة لا تنفع١٥٥٠اللهم إني أعوذ بك من شر ما عملت ومن شر ما لم أعمل١٥٥١قل اللهم إني أعوذ بك من شر سمعي ومن شر بصري ومن شر لساني ومن شر قلبي…١٥٥٢اللهم إني أعوذ بك من الهدم وأعوذ بك من التردي وأعوذ بك من الغرق…١٥٥٣زاد فيه " والغم " .١٥٥٤اللهم إني أعوذ بك من البرص والجنون والجذام ومن سيئ
