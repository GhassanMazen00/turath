سنن أبي داود

كتاب الترجل

٥٥ حديثًا
٤١٥٩حدثنا مسدد، حدثنا يحيى، عن هشام بن حسان، عن الحسن، عن عبد الله بن…٤١٦٠أن رجلا، من أصحاب النبي صلى الله عليه وسلم رحل إلى فضالة بن عبيد وهو…٤١٦١ألا تسمعون ألا تسمعون إن البذاذة من الإيمان إن البذاذة من الإيمان٤١٦٢قال كانت للنبي صلى الله عليه وسلم سكة يتطيب منها .٤١٦٣من كان له شعر فليكرمه٤١٦٤قال حدثتني كريمة بنت همام، أن امرأة، أتت عائشة - رضى الله عنها -…٤١٦٥لا أبايعك حتى تغيري كفيك كأنهما كفا سبع٤١٦٦ما أدري أيد رجل أم يد امرأة٤١٦٧إنما هلكت بنو إسرائيل حين اتخذ هذه نساؤهم٤١٦٨ومسدد، قالا حدثنا يحيى، عن عبيد الله، قال حدثني نافع، عن عبد الله،…٤١٦٩وعثمان بن أبي شيبة، - المعنى - قالا حدثنا جرير، عن منصور، عن إبراهيم،…٤١٧٠قال لعنت الواصلة والمستوصلة والنامصة والمتنمصة والواشمة والمستوشمة من…٤١٧١قال لا بأس بالقرامل . قال أبو داود كأنه يذهب إلى أن المنهي، عنه شعور…٤١٧٢من عرض عليه طيب فلا يرده فإنه طيب الريح خفيف المحمل٤١٧٣إذا استعطرت المرأة فمرت على القوم ليجدوا ريحها فهي كذا وكذا٤١٧٤لا تقبل صلاة لامرأة تطيبت لهذا المسجد حتى ترجع فتغتسل غسلها من الجنابة٤١٧٥أيما امرأة أصابت بخورا فلا تشهدن معنا العشاء٤١٧٦اذهب فاغسل هذا عنك٤١٧٧أنه سمع يحيى بن يعمر، يخبر عن رجل، أخبره عن عمار بن ياسر، - زعم عمر…٤١٧٨لا يقبل الله تعالى صلاة رجل في جسده شىء من خلوق٤١٧٩أن حماد بن زيد، وإسماعيل بن إبراهيم، حدثاهم عن عبد العزيز بن صهيب، عن…٤١٨٠ثلاثة لا تقربهم الملائكة جيفة الكافر والمتضمخ بالخلوق والجنب إلا أن…٤١٨١حدثنا أيوب بن محمد الرقي، حدثنا عمر بن أيوب، عن جعفر بن برقان، عن…٤١٨٢لو أمرتم هذا أن يغسل هذا عنه٤١٨٣ومحمد بن سليمان الأنباري، قالا حدثنا وكيع، عن سفيان، عن أبي إسحاق، عن…٤١٨٤قال كان رسول الله صلى الله عليه وسلم له شعر يبلغ شحمة أذنيه .٤١٨٥قال كان شعر رسول الله صلى الله عليه وسلم إلى شحمة أذنيه .٤١٨٦قال كان شعر رسول الله صلى الله عليه وسلم إلى أنصاف أذنيه .٤١٨٧قالت كان شعر رسول الله صلى الله عليه وسلم فوق الوفرة ودون الجمة .٤١٨٨قال كان أهل الكتاب - يعني - يسدلون أشعارهم وكان المشركون يفرقون…٤١٨٩رضى الله عنها قالت كنت إذا أردت أن أفرق رأس رسول الله صلى الله عليه…٤١٩٠وسفيان بن عقبة السوائي، - هو أخو قبيصة - وحميد بن خوار عن سفيان…٤١٩١قال قالت أم هانئ قدم النبي صلى الله عليه وسلم إلى مكة وله أربع غدائر…٤١٩٢لا تبكوا على أخي بعد اليوم٤١٩٣حدثنا أحمد بن حنبل، حدثنا عثمان بن عثمان، - قال أحمد كان رجلا صالحا -…٤١٩٤حدثنا موسى بن إسماعيل، حدثنا حماد، حدثنا أيوب، عن نافع، عن ابن عمر،…٤١٩٥احلقوه كله أو اتركوه كله٤١٩٦قال كانت لي ذؤابة فقالت لي أمي لا أجزها كان رسول الله صلى الله عليه…٤١٩٧احلقوا هذين أو قصوهما فإن هذا زي اليهود٤١٩٨الفطرة خمس أو خمس من الفطرة الختان والاستحداد ونتف الإبط وتقليم…٤١٩٩أن رسول الله صلى الله عليه وسلم أمر بإحفاء الشوارب وإعفاء اللحى .٤٢٠٠حدثنا مسلم بن إبراهيم، حدثنا صدقة الدقيقي، حدثنا أبو عمران الجوني، عن…٤٢٠١حدثنا ابن نفيل، حدثنا زهير، قرأت على عبد الملك بن أبي سليمان وقرأه…٤٢٠٢لا تنتفوا الشيب ما من مسلم يشيب شيبة في الإسلام٤٢٠٣إن اليهود والنصارى لا يصبغون فخالفوهم٤٢٠٤غيروا هذا بشىء واجتنبوا السواد٤٢٠٥إن أحسن ما غير به هذا الشيب الحناء والكتم٤٢٠٦قال انطلقت مع أبي نحو النبي صلى الله عليه وسلم فإذا هو ذو وفرة بها…٤٢٠٧الله الطبيب بل أنت رجل رفيق طبيبها الذي خلقها٤٢٠٨حدثنا ابن بشار، حدثنا عبد الرحمن، حدثنا سفيان، عن إياد بن لقيط، عن…٤٢٠٩صلى الله عليه وسلم فذكر أنه لم يخضب ولكن قد خضب أبو بكر وعمر رضي الله…٤٢١٠أن النبي صلى الله عليه وسلم كان يلبس النعال السبتية ويصفر لحيته…٤٢١١قال مر على النبي صلى الله عليه وسلم رجل قد خضب بالحناء فقال " ما أحسن…٤٢١٢يكون قوم يخضبون في آخر الزمان بالسواد كحواصل الحمام لا يريحون رائحة…٤٢١٣يا ثوبان اذهب بهذا إلى آل فلان
