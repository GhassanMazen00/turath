سنن أبي داود

كتاب الجهاد

٣١١ حديثًا•صفحة ١ من ٣
٢٤٧٧ويحك إن شأن الهجرة شديد فهل لك من إبل٢٤٧٨يا عائشة ارفقي فإن الرفق لم يكن في شىء قط إلا زانه ولا نزع من شىء قط…٢٤٧٩لا تنقطع الهجرة حتى تنقطع التوبة ولا تنقطع التوبة حتى تطلع الشمس من…٢٤٨٠لا هجرة ولكن جهاد ونية وإذا استنفرتم فانفروا٢٤٨١المسلم من سلم المسلمون من لسانه ويده والمهاجر من هجر ما نهى الله عنه٢٤٨٢ستكون هجرة بعد هجرة فخيار أهل الأرض ألزمهم مهاجر إبراهيم ويبقى في…٢٤٨٣سيصير الأمر إلى أن تكونوا جنودا مجندة جند بالشام وجند باليمن وجند…٢٤٨٤لا تزال طائفة من أمتي يقاتلون على الحق ظاهرين على من ناوأهم حتى يقاتل…٢٤٨٥رجل يجاهد في سبيل الله بنفسه وماله ورجل يعبد الله في شعب من الشعاب قد…٢٤٨٦إن سياحة أمتي الجهاد في سبيل الله تعالى٢٤٨٧حدثنا محمد بن المصفى، حدثنا علي بن عياش، عن الليث بن سعد، حدثنا حيوة،…٢٤٨٨ابنك له أجر شهيدين٢٤٨٩لا يركب البحر إلا حاج أو معتمر أو غاز في سبيل الله فإن تحت البحر نارا…٢٤٩٠رأيت قوما ممن يركب ظهر هذا البحر كالملوك على الأسرة٢٤٩١أنه سمعه يقول كان رسول الله صلى الله عليه وسلم إذا ذهب إلى قباء يدخل…٢٤٩٢قالت نام النبي صلى الله عليه وسلم فاستيقظ وكانت تغسل رأسها فاستيقظ…٢٤٩٣المائد في البحر الذي يصيبه القىء له أجر شهيد والغرق له أجر شهيدين٢٤٩٤ثلاثة كلهم ضامن على الله عز وجل رجل خرج غازيا في سبيل الله فهو ضامن…٢٤٩٥لا يجتمع في النار كافر وقاتله أبدا٢٤٩٦حرمة نساء المجاهدين على القاعدين كحرمة أمهاتهم وما من رجل من القاعدين…٢٤٩٧ما من غازية تغزو في سبيل الله فيصيبون غنيمة إلا تعجلوا ثلثى أجرهم من…٢٤٩٨إن الصلاة والصيام والذكر تضاعف على النفقة في سبيل الله بسبعمائة ضعف٢٤٩٩من فصل في سبيل الله فمات أو قتل فهو شهيد أو وقصه فرسه أو بعيره أو…٢٥٠٠كل الميت يختم على عمله، إلا المرابط فإنه ينمو له عمله إلى يوم القيامة…٢٥٠١تلك غنيمة المسلمين غدا إن شاء الله٢٥٠٢من مات ولم يغز ولم يحدث نفسه بالغزو مات على شعبة من نفاق٢٥٠٣من لم يغز أو يجهز غازيا أو يخلف غازيا في أهله بخير أصابه الله بقارعة٢٥٠٤جاهدوا المشركين بأموالكم وأنفسكم وألسنتكم٢٥٠٥قال { إلا تنفروا يعذبكم عذابا أليما } و { ما كان لأهل المدينة } إلى…٢٥٠٦{ إلا تنفروا يعذبكم عذابا أليما } قال : فأمسك عنهم المطر وكان عذابهم .٢٥٠٧قال : كنت إلى جنب رسول الله صلى الله عليه وسلم فغشيته السكينة فوقعت…٢٥٠٨لقد تركتم بالمدينة أقواما ما سرتم مسيرا ولا أنفقتم من نفقة ولا قطعتم…٢٥٠٩من جهز غازيا في سبيل الله فقد غزا، ومن خلفه في أهله بخير فقد غزا٢٥١٠ليخرج من كل رجلين رجل٢٥١١شر ما في رجل شح هالع وجبن خالع٢٥١٢وابن، لهيعة عن يزيد بن أبي حبيب، عن أسلم أبي عمران، قال : غزونا من…٢٥١٣إن الله عز وجل يدخل بالسهم الواحد ثلاثة نفر الجنة : صانعه يحتسب في…٢٥١٤{ وأعدوا لهم ما استطعتم من قوة } ألا إن القوة الرمى، ألا إن القوة…٢٥١٥الغزو غزوان فأما من ابتغى وجه الله، وأطاع الإمام، وأنفق الكريمة،…٢٥١٦: أن رجلا، قال : يا رسول الله، رجل يريد الجهاد في سبيل الله وهو يبتغي…٢٥١٧من قاتل حتى تكون كلمة الله هي أعلى فهو في سبيل الله عز وجل٢٥١٨حديثا أعجبني . فذكر معناه .٢٥١٩يا عبد الله بن عمرو، إن قاتلت صابرا محتسبا بعثك الله صابرا محتسبا،…٢٥٢٠لما أصيب إخوانكم بأحد جعل الله أرواحهم في جوف طير خضر ترد أنهار…٢٥٢١النبي في الجنة، والشهيد في الجنة، والمولود في الجنة، والوئيد في الجنة٢٥٢٢يشفع الشهيد في سبعين من أهل بيته٢٥٢٣حدثنا محمد بن عمرو الرازي، حدثنا سلمة، - يعني ابن الفضل - عن محمد بن…٢٥٢٤قال : آخى رسول الله صلى الله عليه وسلم بين رجلين فقتل أحدهما ومات…٢٥٢٥ستفتح عليكم الأمصار، وستكون جنود مجندة تقطع عليكم فيها بعوث فيكره…٢٥٢٦للغازي أجره، وللجاعل أجره وأجر الغازي٢٥٢٧ما أجد له في غزوته هذه في الدنيا والآخرة إلا دنانيره التي سمى٢٥٢٨ارجع عليهما فأضحكهما كما أبكيتهما٢٥٢٩قال : جاء رجل إلى النبي صلى الله عليه وسلم فقال : يا رسول الله أجاهد…٢٥٣٠هل لك أحد باليمن٢٥٣١قال : كان رسول الله صلى الله عليه وسلم يغزو بأم سليم ونسوة من الأنصار…٢٥٣٢ثلاثة من أصل الإيمان : الكف عمن قال لا إله إلا الله ولا تكفره بذنب…٢٥٣٣الجهاد واجب عليكم مع كل أمير برا كان أو فاجرا، والصلاة واجبة عليكم…٢٥٣٤يا معشر المهاجرين والأنصار، إن من إخوانكم قوما ليس لهم مال ولا عشيرة…٢٥٣٥اللهم لا تكلهم إلى فأضعف عنهم، ولا تكلهم إلى أنفسهم فيعجزوا عنها، ولا…٢٥٣٦عجب ربنا من رجل غزا في سبيل الله فانهزم٢٥٣٧أن عمرو بن أقيش، كان له ربا في الجاهلية فكره أن يسلم حتى يأخذه فجاء…٢٥٣٨مات جاهدا مجاهدا٢٥٣٩أخوكم يا معشر المسلمين٢٥٤٠ثنتان لا تردان، أو قلما تردان : الدعاء عند النداء، وعند البأس حين…٢٥٤١من قاتل في سبيل الله فواق ناقة فقد وجبت له الجنة، ومن سأل الله القتل…٢٥٤٢لا تقصوا نواصي الخيل ولا معارفها ولا أذنابها، فإن أذنابها مذابها،…٢٥٤٣عليكم بكل كميت أغر محجل، أو أشقر أغر محجل، أو أدهم أغر محجل٢٥٤٤عليكم بكل أشقر أغر محجل، أو كميت أغر٢٥٤٥يمن الخيل في شقرها٢٥٤٦: أن رسول الله صلى الله عليه وسلم كان يسمي الأنثى من الخيل فرسا .٢٥٤٧قال : كان النبي صلى الله عليه وسلم يكره الشكال من الخيل . والشكال :…٢٥٤٨اتقوا الله في هذه البهائم المعجمة فاركبوها وكلوها صالحة٢٥٤٩من رب هذا الجمل، لمن هذا الجمل٢٥٥٠بينما رجل يمشي بطريق فاشتد عليه العطش، فوجد بئرا فنزل فيها فشرب ثم…٢٥٥١حدثنا محمد بن المثنى، حدثني محمد بن جعفر، حدثنا شعبة، عن حمزة الضبي،…٢٥٥٢لا يبقين في رقبة بعير قلادة من وتر ولا قلادة إلا قطعت٢٥٥٣ارتبطوا الخيل وامسحوا بنواصيها وأعجازها٢٥٥٤لا تصحب الملائكة رفقة فيها جرس٢٥٥٥لا تصحب الملائكة رفقة فيها كلب أو جرس٢٥٥٦أن النبي صلى الله عليه وسلم قال في الجرس " مزمار الشيطان " .٢٥٥٧حدثنا مسدد، حدثنا عبد الوارث، عن أيوب، عن نافع، عن ابن عمر، قال نهي…٢٥٥٨حدثنا أحمد بن أبي سريج الرازي، أخبرني عبد الله بن الجهم، حدثنا عمرو،…٢٥٥٩قال كنت ردف رسول الله صلى الله عليه وسلم على حمار يقال له عفير .٢٥٦٠حدثنا محمد بن داود بن سفيان، حدثني يحيى بن حسان، أخبرنا سليمان بن…٢٥٦١أن النبي صلى الله عليه وسلم كان في سفر فسمع لعنة فقال " ما هذه " .…٢٥٦٢حدثنا محمد بن العلاء، أخبرنا يحيى بن آدم، عن قطبة بن عبد العزيز بن…٢٥٦٣قال أتيت النبي صلى الله عليه وسلم بأخ لي حين ولد ليحنكه فإذا هو في…٢٥٦٤أما بلغكم أني قد لعنت من وسم البهيمة في وجهها أو ضربها في وجهها٢٥٦٥إنما يفعل ذلك الذين لا يعلمون٢٥٦٦حدثنا أبو صالح، محبوب بن موسى أخبرنا أبو إسحاق الفزاري، عن عاصم بن…٢٥٦٧إياكم أن تتخذوا ظهور دوابكم منابر فإن الله إنما سخرها لكم لتبلغكم إلى…٢٥٦٨تكون إبل للشياطين وبيوت للشياطين فأما إبل الشياطين فقد رأيتها يخرج…٢٥٦٩إذا سافرتم في الخصب فأعطوا الإبل حقها وإذا سافرتم في الجدب فأسرعوا…٢٥٧٠حدثنا عثمان بن أبي شيبة، حدثنا يزيد بن هارون، أخبرنا هشام، عن الحسن،…٢٥٧١عليكم بالدلجة فإن الأرض تطوى بالليل٢٥٧٢لا أنت أحق بصدر دابتك مني إلا أن تجعله لي٢٥٧٣حدثنا عبد الله بن محمد النفيلي، حدثنا محمد بن سلمة، عن محمد بن إسحاق،…٢٥٧٤لا سبق إلا في خف أو في حافر أو نصل٢٥٧٥أن رسول الله صلى الله عليه وسلم سابق بين الخيل التي قد ضمرت من…٢٥٧٦أن نبي الله صلى الله عليه وسلم كان يضمر الخيل يسابق بها .٢٥٧٧أن النبي صلى الله عليه وسلم سبق بين الخيل وفضل القرح في الغاية .٢٥٧٨هذه بتلك السبقة٢٥٧٩من أدخل فرسا بين فرسين٢٥٨٠حدثنا محمود بن خالد، حدثنا الوليد بن مسلم، عن سعيد بن بشير، عن…٢٥٨١حدثنا يحيى بن خلف، حدثنا عبد الوهاب بن عبد المجيد، حدثنا عنبسة، ح…٢٥٨٢قال الجلب والجنب في الرهان .٢٥٨٣قال كانت قبيعة سيف رسول الله صلى الله عليه وسلم فضة .٢٥٨٤قال كانت قبيعة سيف رسول الله صلى الله عليه وسلم فضة . قال قتادة وما…٢٥٨٥قال كانت . فذكر مثله . قال أبو داود أقوى هذه الأحاديث حديث سعيد بن…٢٥٨٦حدثنا قتيبة بن سعيد، حدثنا الليث، عن أبي الزبير، عن جابر، عن رسول…٢٥٨٧إذا مر أحدكم في مسجدنا أو في سوقنا ومعه نبل فليمسك على نصالها٢٥٨٨أن النبي صلى الله عليه وسلم نهى أن يتعاطى السيف مسلولا .٢٥٨٩أن رسول الله صلى الله عليه وسلم نهى أن يقد السير بين أصبعين .٢٥٩٠قد سماه أن رسول الله صلى الله عليه وسلم ظاهر يوم أحد بين درعين أو لبس…٢٥٩١رسول الله صلى الله عليه وسلم ما كانت فقال كانت سوداء مربعة من نمرة .٢٥٩٢يرفعه إلى النبي صلى الله عليه وسلم أنه كان لواؤه يوم دخل مكة أبيض .٢٥٩٣منهم قال رأيت راية رسول الله صلى الله عليه وسلم صفراء .٢٥٩٤ابغوني الضعفاء فإنما ترزقون وتنصرون بضعفائكم٢٥٩٥قال كان شعار المهاجرين عبد الله وشعار الأنصار عبد الرحمن .٢٥٩٦حدثنا هناد، عن ابن المبارك، عن عكرمة بن عمار، عن إياس بن سلمة، عن…٢٥٩٧إن بيتم فليكن شعاركم حم لا ينصرون٢٥٩٨اللهم أنت الصاحب في السفر والخليفة في الأهل اللهم إني أعوذ بك من…٢٥٩٩{ سبحان الذي سخر لنا هذا وما كنا له مقرنين * وإنا إلى ربنا لمنقلبون }…٢٦٠٠أستودع الله دينك وأمانتك وخواتيم عملك٢٦٠١أستودع الله دينكم وأمانتكم وخواتيم أعمالكم٢٦٠٢إن ربك يعجب من عبده إذا قال اغفر لي ذنوبي يعلم أنه لا يغفر الذنوب غيري٢٦٠٣يا أرض ربي وربك الله أعوذ بالله من شرك وشر ما فيك وشر ما خلق فيك ومن…٢٦٠٤لا ترسلوا فواشيكم إذا غابت الشمس حتى تذهب فحمة العشاء فإن الشياطين…٢٦٠٥قال قلما كان رسول الله صلى الله عليه وسلم يخرج في سفر إلا يوم الخميس .٢٦٠٦اللهم بارك لأمتي في بكورها٢٦٠٧الراكب شيطان والراكبان شيطانان والثلاثة ركب٢٦٠٨إذا خرج ثلاثة في سفر فليؤمروا أحدهم٢٦٠٩إذا كان ثلاثة في سفر فليؤمروا أحدهم٢٦١٠أن عبد الله بن عمر، قال نهى رسول الله صلى الله عليه وسلم أن يسافر…٢٦١١خير الصحابة أربعة وخير السرايا أربعمائة وخير الجيوش أربعة آلاف ولن…٢٦١٢إذا لقيت عدوك من المشركين فادعهم إلى إحدى ثلاث خصال أو خلال فأيتها…٢٦١٣اغزوا باسم الله وفي سبيل الله وقاتلوا من كفر بالله اغزوا ولا تغدروا…٢٦١٤انطلقوا باسم الله وبالله وعلى ملة رسول الله ولا تقتلوا شيخا فانيا ولا…٢٦١٥أن رسول الله صلى الله عليه وسلم حرق نخل بني النضير وقطع وهي البويرة…٢٦١٦أغر على أبنى صباحا وحرق٢٦١٧قيل له أبنى . قال نحن أعلم هي يبنى فلسطين .٢٦١٨حدثنا هارون بن عبد الله، حدثنا هاشم بن القاسم، حدثنا سليمان، - يعني…٢٦١٩إذا أتى أحدكم على ماشية فإن كان فيها صاحبها فليستأذنه فإن أذن له…٢٦٢٠ما علمت إذ كان جاهلا ولا أطعمت إذ كان جائعا٢٦٢١حدثني محمد بن بشار، حدثنا محمد بن جعفر، عن شعبة، عن أبي بشر، قال سمعت…٢٦٢٢يا غلام لم ترمي النخل٢٦٢٣لا يحلبن أحد ماشية أحد بغير إذنه أيحب أحدكم أن تؤتى مشربته فتكسر…٢٦٢٤حدثنا زهير بن حرب، حدثنا حجاج، قال قال ابن جريج { يا أيها الذين آمنوا…٢٦٢٥لو دخلوها - أو دخلوا فيها - لم يزالوا فيها٢٦٢٦السمع والطاعة على المرء المسلم فيما أحب وكره ما لم يؤمر بمعصية فإذا…
