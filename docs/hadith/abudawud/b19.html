سنن أبي داود

كتاب الفرائض

٤٣ حديثًا
٢٨٨٥العلم ثلاثة وما سوى ذلك فهو فضل آية محكمة أو سنة قائمة أو فريضة عادلة٢٨٨٦أنه سمع جابرا، يقول مرضت فأتاني النبي صلى الله عليه وسلم يعودني هو…٢٨٨٧قال اشتكيت وعندي سبع أخوات فدخل على رسول الله صلى الله عليه وسلم فنفخ…٢٨٨٨قال آخر آية نزلت في الكلالة { يستفتونك قل الله يفتيكم في الكلالة } .٢٨٨٩تجزيك آية الصيف٢٨٩٠حدثنا عبد الله بن عامر بن زرارة، حدثنا علي بن مسهر، عن الأعمش، عن أبي…٢٨٩١يقضي الله في ذلك٢٨٩٢وغيره، من أهل العلم عن عبد الله بن محمد بن عقيل، عن جابر بن عبد الله،…٢٨٩٣أن معاذ بن جبل، ورث أختا وابنة فجعل لكل واحدة منهما النصف وهو باليمن…٢٨٩٤حدثنا القعنبي، عن مالك، عن ابن شهاب، عن عثمان بن إسحاق بن خرشة، عن…٢٨٩٥أن النبي صلى الله عليه وسلم جعل للجدة السدس إذا لم تكن دونها أم .٢٨٩٦أن رجلا، أتى النبي صلى الله عليه وسلم فقال إن ابن ابني مات فما لي من…٢٨٩٧أن عمر، قال أيكم يعلم ما ورث رسول الله صلى الله عليه وسلم الجد فقال…٢٨٩٨اقسم المال بين أهل الفرائض على كتاب الله فما تركت الفرائض فلأولى ذكر٢٨٩٩من ترك كلا فإلى٢٩٠٠أنا أولى بكل مؤمن من نفسه فمن ترك دينا أو ضيعة فإلى ومن ترك مالا…٢٩٠١أنا وارث من لا وارث له أفك عانيه وأرث ماله والخال وارث من لا وارث له…٢٩٠٢أعطوا ميراثه رجلا من أهل قريته٢٩٠٣اذهب فالتمس أزديا حولا٢٩٠٤التمسوا له وارثا أو ذا رحم٢٩٠٥أن رجلا، مات ولم يدع وارثا إلا غلاما له كان أعتقه فقال رسول الله صلى…٢٩٠٦المرأة تحرز ثلاثة مواريث عتيقها ولقيطها وولدها الذي لاعنت عنه٢٩٠٧وموسى بن عامر، قالا حدثنا الوليد، أخبرنا ابن جابر، حدثنا مكحول، قال…٢٩٠٨حدثنا موسى بن عامر، حدثنا الوليد، أخبرني عيسى أبو محمد، عن العلاء بن…٢٩٠٩لا يرث المسلم الكافر ولا الكافر المسلم٢٩١٠وهل ترك لنا عقيل منزلا٢٩١١لا يتوارث أهل ملتين شتى٢٩١٢الإسلام يزيد ولا ينقص٢٩١٣أن معاذا، أتي بميراث يهودي وارثه مسلم بمعناه عن النبي صلى الله عليه…٢٩١٤كل قسم قسم في الجاهلية فهو على ما قسم له وكل قسم أدركه الإسلام فهو…٢٩١٥لا يمنعك ذلك فإن الولاء لمن أعتق٢٩١٦الولاء لمن أعطى الثمن وولي النعمة٢٩١٧ما أحرز الولد أو الوالد فهو لعصبته من كان٢٩١٨هو أولى الناس بمحياه ومماته٢٩١٩حدثنا حفص بن عمر، حدثنا شعبة، عن عبد الله بن دينار، عن ابن عمر، رضى…٢٩٢٠إذا استهل المولود ورث٢٩٢١رضى الله عنهما قال { والذين عقدت أيمانكم فآتوهم نصيبهم } كان الرجل…٢٩٢٢حدثنا هارون بن عبد الله، حدثنا أبو أسامة، حدثني إدريس بن يزيد، حدثنا…٢٩٢٣وعبد العزيز بن يحيى المعنى، - قال أحمد - حدثنا محمد بن سلمة، عن ابن…٢٩٢٤والذين، آمنوا وهاجروا والذين آمنوا ولم يهاجروا فكان الأعرابي لا يرث…٢٩٢٥لا حلف في الإسلام وأيما حلف كان في الجاهلية لم يزده الإسلام إلا شدة٢٩٢٦لا حلف في الإسلام٢٩٢٧حدثنا أحمد بن صالح، حدثنا سفيان، عن الزهري، عن سعيد، قال كان عمر بن…
