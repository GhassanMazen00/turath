سنن أبي داود

كتاب الديات

١٠٢ حديثًا
٤٤٩٤حدثنا محمد بن العلاء، حدثنا عبيد الله، - يعني ابن موسى - عن علي بن…٤٤٩٥قال انطلقت مع أبي نحو النبي صلى الله عليه وسلم ثم إن رسول الله صلى…٤٤٩٦من أصيب بقتل أو خبل فإنه يختار إحدى ثلاث إما أن يقتص وإما أن يعفو…٤٤٩٧قال ما رأيت النبي صلى الله عليه وسلم رفع إليه شىء فيه قصاص إلا أمر…٤٤٩٨أما إنه إن كان صادقا ثم قتلته دخلت النار٤٤٩٩حدثنا عبيد الله بن عمر بن ميسرة الجشمي، حدثنا يحيى بن سعيد، عن عوف،…٤٥٠٠بإسناده ومعناه .٤٥٠١قال جاء رجل إلى النبي صلى الله عليه وسلم بحبشي فقال إن هذا قتل ابن…٤٥٠٢لا يحل دم امرئ مسلم إلا بإحدى ثلاث كفر بعد إسلام أو زنا بعد إحصان أو…٤٥٠٣يا عيينة ألا تقبل الغير٤٥٠٤ألا إنكم يا معشر خزاعة قتلتم هذا القتيل من هذيل وإني عاقله فمن قتل له…٤٥٠٥من قتل له قتيل فهو بخير النظرين إما أن يودى أو يقاد٤٥٠٦لا يقتل مؤمن بكافر ومن قتل مؤمنا متعمدا دفع إلى أولياء المقتول فإن…٤٥٠٧لا أعفي من قتل بعد أخذه الدية٤٥٠٨ما كان الله ليسلطك على ذلك٤٥٠٩وأبي، سلمة - قال هارون - عن أبي هريرة، أن امرأة، من اليهود أهدت إلى…٤٥١٠قال كان جابر بن عبد الله يحدث أن يهودية، من أهل خيبر سمت شاة مصلية ثم…٤٥١١ما حملك على الذي صنعت٤٥١٢ارفعوا أيديكم فإنها أخبرتني أنها مسمومة٤٥١٣وأنا لا أتهم بنفسي إلا ذلك فهذا أوان قطعت أبهري٤٥١٤ما حملك على الذي صنعت٤٥١٥من قتل عبده قتلناه ومن جدع عبده جدعناه٤٥١٦من خصى عبده خصيناه٤٥١٧لا يقتل حر بعبد٤٥١٨قال لا يقاد الحر بالعبد .٤٥١٩قال جاء رجل مستصرخ إلى النبي صلى الله عليه وسلم فقال جارية له يا رسول…٤٥٢٠ومحمد بن عبيد، - المعنى - قالا حدثنا حماد بن زيد، عن يحيى بن سعيد، عن…٤٥٢١أنه أخبره هو، ورجال، من كبراء قومه أن عبد الله بن سهل ومحيصة خرجا إلى…٤٥٢٢وكثير بن عبيد، قالا حدثنا ح، وحدثنا محمد بن الصباح بن سفيان، أخبرنا…٤٥٢٣تأتوني بالبينة على من قتل هذا٤٥٢٤لكم شاهدان يشهدان على قتل صاحبكم٤٥٢٥أنه قد وجد بين أظهركم قتيل فدوه٤٥٢٦يحلف منكم خمسون رجلا٤٥٢٧أن جارية، وجدت، قد رض رأسها بين حجرين فقيل لها من فعل بك هذا أفلان…٤٥٢٨أن يهوديا، قتل جارية من الأنصار على حلي لها ثم ألقاها في قليب ورضخ…٤٥٢٩من قتلك فلان قتلك٤٥٣٠المؤمنون تكافأ دماؤهم وهم يد على من سواهم ويسعى بذمتهم أدناهم ألا لا…٤٥٣١ويجير عليهم أقصاهم ويرد مشدهم على مضعفهم ومتسريهم على قاعدهم٤٥٣٢وعبد الوهاب بن نجدة الحوطي، - المعنى واحد - قالا حدثنا عبد العزيز بن…٤٥٣٣أن سعد بن عبادة، قال لرسول الله صلى الله عليه وسلم أرأيت لو وجدت مع…٤٥٣٤أن النبي صلى الله عليه وسلم بعث أبا جهم بن حذيفة مصدقا فلاجه رجل في…٤٥٣٥أن جارية، وجدت، قد رض رأسها بين حجرين فقيل لها من فعل بك هذا أفلان…٤٥٣٦قال بينما رسول الله صلى الله عليه وسلم يقسم قسما أقبل رجل فأكب عليه…٤٥٣٧حدثنا أبو صالح، أخبرنا أبو إسحاق الفزاري، عن الجريري، عن أبي نضرة، عن…٤٥٣٨على المقتتلين أن ينحجزوا الأول فالأول وإن كانت امرأة٤٥٣٩من قتل في عميا في رمى يكون بينهم بحجارة أو ضرب بالسياط أو ضرب بعصا…٤٥٤٠قال قال رسول الله صلى الله عليه وسلم فذكر معنى حديث سفيان .٤٥٤١أن رسول الله صلى الله عليه وسلم قضى أن من قتل خطأ فديته مائة من الإبل…٤٥٤٢قال كانت قيمة الدية على عهد رسول الله صلى الله عليه وسلم ثمانمائة…٤٥٤٣أن رسول الله صلى الله عليه وسلم قضى في الدية على أهل الإبل مائة من…٤٥٤٤قال فرض رسول الله صلى الله عليه وسلم فذكر مثل حديث موسى . قال وعلى…٤٥٤٥في دية الخطإ عشرون حقة وعشرون جذعة وعشرون بنت مخاض وعشرون بنت لبون…٤٥٤٦أن رجلا، من بني عدي قتل فجعل النبي صلى الله عليه وسلم ديته اثنى عشر…٤٥٤٧لا إله إلا الله وحده صدق وعده ونصر عبده وهزم الأحزاب وحده٤٥٤٨بهذا الإسناد نحو معناه .٤٥٤٩حدثنا مسدد، حدثنا عبد الوارث، عن علي بن زيد، عن القاسم بن ربيعة، عن…٤٥٥٠قال قضى عمر في شبه العمد ثلاثين حقة وثلاثين جذعة وأربعين خلفة ما بين…٤٥٥١رضى الله عنه أنه قال في شبه العمد أثلاث ثلاث وثلاثون حقة وثلاث…٤٥٥٢والأسود، قال عبد الله في شبه العمد خمس وعشرون حقة وخمس وعشرون جذعة…٤٥٥٣قال قال علي رضى الله عنه في الخطإ أرباعا خمس وعشرون حقة وخمس وعشرون…٤٥٥٤وزيد بن ثابت، في المغلظة أربعون جذعة خلفة وثلاثون حقة وثلاثون بنات…٤٥٥٥حدثنا محمد بن المثنى، حدثنا محمد بن عبد الله، حدثنا سعيد، عن قتادة،…٤٥٥٦الأصابع سواء عشر عشر من الإبل٤٥٥٧حدثنا أبو الوليد، حدثنا شعبة، عن غالب التمار، عن مسروق بن أوس، عن…٤٥٥٨قال قال رسول الله صلى الله عليه وسلم " هذه وهذه سواء " . قال يعني…٤٥٥٩الأصابع سواء والأسنان سواء الثنية والضرس سواء هذه وهذه سواء٤٥٦٠الأسنان سواء والأصابع سواء٤٥٦١قال جعل رسول الله صلى الله عليه وسلم أصابع اليدين والرجلين سواء .٤٥٦٢في الأصابع عشر عشر٤٥٦٣في الأسنان خمس خمس٤٥٦٤إن العقل ميراث بين ورثة القتيل على قرابتهم فما فضل فللعصبة٤٥٦٥عقل شبه العمد مغلظ مثل عقل العمد ولا يقتل صاحبه٤٥٦٦فضيل بن حسين أن خالد بن الحارث، حدثهم قال أخبرنا حسين، - يعني المعلم…٤٥٦٧قال قضى رسول الله صلى الله عليه وسلم في العين القائمة السادة لمكانها…٤٥٦٨أسجع كسجع الأعراب٤٥٦٩حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن منصور، بإسناده ومعناه . وزاد…٤٥٧٠وهارون بن عباد الأزدي، - المعنى - قالا حدثنا وكيع، عن هشام، عن عروة،…٤٥٧١أن عمر، قال .٤٥٧٢أنه سمع طاوسا، عن ابن عباس، عن عمر، أنه سأل عن قضية النبي، صلى الله…٤٥٧٣حدثنا عبد الله بن محمد الزهري، حدثنا سفيان، عن عمرو، عن طاوس، قال قام…٤٥٧٤أسجع الجاهلية وكهانتها أد في الصبي غرة٤٥٧٥لا ميراثها لزوجها وولدها٤٥٧٦إنما هذا من إخوان الكهان٤٥٧٧في هذه القصة قال ثم إن المرأة التي قضى عليها بالغرة توفيت فقضى رسول…٤٥٧٨أن امرأة، حذفت امرأة فأسقطت فرفع ذلك إلى رسول الله صلى الله عليه وسلم…٤٥٧٩حدثنا إبراهيم بن موسى الرازي، حدثنا عيسى، عن محمد، - يعني ابن عمرو -…٤٥٨٠وجابر، عن الشعبي، قال الغرة خمسمائة درهم . قال أبو داود قال ربيعة…٤٥٨١قال قضى رسول الله صلى الله عليه وسلم في دية المكاتب يقتل يودى ما أدى…٤٥٨٢إذا أصاب المكاتب حدا أو ورث ميراثا يرث على قدر ما عتق منه٤٥٨٣دية المعاهد نصف دية الحر٤٥٨٤أتريد أن يضع يده في فيك تقضمها كالفحل٤٥٨٥إن شئت أن تمكنه من يدك فيعضها ثم تنزعها من فيه٤٥٨٦من تطبب ولا يعلم منه طب فهو ضامن٤٥٨٧أيما طبيب تطبب على قوم لا يعرف له تطبب قبل ذلك فأعنت فهو ضامن٤٥٨٨ألا إن كل مأثرة كانت في الجاهلية من دم أو مال تذكر وتدعى تحت قدمى إلا…٤٥٨٩بهذا الإسناد نحو معناه .٤٥٩٠أن غلاما، لأناس فقراء قطع أذن غلام لأناس أغنياء فأتى أهله النبي صلى…٤٥٩١من قتل في عميا أو رميا يكون بينهم بحجر أو بسوط فعقله عقل خطإ ومن قتل…٤٥٩٢حدثنا عثمان بن أبي شيبة، حدثنا محمد بن يزيد، حدثنا سفيان بن حسين، عن…٤٥٩٣العجماء جرحها جبار والمعدن جبار والبئر جبار وفي الركاز الخمس٤٥٩٤قال قال رسول الله صلى الله عليه وسلم " النار جبار " .٤٥٩٥يا أنس كتاب الله القصاص
