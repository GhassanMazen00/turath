سنن أبي داود

كتاب الضحايا

٥٦ حديثًا
٢٧٨٨يا أيها الناس إن على كل أهل بيت في كل عام أضحية وعتيرة أتدرون ما…٢٧٨٩أمرت بيوم الأضحى عيدا جعله الله عز وجل لهذه الأمة٢٧٩٠حدثنا عثمان بن أبي شيبة، حدثنا شريك، عن أبي الحسناء، عن الحكم، عن…٢٧٩١من كان له ذبح يذبحه فإذا أهل هلال ذي الحجة فلا يأخذن من شعره ولا من…٢٧٩٢يا عائشة هلمي المدية٢٧٩٣أن النبي صلى الله عليه وسلم نحر سبع بدنات بيده قياما وضحى بالمدينة…٢٧٩٤أن النبي صلى الله عليه وسلم ضحى بكبشين أقرنين أملحين يذبح ويكبر ويسمي…٢٧٩٥إني وجهت وجهي للذي فطر السموات والأرض على ملة إبراهيم حنيفا وما أنا…٢٧٩٦قال كان رسول الله صلى الله عليه وسلم يضحي بكبش أقرن فحيل ينظر في سواد…٢٧٩٧لا تذبحوا إلا مسنة إلا أن يعسر عليكم فتذبحوا جذعة من الضأن٢٧٩٨قال قسم رسول الله صلى الله عليه وسلم في أصحابه ضحايا فأعطاني عتودا…٢٧٩٩إن الجذع يوفي مما يوفي منه الثني٢٨٠٠من صلى صلاتنا ونسك نسكنا فقد أصاب النسك ومن نسك قبل الصلاة فتلك شاة…٢٨٠١حدثنا مسدد، حدثنا خالد، عن مطرف، عن عامر، عن البراء بن عازب، قال ضحى…٢٨٠٢أربع لا تجوز في الأضاحي العوراء بين عورها والمريضة بين مرضها والعرجاء…٢٨٠٣حدثنا إبراهيم بن موسى الرازي، قال أخبرنا ح، وحدثنا علي بن بحر بن بري،…٢٨٠٤حدثنا عبد الله بن محمد النفيلي، حدثنا زهير، حدثنا أبو إسحاق، عن شريح…٢٨٠٥ويقال، له هشام بن سنبر عن قتادة، عن جرى بن كليب، عن علي، أن النبي صلى…٢٨٠٦قال قلت لسعيد بن المسيب ما الأعضب قال النصف فما فوقه .٢٨٠٧حدثنا أحمد بن حنبل، حدثنا هشيم، حدثنا عبد الملك، عن عطاء، عن جابر بن…٢٨٠٨البقرة عن سبعة والجزور عن سبعة٢٨٠٩حدثنا القعنبي، عن مالك، عن أبي الزبير المكي، عن جابر بن عبد الله، أنه…٢٨١٠بسم الله والله أكبر هذا عني وعمن لم يضح من أمتي٢٨١١أن أبا أسامة، حدثهم عن أسامة، عن نافع، عن ابن عمر، أن النبي صلى الله…٢٨١٢ادخروا الثلث وتصدقوا بما بقي٢٨١٣إنا كنا نهيناكم عن لحومها أن تأكلوها فوق ثلاث لكى تسعكم فقد جاء الله…٢٨١٤يا ثوبان أصلح لنا لحم هذه الشاة٢٨١٥إن الله كتب الإحسان على كل شىء فإذا قتلتم فأحسنوا٢٨١٦حدثنا أبو الوليد الطيالسي، حدثنا شعبة، عن هشام بن زيد، قال دخلت مع…٢٨١٧قال { فكلوا مما ذكر اسم الله عليه } { ولا تأكلوا مما لم يذكر اسم الله…٢٨١٨في قوله { وإن الشياطين ليوحون إلى أوليائهم } يقولون ما ذبح الله فلا…٢٨١٩حدثنا عثمان بن أبي شيبة، حدثنا عمران بن عيينة، عن عطاء بن السائب، عن…٢٨٢٠حدثنا هارون بن عبد الله، حدثنا حماد بن مسعدة، عن عوف، عن أبي ريحانة،…٢٨٢١أرن أو أعجل ما أنهر الدم وذكر اسم الله عليه فكلوا ما لم يكن سنا أو…٢٨٢٢أن عبد الواحد بن زياد، وحمادا، حدثاهم - المعنى، واحد، - عن عاصم، عن…٢٨٢٣من بني حارثة أنه كان يرعى لقحة بشعب من شعاب أحد فأخذها الموت فلم يجد…٢٨٢٤أمرر الدم بما شئت واذكر اسم الله عز وجل٢٨٢٥لو طعنت في فخذها لأجزأ عنك٢٨٢٦والحسن بن عيسى، مولى ابن المبارك عن ابن المبارك، عن معمر، عن عمرو بن…٢٨٢٧حدثنا القعنبي، حدثنا ابن المبارك، ح وحدثنا مسدد، حدثنا هشيم، عن…٢٨٢٨ذكاة الجنين ذكاة أمه٢٨٢٩ومحاضر، - المعنى - عن هشام بن عروة، عن أبيه، عن عائشة، ولم يذكرا عن…٢٨٣٠اذبحوا لله في أى شهر كان وبروا الله عز وجل وأطعموا٢٨٣١لا فرع ولا عتيرة٢٨٣٢قال الفرع أول النتاج كان ينتج لهم فيذبحونه .٢٨٣٣حدثنا موسى بن إسماعيل، حدثنا حماد، عن عبد الله بن عثمان بن خثيم، عن…٢٨٣٤عن الغلام شاتان مكافئتان وعن الجارية شاة٢٨٣٥أقروا الطير على مكناتها٢٨٣٦عن الغلام شاتان مثلان وعن الجارية شاة٢٨٣٧كل غلام رهينة بعقيقته تذبح عنه يوم السابع ويحلق رأسه ويدمى٢٨٣٨كل غلام رهينة بعقيقته تذبح عنه يوم سابعه ويحلق ويسمى٢٨٣٩مع الغلام عقيقته فأهريقوا عنه دما وأميطوا عنه الأذى٢٨٤٠أنه كان يقول إماطة الأذى حلق الرأس .٢٨٤١حدثنا أبو معمر عبد الله بن عمرو، حدثنا عبد الوارث، حدثنا أيوب، عن…٢٨٤٢لا يحب الله العقوق٢٨٤٣حدثنا أحمد بن محمد بن ثابت، حدثنا علي بن الحسين، حدثني أبي، حدثنا عبد…
