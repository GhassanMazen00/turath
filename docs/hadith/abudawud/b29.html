سنن أبي داود

كتاب الطب

٤٩ حديثًا
٣٨٥٥تداووا فإن الله عز وجل لم يضع داء إلا وضع له دواء غير داء واحد الهرم٣٨٥٦حدثنا هارون بن عبد الله، حدثنا أبو داود، وأبو عامر - وهذا لفظ أبي…٣٨٥٧إن كان في شىء مما تداويتم به خير فالحجامة٣٨٥٨سلمى خادم رسول الله صلى الله عليه وسلم قالت ما كان أحد يشتكي إلى رسول…٣٨٥٩من أهراق من هذه الدماء فلا يضره أن لا يتداوى بشىء لشىء٣٨٦٠أن النبي صلى الله عليه وسلم احتجم ثلاثا في الأخدعين والكاهل . قال…٣٨٦١من احتجم لسبع عشرة وتسع عشرة وإحدى وعشرين كان شفاء من كل داء٣٨٦٢بكار بن عبد العزيز أخبرتني عمتي، كبشة بنت أبي بكرة - وقال غير موسى…٣٨٦٣أن النبي صلى الله عليه وسلم احتجم على وركه من وثء كان به .٣٨٦٤قال بعث النبي صلى الله عليه وسلم إلى أبى طبيبا فقطع منه عرقا .٣٨٦٥حدثنا موسى بن إسماعيل، حدثنا حماد، عن ثابت، عن مطرف، عن عمران بن…٣٨٦٦أن النبي صلى الله عليه وسلم كوى سعد بن معاذ من رميته .٣٨٦٧أن رسول الله صلى الله عليه وسلم استعط .٣٨٦٨هو من عمل الشيطان٣٨٦٩ما أبالي ما أتيت إن أنا شربت ترياقا أو تعلقت تميمة أو قلت الشعر من…٣٨٧٠حدثنا هارون بن عبد الله، حدثنا محمد بن بشر، حدثنا يونس بن أبي إسحاق،…٣٨٧١أن طبيبا، سأل النبي صلى الله عليه وسلم عن ضفدع يجعلها في دواء فنهاه…٣٨٧٢من حسا سما فسمه في يده يتحساه في نار جهنم خالدا مخلدا فيها أبدا٣٨٧٣حدثنا مسلم بن إبراهيم، حدثنا شعبة، عن سماك، عن علقمة بن وائل، عن…٣٨٧٤إن الله أنزل الداء والدواء وجعل لكل داء دواء فتداووا ولا تداووا بحرام٣٨٧٥إنك رجل مفئود ائت الحارث بن كلدة أخا ثقيف فإنه رجل يتطبب فليأخذ سبع…٣٨٧٦من تصبح سبع تمرات عجوة لم يضره ذلك اليوم سم ولا سحر٣٨٧٧علام تدغرن أولادكن بهذا العلاق عليكن بهذا العود الهندي فإن فيه سبعة…٣٨٧٨البسوا من ثيابكم البياض فإنها من خير ثيابكم وكفنوا فيها موتاكم وإن…٣٨٧٩حدثنا أحمد بن حنبل، حدثنا عبد الرزاق، حدثنا معمر، عن همام بن منبه،…٣٨٨٠رضى الله عنها قالت كان يؤمر العائن فيتوضأ ثم يغتسل منه المعين .٣٨٨١لا تقتلوا أولادكم سرا فإن الغيل يدرك الفارس فيدعثره عن فرسه٣٨٨٢لقد هممت أن أنهى عن الغيلة حتى ذكرت أن الروم وفارس يفعلون ذلك فلا يضر…٣٨٨٣إن الرقى والتمائم والتولة شرك٣٨٨٤لا رقية إلا من عين أو حمة٣٨٨٥اكشف الباس رب الناس٣٨٨٦اعرضوا على رقاكم لا بأس بالرقى ما لم تكن شركا٣٨٨٧ألا تعلمين هذه رقية النملة كما علمتيها الكتابة٣٨٨٨مروا أبا ثابت يتعوذ٣٨٨٩لا رقية إلا من عين أو حمة أو دم يرقأ٣٨٩٠اللهم رب الناس مذهب الباس اشف أنت الشافي لا شافي إلا أنت اشفه شفاء لا…٣٨٩١امسحه بيمينك سبع مرات وقل أعوذ بعزة الله وقدرته من شر ما أجد٣٨٩٢من اشتكى منكم شيئا أو اشتكاه أخ له فليقل ربنا الله الذي في السماء…٣٨٩٣أعوذ بكلمات الله التامة من غضبه وشر عباده ومن همزات الشياطين وأن…٣٨٩٤قال رأيت أثر ضربة في ساق سلمة فقلت ما هذه قال أصابتني يوم خيبر فقال…٣٨٩٥تربة أرضنا بريقة بعضنا يشفى سقيمنا بإذن ربنا٣٨٩٦حدثنا مسدد، حدثنا يحيى، عن زكريا، قال حدثني عامر، عن خارجة بن الصلت…٣٨٩٧أنه مر - قال - فرقاه بفاتحة الكتاب ثلاثة أيام غدوة وعشية كلما ختمها…٣٨٩٨من أسلم قال كنت جالسا عند رسول الله صلى الله عليه وسلم فجاء رجل من…٣٨٩٩لو قال أعوذ بكلمات الله التامة من شر ما خلق لم يلدغ٣٩٠٠من أين علمتم أنها رقية أحسنتم اقتسموا واضربوا لي معكم بسهم٣٩٠١كل فلعمري من أكل برقية باطل لقد أكلت برقية حق٣٩٠٢زوج النبي صلى الله عليه وسلم أن رسول الله صلى الله عليه وسلم كان إذا…٣٩٠٣رضى الله عنها قالت أرادت أمي أن تسمني لدخولي على رسول الله صلى الله…
