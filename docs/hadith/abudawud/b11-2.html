١٨٧١أن النبي صلى الله عليه وسلم لما دخل مكة طاف بالبيت وصلى ركعتين خلف…١٨٧٢وهاشم، - يعني ابن القاسم - قالا حدثنا سليمان بن المغيرة، عن ثابت، عن…١٨٧٣أنه جاء إلى الحجر فقبله فقال إني أعلم أنك حجر لا تنفع ولا تضر ولولا…١٨٧٤قال لم أر رسول الله صلى الله عليه وسلم يمسح من البيت إلا الركنين…١٨٧٥أنه أخبر بقول، عائشة رضى الله عنها إن الحجر بعضه من البيت . فقال ابن…١٨٧٦قال كان رسول الله صلى الله عليه وسلم لا يدع أن يستلم الركن اليماني…١٨٧٧أن رسول الله صلى الله عليه وسلم طاف في حجة الوداع على بعير يستلم…١٨٧٨حدثنا مصرف بن عمرو اليامي، حدثنا يونس، - يعني ابن بكير - حدثنا ابن…١٨٧٩ومحمد بن رافع، - المعنى - قالا حدثنا أبو عاصم، عن معروف، - يعني ابن…١٨٨٠أنه سمع جابر بن عبد الله، يقول طاف النبي صلى الله عليه وسلم في حجة…١٨٨١أن رسول الله صلى الله عليه وسلم قدم مكة وهو يشتكي فطاف على راحلته…١٨٨٢طوفي من وراء الناس وأنت راكبة١٨٨٣قال طاف النبي صلى الله عليه وسلم مضطبعا ببرد أخضر .١٨٨٤أن رسول الله صلى الله عليه وسلم وأصحابه اعتمروا من الجعرانة فرملوا…١٨٨٥ارملوا بالبيت ثلاثا١٨٨٦حدثنا مسدد، حدثنا حماد بن زيد، عن أيوب، عن سعيد بن جبير، أنه حدث عن…١٨٨٧حدثنا أحمد بن حنبل، حدثنا عبد الملك بن عمرو، حدثنا هشام بن سعد، عن…١٨٨٨إنما جعل الطواف بالبيت وبين الصفا والمروة ورمى الجمار لإقامة ذكر الله١٨٨٩أن النبي صلى الله عليه وسلم اضطبع فاستلم وكبر ثم رمل ثلاثة أطواف…١٨٩٠أن رسول الله صلى الله عليه وسلم وأصحابه اعتمروا من الجعرانة فرملوا…١٨٩١أن ابن عمر، رمل من الحجر إلى الحجر وذكر أن رسول الله صلى الله عليه…١٨٩٢حدثنا مسدد، حدثنا عيسى بن يونس، حدثنا ابن جريج، عن يحيى بن عبيد، عن…١٨٩٣أن رسول الله صلى الله عليه وسلم كان إذا طاف في الحج والعمرة أول ما…١٨٩٤لا تمنعوا أحدا يطوف بهذا البيت ويصلي أى ساعة شاء من ليل أو نهار١٨٩٥يقول لم يطف النبي صلى الله عليه وسلم ولا أصحابه بين الصفا والمروة إلا…١٨٩٦أن أصحاب، رسول الله صلى الله عليه وسلم الذين كانوا معه لم يطوفوا حتى…١٨٩٧طوافك بالبيت وبين الصفا والمروة يكفيك لحجتك وعمرتك١٨٩٨قال لما فتح رسول الله صلى الله عليه وسلم مكة قلت لألبسن ثيابي - وكانت…١٨٩٩حدثنا مسدد، حدثنا عيسى بن يونس، حدثنا المثنى بن الصباح، عن عمرو بن…١٩٠٠حدثنا عبيد الله بن عمر بن ميسرة، حدثنا يحيى بن سعيد، حدثنا السائب بن…١٩٠١حدثنا القعنبي، عن مالك، عن هشام بن عروة، ح وحدثنا ابن السرح، حدثنا…١٩٠٢أن رسول الله صلى الله عليه وسلم اعتمر فطاف بالبيت وصلى خلف المقام…١٩٠٣بهذا الحديث زاد ثم أتى الصفا والمروة فسعى بينهما سبعا ثم حلق رأسه .١٩٠٤أن رجلا، قال لعبد الله بن عمر بين الصفا والمروة يا أبا عبد الرحمن إني…١٩٠٥اغتسلي واستذفري بثوب وأحرمي١٩٠٦حدثنا عبد الله بن مسلمة، حدثنا سليمان يعني ابن بلال، ح وحدثنا أحمد بن…١٩٠٧قد نحرت ها هنا ومنى كلها منحر١٩٠٨فانحروا في رحالكم١٩٠٩فذكر هذا الحديث وأدرج في الحديث عند قوله { واتخذوا من مقام إبراهيم…١٩١٠قالت كانت قريش ومن دان دينها يقفون بالمزدلفة وكانوا يسمون الحمس وكان…١٩١١قال صلى رسول الله صلى الله عليه وسلم الظهر يوم التروية والفجر يوم…١٩١٢حدثنا أحمد بن إبراهيم، حدثنا إسحاق الأزرق، عن سفيان، عن عبد العزيز بن…١٩١٣قال غدا رسول الله صلى الله عليه وسلم من منى حين صلى الصبح صبيحة يوم…١٩١٤قال لما أن قتل الحجاج ابن الزبير، أرسل إلى ابن عمر أية ساعة كان رسول…١٩١٥أو عمه قال رأيت رسول الله صلى الله عليه وسلم وهو على المنبر بعرفة .١٩١٦نبيط أنه رأى النبي صلى الله عليه وسلم واقفا بعرفة على بعير أحمر يخطب .١٩١٧وعثمان بن أبي شيبة، قالا حدثنا وكيع، عن عبد المجيد، قال حدثني العداء…١٩١٨بمعناه .١٩١٩قفوا على مشاعركم فإنكم على إرث من إرث أبيكم إبراهيم١٩٢٠أيها الناس عليكم بالسكينة فإن البر ليس بإيجاف الخيل والإبل١٩٢١حدثنا أحمد بن عبد الله بن يونس، حدثنا زهير، ح وحدثنا محمد بن كثير،…١٩٢٢السكينة أيها الناس١٩٢٣كيف كان رسول الله صلى الله عليه وسلم يسير في حجة الوداع حين دفع قال…١٩٢٤قال كنت ردف النبي صلى الله عليه وسلم فلما وقعت الشمس دفع رسول الله…١٩٢٥أنه سمعه يقول دفع رسول الله صلى الله عليه وسلم من عرفة حتى إذا كان…١٩٢٦أن رسول الله صلى الله عليه وسلم صلى المغرب والعشاء بالمزدلفة جميعا .١٩٢٧بإسناده ومعناه وقال بإقامة إقامة جمع بينهما . قال أحمد قال وكيع صلى…١٩٢٨ومعناه، قال بإقامة واحدة لكل صلاة ولم يناد في الأولى ولم يسبح على أثر…١٩٢٩قال صليت مع ابن عمر المغرب ثلاثا والعشاء ركعتين فقال له مالك بن…١٩٣٠وعبد الله بن مالك، قالا صلينا مع ابن عمر بالمزدلفة المغرب والعشاء…١٩٣١حدثنا ابن العلاء، حدثنا أبو أسامة، عن إسماعيل، عن أبي إسحاق، عن سعيد…١٩٣٢قال رأيت سعيد بن جبير أقام بجمع فصلى المغرب ثلاثا ثم صلى العشاء…١٩٣٣حدثنا مسدد، حدثنا أبو الأحوص، حدثنا أشعث بن سليم، عن أبيه، قال أقبلت…١٩٣٤أن عبد الواحد بن زياد، وأبا، عوانة وأبا معاوية حدثوهم عن الأعمش، عن…١٩٣٥هذا قزح وهو الموقف وجمع كلها موقف ونحرت ها هنا ومنى كلها منحر فانحروا…١٩٣٦وقفت ها هنا بعرفة وعرفة كلها موقف ووقفت ها هنا بجمع وجمع كلها موقف…١٩٣٧كل عرفة موقف وكل منى منحر وكل المزدلفة موقف وكل فجاج مكة طريق ومنحر١٩٣٨قال قال عمر بن الخطاب كان أهل الجاهلية لا يفيضون حتى يروا الشمس على…١٩٣٩أنه سمع ابن عباس، يقول أنا ممن، قدم رسول الله صلى الله عليه وسلم ليلة…١٩٤٠أبيني لا ترموا الجمرة حتى تطلع الشمس١٩٤١قال كان رسول الله صلى الله عليه وسلم يقدم ضعفاء أهله بغلس ويأمرهم…١٩٤٢أنها قالت أرسل النبي صلى الله عليه وسلم بأم سلمة ليلة النحر فرمت…١٩٤٣حدثنا محمد بن خلاد الباهلي، حدثنا يحيى، عن ابن جريج، أخبرني عطاء،…١٩٤٤قال أفاض رسول الله صلى الله عليه وسلم وعليه السكينة وأمرهم أن يرموا…١٩٤٥أن رسول الله صلى الله عليه وسلم وقف يوم النحر بين الجمرات في الحجة…١٩٤٦أن الحكم بن نافع، حدثهم حدثنا شعيب، عن الزهري، حدثني حميد بن عبد…١٩٤٧إن الزمان قد استدار كهيئته يوم خلق الله السموات والأرض السنة اثنا عشر…١٩٤٨حدثنا محمد بن يحيى بن فياض، حدثنا عبد الوهاب، حدثنا أيوب السختياني،…١٩٤٩الحج الحج يوم عرفة من جاء قبل صلاة الصبح من ليلة جمع فتم حجه أيام منى…١٩٥٠من أدرك معنا هذه الصلاة وأتى عرفات قبل ذلك ليلا أو نهارا فقد تم حجه…١٩٥١لينزل المهاجرون ها هنا١٩٥٢حدثنا محمد بن العلاء، حدثنا ابن المبارك، عن إبراهيم بن نافع، عن ابن…١٩٥٣حدثتني جدتي، سراء بنت نبهان - وكانت ربة بيت في الجاهلية - قالت خطبنا…١٩٥٤قال رأيت النبي صلى الله عليه وسلم يخطب الناس على ناقته العضباء يوم…١٩٥٥رسول الله صلى الله عليه وسلم بمنى يوم النحر .١٩٥٦قال رأيت رسول الله صلى الله عليه وسلم يخطب الناس بمنى حين ارتفع الضحى…١٩٥٧حدثنا مسدد، حدثنا عبد الوارث، عن حميد الأعرج، عن محمد بن إبراهيم…١٩٥٨أو أبو حريز - الشك من يحيى - أنه سمع عبد الرحمن بن فروخ، يسأل ابن عمر…١٩٥٩قال استأذن العباس رسول الله صلى الله عليه وسلم أن يبيت بمكة ليالي منى…١٩٦٠أن أبا معاوية، وحفص بن غياث، حدثاه - وحديث أبي معاوية، أتم - عن…١٩٦١أن عثمان، إنما صلى بمنى أربعا لأنه أجمع على الإقامة بعد الحج .١٩٦٢حدثنا هناد بن السري، عن أبي الأحوص، عن المغيرة، عن إبراهيم، قال إن…١٩٦٣قال لما اتخذ عثمان الأموال بالطائف وأراد أن يقيم بها صلى أربعا قال ثم…١٩٦٤أن عثمان بن عفان، أتم الصلاة بمنى من أجل الأعراب لأنهم كثروا عامئذ…١٩٦٥حدثنا النفيلي، حدثنا زهير، حدثنا أبو إسحاق، حدثني حارثة بن وهب…١٩٦٦يا أيها الناس لا يقتل بعضكم بعضا وإذا رميتم الجمرة فارموا بمثل حصى…١٩٦٧قالت رأيت رسول الله صلى الله عليه وسلم عند جمرة العقبة راكبا ورأيت…١٩٦٨بإسناده في مثل هذا الحديث زاد ولم يقم عندها .١٩٦٩أنه كان يأتي الجمار في الأيام الثلاثة بعد يوم النحر ماشيا ذاهبا…١٩٧٠لتأخذوا مناسككم فإني لا أدري لعلي لا أحج بعد حجتي هذه١٩٧١أنه سمع جابر بن عبد الله، يقول رأيت رسول الله صلى الله عليه وسلم يرمي…١٩٧٢حدثنا عبد الله بن محمد الزهري، حدثنا سفيان، عن مسعر، عن وبرة، قال…١٩٧٣وعبد الله بن سعيد، - المعنى - قالا حدثنا أبو خالد الأحمر، عن محمد بن…١٩٧٤ومسلم بن إبراهيم، - المعنى - قالا حدثنا شعبة، عن الحكم، عن إبراهيم،…١٩٧٥أن رسول الله صلى الله عليه وسلم رخص لرعاء الإبل في البيتوتة يرمون يوم…١٩٧٦ومحمد، ابنى أبي بكر عن أبيهما، عن أبي البداح بن عدي، عن أبيه، أن…١٩٧٧من أمر الجمار فقال ما أدري أرماها رسول الله صلى الله عليه وسلم بست أو…١٩٧٨إذا رمى أحدكم جمرة العقبة فقد حل له كل شىء إلا النساء١٩٧٩اللهم ارحم المحلقين١٩٨٠أن رسول الله صلى الله عليه وسلم حلق رأسه في حجة الوداع .١٩٨١حدثنا محمد بن العلاء، حدثنا حفص، عن هشام، عن ابن سيرين، عن أنس بن…١٩٨٢ابدأ بشقي الأيمن فاحلقه١٩٨٣أن النبي صلى الله عليه وسلم كان يسأل يوم منى فيقول " لا حرج " . فسأله…١٩٨٤ليس على النساء حلق إنما على النساء التقصير١٩٨٥ليس على النساء الحلق إنما على النساء التقصير١٩٨٦ويحيى بن زكريا، عن ابن جريج، عن عكرمة بن خالد، عن ابن عمر، قال اعتمر…١٩٨٧ومحمد بن إسحاق، عن عبد الله بن طاوس، عن أبيه، عن ابن عباس، قال والله…١٩٨٨أعطها فلتحج عليه فإنه في سبيل الله١٩٨٩يا أم معقل ما منعك أن تخرجي معنا١٩٩٠أما إنك لو أحججتها عليه كان في سبيل الله١٩٩١أن رسول الله صلى الله عليه وسلم اعتمر عمرتين عمرة في ذي القعدة وعمرة…١٩٩٢قال سئل ابن عمر كم اعتمر رسول الله صلى الله عليه وسلم فقال مرتين .…١٩٩٣وقتيبة، قالا حدثنا داود بن عبد الرحمن العطار، عن عمرو بن دينار، عن…١٩٩٤وهدبة بن خالد، قالا حدثنا همام، عن قتادة، عن أنس، أن رسول الله صلى…١٩٩٥يا عبد الرحمن أردف أختك عائشة فأعمرها من التنعيم فإذا هبطت بها من…١٩٩٦قال دخل النبي صلى الله عليه وسلم الجعرانة فجاء إلى المسجد فركع ما شاء…١٩٩٧أن رسول الله صلى الله عليه وسلم أقام في عمرة القضاء ثلاثا .١٩٩٨أن النبي صلى الله عليه وسلم أفاض يوم النحر ثم صلى الظهر بمنى يعني…١٩٩٩هل أفضت أبا عبد الله٢٠٠٠وابن، عباس أن النبي صلى الله عليه وسلم أخر طواف يوم النحر إلى الليل .٢٠٠١أن النبي صلى الله عليه وسلم لم يرمل في السبع الذي أفاض فيه .٢٠٠٢لا ينفرن أحد حتى يكون آخر عهده الطواف بالبيت٢٠٠٣حدثنا القعنبي، عن مالك، عن هشام بن عروة، عن أبيه، عن عائشة، أن رسول…٢٠٠٤حدثنا عمرو بن عون، أخبرنا أبو عوانة، عن يعلى بن عطاء، عن الوليد بن…٢٠٠٥- رضى الله عنها - قالت أحرمت من التنعيم بعمرة فدخلت فقضيت عمرتي…٢٠٠٦قالت خرجت معه - تعني مع النبي صلى الله عليه وسلم - في النفر الآخر…٢٠٠٧أن عبد الرحمن بن طارق، أخبره عن أمه، أن رسول الله صلى الله عليه وسلم…٢٠٠٨قالت إنما نزل رسول الله صلى الله عليه وسلم المحصب ليكون أسمح لخروجه…٢٠٠٩وعثمان بن أبي شيبة المعنى، ح وحدثنا مسدد، قالوا حدثنا سفيان، حدثنا…٢٠١٠هل ترك لنا عقيل منزلا٢٠١١أن رسول الله صلى الله عليه وسلم قال حين أراد أن ينفر من منى " نحن…٢٠١٢وأيوب، عن نافع، أن ابن عمر، كان يهجع هجعة بالبطحاء ثم يدخل مكة ويزعم…٢٠١٣وأيوب، عن نافع، عن ابن عمر، أن النبي صلى الله عليه وسلم صلى الظهر…٢٠١٤حدثنا القعنبي، عن مالك، عن ابن شهاب، عن عيسى بن طلحة بن عبيد الله، عن…٢٠١٥لا حرج لا حرج إلا على رجل اقترض عرض رجل مسلم وهو ظالم فذلك الذي حرج…٢٠١٦حدثنا أحمد بن حنبل، حدثنا سفيان بن عيينة، حدثني كثير بن كثير بن…٢٠١٧إن الله حبس عن مكة الفيل وسلط عليها رسوله والمؤمنين وإنما أحلت لي…٢٠١٨في هذه القصة قال " ولا يختلى خلاها " .٢٠١٩لا إنما هو مناخ من سبق إليه٢٠٢٠احتكار الطعام في الحرم إلحاد
