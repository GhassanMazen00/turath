سنن أبي داود

كتاب البيوع

٩٠ حديثًا
٣٣٢٦يا معشر التجار إن البيع يحضره اللغو والحلف فشوبوه بالصدقة٣٣٢٧يحضره الكذب والحلف٣٣٢٨من أين أصبت هذا الذهب٣٣٢٩إن الحلال بين وإن الحرام بين وبينهما أمور مشتبهات٣٣٣٠وبينهما مشبهات لا يعلمها كثير من الناس فمن اتقى الشبهات استبرأ عرضه…٣٣٣١ليأتين على الناس زمان لا يبقى أحد إلا أكل الربا فإن لم يأكله أصابه من…٣٣٣٢أوسع من قبل رجليه أوسع من قبل رأسه٣٣٣٣حدثنا أحمد بن يونس، حدثنا زهير، حدثنا سماك، حدثني عبد الرحمن بن عبد…٣٣٣٤ألا إن كل ربا من ربا الجاهلية موضوع لكم رءوس أموالكم لا تظلمون ولا…٣٣٣٥الحلف منفقة للسلعة ممحقة للبركة٣٣٣٦حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا سفيان، عن سماك بن حرب،…٣٣٣٧ومسلم بن إبراهيم، - المعنى قريب - قالا حدثنا شعبة، عن سماك بن حرب، عن…٣٣٣٨حدثنا ابن أبي رزمة، سمعت أبي يقول، قال رجل لشعبة خالفك سفيان . قال…٣٣٣٩قال كان سفيان أحفظ مني .٣٣٤٠الوزن وزن أهل مكة والمكيال مكيال أهل المدينة٣٣٤١ها هنا أحد من بني فلان٣٣٤٢إن أعظم الذنوب عند الله أن يلقاه بها عبد - بعد الكبائر التي نهى الله…٣٣٤٣حدثنا محمد بن المتوكل العسقلاني، حدثنا عبد الرزاق، أخبرنا معمر، عن…٣٣٤٤وقتيبة بن سعيد، عن شريك، عن سماك، عن عكرمة، رفعه - قال عثمان وحدثنا…٣٣٤٥مطل الغني ظلم وإذا أتبع أحدكم على مليء فليتبع٣٣٤٦أعطه إياه فإن خيار الناس أحسنهم قضاء٣٣٤٧قال كان لي على النبي صلى الله عليه وسلم دين فقضاني وزادني .٣٣٤٨الذهب بالذهب ربا إلا هاء وهاء والبر بالبر ربا إلا هاء وهاء والتمر…٣٣٤٩الذهب بالذهب تبرها وعينها والفضة بالفضة تبرها وعينها والبر بالبر مدى…٣٣٥٠حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، حدثنا سفيان، عن خالد، عن أبي…٣٣٥١لا حتى تميز بينه وبينه٣٣٥٢لا تباع حتى تفصل٣٣٥٣لا تبيعوا الذهب بالذهب إلا وزنا بوزن٣٣٥٤لا بأس أن تأخذها بسعر يومها ما لم تفترقا وبينكما شىء٣٣٥٥بإسناده ومعناه والأول أتم لم يذكر " بسعر يومها " .٣٣٥٦حدثنا موسى بن إسماعيل، حدثنا حماد، عن قتادة، عن الحسن، عن سمرة، أن…٣٣٥٧أن رسول الله صلى الله عليه وسلم أمره أن يجهز جيشا فنفدت الإبل فأمره…٣٣٥٨وقتيبة بن سعيد الثقفي، أن الليث، حدثهم عن أبي الزبير، عن جابر، أن…٣٣٥٩أينقص الرطب إذا يبس٣٣٦٠أن أبا عياش، أخبره أنه، سمع سعد بن أبي وقاص، يقول نهى رسول الله صلى…٣٣٦١حدثنا أبو بكر بن أبي شيبة، حدثنا ابن أبي زائدة، عن عبيد الله، عن…٣٣٦٢أن النبي صلى الله عليه وسلم رخص في بيع العرايا بالتمر والرطب .٣٣٦٣حدثنا عثمان بن أبي شيبة، حدثنا ابن عيينة، عن يحيى بن سعيد، عن بشير بن…٣٣٦٤حدثنا عبد الله بن مسلمة، حدثنا مالك، عن داود بن الحصين، عن مولى ابن…٣٣٦٥أنه قال العرية الرجل يعري الرجل النخلة أو الرجل يستثني من ماله النخلة…٣٣٦٦قال العرايا أن يهب الرجل، للرجل النخلات فيشق عليه أن يقوم عليها…٣٣٦٧حدثنا عبد الله بن مسلمة القعنبي، عن مالك، عن نافع، عن عبد الله بن…٣٣٦٨حدثنا عبد الله بن محمد النفيلي، حدثنا ابن علية، عن أيوب، عن نافع، عن…٣٣٦٩حدثنا حفص بن عمر النمري، حدثنا شعبة، عن يزيد بن خمير، عن مولى، لقريش…٣٣٧٠يقول نهى رسول الله صلى الله عليه وسلم أن تباع الثمرة حتى تشقح . قيل…٣٣٧١حدثنا الحسن بن علي، حدثنا أبو الوليد، عن حماد بن سلمة، عن حميد، عن…٣٣٧٢فإما لا فلا تتبايعوا الثمرة حتى يبدو صلاحها٣٣٧٣حدثنا إسحاق بن إسماعيل الطالقاني، حدثنا سفيان، عن ابن جريج، عن عطاء،…٣٣٧٤ويحيى بن معين، قالا حدثنا سفيان، عن حميد الأعرج، عن سليمان بن عتيق،…٣٣٧٥وسعيد بن ميناء، عن جابر بن عبد الله، أن النبي صلى الله عليه وسلم نهى…٣٣٧٦وعثمان، ابنا أبي شيبة قالا حدثنا ابن إدريس، عن عبيد الله، عن أبي…٣٣٧٧وأحمد بن عمرو بن السرح، - وهذا لفظه - قالا حدثنا سفيان، عن الزهري، عن…٣٣٧٨حدثنا الحسن بن علي، حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن عطاء…٣٣٧٩أن أبا سعيد الخدري، قال نهى رسول الله صلى الله عليه وسلم بمعنى حديث…٣٣٨٠حدثنا عبد الله بن مسلمة، عن مالك، عن نافع، عن عبد الله بن عمر، أن…٣٣٨١حدثنا أحمد بن حنبل، حدثنا يحيى، عن عبيد الله، عن نافع، عن ابن عمر، عن…٣٣٨٢حدثنا محمد بن عيسى، حدثنا هشيم، أخبرنا صالح أبو عامر، - قال أبو داود…٣٣٨٣إن الله يقول أنا ثالث الشريكين، ما لم يخن أحدهما صاحبه فإذا خانه خرجت…٣٣٨٤- يعني ابن أبي الجعد البارقي - قال أعطاه النبي صلى الله عليه وسلم…٣٣٨٥بهذا الخبر ولفظه مختلف .٣٣٨٦أن رسول الله صلى الله عليه وسلم بعث معه بدينار يشتري له أضحية…٣٣٨٧من استطاع منكم أن يكون مثل صاحب فرق الأرز فليكن مثله٣٣٨٨وسعد، فيما نصيب يوم بدر قال فجاء سعد بأسيرين ولم أجئ أنا وعمار بشىء .٣٣٨٩لأن يمنح أحدكم أرضه خير من أن يأخذ عليها خراجا معلوما٣٣٩٠إن كان هذا شأنكم فلا تكروا المزارع٣٣٩١حدثنا عثمان بن أبي شيبة، حدثنا يزيد بن هارون، أخبرنا إبراهيم بن سعد،…٣٣٩٢حدثنا إبراهيم بن موسى الرازي، أخبرنا عيسى، حدثنا الأوزاعي، ح وحدثنا…٣٣٩٣حدثنا قتيبة بن سعيد، عن مالك، عن ربيعة بن أبي عبد الرحمن، عن حنظلة بن…٣٣٩٤أن ابن عمر، كان يكري أرضه حتى بلغه أن رافع بن خديج الأنصاري حدث أن…٣٣٩٥من كانت له أرض فليزرعها أو فليزرعها أخاه ولا يكاريها بثلث ولا بربع…٣٣٩٦بمعنى إسناد عبيد الله وحديثه .٣٣٩٧حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، حدثنا عمر بن ذر، عن مجاهد، عن…٣٣٩٨من استغنى عن أرضه فليمنحها أخاه أو ليدع٣٣٩٩ما أحسن زرع ظهير٣٤٠٠إنما يزرع ثلاثة رجل له أرض فهو يزرعها ورجل منح أرضا فهو يزرع ما منح…٣٤٠١قال أبو داود قرأت على سعيد بن يعقوب الطالقاني قلت له حدثكم ابن…٣٤٠٢لمن الزرع ولمن الأرض٣٤٠٣من زرع في أرض قوم بغير إذنهم فليس له من الزرع شىء وله نفقته٣٤٠٤أن حمادا، وعبد الوارث، حدثاهم كلهم، عن أيوب، عن أبي الزبير، - قال عن…٣٤٠٥حدثنا أبو حفص، عمر بن يزيد السياري حدثنا عباد بن العوام، عن سفيان بن…٣٤٠٦من لم يذر المخابرة فليأذن بحرب من الله ورسوله٣٤٠٧حدثنا أبو بكر بن أبي شيبة، حدثنا عمر بن أيوب، عن جعفر بن برقان، عن…٣٤٠٨أن النبي صلى الله عليه وسلم عامل أهل خيبر بشطر ما يخرج من ثمر أو زرع .٣٤٠٩أن النبي صلى الله عليه وسلم دفع إلى يهود خيبر نخل خيبر وأرضها على أن…٣٤١٠حدثنا أيوب بن محمد الرقي، حدثنا عمر بن أيوب، حدثنا جعفر بن برقان، عن…٣٤١١بإسناده ومعناه قال فحزر وقال عند قوله وكل صفراء وبيضاء يعني الذهب…٣٤١٢حدثنا محمد بن سليمان الأنباري، حدثنا كثير، - يعني ابن هشام - عن جعفر…٣٤١٣- رضى الله عنها - قالت كان النبي صلى الله عليه وسلم يبعث عبد الله بن…٣٤١٤أنه قال أفاء الله على رسوله خيبر فأقرهم رسول الله صلى الله عليه وسلم…٣٤١٥ومحمد بن بكر، قالا حدثنا ابن جريج، أخبرني أبو الزبير، أنه سمع جابر بن…
