سنن أبي داود

كتاب الأدب

٥٠٢ حديثًا•صفحة ٤ من ٤
٥٢٢٣أن عبد الرحمن بن أبي ليلى، حدثه أن عبد الله بن عمر حدثه وذكر، قصة قال…٥٢٢٤حدثنا عمرو بن عون، أخبرنا خالد، عن حصين، عن عبد الرحمن بن أبي ليلى،…٥٢٢٥إن فيك خلتين يحبهما الله الحلم والأناة٥٢٢٦حدثنا موسى بن إسماعيل، حدثنا حماد، ح وحدثنا مسلم، حدثنا هشام، عن…٥٢٢٧أو غيره أن عمران بن حصين، قال كنا نقول في الجاهلية أنعم الله بك عينا…٥٢٢٨حفظك الله بما حفظت به نبيه٥٢٢٩من أحب أن يمثل له الرجال قياما فليتبوأ مقعده من النار٥٢٣٠لا تقوموا كما تقوم الأعاجم يعظم بعضها بعضا٥٢٣١عليك وعلى أبيك السلام٥٢٣٢إن جبريل يقرأ عليك السلام٥٢٣٣أن أبا عبد الرحمن الفهري، قال شهدت مع رسول الله صلى الله عليه وسلم…٥٢٣٤قال ضحك رسول الله صلى الله عليه وسلم فقال له أبو بكر أو عمر أضحك الله…٥٢٣٥ما هذا يا عبد الله٥٢٣٦وهناد، - المعنى - قالا حدثنا أبو معاوية، عن الأعمش، بإسناده بهذا قال…٥٢٣٧حدثنا أحمد بن يونس، حدثنا زهير، حدثنا عثمان بن حكيم، قال أخبرني…٥٢٣٨يا عمر اذهب فأعطهم٥٢٣٩من قطع سدرة صوب الله رأسه في النار٥٢٤٠وسلمة، - يعني ابن شبيب - قالا حدثنا عبد الرزاق، أخبرنا معمر، عن عثمان…٥٢٤١وحميد بن مسعدة، قالا حدثنا حسان بن إبراهيم، قال سألت هشام بن عروة عن…٥٢٤٢في الإنسان ثلاثمائة وستون مفصلا فعليه أن يتصدق عن كل مفصل منه بصدقة٥٢٤٣يصبح على كل سلامى من ابن آدم صدقة تسليمه على من لقي صدقة وأمره…٥٢٤٤بهذا الحديث وذكر النبي صلى الله عليه وسلم في وسطه .٥٢٤٥نزع رجل لم يعمل خيرا قط غصن شوك عن الطريق إما كان في شجرة فقطعه…٥٢٤٦لا تتركوا النار في بيوتكم حين تنامون٥٢٤٧إذا نمتم فأطفئوا سرجكم فإن الشيطان يدل مثل هذه على هذا فتحرقكم٥٢٤٨ما سالمناهن منذ حاربناهن ومن ترك شيئا منهن خيفة فليس منا٥٢٤٩اقتلوا الحيات كلهن فمن خاف ثأرهن فليس مني٥٢٥٠من ترك الحيات مخافة طلبهن فليس منا ما سالمناهن منذ حاربناهن٥٢٥١حدثنا أحمد بن منيع، حدثنا مروان بن معاوية، عن موسى الطحان، قال حدثنا…٥٢٥٢اقتلوا الحيات وذا الطفيتين والأبتر فإنهما يلتمسان البصر ويسقطان الحبل٥٢٥٣حدثنا القعنبي، عن مالك، عن نافع، عن أبي لبابة، أن رسول الله صلى الله…٥٢٥٤أن ابن عمر، وجد بعد ذلك - يعني بعد ما حدثه أبو لبابة - حية في داره…٥٢٥٥وأحمد بن سعيد الهمداني، قالا أخبرنا ابن وهب، قال أخبرني أسامة، عن…٥٢٥٦إن الهوام من الجن فمن رأى في بيته شيئا فليحرج عليه ثلاث مرات فإن عاد…٥٢٥٧استغفروا لصاحبكم٥٢٥٨فليؤذنه ثلاثا فإن بدا له بعد فليقتله فإنه شيطان٥٢٥٩فآذنوه ثلاثة أيام فإن بدا لكم بعد ذلك فاقتلوه فإنما هو شيطان٥٢٦٠إذا رأيتم منهن شيئا في مساكنكم فقولوا أنشدكن العهد الذي أخذ عليكن نوح…٥٢٦١أنه قال اقتلوا الحيات كلها إلا الجان الأبيض الذي كأنه قضيب فضة . قال…٥٢٦٢قال أمر رسول الله صلى الله عليه وسلم بقتل الوزغ وسماه فويسقا .٥٢٦٣من قتل وزغة في أول ضربة فله كذا وكذا حسنة ومن قتلها في الضربة الثانية…٥٢٦٤في أول ضربة سبعين حسنة٥٢٦٥نزل نبي من الأنبياء تحت شجرة فلدغته نملة فأمر بجهازه فأخرج من تحتها…٥٢٦٦إن نملة قرصت نبيا من الأنبياء فأمر بقرية النمل فأحرقت فأوحى الله إليه…٥٢٦٧حدثنا أحمد بن حنبل، حدثنا عبد الرزاق، حدثنا معمر، عن الزهري، عن عبيد…٥٢٦٨من فجع هذه بولدها ردوا ولدها إليها٥٢٦٩أن طبيبا، سأل النبي صلى الله عليه وسلم عن ضفدع يجعلها في دواء فنهاه…٥٢٧٠إنه لا يصيد صيدا ولا ينكأ عدوا وإنما يفقأ العين ويكسر السن٥٢٧١لا تنهكي فإن ذلك أحظى للمرأة وأحب إلى البعل٥٢٧٢استأخرن فإنه ليس لكن أن تحققن الطريق عليكن بحافات الطريق٥٢٧٣أن النبي صلى الله عليه وسلم نهى أن يمشي - يعني الرجل - بين المرأتين .٥٢٧٤يقول الله عز وجل يؤذيني ابن آدم يسب الدهر وأنا الدهر بيدي الأمر أقلب…
