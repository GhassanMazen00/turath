سنن أبي داود

كتاب العتق

٤٣ حديثًا
٣٩٢٦المكاتب عبد ما بقي عليه من مكاتبته درهم٣٩٢٧أيما عبد كاتب على مائة أوقية فأداها إلا عشرة أواق فهو عبد وأيما عبد…٣٩٢٨إن كان لإحداكن مكاتب فكان عنده ما يؤدي فلتحتجب منه٣٩٢٩ابتاعي فأعتقي فإنما الولاء لمن أعتق٣٩٣٠ما بال رجال يقول أحدهم أعتق يا فلان والولاء لي إنما الولاء لمن أعتق٣٩٣١فهل لك إلى ما هو خير منه٣٩٣٢قال كنت مملوكا لأم سلمة فقالت أعتقك وأشترط عليك أن تخدم رسول الله صلى…٣٩٣٣أن رجلا، أعتق شقصا له من غلام فذكر ذلك للنبي صلى الله عليه وسلم فقال…٣٩٣٤أن رجلا، أعتق شقصا له من غلام فأجاز النبي صلى الله عليه وسلم عتقه…٣٩٣٥من أعتق مملوكا بينه وبين آخر فعليه خلاصه٣٩٣٦من أعتق نصيبا له في مملوك عتق من ماله إن كان له مال٣٩٣٧من أعتق شقيصا في مملوكه فعليه أن يعتقه كله إن كان له مال وإلا استسعي…٣٩٣٨من أعتق شقصا له - أو شقيصا له - في مملوك فخلاصه عليه في ماله إن كان…٣٩٣٩وابن أبي عدي، عن سعيد، بإسناده ومعناه . قال أبو داود ورواه روح بن…٣٩٤٠من أعتق شركا له في مملوك أقيم عليه قيمة العدل فأعطى شركاءه حصصهم…٣٩٤١فقد عتق منه ما عتق٣٩٤٢حدثنا سليمان بن داود، حدثنا حماد، عن أيوب، عن نافع، عن ابن عمر، عن…٣٩٤٣من أعتق شركا من مملوك له فعليه عتقه كله إن كان له ما يبلغ ثمنه وإن لم…٣٩٤٤حدثنا مخلد بن خالد، حدثنا يزيد بن هارون، أخبرنا يحيى بن سعيد، عن…٣٩٤٥وإلا فقد عتق منه ما عتق٣٩٤٦من أعتق شركا له في عبد عتق منه ما بقي في ماله إذا كان له ما يبلغ ثمن…٣٩٤٧إذا كان العبد بين اثنين فأعتق أحدهما نصيبه فإن كان موسرا يقوم عليه…٣٩٤٨أن رجلا، أعتق نصيبا له من مملوك فلم يضمنه النبي صلى الله عليه وسلم .…٣٩٤٩من ملك ذا رحم محرم فهو حر٣٩٥٠أن عمر بن الخطاب، - رضى الله عنه - قال من ملك ذا رحم محرم فهو حر .٣٩٥١قال من ملك ذا رحم محرم فهو حر .٣٩٥٢والحسن، مثله . قال أبو داود سعيد أحفظ من حماد .٣٩٥٣حدثنا عبد الله بن محمد النفيلي، حدثنا محمد بن سلمة، عن محمد بن إسحاق،…٣٩٥٤حدثنا موسى بن إسماعيل، حدثنا حماد، عن قيس، عن عطاء، عن جابر بن عبد…٣٩٥٥وإسماعيل بن أبي خالد، عن سلمة بن كهيل، عن عطاء، عن جابر بن عبد الله،…٣٩٥٦أنت أحق بثمنه والله أغنى عنه٣٩٥٧أن رجلا، من الأنصار يقال له أبو مذكور أعتق غلاما له يقال له يعقوب عن…٣٩٥٨أن رجلا، أعتق ستة أعبد عند موته ولم يكن له مال غيرهم فبلغ ذلك النبي…٣٩٥٩بإسناده ومعناه ولم يقل فقال له قولا شديدا .٣٩٦٠لو شهدته قبل أن يدفن لم يدفن في مقابر المسلمين٣٩٦١وأيوب، عن محمد بن سيرين، عن عمران بن حصين، أن رجلا، أعتق ستة أعبد عند…٣٩٦٢من أعتق عبدا وله مال فمال العبد له إلا أن يشترطه السيد٣٩٦٣ولد الزنا شر الثلاثة٣٩٦٤أعتقوا عنه يعتق الله بكل عضو منه عضوا منه من النار٣٩٦٥من بلغ بسهم في سبيل الله عز وجل فله درجة٣٩٦٦من أعتق رقبة مؤمنة كانت فداءه من النار٣٩٦٧وأيما امرئ أعتق مسلما وأيما امرأة أعتقت امرأة مسلمة٣٩٦٨مثل الذي يعتق عند الموت كمثل الذي يهدي إذا شبع
