سنن أبي داود

كتاب صلاة السفر

٥٢ حديثًا
١١٩٨- رضى الله عنها - قالت فرضت الصلاة ركعتين ركعتين في الحضر والسفر…١١٩٩صدقة تصدق الله بها عليكم فاقبلوا صدقته١٢٠٠ومحمد بن بكر، قالا أخبرنا ابن جريج، سمعت عبد الله بن أبي عمار، يحدث…١٢٠١فقال أنس كان رسول الله صلى الله عليه وسلم إذا خرج مسيرة ثلاثة أميال…١٢٠٢وإبراهيم بن ميسرة، سمعا أنس بن مالك، يقول صليت مع رسول الله صلى الله…١٢٠٣يعجب ربكم من راعي غنم في رأس شظية بجبل يؤذن بالصلاة ويصلي فيقول الله…١٢٠٤حدثنا مسدد، حدثنا أبو معاوية، عن المسحاج بن موسى، قال قلت لأنس بن…١٢٠٥يقول كان رسول الله صلى الله عليه وسلم إذا نزل منزلا لم يرتحل حتى يصلي…١٢٠٦عامر بن واثلة أن معاذ بن جبل، أخبرهم أنهم، خرجوا مع رسول الله صلى…١٢٠٧أن ابن عمر، استصرخ على صفية وهو بمكة فسار حتى غربت الشمس وبدت النجوم…١٢٠٨والليث بن سعد، عن هشام بن سعد، عن أبي الزبير، عن أبي الطفيل، عن معاذ…١٢٠٩حدثنا قتيبة، حدثنا عبد الله بن نافع، عن أبي مودود، عن سليمان بن أبي…١٢١٠حدثنا القعنبي، عن مالك، عن أبي الزبير المكي، عن سعيد بن جبير، عن عبد…١٢١١قال جمع رسول الله صلى الله عليه وسلم بين الظهر والعصر والمغرب والعشاء…١٢١٢وعبد الله بن واقد، أن مؤذن ابن عمر، قال الصلاة . قال سر سر . حتى إذا…١٢١٣قال حتى إذا كان عند ذهاب الشفق نزل فجمع بينهما .١٢١٤ومسدد، قالا حدثنا حماد بن زيد، ح وحدثنا عمرو بن عون، أخبرنا حماد بن…١٢١٥أن رسول الله صلى الله عليه وسلم غابت له الشمس بمكة فجمع بينهما بسرف .١٢١٦قال بينهما عشرة أميال يعني بين مكة وسرف .١٢١٧حدثنا عبد الملك بن شعيب، حدثنا ابن وهب، عن الليث، قال قال ربيعة -…١٢١٨وابن، موهب - المعنى - قالا حدثنا المفضل، عن عقيل، عن ابن شهاب، عن أنس…١٢١٩بهذا الحديث بإسناده قال ويؤخر المغرب حتى يجمع بينها وبين العشاء حين…١٢٢٠أن النبي صلى الله عليه وسلم كان في غزوة تبوك إذا ارتحل قبل أن تزيغ…١٢٢١حدثنا حفص بن عمر، حدثنا شعبة، عن عدي بن ثابت، عن البراء، قال خرجنا مع…١٢٢٢قال صحبت رسول الله صلى الله عليه وسلم ثمانية عشر سفرا فما رأيته ترك…١٢٢٣حدثنا القعنبي، حدثنا عيسى بن حفص بن عاصم بن عمر بن الخطاب، عن أبيه،…١٢٢٤قال كان رسول الله صلى الله عليه وسلم يسبح على الراحلة أى وجه توجه…١٢٢٥أن رسول الله صلى الله عليه وسلم كان إذا سافر فأراد أن يتطوع استقبل…١٢٢٦أنه قال رأيت رسول الله صلى الله عليه وسلم يصلي على حمار وهو متوجه إلى…١٢٢٧- قال - بعثني رسول الله صلى الله عليه وسلم في حاجة قال فجئت وهو يصلي…١٢٢٨أنه سأل عائشة - رضى الله عنها - هل رخص للنساء أن يصلين على الدواب…١٢٢٩يا أهل البلد صلوا أربعا فإنا قوم سفر١٢٣٠وعثمان بن أبي شيبة، - المعنى واحد - قالا حدثنا حفص، عن عاصم، عن…١٢٣١حدثنا النفيلي، حدثنا محمد بن سلمة، عن محمد بن إسحاق، عن الزهري، عن…١٢٣٢أن رسول الله صلى الله عليه وسلم أقام بمكة سبع عشرة يصلي ركعتين .١٢٣٣ومسلم بن إبراهيم، - المعنى - قالا حدثنا وهيب، حدثني يحيى بن أبي…١٢٣٤وابن المثنى، - وهذا لفظ ابن المثنى - قالا حدثنا أبو أسامة، - قال ابن…١٢٣٥قال أقام رسول الله صلى الله عليه وسلم بتبوك عشرين يوما يقصر الصلاة .…١٢٣٦حدثنا سعيد بن منصور، حدثنا جرير بن عبد الحميد، عن منصور، عن مجاهد، عن…١٢٣٧أن النبي صلى الله عليه وسلم صلى بأصحابه في خوف فجعلهم خلفه صفين فصلى…١٢٣٨حدثنا القعنبي، عن مالك، عن يزيد بن رومان، عن صالح بن خوات، عمن صلى مع…١٢٣٩أن سهل بن أبي حثمة الأنصاري، حدثه أن صلاة الخوف أن يقوم الإمام وطائفة…١٢٤٠وابن، لهيعة قالا أخبرنا أبو الأسود، أنه سمع عروة بن الزبير، يحدث عن…١٢٤١ومحمد بن الأسود، عن عروة بن الزبير، عن أبي هريرة، قال خرجنا مع رسول…١٢٤٢أن عروة بن الزبير، حدثه أن عائشة حدثته بهذه القصة، قالت كبر رسول الله…١٢٤٣حدثنا مسدد، حدثنا يزيد بن زريع، عن معمر، عن الزهري، عن سالم، عن ابن…١٢٤٤حدثنا عمران بن ميسرة، حدثنا ابن فضيل، حدثنا خصيف، عن أبي عبيدة، عن…١٢٤٥حدثنا تميم بن المنتصر، أخبرنا إسحاق، - يعني ابن يوسف - عن شريك، عن…١٢٤٦حدثنا مسدد، حدثنا يحيى، عن سفيان، حدثني الأشعث بن سليم، عن الأسود بن…١٢٤٧وسعيد بن منصور، قالا حدثنا أبو عوانة، عن بكير بن الأخنس، عن مجاهد، عن…١٢٤٨حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا الأشعث، عن الحسن، عن أبي…١٢٤٩حدثنا أبو معمر عبد الله بن عمرو، حدثنا عبد الوارث، حدثنا محمد بن…
