سنن أبي داود

كتاب النكاح

١٢٩ حديثًا
٢٠٤٦من استطاع منكم الباءة فليتزوج فإنه أغض للبصر وأحصن للفرج ومن لم يستطع…٢٠٤٧تنكح النساء لأربع لمالها ولحسبها ولجمالها ولدينها فاظفر بذات الدين…٢٠٤٨قال قال لي رسول الله صلى الله عليه وسلم " أتزوجت " . قلت نعم . قال "…٢٠٤٩قال جاء رجل إلى النبي صلى الله عليه وسلم فقال إن امرأتي لا تمنع يد…٢٠٥٠قال جاء رجل إلى النبي صلى الله عليه وسلم فقال إني أصبت امرأة ذات حسب…٢٠٥١أن مرثد بن أبي مرثد الغنوي، كان يحمل الأسارى بمكة وكان بمكة بغي يقال…٢٠٥٢لا ينكح الزاني المجلود إلا مثله٢٠٥٣من أعتق جاريته وتزوجها كان له أجران٢٠٥٤وعبد العزيز بن صهيب، عن أنس بن مالك، أن النبي صلى الله عليه وسلم أعتق…٢٠٥٥يحرم من الرضاعة ما يحرم من الولادة٢٠٥٦أن أم حبيبة، قالت يا رسول الله هل لك في أختي قال " فأفعل ماذا " .…٢٠٥٧إنه عمك فليلج عليك٢٠٥٨انظرن من إخوانكن فإنما الرضاعة من المجاعة٢٠٥٩أن سليمان بن المغيرة، حدثهم عن أبي موسى، عن أبيه، عن ابن لعبد الله بن…٢٠٦٠حدثنا محمد بن سليمان الأنباري، حدثنا وكيع، عن سليمان بن المغيرة، عن…٢٠٦١حدثنا أحمد بن صالح، حدثنا عنبسة، حدثني يونس، عن ابن شهاب، حدثني عروة…٢٠٦٢أنها قالت كان فيما أنزل الله عز وجل من القرآن عشر رضعات يحرمن ثم نسخن…٢٠٦٣لا تحرم المصة ولا المصتان٢٠٦٤الغرة العبد أو الأمة٢٠٦٥لا تنكح المرأة على عمتها ولا العمة على بنت أخيها ولا المرأة على…٢٠٦٦أنه سمع أبا هريرة، يقول نهى رسول الله صلى الله عليه وسلم أن يجمع بين…٢٠٦٧حدثنا عبد الله بن محمد النفيلي، حدثنا خطاب بن القاسم، عن خصيف، عن…٢٠٦٨حدثنا أحمد بن عمرو بن السرح المصري، حدثنا ابن وهب، أخبرني يونس، عن…٢٠٦٩إن فاطمة مني وأنا أتخوف أن تفتن في دينها٢٠٧٠حدثنا محمد بن يحيى بن فارس، حدثني عبد الرزاق، أخبرنا معمر، عن الزهري،…٢٠٧١إن بني هشام بن المغيرة استأذنوني أن ينكحوا ابنتهم من علي بن أبي طالب…٢٠٧٢حدثنا مسدد بن مسرهد، حدثنا عبد الوارث، عن إسماعيل بن أمية، عن الزهري،…٢٠٧٣أن النبي صلى الله عليه وسلم حرم متعة النساء .٢٠٧٤حدثنا القعنبي، عن مالك، ح وحدثنا مسدد بن مسرهد، حدثنا يحيى، عن عبيد…٢٠٧٥أن العباس بن عبد الله بن العباس، أنكح عبد الرحمن بن الحكم ابنته…٢٠٧٦لعن الله المحلل والمحلل له٢٠٧٧حدثنا وهب بن بقية، عن خالد، عن حصين، عن عامر، عن الحارث الأعور، عن…٢٠٧٨أيما عبد تزوج بغير إذن مواليه فهو عاهر٢٠٧٩إذا نكح العبد بغير إذن مولاه فنكاحه باطل٢٠٨٠لا يخطب الرجل على خطبة أخيه٢٠٨١لا يخطب أحدكم على خطبة أخيه ولا يبيع على بيع أخيه إلا بإذنه٢٠٨٢إذا خطب أحدكم المرأة فإن استطاع أن ينظر إلى ما يدعوه إلى نكاحها فليفعل٢٠٨٣أيما امرأة نكحت بغير إذن مواليها فنكاحها باطل٢٠٨٤حدثني القعنبي، حدثنا ابن لهيعة، عن جعفر، - يعني ابن ربيعة - عن ابن…٢٠٨٥لا نكاح إلا بولي٢٠٨٦أنها كانت عند ابن جحش فهلك عنها - وكان فيمن هاجر إلى أرض الحبشة -…٢٠٨٧حدثنا محمد بن المثنى، حدثني أبو عامر، حدثنا عباد بن راشد، عن الحسن،…٢٠٨٨أيما امرأة زوجها وليان فهي للأول منهما وأيما رجل باع بيعا من رجلين…٢٠٨٩- في هذه الآية { لا يحل لكم أن ترثوا النساء كرها ولا تعضلوهن } قال…٢٠٩٠حدثنا أحمد بن محمد بن ثابت المروزي، حدثني علي بن حسين بن واقد، عن…٢٠٩١بمعناه قال فوعظ الله ذلك .٢٠٩٢لا تنكح الثيب حتى تستأمر ولا البكر إلا بإذنها٢٠٩٣تستأمر اليتيمة في نفسها فإن سكتت فهو إذنها وإن أبت فلا جواز عليها٢٠٩٤فإن بكت أو سكتت٢٠٩٥آمروا النساء في بناتهن٢٠٩٦أن جارية، بكرا أتت النبي صلى الله عليه وسلم فذكرت أن أباها زوجها وهي…٢٠٩٧حدثنا محمد بن عبيد، حدثنا حماد بن زيد، عن أيوب، عن عكرمة، عن النبي…٢٠٩٨الأيم أحق بنفسها من وليها والبكر تستأذن في نفسها وإذنها صماتها٢٠٩٩الثيب أحق بنفسها من وليها والبكر يستأمرها أبوها٢١٠٠ليس للولي مع الثيب أمر واليتيمة تستأمر وصمتها إقرارها٢١٠١ومجمع، ابنى يزيد الأنصاريين عن خنساء بنت خدام الأنصارية، أن أباها،…٢١٠٢يا بني بياضة أنكحوا أبا هند وانكحوا إليه٢١٠٣وبقرن أى النساء هي اليوم٢١٠٤أن خالته، أخبرته عن امرأة، قالت هي مصدقة امرأة صدق قالت بينا أبي في…٢١٠٥حدثنا عبد الله بن محمد النفيلي، حدثنا عبد العزيز بن محمد، حدثنا يزيد…٢١٠٦حدثنا محمد بن عبيد، حدثنا حماد بن زيد، عن أيوب، عن محمد، عن أبي…٢١٠٧أنها كانت تحت عبيد الله بن جحش فمات بأرض الحبشة فزوجها النجاشي النبي…٢١٠٨أن النجاشي، زوج أم حبيبة بنت أبي سفيان من رسول الله صلى الله عليه…٢١٠٩وحميد، عن أنس، أن رسول الله صلى الله عليه وسلم رأى عبد الرحمن بن عوف…٢١١٠من أعطى في صداق امرأة ملء كفيه سويقا أو تمرا فقد استحل٢١١١هل عندك من شىء تصدقها إياه٢١١٢ما تحفظ من القرآن٢١١٣نحو خبر سهل قال وكان مكحول يقول ليس ذلك لأحد بعد رسول الله صلى الله…٢١١٤حدثنا عثمان بن أبي شيبة، حدثنا عبد الرحمن بن مهدي، عن سفيان، عن فراس،…٢١١٥وابن، مهدي عن سفيان، عن منصور، عن إبراهيم، عن علقمة، عن عبد الله،…٢١١٦وأبي، حسان عن عبد الله بن عتبة بن مسعود، أن عبد الله بن مسعود، أتي في…٢١١٧أترضى أن أزوجك فلانة٢١١٨إن الحمد لله نستعينه ونستغفره ونعوذ به من شرور أنفسنا من يهد الله فلا…٢١١٩أن رسول الله صلى الله عليه وسلم كان إذا تشهد ذكر نحوه وقال بعد قوله "…٢١٢٠من بني سليم قال خطبت إلى النبي صلى الله عليه وسلم أمامة بنت عبد…٢١٢١حدثنا سليمان بن حرب، وأبو كامل قالا حدثنا حماد بن زيد، عن هشام بن…٢١٢٢ليس بك على أهلك هوان إن شئت سبعت لك وإن سبعت لك سبعت لنسائي٢١٢٣وعثمان بن أبي شيبة، عن هشيم، عن حميد، عن أنس بن مالك، قال لما أخذ…٢١٢٤وإسماعيل ابن علية، عن خالد الحذاء، عن أبي قلابة، عن أنس بن مالك، قال…٢١٢٥قال لما تزوج علي فاطمة قال له رسول الله صلى الله عليه وسلم " أعطها…٢١٢٦من أصحاب النبي صلى الله عليه وسلم أن عليا عليه السلام لما تزوج فاطمة…٢١٢٧مثله .٢١٢٨قالت أمرني رسول الله صلى الله عليه وسلم أن أدخل امرأة على زوجها قبل…٢١٢٩أيما امرأة نكحت على صداق أو حباء أو عدة قبل عصمة النكاح فهو لها وما…٢١٣٠بارك الله لك وبارك عليك وجمع بينكما في خير٢١٣١لها الصداق بما استحللت من فرجها والولد عبد لك فإذا ولدت٢١٣٢أن رجلا، يقال له بصرة بن أكثم نكح امرأة فذكر معناه . وزاد وفرق بينهما…٢١٣٣من كانت له امرأتان فمال إلى إحداهما جاء يوم القيامة وشقه مائل٢١٣٤اللهم هذا قسمي فيما أملك فلا تلمني فيما تملك ولا أملك٢١٣٥حدثنا أحمد بن يونس، حدثنا عبد الرحمن، - يعني ابن أبي الزناد - عن هشام…٢١٣٦ومحمد بن عيسى، - المعنى - قالا حدثنا عباد بن عباد، عن عاصم، عن معاذة،…٢١٣٧إني لا أستطيع أن أدور بينكن فإن رأيتن أن تأذن لي فأكون عند عائشة فعلتن٢١٣٨أن عروة بن الزبير، حدثه أن عائشة زوج النبي صلى الله عليه وسلم قالت…٢١٣٩إن أحق الشروط أن توفوا به ما استحللتم به الفروج٢١٤٠أرأيت لو مررت بقبري أكنت تسجد له٢١٤١إذا دعا الرجل امرأته إلى فراشه فأبت فلم تأته فبات غضبان عليها لعنتها…٢١٤٢أن تطعمها إذا طعمت وتكسوها إذا اكتسيت - أو اكتسبت - ولا تضرب الوجه…٢١٤٣ائت حرثك أنى شئت وأطعمها إذا طعمت واكسها إذا اكتسيت ولا تقبح الوجه…٢١٤٤أطعموهن مما تأكلون واكسوهن مما تكتسون ولا تضربوهن ولا تقبحوهن٢١٤٥فإن خفتم نشوزهن فاهجروهن في المضاجع٢١٤٦لا تضربوا إماء الله٢١٤٧لا يسأل الرجل فيما ضرب امرأته٢١٤٨حدثنا محمد بن كثير، أخبرنا سفيان، حدثني يونس بن عبيد، عن عمرو بن…٢١٤٩يا علي لا تتبع النظرة النظرة فإن لك الأولى وليست لك الآخرة٢١٥٠لا تباشر المرأة المرأة لتنعتها لزوجها كأنما ينظر إليها٢١٥١إن المرأة تقبل في صورة شيطان فمن وجد من ذلك شيئا فليأت أهله فإنه يضمر…٢١٥٢إن الله كتب على ابن آدم حظه من الزنا أدرك ذلك لا محالة فزنا العينين…٢١٥٣لكل ابن آدم حظه من الزنا٢١٥٤والأذن زناها الاستماع٢١٥٥أن رسول الله صلى الله عليه وسلم بعث يوم حنين بعثا إلى أوطاس فلقوا…٢١٥٦لعل صاحبها ألم بها٢١٥٧لا توطأ حامل حتى تضع ولا غير ذات حمل حتى تحيض حيضة٢١٥٨لا يحل لامرئ يؤمن بالله واليوم الآخر أن يسقي ماءه زرع غيره٢١٥٩حتى يستبرئها بحيضة٢١٦٠إذا تزوج أحدكم امرأة أو اشترى خادما فليقل اللهم إني أسألك خيرها وخير…٢١٦١لو أن أحدكم إذا أراد أن يأتي أهله قال بسم الله اللهم جنبنا الشيطان…٢١٦٢ملعون من أتى امرأته في دبرها٢١٦٣يقول إن اليهود يقولون إذا جامع الرجل أهله في فرجها من ورائها كان ولده…٢١٦٤حدثنا عبد العزيز بن يحيى أبو الأصبغ، حدثني محمد، - يعني ابن سلمة - عن…٢١٦٥جامعوهن في البيوت واصنعوا كل شىء غير النكاح٢١٦٦حدثنا مسدد، حدثنا يحيى، عن جابر بن صبح، قال سمعت خلاسا الهجري، قال…٢١٦٧ومسدد، قالا حدثنا حفص، عن الشيباني، عن عبد الله بن شداد، عن خالته،…٢١٦٨يتصدق بدينار أو بنصف دينار٢١٦٩قال إذا أصابها في الدم فدينار وإذا أصابها في انقطاع الدم فنصف دينار .٢١٧٠فلم يفعل أحدكم٢١٧١كذبت يهود لو أراد الله أن يخلقه ما استطعت أن تصرفه٢١٧٢ما عليكم أن لا تفعلوا ما من نسمة كائنة إلى يوم القيامة إلا وهي كائنة٢١٧٣اعزل عنها إن شئت فإنه سيأتيها ما قدر لها٢١٧٤من أحس الفتى الدوسي
