سنن أبي داود

كتاب الجهاد

٣١١ حديثًا•صفحة ٢ من ٣
٢٦٢٧أعجزتم إذ بعثت رجلا منكم فلم يمض لأمري أن تجعلوا مكانه من يمضي لأمري٢٦٢٨إن تفرقكم في هذه الشعاب والأودية إنما ذلكم من الشيطان٢٦٢٩قال غزوت مع نبي الله صلى الله عليه وسلم غزوة كذا وكذا فضيق الناس…٢٦٣٠حدثنا عمرو بن عثمان، حدثنا بقية، عن الأوزاعي، عن أسيد بن عبد الرحمن،…٢٦٣١يا أيها الناس لا تتمنوا لقاء العدو وسلوا الله تعالى العافية فإذا…٢٦٣٢اللهم أنت عضدي ونصيري بك أحول وبك أصول وبك أقاتل٢٦٣٣حدثنا سعيد بن منصور، حدثنا إسماعيل بن إبراهيم، أخبرنا ابن عون، قال…٢٦٣٤حدثنا موسى بن إسماعيل، حدثنا حماد، أخبرنا ثابت، عن أنس، أن النبي صلى…٢٦٣٥إذا رأيتم مسجدا أو سمعتم مؤذنا فلا تقتلوا أحدا٢٦٣٦أنه سمع جابرا، أن رسول الله صلى الله عليه وسلم قال " الحرب خدعة " .٢٦٣٧حدثنا محمد بن عبيد، حدثنا ابن ثور، عن معمر، عن الزهري، عن عبد الرحمن…٢٦٣٨حدثنا الحسن بن علي، حدثنا عبد الصمد، وأبو عامر عن عكرمة بن عمار،…٢٦٣٩أن جابر بن عبد الله، حدثهم قال كان رسول الله صلى الله عليه وسلم يتخلف…٢٦٤٠أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فإذا قالوها منعوا مني…٢٦٤١أمرت أن أقاتل الناس حتى يشهدوا أن لا إله إلا الله وأن محمدا عبده…٢٦٤٢أمرت أن أقاتل المشركين٢٦٤٣من لك بلا إله إلا الله يوم القيامة٢٦٤٤أنه أخبره أنه، قال يا رسول الله أرأيت إن لقيت رجلا من الكفار فقاتلني…٢٦٤٥أنا بريء من كل مسلم يقيم بين أظهر المشركين٢٦٤٦قال نزلت { إن يكن منكم عشرون صابرون يغلبوا مائتين } فشق ذلك على…٢٦٤٧لا بل أنتم العكارون٢٦٤٨قال نزلت في يوم بدر { ومن يولهم يومئذ دبره } .٢٦٤٩قد كان من قبلكم يؤخذ الرجل فيحفر له في الأرض ثم يؤتى بالمنشار فيجعل…٢٦٥٠انطلقوا حتى تأتوا روضة خاخ فإن بها ظعينة معها كتاب فخذوه منها…٢٦٥١حدثنا وهب بن بقية، عن خالد، عن حصين، عن سعد بن عبيدة، عن أبي عبد…٢٦٥٢إن منكم رجالا نكلهم إلى إيمانهم منهم فرات بن حيان٢٦٥٣قال أتى النبي صلى الله عليه وسلم عين من المشركين - وهو في سفر - فجلس…٢٦٥٤أن هاشم بن القاسم، وهشاما، حدثاهم قالا، حدثنا عكرمة بن عمار، قال…٢٦٥٥أن النعمان، - يعني ابن مقرن - قال شهدت رسول الله صلى الله عليه وسلم…٢٦٥٦قال كان أصحاب النبي صلى الله عليه وسلم يكرهون الصوت عند القتال .٢٦٥٧حدثنا عبيد الله بن عمر، حدثنا عبد الرحمن، عن همام، حدثني مطر، عن…٢٦٥٨حدثنا عثمان بن أبي شيبة، حدثنا وكيع، عن إسرائيل، عن أبي إسحاق، عن…٢٦٥٩من الغيرة ما يحب الله ومنها ما يبغض الله فأما التي يحبها الله فالغيرة…٢٦٦٠حدثنا موسى بن إسماعيل، حدثنا إبراهيم، - يعني ابن سعد - أخبرنا ابن…٢٦٦١- وهو حليف لبني زهرة - وكان من أصحاب أبي هريرة فذكر الحديث .٢٦٦٢إن رأيتمونا تخطفنا الطير فلا تبرحوا من مكانكم هذا حتى أرسل إليكم وإن…٢٦٦٣إذا أكثبوكم - يعني إذا غشوكم - فارموهم بالنبل واستبقوا نبلكم٢٦٦٤إذا أكثبوكم فارموهم بالنبل ولا تسلوا السيوف حتى يغشوكم٢٦٦٥قم يا حمزة قم يا علي قم يا عبيدة بن الحارث٢٦٦٦أعف الناس قتلة أهل الإيمان٢٦٦٧أن عمران، أبق له غلام فجعل لله عليه لئن قدر عليه ليقطعن يده فأرسلني…٢٦٦٨وقتيبة، - يعني ابن سعيد - قالا حدثنا الليث، عن نافع، عن عبد الله، أن…٢٦٦٩انظر علام اجتمع هؤلاء٢٦٧٠اقتلوا شيوخ المشركين واستبقوا شرخهم٢٦٧١حدثنا عبد الله بن محمد النفيلي، حدثنا محمد بن سلمة، عن محمد بن إسحاق،…٢٦٧٢حدثنا أحمد بن عمرو بن السرح، حدثنا سفيان، عن الزهري، عن عبيد الله، -…٢٦٧٣إن وجدتم فلانا فاحرقوه بالنار٢٦٧٤إن وجدتم فلانا وفلانا٢٦٧٥من فجع هذه بولدها ردوا ولدها إليها٢٦٧٦حدثنا إسحاق بن إبراهيم الدمشقي أبو النضر، حدثنا محمد بن شعيب، أخبرني…٢٦٧٧لقد عجب ربنا عز وجل من قوم يقادون إلى الجنة في السلاسل٢٦٧٨حدثنا عبد الله بن عمرو بن أبي الحجاج أبو معمر، حدثنا عبد الوارث،…٢٦٧٩ماذا عندك يا ثمامة٢٦٨٠حدثنا محمد بن عمرو الرازي، قال حدثنا سلمة، - يعني ابن الفضل - عن ابن…٢٦٨١والذي نفسي بيده إنكم لتضربونه إذا صدقكم وتدعونه إذا كذبكم هذه قريش قد…٢٦٨٢وهذا، لفظه ح وحدثنا الحسن بن علي، قال حدثنا وهب بن جرير، عن شعبة، عن…٢٦٨٣أما كان فيكم رجل رشيد يقوم إلى هذا حيث رآني كففت يدي عن بيعته فيقتله٢٦٨٤أربعة لا أؤمنهم في حل ولا حرم٢٦٨٥أن رسول الله صلى الله عليه وسلم دخل مكة عام الفتح وعلى رأسه المغفر…٢٦٨٦قال أراد الضحاك بن قيس أن يستعمل، مسروقا فقال له عمارة بن عقبة…٢٦٨٧حدثنا سعيد بن منصور، قال حدثنا عبد الله بن وهب، قال أخبرني عمرو بن…٢٦٨٨أن ثمانين، رجلا من أهل مكة هبطوا على النبي صلى الله عليه وسلم وأصحابه…٢٦٨٩لو كان مطعم بن عدي حيا ثم كلمني في هؤلاء النتنى لأطلقتهم له٢٦٩٠حدثنا أحمد بن محمد بن حنبل، قال حدثنا أبو نوح، قال أخبرنا عكرمة بن…٢٦٩١أن النبي صلى الله عليه وسلم جعل فداء أهل الجاهلية يوم بدر أربعمائة .٢٦٩٢إن رأيتم أن تطلقوا لها أسيرها وتردوا عليها الذي لها٢٦٩٣معي من ترون وأحب الحديث إلى أصدقه فاختاروا إما السبى وإما المال٢٦٩٤ردوا عليهم نساءهم وأبناءهم فمن مسك بشىء من هذا الفىء فإن له به علينا…٢٦٩٥حدثنا محمد بن المثنى، قال حدثنا معاذ بن معاذ، ح وحدثنا هارون بن عبد…٢٦٩٦حدثنا عثمان بن أبي شيبة، قال حدثنا إسحاق بن منصور، حدثنا عبد السلام…٢٦٩٧يا سلمة هب لي المرأة٢٦٩٨أن غلاما، لابن عمر أبق إلى العدو فظهر عليه المسلمون فرده رسول الله…٢٦٩٩والحسن بن علي، - المعنى - قالا حدثنا ابن نمير، عن عبيد الله، عن نافع،…٢٧٠٠ما أراكم تنتهون يا معشر قريش حتى يبعث الله عليكم من يضرب رقابكم على…٢٧٠١أن جيشا، غنموا في زمان رسول الله صلى الله عليه وسلم طعاما وعسلا فلم…٢٧٠٢والقعنبي، قالا حدثنا سليمان، عن حميد، - يعني ابن هلال - عن عبد الله…٢٧٠٣حدثنا سليمان بن حرب، قال حدثنا جرير، - يعني ابن حازم - عن يعلى بن…٢٧٠٤حدثنا محمد بن العلاء، حدثنا أبو معاوية، حدثنا أبو إسحاق الشيباني، عن…٢٧٠٥إن النهبة ليست بأحل من الميتة٢٧٠٦أن ابن حرشف الأزدي، حدثه عن القاسم، مولى عبد الرحمن عن بعض، أصحاب…٢٧٠٧حدثنا محمد بن المصفى، حدثنا محمد بن المبارك، عن يحيى بن حمزة، قال…٢٧٠٨من كان يؤمن بالله وباليوم الآخر فلا يركب دابة من فىء المسلمين حتى إذا…٢٧٠٩قال مررت فإذا أبو جهل صريع قد ضربت رجله فقلت يا عدو الله يا أبا جهل…٢٧١٠صلوا على صاحبكم٢٧١١كلا والذي نفسي بيده إن الشملة التي أخذها يوم خيبر من المغانم لم تصبها…٢٧١٢أسمعت بلالا ينادي٢٧١٣إذا وجدتم الرجل قد غل فأحرقوا متاعه واضربوه٢٧١٤حدثنا أبو صالح، محبوب بن موسى الأنطاكي قال أخبرنا أبو إسحاق، عن صالح…٢٧١٥حدثنا محمد بن عوف، قال حدثنا موسى بن أيوب، قال حدثنا الوليد بن مسلم،…٢٧١٦من كتم غالا فإنه مثله٢٧١٧من قتل قتيلا له عليه بينة فله سلبه٢٧١٨من قتل كافرا فله سلبه٢٧١٩يا خالد ما حملك على ما صنعت٢٧٢٠نحوه .٢٧٢١وخالد بن الوليد، أن رسول الله صلى الله عليه وسلم قضى بالسلب للقاتل…٢٧٢٢قال نفلني رسول الله صلى الله عليه وسلم يوم بدر سيف أبي جهل كان قتله .٢٧٢٣أن عنبسة بن سعيد، أخبره أنه، سمع أبا هريرة، يحدث سعيد بن العاص أن…٢٧٢٤وسأله، إسماعيل بن أمية فحدثناه الزهري، أنه سمع عنبسة بن سعيد القرشي،…٢٧٢٥حدثنا محمد بن العلاء، قال حدثنا أبو أسامة، حدثنا بريد، عن أبي بردة،…٢٧٢٦إن عثمان انطلق في حاجة الله وحاجة رسول الله وإني أبايع له٢٧٢٧وكذا، وذكر، أشياء وعن المملوك، أله في الفىء شىء وعن النساء، هل كن…٢٧٢٨والزهري، عن يزيد بن هرمز، قال كتب نجدة الحروري إلى ابن عباس يسأله عن…٢٧٢٩مع من خرجتن وبإذن من خرجتن٢٧٣٠حدثنا أحمد بن حنبل، حدثنا بشر، - يعني ابن المفضل - عن محمد بن زيد،…٢٧٣١قال كنت أميح أصحابي الماء يوم بدر .٢٧٣٢ويحيى بن معين، قالا حدثنا يحيى، عن مالك، عن الفضيل، عن عبد الله بن…٢٧٣٣أن رسول الله صلى الله عليه وسلم أسهم لرجل ولفرسه ثلاثة أسهم سهما له…٢٧٣٤حدثنا أحمد بن حنبل، حدثنا أبو معاوية، حدثنا عبد الله بن يزيد، حدثني…٢٧٣٥بمعناه إلا أنه قال ثلاثة نفر . زاد فكان للفارس ثلاثة أسهم .٢٧٣٦نعم والذي نفس محمد بيده إنه لفتح٢٧٣٧من فعل كذا وكذا فله من النفل كذا وكذا٢٧٣٨من قتل قتيلا فله كذا وكذا ومن أسر أسيرا فله كذا وكذا٢٧٣٩بهذا الحديث بإسناده قال فقسمها رسول الله صلى الله عليه وسلم بالسواء .…٢٧٤٠إن هذا السيف ليس لي ولا لك٢٧٤١أن الحكم بن نافع، حدثهم - المعنى، - كلهم عن شعيب بن أبي حمزة، عن…٢٧٤٢قال قال الوليد - يعني ابن مسلم - حدثت ابن المبارك، بهذا الحديث قلت…٢٧٤٣حدثنا هناد، قال حدثنا عبدة، - يعني ابن سليمان الكلابي - عن محمد بن…٢٧٤٤ويزيد بن خالد بن موهب، قالا حدثنا الليث، - المعنى - عن نافع، عن عبد…٢٧٤٥حدثنا مسدد، حدثنا يحيى، عن عبيد الله، قال حدثني نافع، عن عبد الله،…٢٧٤٦أن رسول الله صلى الله عليه وسلم قد كان ينفل بعض من يبعث من السرايا…٢٧٤٧اللهم إنهم حفاة فاحملهم اللهم إنهم عراة فاكسهم اللهم إنهم جياع فأشبعهم٢٧٤٨أنه قال كان رسول الله صلى الله عليه وسلم ينفل الثلث بعد الخمس .٢٧٤٩أن رسول الله صلى الله عليه وسلم كان ينفل الربع بعد الخمس والثلث بعد…٢٧٥٠ومحمود بن خالد الدمشقيان، - المعنى - قالا حدثنا مروان بن محمد، قال…٢٧٥١المسلمون تتكافأ دماؤهم يسعى بذمتهم أدناهم ويجير عليهم أقصاهم وهم يد…٢٧٥٢حدثنا هارون بن عبد الله، حدثنا هاشم بن القاسم، حدثنا عكرمة، حدثني…٢٧٥٣لا نفل إلا بعد الخمس٢٧٥٤بإسناده ومعناه .٢٧٥٥ولا يحل لي من غنائمكم مثل هذا إلا الخمس والخمس مردود فيكم٢٧٥٦إن الغادر ينصب له لواء يوم القيامة فيقال هذه غدرة فلان بن فلان٢٧٥٧إنما الإمام جنة يقاتل به٢٧٥٨إني لا أخيس بالعهد ولا أحبس البرد ولكن ارجع فإن كان في نفسك الذي في…٢٧٥٩من كان بينه وبين قوم عهد فلا يشد عقدة ولا يحلها حتى ينقضي أمدها أو…٢٧٦٠من قتل معاهدا في غير كنهه حرم الله عليه الجنة٢٧٦١حدثنا محمد بن عمرو الرازي، حدثنا سلمة، - يعني ابن الفضل - عن محمد بن…٢٧٦٢لولا أنك رسول لضربت عنقك٢٧٦٣قد أجرنا من أجرت وأمنا من أمنت٢٧٦٤قالت إن كانت المرأة لتجير على المؤمنين فيجوز .٢٧٦٥ما خلأت وما ذلك لها بخلق ولكن حبسها حابس الفيل٢٧٦٦ومروان بن الحكم، أنهم اصطلحوا على وضع الحرب عشر سنين يأمن فيهن الناس…٢٧٦٧ستصالحون الروم صلحا آمنا وتغزون أنتم وهم عدوا من ورائكم٢٧٦٨من لكعب بن الأشرف فإنه قد آذى الله ورسوله٢٧٦٩الإيمان قيد الفتك لا يفتك مؤمن٢٧٧٠لا إله إلا الله وحده لا شريك له له الملك وله الحمد وهو على كل شىء…٢٧٧١قال { لا يستأذنك الذين يؤمنون بالله واليوم الآخر } الآية نسختها التي…٢٧٧٢ألا تريحني من ذي الخلصة٢٧٧٣أن عبد الله بن كعب، قال سمعت كعب بن مالك، قال كان النبي صلى الله عليه…٢٧٧٤حدثنا مخلد بن خالد، حدثنا أبو عاصم، عن أبي بكرة، بكار بن عبد العزيز…٢٧٧٥إني سألت ربي وشفعت لأمتي فأعطاني ثلث أمتي فخررت ساجدا شكرا لربي ثم…٢٧٧٦ومسلم بن إبراهيم، قالا حدثنا شعبة، عن محارب بن دثار، عن جابر بن عبد…
