سنن أبي داود

كتاب الأدب

٥٠٢ حديثًا•صفحة ١ من ٤
٤٧٧٣يا أنيس اذهب حيث أمرتك٤٧٧٤حدثنا عبد الله بن مسلمة، حدثنا سليمان، - يعني ابن المغيرة - عن ثابت،…٤٧٧٥لا وأستغفر الله لا وأستغفر الله لا وأستغفر الله لا أحمل لك حتى تقيدني…٤٧٧٦إن الهدى الصالح والسمت الصالح والاقتصاد جزء من خمسة وعشرين جزءا من…٤٧٧٧من كظم غيظا - وهو قادر على أن ينفذه - دعاه الله عز وجل على رءوس…٤٧٧٨ملأه الله أمنا وإيمانا٤٧٧٩ما تعدون الصرعة فيكم٤٧٨٠إني لأعلم كلمة لو قالها لذهب عنه ما يجده من الغضب٤٧٨١إني لأعرف كلمة لو قالها هذا لذهب عنه الذي يجد أعوذ بالله من الشيطان…٤٧٨٢إذا غضب أحدكم وهو قائم فليجلس فإن ذهب عنه الغضب وإلا فليضطجع٤٧٨٣أن النبي صلى الله عليه وسلم بعث أبا ذر بهذا الحديث . قال أبو داود…٤٧٨٤إن الغضب من الشيطان وإن الشيطان خلق من النار وإنما تطفأ النار بالماء…٤٧٨٥رضى الله عنها أنها قالت ما خير رسول الله صلى الله عليه وسلم في أمرين…٤٧٨٦رضى الله عنها، قالت ما ضرب رسول الله صلى الله عليه وسلم خادما ولا…٤٧٨٧- يعني ابن الزبير - في قوله { خذ العفو } قال أمر نبي الله صلى الله…٤٧٨٨ما بال أقوام يقولون كذا وكذا٤٧٨٩لو أمرتم هذا أن يغسل ذا عنه٤٧٩٠المؤمن غر كريم والفاجر خب لئيم٤٧٩١بئس ابن العشيرة٤٧٩٢بئس أخو العشيرة٤٧٩٣يا عائشة إن من شرار الناس الذين يكرمون اتقاء ألسنتهم٤٧٩٤قال ما رأيت رجلا التقم أذن رسول الله صلى الله عليه وسلم فينحي رأسه…٤٧٩٥دعه فإن الحياء من الإيمان٤٧٩٦الحياء خير كله٤٧٩٧إن مما أدرك الناس من كلام النبوة الأولى إذا لم تستح فافعل ما شئت٤٧٩٨إن المؤمن ليدرك بحسن خلقه درجة الصائم القائم٤٧٩٩ما من شىء أثقل في الميزان من حسن الخلق٤٨٠٠أنا زعيم ببيت في ربض الجنة لمن ترك المراء وإن كان محقا وببيت في وسط…٤٨٠١لا يدخل الجنة الجواظ ولا الجعظري٤٨٠٢حق على الله عز وجل أن لا يرفع شيئا من الدنيا إلا وضعه٤٨٠٣إن حقا على الله عز وجل أن لا يرتفع شىء من الدنيا إلا وضعه٤٨٠٤إذا لقيتم المداحين فاحثوا في وجوههم التراب٤٨٠٥قطعت عنق صاحبك٤٨٠٦السيد الله تبارك وتعالى٤٨٠٧إن الله رفيق يحب الرفق ويعطي عليه ما لا يعطي على العنف٤٨٠٨يا عائشة ارفقي فإن الرفق لم يكن في شىء قط إلا زانه ولا نزع من شىء قط…٤٨٠٩من يحرم الرفق يحرم الخير كله٤٨١٠التؤدة في كل شىء إلا في عمل الآخرة٤٨١١لا يشكر الله من لا يشكر الناس٤٨١٢لا ما دعوتم الله لهم وأثنيتم عليهم٤٨١٣من أعطي عطاء فوجد فليجز به فإن لم يجد فليثن به فمن أثنى به فقد شكره…٤٨١٤من أبلى بلاء فذكره فقد شكره وإن كتمه فقد كفره٤٨١٥إياكم والجلوس بالطرقات٤٨١٦حدثنا مسدد، حدثنا بشر، - يعني ابن المفضل - حدثنا عبد الرحمن بن إسحاق،…٤٨١٧وتغيثوا الملهوف وتهدوا الضال٤٨١٨يا أم فلان اجلسي في أى نواحي السكك شئت حتى أجلس إليك٤٨١٩أن امرأة، كان في عقلها شىء بمعناه .٤٨٢٠خير المجالس أوسعها٤٨٢١إذا كان أحدكم في الشمس٤٨٢٢أنه جاء ورسول الله صلى الله عليه وسلم يخطب فقام في الشمس فأمر به فحول…٤٨٢٣ما لي أراكم عزين٤٨٢٤بهذا قال كأنه يحب الجماعة .٤٨٢٥وهناد، أن شريكا، أخبرهم عن سماك، عن جابر بن سمرة، قال كنا إذا أتينا…٤٨٢٦حدثنا موسى بن إسماعيل، حدثنا أبان، حدثنا قتادة، قال حدثني أبو مجلز،…٤٨٢٧حدثنا مسلم بن إبراهيم، حدثنا شعبة، عن عبد ربه بن سعيد، عن أبي عبد…٤٨٢٨أن محمد بن جعفر، حدثهم عن شعبة، عن عقيل بن طلحة، قال سمعت أبا الخصيب،…٤٨٢٩مثل المؤمن الذي يقرأ القرآن مثل الأترجة ريحها طيب وطعمها طيب ومثل…٤٨٣٠حدثنا مسدد، حدثنا يحيى، - المعنى - ح وحدثنا ابن معاذ، حدثنا أبي،…٤٨٣١مثل الجليس الصالح٤٨٣٢لا تصاحب إلا مؤمنا ولا يأكل طعامك إلا تقي٤٨٣٣الرجل على دين خليله فلينظر أحدكم من يخالل٤٨٣٤الأرواح جنود مجندة فما تعارف منها ائتلف وما تناكر منها اختلف٤٨٣٥بشروا ولا تنفروا ويسروا ولا تعسروا٤٨٣٦حدثنا مسدد، حدثنا يحيى، عن سفيان، قال حدثني إبراهيم بن المهاجر، عن…٤٨٣٧قال كان رسول الله صلى الله عليه وسلم إذا جلس يتحدث يكثر أن يرفع طرفه…٤٨٣٨يقول كان في كلام رسول الله صلى الله عليه وسلم ترتيل أو ترسيل .٤٨٣٩رحمها الله قالت كان كلام رسول الله صلى الله عليه وسلم كلاما فصلا…٤٨٤٠كل كلام لا يبدأ فيه ب الحمد لله فهو أجذم٤٨٤١كل خطبة ليس فيها تشهد فهي كاليد الجذماء٤٨٤٢أنزلوا الناس منازلهم٤٨٤٣إن من إجلال الله إكرام ذي الشيبة المسلم وحامل القرآن غير الغالي فيه…٤٨٤٤لا يجلس بين رجلين إلا بإذنهما٤٨٤٥لا يحل لرجل أن يفرق بين اثنين إلا بإذنهما٤٨٤٦أن رسول الله صلى الله عليه وسلم كان إذا جلس احتبى بيده . قال أبو داود…٤٨٤٧وموسى بن إسماعيل، قالا حدثنا عبد الله بن حسان العنبري، قال حدثتني…٤٨٤٨أتقعد قعدة المغضوب عليهم٤٨٤٩حدثنا مسدد، حدثنا يحيى، عن عوف، قال حدثني أبو المنهال، عن أبي برزة،…٤٨٥٠قال كان النبي صلى الله عليه وسلم إذا صلى الفجر تربع في مجلسه حتى تطلع…٤٨٥١لا ينتجي اثنان دون الثالث فإن ذلك يحزنه٤٨٥٢قال قال رسول الله صلى الله عليه وسلم مثله . قال أبو صالح فقلت لابن…٤٨٥٣إذا قام الرجل من مجلس ثم رجع إليه فهو أحق به٤٨٥٤حدثنا إبراهيم بن موسى الرازي، حدثنا مبشر الحلبي، عن تمام بن نجيح، عن…٤٨٥٥ما من قوم يقومون من مجلس لا يذكرون الله فيه إلا قاموا عن مثل جيفة…٤٨٥٦من قعد مقعدا لم يذكر الله فيه كانت عليه من الله ترة ومن اضطجع مضجعا…٤٨٥٧أن سعيد بن أبي هلال، حدثه أن سعيد بن أبي سعيد المقبري حدثه عن عبد…٤٨٥٨حدثنا أحمد بن صالح، حدثنا ابن وهب، قال قال عمرو وحدثني بنحو، ذلك عبد…٤٨٥٩سبحانك اللهم وبحمدك أشهد أن لا إله إلا أنت أستغفرك وأتوب إليك٤٨٦٠لا يبلغني أحد من أصحابي عن أحد شيئا فإني أحب أن أخرج إليكم وأنا سليم…٤٨٦١حدثنا محمد بن يحيى بن فارس، حدثنا نوح بن يزيد بن سيار المؤدب، حدثنا…٤٨٦٢لا يلدغ المؤمن من جحر واحد مرتين٤٨٦٣قال كان النبي صلى الله عليه وسلم إذا مشى كأنه يتوكأ .٤٨٦٤قال رأيت رسول الله صلى الله عليه وسلم . قلت كيف رأيته قال كان أبيض…٤٨٦٥قال نهى رسول الله صلى الله عليه وسلم أن يضع - وقال قتيبة يرفع الرجل…٤٨٦٦أنه رأى رسول الله صلى الله عليه وسلم مستلقيا - قال القعنبي - في…٤٨٦٧أن عمر بن الخطاب، رضى الله عنه وعثمان بن عفان كانا يفعلان ذلك .٤٨٦٨إذا حدث الرجل بالحديث ثم التفت فهي أمانة٤٨٦٩المجالس بالأمانة إلا ثلاثة مجالس سفك دم حرام أو فرج حرام أو اقتطاع…٤٨٧٠إن من أعظم الأمانة عند الله يوم القيامة الرجل يفضي إلى امرأته وتفضي…٤٨٧١لا يدخل الجنة قتات٤٨٧٢من شر الناس ذو الوجهين الذي يأتي هؤلاء بوجه وهؤلاء بوجه٤٨٧٣من كان له وجهان في الدنيا كان له يوم القيامة لسانان من نار٤٨٧٤ذكرك أخاك بما يكره٤٨٧٥لقد قلت كلمة لو مزجت بماء البحر لمزجته٤٨٧٦إن من أربى الربا الاستطالة في عرض المسلم بغير حق٤٨٧٧إن من أكبر الكبائر استطالة المرء في عرض رجل مسلم بغير حق ومن الكبائر…٤٨٧٨لما عرج بي مررت بقوم لهم أظفار من نحاس يخمشون وجوههم وصدورهم فقلت من…٤٨٧٩كما قال ابن المصفى .٤٨٨٠يا معشر من آمن بلسانه ولم يدخل الإيمان قلبه لا تغتابوا المسلمين ولا…٤٨٨١من أكل برجل مسلم أكلة فإن الله يطعمه مثلها من جهنم ومن كسي ثوبا برجل…٤٨٨٢كل المسلم على المسلم حرام ماله وعرضه ودمه حسب امرئ من الشر أن يحقر…٤٨٨٣من حمى مؤمنا من منافق٤٨٨٤ما من امرئ يخذل امرأ مسلما في موضع تنتهك فيه حرمته وينتقص فيه من عرضه…٤٨٨٥أتقولون هو أضل أم بعيره ألم تسمعوا إلى ما قال٤٨٨٦قال أيعجز أحدكم أن يكون، مثل أبي ضيغم - أو ضمضم شك ابن عبيد - كان إذا…٤٨٨٧أيعجز أحدكم أن يكون مثل أبي ضمضم٤٨٨٨إنك إن اتبعت عورات الناس أفسدتهم أو كدت أن تفسدهم٤٨٨٩إن الأمير إذا ابتغى الريبة في الناس أفسدهم٤٨٩٠حدثنا أبو بكر بن أبي شيبة، حدثنا أبو معاوية، عن الأعمش، عن زيد بن…٤٨٩١من رأى عورة فسترها كان كمن أحيا موءودة٤٨٩٢أنه سمع أبا الهيثم، يذكر أنه سمع دخينا، كاتب عقبة بن عامر قال كان لنا…٤٨٩٣المسلم أخو المسلم لا يظلمه ولا يسلمه من كان في حاجة أخيه فإن الله في…٤٨٩٤المستبان ما قالا فعلى البادي منهما ما لم يعتد المظلوم٤٨٩٥إن الله أوحى إلى أن تواضعوا حتى لا يبغي أحد على أحد ولا يفخر أحد على…٤٨٩٦نزل ملك من السماء يكذبه بما قال لك فلما انتصرت وقع الشيطان فلم أكن…٤٨٩٧أن رجلا، كان يسب أبا بكر وساق نحوه . قال أبو داود وكذلك رواه صفوان بن…٤٨٩٨حدثنا عبيد الله بن معاذ، حدثنا أبي ح، وحدثنا عبيد الله بن عمر بن…٤٨٩٩إذا مات صاحبكم فدعوه لا تقعوا فيه٤٩٠٠اذكروا محاسن موتاكم وكفوا عن مساويهم٤٩٠١كان رجلان في بني إسرائيل متآخيين فكان أحدهما يذنب والآخر مجتهد في…٤٩٠٢ما من ذنب أجدر أن يعجل الله تعالى لصاحبه العقوبة في الدنيا - مع ما…٤٩٠٣إياكم والحسد فإن الحسد يأكل الحسنات كما تأكل النار الحطب٤٩٠٤لا تشددوا على أنفسكم فيشدد عليكم فإن قوما شددوا على أنفسهم فشدد الله…٤٩٠٥إن العبد إذا لعن شيئا صعدت اللعنة إلى السماء فتغلق أبواب السماء دونها…٤٩٠٦لا تلاعنوا بلعنة الله ولا بغضب الله ولا بالنار٤٩٠٧لا يكون اللعانون شفعاء ولا شهداء٤٩٠٨لا تلعنها فإنها مأمورة وإنه من لعن شيئا ليس له بأهل رجعت اللعنة عليه٤٩٠٩رضى الله عنها قالت سرق لها شىء فجعلت تدعو عليه فقال لها رسول الله صلى…٤٩١٠لا تباغضوا ولا تحاسدوا ولا تدابروا وكونوا عباد الله إخوانا ولا يحل…٤٩١١لا يحل لمسلم أن يهجر أخاه فوق ثلاثة أيام يلتقيان فيعرض هذا ويعرض هذا…٤٩١٢لا يحل لمؤمن أن يهجر مؤمنا فوق ثلاث فإن مرت به ثلاث فليلقه فليسلم…٤٩١٣لا يكون لمسلم أن يهجر مسلما فوق ثلاثة فإذا لقيه سلم عليه ثلاث مرار كل…٤٩١٤لا يحل لمسلم أن يهجر أخاه فوق ثلاث فمن هجر فوق ثلاث فمات دخل النار٤٩١٥من هجر أخاه سنة فهو كسفك دمه٤٩١٦تفتح أبواب الجنة كل يوم اثنين وخميس فيغفر في ذلك اليومين لكل عبد لا…٤٩١٧إياكم والظن فإن الظن أكذب الحديث ولا تحسسوا ولا تجسسوا٤٩١٨المؤمن مرآة المؤمن والمؤمن أخو المؤمن يكف عليه ضيعته ويحوطه من ورائه٤٩١٩ألا أخبركم بأفضل من درجة الصيام والصلاة والصدقة٤٩٢٠لم يكذب من نمى بين اثنين ليصلح٤٩٢١لا أعده كاذبا الرجل يصلح بين الناس يقول القول ولا يريد به إلا الإصلاح…٤٩٢٢دعي هذه وقولي الذي كنت تقولين
