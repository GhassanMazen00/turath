سنن أبي داود

كتاب الصلاة

٧٧٠ حديثًا•صفحة ٤ من ٦
٨٤١يعمد أحدكم في صلاته فيبرك كما يبرك الجمل٨٤٢حدثنا مسدد، حدثنا إسماعيل، - يعني ابن إبراهيم - عن أيوب، عن أبي…٨٤٣حدثنا زياد بن أيوب، حدثنا إسماعيل، عن أيوب، عن أبي قلابة، قال جاءنا…٨٤٤أنه رأى النبي صلى الله عليه وسلم إذا كان في وتر من صلاته لم ينهض حتى…٨٤٥أنه سمع طاوسا، يقول قلنا لابن عباس في الإقعاء على القدمين في السجود .…٨٤٦سمع الله لمن حمده اللهم ربنا لك الحمد ملء السموات وملء الأرض وملء ما…٨٤٧اللهم ربنا لك الحمد ملء السماء٨٤٨إذا قال الإمام سمع الله لمن حمده فقولوا اللهم ربنا لك الحمد فإنه من…٨٤٩حدثنا بشر بن عمار، حدثنا أسباط، عن مطرف، عن عامر، قال لا يقول القوم…٨٥٠اللهم اغفر لي وارحمني وعافني واهدني وارزقني٨٥١من كان منكن يؤمن بالله واليوم الآخر فلا ترفع رأسها حتى يرفع الرجال…٨٥٢أن رسول الله صلى الله عليه وسلم كان سجوده وركوعه وقعوده وما بين…٨٥٣سمع الله لمن حمده٨٥٤حدثنا مسدد، وأبو كامل - دخل حديث أحدهما في الآخر - قالا حدثنا أبو…٨٥٥لا تجزئ صلاة الرجل حتى يقيم ظهره في الركوع والسجود٨٥٦ارجع فصل فإنك لم تصل٨٥٧إنه لا تتم صلاة لأحد من الناس حتى يتوضأ فيضع الوضوء٨٥٨إنها لا تتم صلاة أحدكم حتى يسبغ الوضوء كما أمره الله عز وجل فيغسل…٨٥٩إذا قمت فتوجهت إلى القبلة فكبر ثم اقرأ بأم القرآن وبما شاء الله أن…٨٦٠إذا أنت قمت في صلاتك فكبر الله تعالى ثم اقرأ ما تيسر عليك من القرآن٨٦١فتوضأ كما أمرك الله جل وعز ثم تشهد فأقم ثم كبر فإن كان معك قرآن فاقرأ…٨٦٢حدثنا أبو الوليد الطيالسي، حدثنا الليث، عن يزيد بن أبي حبيب، عن جعفر…٨٦٣حدثنا زهير بن حرب، حدثنا جرير، عن عطاء بن السائب، عن سالم البراد، قال…٨٦٤إن أول ما يحاسب الناس به يوم القيامة من أعمالهم الصلاة قال يقول ربنا…٨٦٥حدثنا موسى بن إسماعيل، حدثنا حماد، عن حميد، عن الحسن، عن رجل، من بني…٨٦٦ثم الزكاة مثل ذلك ثم تؤخذ الأعمال على حسب ذلك٨٦٧حدثنا حفص بن عمر، حدثنا شعبة، عن أبي يعفور، - قال أبو داود واسمه…٨٦٨والأسود، عن عبد الله، قال إذا ركع أحدكم فليفرش ذراعيه على فخذه وليطبق…٨٦٩اجعلوها في ركوعكم٨٧٠سبحان ربي العظيم وبحمده٨٧١سبحان ربي العظيم٨٧٢سبوح قدوس رب الملائكة والروح٨٧٣سبحان ذي الجبروت والملكوت والكبرياء والعظمة٨٧٤الله أكبر - ثلاثا - ذو الملكوت والجبروت والكبرياء والعظمة٨٧٥أقرب ما يكون العبد من ربه وهو ساجد فأكثروا الدعاء٨٧٦يا أيها الناس إنه لم يبق من مبشرات النبوة إلا الرؤيا الصالحة يراها…٨٧٧سبحانك اللهم ربنا وبحمدك اللهم اغفر لي٨٧٨اللهم اغفر لي ذنبي كله دقه وجله وأوله وآخره٨٧٩أعوذ برضاك من سخطك وأعوذ بمعافاتك من عقوبتك وأعوذ بك منك لا أحصي ثناء…٨٨٠اللهم إني أعوذ بك من عذاب القبر وأعوذ بك من فتنة المسيح الدجال وأعوذ…٨٨١أعوذ بالله من النار ويل لأهل النار٨٨٢لقد تحجرت واسعا٨٨٣سبحان ربي الأعلى٨٨٤حدثنا محمد بن المثنى، حدثني محمد بن جعفر، حدثنا شعبة، عن موسى بن أبي…٨٨٥سبحان الله وبحمده٨٨٦إذا ركع أحدكم فليقل ثلاث مرات سبحان ربي العظيم وذلك أدناه وإذا سجد…٨٨٧من قرأ منكم { والتين والزيتون } فانتهى إلى آخرها { أليس الله بأحكم…٨٨٨وابن، رافع قالا حدثنا عبد الله بن إبراهيم بن عمر بن كيسان، حدثني أبي،…٨٨٩وسليمان بن حرب، قالا حدثنا حماد بن زيد، عن عمرو بن دينار، عن طاوس، عن…٨٩٠حدثنا محمد بن كثير، أخبرنا شعبة، عن عمرو بن دينار، عن طاوس، عن ابن…٨٩١إذا سجد العبد سجد معه سبعة آراب وجهه وكفاه وركبتاه وقدماه٨٩٢إن اليدين تسجدان كما يسجد الوجه فإذا وضع أحدكم وجهه فليضع يديه وإذا…٨٩٣إذا جئتم إلى الصلاة ونحن سجود فاسجدوا ولا تعدوها شيئا ومن أدرك الركعة…٨٩٤أن رسول الله صلى الله عليه وسلم رئي على جبهته وعلى أرنبته أثر طين من…٨٩٥نحوه .٨٩٦حدثنا الربيع بن نافع أبو توبة، حدثنا شريك، عن أبي إسحاق، قال وصف لنا…٨٩٧اعتدلوا في السجود ولا يفترش أحدكم ذراعيه افتراش الكلب٨٩٨أن النبي صلى الله عليه وسلم كان إذا سجد جافى بين يديه حتى لو أن بهمة…٨٩٩قال أتيت النبي صلى الله عليه وسلم من خلفه فرأيت بياض إبطيه وهو مجخ قد…٩٠٠حدثنا مسلم بن إبراهيم، حدثنا عباد بن راشد، حدثنا الحسن، حدثنا أحمر بن…٩٠١إذا سجد أحدكم فلا يفترش يديه افتراش الكلب وليضم فخذيه٩٠٢قال اشتكى أصحاب النبي صلى الله عليه وسلم إلى النبي صلى الله عليه وسلم…٩٠٣قال صليت إلى جنب ابن عمر فوضعت يدى على خاصرتى فلما صلى قال هذا الصلب…٩٠٤قال رأيت رسول الله صلى الله عليه وسلم يصلي وفي صدره أزيز كأزيز الرحى…٩٠٥من توضأ فأحسن وضوءه ثم صلى ركعتين لا يسهو فيهما غفر له ما تقدم من ذنبه٩٠٦ما من أحد يتوضأ فيحسن الوضوء ويصلي ركعتين يقبل بقلبه ووجهه عليهما إلا…٩٠٧حديث ٩٠٧٩٠٨يا علي لا تفتح على الإمام في الصلاة٩٠٩لا يزال الله عز وجل مقبلا على العبد وهو في صلاته ما لم يلتفت فإذا…٩١٠إنما هو اختلاس يختلسه الشيطان من صلاة العبد٩١١أن رسول الله صلى الله عليه وسلم رئي على جبهته وعلى أرنبته أثر طين من…٩١٢لينتهين رجال يشخصون أبصارهم إلى السماء - قال مسدد في الصلاة - أو لا…٩١٣ما بال أقوام يرفعون أبصارهم في صلاتهم٩١٤شغلتني أعلام هذه اذهبوا بها إلى أبي جهم وأتوني بأنبجانيته٩١٥بهذا الخبر قال وأخذ كرديا كان لأبي جهم فقيل يا رسول الله الخميصة كانت…٩١٦أنه سمع أبا سلام، قال حدثني السلولي، - هو أبو كبشة - عن سهل ابن…٩١٧أن رسول الله صلى الله عليه وسلم كان يصلي وهو حامل أمامة بنت زينب بنت…٩١٨أنه سمع أبا قتادة، يقول بينا نحن في المسجد جلوس خرج علينا رسول الله…٩١٩يقول رأيت رسول الله صلى الله عليه وسلم يصلي للناس وأمامة بنت أبي…٩٢٠حدثنا يحيى بن خلف، حدثنا عبد الأعلى، حدثنا محمد، - يعني ابن إسحاق -…٩٢١اقتلوا الأسودين في الصلاة الحية والعقرب٩٢٢ومسدد، - وهذا لفظه - قال حدثنا بشر، - يعني ابن المفضل - حدثنا برد، عن…٩٢٣إن في الصلاة لشغلا٩٢٤إن الله يحدث من أمره ما يشاء وإن الله جل وعز قد أحدث من أمره أن لا…٩٢٥وقتيبة بن سعيد، أن الليث، حدثهم عن بكير، عن نابل، صاحب العباء عن ابن…٩٢٦ما فعلت في الذي أرسلتك فإنه لم يمنعني أن أكلمك إلا أني كنت أصلي٩٢٧يقول خرج رسول الله صلى الله عليه وسلم إلى قباء يصلي فيه - قال -…٩٢٨لا غرار في صلاة ولا تسليم٩٢٩لا غرار في تسليم ولا صلاة٩٣٠إن هذه الصلاة لا يحل فيها شىء من كلام الناس هذا إنما هو التسبيح…٩٣١إذا عطست فاحمد الله وإذا عطس العاطس فحمد الله فقل يرحمك الله٩٣٢قال كان رسول الله صلى الله عليه وسلم إذا قرأ { ولا الضالين } قال "…٩٣٣حدثنا مخلد بن خالد الشعيري، حدثنا ابن نمير، حدثنا علي بن صالح، عن…٩٣٤قال كان رسول الله صلى الله عليه وسلم إذا تلا { غير المغضوب عليهم ولا…٩٣٥إذا قال الإمام { غير المغضوب عليهم ولا الضالين } فقولوا٩٣٦إذا أمن الإمام فأمنوا فإنه من وافق تأمينه تأمين الملائكة غفر له ما…٩٣٧أنه قال يا رسول الله لا تسبقني " بآمين " .٩٣٨ومحمود بن خالد، قالا حدثنا الفريابي، عن صبيح بن محرز الحمصي، حدثني…٩٣٩التسبيح للرجال والتصفيق للنساء٩٤٠يا أبا بكر ما منعك أن تثبت إذ أمرتك٩٤١إن حضرت صلاة العصر ولم آتك فمر أبا بكر فليصل بالناس٩٤٢قال قوله " التصفيح للنساء " . تضرب بأصبعين من يمينها على كفها اليسرى .٩٤٣ومحمد بن رافع، قالا حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن أنس…٩٤٤قال قال رسول الله صلى الله عليه وسلم " التسبيح للرجال " . يعني في…٩٤٥إذا قام أحدكم إلى الصلاة فإن الرحمة تواجهه فلا يمسح الحصى٩٤٦لا تمسح وأنت تصلي فإن كنت لا بد فاعلا فواحدة تسوية الحصى٩٤٧حدثنا يعقوب بن كعب، حدثنا محمد بن سلمة، عن هشام، عن محمد، عن أبي…٩٤٨حدثنا عبد السلام بن عبد الرحمن الوابصي، حدثنا أبي، عن شيبان، عن حصين…٩٤٩حدثنا محمد بن عيسى، حدثنا هشيم، أخبرنا إسماعيل بن أبي خالد، عن الحارث…٩٥٠صلاة الرجل قاعدا نصف الصلاة٩٥١صلاته قائما أفضل من صلاته قاعدا وصلاته قاعدا على النصف من صلاته قائما…٩٥٢صل قائما فإن لم تستطع فقاعدا فإن لم تستطع فعلى جنب٩٥٣قالت ما رأيت رسول الله صلى الله عليه وسلم يقرأ في شىء من صلاة الليل…٩٥٤وأبي النضر، عن أبي سلمة بن عبد الرحمن، عن عائشة، زوج النبي صلى الله…٩٥٥وأيوب، يحدثان عن عبد الله بن شقيق، عن عائشة، قالت كان رسول الله صلى…٩٥٦قال سألت عائشة أكان رسول الله صلى الله عليه وسلم يقرأ السورة في ركعة…٩٥٧قال قلت لأنظرن إلى صلاة رسول الله صلى الله عليه وسلم كيف يصلي فقام…٩٥٨قال سنة الصلاة أن تنصب، رجلك اليمنى وتثني رجلك اليسرى .٩٥٩أنه سمع عبد الله بن عمر، يقول من سنة الصلاة أن تضجع رجلك اليسرى وتنصب…٩٦٠أيضا من السنة كما قال جرير .٩٦١أن القاسم بن محمد، أراهم الجلوس في التشهد فذكر الحديث .٩٦٢قال كان النبي صلى الله عليه وسلم إذا جلس في الصلاة افترش رجله اليسرى…٩٦٣حدثنا أحمد بن حنبل، حدثنا أبو عاصم الضحاك بن مخلد، أخبرنا عبد الحميد…٩٦٤ويزيد بن أبي حبيب، عن محمد بن عمرو بن حلحلة، عن محمد بن عمرو بن عطاء،…٩٦٥قال كنت في مجلس بهذا الحديث قال فيه فإذا قعد في الركعتين قعد على بطن…٩٦٦حدثنا علي بن الحسين بن إبراهيم، حدثنا أبو بدر، حدثني زهير أبو خيثمة،…٩٦٧قال اجتمع أبو حميد وأبو أسيد وسهل بن سعد ومحمد بن مسلمة فذكر هذا…٩٦٨لا تقولوا السلام على الله فإن الله هو السلام ولكن إذا جلس أحدكم فليقل…٩٦٩اللهم ألف بين قلوبنا وأصلح ذات بيننا واهدنا سبل السلام ونجنا من…٩٧٠إذا قلت هذا أو قضيت هذا فقد قضيت صلاتك إن شئت أن تقوم فقم وإن شئت أن…٩٧١التحيات لله الصلوات الطيبات السلام عليك أيها النبي ورحمة الله وبركاته٩٧٢إذا صليتم فأقيموا صفوفكم ثم ليؤمكم أحدكم فإذا كبر فكبروا وإذا قرأ {…٩٧٣فإذا قرأ فأنصتوا٩٧٤التحيات المباركات الصلوات الطيبات لله السلام عليك أيها النبي ورحمة…٩٧٥التحيات الطيبات والصلوات والملك لله ثم سلموا على اليمين ثم سلموا على…٩٧٦قولوا اللهم صل على محمد وآل محمد كما صليت على إبراهيم وبارك على محمد…٩٧٧صل على محمد وعلى آل محمد كما صليت على إبراهيم٩٧٨اللهم صل على محمد وعلى آل محمد كما صليت على إبراهيم إنك حميد مجيد…٩٧٩قولوا اللهم صل على محمد وأزواجه وذريته كما صليت على آل إبراهيم وبارك…٩٨٠أن محمد بن عبد الله بن زيد، - وعبد الله بن زيد هو الذي أري النداء…٩٨١قولوا اللهم صل على محمد النبي الأمي وعلى آل محمد٩٨٢من سره أن يكتال بالمكيال الأوفى إذا صلى علينا أهل البيت فليقل اللهم…٩٨٣إذا فرغ أحدكم من التشهد الآخر فليتعوذ بالله من أربع من عذاب جهنم ومن…٩٨٤اللهم إني أعوذ بك من عذاب جهنم وأعوذ بك من عذاب القبر وأعوذ بك من…٩٨٥قد غفر له قد غفر له٩٨٦قال من السنة أن يخفى التشهد .٩٨٧حدثنا القعنبي، عن مالك، عن مسلم بن أبي مريم، عن علي بن عبد الرحمن…٩٨٨حدثنا محمد بن عبد الرحيم البزاز، حدثنا عفان، حدثنا عبد الواحد بن…٩٨٩حدثنا إبراهيم بن الحسن المصيصي، حدثنا حجاج، عن ابن جريج، عن زياد، عن…٩٩٠بهذا الحديث قال لا يجاوز بصره إشارته . وحديث حجاج أتم .
