٣٢٥ حديثًا•صفحة ١ من ٣
١٧٢١بل مرة واحدة فمن زاد فهو تطوع١٧٢٢هذه ثم ظهور الحصر١٧٢٣لا يحل لامرأة مسلمة تسافر مسيرة ليلة إلا ومعها رجل ذو حرمة منها١٧٢٤لا يحل لامرأة تؤمن بالله واليوم الآخر أن تسافر يوما وليلة١٧٢٥قال قال رسول الله صلى الله عليه وسلم فذكر نحوه إلا أنه قال " بريدا " .١٧٢٦لا يحل لامرأة تؤمن بالله واليوم الآخر أن تسافر سفرا فوق ثلاثة أيام…١٧٢٧لا تسافر المرأة ثلاثا إلا ومعها ذو محرم١٧٢٨أن ابن عمر، كان يردف مولاة له يقال لها صفية تسافر معه إلى مكة .١٧٢٩لا صرورة في الإسلام١٧٣٠قال كانوا يحجون ولا يتزودون - قال أبو مسعود كان أهل اليمن أو ناس من…١٧٣١قال قرأ هذه الآية { ليس عليكم جناح أن تبتغوا فضلا من ربكم } قال كانوا…١٧٣٢من أراد الحج فليتعجل١٧٣٣حدثنا مسدد، حدثنا عبد الواحد بن زياد، حدثنا العلاء بن المسيب، حدثنا…١٧٣٤أن الناس، في أول الحج كانوا يتبايعون بمنى وعرفة وسوق ذي المجاز ومواسم…١٧٣٥أن الناس، في أول ما كان الحج كانوا يبيعون فذكر معناه إلى قوله مواسم…١٧٣٦قال كان رسول الله صلى الله عليه وسلم بالروحاء فلقي ركبا فسلم عليهم…١٧٣٧حدثنا القعنبي، عن مالك، ح وحدثنا أحمد بن يونس، حدثنا مالك، عن نافع،…١٧٣٨فهن لهم ولمن أتى عليهن من غير أهلهن ممن كان يريد الحج والعمرة ومن كان…١٧٣٩رضى الله عنها أن رسول الله صلى الله عليه وسلم وقت لأهل العراق ذات عرق…١٧٤٠قال وقت رسول الله صلى الله عليه وسلم لأهل المشرق العقيق .١٧٤١من أهل بحجة أو عمرة من المسجد الأقصى إلى المسجد الحرام غفر له ما تقدم…١٧٤٢أن الحارث بن عمرو السهمي، حدثه قال أتيت رسول الله صلى الله عليه وسلم…١٧٤٣قالت نفست أسماء بنت عميس بمحمد بن أبي بكر بالشجرة فأمر رسول الله صلى…١٧٤٤الحائض والنفساء إذا أتتا على الوقت تغتسلان وتحرمان وتقضيان المناسك…١٧٤٥وأحمد بن يونس، قالا حدثنا مالك، عن عبد الرحمن بن القاسم، عن أبيه، عن…١٧٤٦قالت كأني أنظر إلى وبيص المسك في مفرق رسول الله صلى الله عليه وسلم…١٧٤٧حدثنا سليمان بن داود المهري، حدثنا ابن وهب، أخبرني يونس، عن ابن شهاب،…١٧٤٨أن النبي صلى الله عليه وسلم لبد رأسه بالعسل .١٧٤٩حدثنا النفيلي، حدثنا محمد بن سلمة، حدثنا محمد بن إسحاق، ح وحدثنا محمد…١٧٥٠حدثنا ابن السرح، حدثنا ابن وهب، أخبرني يونس، عن ابن شهاب، عن عمرة بنت…١٧٥١ومحمد بن مهران الرازي، قالا حدثنا الوليد، عن الأوزاعي، عن يحيى، عن…١٧٥٢وحفص بن عمر، - المعنى - قالا حدثنا شعبة، عن قتادة، - قال أبو الوليد -…١٧٥٣بهذا الحديث بمعنى أبي الوليد قال ثم سلت الدم بيده . قال أبو داود رواه…١٧٥٤ومروان، أنهما قالا خرج رسول الله صلى الله عليه وسلم عام الحديبية فلما…١٧٥٥والأعمش، عن إبراهيم، عن الأسود، عن عائشة، أن رسول الله صلى الله عليه…١٧٥٦- قال أبو داود أبو عبد الرحيم خالد بن أبي يزيد خال محمد بن سلمة روى…١٧٥٧قالت فتلت قلائد بدن رسول الله صلى الله عليه وسلم بيدى ثم أشعرها…١٧٥٨وقتيبة بن سعيد، أن الليث بن سعد، حدثهم عن ابن شهاب، عن عروة، وعمرة…١٧٥٩- زعم أنه سمعه منهما، جميعا ولم يحفظ حديث هذا من حديث هذا ولا حديث…١٧٦٠أن رسول الله صلى الله عليه وسلم رأى رجلا يسوق بدنة . فقال " اركبها "…١٧٦١اركبها بالمعروف إذا ألجئت إليها حتى تجد ظهرا١٧٦٢إن عطب منها شىء فانحره ثم اصبغ نعله في دمه ثم خل بينه وبين الناس١٧٦٣تنحرها ثم تصبغ نعلها في دمها ثم اضربها على صفحتها ولا تأكل منها أنت…١٧٦٤ويعلى، ابنا عبيد قالا حدثنا محمد بن إسحاق، عن ابن أبي نجيح، عن مجاهد،…١٧٦٥إن أعظم الأيام عند الله تبارك وتعالى يوم النحر ثم يوم القر١٧٦٦حدثنا محمد بن حاتم، حدثنا عبد الرحمن بن مهدي، حدثنا عبد الله بن…١٧٦٧أن النبي صلى الله عليه وسلم وأصحابه كانوا ينحرون البدنة معقولة اليسرى…١٧٦٨قال كنت مع ابن عمر بمنى فمر برجل وهو ينحر بدنته وهي باركة فقال ابعثها…١٧٦٩نحن نعطيه من عندنا١٧٧٠قال قلت لعبد الله بن عباس يا أبا العباس عجبت لاختلاف أصحاب رسول الله…١٧٧١أنه قال بيداؤكم هذه التي تكذبون على رسول الله صلى الله عليه وسلم فيها…١٧٧٢حدثنا القعنبي، عن مالك، عن سعيد بن أبي سعيد المقبري، عن عبيد بن جريج،…١٧٧٣قال صلى رسول الله صلى الله عليه وسلم الظهر بالمدينة أربعا وصلى العصر…١٧٧٤أن النبي صلى الله عليه وسلم صلى الظهر ثم ركب راحلته فلما علا على جبل…١٧٧٥قالت قال سعد بن أبي وقاص كان نبي الله صلى الله عليه وسلم إذا أخذ طريق…١٧٧٦أن ضباعة بنت الزبير بن عبد المطلب، أتت رسول الله صلى الله عليه وسلم…١٧٧٧أن رسول الله صلى الله عليه وسلم أفرد الحج .١٧٧٨من شاء أن يهل بحج فليهل ومن شاء أن يهل بعمرة فليهل بعمرة١٧٧٩حدثنا القعنبي عبد الله بن مسلمة، عن مالك، عن أبي الأسود، محمد بن عبد…١٧٨٠بإسناده مثله زاد فأما من أهل بعمرة فأحل .١٧٨١من كان معه هدى فليهل بالحج مع العمرة ثم لا يحل حتى يحل منهما جميعا١٧٨٢ما يبكيك يا عائشة١٧٨٣حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن منصور، عن إبراهيم، عن الأسود،…١٧٨٤. قال محمد أحسبه قال١٧٨٥حدثنا قتيبة بن سعيد، حدثنا الليث، عن أبي الزبير، عن جابر، قال أقبلنا…١٧٨٦أنه سمع جابرا، قال دخل النبي صلى الله عليه وسلم على عائشة ببعض هذه…١٧٨٧لولا هديي لحللت١٧٨٨اجعلوها عمرة إلا من كان معه الهدى١٧٨٩لو أني استقبلت من أمري ما استدبرت ما أهديت ولولا أن معي الهدى لأحللت١٧٩٠هذه عمرة استمتعنا بها فمن لم يكن عنده هدى فليحل الحل كله وقد دخلت…١٧٩١إذا أهل الرجل بالحج ثم قدم مكة فطاف بالبيت وبالصفا والمروة فقد حل وهي…١٧٩٢وأحمد بن منيع، قالا حدثنا هشيم، عن يزيد بن أبي زياد، - قال ابن منيع…١٧٩٣أن رجلا، من أصحاب النبي صلى الله عليه وسلم أتى عمر بن الخطاب - رضي…١٧٩٤حدثنا موسى أبو سلمة، حدثنا حماد، عن قتادة، عن أبي شيخ الهنائي، خيوان…١٧٩٥لبيك عمرة وحجا لبيك عمرة وحجا١٧٩٦حدثنا أبو سلمة، موسى بن إسماعيل حدثنا وهيب، حدثنا أيوب، عن أبي قلابة،…١٧٩٧قال كنت مع علي حين أمره رسول الله صلى الله عليه وسلم على اليمن قال…١٧٩٨قال قال الصبى بن معبد أهللت بهما معا . فقال عمر هديت لسنة نبيك صلى…١٧٩٩وعثمان بن أبي شيبة، قالا حدثنا جرير بن عبد الحميد، عن منصور، عن أبي…١٨٠٠أتاني الليلة آت من عند ربي عز وجل١٨٠١إن الله تعالى قد أدخل عليكم في حجكم هذا عمرة فإذا قدمتم فمن تطوف…١٨٠٢أن معاوية بن أبي سفيان، أخبره قال قصرت عن النبي صلى الله عليه وسلم…١٨٠٣ومخلد بن خالد، ومحمد بن يحيى، - المعنى - قالوا حدثنا عبد الرزاق،…١٨٠٤سمع ابن عباس، يقول أهل النبي صلى الله عليه وسلم بعمرة وأهل أصحابه بحج…١٨٠٥من كان منكم أهدى فإنه لا يحل له من شىء حرم منه حتى يقضي حجه ومن لم…١٨٠٦إني لبدت رأسي وقلدت هديي فلا أحل حتى أنحر الهدى١٨٠٧أن أبا ذر، كان يقول فيمن حج ثم فسخها بعمرة لم يكن ذلك إلا للركب الذين…١٨٠٨حدثنا النفيلي، حدثنا عبد العزيز، - يعني ابن محمد - أخبرني ربيعة بن…١٨٠٩قال كان الفضل بن عباس رديف رسول الله صلى الله عليه وسلم فجاءته امرأة…١٨١٠احجج عن أبيك واعتمر١٨١١وهناد بن السري، - المعنى واحد - قال إسحاق - حدثنا عبدة بن سليمان، عن…١٨١٢لبيك اللهم لبيك لبيك لا شريك لك لبيك إن الحمد والنعمة لك والملك لا…١٨١٣قال أهل رسول الله صلى الله عليه وسلم فذكر التلبية مثل حديث ابن عمر…١٨١٤أتاني جبريل صلى الله عليه وسلم فأمرني أن آمر أصحابي ومن معي أن يرفعوا…١٨١٥أن رسول الله صلى الله عليه وسلم لبى حتى رمى جمرة العقبة .١٨١٦حدثنا أحمد بن حنبل، حدثنا عبد الله بن نمير، حدثنا يحيى بن سعيد، عن…١٨١٧يلبي المعتمر حتى يستلم الحجر١٨١٨انظروا إلى هذا المحرم ما يصنع١٨١٩أين السائل عن العمرة١٨٢٠وهشيم، عن الحجاج، عن عطاء، عن صفوان بن يعلى، عن أبيه، بهذه القصة قال…١٨٢١بهذا الخبر قال فيه فأمره رسول الله صلى الله عليه وسلم أن ينزعها نزعا…١٨٢٢أن رجلا، أتى النبي صلى الله عليه وسلم بالجعرانة وقد أحرم بعمرة وعليه…١٨٢٣لا يلبس القميص ولا البرنس ولا السراويل ولا العمامة ولا ثوبا مسه ورس…١٨٢٤حدثنا عبد الله بن مسلمة، عن مالك، عن نافع، عن ابن عمر، عن النبي صلى…١٨٢٥ولا تنتقب المرأة الحرام ولا تلبس القفازين١٨٢٦المحرمة لا تنتقب ولا تلبس القفازين١٨٢٧حدثنا أحمد بن حنبل، حدثنا يعقوب، حدثنا أبي، عن ابن إسحاق، قال فإن…١٨٢٨أنه وجد القر فقال ألق على ثوبا يا نافع . فألقيت عليه برنسا فقال تلقي…١٨٢٩السراويل لمن لا يجد الإزار والخف لمن لا يجد النعلين١٨٣٠قال حدثتني عائشة بنت طلحة، أن عائشة أم المؤمنين، - رضى الله عنها -…١٨٣١أن عبد الله، - يعني ابن عمر - كان يصنع ذلك - يعني يقطع الخفين للمرأة…١٨٣٢يقول لما صالح رسول الله صلى الله عليه وسلم أهل الحديبية صالحهم على أن…١٨٣٣حدثنا أحمد بن حنبل، حدثنا هشيم، أخبرنا يزيد بن أبي زياد، عن مجاهد، عن…١٨٣٤حدثته قالت، حججنا مع النبي صلى الله عليه وسلم حجة الوداع فرأيت أسامة…١٨٣٥وطاوس، عن ابن عباس، أن النبي صلى الله عليه وسلم احتجم وهو محرم .١٨٣٦أن رسول الله صلى الله عليه وسلم احتجم وهو محرم في رأسه من داء كان به .١٨٣٧حدثنا أحمد بن حنبل، حدثنا عبد الرزاق، أخبرنا معمر، عن قتادة، عن أنس،…١٨٣٨حدثنا أحمد بن حنبل، حدثنا سفيان، عن أيوب بن موسى، عن نبيه بن وهب، قال…١٨٣٩بهذا الحديث .١٨٤٠أن عبد الله بن عباس، والمسور بن مخرمة، اختلفا بالأبواء فقال ابن عباس…١٨٤١لا ينكح المحرم ولا ينكح١٨٤٢أن محمد بن جعفر، حدثهم حدثنا سعيد، عن مطر، ويعلى بن حكيم، عن نافع، عن…١٨٤٣قالت تزوجني رسول الله صلى الله عليه وسلم ونحن حلالان بسرف .١٨٤٤أن النبي صلى الله عليه وسلم تزوج ميمونة وهو محرم .١٨٤٥قال وهم ابن عباس في تزويج ميمونة وهو محرم .١٨٤٦خمس لا جناح في قتلهن على من قتلهن في الحل والحرم العقرب والفأرة…١٨٤٧خمس قتلهن حلال في الحرم الحية والعقرب والحدأة والفأرة والكلب العقور١٨٤٨الحية والعقرب والفويسقة ويرمي الغراب ولا يقتله والكلب العقور والحدأة…١٨٤٩وكان الحارث، خليفة عثمان على الطائف فصنع لعثمان طعاما فيه من الحجل…١٨٥٠حدثنا أبو سلمة، موسى بن إسماعيل حدثنا حماد، عن قيس، عن عطاء، عن ابن…١٨٥١صيد البر لكم حلال ما لم تصيدوه أو يصد لكم١٨٥٢إنما هي طعمة أطعمكموها الله تعالى١٨٥٣الجراد من صيد البحر١٨٥٤إنما هو من صيد البحر١٨٥٥قال الجراد من صيد البحر .١٨٥٦قد آذاك هوام رأسك١٨٥٧إن شئت فانسك نسيكة وإن شئت فصم ثلاثة أيام وإن شئت فأطعم ثلاثة آصع من…١٨٥٨أن رسول الله صلى الله عليه وسلم مر به زمن الحديبية فذكر القصة فقال "…١٨٥٩أن رجلا، من الأنصار أخبره عن كعب بن عجرة، - وكان قد أصابه في رأسه أذى…١٨٦٠احلق رأسك وصم ثلاثة أيام أو أطعم ستة مساكين فرقا من زبيب أو انسك شاة١٨٦١أى ذلك فعلت أجزأ عنك١٨٦٢من كسر أو عرج فقد حل وعليه الحج من قابل١٨٦٣من كسر أو عرج أو مرض١٨٦٤حدثنا النفيلي، حدثنا محمد بن سلمة، عن محمد بن إسحاق، عن عمرو بن…١٨٦٥أن ابن عمر، كان إذا قدم مكة بات بذي طوى حتى يصبح ويغتسل ثم يدخل مكة…١٨٦٦وابن، حنبل عن يحيى، ح وحدثنا عثمان بن أبي شيبة، حدثنا أبو أسامة،…١٨٦٧أن النبي صلى الله عليه وسلم كان يخرج من طريق الشجرة ويدخل من طريق…١٨٦٨- رضى الله عنها - قالت دخل رسول الله صلى الله عليه وسلم عام الفتح من…١٨٦٩أن النبي صلى الله عليه وسلم كان إذا دخل مكة دخل من أعلاها وخرج من…١٨٧٠أن محمد بن جعفر، حدثهم حدثنا شعبة، قال سمعت أبا قزعة، يحدث عن المهاجر…
