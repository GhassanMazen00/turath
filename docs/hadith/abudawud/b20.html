سنن أبي داود

كتاب الخراج والإمارة والفىء

١٦١ حديثًا•صفحة ١ من ٢
٢٩٢٨ألا كلكم راع وكلكم مسئول عن رعيته فالأمير الذي على الناس راع عليهم…٢٩٢٩يا عبد الرحمن بن سمرة لا تسأل الإمارة فإنك إذا أعطيتها عن مسألة وكلت…٢٩٣٠إن أخونكم عندنا من طلبه٢٩٣١أن النبي صلى الله عليه وسلم استخلف ابن أم مكتوم على المدينة مرتين .٢٩٣٢إذا أراد الله بالأمير خيرا جعل له وزير صدق إن نسي ذكره وإن ذكر أعانه…٢٩٣٣أفلحت يا قديم إن مت ولم تكن أميرا ولا كاتبا ولا عريفا٢٩٣٤وعليك وعلى أبيك السلام٢٩٣٥قال السجل كاتب كان للنبي صلى الله عليه وسلم .٢٩٣٦العامل على الصدقة بالحق كالغازي في سبيل الله حتى يرجع إلى بيته٢٩٣٧لا يدخل الجنة صاحب مكس٢٩٣٨قال الذي يعشر الناس يعني صاحب المكس .٢٩٣٩وسلمة، قالا حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن سالم، عن ابن…٢٩٤٠حدثنا حفص بن عمر، حدثنا شعبة، عن عبد الله بن دينار، عن ابن عمر، قال…٢٩٤١اذهبي فقد بايعتك٢٩٤٢قال وكان قد أدرك النبي صلى الله عليه وسلم وذهبت به أمه زينب بنت حميد…٢٩٤٣من استعملناه على عمل فرزقناه رزقا فما أخذ بعد ذلك فهو غلول٢٩٤٤قال استعملني عمر على الصدقة فلما فرغت أمر لي بعمالة فقلت إنما عملت…٢٩٤٥من كان لنا عاملا فليكتسب زوجة فإن لم يكن له خادم فليكتسب خادما فإن لم…٢٩٤٦ما بال العامل نبعثه فيجيء فيقول هذا لكم وهذا أهدي لي . ألا جلس في بيت…٢٩٤٧انطلق أبا مسعود ولا ألفينك يوم القيامة تجيء على ظهرك بعير من إبل…٢٩٤٨من ولاه الله عز وجل شيئا من أمر المسلمين فاحتجب دون حاجتهم وخلتهم…٢٩٤٩ما أوتيكم من شىء وما أمنعكموه إن أنا إلا خازن أضع حيث أمرت٢٩٥٠حدثنا النفيلي، حدثنا محمد بن سلمة، عن محمد بن إسحاق، عن محمد بن عمرو…٢٩٥١أن عبد الله بن عمر، دخل على معاوية فقال حاجتك يا أبا عبد الرحمن فقال…٢٩٥٢رضى الله عنها أن النبي صلى الله عليه وسلم أتي بظبية فيها خرز فقسمها…٢٩٥٣حدثنا سعيد بن منصور، حدثنا عبد الله بن المبارك، ح وحدثنا ابن المصفى،…٢٩٥٤أنا أولى بالمؤمنين من أنفسهم من ترك مالا فلأهله ومن ترك دينا أو ضياعا…٢٩٥٥من ترك مالا فلورثته ومن ترك كلا فإلينا٢٩٥٦أنا أولى بكل مؤمن من نفسه فأيما رجل مات وترك دينا فإلى ومن ترك مالا…٢٩٥٧أن النبي صلى الله عليه وسلم عرضه يوم أحد وهو ابن أربع عشرة فلم يجزه…٢٩٥٨يا أيها الناس خذوا العطاء ما كان عطاء فإذا تجاحفت قريش على الملك وكان…٢٩٥٩اللهم هل بلغت٢٩٦٠أن جيشا، من الأنصار كانوا بأرض فارس مع أميرهم وكان عمر يعقب الجيوش في…٢٩٦١جعل الله الحق على لسان عمر وقلبه٢٩٦٢إن الله وضع الحق على لسان عمر يقول به٢٩٦٣لا نورث ما تركنا صدقة٢٩٦٤بهذه القصة قال وهما - يعني عليا والعباس رضي الله عنهما - يختصمان فيما…٢٩٦٥وأحمد بن عبدة، - المعنى - أن سفيان بن عيينة، أخبرهم عن عمرو بن دينار،…٢٩٦٦قال قال عمر { وما أفاء الله على رسوله منهم فما أوجفتم عليه من خيل ولا…٢٩٦٧حدثنا هشام بن عمار، حدثنا حاتم بن إسماعيل، ح وحدثنا سليمان بن داود…٢٩٦٨لا نورث ما تركنا صدقة إنما يأكل آل محمد من هذا المال٢٩٦٩لا نورث ما تركنا صدقة وإنما يأكل آل محمد في هذا المال٢٩٧٠أن عائشة، رضى الله عنها أخبرته بهذا الحديث، قال فيه فأبى أبو بكر رضى…٢٩٧١في قوله { فما أوجفتم عليه من خيل ولا ركاب } قال صالح النبي صلى الله…٢٩٧٢حدثنا عبد الله بن الجراح، حدثنا جرير، عن المغيرة، قال جمع عمر بن عبد…٢٩٧٣إن الله عز وجل إذا أطعم نبيا طعمة فهي للذي يقوم من بعده٢٩٧٤لا تقتسم ورثتي دينارا ما تركت بعد نفقة نسائي ومؤنة عاملي فهو صدقة٢٩٧٥كل مال النبي صلى الله عليه وسلم صدقة إلا ما أطعمه أهله وكساهم إنا لا…٢٩٧٦لا نورث ما تركنا فهو صدقة٢٩٧٧لا نورث ما تركنا فهو صدقة وإنما هذا المال لآل محمد لنائبتهم ولضيفهم…٢٩٧٨إنما بنو هاشم وبنو المطلب شىء واحد٢٩٧٩أن رسول الله صلى الله عليه وسلم لم يقسم لبني عبد شمس ولا لبني نوفل من…٢٩٨٠أنا وبنو المطلب لا نفترق في جاهلية ولا إسلام وإنما نحن وهم شىء واحد٢٩٨١في ذي القربى قال هم بنو عبد المطلب .٢٩٨٢أن نجدة الحروري، حين حج في فتنة ابن الزبير أرسل إلى ابن عباس يسأله عن…٢٩٨٣حدثنا عباس بن عبد العظيم، حدثنا يحيى بن أبي بكير، حدثنا أبو جعفر…٢٩٨٤وفاطمة، وزيد بن حارثة، عند النبي صلى الله عليه وسلم فقلت يا رسول الله…٢٩٨٥لا والله لا نستعمل منكم أحدا على الصدقة٢٩٨٦أن حسين بن علي، أخبره أن علي بن أبي طالب قال كانت لي شارف من نصيبي من…٢٩٨٧سبقكن يتامى بدر لكن سأدلكن على ما هو خير لكن من ذلك تكبرن الله على…٢٩٨٨حدثنا يحيى بن خلف، حدثنا عبد الأعلى، عن سعيد، - يعني الجريري - عن أبي…٢٩٨٩بهذه القصة قال ولم يخدمها .٢٩٩٠لو كنت جاعلا لمشرك دية جعلت لأخيك ولكن سأعطيك منه عقبى٢٩٩١قال كان للنبي صلى الله عليه وسلم سهم يدعى الصفي إن شاء عبدا وإن شاء…٢٩٩٢وأزهر، قالا حدثنا ابن عون، قال سألت محمدا عن سهم النبي، صلى الله عليه…٢٩٩٣قال كان رسول الله صلى الله عليه وسلم إذا غزا كان له سهم صاف يأخذه من…٢٩٩٤قالت كانت صفية من الصفي .٢٩٩٥حدثنا سعيد بن منصور، حدثنا يعقوب بن عبد الرحمن الزهري، عن عمرو بن أبي…٢٩٩٦قال صارت صفية لدحية الكلبي ثم صارت لرسول الله صلى الله عليه وسلم .٢٩٩٧قال وقع في سهم دحية جارية جميلة فاشتراها رسول الله صلى الله عليه وسلم…٢٩٩٨قال جمع السبى - يعني بخيبر - فجاء دحية فقال يا رسول الله أعطني جارية…٢٩٩٩من محمد رسول الله إلى بني زهير بن أقيش إنكم إن شهدتم أن لا إله إلا…٣٠٠٠أن الحكم بن نافع، حدثهم قال أخبرنا شعيب، عن الزهري، عن عبد الرحمن بن…٣٠٠١يا معشر يهود أسلموا قبل أن يصيبكم مثل ما أصاب قريشا٣٠٠٢من ظفرتم به من رجال يهود فاقتلوه٣٠٠٣انطلقوا إلى يهود٣٠٠٤لقد بلغ وعيد قريش منكم المبالغ ما كانت تكيدكم بأكثر مما تريدون أن…٣٠٠٥أن يهود بني النضير، وقريظة، حاربوا رسول الله صلى الله عليه وسلم فأجلى…٣٠٠٦أين مسك حيى بن أخطب٣٠٠٧أن عمر، قال أيها الناس إن رسول الله صلى الله عليه وسلم كان عامل يهود…٣٠٠٨أقركم فيها على ذلك ما شئنا٣٠٠٩وزياد بن أيوب، أن إسماعيل بن إبراهيم، حدثهم عن عبد العزيز بن صهيب، عن…٣٠١٠قال قسم رسول الله صلى الله عليه وسلم خيبر نصفين نصفا لنوائبه وحاجته…٣٠١١أن يحيى بن آدم، حدثهم عن أبي شهاب، عن يحيى بن سعيد، عن بشير بن يسار،…٣٠١٢من أصحاب النبي صلى الله عليه وسلم أن رسول الله صلى الله عليه وسلم لما…٣٠١٣قال لما أفاء الله على نبيه صلى الله عليه وسلم خيبر قسمها على ستة…٣٠١٤أن رسول الله صلى الله عليه وسلم لما أفاء الله عليه خيبر قسمها ستة…٣٠١٥مجمع بن جارية الأنصاري - وكان أحد القراء الذين قرءوا القرآن - قال…٣٠١٦وعبد الله بن أبي بكر، وبعض، ولد محمد بن مسلمة قالوا بقيت بقية من أهل…٣٠١٧أن سعيد بن المسيب، أخبره أن رسول الله صلى الله عليه وسلم افتتح بعض…٣٠١٨قال بلغني أن رسول الله صلى الله عليه وسلم افتتح خيبر عنوة بعد القتال…٣٠١٩قال خمس رسول الله صلى الله عليه وسلم خيبر ثم قسم سائرها على من شهدها…٣٠٢٠قال لولا آخر المسلمين ما فتحت قرية إلا قسمتها كما قسم رسول الله صلى…٣٠٢١نعم من دخل دار أبي سفيان فهو آمن ومن أغلق عليه بابه فهو آمن٣٠٢٢نعم من دخل دار أبي سفيان فهو آمن ومن أغلق عليه داره فهو آمن ومن دخل…٣٠٢٣قال سألت جابرا هل غنموا يوم الفتح شيئا قال لا .٣٠٢٤يا أبا هريرة اهتف بالأنصار٣٠٢٥سيتصدقون ويجاهدون إذا أسلموا٣٠٢٦لكم أن لا تحشروا ولا تعشروا ولا خير في دين ليس فيه ركوع٣٠٢٧بسم الله الرحمن الرحيم من محمد رسول الله لعك ذي خيوان إن كان صادقا في…٣٠٢٨يا أخا سبإ لا بد من صدقة٣٠٢٩أخرجوا المشركين من جزيرة العرب وأجيزوا الوفد بنحو مما كنت أجيزهم٣٠٣٠لأخرجن اليهود والنصارى من جزيرة العرب فلا أترك فيها إلا مسلما٣٠٣١قال قال رسول الله صلى الله عليه وسلم بمعناه والأول أتم .٣٠٣٢لا تكون قبلتان في بلد واحد٣٠٣٣حدثنا محمود بن خالد، حدثنا عمر، - يعني ابن عبد الواحد - قال قال سعيد…٣٠٣٤قال قال مالك قد أجلى عمر رحمه الله يهود نجران وفدك .٣٠٣٥منعت العراق قفيزها ودرهمها ومنعت الشام مديها ودينارها ومنعت مصر…٣٠٣٦أيما قرية أتيتموها وأقمتم فيها فسهمكم فيها وأيما قرية عصت الله ورسوله…٣٠٣٧أن النبي صلى الله عليه وسلم بعث خالد بن الوليد إلى أكيدر دومة فأخذ…٣٠٣٨أن النبي صلى الله عليه وسلم لما وجهه إلى اليمن أمره أن يأخذ من كل…٣٠٣٩حدثنا النفيلي، حدثنا أبو معاوية، حدثنا الأعمش، عن إبراهيم، عن مسروق،…٣٠٤٠حدثنا العباس بن عبد العظيم، حدثنا عبد الرحمن بن هانئ أبو نعيم النخعي،…٣٠٤١حدثنا مصرف بن عمرو اليامي، حدثنا يونس، - يعني ابن بكير - حدثنا أسباط…٣٠٤٢قال إن أهل فارس لما مات نبيهم كتب لهم إبليس المجوسية .٣٠٤٣سمع بجالة، يحدث عمرو بن أوس وأبا الشعثاء قال كنت كاتبا لجزء بن معاوية…٣٠٤٤حدثنا محمد بن مسكين اليمامي، حدثنا يحيى بن حسان، حدثنا هشيم، أخبرنا…٣٠٤٥إن الله يعذب الذين يعذبون الناس في الدنيا٣٠٤٦إنما العشور على اليهود والنصارى وليس على المسلمين عشور٣٠٤٧حدثنا محمد بن عبيد المحاربي، حدثنا وكيع، عن سفيان، عن عطاء بن السائب،…٣٠٤٨إنما العشور على اليهود والنصارى٣٠٤٩لا إنما العشور على النصارى واليهود٣٠٥٠يا ابن عوف اركب فرسك ثم ناد ألا إن الجنة لا تحل إلا لمؤمن وأن اجتمعوا…٣٠٥١لعلكم تقاتلون قوما فتظهرون عليهم فيتقونكم بأموالهم دون أنفسهم وأبنائهم٣٠٥٢ألا من ظلم معاهدا أو انتقصه أو كلفه فوق طاقته أو أخذ منه شيئا بغير…٣٠٥٣ليس على المسلم جزية٣٠٥٤هذا فقال إذا أسلم فلا جزية عليه .٣٠٥٥أبشر فقد جاءك الله بقضائك٣٠٥٦بمعنى إسناد أبي توبة وحديثه قال عند قوله " ما يقضي عني " . فسكت عني…٣٠٥٧حدثنا هارون بن عبد الله، حدثنا أبو داود، حدثنا عمران، عن قتادة، عن…٣٠٥٨أن النبي صلى الله عليه وسلم أقطعه أرضا بحضرموت .٣٠٥٩بإسناده مثله .٣٠٦٠قال خط لي رسول الله صلى الله عليه وسلم دارا بالمدينة بقوس وقال "…٣٠٦١واحد، أن رسول الله صلى الله عليه وسلم أقطع بلال بن الحارث المزني…٣٠٦٢بسم الله الرحمن الرحيم هذا ما أعطى محمد رسول الله بلال بن الحارث…٣٠٦٣هذا ما أعطى رسول الله صلى الله عليه وسلم بلال بن الحارث المزني أعطاه…٣٠٦٤ما لم تنله خفاف٣٠٦٥ما لم تنله أخفاف الإبل٣٠٦٦حدثنا محمد بن أحمد القرشي، حدثنا عبد الله بن الزبير، حدثنا فرج بن…٣٠٦٧اللهم بارك لأحمس في خيلها ورجالها٣٠٦٨من أهل ذي المروة٣٠٦٩أن رسول الله صلى الله عليه وسلم أقطع الزبير نخلا .٣٠٧٠اكتب له يا غلام بالدهناء٣٠٧١من سبق إلى ماء لم يسبقه إليه مسلم فهو له٣٠٧٢أعطوه من حيث بلغ السوط٣٠٧٣من أحيا أرضا ميتة فهي له وليس لعرق ظالم حق٣٠٧٤من أحيا أرضا ميتة فهي له٣٠٧٥حدثنا أحمد بن سعيد الدارمي، حدثنا وهب، عن أبيه، عن ابن إسحاق، بإسناده…٣٠٧٦حدثنا أحمد بن عبدة الآملي، حدثنا عبد الله بن عثمان، حدثنا عبد الله بن…٣٠٧٧من أحاط حائطا على أرض فهي له
