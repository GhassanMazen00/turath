كتاب الأيمان والنذور

٨٤ حديثًا
٣٢٤٢من حلف على يمين مصبورة كاذبا فليتبوأ بوجهه مقعده من النار٣٢٤٣من حلف على يمين هو فيها فاجر ليقتطع بها مال امرئ مسلم لقي الله وهو…٣٢٤٤أن رجلا، من كندة ورجلا من حضرموت اختصما إلى النبي صلى الله عليه وسلم…٣٢٤٥قال جاء رجل من حضرموت ورجل من كندة إلى رسول الله صلى الله عليه وسلم…٣٢٤٦لا يحلف أحد عند منبري هذا على يمين آثمة ولو على سواك أخضر إلا تبوأ…٣٢٤٧من حلف فقال في حلفه واللات فليقل لا إله إلا الله ومن قال لصاحبه تعال…٣٢٤٨لا تحلفوا بآبائكم ولا بأمهاتكم ولا بالأنداد ولا تحلفوا إلا بالله ولا…٣٢٤٩إن الله ينهاكم أن تحلفوا بآبائكم فمن كان حالفا فليحلف بالله أو ليسكت٣٢٥٠- رضى الله عنه - قال سمعني رسول الله صلى الله عليه وسلم نحو معناه إلى…٣٢٥١من حلف بغير الله فقد أشرك٣٢٥٢أفلح وأبيه إن صدق دخل الجنة وأبيه إن صدق٣٢٥٣من حلف بالأمانة فليس منا٣٢٥٤هو كلام الرجل في بيته كلا والله وبلى والله٣٢٥٥يمينك على ما يصدقك عليها صاحبك٣٢٥٦صدقت المسلم أخو المسلم٣٢٥٧من حلف بملة غير ملة الإسلام كاذبا فهو كما قال ومن قتل نفسه بشىء عذب…٣٢٥٨من حلف فقال إني بريء من الإسلام فإن كان كاذبا فهو كما قال، وإن كان…٣٢٥٩قال : رأيت النبي صلى الله عليه وسلم وضع تمرة على كسرة فقال : " هذه…٣٢٦٠مثله .٣٢٦١من حلف على يمين فقال إن شاء الله فقد استثنى٣٢٦٢من حلف فاستثنى فإن شاء رجع، وإن شاء ترك غير حنث٣٢٦٣لا، ومقلب القلوب٣٢٦٤والذي نفس أبي القاسم بيده٣٢٦٥لا، وأستغفر الله٣٢٦٦: أن لقيط بن عامر، خرج وافدا إلى النبي صلى الله عليه وسلم قال لقيط :…٣٢٦٧: أن أبا بكر، أقسم على النبي صلى الله عليه وسلم فقال له النبي صلى…٣٢٦٨أصبت بعضا وأخطأت بعضا٣٢٦٩زاد فيه ولم يخبره .٣٢٧٠بل أنت أبرهم وأصدقهم٣٢٧١وعبد الأعلى، عن الجريري، عن أبي عثمان، عن عبد الرحمن بن أبي بكر، بهذا…٣٢٧٢لا يمين عليك، ولا نذر في معصية الرب وفي قطيعة الرحم وفيما لا تملك٣٢٧٣لا نذر إلا فيما يبتغى به وجه الله، ولا يمين في قطيعة رحم٣٢٧٤لا نذر ولا يمين فيما لا يملك ابن آدم ولا في معصية الله ولا في قطيعة…٣٢٧٥بلى قد فعلت، ولكن قد غفر لك بإخلاص قول لا إله إلا الله٣٢٧٦إني والله إن شاء الله لا أحلف على يمين فأرى غيرها خيرا منها إلا كفرت…٣٢٧٧يا عبد الرحمن بن سمرة إذا حلفت على يمين فرأيت غيرها خيرا منها، فأت…٣٢٧٨فكفر عن يمينك، ثم ائت الذي هو خير٣٢٧٩أو قال فحزرته فوجدته مدين ونصفا بمد هشام .٣٢٨٠قال محمد : صاع خالد صاع هشام يعني ابن عبد الملك .٣٢٨١قال : لما ولي خالد القسري أضعف الصاع فصار الصاع ستة عشر رطلا . قال…٣٢٨٢حدثنا مسدد، حدثنا يحيى، عن الحجاج الصواف، حدثني يحيى بن أبي كثير، عن…٣٢٨٣: أن أمه، أوصته أن يعتق، عنها رقبة مؤمنة فأتى النبي صلى الله عليه…٣٢٨٤: أن رجلا، أتى النبي صلى الله عليه وسلم بجارية سوداء فقال : يا رسول…٣٢٨٥والله لأغزون قريشا، والله لأغزون قريشا، والله لأغزون قريشا٣٢٨٦والله لأغزون قريشا٣٢٨٧لا يرد شيئا، وإنما يستخرج به من البخيل٣٢٨٨لا يأتي ابن آدم النذر القدر بشىء لم أكن قدرته له، ولكن يلقيه النذر…٣٢٨٩من نذر أن يطيع الله فليطعه، ومن نذر أن يعصي الله فلا يعصه٣٢٩٠لا نذر في معصية، وكفارته كفارة يمين٣٢٩١يقول قال ابن المبارك - يعني في هذا الحديث - حدث أبو سلمة، فدل ذلك على…٣٢٩٢لا نذر في معصية، وكفارته كفارة يمين٣٢٩٣مروها فلتختمر ولتركب، ولتصم ثلاثة أيام٣٢٩٤مولى لبني ضمرة - وكان أيما رجل - أن أبا سعيد الرعيني أخبره بإسناد…٣٢٩٥إن الله لا يصنع بشقاء أختك شيئا، فلتحج راكبة ولتكفر عن يمينها٣٢٩٦: أن أخت، عقبة بن عامر نذرت أن تمشي، إلى البيت، فأمرها النبي صلى الله…٣٢٩٧إن الله لغني عن نذرها، مرها فلتركب٣٢٩٨مر أختك فلتركب٣٢٩٩أن يزيد بن أبي حبيب، أخبره أن أبا الخير حدثه عن عقبة بن عامر الجهني،…٣٣٠٠مروه فليتكلم وليستظل وليقعد، وليتم صومه٣٣٠١إن الله لغني عن تعذيب هذا نفسه٣٣٠٢أن طاوسا، أخبره عن ابن عباس، : أن النبي صلى الله عليه وسلم مر وهو…٣٣٠٣إن الله لغني عن مشى أختك، فلتركب ولتهد بدنة٣٣٠٤إن الله لا يصنع بمشى أختك إلى البيت شيئا٣٣٠٥: أن رجلا، قام يوم الفتح فقال : يا رسول الله إني نذرت لله إن فتح الله…٣٣٠٦والذي بعث محمدا بالحق لو صليت ها هنا لأجزأ عنك صلاة في بيت المقدس٣٣٠٧: أن سعد بن عبادة، استفتى رسول الله صلى الله عليه وسلم فقال : إن أمي…٣٣٠٨: أن امرأة، ركبت البحر فنذرت إن نجاها الله أن تصوم شهرا، فنجاها الله…٣٣٠٩قد وجب أجرك، ورجعت إليك في الميراث٣٣١٠لو كان على أمك دين أكنت قاضيته٣٣١١من مات وعليه صيام صام عنه وليه٣٣١٢: أن امرأة، أتت النبي صلى الله عليه وسلم فقالت : يا رسول الله إني…٣٣١٣هل كان فيها وثن من أوثان الجاهلية يعبد٣٣١٤هل بها من الأوثان شىء٣٣١٥هل بها وثن أو عيد من أعياد الجاهلية٣٣١٦نأخذك بجريرة حلفائك ثقيف٣٣١٧أمسك عليك بعض مالك فهو خير لك٣٣١٨أنه قال لرسول الله صلى الله عليه وسلم حين تيب عليه : إني أنخلع من…٣٣١٩يجزئ عنك الثلث٣٣٢٠قال : كان أبو لبابة، فذكر معناه والقصة لأبي لبابة . قال أبو داود :…٣٣٢١في قصته قال قلت : يا رسول الله إن من توبتي إلى الله أن أخرج من مالي…٣٣٢٢من نذر نذرا لم يسمه فكفارته كفارة يمين، ومن نذر نذرا في معصية فكفارته…٣٣٢٣كفارة النذر كفارة اليمين٣٣٢٤أن سعيد بن الحكم، حدثهم أخبرنا يحيى بن أيوب، حدثني كعب بن علقمة، أنه…٣٣٢٥رضي الله عنه أنه قال : يا رسول الله إني نذرت في الجاهلية أن أعتكف في…
