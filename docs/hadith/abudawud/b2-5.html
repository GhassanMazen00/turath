٧٧٠ حديثًا•صفحة ٥ من ٦
٩٩١قال رأيت النبي صلى الله عليه وسلم واضعا ذراعه اليمنى على فخذه اليمنى…٩٩٢وأحمد بن محمد بن شبوية، ومحمد بن رافع، ومحمد بن عبد الملك الغزال،…٩٩٣يصلي وهو مشبك يديه قال قال ابن عمر تلك صلاة المغضوب عليهم .٩٩٤أنه رأى رجلا يتكئ على يده اليسرى وهو قاعد في الصلاة - وقال هارون بن…٩٩٥أن النبي صلى الله عليه وسلم كان في الركعتين الأوليين كأنه على الرضف .…٩٩٦السلام عليكم ورحمة الله السلام عليكم ورحمة الله٩٩٧السلام عليكم ورحمة الله وبركاته٩٩٨ما بال أحدكم يرمي بيده كأنها أذناب خيل شمس إنما يكفي أحدكم - أو ألا…٩٩٩أما يكفي أحدكم - أو أحدهم - أن يضع يده على فخذه ثم يسلم على أخيه من…١٠٠٠ما لي أراكم رافعي أيديكم كأنها أذناب خيل شمس اسكنوا في الصلاة١٠٠١حدثنا محمد بن عثمان أبو الجماهر، حدثنا سعيد بن بشير، عن قتادة، عن…١٠٠٢قال كان يعلم انقضاء صلاة رسول الله صلى الله عليه وسلم بالتكبير .١٠٠٣أن أبا معبد، مولى ابن عباس أخبره أن ابن عباس أخبره أن رفع الصوت للذكر…١٠٠٤حذف السلام سنة١٠٠٥إذا فسا أحدكم في الصلاة فلينصرف فليتوضأ وليعد صلاته١٠٠٦وعبد الوارث، عن ليث، عن الحجاج بن عبيد، عن إبراهيم بن إسماعيل، عن أبي…١٠٠٧أصاب الله بك يا ابن الخطاب١٠٠٨لم أنس ولم تقصر الصلاة١٠٠٩حدثنا عبد الله بن مسلمة، عن مالك، عن أيوب، عن محمد، بإسناده - وحديث…١٠١٠حدثنا مسدد، حدثنا بشر، - يعني ابن المفضل - حدثنا سلمة، - يعني ابن…١٠١١وهشام، ويحيى بن عتيق، وابن، عون عن محمد، عن أبي هريرة، عن النبي صلى…١٠١٢وأبي، سلمة وعبيد الله بن عبد الله عن أبي هريرة، بهذه القصة قال ولم…١٠١٣أن أبا بكر بن سليمان بن أبي حثمة، أخبره أنه، بلغه أن رسول الله صلى…١٠١٤سمع أبا سلمة بن عبد الرحمن، عن أبي هريرة، أن النبي صلى الله عليه وسلم…١٠١٥كل ذلك لم أفعل١٠١٦بهذا الخبر قال ثم سجد سجدتى السهو بعد ما سلم .١٠١٧حدثنا أحمد بن محمد بن ثابت، حدثنا أبو أسامة، ح وحدثنا محمد بن العلاء،…١٠١٨حدثنا مسدد، حدثنا يزيد بن زريع، ح وحدثنا مسدد، حدثنا مسلمة بن محمد،…١٠١٩ومسلم بن إبراهيم، - المعنى - قال حفص حدثنا شعبة، عن الحكم، عن…١٠٢٠حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن منصور، عن إبراهيم، عن علقمة،…١٠٢١فإذا نسي أحدكم فليسجد سجدتين١٠٢٢حدثنا نصر بن علي، أخبرنا جرير، ح وحدثنا يوسف بن موسى، حدثنا جرير، -…١٠٢٣أن سويد بن قيس، أخبره عن معاوية بن حديج، أن رسول الله صلى الله عليه…١٠٢٤إذا شك أحدكم في صلاته فليلق الشك وليبن على اليقين فإذا استيقن التمام…١٠٢٥أن النبي صلى الله عليه وسلم سمى سجدتى السهو المرغمتين .١٠٢٦إذا شك أحدكم في صلاته فلا يدري كم صلى ثلاثا أو أربعا فليصل ركعة ويسجد…١٠٢٧إذا شك أحدكم في صلاته فإن استيقن أن قد صلى ثلاثا فليقم فليتم ركعة…١٠٢٨إذا كنت في صلاة فشككت في ثلاث أو أربع وأكبر ظنك على أربع تشهدت ثم…١٠٢٩إذا صلى أحدكم فلم يدر زاد أم نقص فليسجد سجدتين وهو قاعد فإذا أتاه…١٠٣٠إن أحدكم إذا قام يصلي جاءه الشيطان فلبس عليه حتى لا يدري كم صلى فإذا…١٠٣١وهو جالس قبل التسليم١٠٣٢فليسجد سجدتين قبل أن يسلم ثم ليسلم١٠٣٣من شك في صلاته فليسجد سجدتين بعد ما يسلم١٠٣٤حدثنا القعنبي، عن مالك، عن ابن شهاب، عن عبد الرحمن الأعرج، عن عبد…١٠٣٥حدثنا عمرو بن عثمان، حدثنا أبي وبقية، قالا حدثنا شعيب، عن الزهري،…١٠٣٦إذا قام الإمام في الركعتين فإن ذكر قبل أن يستوي قائما فليجلس فإن…١٠٣٧حدثنا عبيد الله بن عمر الجشمي، حدثنا يزيد بن هارون، أخبرنا المسعودي،…١٠٣٨لكل سهو سجدتان بعد ما يسلم١٠٣٩أن النبي صلى الله عليه وسلم صلى بهم فسها فسجد سجدتين ثم تشهد ثم سلم .١٠٤٠ومحمد بن رافع، قالا حدثنا عبد الرزاق، أخبرنا معمر، عن الزهري، عن هند…١٠٤١حدثنا أبو الوليد الطيالسي، حدثنا شعبة، عن سماك بن حرب، عن قبيصة بن…١٠٤٢حدثنا مسلم بن إبراهيم، حدثنا شعبة، عن سليمان، عن عمارة بن عمير، عن…١٠٤٣اجعلوا في بيوتكم من صلاتكم ولا تتخذوها قبورا١٠٤٤صلاة المرء في بيته أفضل من صلاته في مسجدي هذا إلا المكتوبة١٠٤٥وحميد، عن أنس، أن النبي صلى الله عليه وسلم وأصحابه كانوا يصلون نحو…١٠٤٦خير يوم طلعت فيه الشمس يوم الجمعة فيه خلق آدم وفيه أهبط وفيه تيب عليه…١٠٤٧إن من أفضل أيامكم يوم الجمعة فيه خلق آدم وفيه قبض وفيه النفخة وفيه…١٠٤٨يوم الجمعة ثنتا عشرة١٠٤٩هي ما بين أن يجلس الإمام إلى أن تقضى الصلاة١٠٥٠من توضأ فأحسن الوضوء ثم أتى الجمعة فاستمع وأنصت غفر له ما بين الجمعة…١٠٥١إذا كان يوم الجمعة غدت الشياطين براياتها إلى الأسواق فيرمون الناس…١٠٥٢من ترك ثلاث جمع تهاونا بها طبع الله على قلبه١٠٥٣من ترك الجمعة من غير عذر فليتصدق بدينار فإن لم يجد فبنصف دينار١٠٥٤من فاتته الجمعة من غير عذر فليتصدق بدرهم أو نصف درهم أو صاع حنطة أو…١٠٥٥أن محمد بن جعفر، حدثه عن عروة بن الزبير، عن عائشة، زوج النبي صلى الله…١٠٥٦الجمعة على كل من سمع النداء١٠٥٧أن يوم، حنين كان يوم مطر فأمر النبي صلى الله عليه وسلم مناديه أن…١٠٥٨أن ذلك، كان يوم جمعة .١٠٥٩أنه شهد النبي صلى الله عليه وسلم زمن الحديبية في يوم جمعة وأصابهم مطر…١٠٦٠أن ابن عمر، نزل بضجنان في ليلة باردة فأمر المنادي فنادى أن الصلاة في…١٠٦١أن صلوا في رحالكم١٠٦٢أنه نادى بالصلاة بضجنان في ليلة ذات برد وريح فقال في آخر ندائه ألا…١٠٦٣أن ابن عمر، - يعني أذن بالصلاة في ليلة ذات برد وريح - فقال ألا صلوا…١٠٦٤حدثنا عبد الله بن محمد النفيلي، حدثنا محمد بن سلمة، عن محمد بن إسحاق،…١٠٦٥ليصل من شاء منكم في رحله١٠٦٦محمد بن سيرين أن ابن عباس، قال لمؤذنه في يوم مطير إذا قلت أشهد أن…١٠٦٧الجمعة حق واجب على كل مسلم في جماعة إلا أربعة عبد مملوك أو امرأة أو…١٠٦٨ومحمد بن عبد الله المخرمي، - لفظه - قالا حدثنا وكيع، عن إبراهيم بن…١٠٦٩كعب بن مالك أنه كان إذا سمع النداء، يوم الجمعة ترحم لأسعد بن زرارة .…١٠٧٠من شاء أن يصلي فليصل١٠٧١حدثنا محمد بن طريف البجلي، حدثنا أسباط، عن الأعمش، عن عطاء بن أبي…١٠٧٢قال قال عطاء اجتمع يوم جمعة ويوم فطر على عهد ابن الزبير فقال عيدان…١٠٧٣قد اجتمع في يومكم هذا عيدان فمن شاء أجزأه من الجمعة وإنا مجمعون١٠٧٤أن رسول الله صلى الله عليه وسلم كان يقرأ في صلاة الفجر يوم الجمعة…١٠٧٥بإسناده ومعناه وزاد في صلاة الجمعة بسورة الجمعة و { إذا جاءك…١٠٧٦إنما يلبس هذه من لا خلاق له في الآخرة١٠٧٧وعمرو بن الحارث، عن ابن شهاب، عن سالم، عن أبيه، قال وجد عمر بن الخطاب…١٠٧٨ما على أحدكم إن وجد١٠٧٩حدثنا مسدد، حدثنا يحيى، عن ابن عجلان، عن عمرو بن شعيب، عن أبيه، عن…١٠٨٠أن مري غلامك النجار أن يعمل لي أعوادا أجلس عليهن إذا كلمت الناس١٠٨١أن النبي صلى الله عليه وسلم لما بدن قال له تميم الداري ألا أتخذ لك…١٠٨٢قال كان بين منبر رسول الله صلى الله عليه وسلم وبين الحائط كقدر ممر…١٠٨٣إن جهنم تسجر إلا يوم الجمعة١٠٨٤يقول كان رسول الله صلى الله عليه وسلم يصلي الجمعة إذا مالت الشمس .١٠٨٥حدثنا أحمد بن يونس، حدثنا يعلى بن الحارث، سمعت إياس بن سلمة بن…١٠٨٦حدثنا محمد بن كثير، أخبرنا سفيان، عن أبي حازم، عن سهل بن سعد، قال كنا…١٠٨٧أن الأذان، كان أوله حين يجلس الإمام على المنبر يوم الجمعة في عهد…١٠٨٨قال كان يؤذن بين يدى رسول الله صلى الله عليه وسلم إذا جلس على المنبر…١٠٨٩قال لم يكن لرسول الله صلى الله عليه وسلم إلا مؤذن واحد بلال ثم ذكر…١٠٩٠أن السائب بن يزيد ابن أخت، نمر أخبره قال ولم يكن لرسول الله صلى الله…١٠٩١حدثنا يعقوب بن كعب الأنطاكي، حدثنا مخلد بن يزيد، حدثنا ابن جريج، عن…١٠٩٢قال كان النبي صلى الله عليه وسلم يخطب خطبتين كان يجلس إذا صعد المنبر…١٠٩٣أن رسول الله صلى الله عليه وسلم كان يخطب قائما ثم يجلس ثم يقوم فيخطب…١٠٩٤وعثمان بن أبي شيبة، - المعنى - عن أبي الأحوص، حدثنا سماك، عن جابر بن…١٠٩٥قال رأيت النبي صلى الله عليه وسلم يخطب قائما ثم يقعد قعدة لا يتكلم .…١٠٩٦أيها الناس إنكم لن تطيقوا أو لن تفعلوا كل ما أمرتم به ولكن سددوا…١٠٩٧الحمد لله نستعينه ونستغفره ونعوذ بالله من شرور أنفسنا من يهده الله…١٠٩٨ومن يعصهما فقد غوى١٠٩٩قم - أو اذهب - بئس الخطيب أنت١١٠٠حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، عن خبيب، عن عبد…١١٠١قال كانت صلاة رسول الله صلى الله عليه وسلم قصدا وخطبته قصدا يقرأ آيات…١١٠٢حدثنا محمود بن خالد، حدثنا مروان، حدثنا سليمان بن بلال، عن يحيى بن…١١٠٣لعمرة بنت عبد الرحمن كانت أكبر منها بمعناه .١١٠٤حدثنا أحمد بن يونس، حدثنا زائدة، عن حصين بن عبد الرحمن، قال رأى عمارة…١١٠٥قال ما رأيت رسول الله صلى الله عليه وسلم شاهرا يديه قط يدعو على منبره…١١٠٦حدثنا محمد بن عبد الله بن نمير، حدثنا أبي، حدثنا العلاء بن صالح، عن…١١٠٧قال كان رسول الله صلى الله عليه وسلم لا يطيل الموعظة يوم الجمعة إنما…١١٠٨احضروا الذكر وادنوا من الإمام فإن الرجل لا يزال يتباعد حتى يؤخر في…١١٠٩صدق الله { إنما أموالكم وأولادكم فتنة } رأيت هذين فلم أصبر١١١٠حدثنا محمد بن عوف، حدثنا المقرئ، حدثنا سعيد بن أبي أيوب، عن أبي…١١١١حدثنا داود بن رشيد، حدثنا خالد بن حيان الرقي، حدثنا سليمان بن عبد…١١١٢إذا قلت أنصت والإمام يخطب فقد لغوت١١١٣يحضر الجمعة ثلاثة نفر رجل حضرها يلغو وهو حظه منها ورجل حضرها يدعو فهو…١١١٤إذا أحدث أحدكم في صلاته فليأخذ بأنفه ثم لينصرف١١١٥أن رجلا، جاء يوم الجمعة والنبي صلى الله عليه وسلم يخطب فقال " أصليت…١١١٦وإسماعيل بن إبراهيم، - المعنى - قالا حدثنا حفص بن غياث، عن الأعمش، عن…١١١٧إذا جاء أحدكم والإمام يخطب فليصل ركعتين يتجوز فيهما١١١٨حدثنا هارون بن معروف، حدثنا بشر بن السري، حدثنا معاوية بن صالح، عن…١١١٩إذا نعس أحدكم وهو في المسجد فليتحول من مجلسه ذلك إلى غيره١١٢٠حدثنا مسلم بن إبراهيم، عن جرير، - هو ابن حازم لا أدري كيف قاله مسلم…١١٢١من أدرك ركعة من الصلاة فقد أدرك الصلاة١١٢٢أن رسول الله صلى الله عليه وسلم كان يقرأ في العيدين ويوم الجمعة ب {…١١٢٣أن الضحاك بن قيس، سأل النعمان بن بشير ماذا كان يقرأ به رسول الله صلى…١١٢٤حدثنا القعنبي، حدثنا سليمان، - يعني ابن بلال - عن جعفر، عن أبيه، عن…١١٢٥أن رسول الله صلى الله عليه وسلم كان يقرأ في صلاة الجمعة { سبح اسم ربك…١١٢٦- رضى الله عنها - قالت صلى رسول الله صلى الله عليه وسلم في حجرته…١١٢٧وسليمان بن داود، - المعنى - قالا حدثنا حماد بن زيد، حدثنا أيوب، عن…١١٢٨قال كان ابن عمر يطيل الصلاة قبل الجمعة ويصلي بعدها ركعتين في بيته…١١٢٩أن نافع بن جبير، أرسله إلى السائب بن يزيد ابن أخت نمر يسأله عن شىء،…١١٣٠قال
