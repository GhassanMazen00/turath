سنن أبي داود

كتاب الأقضية

٧٠ حديثًا
٣٥٧١من ولي القضاء فقد ذبح بغير سكين٣٥٧٢من جعل قاضيا بين الناس فقد ذبح بغير سكين٣٥٧٣القضاة ثلاثة واحد في الجنة واثنان في النار فأما الذي في الجنة فرجل…٣٥٧٤إذا حكم الحاكم فاجتهد فأصاب فله أجران وإذا حكم فاجتهد فأخطأ فله أجر٣٥٧٥من طلب قضاء المسلمين حتى يناله ثم غلب عدله جوره فله الجنة ومن غلب…٣٥٧٦قال { ومن لم يحكم بما أنزل الله فأولئك هم الكافرون } إلى قوله {…٣٥٧٧ومحمد بن المثنى، قالا أخبرنا أبو معاوية، عن الأعمش، عن رجاء الأنصاري،…٣٥٧٨من طلب القضاء واستعان عليه وكل إليه ومن لم يطلبه ولم يستعن عليه أنزل…٣٥٧٩لن نستعمل - أو لا نستعمل - على عملنا من أراده٣٥٨٠حدثنا أحمد بن يونس، حدثنا ابن أبي ذئب، عن الحارث بن عبد الرحمن، عن…٣٥٨١يا أيها الناس من عمل منكم لنا على عمل فكتمنا منه مخيطا فما فوقه فهو…٣٥٨٢إن الله سيهدي قلبك ويثبت لسانك فإذا جلس بين يديك الخصمان فلا تقضين…٣٥٨٣إنما أنا بشر وإنكم تختصمون إلى ولعل بعضكم أن يكون ألحن بحجته من بعض…٣٥٨٤أما إذ فعلتما ما فعلتما فاقتسما وتوخيا الحق . ثم استهما ثم تحالا٣٥٨٥إني إنما أقضي بينكم برأيي فيما لم ينزل على فيه٣٥٨٦أن عمر بن الخطاب، - رضى الله عنه - قال وهو على المنبر يا أيها الناس…٣٥٨٧ولا إخالني رأيت شاميا أفضل منه يعني حريز بن عثمان .٣٥٨٨قال قضى رسول الله صلى الله عليه وسلم أن الخصمين يقعدان بين يدى الحكم .٣٥٨٩لا يقضي الحكم بين اثنين وهو غضبان٣٥٩٠قال { فإن جاءوك فاحكم بينهم أو أعرض عنهم } فنسخت قال { فاحكم بينهم…٣٥٩١قال لما نزلت هذه الآية { فإن جاءوك فاحكم بينهم أو أعرض عنهم } { وإن…٣٥٩٢كيف تقضي إذا عرض لك قضاء٣٥٩٣أن رسول الله صلى الله عليه وسلم لما بعثه إلى اليمن فذكر معناه .٣٥٩٤الصلح جائز بين المسلمين٣٥٩٥أن كعب بن مالك، أخبره أنه، تقاضى ابن أبي حدرد دينا كان له عليه في عهد…٣٥٩٦ألا أخبركم بخير الشهداء الذي يأتي بشهادته أو يخبر بشهادته قبل أن…٣٥٩٧من حالت شفاعته دون حد من حدود الله فقد ضاد الله ومن خاصم في باطل وهو…٣٥٩٨ومن أعان على خصومة بظلم فقد باء بغضب من الله عز وجل٣٥٩٩عدلت شهادة الزور بالإشراك بالله٣٦٠٠أن رسول الله صلى الله عليه وسلم رد شهادة الخائن والخائنة وذي الغمر…٣٦٠١لا تجوز شهادة خائن ولا خائنة ولا زان ولا زانية ولا ذي غمر على أخيه٣٦٠٢لا تجوز شهادة بدوي على صاحب قرية٣٦٠٣وما يدريك وقد قالت ما قالت دعها عنك٣٦٠٤عقبة ولكني لحديث عبيد أحفظ - فذكر معناه . قال أبو داود نظر حماد بن…٣٦٠٥أن رجلا، من المسلمين حضرته الوفاة بدقوقاء هذه ولم يجد أحدا من…٣٦٠٦حدثنا الحسن بن علي، حدثنا يحيى بن آدم، حدثنا ابن أبي زائدة، عن محمد…٣٦٠٧أوليس قد ابتعته منك٣٦٠٨والحسن بن علي، أن زيد بن الحباب، حدثهم حدثنا سيف المكي، - قال عثمان…٣٦٠٩وسلمة بن شبيب، قالا حدثنا عبد الرزاق، أخبرنا محمد بن مسلم، عن عمرو بن…٣٦١٠حدثنا أحمد بن أبي بكر أبو مصعب الزهري، حدثنا الدراوردي، عن ربيعة بن…٣٦١١حدثنا محمد بن داود الإسكندراني، حدثنا زياد، - يعني ابن يونس - حدثني…٣٦١٢هل لكم بينة على أنكم أسلمتم قبل أن تؤخذوا في هذه الأيام٣٦١٣أن رجلين، ادعيا بعيرا أو دابة إلى النبي صلى الله عليه وسلم ليست لواحد…٣٦١٤بإسناده ومعناه .٣٦١٥بمعنى إسناده أن رجلين، ادعيا بعيرا على عهد النبي صلى الله عليه وسلم…٣٦١٦استهما على اليمين ما كان أحبا ذلك أو كرها٣٦١٧إذا كره الاثنان اليمين أو استحباها فليستهما عليها٣٦١٨بإسناد ابن منهال مثله قال في دابة وليس لهما بينة فأمرهما رسول الله…٣٦١٩قال كتب إلى ابن عباس أن رسول الله صلى الله عليه وسلم قضى باليمين على…٣٦٢٠احلف بالله الذي لا إله إلا هو ما له عندك شىء٣٦٢١قال كان بيني وبين رجل من اليهود أرض فجحدني فقدمته إلى النبي صلى الله…٣٦٢٢أن رجلا، من كندة ورجلا من حضرموت اختصما إلى النبي صلى الله عليه وسلم…٣٦٢٣قال جاء رجل من حضرموت ورجل من كندة إلى رسول الله صلى الله عليه وسلم…٣٦٢٤أنشدكم بالله الذي أنزل التوراة على موسى ما تجدون في التوراة على من زنى٣٦٢٥من مزينة ممن كان يتبع العلم ويعيه يحدث سعيد بن المسيب وساق الحديث…٣٦٢٦أذكركم بالله الذي نجاكم من آل فرعون وأقطعكم البحر وظلل عليكم الغمام…٣٦٢٧إن الله يلوم على العجز ولكن عليك بالكيس فإذا غلبك أمر فقل حسبي الله…٣٦٢٨لى الواجد يحل عرضه وعقوبته٣٦٢٩قال أتيت النبي صلى الله عليه وسلم بغريم لي فقال لي " الزمه " . ثم قال…٣٦٣٠أن النبي صلى الله عليه وسلم حبس رجلا في تهمة .٣٦٣١خلوا له عن جيرانه٣٦٣٢إذا أتيت وكيلي فخذ منه خمسة عشر وسقا فإن ابتغى منك آية فضع يدك على…٣٦٣٣إذا تدارأتم في طريق فاجعلوه سبعة أذرع٣٦٣٤إذا استأذن أحدكم أخاه أن يغرز خشبة في جداره فلا يمنعه٣٦٣٥من ضار أضر الله به ومن شاق شاق الله عليه٣٦٣٦فهبه له ولك كذا وكذا٣٦٣٧اسق يا زبير ثم أرسل إلى جارك٣٦٣٨ثعلبة بن أبي مالك أنه سمع كبراءهم، يذكرون أن رجلا، من قريش كان له سهم…٣٦٣٩أن رسول الله صلى الله عليه وسلم قضى في السيل المهزور أن يمسك حتى يبلغ…٣٦٤٠أن محمد بن عثمان، حدثهم حدثنا عبد العزيز بن محمد، عن أبي طوالة، وعمرو…
