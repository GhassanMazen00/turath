سنن أبي داود

كتاب الصوم

١٦٤ حديثًا•صفحة ١ من ٢
٢٣١٣{ يا أيها الذين آمنوا كتب عليكم الصيام كما كتب على الذين من قبلكم }…٢٣١٤قال كان الرجل إذا صام فنام لم يأكل إلى مثلها وإن صرمة بن قيس الأنصاري…٢٣١٥حدثنا قتيبة بن سعيد، حدثنا بكر، - يعني ابن مضر - عن عمرو بن الحارث،…٢٣١٦{ وعلى الذين يطيقونه فدية طعام مسكين } فكان من شاء منهم أن يفتدي…٢٣١٧أن عكرمة، حدثه أن ابن عباس قال أثبتت للحبلى والمرضع .٢٣١٨حدثنا ابن المثنى، حدثنا ابن أبي عدي، عن سعيد، عن قتادة، عن عزرة، عن…٢٣١٩إنا أمة أمية لا نكتب ولا نحسب الشهر هكذا و هكذا وهكذا٢٣٢٠الشهر تسع وعشرون فلا تصوموا حتى تروه ولا تفطروا حتى تروه فإن غم عليكم…٢٣٢١حدثنا حميد بن مسعدة، حدثنا عبد الوهاب، حدثني أيوب، قال كتب عمر بن عبد…٢٣٢٢حدثنا أحمد بن منيع، عن ابن أبي زائدة، عن عيسى بن دينار، عن أبيه، عن…٢٣٢٣أن يزيد بن زريع، حدثهم حدثنا خالد الحذاء، عن عبد الرحمن بن أبي بكرة،…٢٣٢٤وفطركم يوم تفطرون وأضحاكم يوم تضحون وكل عرفة موقف وكل منى منحر وكل…٢٣٢٥- رضى الله عنها - تقول كان رسول الله صلى الله عليه وسلم يتحفظ من…٢٣٢٦لا تقدموا الشهر حتى تروا الهلال أو تكملوا العدة ثم صوموا حتى تروا…٢٣٢٧لا تقدموا الشهر بصيام يوم ولا يومين إلا أن يكون شىء يصومه أحدكم ولا…٢٣٢٨هل صمت من سرر شعبان شيئا٢٣٢٩صوموا الشهر وسره٢٣٣٠حدثنا سليمان بن عبد الرحمن الدمشقي، - في هذا الحديث - قال قال الوليد…٢٣٣١قال كان سعيد - يعني ابن عبد العزيز - يقول سره أوله . قال أبو داود…٢٣٣٢أن أم الفضل ابنة الحارث، بعثته إلى معاوية بالشام قال فقدمت الشام…٢٣٣٣حدثنا عبيد الله بن معاذ، حدثني أبي، حدثنا الأشعث، عن الحسنفي رجل كان…٢٣٣٤حدثنا محمد بن عبد الله بن نمير، حدثنا أبو خالد الأحمر، عن عمرو بن…٢٣٣٥لا تقدموا صوم رمضان بيوم ولا يومين إلا أن يكون صوما يصومه رجل فليصم…٢٣٣٦حدثنا أحمد بن حنبل، حدثنا محمد بن جعفر، حدثنا شعبة، عن توبة العنبري،…٢٣٣٧إذا انتصف شعبان فلا تصوموا٢٣٣٨حدثنا محمد بن عبد الرحيم أبو يحيى البزاز، حدثنا سعيد بن سليمان، حدثنا…٢٣٣٩وخلف بن هشام المقرئ، قالا حدثنا أبو عوانة، عن منصور، عن ربعي بن حراش،…٢٣٤٠أتشهد أن لا إله إلا الله٢٣٤١أتشهد أن لا إله إلا الله وأني رسول الله٢٣٤٢وعبد الله بن عبد الرحمن السمرقندي، - وأنا لحديثه، أتقن - قالا حدثنا…٢٣٤٣إن فصل ما بين صيامنا وصيام أهل الكتاب أكلة السحر٢٣٤٤هلم إلى الغداء المبارك٢٣٤٥نعم سحور المؤمن التمر٢٣٤٦لا يمنعن من سحوركم أذان بلال ولا بياض الأفق الذي هكذا حتى يستطير٢٣٤٧لا يمنعن أحدكم أذان بلال من سحوره فإنه يؤذن - أو قال ينادي - ليرجع…٢٣٤٨كلوا واشربوا ولا يهيدنكم الساطع المصعد فكلوا واشربوا حتى يعترض لكم…٢٣٤٩إن وسادك لعريض طويل إنما هو الليل والنهار٢٣٥٠إذا سمع أحدكم النداء والإناء على يده فلا يضعه حتى يقضي حاجته منه٢٣٥١إذا جاء الليل من ها هنا وذهب النهار من ها هنا٢٣٥٢يا بلال انزل فاجدح لنا٢٣٥٣لا يزال الدين ظاهرا ما عجل الناس الفطر لأن اليهود والنصارى يؤخرون٢٣٥٤حدثنا مسدد، حدثنا أبو معاوية، عن الأعمش، عن عمارة بن عمير، عن أبي…٢٣٥٥إذا كان أحدكم صائما فليفطر على التمر فإن لم يجد التمر فعلى الماء فإن…٢٣٥٦أنه سمع أنس بن مالك، يقول كان رسول الله صلى الله عليه وسلم يفطر على…٢٣٥٧ذهب الظمأ وابتلت العروق وثبت الأجر إن شاء الله٢٣٥٨اللهم لك صمت وعلى رزقك أفطرت٢٣٥٩ومحمد بن العلاء، - المعنى - قالا حدثنا أبو أسامة، حدثنا هشام بن عروة،…٢٣٦٠إني لست كهيئتكم إني أطعم وأسقى٢٣٦١لا تواصلوا فأيكم أراد أن يواصل فليواصل حتى السحر٢٣٦٢من لم يدع قول الزور والعمل به فليس لله حاجة أن يدع طعامه وشرابه٢٣٦٣الصيام جنة إذا كان أحدكم صائما فلا يرفث ولا يجهل فإن امرؤ قاتله أو…٢٣٦٤قال رأيت رسول الله صلى الله عليه وسلم يستاك وهو صائم . زاد مسدد ما لا…٢٣٦٥حدثنا عبد الله بن مسلمة القعنبي، عن مالك، عن سمى، مولى أبي بكر بن عبد…٢٣٦٦بالغ في الاستنشاق إلا أن تكون صائما٢٣٦٧أفطر الحاجم والمحجوم٢٣٦٨أنه أخبره أن شداد بن أوس بينما هو يمشي مع النبي صلى الله عليه وسلم…٢٣٦٩أفطر الحاجم والمحجوم٢٣٧٠أفطر الحاجم والمحجوم٢٣٧١أفطر الحاجم والمحجوم٢٣٧٢حدثنا أبو معمر عبد الله بن عمرو، حدثنا عبد الوارث، عن أيوب، عن عكرمة،…٢٣٧٣أن رسول الله صلى الله عليه وسلم احتجم وهو صائم محرم .٢٣٧٤إني أواصل إلى السحر وربي يطعمني ويسقيني٢٣٧٥حدثنا عبد الله بن مسلمة، حدثنا سليمان، - يعني ابن المغيرة - عن ثابت،…٢٣٧٦لا يفطر من قاء ولا من احتلم ولا من احتجم٢٣٧٧حدثنا النفيلي، حدثنا علي بن ثابت، حدثني عبد الرحمن بن النعمان بن معبد…٢٣٧٨أنه كان يكتحل وهو صائم .٢٣٧٩ويحيى بن موسى البلخي، قالا حدثنا يحيى بن عيسى، عن الأعمش، قال ما رأيت…٢٣٨٠من ذرعه قىء وهو صائم فليس عليه قضاء وإن استقاء فليقض٢٣٨١أن أباه، حدثه حدثني معدان بن طلحة، أن أبا الدرداء، حدثه أن رسول الله…٢٣٨٢وعلقمة، عن عائشة، قالت كان رسول الله صلى الله عليه وسلم يقبل وهو صائم…٢٣٨٣قالت كان رسول الله صلى الله عليه وسلم يقبل في شهر الصوم .٢٣٨٤حدثنا محمد بن كثير، أخبرنا سفيان، عن سعد بن إبراهيم، عن طلحة بن عبد…٢٣٨٥أرأيت لو مضمضت من الماء وأنت صائم٢٣٨٦أن النبي صلى الله عليه وسلم كان يقبلها وهو صائم ويمص لسانها .٢٣٨٧أن رجلا، سأل النبي صلى الله عليه وسلم عن المباشرة للصائم فرخص له…٢٣٨٨وأم سلمة زوجى النبي صلى الله عليه وسلم أنهما قالتا كان رسول الله صلى…٢٣٨٩وأنا أصبح جنبا وأنا أريد الصيام فأغتسل وأصوم٢٣٩٠ومحمد بن عيسى، - المعنى - قالا حدثنا سفيان، - قال مسدد - حدثنا…٢٣٩١بهذا الحديث بمعناه . زاد الزهري وإنما كان هذا رخصة له خاصة فلو أن…٢٣٩٢أن رجلا، أفطر في رمضان فأمره رسول الله صلى الله عليه وسلم أن يعتق…٢٣٩٣كله أنت وأهل بيتك وصم يوما واستغفر الله٢٣٩٤أن عبد الرحمن بن القاسم، حدثه أن محمد بن جعفر بن الزبير حدثه أن عباد…٢٣٩٥بهذه القصة قال فأتي بعرق فيه عشرون صاعا .٢٣٩٦من أفطر يوما من رمضان من غير رخصة رخصها الله له لم يقض عنه صيام الدهر٢٣٩٧قال قال النبي صلى الله عليه وسلم مثل حديث ابن كثير وسليمان . قال أبو…٢٣٩٨أطعمك الله وسقاك٢٣٩٩أنه سمع عائشة، - رضى الله عنها - تقول إن كان ليكون على الصوم من رمضان…٢٤٠٠من مات وعليه صيام صام عنه وليه٢٤٠١قال إذا مرض الرجل في رمضان ثم مات ولم يصم أطعم عنه ولم يكن عليه قضاء…٢٤٠٢صم إن شئت وأفطر إن شئت٢٤٠٣أى ذلك شئت يا حمزة٢٤٠٤قال خرج النبي صلى الله عليه وسلم من المدينة إلى مكة حتى بلغ عسفان ثم…٢٤٠٥حدثنا أحمد بن يونس، حدثنا زائدة، عن حميد الطويل، عن أنس، قال سافرنا…٢٤٠٦إنكم قد دنوتم من عدوكم والفطر أقوى لكم٢٤٠٧ليس من البر الصيام في السفر٢٤٠٨اجلس فأصب من طعامنا هذا٢٤٠٩حدثتني أم الدرداء، عن أبي الدرداء، قال خرجنا مع رسول الله صلى الله…٢٤١٠من كانت له حمولة تأوي إلى شبع فليصم رمضان حيث أدركه٢٤١١من أدركه رمضان في السفر٢٤١٢وزاد، جعفر والليث حدثني يزيد بن أبي حبيب، أن كليب بن ذهل الحضرمي،…٢٤١٣أن دحية بن خليفة، خرج من قرية من دمشق مرة إلى قدر قرية عقبة من…٢٤١٤أن ابن عمر، كان يخرج إلى الغابة فلا يفطر ولا يقصر .٢٤١٥لا يقولن أحدكم إني صمت رمضان كله و قمته كله٢٤١٦وزهير بن حرب، - وهذا حديثه - قال حدثنا سفيان، عن الزهري، عن أبي عبيد،…٢٤١٧حدثنا موسى بن إسماعيل، حدثنا وهيب، حدثنا عمرو بن يحيى، عن أبيه، عن…٢٤١٨حدثنا عبد الله بن مسلمة القعنبي، عن مالك، عن يزيد بن الهاد، عن أبي…٢٤١٩يوم عرفة ويوم النحر وأيام التشريق عيدنا أهل الإسلام وهي أيام أكل وشرب٢٤٢٠لا يصم أحدكم يوم الجمعة إلا أن يصوم قبله بيوم أو بعده٢٤٢١لا تصوموا يوم السبت إلا فيما افترض عليكم وإن لم يجد أحدكم إلا لحاء…٢٤٢٢أن النبي صلى الله عليه وسلم دخل عليها يوم الجمعة وهي صائمة فقال "…٢٤٢٣حدثنا عبد الملك بن شعيب، حدثنا ابن وهب، قال سمعت الليث، يحدث عن ابن…٢٤٢٤قال ما زلت له كاتما حتى رأيته انتشر . يعني حديث عبد الله بن بسر هذا…٢٤٢٥ومسدد، قالا حدثنا حماد بن زيد، عن غيلان بن جرير، عن عبد الله بن معبد…٢٤٢٦فيه ولدت وفيه أنزل على القرآن٢٤٢٧ألم أحدث أنك تقول لأقومن الليل ولأصومن النهار٢٤٢٨حدثنا موسى بن إسماعيل، حدثنا حماد، عن سعيد الجريري، عن أبي السليل، عن…٢٤٢٩أفضل الصيام بعد شهر رمضان شهر الله المحرم وإن أفضل الصلاة بعد…٢٤٣٠أن رسول الله صلى الله عليه وسلم كان يصوم حتى نقول لا يفطر ويفطر حتى…٢٤٣١سمع عائشة، تقول كان أحب الشهور إلى رسول الله صلى الله عليه وسلم أن…٢٤٣٢إن لأهلك عليك حقا صم رمضان والذي يليه وكل أربعاء وخميس فإذا أنت قد…٢٤٣٣من صام رمضان ثم أتبعه بست من شوال فكأنما صام الدهر٢٤٣٤زوج النبي صلى الله عليه وسلم أنها قالت كان رسول الله صلى الله عليه…٢٤٣٥حدثنا موسى بن إسماعيل، حدثنا حماد، عن محمد بن عمرو، عن أبي سلمة، عن…٢٤٣٦إن أعمال العباد تعرض يوم الاثنين ويوم الخميس٢٤٣٧أزواج النبي صلى الله عليه وسلم قالت كان رسول الله صلى الله عليه وسلم…٢٤٣٨ما من أيام العمل الصالح فيها أحب إلى الله من هذه الأيام٢٤٣٩قالت ما رأيت رسول الله صلى الله عليه وسلم صائما العشر قط .٢٤٤٠حدثنا سليمان بن حرب، حدثنا حوشب بن عقيل، عن مهدي الهجري، حدثنا عكرمة،…٢٤٤١أن ناسا، تماروا عندها يوم عرفة في صوم رسول الله صلى الله عليه وسلم…٢٤٤٢- رضى الله عنها - قالت كان يوم عاشوراء يوما تصومه قريش في الجاهلية…٢٤٤٣هذا يوم من أيام الله فمن شاء صامه ومن شاء تركه٢٤٤٤نحن أولى بموسى منكم٢٤٤٥فإذا كان العام المقبل صمنا يوم التاسع٢٤٤٦حدثنا مسدد، حدثنا يحيى، - يعني ابن سعيد - عن معاوية بن غلاب، ح وحدثنا…٢٤٤٧صمتم يومكم هذا٢٤٤٨أحب الصيام إلى الله تعالى صيام داود وأحب الصلاة إلى الله صلاة داود…٢٤٤٩هن كهيئة الدهر٢٤٥٠قال كان رسول الله صلى الله عليه وسلم يصوم - يعني من غرة كل شهر -…٢٤٥١قالت كان رسول الله صلى الله عليه وسلم يصوم ثلاثة أيام من الشهر…٢٤٥٢حدثنا زهير بن حرب، حدثنا محمد بن فضيل، حدثنا الحسن بن عبيد الله، عن…٢٤٥٣قالت قلت لعائشة أكان رسول الله صلى الله عليه وسلم يصوم من كل شهر…٢٤٥٤من لم يجمع الصيام قبل الفجر فلا صيام له٢٤٥٥حدثنا محمد بن كثير، حدثنا سفيان، ح وحدثنا عثمان بن أبي شيبة، حدثنا…٢٤٥٦أكنت تقضين شيئا٢٤٥٧لا عليكما صوما مكانه يوما آخر٢٤٥٨لا تصوم المرأة وبعلها شاهد إلا بإذنه غير رمضان ولا تأذن في بيته وهو…٢٤٥٩لو كانت سورة واحدة لكفت الناس٢٤٦٠إذا دعي أحدكم فليجب فإن كان مفطرا فليطعم وإن كان صائما فليصل٢٤٦١إذا دعي أحدكم إلى طعام وهو صائم فليقل إني صائم٢٤٦٢أن النبي صلى الله عليه وسلم كان يعتكف العشر الأواخر من رمضان حتى قبضه…
