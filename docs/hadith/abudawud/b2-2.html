سنن أبي داود

كتاب الصلاة

٧٧٠ حديثًا•صفحة ٢ من ٦
٥٤١أن الصلاة، كانت تقام لرسول الله صلى الله عليه وسلم فيأخذ الناس مقامهم…٥٤٢قال أقيمت الصلاة فعرض لرسول الله صلى الله عليه وسلم رجل فحبسه بعد ما…٥٤٣إن الله وملائكته يصلون على الذين يلون الصفوف الأول وما من خطوة أحب…٥٤٤قال أقيمت الصلاة ورسول الله صلى الله عليه وسلم نجي في جانب المسجد فما…٥٤٥قال كان رسول الله صلى الله عليه وسلم حين تقام الصلاة في المسجد إذا…٥٤٦- رضى الله عنه - مثل ذلك .٥٤٧ما من ثلاثة في قرية ولا بدو لا تقام فيهم الصلاة إلا قد استحوذ عليهم…٥٤٨لقد هممت أن آمر بالصلاة فتقام ثم آمر رجلا فيصلي بالناس ثم أنطلق معي…٥٤٩لقد هممت أن آمر فتيتي فيجمعوا حزما من حطب ثم آتي قوما يصلون في بيوتهم…٥٥٠حدثنا هارون بن عباد الأزدي، حدثنا وكيع، عن المسعودي، عن علي بن…٥٥١من سمع المنادي فلم يمنعه من اتباعه عذر٥٥٢أنه سأل النبي صلى الله عليه وسلم فقال يا رسول الله إني رجل ضرير البصر…٥٥٣أتسمع حى على الصلاة حى على الفلاح فحى هلا٥٥٤حدثنا حفص بن عمر، حدثنا شعبة، عن أبي إسحاق، عن عبد الله بن أبي بصير،…٥٥٥من صلى العشاء في جماعة كان كقيام نصف ليلة ومن صلى العشاء والفجر في…٥٥٦الأبعد فالأبعد من المسجد أعظم أجرا٥٥٧أعطاك الله ذلك كله أنطاك الله جل وعز ما احتسبت كله أجمع٥٥٨من خرج من بيته متطهرا إلى صلاة مكتوبة فأجره كأجر الحاج المحرم ومن خرج…٥٥٩صلاة الرجل في جماعة تزيد على صلاته في بيته وصلاته في سوقه خمسا وعشرين…٥٦٠الصلاة في جماعة تعدل خمسا وعشرين صلاة فإذا صلاها في فلاة فأتم ركوعها…٥٦١بشر المشائين في الظلم إلى المساجد بالنور التام يوم القيامة٥٦٢إذا توضأ أحدكم فأحسن وضوءه ثم خرج عامدا إلى المسجد فلا يشبكن يديه…٥٦٣إذا توضأ أحدكم فأحسن الوضوء ثم خرج إلى الصلاة لم يرفع قدمه اليمنى إلا…٥٦٤من توضأ فأحسن وضوءه ثم راح فوجد الناس قد صلوا أعطاه الله جل وعز مثل…٥٦٥لا تمنعوا إماء الله مساجد الله ولكن ليخرجن وهن تفلات٥٦٦لا تمنعوا إماء الله مساجد الله٥٦٧لا تمنعوا نساءكم المساجد وبيوتهن خير لهن٥٦٨ائذنوا للنساء إلى المساجد بالليل٥٦٩أنها أخبرته أن عائشة زوج النبي صلى الله عليه وسلم قالت لو أدرك رسول…٥٧٠صلاة المرأة في بيتها أفضل من صلاتها في حجرتها وصلاتها في مخدعها أفضل…٥٧١لو تركنا هذا الباب للنساء٥٧٢إذا أقيمت الصلاة فلا تأتوها تسعون وأتوها تمشون وعليكم السكينة فما…٥٧٣ائتوا الصلاة وعليكم السكينة فصلوا ما أدركتم واقضوا ما سبقكم٥٧٤ألا رجل يتصدق على هذا فيصلي معه٥٧٥ما منعكما أن تصليا معنا٥٧٦قال صليت مع النبي صلى الله عليه وسلم الصبح بمنى بمعناه .٥٧٧ألم تسلم يا يزيد٥٧٨ذلك له سهم جمع٥٧٩لا تصلوا صلاة في يوم مرتين٥٨٠من أم الناس فأصاب الوقت فله ولهم ومن انتقص من ذلك شيئا فعليه ولا عليهم٥٨١إن من أشراط الساعة أن يتدافع أهل المسجد لا يجدون إماما يصلي بهم٥٨٢يؤم القوم أقرؤهم لكتاب الله وأقدمهم قراءة فإن كانوا في القراءة سواء…٥٨٣ولا يؤم الرجل الرجل في سلطانه٥٨٤فإن كانوا في القراءة سواء فأعلمهم بالسنة فإن كانوا في السنة سواء…٥٨٥حدثنا موسى بن إسماعيل، حدثنا حماد، أخبرنا أيوب، عن عمرو بن سلمة، قال…٥٨٦بهذا الخبر قال فكنت أؤمهم في بردة موصلة فيها فتق فكنت إذا سجدت خرجت…٥٨٧أكثركم جمعا للقرآن٥٨٨حدثنا القعنبي، حدثنا أنس يعني ابن عياض، ح وحدثنا الهيثم بن خالد…٥٨٩إذا حضرت الصلاة فأذنا ثم أقيما ثم ليؤمكما أكبركما٥٩٠ليؤذن لكم خياركم وليؤمكم قراؤكم٥٩١قري في بيتك فإن الله تعالى يرزقك الشهادة٥٩٢حدثنا الحسن بن حماد الحضرمي، حدثنا محمد بن فضيل، عن الوليد بن جميع،…٥٩٣ثلاثة لا يقبل الله منهم صلاة من تقدم قوما وهم له كارهون ورجل أتى…٥٩٤الصلاة المكتوبة واجبة خلف كل مسلم برا كان أو فاجرا وإن عمل الكبائر٥٩٥أن النبي صلى الله عليه وسلم استخلف ابن أم مكتوم يؤم الناس وهو أعمى .٥٩٦من زار قوما فلا يؤمهم وليؤمهم رجل منهم٥٩٧وأحمد بن الفرات أبو مسعود الرازي، - المعنى - قالا حدثنا يعلى، حدثنا…٥٩٨إذا أم الرجل القوم فلا يقم في مكان أرفع من مقامهم٥٩٩أن معاذ بن جبل، كان يصلي مع رسول الله صلى الله عليه وسلم العشاء ثم…٦٠٠سمع جابر بن عبد الله، يقول إن معاذا كان يصلي مع النبي صلى الله عليه…٦٠١إنما جعل الإمام ليؤتم به فإذا صلى قائما فصلوا قياما وإذا ركع فاركعوا…٦٠٢إذا صلى الإمام جالسا فصلوا جلوسا وإذا صلى الإمام قائما فصلوا قياما…٦٠٣إنما جعل الإمام ليؤتم به فإذا كبر فكبروا ولا تكبروا حتى يكبر وإذا ركع…٦٠٤إنما جعل الإمام ليؤتم به٦٠٥إنما جعل الإمام ليؤتم به فإذا ركع فاركعوا وإذا رفع فارفعوا وإذا صلى…٦٠٦ويزيد بن خالد بن موهب، - المعنى - أن الليث، حدثهم عن أبي الزبير، عن…٦٠٧إذا صلى قاعدا فصلوا قعودا٦٠٨ردوا هذا في وعائه وهذا في سقائه فإني صائم٦٠٩حدثنا حفص بن عمر، حدثنا شعبة، عن عبد الله بن المختار، عن موسى بن أنس،…٦١٠حدثنا مسدد، حدثنا يحيى، عن عبد الملك بن أبي سليمان، عن عطاء، عن ابن…٦١١حدثنا عمرو بن عون، أخبرنا هشيم، عن أبي بشر، عن سعيد بن جبير، عن ابن…٦١٢قوموا فلأصلي لكم٦١٣حدثنا عثمان بن أبي شيبة، حدثنا محمد بن فضيل، عن هارون بن عنترة، عن…٦١٤قال صليت خلف رسول الله صلى الله عليه وسلم فكان إذا انصرف انحرف .٦١٥حدثنا محمد بن رافع، حدثنا أبو أحمد الزبيري، حدثنا مسعر، عن ثابت بن…٦١٦لا يصلي الإمام في الموضع الذي صلى فيه حتى يتحول٦١٧إذا قضى الإمام الصلاة وقعد فأحدث قبل أن يتكلم فقد تمت صلاته ومن كان…٦١٨مفتاح الصلاة الطهور وتحريمها التكبير وتحليلها التسليم٦١٩لا تبادروني بركوع ولا بسجود فإنه مهما أسبقكم به إذا ركعت تدركوني به…٦٢٠- وهو غير كذوب - أنهم كانوا إذا رفعوا رءوسهم من الركوع مع رسول الله…٦٢١وهارون بن معروف، - المعنى - قالا حدثنا سفيان، عن أبان بن تغلب، - قال…٦٢٢سمع الله لمن حمده٦٢٣أما يخشى - أو ألا يخشى - أحدكم إذا رفع رأسه والإمام ساجد أن يحول الله…٦٢٤أن النبي صلى الله عليه وسلم حضهم على الصلاة ونهاهم أن ينصرفوا قبل…٦٢٥أولكلكم ثوبان٦٢٦لا يصل أحدكم في الثوب الواحد ليس على منكبيه منه شىء٦٢٧إذا صلى أحدكم في ثوب فليخالف بطرفيه على عاتقيه٦٢٨قال رأيت رسول الله صلى الله عليه وسلم يصلي في ثوب واحد ملتحفا مخالفا…٦٢٩أوكلكم يجد ثوبين٦٣٠حدثنا محمد بن سليمان الأنباري، حدثنا وكيع، عن سفيان، عن أبي حازم، عن…٦٣١رضى الله عنها أن النبي صلى الله عليه وسلم صلى في ثوب واحد بعضه على .٦٣٢نعم وازرره ولو بشوكة٦٣٣حدثنا محمد بن حاتم بن بزيع، حدثنا يحيى بن أبي بكير، عن إسرائيل، عن…٦٣٤وسليمان بن عبد الرحمن الدمشقي، ويحيى بن الفضل السجستاني، قالوا حدثنا…٦٣٥إذا كان لأحدكم ثوبان فليصل فيهما فإن لم يكن إلا ثوب واحد فليتزر به…٦٣٦قال نهى رسول الله صلى الله عليه وسلم أن يصلى في لحاف لا يتوشح به…٦٣٧من أسبل إزاره في صلاته خيلاء فليس من الله في حل ولا حرام٦٣٨قال بينما رجل يصلي مسبلا إزاره إذ قال له رسول الله صلى الله عليه وسلم…٦٣٩أنها سألت أم سلمة ماذا تصلي فيه المرأة من الثياب فقالت تصلي في الخمار…٦٤٠إذا كان الدرع سابغا يغطي ظهور قدميها٦٤١لا يقبل الله صلاة حائض إلا بخمار٦٤٢شقيه بشقتين فأعطي هذه نصفا والفتاة التي عند أم سلمة نصفا فإني لا…٦٤٣وإبراهيم بن موسى، عن ابن المبارك، عن الحسن بن ذكوان، عن سليمان…٦٤٤قال أكثر ما رأيت عطاء يصلي سادلا . قال أبو داود وهذا يضعف ذلك الحديث .٦٤٥حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا الأشعث، عن محمد، - يعني ابن…٦٤٦ذلك كفل الشيطان٦٤٧إنما مثل هذا مثل الذي يصلي وهو مكتوف٦٤٨حدثنا مسدد، حدثنا يحيى، عن ابن جريج، حدثني محمد بن عباد بن جعفر، عن…٦٤٩وعبد الله بن المسيب العابدي، وعبد الله بن عمرو، عن عبد الله بن…٦٥٠ما حملكم على إلقائكم نعالكم٦٥١حدثنا موسى، - يعني ابن إسماعيل - حدثنا أبان، حدثنا قتادة، حدثني بكر…٦٥٢خالفوا اليهود فإنهم لا يصلون في نعالهم ولا خفافهم٦٥٣قال رأيت رسول الله صلى الله عليه وسلم يصلي حافيا ومنتعلا .٦٥٤إذا صلى أحدكم فلا يضع نعليه عن يمينه ولا عن يساره فتكون عن يمين غيره…٦٥٥إذا صلى أحدكم فخلع نعليه فلا يؤذ بهما أحدا ليجعلهما بين رجليه أو ليصل…٦٥٦حدثتني ميمونة بنت الحارث، قالت كان رسول الله صلى الله عليه وسلم يصلي…٦٥٧قال قال رجل من الأنصار يا رسول الله إني رجل ضخم - وكان ضخما - لا…٦٥٨حدثنا مسلم بن إبراهيم، حدثنا المثنى بن سعيد الذراع، حدثنا قتادة، عن…٦٥٩وعثمان بن أبي شيبة، - بمعنى الإسناد والحديث - قالا حدثنا أبو أحمد…٦٦٠حدثنا أحمد بن حنبل، حدثنا بشر، - يعني ابن المفضل - حدثنا غالب القطان،…٦٦١ألا تصفون كما تصف الملائكة عند ربهم جل وعز٦٦٢يقول أقبل رسول الله صلى الله عليه وسلم على الناس بوجهه فقال " أقيموا…٦٦٣لتسون صفوفكم أو ليخالفن الله بين وجوهكم٦٦٤لا تختلفوا فتختلف قلوبكم٦٦٥حدثنا ابن معاذ، حدثنا خالد بن الحارث، حدثنا حاتم، - يعني ابن أبي…٦٦٦أقيموا الصفوف وحاذوا بين المناكب وسدوا الخلل ولينوا بأيدي إخوانكم٦٦٧رصوا صفوفكم وقاربوا بينها وحاذوا بالأعناق فوالذي نفسي بيده إني لأرى…٦٦٨سووا صفوفكم فإن تسوية الصف من تمام الصلاة٦٦٩استووا وعدلوا صفوفكم٦٧٠اعتدلوا سووا صفوفكم٦٧١أتموا الصف المقدم ثم الذي يليه فما كان من نقص فليكن في الصف المؤخر٦٧٢خياركم ألينكم مناكب في الصلاة٦٧٣حدثنا محمد بن بشار، حدثنا عبد الرحمن، حدثنا سفيان، عن يحيى بن هانئ،…٦٧٤ليلني منكم أولو الأحلام والنهى ثم الذين يلونهم ثم الذين يلونهم٦٧٥ولا تختلفوا فتختلف قلوبكم وإياكم وهيشات الأسواق٦٧٦إن الله وملائكته يصلون على ميامن الصفوف٦٧٧قال قال أبو مالك الأشعري ألا أحدثكم بصلاة النبي صلى الله عليه وسلم…٦٧٨خير صفوف الرجال أولها وشرها آخرها وخير صفوف النساء آخرها وشرها أولها٦٧٩لا يزال قوم يتأخرون عن الصف الأول حتى يؤخرهم الله في النار٦٨٠تقدموا فائتموا بي وليأتم بكم من بعدكم ولا يزال قوم يتأخرون حتى يؤخرهم…٦٨١وسطوا الإمام وسدوا الخلل٦٨٢وحفص بن عمر، قالا حدثنا شعبة، عن عمرو بن مرة، عن هلال بن يساف، عن…٦٨٣زادك الله حرصا ولا تعد٦٨٤أيكم الذي ركع دون الصف ثم مشى إلى الصف٦٨٥إذا جعلت بين يديك مثل مؤخرة الرحل فلا يضرك من مر بين يديك٦٨٦قال آخرة الرحل ذراع فما فوقه .٦٨٧أن رسول الله صلى الله عليه وسلم كان إذا خرج يوم العيد أمر بالحربة…٦٨٨أن النبي صلى الله عليه وسلم صلى بهم بالبطحاء وبين يديه عنزة الظهر…٦٨٩إذا صلى أحدكم فليجعل تلقاء وجهه شيئا فإن لم يجد فلينصب عصا فإن لم يكن…٦٩٠حدثنا محمد بن يحيى بن فارس، حدثنا علي، - يعني ابن المديني - عن سفيان،…
