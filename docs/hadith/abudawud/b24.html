سنن أبي داود

كتاب الإجارة

١٥٥ حديثًا•صفحة ١ من ٢
٣٤١٦إن كنت تحب أن تطوق طوقا من نار فاقبلها٣٤١٧جمرة بين كتفيك تقلدتها٣٤١٨من أين علمتم أنها رقية أحسنتم واضربوا لي معكم بسهم٣٤١٩حدثنا الحسن بن علي، حدثنا يزيد بن هارون، أخبرنا هشام بن حسان، عن محمد…٣٤٢٠كل فلعمري لمن أكل برقية باطل لقد أكلت برقية حق٣٤٢١كسب الحجام خبيث وثمن الكلب خبيث ومهر البغي خبيث٣٤٢٢أنه استأذن رسول الله صلى الله عليه وسلم في إجارة الحجام فنهاه عنها…٣٤٢٣قال احتجم رسول الله صلى الله عليه وسلم وأعطى الحجام أجره ولو علمه…٣٤٢٤أنه قال حجم أبو طيبة رسول الله صلى الله عليه وسلم فأمر له بصاع من تمر…٣٤٢٥سمع أبا هريرة، قال نهى رسول الله صلى الله عليه وسلم عن كسب الإماء .٣٤٢٦حدثنا هارون بن عبد الله، حدثنا هاشم بن القاسم، حدثنا عكرمة، حدثني…٣٤٢٧حدثني أحمد بن صالح، حدثنا ابن أبي فديك، عن عبيد الله، - يعني ابن هرير…٣٤٢٨حدثنا قتيبة، عن سفيان، عن الزهري، عن أبي بكر بن عبد الرحمن، عن أبي…٣٤٢٩حدثنا مسدد بن مسرهد، حدثنا إسماعيل، عن علي بن الحكم، عن نافع، عن ابن…٣٤٣٠إني وهبت لخالتي غلاما وأنا أرجو أن يبارك لها فيه فقلت لها لا تسلميه…٣٤٣١حدثنا يوسف بن موسى، حدثنا سلمة بن الفضل، حدثنا ابن إسحاق، عن العلاء…٣٤٣٢حدثنا الفضل بن يعقوب، حدثنا عبد الأعلى، عن محمد بن إسحاق، حدثنا…٣٤٣٣من باع عبدا وله مال فماله للبائع إلا أن يشترطه المبتاع ومن باع نخلا…٣٤٣٤حدثنا القعنبي، عن مالك، عن نافع، عن ابن عمر، عن عمر، عن رسول الله صلى…٣٤٣٥من باع عبدا وله مال فماله للبائع إلا أن يشترط المبتاع٣٤٣٦لا يبع بعضكم على بيع بعض ولا تلقوا السلع حتى يهبط بها الأسواق٣٤٣٧حدثنا الربيع بن نافع أبو توبة، حدثنا عبيد الله، - يعني ابن عمرو الرقي…٣٤٣٨قال قال النبي صلى الله عليه وسلم " لا تناجشوا " .٣٤٣٩قال نهى رسول الله صلى الله عليه وسلم أن يبيع حاضر لباد . فقلت ما يبيع…٣٤٤٠لا يبيع حاضر لباد وإن كان أخاه أو أباه٣٤٤١أن أعرابيا، حدثه أنه، قدم بحلوبة له على عهد رسول الله صلى الله عليه…٣٤٤٢لا يبع حاضر لباد وذروا الناس يرزق الله بعضهم من بعض٣٤٤٣لا تلقوا الركبان للبيع ولا يبع بعضكم على بيع بعض ولا تصروا الإبل…٣٤٤٤من اشترى شاة مصراة فهو بالخيار ثلاثة أيام إن شاء ردها وصاعا من طعام…٣٤٤٥من اشترى غنما مصراة احتلبها فإن رضيها أمسكها وإن سخطها ففي حلبتها صاع…٣٤٤٦من ابتاع محفلة فهو بالخيار ثلاثة أيام فإن ردها رد معها مثل أو مثلى…٣٤٤٧لا يحتكر إلا خاطئ٣٤٤٨حدثنا محمد بن يحيى بن فياض، حدثنا أبي ح، وحدثنا ابن المثنى، حدثنا…٣٤٤٩قال نهى رسول الله صلى الله عليه وسلم أن تكسر سكة المسلمين الجائزة…٣٤٥٠أن سليمان بن بلال، حدثهم حدثني العلاء بن عبد الرحمن، عن أبيه، عن أبي…٣٤٥١إن الله هو المسعر القابض الباسط الرازق وإني لأرجو أن ألقى الله وليس…٣٤٥٢حدثنا أحمد بن محمد بن حنبل، حدثنا سفيان بن عيينة، عن العلاء، عن أبيه،…٣٤٥٣حدثنا الحسن بن الصباح، عن علي، عن يحيى، قال كان سفيان يكره هذا…٣٤٥٤المتبايعان كل واحد منهما بالخيار على صاحبه ما لم يفترقا إلا بيع الخيار٣٤٥٥أو يقول أحدهما لصاحبه اختر٣٤٥٦المتبايعان بالخيار ما لم يفترقا إلا أن تكون صفقة خيار ولا يحل له أن…٣٤٥٧البيعان بالخيار ما لم يتفرقا٣٤٥٨لا يفترقن اثنان إلا عن تراض٣٤٥٩البيعان بالخيار ما لم يفترقا فإن صدقا وبينا بورك لهما في بيعهما وإن…٣٤٦٠من أقال مسلما أقاله الله عثرته٣٤٦١من باع بيعتين في بيعة فله أوكسهما أو الربا٣٤٦٢إذا تبايعتم بالعينة وأخذتم أذناب البقر ورضيتم بالزرع وتركتم الجهاد…٣٤٦٣من أسلف في تمر فليسلف في كيل معلوم ووزن معلوم إلى أجل معلوم٣٤٦٤حدثنا حفص بن عمر، حدثنا شعبة، ح وحدثنا ابن كثير، أخبرنا شعبة، أخبرني…٣٤٦٥وابن، مهدي قالا حدثنا شعبة، عن عبد الله بن أبي المجالد، وقال عبد…٣٤٦٦حدثنا محمد بن المصفى، حدثنا أبو المغيرة، حدثنا عبد الملك بن أبي غنية،…٣٤٦٧بم تستحل ماله اردد عليه ماله٣٤٦٨من أسلف في شىء فلا يصرفه إلى غيره٣٤٦٩أنه قال أصيب رجل في عهد رسول الله صلى الله عليه وسلم في ثمار ابتاعها…٣٤٧٠إن بعت من أخيك تمرا فأصابتها جائحة فلا يحل لك أن تأخذ منه شيئا بم…٣٤٧١قال الجوائح كل ظاهر مفسد من مطر أو برد أو جراد أو ريح أو حريق .٣٤٧٢أنه قال لا جائحة فيما أصيب دون ثلث رأس المال - قال يحيى - وذلك في سنة…٣٤٧٣لا يمنع فضل الماء ليمنع به الكلأ٣٤٧٤ثلاثة لا يكلمهم الله يوم القيامة رجل منع ابن السبيل فضل ماء عنده ورجل…٣٤٧٥ولا يزكيهم ولهم عذاب أليم٣٤٧٦قالت استأذن أبي النبي صلى الله عليه وسلم فدخل بينه وبين قميصه فجعل…٣٤٧٧المسلمون شركاء في ثلاث في الكلإ والماء والنار٣٤٧٨حدثنا عبد الله بن محمد النفيلي، حدثنا داود بن عبد الرحمن العطار، عن…٣٤٧٩وعلي بن بحر، قالا حدثنا عيسى، وقال، إبراهيم أخبرنا عن الأعمش، عن أبي…٣٤٨٠أنه سمع أبا الزبير، عن جابر، أن النبي صلى الله عليه وسلم نهى عن ثمن…٣٤٨١حدثنا قتيبة بن سعيد، حدثنا سفيان، عن الزهري، عن أبي بكر بن عبد…٣٤٨٢حدثنا الربيع بن نافع أبو توبة، حدثنا عبيد الله، - يعني ابن عمرو - عن…٣٤٨٣أن أباه، قال إن رسول الله صلى الله عليه وسلم نهى عن ثمن الكلب .٣٤٨٤لا يحل ثمن الكلب ولا حلوان الكاهن ولا مهر البغي٣٤٨٥إن الله حرم الخمر وثمنها وحرم الميتة وثمنها وحرم الخنزير وثمنه٣٤٨٦إن الله حرم بيع الخمر والميتة والخنزير والأصنام٣٤٨٧نحوه لم يقل " هو حرام " .٣٤٨٨لعن الله اليهود٣٤٨٩من باع الخمر فليشقص الخنازير٣٤٩٠حرمت التجارة في الخمر٣٤٩١بإسناده ومعناه قال الآيات الأواخر في الربا .٣٤٩٢من ابتاع طعاما فلا يبعه حتى يستوفيه٣٤٩٣حدثنا عبد الله بن مسلمة، عن مالك، عن نافع، عن ابن عمر، أنه قال كنا في…٣٤٩٤قال كانوا يتبايعون الطعام جزافا بأعلى السوق فنهى رسول الله صلى الله…٣٤٩٥أن القاسم بن محمد، حدثه أن عبد الله بن عمر حدثه أن رسول الله صلى الله…٣٤٩٦من ابتاع طعاما فلا يبعه حتى يكتاله٣٤٩٧إذا اشترى أحدكم طعاما فلا يبعه حتى يقبضه٣٤٩٨قال رأيت الناس يضربون على عهد رسول الله صلى الله عليه وسلم إذا اشتروا…٣٤٩٩حدثنا محمد بن عوف الطائي، حدثنا أحمد بن خالد الوهبي، حدثنا محمد بن…٣٥٠٠إذا بايعت فقل لا خلابة٣٥٠١إن كنت غير تارك البيع فقل هاء وهاء ولا خلابة٣٥٠٢حدثنا عبد الله بن مسلمة، قال قرأت على مالك بن أنس أنه بلغه عن عمرو بن…٣٥٠٣لا تبع ما ليس عندك٣٥٠٤لا يحل سلف وبيع ولا شرطان في بيع ولا ربح ما لم تضمن ولا بيع ما ليس…٣٥٠٥تراني إنما ماكستك لأذهب بجملك خذ جملك وثمنه فهما لك٣٥٠٦عهدة الرقيق ثلاثة أيام٣٥٠٧بإسناده ومعناه زاد إن وجد داء في الثلاث الليالي رد بغير بينة وإن وجد…٣٥٠٨- رضى الله عنها - قالت قال رسول الله صلى الله عليه وسلم " الخراج…٣٥٠٩حدثنا محمود بن خالد الفريابي، عن سفيان، عن محمد بن عبد الرحمن، عن…٣٥١٠رضى الله عنها أن رجلا، ابتاع غلاما فأقام عنده ما شاء الله أن يقيم ثم…٣٥١١إذا اختلف البيعان وليس بينهما بينة فهو ما يقول رب السلعة أو يتتاركان٣٥١٢أن القاسم بن عبد الرحمن، عن أبيه، أن ابن مسعود، باع من الأشعث بن قيس…٣٥١٣الشفعة في كل شرك ربعة أو حائط لا يصلح أن يبيع حتى يؤذن شريكه فإن باع…٣٥١٤قال إنما جعل رسول الله صلى الله عليه وسلم الشفعة في كل مال لم يقسم…٣٥١٥إذا قسمت الأرض وحدت فلا شفعة فيها٣٥١٦الجار أحق بسقبه٣٥١٧جار الدار أحق بدار الجار أو الأرض٣٥١٨الجار أحق بشفعة جاره ينتظر بها وإن كان غائبا إذا كان طريقهما واحدا٣٥١٩أيما رجل أفلس فأدرك الرجل متاعه بعينه فهو أحق به من غيره٣٥٢٠أيما رجل باع متاعا فأفلس الذي ابتاعه ولم يقبض الذي باعه من ثمنه شيئا…٣٥٢١وإن قضى من ثمنها شيئا فهو أسوة الغرماء فيها٣٥٢٢فإن كان قضاه من ثمنها شيئا فما بقي فهو أسوة الغرماء وأيما امرئ هلك…٣٥٢٣من أفلس أو مات فوجد رجل متاعه بعينه فهو أحق به٣٥٢٤من وجد دابة قد عجز عنها أهلها أن يعلفوها فسيبوها فأخذها فأحياها فهي له٣٥٢٥من ترك دابة بمهلك فأحياها رجل فهي لمن أحياها٣٥٢٦لبن الدر يحلب بنفقته إذا كان مرهونا والظهر يركب بنفقته إذا كان مرهونا…٣٥٢٧إن من عباد الله لأناسا ما هم بأنبياء ولا شهداء يغبطهم الأنبياء…٣٥٢٨إن من أطيب ما أكل الرجل من كسبه وولده من كسبه٣٥٢٩ولد الرجل من كسبه من أطيب كسبه فكلوا من أموالهم٣٥٣٠أنت ومالك لوالدك إن أولادكم من أطيب كسبكم فكلوا من كسب أولادكم٣٥٣١من وجد عين ماله عند رجل فهو أحق به ويتبع البيع من باعه٣٥٣٢خذي ما يكفيك وبنيك بالمعروف٣٥٣٣لا حرج عليك أن تنفقي عليهم بالمعروف٣٥٣٤أد الأمانة إلى من ائتمنك ولا تخن من خانك٣٥٣٥أد الأمانة إلى من ائتمنك ولا تخن من خانك٣٥٣٦وعبد الرحيم بن مطرف الرؤاسي، قالا حدثنا عيسى، - وهو ابن يونس بن أبي…٣٥٣٧وايم الله لا أقبل بعد يومي هذا من أحد هدية إلا أن يكون مهاجرا قرشيا…٣٥٣٨العائد في هبته كالعائد في قيئه٣٥٣٩لا يحل لرجل أن يعطي عطية أو يهب هبة فيرجع فيها إلا الوالد فيما يعطي…٣٥٤٠مثل الذي يسترد ما وهب كمثل الكلب يقيء فيأكل قيئه فإذا استرد الواهب…٣٥٤١من شفع لأخيه بشفاعة فأهدى له هدية عليها فقبلها فقد أتى بابا عظيما من…٣٥٤٢وإسماعيل بن سالم، عن الشعبي، عن النعمان بن بشير، قال أنحلني أبي نحلا…٣٥٤٣قال أعطاه أبوه غلاما فقال له رسول الله صلى الله عليه وسلم " ما هذا…٣٥٤٤اعدلوا بين أولادكم اعدلوا بين أبنائكم٣٥٤٥قال قالت امرأة بشير انحل ابني غلامك وأشهد لي رسول الله صلى الله عليه…٣٥٤٦لا يجوز لامرأة أمر في مالها إذا ملك زوجها عصمتها٣٥٤٧لا يجوز لامرأة عطية إلا بإذن زوجها٣٥٤٨حدثنا أبو الوليد الطيالسي، حدثنا همام، عن قتادة، عن النضر بن أنس، عن…٣٥٤٩حدثنا أبو الوليد، حدثنا همام، عن قتادة، عن الحسن، عن سمرة، عن النبي…٣٥٥٠العمرى لمن وهبت له٣٥٥١من أعمر عمرى فهي له ولعقبه يرثها من يرثه من عقبه٣٥٥٢وعروة، عن جابر، عن النبي صلى الله عليه وسلم بمعناه . قال أبو داود…٣٥٥٣أيما رجل أعمر عمرى له ولعقبه فإنها للذي يعطاها لا ترجع إلى الذي…٣٥٥٤ويزيد بن أبي حبيب، عن ابن شهاب، واختلف، على الأوزاعي في لفظه عن ابن…٣٥٥٥قال إنما العمرى التي أجازها رسول الله صلى الله عليه وسلم أن يقول هي…٣٥٥٦لا ترقبوا ولا تعمروا فمن أرقب شيئا أو أعمره فهو لورثته٣٥٥٧هي لها حياتها وموتها٣٥٥٨العمرى جائزة لأهلها والرقبى جائزة لأهلها٣٥٥٩من أعمر شيئا فهو لمعمره محياه ومماته ولا ترقبوا فمن أرقب شيئا فهو…٣٥٦٠قال العمرى أن يقول الرجل للرجل هو لك ما عشت فإذا قال ذلك فهو له…٣٥٦١على اليد ما أخذت حتى تؤدي٣٥٦٢لا بل عارية مضمونة٣٥٦٣يا صفوان هل عندك من سلاح٣٥٦٤من آل صفوان قال استعار النبي صلى الله عليه وسلم فذكر معناه .٣٥٦٥إن الله عز وجل قد أعطى كل ذي حق حقه فلا وصية لوارث ولا تنفق المرأة…
