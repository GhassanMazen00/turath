سنن أبي داود

كتاب الطهارة

٣٩٠ حديثًا•صفحة ٢ من ٣
١٥١دع الخفين فإني أدخلت القدمين الخفين وهما طاهرتان١٥٢أن المغيرة بن شعبة، قال تخلف رسول الله صلى الله عليه وسلم فذكر هذه…١٥٣- يعني ابن حفص بن عمر بن سعد - سمع أبا عبد الله، عن أبي عبد الرحمن…١٥٤أن جريرا، بال ثم توضأ فمسح على الخفين وقال ما يمنعني أن أمسح وقد رأيت…١٥٥وأحمد بن أبي شعيب الحراني، قالا حدثنا وكيع، حدثنا دلهم بن صالح، عن…١٥٦بل أنت نسيت بهذا أمرني ربي عز وجل١٥٧المسح على الخفين للمسافر ثلاثة أيام وللمقيم يوم وليلة١٥٨حدثنا يحيى بن معين، حدثنا عمرو بن الربيع بن طارق، أخبرنا يحيى بن…١٥٩حدثنا عثمان بن أبي شيبة، عن وكيع، عن سفيان الثوري، عن أبي قيس الأودي،…١٦٠وعباد بن موسى، قالا حدثنا هشيم، عن يعلى بن عطاء، عن أبيه، - قال عباد…١٦١أن رسول الله صلى الله عليه وسلم كان يمسح على الخفين . وقال غير محمد…١٦٢- رضى الله عنه - قال لو كان الدين بالرأى لكان أسفل الخف أولى بالمسح…١٦٣بإسناده بهذا الحديث قال ما كنت أرى باطن القدمين إلا أحق بالغسل حتى…١٦٤حدثنا محمد بن العلاء، حدثنا حفص بن غياث، عن الأعمش، بهذا الحديث قال…١٦٥ومحمود بن خالد الدمشقي، - المعنى - قالا حدثنا الوليد، - قال محمود -…١٦٦أو الحكم بن سفيان قال كان رسول الله صلى الله عليه وسلم إذا بال يتوضأ…١٦٧قال رأيت رسول الله صلى الله عليه وسلم بال ثم نضح فرجه .١٦٨أن رسول الله صلى الله عليه وسلم بال ثم توضأ ونضح فرجه .١٦٩ما منكم من أحد يتوضأ فيحسن الوضوء ثم يقوم فيركع ركعتين يقبل عليهما…١٧٠حدثنا الحسين بن عيسى، حدثنا عبد الله بن يزيد المقرئ، عن حيوة، - وهو…١٧١حدثنا محمد بن عيسى، حدثنا شريك، عن عمرو بن عامر البجلي، - قال محمد هو…١٧٢قال صلى رسول الله صلى الله عليه وسلم يوم الفتح خمس صلوات بوضوء واحد…١٧٣ارجع فأحسن وضوءك١٧٤وحميد، عن الحسن، عن النبي صلى الله عليه وسلم بمعنى قتادة .١٧٥أصحاب النبي أن النبي صلى الله عليه وسلم رأى رجلا يصلي وفي ظهر قدمه…١٧٦لا ينفتل حتى يسمع صوتا أو يجد ريحا١٧٧إذا كان أحدكم في الصلاة فوجد حركة في دبره أحدث أو لم يحدث فأشكل عليه…١٧٨وعبد الرحمن، قالا حدثنا سفيان، عن أبي روق، عن إبراهيم التيمي، عن…١٧٩حدثنا عثمان بن أبي شيبة، حدثنا وكيع، حدثنا الأعمش، عن حبيب، عن عروة،…١٨٠حدثنا إبراهيم بن مخلد الطالقاني، حدثنا عبد الرحمن، - يعني ابن مغراء -…١٨١من مس ذكره فليتوضأ١٨٢هل هو إلا مضغة منه١٨٣بإسناده ومعناه وقال في الصلاة .١٨٤حدثنا عثمان بن أبي شيبة، حدثنا أبو معاوية، حدثنا الأعمش، عن عبد الله…١٨٥تنح حتى أريك فأدخل يده بين الجلد واللحم فدحس بها حتى توارت إلى الإبط…١٨٦أيكم يحب أن هذا له١٨٧أن رسول الله صلى الله عليه وسلم أكل كتف شاة ثم صلى ولم يتوضأ .١٨٨ما له تربت يداه١٨٩قال أكل رسول الله صلى الله عليه وسلم كتفا ثم مسح يده بمسح كان تحته ثم…١٩٠أن رسول الله صلى الله عليه وسلم انتهش من كتف ثم صلى ولم يتوضأ .١٩١يقول قربت للنبي صلى الله عليه وسلم خبزا ولحما فأكل ثم دعا بوضوء فتوضأ…١٩٢قال كان آخر الأمرين من رسول الله صلى الله عليه وسلم ترك الوضوء مما…١٩٣حدثنا أحمد بن عمرو بن السرح، حدثنا عبد الملك بن أبي كريمة، - قال ابن…١٩٤الوضوء مما أنضجت النار١٩٥توضئوا مما غيرت النار١٩٦حدثنا قتيبة بن سعيد، حدثنا الليث، عن عقيل، عن الزهري، عن عبيد الله بن…١٩٧أنه سمع أنس بن مالك، يقول إن رسول الله صلى الله عليه وسلم شرب لبنا…١٩٨حدثنا أبو توبة الربيع بن نافع، حدثنا ابن المبارك، عن محمد بن إسحاق،…١٩٩ليس أحد ينتظر الصلاة غيركم٢٠٠حدثنا شاذ بن فياض، حدثنا هشام الدستوائي، عن قتادة، عن أنس، قال كان…٢٠١وداود بن شبيب، قالا حدثنا حماد بن سلمة، عن ثابت البناني، أن أنس بن…٢٠٢إنما الوضوء على من نام مضطجعا٢٠٣وكاء السه العينان فمن نام فليتوضأ٢٠٤وإبراهيم بن أبي معاوية، عن أبي معاوية، ح وحدثنا عثمان بن أبي شيبة،…٢٠٥إذا فسا أحدكم في الصلاة فلينصرف فليتوضأ وليعد الصلاة٢٠٦لا تفعل إذا رأيت المذى فاغسل ذكرك وتوضأ وضوءك للصلاة فإذا فضخت الماء…٢٠٧إذا وجد أحدكم ذلك فلينضح فرجه وليتوضأ وضوءه للصلاة٢٠٨ليغسل ذكره وأنثييه٢٠٩حدثنا عبد الله بن مسلمة القعنبي، قال حدثنا أبي، عن هشام بن عروة، عن…٢١٠إنما يجزيك من ذلك الوضوء٢١١ذاك المذى وكل فحل يمذي فتغسل من ذلك فرجك وأنثييك وتوضأ وضوءك للصلاة٢١٢لك ما فوق الإزار٢١٣ما فوق الإزار والتعفف عن ذلك أفضل٢١٤من أرضى أن سهل بن سعد الساعدي، أخبره أن أبى بن كعب أخبره أن رسول الله…٢١٥أن الفتيا التي، كانوا يفتون أن الماء من الماء كانت رخصة رخصها رسول…٢١٦إذا قعد بين شعبها الأربع وألزق الختان بالختان فقد وجب الغسل٢١٧أن رسول الله صلى الله عليه وسلم قال " الماء من الماء " . وكان أبو…٢١٨حدثنا مسدد بن مسرهد، حدثنا إسماعيل، حدثنا حميد الطويل، عن أنس، أن…٢١٩هذا أزكى وأطيب وأطهر٢٢٠إذا أتى أحدكم أهله ثم بدا له أن يعاود فليتوضأ بينهما وضوءا٢٢١توضأ واغسل ذكرك ثم نم٢٢٢وقتيبة بن سعيد، قالا حدثنا سفيان، عن الزهري، عن أبي سلمة، عن عائشة،…٢٢٣وإذا أراد أن يأكل وهو جنب غسل يديه٢٢٤أن النبي صلى الله عليه وسلم كان إذا أراد أن يأكل أو ينام توضأ . تعني…٢٢٥أن النبي صلى الله عليه وسلم رخص للجنب إذا أكل أو شرب أو نام أن يتوضأ…٢٢٦قال قلت لعائشة أرأيت رسول الله صلى الله عليه وسلم كان يغتسل من…٢٢٧لا تدخل الملائكة بيتا فيه صورة ولا كلب ولا جنب٢٢٨حدثنا محمد بن كثير، أخبرنا سفيان، عن أبي إسحاق، عن الأسود، عن عائشة،…٢٢٩حدثنا حفص بن عمر، حدثنا شعبة، عن عمرو بن مرة، عن عبد الله بن سلمة،…٢٣٠إن المسلم لا ينجس٢٣١أين كنت يا أبا هريرة٢٣٢وجهوا هذه البيوت عن المسجد٢٣٣أن رسول الله صلى الله عليه وسلم دخل في صلاة الفجر فأومأ بيده أن…٢٣٤إنما أنا بشر وإني كنت جنبا٢٣٥حدثنا عمرو بن عثمان، حدثنا محمد بن حرب، حدثنا الزبيدي، ح وحدثنا عياش…٢٣٦حدثنا قتيبة بن سعيد، حدثنا حماد بن خالد الخياط، حدثنا عبد الله…٢٣٧نعم فلتغتسل إذا وجدت الماء٢٣٨حدثنا عبد الله بن مسلمة القعنبي، عن مالك، عن ابن شهاب، عن عروة، عن…٢٣٩أما أنا فأفيض على رأسي ثلاثا٢٤٠قالت كان رسول الله صلى الله عليه وسلم إذا اغتسل من الجنابة دعا بشىء…٢٤١حدثنا يعقوب بن إبراهيم، حدثنا عبد الرحمن، - يعني ابن مهدي - عن زائدة…٢٤٢ومسدد، قالا حدثنا حماد، عن هشام بن عروة، عن أبيه، عن عائشة، قالت كان…٢٤٣قالت كان رسول الله صلى الله عليه وسلم إذا أراد أن يغتسل من الجنابة…٢٤٤قال قالت عائشة رضى الله عنها لئن شئتم لأرينكم أثر يد رسول الله صلى…٢٤٥حدثنا مسدد بن مسرهد، حدثنا عبد الله بن داود، عن الأعمش، عن سالم، عن…٢٤٦قال إن ابن عباس كان إذا اغتسل من الجنابة يفرغ بيده اليمنى على يده…٢٤٧قال كانت الصلاة خمسين والغسل من الجنابة سبع مرار وغسل البول من الثوب…٢٤٨إن تحت كل شعرة جنابة فاغسلوا الشعر وأنقوا البشر٢٤٩من ترك موضع شعرة من جنابة لم يغسلها فعل به كذا وكذا من النار٢٥٠قالت كان رسول الله صلى الله عليه وسلم يغتسل ويصلي الركعتين وصلاة…٢٥١إنما يكفيك أن تحفني عليه ثلاثا٢٥٢واغمزي قرونك عند كل حفنة٢٥٣حدثنا عثمان بن أبي شيبة، حدثنا يحيى بن أبي بكير، حدثنا إبراهيم بن…٢٥٤حدثنا نصر بن علي، حدثنا عبد الله بن داود، عن عمر بن سويد، عن عائشة…٢٥٥أما الرجل فلينشر رأسه فليغسله حتى يبلغ أصول الشعر وأما المرأة فلا…٢٥٦حدثنا محمد بن جعفر بن زياد، حدثنا شريك، عن قيس بن وهب، عن رجل، من بني…٢٥٧فيما يفيض بين الرجل والمرأة من الماء قالت كان رسول الله صلى الله عليه…٢٥٨جامعوهن في البيوت واصنعوا كل شىء غير النكاح٢٥٩فأعطيه النبي صلى الله عليه وسلم فيضع فمه في الموضع الذي فيه وضعته…٢٦٠حدثنا محمد بن كثير، حدثنا سفيان، عن منصور بن عبد الرحمن، عن صفية، عن…٢٦١ناوليني الخمرة من المسجد٢٦٢أن امرأة، سألت عائشة أتقضي الحائض الصلاة فقالت أحرورية أنت لقد كنا…٢٦٣بهذا الحديث . قال أبو داود وزاد فيه فنؤمر بقضاء الصوم ولا نؤمر بقضاء…٢٦٤يتصدق بدينار أو نصف دينار٢٦٥حدثنا عبد السلام بن مطهر، حدثنا جعفر، - يعني ابن سليمان - عن علي بن…٢٦٦إذا وقع الرجل بأهله وهي حائض فليتصدق بنصف دينار٢٦٧أن رسول الله صلى الله عليه وسلم كان يباشر المرأة من نسائه وهي حائض…٢٦٨حدثنا مسلم بن إبراهيم، حدثنا شعبة، عن منصور، عن إبراهيم، عن الأسود،…٢٦٩حدثنا مسدد، حدثنا يحيى، عن جابر بن صبح، سمعت خلاسا الهجري، قال سمعت…٢٧٠أن عمة، له حدثته أنها، سألت عائشة قالت إحدانا تحيض وليس لها ولزوجها…٢٧١حدثنا سعيد بن عبد الجبار، حدثنا عبد العزيز، - يعني ابن محمد - عن أبي…٢٧٢أزواج النبي صلى الله عليه وسلم أن النبي صلى الله عليه وسلم كان إذا…٢٧٣حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن الشيباني، عن عبد الرحمن بن…٢٧٤لتنظر عدة الليالي والأيام التي كانت تحيضهن من الشهر قبل أن يصيبها…٢٧٥فإذا خلفت ذلك وحضرت الصلاة فلتغتسل٢٧٦فإذا خلفتهن وحضرت الصلاة فلتغتسل٢٧٧فلتترك الصلاة قدر ذلك ثم إذا حضرت الصلاة فلتغتسل ولتستثفر بثوب ثم تصلي٢٧٨تدع الصلاة وتغتسل فيما سوى ذلك وتستثفر بثوب وتصلي٢٧٩امكثي قدر ما كانت تحبسك حيضتك ثم اغتسلي٢٨٠إنما ذلك عرق فانظري إذا أتى قرؤك فلا تصلي فإذا مر قرؤك فتطهري ثم صلي…٢٨١تدع الصلاة أيام أقرائها٢٨٢إنما ذلك عرق وليست بالحيضة فإذا أقبلت الحيضة فدعي الصلاة وإذا أدبرت…٢٨٣فإذا أقبلت الحيضة فاتركي الصلاة فإذا ذهب قدرها فاغسلي الدم عنك وصلي٢٨٤فسد حيضها وأهريقت دما فأمرني رسول الله صلى الله عليه وسلم أن آمرها…٢٨٥إن هذه ليست بالحيضة ولكن هذا عرق فاغتسلي وصلي٢٨٦إذا كان دم الحيضة فإنه دم أسود يعرف فإذا كان ذلك فأمسكي عن الصلاة…٢٨٧أنعت لك الكرسف فإنه يذهب الدم٢٨٨إن هذه ليست بالحيضة ولكن هذا عرق فاغتسلي وصلي٢٨٩أخبرتني عمرة بنت عبد الرحمن، عن أم حبيبة، بهذا الحديث قالت عائشة رضي…٢٩٠حدثنا يزيد بن خالد بن عبد الله بن موهب الهمداني، حدثني الليث بن سعد،…٢٩١وعمرة بنت عبد الرحمن، عن عائشة، أن أم حبيبة، استحيضت سبع سنين فأمرها…٢٩٢اغتسلي لكل صلاة٢٩٣إنما هي - أو قال إنما هو - عرق أو قال عروق٢٩٤حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا شعبة، عن عبد الرحمن بن…٢٩٥أن سهلة بنت سهيل، استحيضت فأتت النبي صلى الله عليه وسلم فأمرها أن…٢٩٦سبحان الله إن هذا من الشيطان لتجلس في مركن فإذا رأت صفرة فوق الماء…٢٩٧تدع الصلاة أيام أقرائها ثم تغتسل وتصلي والوضوء عند كل صلاة٢٩٨ثم اغتسلي ثم توضئي لكل صلاة وصلي٢٩٩في المستحاضة تغتسل - تعني مرة واحدة - ثم توضأ إلى أيام أقرائها .٣٠٠توضئي لكل صلاة
