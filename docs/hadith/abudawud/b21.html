سنن أبي داود

كتاب الجنائز

١٥٣ حديثًا•صفحة ١ من ٢
٣٠٨٩إن المؤمن إذا أصابه السقم ثم أعفاه الله منه كان كفارة لما مضى من…٣٠٩٠إن العبد إذا سبقت له من الله منزلة لم يبلغها بعمله ابتلاه الله في…٣٠٩١إذا كان العبد يعمل عملا صالحا فشغله عنه مرض أو سفر كتب له كصالح ما…٣٠٩٢أبشري يا أم العلاء فإن مرض المسلم يذهب الله به خطاياه كما تذهب النار…٣٠٩٣أية آية يا عائشة٣٠٩٤قد كنت أنهاك عن حب يهود٣٠٩٥أن غلاما، من اليهود كان مرض فأتاه النبي صلى الله عليه وسلم يعوده فقعد…٣٠٩٦قال كان النبي صلى الله عليه وسلم يعودني ليس براكب بغل ولا برذون .٣٠٩٧من توضأ فأحسن الوضوء وعاد أخاه المسلم محتسبا بوعد من جهنم مسيرة سبعين…٣٠٩٨قال ما من رجل يعود مريضا ممسيا إلا خرج معه سبعون ألف ملك يستغفرون له…٣٠٩٩حدثنا عثمان بن أبي شيبة، حدثنا أبو معاوية، حدثنا الأعمش، عن الحكم، عن…٣١٠٠حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن منصور، عن الحكم، عن أبي جعفر…٣١٠١قالت لما أصيب سعد بن معاذ يوم الخندق رماه رجل في الأكحل فضرب عليه…٣١٠٢قال عادني رسول الله صلى الله عليه وسلم من وجع كان بعينى .٣١٠٣إذا سمعتم به بأرض فلا تقدموا عليه وإذا وقع بأرض وأنتم بها فلا تخرجوا…٣١٠٤اللهم اشف سعدا وأتمم له هجرته٣١٠٥أطعموا الجائع وعودوا المريض وفكوا العاني٣١٠٦من عاد مريضا لم يحضر أجله فقال عنده سبع مرار أسأل الله العظيم رب…٣١٠٧إذا جاء الرجل يعود مريضا فليقل اللهم اشف عبدك ينكأ لك عدوا أو يمشي لك…٣١٠٨لا يدعون أحدكم بالموت لضر نزل به ولكن ليقل اللهم أحيني ما كانت الحياة…٣١٠٩لا يتمنين أحدكم الموت٣١١٠موت الفجأة أخذة أسف٣١١١غلبنا عليك يا أبا الربيع٣١١٢حدثنا موسى بن إسماعيل، حدثنا إبراهيم بن سعد، أخبرنا ابن شهاب، أخبرني…٣١١٣لا يموت أحدكم إلا وهو يحسن الظن بالله٣١١٤إن الميت يبعث في ثيابه التي يموت فيها٣١١٥إذا حضرتم الميت فقولوا خيرا فإن الملائكة يؤمنون على ما تقولون٣١١٦من كان آخر كلامه لا إله إلا الله دخل الجنة٣١١٧لقنوا موتاكم قول لا إله إلا الله٣١١٨لا تدعوا على أنفسكم إلا بخير فإن الملائكة يؤمنون على ما تقولون٣١١٩إذا أصابت أحدكم مصيبة فليقل { إنا لله وإنا إليه راجعون } اللهم عندك…٣١٢٠أن النبي صلى الله عليه وسلم سجي في ثوب حبرة .٣١٢١اقرءوا { يس } على موتاكم٣١٢٢قالت لما قتل زيد بن حارثة وجعفر وعبد الله بن رواحة جلس رسول الله صلى…٣١٢٣ما أخرجك يا فاطمة من بيتك٣١٢٤اتقي الله واصبري٣١٢٥قل لله ما أخذ وما أعطى وكل شىء عنده إلى أجل٣١٢٦ولد لي الليلة غلام فسميته باسم أبي إبراهيم٣١٢٧حدثنا مسدد، حدثنا عبد الوارث، عن أيوب، عن حفصة، عن أم عطية، قالت إن…٣١٢٨حدثنا إبراهيم بن موسى، أخبرنا محمد بن ربيعة، عن محمد بن الحسن بن…٣١٢٩إن الميت ليعذب ببكاء أهله عليه٣١٣٠ليس منا من حلق ومن سلق ومن خرق٣١٣١حدثنا مسدد، حدثنا حميد بن الأسود، حدثنا الحجاج، - عامل لعمر بن عبد…٣١٣٢اصنعوا لآل جعفر طعاما فإنه قد أتاهم أمر شغلهم٣١٣٣قال رمي رجل بسهم في صدره أو في حلقه فمات فأدرج في ثيابه كما هو - قال…٣١٣٤وعيسى بن يونس، قالا حدثنا علي بن عاصم، عن عطاء بن السائب، عن سعيد بن…٣١٣٥أن ابن شهاب، أخبره أن أنس بن مالك حدثهم أن شهداء أحد لم يغسلوا ودفنوا…٣١٣٦لولا أن تجد صفية في نفسها لتركته حتى تأكله العافية حتى يحشر من بطونها٣١٣٧أن النبي صلى الله عليه وسلم مر بحمزة وقد مثل به ولم يصل على أحد من…٣١٣٨أيهما أكثر أخذا للقرآن٣١٣٩بهذا الحديث بمعناه قال يجمع بين الرجلين من قتلى أحد في ثوب واحد .٣١٤٠لا تبرز فخذك ولا تنظرن إلى فخذ حى ولا ميت٣١٤١حدثنا النفيلي، حدثنا محمد بن سلمة، عن محمد بن إسحاق، حدثني يحيى بن…٣١٤٢اغسلنها ثلاثا أو خمسا أو أكثر من ذلك - إن رأيتن ذلك - بماء وسدر…٣١٤٣وأبو كامل - بمعنى الإسناد - أن يزيد بن زريع، حدثهم حدثنا أيوب، عن…٣١٤٤حدثنا محمد بن المثنى، حدثنا عبد الأعلى، حدثنا هشام، عن حفصة بنت…٣١٤٥ابدأن بميامنها ومواضع الوضوء منها٣١٤٦أو سبعا أو أكثر من ذلك إن رأيتنه٣١٤٧يغسل بالسدر مرتين والثالثة بالماء والكافور .٣١٤٨إذا كفن أحدكم أخاه فليحسن كفنه٣١٤٩قالت أدرج رسول الله صلى الله عليه وسلم في ثوب حبرة ثم أخر عنه .٣١٥٠إذا توفي أحدكم فوجد شيئا فليكفن في ثوب حبرة٣١٥١أخبرتني عائشة، قالت كفن رسول الله صلى الله عليه وسلم في ثلاثة أثواب…٣١٥٢مثله زاد من كرسف . قال فذكر لعائشة قولهم في ثوبين وبرد حبرة فقالت قد…٣١٥٣وعثمان بن أبي شيبة، قالا حدثنا ابن إدريس، عن يزيد، - يعني ابن أبي…٣١٥٤لا تغالوا في الكفن فإنه يسلبه سلبا سريعا٣١٥٥غطوا بها رأسه واجعلوا على رجليه شيئا من الإذخر٣١٥٦خير الكفن الحلة وخير الأضحية الكبش الأقرن٣١٥٧حدثنا أحمد بن حنبل، حدثنا يعقوب بن إبراهيم، حدثنا أبي، عن ابن إسحاق،…٣١٥٨أطيب طيبكم المسك٣١٥٩إني لا أرى طلحة إلا قد حدث فيه الموت فآذنوني به وعجلوا فإنه لا ينبغي…٣١٦٠أنها حدثته أن النبي صلى الله عليه وسلم كان يغتسل من أربع من الجنابة…٣١٦١من غسل الميت فليغتسل ومن حمله فليتوضأ٣١٦٢حدثنا حامد بن يحيى، عن سفيان، عن سهيل بن أبي صالح، عن أبيه، عن إسحاق،…٣١٦٣قالت رأيت رسول الله صلى الله عليه وسلم يقبل عثمان بن مظعون وهو ميت…٣١٦٤- قال رأى ناس نارا في المقبرة فأتوها فإذا رسول الله صلى الله عليه…٣١٦٥حدثنا محمد بن كثير، أخبرنا سفيان، عن الأسود بن قيس، عن نبيح، عن جابر…٣١٦٦ما من مسلم يموت فيصلي عليه ثلاثة صفوف من المسلمين إلا أوجب٣١٦٧حدثنا سليمان بن حرب، حدثنا حماد، عن أيوب، عن حفصة، عن أم عطية، قالت…٣١٦٨يرويه قال من تبع جنازة فصلى عليها فله قيراط ومن تبعها حتى يفرغ منها…٣١٦٩من خرج مع جنازة من بيتها وصلى عليها٣١٧٠ما من مسلم يموت فيقوم على جنازته أربعون رجلا لا يشركون بالله شيئا إلا…٣١٧١لا تتبع الجنازة بصوت ولا نار٣١٧٢إذا رأيتم الجنازة فقوموا لها حتى تخلفكم أو توضع٣١٧٣إذا تبعتم الجنازة فلا تجلسوا حتى توضع٣١٧٤إن الموت فزع فإذا رأيتم جنازة فقوموا٣١٧٥أن النبي صلى الله عليه وسلم قام في الجنائز ثم قعد بعد .٣١٧٦قال كان رسول الله صلى الله عليه وسلم يقوم في الجنازة حتى توضع في…٣١٧٧إن الملائكة كانت تمشي فلم أكن لأركب وهم يمشون فلما ذهبوا ركبت٣١٧٨سمع جابر بن سمرة، قال صلى النبي صلى الله عليه وسلم على ابن الدحداح…٣١٧٩قال رأيت النبي صلى الله عليه وسلم وأبا بكر وعمر يمشون أمام الجنازة .٣١٨٠الراكب يسير خلف الجنازة والماشي يمشي خلفها وأمامها وعن يمينها وعن…٣١٨١أسرعوا بالجنازة فإن تك صالحة فخير تقدمونها إليه وإن تك سوى ذلك فشر…٣١٨٢حدثنا مسلم بن إبراهيم، حدثنا شعبة، عن عيينة بن عبد الرحمن، عن أبيه،…٣١٨٣بهذا الحديث قالا في جنازة عبد الرحمن بن سمرة وقال فحمل عليهم بغلته…٣١٨٤ما دون الخبب إن يكن خيرا تعجل إليه وإن يكن غير ذلك فبعدا لأهل النار…٣١٨٥حدثنا ابن نفيل، حدثنا زهير، حدثنا سماك، حدثني جابر بن سمرة، قال مرض…٣١٨٦حدثنا أبو كامل، حدثنا أبو عوانة، عن أبي بشر، حدثني نفر، من أهل البصرة…٣١٨٧قالت مات إبراهيم ابن النبي صلى الله عليه وسلم وهو ابن ثمانية عشر شهرا…٣١٨٨حدثنا هناد بن السري، حدثنا محمد بن عبيد، عن وائل بن داود، قال سمعت…٣١٨٩ومحمد بن عبد الله بن عباد، عن عباد بن عبد الله بن الزبير، عن عائشة،…٣١٩٠قالت والله لقد صلى رسول الله صلى الله عليه وسلم على ابنى بيضاء في…٣١٩١من صلى على جنازة في المسجد فلا شىء عليه٣١٩٢أنه سمع عقبة بن عامر، قال ثلاث ساعات كان رسول الله صلى الله عليه وسلم…٣١٩٣مولى الحارث بن نوفل أنه شهد جنازة أم كلثوم وابنها فجعل الغلام مما يلي…٣١٩٤إني لم أمسك عنه منذ اليوم إلا لتوفي بنذرك٣١٩٥قال صليت وراء النبي صلى الله عليه وسلم على امرأة ماتت في نفاسها فقام…٣١٩٦أن رسول الله صلى الله عليه وسلم مر بقبر رطب فصفوا عليه وكبر عليه…٣١٩٧حدثنا أبو الوليد الطيالسي، حدثنا شعبة، ح وحدثنا محمد بن المثنى، حدثنا…٣١٩٨قال صليت مع ابن عباس على جنازة فقرأ بفاتحة الكتاب فقال إنها من السنة .٣١٩٩إذا صليتم على الميت فأخلصوا له الدعاء٣٢٠٠اللهم أنت ربها وأنت خلقتها وأنت هديتها للإسلام وأنت قبضت روحها وأنت…٣٢٠١اللهم اغفر لحينا وميتنا وصغيرنا وكبيرنا وذكرنا وأنثانا وشاهدنا…٣٢٠٢اللهم إن فلان بن فلان في ذمتك فقه فتنة القبر٣٢٠٣ألا آذنتموني به٣٢٠٤أن رسول الله صلى الله عليه وسلم نعى للناس النجاشي في اليوم الذي مات…٣٢٠٥حدثنا عباد بن موسى، حدثنا إسماعيل، - يعني ابن جعفر - عن إسرائيل، عن…٣٢٠٦أتعلم بها قبر أخي وأدفن إليه من مات من أهلي٣٢٠٧كسر عظم الميت ككسره حيا٣٢٠٨اللحد لنا والشق لغيرنا٣٢٠٩حدثنا أحمد بن يونس، حدثنا زهير، حدثنا إسماعيل بن أبي خالد، عن عامر،…٣٢١٠أن عبد الرحمن بن عوف، نزل في قبر النبي صلى الله عليه وسلم قال كأني…٣٢١١قال أوصى الحارث أن يصلي، عليه عبد الله بن يزيد فصلى عليه ثم أدخله…٣٢١٢حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن الأعمش، عن المنهال بن عمرو،…٣٢١٣بسم الله وعلى سنة رسول الله صلى الله عليه وسلم٣٢١٤اذهب فوار أباك ثم لا تحدثن شيئا حتى تأتيني٣٢١٥احفروا وأوسعوا واجعلوا الرجلين والثلاثة في القبر٣٢١٦بإسناده ومعناه زاد فيه " وأعمقوا " .٣٢١٧بهذا الحديث .٣٢١٨قال بعثني علي قال لي أبعثك على ما بعثني عليه رسول الله صلى الله عليه…٣٢١٩أن أبا علي الهمداني، حدثه قال كنا مع فضالة بن عبيد برودس من أرض الروم…٣٢٢٠حدثنا أحمد بن صالح، حدثنا ابن أبي فديك، أخبرني عمرو بن عثمان بن هانئ،…٣٢٢١استغفروا لأخيكم وسلوا له التثبيت فإنه الآن يسأل٣٢٢٢لا عقر في الإسلام٣٢٢٣أن رسول الله صلى الله عليه وسلم خرج يوما فصلى على أهل أحد صلاته على…٣٢٢٤بهذا الحديث قال إن النبي صلى الله عليه وسلم صلى على قتلى أحد بعد ثمان…٣٢٢٥أنه سمع جابرا، يقول سمعت رسول الله صلى الله عليه وسلم نهى أن يقعد على…٣٢٢٦وعثمان بن أبي شيبة، قالا حدثنا حفص بن غياث، عن ابن جريج، عن سليمان بن…٣٢٢٧قاتل الله اليهود اتخذوا قبور أنبيائهم مساجد٣٢٢٨لأن يجلس أحدكم على جمرة فتحرق ثيابه حتى تخلص إلى جلده خير له من أن…٣٢٢٩لا تجلسوا على القبور ولا تصلوا إليها٣٢٣٠حدثنا سهل بن بكار، حدثنا الأسود بن شيبان، عن خالد بن سمير السدوسي، عن…٣٢٣١إن العبد إذا وضع في قبره وتولى عنه أصحابه إنه ليسمع قرع نعالهم٣٢٣٢قال دفن مع أبي رجل فكان في نفسي من ذلك حاجة فأخرجته بعد ستة أشهر فما…٣٢٣٣قال مروا على رسول الله صلى الله عليه وسلم بجنازة فأثنوا عليها خيرا…٣٢٣٤استأذنت ربي تعالى على أن أستغفر لها فلم يؤذن لي فاستأذنت أن أزور…٣٢٣٥نهيتكم عن زيارة القبور فزوروها فإن في زيارتها تذكرة٣٢٣٦حدثنا محمد بن كثير، أخبرنا شعبة، عن محمد بن جحادة، قال سمعت أبا صالح،…٣٢٣٧السلام عليكم دار قوم مؤمنين وإنا إن شاء الله بكم لاحقون٣٢٣٨كفنوه في ثوبيه واغسلوه بماء وسدر ولا تخمروا رأسه فإن الله يبعثه يوم…
