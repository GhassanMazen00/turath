سنن أبي داود

كتاب الأدب

٥٠٢ حديثًا•صفحة ٣ من ٤
٥٠٧٣من قال حين يصبح اللهم ما أصبح بي من نعمة فمنك وحدك لا شريك لك فلك…٥٠٧٤اللهم إني أسألك العافية في الدنيا والآخرة اللهم إني أسألك العفو…٥٠٧٥قولي حين تصبحين سبحان الله وبحمده لا قوة إلا بالله ما شاء الله كان…٥٠٧٦من قال حين يصبح { فسبحان الله حين تمسون وحين تصبحون * وله الحمد في…٥٠٧٧من قال إذا أصبح لا إله إلا الله وحده لا شريك له له الملك وله الحمد…٥٠٧٨من قال حين يصبح اللهم إني أصبحت أشهدك وأشهد حملة عرشك وملائكتك وجميع…٥٠٧٩إذا انصرفت من صلاة المغرب فقل اللهم أجرني من النار . سبع مرات فإنك…٥٠٨٠ومؤمل بن الفضل الحراني، وعلي بن سهل الرملي، ومحمد بن المصفى الحمصي،…٥٠٨١حدثنا يزيد بن محمد الدمشقي، حدثنا عبد الرزاق بن مسلم الدمشقي، - وكان…٥٠٨٢حدثنا محمد بن المصفى، حدثنا ابن أبي فديك، قال أخبرني ابن أبي ذئب، عن…٥٠٨٣حدثنا محمد بن عوف، حدثنا محمد بن إسماعيل، قال حدثني أبي، - قال ابن…٥٠٨٤إذا أصبح أحدكم فليقل أصبحنا وأصبح الملك لله رب العالمين اللهم إني…٥٠٨٥سبحان الله وبحمده٥٠٨٦سمع سامع بحمد الله ونعمته وحسن بلائه علينا اللهم صاحبنا فأفضل علينا٥٠٨٧قال كان أبو ذر يقول من قال حين يصبح اللهم ما حلفت من حلف أو قلت من…٥٠٨٨من قال بسم الله الذي لا يضر مع اسمه شىء في الأرض ولا في السماء وهو…٥٠٨٩حدثنا نصر بن عاصم الأنطاكي، حدثنا أنس بن عياض، قال حدثني أبو مودود،…٥٠٩٠دعوات المكروب اللهم رحمتك أرجو فلا تكلني إلى نفسي طرفة عين وأصلح لي…٥٠٩١من قال حين يصبح سبحان الله العظيم وبحمده مائة مرة وإذا أمسى كذلك لم…٥٠٩٢هلال خير ورشد هلال خير ورشد هلال خير ورشد آمنت بالذي خلقك٥٠٩٣أن زيد بن حباب، أخبرهم عن أبي هلال، عن قتادة، أن رسول الله صلى الله…٥٠٩٤اللهم إني أعوذ بك أن أضل أو أضل أو أزل أو أزل أو أظلم أو أظلم أو أجهل…٥٠٩٥إذا خرج الرجل من بيته فقال بسم الله توكلت على الله لا حول ولا قوة إلا…٥٠٩٦إذا ولج الرجل في بيته فليقل اللهم إني أسألك خير المولج وخير المخرج…٥٠٩٧الريح من روح الله٥٠٩٨يا عائشة ما يؤمنني أن يكون فيه عذاب قد عذب قوم بالريح وقد رأى قوم…٥٠٩٩اللهم إني أعوذ بك من شرها٥١٠٠لأنه حديث عهد بربه٥١٠١لا تسبوا الديك فإنه يوقظ للصلاة٥١٠٢إذا سمعتم صياح الديكة فسلوا الله تعالى من فضله فإنها رأت ملكا وإذا…٥١٠٣إذا سمعتم نباح الكلاب ونهيق الحمر بالليل فتعوذوا بالله فإنهن يرين ما…٥١٠٤أقلوا الخروج بعد هدأة الرجل فإن لله تعالى دواب يبثهن في الأرض٥١٠٥قال رأيت رسول الله صلى الله عليه وسلم أذن في أذن الحسن بن علي - حين…٥١٠٦رضى الله عنها قالت كان رسول الله صلى الله عليه وسلم يؤتى بالصبيان…٥١٠٧هل رئي - أو كلمة غيرها - فيكم المغربون٥١٠٨من استعاذ بالله فأعيذوه ومن سألكم بوجه الله فأعطوه٥١٠٩من استعاذكم بالله فأعيذوه ومن سألكم بالله فأعطوه٥١١٠حدثنا عباس بن عبد العظيم، حدثنا النضر بن محمد، حدثنا عكرمة، - يعني…٥١١١حدثنا أحمد بن يونس، حدثنا زهير، حدثنا سهيل، عن أبيه، عن أبي هريرة،…٥١١٢الله أكبر الله أكبر الله أكبر الحمد لله الذي رد كيده إلى الوسوسة٥١١٣من ادعى إلى غير أبيه وهو يعلم أنه غير أبيه فالجنة عليه حرام٥١١٤من تولى قوما بغير إذن مواليه فعليه لعنة الله والملائكة والناس أجمعين…٥١١٥من ادعى إلى غير أبيه أو انتمى إلى غير مواليه فعليه لعنة الله…٥١١٦إن الله عز وجل قد أذهب عنكم عبية الجاهلية وفخرها بالآباء مؤمن تقي…٥١١٧قال من نصر قومه على غير الحق فهو كالبعير الذي ردي فهو ينزع بذنبه .٥١١٨قال انتهيت إلى النبي صلى الله عليه وسلم وهو في قبة من أدم فذكر نحوه .٥١١٩أن تعين قومك على الظلم٥١٢٠خيركم المدافع عن عشيرته ما لم يأثم٥١٢١ليس منا من دعا إلى عصبية وليس منا من قاتل على عصبية وليس منا من مات…٥١٢٢ابن أخت القوم منهم٥١٢٣فهلا قلت خذها مني وأنا الغلام الأنصاري٥١٢٤إذا أحب الرجل أخاه فليخبره أنه يحبه٥١٢٥أن رجلا، كان عند النبي صلى الله عليه وسلم فمر به رجل فقال يا رسول…٥١٢٦أنت يا أبا ذر مع من أحببت٥١٢٧المرء مع من أحب٥١٢٨المستشار مؤتمن٥١٢٩لا أجد ما أحملك عليه ولكن ائت فلانا فلعله أن يحملك٥١٣٠حبك الشىء يعمي ويصم٥١٣١اشفعوا إلى لتؤجروا وليقض الله على لسان نبيه ما شاء٥١٣٢وأحمد بن عمرو بن السرح، قالا حدثنا سفيان بن عيينة، عن عمرو بن دينار،…٥١٣٣حدثنا أبو معمر، حدثنا سفيان، عن بريد، عن أبي بردة، عن أبي موسى، عن…٥١٣٤حدثنا أحمد بن حنبل، حدثنا هشيم، عن منصور، عن ابن سيرين، - قال أحمد…٥١٣٥- يعني ابن الحضرمي - أنه كتب إلى النبي صلى الله عليه وسلم فبدأ باسمه .٥١٣٦من محمد رسول الله إلى هرقل عظيم الروم سلام على من اتبع الهدى٥١٣٧لا يجزي ولد والده إلا أن يجده مملوكا فيشتريه فيعتقه٥١٣٨قال كانت تحتي امرأة وكنت أحبها وكان عمر يكرهها فقال لي طلقها فأبيت…٥١٣٩أمك ثم أمك ثم أمك ثم أباك ثم الأقرب فالأقرب٥١٤٠أمك وأباك وأختك وأخاك ومولاك الذي يلي ذاك حق واجب ورحم موصولة٥١٤١إن من أكبر الكبائر أن يلعن الرجل والديه٥١٤٢نعم الصلاة عليهما والاستغفار لهما وإنفاذ عهدهما من بعدهما وصلة الرحم…٥١٤٣إن أبر البر صلة المرء أهل ود أبيه بعد أن يولي٥١٤٤أن أبا الطفيل، أخبره قال رأيت النبي صلى الله عليه وسلم يقسم لحما…٥١٤٥أن عمر بن السائب، حدثه أنه، بلغه أن رسول الله صلى الله عليه وسلم كان…٥١٤٦من كانت له أنثى فلم يئدها ولم يهنها ولم يؤثر ولده عليها - قال يعني…٥١٤٧من عال ثلاث بنات فأدبهن وزوجهن وأحسن إليهن فله الجنة٥١٤٨ثلاث أخوات أو ثلاث بنات أو بنتان أو أختان٥١٤٩أنا وامرأة سفعاء الخدين كهاتين يوم القيامة٥١٥٠أنا وكافل اليتيم كهاتين في الجنة٥١٥١ما زال جبريل يوصيني بالجار حتى قلت ليورثنه٥١٥٢ما زال جبريل يوصيني بالجار حتى ظننت أنه سيورثه٥١٥٣قال جاء رجل إلى النبي صلى الله عليه وسلم يشكو جاره فقال " اذهب فاصبر…٥١٥٤من كان يؤمن بالله واليوم الآخر فليكرم ضيفه ومن كان يؤمن بالله واليوم…٥١٥٥وسعيد بن منصور، أن الحارث بن عبيد، حدثهم عن أبي عمران الجوني، عن…٥١٥٦الصلاة الصلاة اتقوا الله فيما ملكت أيمانكم٥١٥٧يا أبا ذر إنك امرؤ فيك جاهلية٥١٥٨إخوانكم جعلهم الله تحت أيديكم فمن كان أخوه تحت يديه فليطعمه مما يأكل…٥١٥٩حدثنا محمد بن العلاء، قال حدثنا أبو معاوية، ح وحدثنا ابن المثنى، قال…٥١٦٠بإسناده ومعناه نحوه قال كنت أضرب غلاما لي أسود بالسوط ولم يذكر أمر…٥١٦١من لاءمكم من مملوكيكم فأطعموه مما تأكلون واكسوه مما تكتسون ومن لم…٥١٦٢حسن الملكة نماء وسوء الخلق شؤم٥١٦٣حسن الملكة نماء وسوء الخلق شؤم٥١٦٤اعفوا عنه في كل يوم سبعين مرة٥١٦٥من قذف مملوكه وهو بريء مما قال جلد له يوم القيامة حدا٥١٦٦حدثنا مسدد، حدثنا فضيل بن عياض، عن حصين، عن هلال بن يساف، قال كنا…٥١٦٧حدثنا مسدد، حدثنا يحيى، عن سفيان، قال حدثني سلمة بن كهيل، قال حدثني…٥١٦٨من لطم مملوكه أو ضربه فكفارته أن يعتقه٥١٦٩إن العبد إذا نصح لسيده وأحسن عبادة الله فله أجره مرتين٥١٧٠من خبب زوجة امرئ أو مملوكه فليس منا٥١٧١أن رجلا، اطلع من بعض حجر النبي صلى الله عليه وسلم فقام إليه رسول الله…٥١٧٢من اطلع في دار قوم بغير إذنهم ففقأوا عينه فقد هدرت عينه٥١٧٣إذا دخل البصر فلا إذن٥١٧٤هكذا عنك أو هكذا فإنما الاستئذان من النظر٥١٧٥حدثنا هارون بن عبد الله، حدثنا أبو داود الحفري، عن سفيان، عن الأعمش،…٥١٧٦ارجع فقل السلام عليكم٥١٧٧اخرج إلى هذا فعلمه الاستئذان فقل له قل السلام عليكم أأدخل٥١٧٨حدثنا هناد بن السري، عن أبي الأحوص، عن منصور، عن ربعي بن حراش، قال…٥١٧٩حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا شعبة، عن منصور، عن ربعي، عن…٥١٨٠إذا استأذن أحدكم ثلاثا فلم يؤذن له فليرجع٥١٨١يستأذن أحدكم ثلاثا فإن أذن له وإلا فليرجع٥١٨٢أن أبا موسى، استأذن على عمر بهذه القصة . قال فيه فانطلق بأبي سعيد…٥١٨٣حدثنا زيد بن أخزم، حدثنا عبد القاهر بن شعيب، حدثنا هشام، عن حميد بن…٥١٨٤واحد، من علمائهم في هذا فقال عمر لأبي موسى أما إني لم أتهمك ولكن خشيت…٥١٨٥السلام عليكم ورحمة الله٥١٨٦السلام عليكم السلام عليكم٥١٨٧حدثنا مسدد، حدثنا بشر، عن شعبة، عن محمد بن المنكدر، عن جابر، أنه ذهب…٥١٨٨قال خرجت مع رسول الله صلى الله عليه وسلم حتى دخلت حائطا فقال لي "…٥١٨٩رسول الرجل إلى الرجل إذنه٥١٩٠إذا دعي أحدكم إلى طعام فجاء مع الرسول فإن ذلك له إذن٥١٩١وابن، عبدة - وهذا حديثه - قالا أخبرنا سفيان، عن عبيد الله بن أبي…٥١٩٢أن نفرا، من أهل العراق قالوا يا ابن عباس كيف ترى في هذه الآية التي…٥١٩٣والذي نفسي بيده لا تدخلوا الجنة حتى تؤمنوا ولا تؤمنوا حتى تحابوا أفلا…٥١٩٤تطعم الطعام وتقرأ السلام على من عرفت ومن لم تعرف٥١٩٥قال جاء رجل إلى النبي صلى الله عليه وسلم فقال السلام عليكم . فرد عليه…٥١٩٦حدثنا إسحاق بن سويد الرملي، حدثنا ابن أبي مريم، قال أظن أني سمعت نافع…٥١٩٧إن أولى الناس بالله من بدأهم بالسلام٥١٩٨يسلم الصغير على الكبير والمار على القاعد والقليل على الكثير٥١٩٩يسلم الراكب على الماشي٥٢٠٠حدثنا أحمد بن سعيد الهمداني، حدثنا ابن وهب، قال أخبرني معاوية بن…٥٢٠١أنه أتى النبي صلى الله عليه وسلم وهو في مشربة له فقال السلام عليك يا…٥٢٠٢قال قال أنس أتى رسول الله صلى الله عليه وسلم على غلمان يلعبون فسلم…٥٢٠٣حدثنا ابن المثنى، حدثنا خالد، - يعني ابن الحارث - حدثنا حميد، قال قال…٥٢٠٤سمعه من، شهر بن حوشب يقول أخبرته أسماء بنت يزيد، مر علينا النبي صلى…٥٢٠٥لا تبدءوهم بالسلام وإذا لقيتموهم في الطريق فاضطروهم إلى أضيق الطريق٥٢٠٦إن اليهود إذا سلم عليكم أحدهم فإنما يقول السام عليكم فقولوا وعليكم٥٢٠٧أن أصحاب النبي، صلى الله عليه وسلم قالوا للنبي صلى الله عليه وسلم إن…٥٢٠٨إذا انتهى أحدكم إلى المجلس فليسلم فإذا أراد أن يقوم فليسلم فليست…٥٢٠٩لا تقل عليك السلام فإن عليك السلام تحية الموتى٥٢١٠يجزئ عن الجماعة، إذا مروا أن يسلم، أحدهم ويجزئ عن الجلوس أن يرد أحدهم٥٢١١إذا التقى المسلمان فتصافحا وحمدا الله عز وجل واستغفراه غفر لهما٥٢١٢ما من مسلمين يلتقيان فيتصافحان إلا غفر لهما قبل أن يفترقا٥٢١٣قد جاءكم أهل اليمن وهم أول من جاء
