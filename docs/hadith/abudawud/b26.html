سنن أبي داود

كتاب العلم

٢٨ حديثًا
٣٦٤١من سلك طريقا يطلب فيه علما سلك الله به طريقا من طرق الجنة وإن…٣٦٤٢حدثنا محمد بن الوزير الدمشقي، حدثنا الوليد، قال لقيت شبيب بن شيبة…٣٦٤٣ما من رجل يسلك طريقا يطلب فيه علما إلا سهل الله له به طريق الجنة ومن…٣٦٤٤حدثنا أحمد بن محمد بن ثابت المروزي، حدثنا عبد الرزاق، أخبرنا معمر، عن…٣٦٤٥إني والله ما آمن يهود على كتابي٣٦٤٦اكتب فوالذي نفسي بيده ما يخرج منه إلا حق٣٦٤٧حدثنا نصر بن علي، أخبرنا أبو أحمد، حدثنا كثير بن زيد، عن المطلب بن…٣٦٤٨حدثنا أحمد بن يونس، حدثنا أبو شهاب، عن الحذاء، عن أبي المتوكل الناجي،…٣٦٤٩قال لما فتحت مكة قام النبي صلى الله عليه وسلم فذكر الخطبة خطبة النبي…٣٦٥٠قال قلت لأبي عمرو ما يكتبوه قال الخطبة التي سمعها يومئذ منه .٣٦٥١من كذب على متعمدا فليتبوأ مقعده من النار٣٦٥٢من قال في كتاب الله عز وجل برأيه فأصاب فقد أخطأ٣٦٥٣خدم النبي صلى الله عليه وسلم أن النبي صلى الله عليه وسلم كان إذا حدث…٣٦٥٤قال جلس أبو هريرة إلى جنب حجرة عائشة - رضى الله عنها - وهي تصلي فجعل…٣٦٥٥أن عروة بن الزبير، حدثه أن عائشة زوج النبي صلى الله عليه وسلم قالت…٣٦٥٦حدثنا إبراهيم بن موسى الرازي، حدثنا عيسى، عن الأوزاعي، عن عبد الله بن…٣٦٥٧حدثنا الحسن بن علي، حدثنا أبو عبد الرحمن المقرئ، حدثنا سعيد، - يعني…٣٦٥٨من سئل عن علم فكتمه ألجمه الله بلجام من نار يوم القيامة٣٦٥٩تسمعون ويسمع منكم ويسمع ممن سمع منكم٣٦٦٠نضر الله امرأ سمع منا حديثا فحفظه حتى يبلغه فرب حامل فقه إلى من هو…٣٦٦١والله لأن يهدى بهداك رجل واحد خير لك من حمر النعم٣٦٦٢حدثوا عن بني إسرائيل ولا حرج٣٦٦٣حدثنا محمد بن المثنى، حدثنا معاذ، حدثني أبي، عن قتادة، عن أبي حسان،…٣٦٦٤من تعلم علما مما يبتغى به وجه الله عز وجل لا يتعلمه إلا ليصيب به عرضا…٣٦٦٥لا يقص إلا أمير أو مأمور أو مختال٣٦٦٦حدثنا مسدد، حدثنا جعفر بن سليمان، عن المعلى بن زياد، عن العلاء بن…٣٦٦٧لأن أقعد مع قوم يذكرون الله تعالى من صلاة الغداة حتى تطلع الشمس أحب…٣٦٦٨اقرأ على سورة النساء
