سنن أبي داود

كتاب اللباس

١٣٩ حديثًا
٤٠٢٠اللهم لك الحمد أنت كسوتنيه أسألك من خيره وخير ما صنع له وأعوذ بك من…٤٠٢١بإسناده نحوه .٤٠٢٢حدثنا مسلم بن إبراهيم، حدثنا محمد بن دينار، عن الجريري، بإسناده…٤٠٢٣من أكل طعاما ثم قال الحمد لله الذي أطعمني هذا الطعام ورزقنيه من غير…٤٠٢٤من ترون أحق بهذه٤٠٢٥قالت كان أحب الثياب إلى رسول الله صلى الله عليه وسلم القميص .٤٠٢٦قالت لم يكن ثوب أحب إلى رسول الله صلى الله عليه وسلم من قميص .٤٠٢٧قالت كانت يد كم قميص رسول الله صلى الله عليه وسلم إلى الرصغ .٤٠٢٨ويزيد بن خالد بن موهب، - المعنى - أن الليث، - يعني ابن سعد - حدثهم عن…٤٠٢٩من لبس ثوب شهرة ألبسه الله يوم القيامة ثوبا مثله٤٠٣٠قال ثوب مذلة .٤٠٣١من تشبه بقوم فهو منهم٤٠٣٢وحسين بن علي، قالا حدثنا ابن أبي زائدة، عن أبيه، عن مصعب بن شيبة، عن…٤٠٣٣حدثنا عمرو بن عون، حدثنا أبو عوانة، عن قتادة، عن أبي بردة، قال قال لي…٤٠٣٤أن ملك، ذي يزن أهدى إلى رسول الله صلى الله عليه وسلم حلة أخذها بثلاثة…٤٠٣٥أن رسول الله صلى الله عليه وسلم اشترى حلة ببضعة وعشرين قلوصا فأهداها…٤٠٣٦حدثنا موسى بن إسماعيل، حدثنا حماد، ح وحدثنا موسى، حدثنا سليمان، -…٤٠٣٧قال لما خرجت الحرورية أتيت عليا - رضى الله عنه - فقال ائت هؤلاء القوم…٤٠٣٨سعد قال رأيت رجلا ببخارى على بغلة بيضاء عليه عمامة خز سوداء فقال…٤٠٣٩ليكونن من أمتي أقوام يستحلون الخز والحرير٤٠٤٠إنما يلبس هذه من لا خلاق له في الآخرة٤٠٤١تبيعها وتصيب بها حاجتك٤٠٤٢حدثنا موسى بن إسماعيل، حدثنا حماد، حدثنا عاصم الأحول، عن أبي عثمان…٤٠٤٣إني لم أرسل بها إليك لتلبسها٤٠٤٤حدثنا القعنبي، عن مالك، عن نافع، عن إبراهيم بن عبد الله بن حنين، عن…٤٠٤٥حدثنا أحمد بن محمد، - يعني المروزي - حدثنا عبد الرزاق، أخبرنا معمر،…٤٠٤٦بهذا زاد ولا أقول نهاكم .٤٠٤٧إني لم أعطكها لتلبسها٤٠٤٨لا أركب الأرجوان ولا ألبس المعصفر ولا ألبس القميص المكفف بالحرير٤٠٤٩حدثنا يزيد بن خالد بن عبد الله بن موهب الهمداني، أخبرنا المفضل، -…٤٠٥٠حدثنا يحيى بن حبيب، حدثنا روح، حدثنا هشام، عن محمد، عن عبيدة، عن علي،…٤٠٥١ومسلم بن إبراهيم، قالا حدثنا شعبة، عن أبي إسحاق، عن هبيرة، عن علي، -…٤٠٥٢اذهبوا بخميصتي هذه إلى أبي جهم فإنها ألهتني آنفا في صلاتي وأتوني…٤٠٥٣نحوه والأول أشبع .٤٠٥٤مولى أسماء بنت أبي بكر قال رأيت ابن عمر في السوق اشترى ثوبا شاميا…٤٠٥٥حدثنا ابن نفيل، حدثنا زهير، حدثنا خصيف، عن عكرمة، عن ابن عباس، قال…٤٠٥٦قال رخص رسول الله صلى الله عليه وسلم لعبد الرحمن بن عوف وللزبير بن…٤٠٥٧إن هذين حرام على ذكور أمتي٤٠٥٨وكثير بن عبيد الحمصيان، قالا حدثنا بقية، عن الزبيدي، عن الزهري، عن…٤٠٥٩ونتركه، على الجواري . قال مسعر فسألت عمرو بن دينار عنه فلم يعرفه .٤٠٦٠حدثنا هدبة بن خالد الأزدي، حدثنا همام، عن قتادة، قال قلنا لأنس - يعني…٤٠٦١البسوا من ثيابكم البياض فإنها خير ثيابكم وكفنوا فيها موتاكم وإن خير…٤٠٦٢أما كان يجد هذا ما يسكن به شعره٤٠٦٣قال أتيت النبي صلى الله عليه وسلم في ثوب دون فقال " ألك مال " . قال…٤٠٦٤- يعني ابن أسلم - أن ابن عمر، كان يصبغ لحيته بالصفرة حتى تمتلئ ثيابه…٤٠٦٥قال انطلقت مع أبي نحو النبي صلى الله عليه وسلم فرأيت عليه بردين…٤٠٦٦ما هذه الريطة عليك٤٠٦٧قال قال هشام - يعني ابن الغاز - المضرجة التي ليست بمشبعة ولا الموردة .٤٠٦٨حدثنا محمد بن عثمان الدمشقي، حدثنا إسماعيل بن عياش، عن شرحبيل بن…٤٠٦٩قال مر على النبي صلى الله عليه وسلم رجل عليه ثوبان أحمران فسلم عليه…٤٠٧٠ألا أرى هذه الحمرة قد علتكم٤٠٧١حدثنا ابن عوف الطائي، حدثنا محمد بن إسماعيل، حدثني أبي، - قال ابن عوف…٤٠٧٢قال كان رسول الله صلى الله عليه وسلم له شعر يبلغ شحمة أذنيه ورأيته في…٤٠٧٣قال رأيت رسول الله صلى الله عليه وسلم بمنى يخطب على بغلة وعليه برد…٤٠٧٤- رضى الله عنها - قالت صنعت لرسول الله صلى الله عليه وسلم بردة سوداء…٤٠٧٥- يعني ابن سليم - قال أتيت النبي صلى الله عليه وسلم وهو محتب بشملة…٤٠٧٦ومسلم بن إبراهيم، وموسى بن إسماعيل، قالوا حدثنا حماد، عن أبي الزبير،…٤٠٧٧قال رأيت النبي صلى الله عليه وسلم على المنبر وعليه عمامة سوداء قد…٤٠٧٨فرق ما بيننا وبين المشركين العمائم على القلانس٤٠٧٩يقول عممني رسول الله صلى الله عليه وسلم فسدلها بين يدى ومن خلفي .٤٠٨٠حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن الأعمش، عن أبي صالح، عن أبي…٤٠٨١حدثنا موسى بن إسماعيل، حدثنا حماد، عن أبي الزبير، عن جابر، قال نهى…٤٠٨٢وأحمد بن يونس، قالا حدثنا زهير، حدثنا عروة بن عبد الله، - قال ابن…٤٠٨٣حدثنا محمد بن داود بن سفيان، حدثنا عبد الرزاق، أخبرنا معمر، قال قال…٤٠٨٤لا تقل عليك السلام . فإن عليك السلام تحية الميت قل السلام عليك٤٠٨٥من جر ثوبه خيلاء لم ينظر الله إليه يوم القيامة٤٠٨٦قال بينما رجل يصلي مسبلا إزاره فقال له رسول الله صلى الله عليه وسلم "…٤٠٨٧ثلاثة لا يكلمهم الله ولا ينظر إليهم يوم القيامة ولا يزكيهم ولهم عذاب…٤٠٨٨المنان الذي لا يعطي شيئا إلا منه٤٠٨٩سبحان الله لا بأس أن يؤجر ويحمد٤٠٩٠قال الله عز وجل الكبرياء ردائي والعظمة إزاري فمن نازعني واحدا منهما…٤٠٩١لا يدخل الجنة من كان في قلبه مثقال حبة من خردل من كبر ولا يدخل النار…٤٠٩٢لا ولكن الكبر من بطر الحق وغمط الناس٤٠٩٣إزرة المسلم إلى نصف الساق ولا حرج - أو لا جناح - فيما بينه وبين…٤٠٩٤الإسبال في الإزار والقميص والعمامة من جر منها شيئا خيلاء لم ينظر الله…٤٠٩٥يقول ما قال رسول الله صلى الله عليه وسلم في الإزار فهو في القميص .٤٠٩٦أنه رأى ابن عباس يأتزر فيضع حاشية إزاره من مقدمه على ظهر قدميه ويرفع…٤٠٩٧حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا شعبة، عن قتادة، عن عكرمة،…٤٠٩٨حدثنا زهير بن حرب، حدثنا أبو عامر، عن سليمان بن بلال، عن سهيل، عن…٤٠٩٩حدثنا محمد بن سليمان، لوين - وبعضه قراءة عليه - عن سفيان، عن ابن…٤١٠٠رضى الله عنها أنها ذكرت نساء الأنصار فأثنت عليهن وقالت لهن معروفا…٤١٠١قالت لما نزلت { يدنين عليهن من جلابيبهن } خرج نساء الأنصار كأن على…٤١٠٢وابن السرح، وأحمد بن سعيد الهمداني، قالوا أخبرنا ابن وهب، قال أخبرني…٤١٠٣بإسناده ومعناه .٤١٠٤يا أسماء إن المرأة إذا بلغت المحيض لم تصلح أن يرى منها إلا هذا وهذا٤١٠٥وابن، موهب قالا حدثنا الليث، عن أبي الزبير، عن جابر، أن أم سلمة،…٤١٠٦إنه ليس عليك بأس إنما هو أبوك وغلامك٤١٠٧ألا أرى هذا يعلم ما ها هنا لا يدخلن عليكن هذا٤١٠٨بمعناه .٤١٠٩بهذا الحديث زاد وأخرجه فكان بالبيداء يدخل كل جمعة يستطعم .٤١١٠في هذه القصة فقيل يا رسول الله إنه إذا يموت من الجوع فأذن له أن يدخل…٤١١١{ وقل للمؤمنات يغضضن من أبصارهن } الآية فنسخ واستثني من ذلك {…٤١١٢حدثنا محمد بن العلاء، حدثنا ابن المبارك، عن يونس، عن الزهري، قال…٤١١٣إذا زوج أحدكم عبده أمته فلا ينظر إلى عورتها٤١١٤إذا زوج أحدكم خادمه عبده أو أجيره فلا ينظر إلى ما دون السرة وفوق…٤١١٥أن النبي صلى الله عليه وسلم دخل عليها وهي تختمر فقال " لية لا ليتين "…٤١١٦اصدعها صدعين فاقطع أحدهما قميصا وأعط الآخر امرأتك تختمر به٤١١٧أنها أخبرته أن أم سلمة زوج النبي صلى الله عليه وسلم قالت لرسول الله…٤١١٨حدثنا إبراهيم بن موسى، أخبرنا عيسى، عن عبيد الله، عن نافع، عن سليمان…٤١١٩حدثنا مسدد، حدثنا يحيى بن سعيد، عن سفيان، أخبرني زيد العمي، عن أبي…٤١٢٠ألا دبغتم إهابها فاستمتعتم به٤١٢١ألا انتفعتم بإهابها٤١٢٢قال قال معمر وكان الزهري ينكر الدباغ ويقول يستمتع به على كل حال . قال…٤١٢٣إذا دبغ الإهاب فقد طهر٤١٢٤زوج النبي صلى الله عليه وسلم أن رسول الله صلى الله عليه وسلم أمر أن…٤١٢٥وموسى بن إسماعيل، قالا حدثنا همام، عن قتادة، عن الحسن، عن جون بن…٤١٢٦لو أخذتم إهابها٤١٢٧أن لا تستمتعوا من الميتة بإهاب ولا عصب٤١٢٨حدثنا محمد بن إسماعيل، مولى بني هاشم حدثنا الثقفي، عن خالد، عن الحكم…٤١٢٩لا تركبوا الخز ولا النمار٤١٣٠لا تصحب الملائكة رفقة فيها جلد نمر٤١٣١هذا مني وحسين من علي٤١٣٢أن يحيى بن سعيد، وإسماعيل بن إبراهيم، حدثاهم - المعنى، - عن سعيد بن…٤١٣٣أكثروا من النعال فإن الرجل لا يزال راكبا ما انتعل٤١٣٤أن نعل النبي، صلى الله عليه وسلم كان لها قبالان .٤١٣٥قال نهى رسول الله صلى الله عليه وسلم أن ينتعل الرجل قائما .٤١٣٦لا يمشي أحدكم في النعل الواحدة لينتعلهما جميعا أو ليخلعهما جميعا٤١٣٧إذا انقطع شسع أحدكم فلا يمش في نعل واحدة حتى يصلح شسعه ولا يمش في خف…٤١٣٨قال من السنة إذا جلس الرجل أن يخلع نعليه فيضعهما بجنبه .٤١٣٩إذا انتعل أحدكم فليبدأ باليمين وإذا نزع فليبدأ بالشمال لتكن اليمين…٤١٤٠ومسلم بن إبراهيم، قالا حدثنا شعبة، عن الأشعث بن سليم، عن أبيه، عن…٤١٤١إذا لبستم وإذا توضأتم فابدءوا بأيامنكم٤١٤٢فراش للرجل وفراش للمرأة وفراش للضيف والرابع للشيطان٤١٤٣حدثنا أحمد بن حنبل، حدثنا وكيع، ح وحدثنا عبد الله بن الجراح، عن وكيع،…٤١٤٤أنه رأى رفقة من أهل اليمن رحالهم الأدم فقال من أحب أن ينظر إلى أشبه…٤١٤٥حدثنا ابن السرح، حدثنا سفيان، عن ابن المنكدر، عن جابر، قال قال لي…٤١٤٦وأحمد بن منيع، قالا حدثنا أبو معاوية، عن هشام بن عروة، عن أبيه، عن…٤١٤٧- رضى الله عنها - قالت كانت ضجعة رسول الله صلى الله عليه وسلم من أدم…٤١٤٨قالت كان فراشها حيال مسجد رسول الله صلى الله عليه وسلم .٤١٤٩وما أنا والدنيا وما أنا والرقم٤١٥٠بهذا الحديث قال وكان سترا موشيا .٤١٥١رضى الله عنها أن رسول الله صلى الله عليه وسلم كان لا يترك في بيته…٤١٥٢لا تدخل الملائكة بيتا فيه صورة ولا كلب ولا جنب٤١٥٣لا تدخل الملائكة بيتا فيه كلب ولا تمثال٤١٥٤حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن سهيل، بإسناده مثله قال فقلت…٤١٥٥إن الملائكة لا تدخل بيتا فيه صورة٤١٥٦أن إسماعيل بن عبد الكريم، حدثهم قال حدثني إبراهيم، - يعني ابن عقيل -…٤١٥٧إن جبريل عليه السلام كان وعدني أن يلقاني الليلة فلم يلقني٤١٥٨أتاني جبريل عليه السلام فقال لي أتيتك البارحة فلم يمنعني أن أكون دخلت…
