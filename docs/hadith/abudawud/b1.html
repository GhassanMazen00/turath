٣٩٠ حديثًا•صفحة ١ من ٣
١أن النبي صلى الله عليه وسلم كان إذا ذهب المذهب أبعد .٢أن النبي صلى الله عليه وسلم كان إذا أراد البراز انطلق حتى لا يراه أحد…٣إذا أراد أحدكم أن يبول فليرتد لبوله موضعا٤اللهم إني أعوذ بك٥اللهم إني أعوذ بك٦إن هذه الحشوش محتضرة فإذا أتى أحدكم الخلاء فليقل أعوذ بالله من الخبث…٧حدثنا مسدد بن مسرهد، حدثنا أبو معاوية، عن الأعمش، عن إبراهيم، عن عبد…٨إنما أنا لكم بمنزلة الوالد أعلمكم فإذا أتى أحدكم الغائط فلا يستقبل…٩إذا أتيتم الغائط فلا تستقبلوا القبلة بغائط ولا بول ولكن شرقوا أو غربوا١٠قال نهى رسول الله صلى الله عليه وسلم أن نستقبل القبلتين ببول أو غائط…١١حدثنا محمد بن يحيى بن فارس، حدثنا صفوان بن عيسى، عن الحسن بن ذكوان،…١٢واسع بن حبان، عن عبد الله بن عمر، قال لقد ارتقيت على ظهر البيت فرأيت…١٣قال نهى نبي الله صلى الله عليه وسلم أن نستقبل القبلة ببول فرأيته قبل…١٤حدثنا زهير بن حرب، حدثنا وكيع، عن الأعمش، عن رجل، عن ابن عمر، أن…١٥لا يخرج الرجلان يضربان الغائط كاشفين عن عورتهما يتحدثان فإن الله عز…١٦حدثنا عثمان، وأبو بكر ابنا أبي شيبة قالا حدثنا عمر بن سعد، عن سفيان،…١٧إني كرهت أن أذكر الله عز وجل إلا على طهر١٨قالت كان رسول الله صلى الله عليه وسلم يذكر الله عز وجل على كل أحيانه .١٩حدثنا نصر بن علي، عن أبي علي الحنفي، عن همام، عن ابن جريج، عن الزهري،…٢٠إنهما يعذبان وما يعذبان في كبير أما هذا فكان لا يستنزه من البول وأما…٢١كان لا يستتر من بوله٢٢ألم تعلموا ما لقي صاحب بني إسرائيل كانوا إذا أصابهم البول قطعوا ما…٢٣ومسلم بن إبراهيم، قالا حدثنا شعبة، ح وحدثنا مسدد، حدثنا أبو عوانة، -…٢٤أنها قالت كان للنبي صلى الله عليه وسلم قدح من عيدان تحت سريره يبول…٢٥أن رسول الله صلى الله عليه وسلم قال " اتقوا اللاعنين " . قالوا وما…٢٦اتقوا الملاعن الثلاث البراز في الموارد وقارعة الطريق والظل٢٧لا يبولن أحدكم في مستحمه ثم يغتسل فيه٢٨حدثنا أحمد بن يونس، حدثنا زهير، عن داود بن عبد الله، عن حميد الحميري،…٢٩أن رسول الله صلى الله عليه وسلم نهى أن يبال في الجحر . قال قالوا…٣٠حدثتني عائشة، رضى الله عنها أن النبي صلى الله عليه وسلم كان إذا خرج…٣١إذا بال أحدكم فلا يمس ذكره بيمينه وإذا أتى الخلاء فلا يتمسح بيمينه…٣٢ومعبد، عن حارثة بن وهب الخزاعي، قال حدثتني حفصة، زوج النبي صلى الله…٣٣قالت كانت يد رسول الله صلى الله عليه وسلم اليمنى لطهوره وطعامه وكانت…٣٤حدثنا محمد بن حاتم بن بزيع، حدثنا عبد الوهاب بن عطاء، عن سعيد، عن أبي…٣٥من اكتحل فليوتر من فعل فقد أحسن ومن لا فلا حرج ومن استجمر فليوتر من…٣٦يا رويفع لعل الحياة ستطول بك بعدي فأخبر الناس أنه من عقد لحيته أو…٣٧أن شييم بن بيتان، أخبره بهذا الحديث، أيضا عن أبي سالم الجيشاني، عن…٣٨أنه سمع جابر بن عبد الله، يقول نهانا رسول الله صلى الله عليه وسلم أن…٣٩حدثنا حيوة بن شريح الحمصي، حدثنا ابن عياش، عن يحيى بن أبي عمرو…٤٠إذا ذهب أحدكم إلى الغائط فليذهب معه بثلاثة أحجار يستطيب بهن فإنها…٤١بثلاثة أحجار ليس فيها رجيع٤٢وخلف بن هشام المقرئ، قالا حدثنا عبد الله بن يحيى التوأم، ح وحدثنا…٤٣حدثنا وهب بن بقية، عن خالد، - يعني الواسطي - عن خالد، - يعني الحذاء -…٤٤نزلت هذه الآية في أهل قباء { فيه رجال يحبون أن يتطهروا } قال كانوا…٤٥وهذا، لفظه ح وحدثنا محمد بن عبد الله، - يعني المخرمي - حدثنا وكيع، عن…٤٦لولا أن أشق، على المؤمنين لأمرتهم بتأخير العشاء وبالسواك عند كل صلاة٤٧لولا أن أشق على أمتي لأمرتهم بالسواك عند كل صلاة٤٨حدثنا محمد بن عوف الطائي، حدثنا أحمد بن خالد، حدثنا محمد بن إسحاق، عن…٤٩وسليمان بن داود العتكي، قالا حدثنا حماد بن زيد، عن غيلان بن جرير، عن…٥٠حدثنا محمد بن عيسى، حدثنا عنبسة بن عبد الواحد، عن هشام بن عروة، عن…٥١قال قلت لعائشة بأى شىء كان يبدأ رسول الله صلى الله عليه وسلم إذا دخل…٥٢أنها قالت كان نبي الله صلى الله عليه وسلم يستاك فيعطيني السواك لأغسله…٥٣عشر من الفطرة قص الشارب وإعفاء اللحية والسواك والاستنشاق بالماء وقص…٥٤إن من الفطرة المضمضة والاستنشاق٥٥وحصين، عن أبي وائل، عن حذيفة، أن رسول الله صلى الله عليه وسلم كان إذا…٥٦أن النبي صلى الله عليه وسلم كان يوضع له وضوءه وسواكه فإذا قام من…٥٧أن النبي صلى الله عليه وسلم كان لا يرقد من ليل ولا نهار فيستيقظ إلا…٥٨حدثنا محمد بن عيسى، حدثنا هشيم، أخبرنا حصين، عن حبيب بن أبي ثابت، عن…٥٩لا يقبل الله عز وجل صدقة من غلول ولا صلاة بغير طهور٦٠لا يقبل الله صلاة أحدكم إذا أحدث حتى يتوضأ٦١مفتاح الصلاة الطهور وتحريمها التكبير وتحليلها التسليم٦٢من توضأ على طهر كتب الله له عشر حسنات٦٣إذا كان الماء قلتين لم يحمل الخبث٦٤حدثنا موسى بن إسماعيل، حدثنا حماد، ح وحدثنا أبو كامل، حدثنا يزيد، -…٦٥إذا كان الماء قلتين فإنه لا ينجس٦٦الماء طهور لا ينجسه شىء٦٧إن الماء طهور لا ينجسه شىء٦٨إن الماء لا يجنب٦٩لا يبولن أحدكم في الماء الدائم ثم يغتسل منه٧٠لا يبولن أحدكم في الماء الدائم ولا يغتسل فيه من الجنابة٧١طهور إناء أحدكم إذا ولغ فيه الكلب أن يغسل سبع مرار أولاهن بتراب٧٢وإذا ولغ الهر غسل مرة٧٣إذا ولغ الكلب في الإناء فاغسلوه سبع مرات السابعة بالتراب٧٤أن رسول الله صلى الله عليه وسلم أمر بقتل الكلاب ثم قال " ما لهم ولها…٧٥إنها ليست بنجس إنها من الطوافين عليكم والطوافات٧٦إنها ليست بنجس إنما هي من الطوافين عليكم٧٧صلى الله عليه وسلم من إناء واحد ونحن جنبان .٧٨قالت اختلفت يدي ويد رسول الله صلى الله عليه وسلم في الوضوء من إناء…٧٩قال كان الرجال والنساء يتوضئون في زمان رسول الله صلى الله عليه وسلم -…٨٠حدثنا مسدد، حدثنا يحيى، عن عبيد الله، حدثني نافع، عن عبد الله بن عمر،…٨١قال لقيت رجلا صحب النبي صلى الله عليه وسلم أربع سنين كما صحبه أبو…٨٢وهو الأقرع أن النبي صلى الله عليه وسلم نهى أن يتوضأ الرجل بفضل طهور…٨٣هو الطهور ماؤه الحل ميتته٨٤وسليمان بن داود العتكي، قالا حدثنا شريك، عن أبي فزارة، عن أبي زيد، عن…٨٥حدثنا موسى بن إسماعيل، حدثنا وهيب، عن داود، عن عامر، عن علقمة، قال…٨٦أنه كره الوضوء باللبن والنبيذ وقال إن التيمم أعجب إلى منه .٨٧أصابته جنابة وليس عنده ماء وعنده نبيذ أيغتسل به قال لا .٨٨إذا أراد أحدكم أن يذهب الخلاء وقامت الصلاة فليبدأ بالخلاء٨٩لا يصلى بحضرة الطعام ولا وهو يدافعه الأخبثان٩٠ثلاث لا يحل لأحد أن يفعلهن لا يؤم رجل قوما فيخص نفسه بالدعاء دونهم…٩١لا يحل لرجل يؤمن بالله واليوم الآخر أن يصلي وهو حقن حتى يتخفف٩٢حدثنا محمد بن كثير، حدثنا همام، عن قتادة، عن صفية بنت شيبة، عن عائشة،…٩٣قال كان رسول الله صلى الله عليه وسلم يغتسل بالصاع ويتوضأ بالمد .٩٤وهي أم عمارة أن النبي صلى الله عليه وسلم توضأ فأتي بإناء فيه ماء قدر…٩٥حدثنا محمد بن الصباح البزاز، حدثنا شريك، عن عبد الله بن عيسى، عن عبد…٩٦إنه سيكون في هذه الأمة قوم يعتدون في الطهور والدعاء٩٧ويل للأعقاب من النار أسبغوا الوضوء٩٨أن عائشة، قالت كنت أغتسل أنا ورسول الله، صلى الله عليه وسلم في تور من…٩٩أن إسحاق بن منصور، حدثهم عن حماد بن سلمة، عن رجل، عن هشام بن عروة، عن…١٠٠وسهل بن حماد، قالا حدثنا عبد العزيز بن عبد الله بن أبي سلمة، عن عمرو…١٠١لا صلاة لمن لا وضوء له ولا وضوء لمن لم يذكر اسم الله تعالى عليه١٠٢لا وضوء لمن لم يذكر اسم الله عليه١٠٣إذا قام أحدكم من الليل فلا يغمس يده في الإناء حتى يغسلها ثلاث مرات…١٠٤حدثنا مسدد، حدثنا عيسى بن يونس، عن الأعمش، عن أبي صالح، عن أبي هريرة،…١٠٥إذا استيقظ أحدكم من نومه فلا يدخل يده في الإناء حتى يغسلها ثلاث مرات…١٠٦من توضأ مثل وضوئي هذا ثم صلى ركعتين لا يحدث فيهما نفسه غفر الله له ما…١٠٧من توضأ دون هذا كفاه١٠٨حدثنا محمد بن داود الإسكندراني، حدثنا زياد بن يونس، حدثني سعيد بن…١٠٩أن عثمان، دعا بماء فتوضأ فأفرغ بيده اليمنى على اليسرى ثم غسلهما إلى…١١٠حدثنا هارون بن عبد الله، حدثنا يحيى بن آدم، حدثنا إسرائيل، عن عامر بن…١١١حدثنا مسدد، حدثنا أبو عوانة، عن خالد بن علقمة، عن عبد خير، قال أتانا…١١٢قال صلى علي رضى الله عنه الغداة ثم دخل الرحبة فدعا بماء فأتاه الغلام…١١٣قال رأيت عليا - رضى الله عنه - أتي بكرسي فقعد عليه ثم أتي بكوز من ماء…١١٤أنه سمع عليا، رضى الله عنه وسئل عن وضوء، رسول الله صلى الله عليه وسلم…١١٥قال رأيت عليا - رضى الله عنه - توضأ فغسل وجهه ثلاثا وغسل ذراعيه ثلاثا…١١٦قال رأيت عليا - رضى الله عنه - توضأ فذكر وضوءه كله ثلاثا ثلاثا - قال…١١٧حدثنا عبد العزيز بن يحيى الحراني، حدثنا محمد، - يعني ابن سلمة - عن…١١٨أنه قال لعبد الله بن زيد بن عاصم - وهو جد عمرو بن يحيى المازني هل…١١٩بهذا الحديث قال فمضمض واستنشق من كف واحدة يفعل ذلك ثلاثا . ثم ذكر…١٢٠أن حبان بن واسع، حدثه أن أباه حدثه أنه، سمع عبد الله بن زيد بن عاصم…١٢١قال أتي رسول الله صلى الله عليه وسلم بوضوء فتوضأ فغسل كفيه ثلاثا ثم…١٢٢ويعقوب بن كعب الأنطاكي، - لفظه - قالا حدثنا الوليد بن مسلم، عن حريز…١٢٣وهشام بن خالد، - المعنى - قالا حدثنا الوليد، بهذا الإسناد قال ومسح…١٢٤ويزيد بن أبي مالك، أن معاوية، توضأ للناس كما رأى رسول الله صلى الله…١٢٥بهذا الإسناد قال فتوضأ ثلاثا ثلاثا وغسل رجليه بغير عدد .١٢٦حدثنا مسدد، حدثنا بشر بن المفضل، حدثنا عبد الله بن محمد بن عقيل، عن…١٢٧بهذا الحديث يغير بعض معاني بشر قال فيه وتمضمض واستنثر ثلاثا .١٢٨ويزيد بن خالد الهمداني، قالا حدثنا الليث، عن ابن عجلان، عن عبد الله…١٢٩أن ربيع بنت معوذ ابن عفراء، أخبرته قالت، رأيت رسول الله صلى الله عليه…١٣٠أن النبي صلى الله عليه وسلم مسح برأسه من فضل ماء كان في يده .١٣١أن النبي صلى الله عليه وسلم توضأ فأدخل إصبعيه في جحرى أذنيه .١٣٢ومسدد، قالا حدثنا عبد الوارث، عن ليث، عن طلحة بن مصرف، عن أبيه، عن…١٣٣رأى رسول الله صلى الله عليه وسلم يتوضأ . فذكر الحديث كله ثلاثا ثلاثا…١٣٤الأذنان من الرأس١٣٥هكذا الوضوء فمن زاد على هذا أو نقص فقد أساء وظلم١٣٦أن النبي صلى الله عليه وسلم توضأ مرتين مرتين .١٣٧كيف كان رسول الله صلى الله عليه وسلم يتوضأ فدعا بإناء فيه ماء فاغترف…١٣٨قال ألا أخبركم بوضوء، رسول الله صلى الله عليه وسلم فتوضأ مرة مرة .١٣٩قال دخلت - يعني - على النبي صلى الله عليه وسلم وهو يتوضأ والماء يسيل…١٤٠إذا توضأ أحدكم فليجعل في أنفه ماء ثم لينثر١٤١استنثروا مرتين بالغتين أو ثلاثا١٤٢هل أصبتم شيئا أو أمر لكم بشىء١٤٣وافد بني المنتفق، أنه أتى عائشة فذكر معناه . قال فلم ينشب أن جاء رسول…١٤٤إذا توضأت فمضمض١٤٥هكذا أمرني ربي عز وجل١٤٦قال بعث رسول الله صلى الله عليه وسلم سرية فأصابهم البرد فلما قدموا…١٤٧قال رأيت رسول الله صلى الله عليه وسلم يتوضأ وعليه عمامة قطرية فأدخل…١٤٨قال رأيت رسول الله صلى الله عليه وسلم إذا توضأ يدلك أصابع رجليه…١٤٩أن عروة بن المغيرة بن شعبة، أخبره أنه، سمع أباه المغيرة، يقول عدل…١٥٠حدثنا مسدد، حدثنا يحيى يعني ابن
