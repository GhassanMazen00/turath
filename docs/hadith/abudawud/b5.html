١٢١ حديثًا
١٢٥٠من صلى في يوم ثنتى عشرة ركعة تطوعا بني له بهن بيت في الجنة١٢٥١رسول الله صلى الله عليه وسلم من التطوع فقالت كان يصلي قبل الظهر أربعا…١٢٥٢أن رسول الله صلى الله عليه وسلم كان يصلي قبل الظهر ركعتين وبعدها…١٢٥٣أن النبي صلى الله عليه وسلم كان لا يدع أربعا قبل الظهر وركعتين قبل…١٢٥٤- رضى الله عنها - قالت إن رسول الله صلى الله عليه وسلم لم يكن على شىء…١٢٥٥قالت كان النبي صلى الله عليه وسلم يخفف الركعتين قبل صلاة الفجر حتى…١٢٥٦أن النبي صلى الله عليه وسلم قرأ في ركعتى الفجر { قل يا أيها الكافرون…١٢٥٧إني كنت ركعت ركعتى الفجر١٢٥٨لا تدعوهما وإن طردتكم الخيل١٢٥٩أن كثيرا، مما كان يقرأ رسول الله صلى الله عليه وسلم في ركعتى الفجر ب…١٢٦٠حدثنا محمد بن الصباح بن سفيان، حدثنا عبد العزيز بن محمد، عن عثمان بن…١٢٦١إذا صلى أحدكم الركعتين قبل الصبح فليضطجع على يمينه١٢٦٢حدثنا يحيى بن حكيم، حدثنا بشر بن عمر، حدثنا مالك بن أنس، عن سالم أبي…١٢٦٣عمن حدثه - ابن أبي عتاب، أو غيره - عن أبي سلمة، قال قالت عائشة كان…١٢٦٤وزياد بن يحيى، قالا حدثنا سهل بن حماد، عن أبي مكين، حدثنا أبو الفضل،…١٢٦٥يا فلان أيتهما صلاتك التي صليت وحدك أو التي صليت معنا١٢٦٦إذا أقيمت الصلاة فلا صلاة إلا المكتوبة١٢٦٧صلاة الصبح ركعتان١٢٦٨حدثنا حامد بن يحيى البلخي، قال قال سفيان كان عطاء بن أبي رباح يحدث…١٢٦٩من حافظ على أربع ركعات قبل الظهر وأربع بعدها حرم على النار١٢٧٠أربع قبل الظهر ليس فيهن تسليم تفتح لهن أبواب السماء١٢٧١رحم الله امرأ صلى قبل العصر أربعا١٢٧٢عليه السلام أن النبي صلى الله عليه وسلم كان يصلي قبل العصر ركعتين .١٢٧٣يا بنت أبي أمية سألت عن الركعتين بعد العصر إنه أتاني ناس من عبد القيس…١٢٧٤حدثنا مسلم بن إبراهيم، حدثنا شعبة، عن منصور، عن هلال بن يساف، عن وهب…١٢٧٥قال كان رسول الله صلى الله عليه وسلم يصلي في إثر كل صلاة مكتوبة…١٢٧٦لا صلاة بعد صلاة الصبح حتى تطلع الشمس ولا صلاة بعد صلاة العصر حتى…١٢٧٧جوف الليل الآخر فصل ما شئت فإن الصلاة مشهودة مكتوبة حتى تصلي الصبح ثم…١٢٧٨ليبلغ شاهدكم غائبكم لا تصلوا بعد الفجر إلا سجدتين١٢٧٩ومسروق، قالا نشهد على عائشة - رضى الله عنها - أنها قالت ما من يوم…١٢٨٠حدثنا عبيد الله بن سعد، حدثنا عمي، حدثنا أبي، عن ابن إسحاق، عن محمد…١٢٨١صلوا قبل المغرب ركعتين١٢٨٢حدثنا محمد بن عبد الرحيم البزاز، أخبرنا سعيد بن سليمان، حدثنا منصور…١٢٨٣بين كل أذانين صلاة بين كل أذانين صلاة لمن شاء١٢٨٤حدثنا ابن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، عن أبي شعيب، عن طاوس،…١٢٨٥يصبح على كل سلامى من ابن آدم صدقة تسليمه على من لقي صدقة وأمره…١٢٨٦يصبح على كل سلامى من أحدكم في كل يوم صدقة فله بكل صلاة صدقة وصيام…١٢٨٧من قعد في مصلاه حين ينصرف من صلاة الصبح حتى يسبح ركعتى الضحى لا يقول…١٢٨٨صلاة في أثر صلاة لا لغو بينهما كتاب في عليين١٢٨٩يقول الله عز وجل يا ابن آدم لا تعجزني من أربع ركعات في أول نهارك أكفك…١٢٩٠وأحمد بن عمرو بن السرح، قالا حدثنا ابن وهب، حدثني عياض بن عبد الله،…١٢٩١أنه رأى النبي صلى الله عليه وسلم صلى الضحى غير أم هانئ فإنها ذكرت أن…١٢٩٢قال سألت عائشة هل كان رسول الله صلى الله عليه وسلم يصلي الضحى فقالت…١٢٩٣زوج النبي صلى الله عليه وسلم أنها قالت ما سبح رسول الله صلى الله عليه…١٢٩٤وأحمد بن يونس، قالا حدثنا زهير، حدثنا سماك، قال قلت لجابر بن سمرة…١٢٩٥صلاة الليل والنهار مثنى مثنى١٢٩٦الصلاة مثنى مثنى أن تشهد في كل ركعتين وأن تباءس وتمسكن وتقنع بيديك…١٢٩٧يا عباس يا عماه ألا أعطيك ألا أمنحك ألا أحبوك ألا أفعل بك عشر خصال…١٢٩٨ائتني غدا أحبوك وأثيبك وأعطيك١٢٩٩أن رسول الله صلى الله عليه وسلم قال لجعفر بهذا الحديث فذكر نحوهم قال…١٣٠٠هذه صلاة البيوت١٣٠١حدثنا حسين بن عبد الرحمن الجرجرائي، حدثنا طلق بن غنام، حدثنا يعقوب بن…١٣٠٢وسليمان بن داود العتكي، قالا حدثنا يعقوب، عن جعفر، عن سعيد بن جبير،…١٣٠٣حدثنا محمد بن رافع، حدثنا زيد بن الحباب العكلي، حدثني مالك بن مغول،…١٣٠٤قال في المزمل { قم الليل إلا قليلا * نصفه } نسختها الآية التي فيها {…١٣٠٥قال لما نزلت أول المزمل كانوا يقومون نحوا من قيامهم في شهر رمضان حتى…١٣٠٦يعقد الشيطان على قافية رأس أحدكم إذا هو نام ثلاث عقد يضرب مكان كل…١٣٠٧يقول قالت عائشة رضى الله عنها لا تدع قيام الليل فإن رسول الله صلى…١٣٠٨رحم الله رجلا قام من الليل فصلى وأيقظ امرأته فإن أبت نضح في وجهها…١٣٠٩إذا أيقظ الرجل أهله من الليل فصليا أو صلى ركعتين جميعا كتبا في…١٣١٠إذا نعس أحدكم في الصلاة فليرقد حتى يذهب عنه النوم فإن أحدكم إذا صلى…١٣١١إذا قام أحدكم من الليل فاستعجم القرآن على لسانه فلم يدر ما يقول…١٣١٢وهارون بن عباد الأزدي، أن إسماعيل بن إبراهيم، حدثهم حدثنا عبد العزيز،…١٣١٣من نام عن حزبه أو عن شىء منه فقرأه ما بين صلاة الفجر وصلاة الظهر كتب…١٣١٤ما من امرئ تكون له صلاة بليل يغلبه عليها نوم إلا كتب له أجر صلاته…١٣١٥ينزل ربنا تبارك وتعالى كل ليلة إلى سماء الدنيا حين يبقى ثلث الليل…١٣١٦قالت إن كان رسول الله صلى الله عليه وسلم ليوقظه الله عز وجل بالليل…١٣١٧حدثنا إبراهيم بن موسى، حدثنا أبو الأحوص، ح وحدثنا هناد، عن أبي…١٣١٨قالت : ما ألفاه السحر عندي إلا نائما، تعني النبي صلى الله عليه وسلم .١٣١٩قال : كان النبي صلى الله عليه وسلم إذا حزبه أمر صلى .١٣٢٠يقول : كنت أبيت مع رسول الله صلى الله عليه وسلم آتيه بوضوئه وبحاجته،…١٣٢١يدعون ربهم خوفا وطمعا ومما رزقناهم ينفقون } قال : كانوا يتيقظون ما…١٣٢٢وابن أبي عدي، عن سعيد، عن قتادة، عن أنس، في قوله جل وعز { كانوا قليلا…١٣٢٣إذا قام أحدكم من الليل فليصل ركعتين خفيفتين١٣٢٤وكذلك رواه أيوب وابن عون أوقفوه على أبي هريرة، ورواه ابن عون عن محمد…١٣٢٥: أن رسول الله صلى الله عليه وسلم سئل أى الأعمال أفضل قال : " طول…١٣٢٦صلاة الليل مثنى مثنى، فإذا خشي أحدكم الصبح صلى ركعة واحدة توتر له ما…١٣٢٧قال : كانت قراءة النبي صلى الله عليه وسلم على قدر ما يسمعه من في…١٣٢٨أنه قال : كانت قراءة النبي صلى الله عليه وسلم بالليل يرفع طورا ويخفض…١٣٢٩يا أبا بكر مررت بك وأنت تصلي تخفض صوتك١٣٣٠ارفع من صوتك شيئا١٣٣١يرحم الله فلانا، كأين من آية أذكرنيها الليلة كنت قد أسقطتها١٣٣٢ألا إن كلكم مناج ربه فلا يؤذين بعضكم بعضا، ولا يرفع بعضكم على بعض في…١٣٣٣الجاهر بالقرآن كالجاهر بالصدقة، والمسر بالقرآن كالمسر بالصدقة١٣٣٤قالت : كان رسول الله صلى الله عليه وسلم يصلي من الليل عشر ركعات،…١٣٣٥زوج النبي صلى الله عليه وسلم : أن رسول الله صلى الله عليه وسلم كان…١٣٣٦ونصر بن عاصم، - وهذا لفظه - قالا حدثنا الوليد، حدثنا الأوزاعي، - وقال…١٣٣٧وعمرو بن الحارث، ويونس بن يزيد، أن ابن شهاب، أخبرهم بإسناده، ومعناه،،…١٣٣٨قالت : كان رسول الله صلى الله عليه وسلم يصلي من الليل ثلاث عشرة ركعة…١٣٣٩قالت : كان رسول الله صلى الله عليه وسلم يصلي بالليل ثلاث عشرة ركعة،…١٣٤٠ومسلم بن إبراهيم، قالا حدثنا أبان، عن يحيى، عن أبي سلمة، عن عائشة، :…١٣٤١يا عائشة إن عينى تنامان ولا ينام قلبي١٣٤٢لقد كان لكم في رسول الله أسوة حسنة١٣٤٣بإسناده نحوه قال : يصلي ثمان ركعات لا يجلس فيهن إلا عند الثامنة،…١٣٤٤حدثنا عثمان بن أبي شيبة، حدثنا محمد بن بشر، حدثنا سعيد، بهذا الحديث…١٣٤٥حدثنا محمد بن بشار، حدثنا ابن أبي عدي، عن سعيد، بهذا الحديث قال ابن…١٣٤٦: أن عائشة، - رضى الله عنها - سئلت عن صلاة رسول الله صلى الله عليه…١٣٤٧فذكر هذا الحديث بإسناده قال : يصلي العشاء ثم يأوي إلى فراشه، لم يذكر…١٣٤٨رسول الله صلى الله عليه وسلم فقالت : كان يصلي بالناس العشاء، ثم يرجع…١٣٤٩- رضى الله عنها - بهذا الحديث وليس في تمام حديثهم .١٣٥٠رضى الله عنها : أن رسول الله صلى الله عليه وسلم كان يصلي من الليل…١٣٥١رضى الله عنها : أن رسول الله صلى الله عليه وسلم كان يوتر بتسع ركعات،…١٣٥٢ثم يأوي إلى فراشه فينام، فإذا كان جوف الليل قام إلى حاجته وإلى طهوره…١٣٥٣اللهم اجعل في قلبي نورا، واجعل في لساني نورا، واجعل في سمعي نورا،…١٣٥٤حدثنا وهب بن بقية، عن خالد، عن حصين، نحوه قال : " وأعظم لي نورا " .…١٣٥٥قال : بت ليلة عند النبي صلى الله عليه وسلم لأنظر كيف يصلي فقام فتوضأ…١٣٥٦قال : بت عند خالتي ميمونة فجاء رسول الله صلى الله عليه وسلم بعد ما…١٣٥٧قال : بت في بيت خالتي ميمونة بنت الحارث فصلى النبي صلى الله عليه وسلم…١٣٥٨أن ابن عباس، حدثه في، هذه القصة قال : فقام فصلى ركعتين ركعتين، حتى…١٣٥٩قالت : كان رسول الله صلى الله عليه وسلم يصلي ثلاث عشرة ركعة بركعتيه…١٣٦٠أنها أخبرته : أن النبي صلى الله عليه وسلم كان يصلي بالليل ثلاث عشرة…١٣٦١وجعفر بن مسافر، أن عبد الله بن يزيد المقرئ، أخبرهما عن سعيد بن أبي…١٣٦٢ومحمد بن سلمة المرادي، قالا حدثنا ابن وهب، عن معاوية بن صالح، عن عبد…١٣٦٣رسول الله صلى الله عليه وسلم بالليل . فقالت : كان يصلي ثلاث عشرة ركعة…١٣٦٤أن كريبا، مولى ابن عباس أخبره أنه، قال : سألت ابن عباس كيف كانت صلاة…١٣٦٥ويحيى بن موسى، قالا حدثنا عبد الرزاق، أخبرنا معمر، عن ابن طاوس، عن…١٣٦٦أن عبد الله بن قيس بن مخرمة، أخبره عن زيد بن خالد الجهني، أنه - قال -…١٣٦٧مولى ابن عباس أن عبد الله بن عباس، أخبره : أنه، بات عند ميمونة زوج…١٣٦٨اكلفوا من العمل ما تطيقون، فإن الله لا يمل حتى تملوا، وإن أحب العمل…١٣٦٩يا عثمان أرغبت عن سنتي١٣٧٠قال : سألت عائشة كيف كان عمل رسول الله صلى الله عليه وسلم هل كان يخص…
