سنن أبي داود

كتاب الصلاة

٧٧٠ حديثًا•صفحة ١ من ٦
٣٩١خمس صلوات في اليوم والليلة٣٩٢أفلح وأبيه إن صدق دخل الجنة وأبيه إن صدق٣٩٣أمني جبريل عليه السلام عند البيت مرتين فصلى بي الظهر حين زالت الشمس…٣٩٤نزل جبريل صلى الله عليه وسلم فأخبرني بوقت الصلاة فصليت معه ثم صليت…٣٩٥أين السائل عن وقت الصلاة الوقت فيما بين هذين٣٩٦وقت الظهر ما لم تحضر العصر ووقت العصر ما لم تصفر الشمس ووقت المغرب ما…٣٩٧صلاة النبي صلى الله عليه وسلم فقال كان يصلي الظهر بالهاجرة والعصر…٣٩٨حدثنا حفص بن عمر، حدثنا شعبة، عن أبي المنهال، عن أبي برزة، قال كان…٣٩٩ومسدد، قالا حدثنا عباد بن عباد، حدثنا محمد بن عمرو، عن سعيد بن الحارث…٤٠٠أن عبد الله بن مسعود، قال كانت قدر صلاة رسول الله صلى الله عليه وسلم…٤٠١حدثنا أبو الوليد الطيالسي، حدثنا شعبة، أخبرني أبو الحسن، - قال أبو…٤٠٢إذا اشتد الحر فأبردوا عن الصلاة٤٠٣أن بلالا، كان يؤذن الظهر إذا دحضت الشمس .٤٠٤أنه أخبره أن رسول الله صلى الله عليه وسلم كان يصلي العصر والشمس بيضاء…٤٠٥قال والعوالي على ميلين أو ثلاثة . قال وأحسبه قال أو أربعة .٤٠٦قال حياتها أن تجد، حرها .٤٠٧قال عروة ولقد حدثتني عائشة، أن رسول الله صلى الله عليه وسلم كان يصلي…٤٠٨حدثنا محمد بن عبد الرحمن العنبري، حدثنا إبراهيم بن أبي الوزير، حدثنا…٤٠٩حبسونا عن صلاة الوسطى صلاة العصر ملأ الله بيوتهم وقبورهم نارا٤١٠حدثنا القعنبي، عن مالك، عن زيد بن أسلم، عن القعقاع بن حكيم، عن أبي…٤١١إن قبلها صلاتين وبعدها صلاتين٤١٢من أدرك من العصر ركعة قبل أن تغرب الشمس فقد أدرك ومن أدرك من الفجر…٤١٣تلك صلاة المنافقين تلك صلاة المنافقين تلك صلاة المنافقين يجلس أحدهم…٤١٤الذي تفوته صلاة العصر فكأنما وتر أهله وماله٤١٥قال قال أبو عمرو يعني الأوزاعي وذلك أن ترى، ما على الأرض من الشمس…٤١٦حدثنا داود بن شبيب، حدثنا حماد، عن ثابت البناني، عن أنس بن مالك، قال…٤١٧قال كان النبي صلى الله عليه وسلم يصلي المغرب ساعة تغرب الشمس إذا غاب…٤١٨لا تزال أمتي بخير - أو قال على الفطرة - ما لم يؤخروا المغرب إلى أن…٤١٩بوقت هذه الصلاة صلاة العشاء الآخرة كان رسول الله صلى الله عليه وسلم…٤٢٠أتنتظرون هذه الصلاة لولا أن تثقل على أمتي لصليت بهم هذه الساعة٤٢١أعتموا بهذه الصلاة فإنكم قد فضلتم بها على سائر الأمم ولم تصلها أمة…٤٢٢حدثنا مسدد، حدثنا بشر بن المفضل، حدثنا داود بن أبي هند، عن أبي نضرة،…٤٢٣- رضى الله عنها - أنها قالت إن كان رسول الله صلى الله عليه وسلم ليصلي…٤٢٤أصبحوا بالصبح فإنه أعظم لأجوركم٤٢٥خمس صلوات افترضهن الله تعالى من أحسن وضوءهن وصلاهن لوقتهن وأتم ركوعهن…٤٢٦الصلاة في أول وقتها٤٢٧لا يلج النار رجل صلى قبل طلوع الشمس وقبل أن تغرب٤٢٨وحافظ على الصلوات الخمس٤٢٩خمس من جاء بهن مع إيمان دخل الجنة من حافظ على الصلوات الخمس على…٤٣٠قال الله تعالى إني فرضت على أمتك خمس صلوات وعهدت عندي عهدا أنه من جاء…٤٣١يا أبا ذر كيف أنت إذا كانت عليك أمراء يميتون الصلاة٤٣٢كيف بكم إذا أتت عليكم أمراء يصلون الصلاة لغير ميقاتها٤٣٣إنها ستكون عليكم بعدي أمراء تشغلهم أشياء عن الصلاة لوقتها حتى يذهب…٤٣٤يكون عليكم أمراء من بعدي يؤخرون الصلاة فهي لكم وهي عليهم فصلوا معهم…٤٣٥حدثنا أحمد بن صالح، حدثنا ابن وهب، أخبرني يونس، عن ابن شهاب، عن ابن…٤٣٦تحولوا عن مكانكم الذي أصابتكم فيه الغفلة٤٣٧حدثنا موسى بن إسماعيل، حدثنا حماد، عن ثابت البناني، عن عبد الله بن…٤٣٨حدثنا علي بن نصر، حدثنا وهب بن جرير، حدثنا الأسود بن شيبان، حدثنا…٤٣٩إن الله قبض أرواحكم حيث شاء وردها حيث شاء قم فأذن بالصلاة٤٤٠حدثنا هناد، حدثنا عبثر، عن حصين، عن عبد الله بن أبي قتادة، عن أبيه،…٤٤١ليس في النوم تفريط إنما التفريط في اليقظة أن تؤخر صلاة حتى يدخل وقت…٤٤٢من نسي صلاة فليصلها إذا ذكرها لا كفارة لها إلا ذلك٤٤٣حدثنا وهب بن بقية، عن خالد، عن يونس بن عبيد، عن الحسن، عن عمران بن…٤٤٤تنحوا عن هذا المكان٤٤٥حدثنا إبراهيم بن الحسن، حدثنا حجاج، - يعني ابن محمد - حدثنا حريز، ح…٤٤٦مخبر بن أخي النجاشي في هذا الخبر قال فأذن وهو غير عجل .٤٤٧حدثنا محمد بن المثنى، حدثنا محمد بن جعفر، حدثنا شعبة، عن جامع بن…٤٤٨ما أمرت بتشييد المساجد٤٤٩لا تقوم الساعة حتى يتباهى الناس في المساجد٤٥٠أن النبي صلى الله عليه وسلم أمره أن يجعل مسجد الطائف حيث كان طواغيتهم…٤٥١ومجاهد بن موسى، - وهو أتم - قالا حدثنا يعقوب بن إبراهيم، حدثنا أبي،…٤٥٢أن مسجد النبي، صلى الله عليه وسلم كانت سواريه على عهد رسول الله صلى…٤٥٣يا بني النجار ثامنوني بحائطكم هذا٤٥٤حدثنا موسى بن إسماعيل، حدثنا حماد بن سلمة، عن أبي التياح، عن أنس بن…٤٥٥قالت أمر رسول الله صلى الله عليه وسلم ببناء المساجد في الدور وأن تنظف…٤٥٦حدثنا محمد بن داود بن سفيان، حدثنا يحيى، - يعني ابن حسان - حدثنا…٤٥٧حدثنا النفيلي، حدثنا مسكين، عن سعيد بن عبد العزيز، عن زياد بن أبي…٤٥٨حدثنا سهل بن تمام بن بزيع، حدثنا عمر بن سليم الباهلي، عن أبي الوليد،…٤٥٩ووكيع، قالا حدثنا الأعمش، عن أبي صالح، قال كان يقال إن الرجل إذا أخرج…٤٦٠إن الحصاة لتناشد الذي يخرجها من المسجد٤٦١عرضت على أجور أمتي حتى القذاة يخرجها الرجل من المسجد وعرضت على ذنوب…٤٦٢لو تركنا هذا الباب للنساء٤٦٣قال قال عمر بن الخطاب - رضى الله عنه - فذكره بمعناه وهو أصح .٤٦٤أن عمر بن الخطاب، كان ينهى أن يدخل، من باب النساء .٤٦٥إذا دخل أحدكم المسجد فليسلم على النبي صلى الله عليه وسلم ثم ليقل…٤٦٦أعوذ بالله العظيم وبوجهه الكريم وسلطانه القديم من الشيطان الرجيم٤٦٧إذا جاء أحدكم المسجد فليصل سجدتين من قبل أن يجلس٤٦٨ثم ليقعد بعد إن شاء أو ليذهب لحاجته٤٦٩الملائكة تصلي على أحدكم ما دام في مصلاه الذي صلى فيه ما لم يحدث أو…٤٧٠لا يزال أحدكم في صلاة ما كانت الصلاة تحبسه لا يمنعه أن ينقلب إلى أهله…٤٧١لا يزال العبد في صلاة ما كان في مصلاه ينتظر الصلاة تقول الملائكة…٤٧٢من أتى المسجد لشىء فهو حظه٤٧٣من سمع رجلا ينشد ضالة في المسجد فليقل لا أداها الله إليك فإن المساجد…٤٧٤التفل في المسجد خطيئة وكفارته أن تواريه٤٧٥البزاق في المسجد خطيئة وكفارتها دفنها٤٧٦النخاعة في المسجد٤٧٧من دخل هذا المسجد فبزق فيه أو تنخم فليحفر فليدفنه فإن لم يفعل فليبزق…٤٧٨إذا قام الرجل إلى الصلاة - أو إذا صلى أحدكم فلا يبزق أمامه ولا عن…٤٧٩إن الله قبل وجه أحدكم إذا صلى فلا يبزق بين يديه٤٨٠أيسر أحدكم أن يبصق في وجهه إن أحدكم إذا استقبل القبلة فإنما يستقبل…٤٨١- قال أحمد - من أصحاب النبي صلى الله عليه وسلم أن رجلا أم قوما فبصق…٤٨٢قال أتيت رسول الله صلى الله عليه وسلم وهو يصلي فبزق تحت قدمه اليسرى .٤٨٣بمعناه زاد ثم دلكه بنعله .٤٨٤قال رأيت واثلة بن الأسقع في مسجد دمشق بصق على البوري ثم مسحه برجله…٤٨٥أيكم يحب أن يعرض الله عنه بوجهه٤٨٦أنه سمع أنس بن مالك، يقول دخل رجل على جمل فأناخه في المسجد ثم عقله ثم…٤٨٧أنا ابن عبد المطلب٤٨٨قال اليهود أتوا النبي صلى الله عليه وسلم وهو جالس في المسجد في أصحابه…٤٨٩جعلت لي الأرض طهورا ومسجدا٤٩٠ويحيى بن أزهر، عن عمار بن سعد المرادي، عن أبي صالح الغفاري، أن عليا،…٤٩١وابن، لهيعة عن الحجاج بن شداد، عن أبي صالح الغفاري، عن علي، بمعنى…٤٩٢الأرض كلها مسجد إلا الحمام والمقبرة٤٩٣لا تصلوا في مبارك الإبل فإنها من الشياطين٤٩٤مروا الصبي بالصلاة إذا بلغ سبع سنين وإذا بلغ عشر سنين فاضربوه عليها٤٩٥مروا أولادكم بالصلاة وهم أبناء سبع سنين واضربوهم عليها وهم أبناء عشر…٤٩٦وإذا زوج أحدكم خادمه عبده أو أجيره فلا ينظر إلى ما دون السرة وفوق…٤٩٧إذا عرف يمينه من شماله فمروه بالصلاة٤٩٨هو من أمر اليهود٤٩٩إنها لرؤيا حق إن شاء الله فقم مع بلال فألق عليه ما رأيت فليؤذن به…٥٠٠تقول الله أكبر الله أكبر الله أكبر الله أكبر ترفع بها صوتك ثم تقول…٥٠١الصلاة خير من النوم الصلاة خير من النوم في الأولى من الصبح٥٠٢الله أكبر الله أكبر الله أكبر الله أكبر أشهد أن لا إله إلا الله أشهد…٥٠٣قل الله أكبر الله أكبر الله أكبر الله أكبر أشهد أن لا إله إلا الله…٥٠٤الله أكبر الله أكبر الله أكبر الله أكبر أشهد أن لا إله إلا الله أشهد…٥٠٥الله أكبر الله أكبر أشهد أن لا إله إلا الله أشهد أن لا إله إلا الله…٥٠٦لقد أعجبني أن تكون صلاة المسلمين - أو قال المؤمنين - واحدة حتى لقد…٥٠٧حدثنا محمد بن المثنى، عن أبي داود، ح وحدثنا نصر بن المهاجر، حدثنا…٥٠٨وعبد الرحمن بن المبارك، قالا حدثنا حماد، عن سماك بن عطية، ح وحدثنا…٥٠٩مثل حديث وهيب . قال إسماعيل فحدثت به، أيوب فقال إلا الإقامة .٥١٠حدثنا محمد بن بشار، حدثنا محمد بن جعفر، حدثنا شعبة، سمعت أبا جعفر،…٥١١وساق الحديث٥١٢حدثنا عثمان بن أبي شيبة، حدثنا حماد بن خالد، حدثنا محمد بن عمرو، عن…٥١٣حدثنا عبيد الله بن عمر القواريري، حدثنا عبد الرحمن بن مهدي، حدثنا…٥١٤- يعني الإفريقي - أنه سمع زياد بن نعيم الحضرمي، أنه سمع زياد بن…٥١٥المؤذن يغفر له مدى صوته ويشهد له كل رطب ويابس وشاهد الصلاة يكتب له…٥١٦إذا نودي بالصلاة أدبر الشيطان وله ضراط حتى لا يسمع التأذين فإذا قضي…٥١٧الإمام ضامن والمؤذن مؤتمن اللهم أرشد الأئمة واغفر للمؤذنين٥١٨قال قال رسول الله صلى الله عليه وسلم مثله .٥١٩من بني النجار قالت كان بيتي من أطول بيت حول المسجد وكان بلال يؤذن…٥٢٠حدثنا موسى بن إسماعيل، حدثنا قيس يعني ابن الربيع، ح وحدثنا محمد بن…٥٢١لا يرد الدعاء بين الأذان والإقامة٥٢٢إذا سمعتم النداء فقولوا مثل ما يقول المؤذن٥٢٣إذا سمعتم المؤذن فقولوا مثل ما يقول ثم صلوا على فإنه من صلى على صلاة…٥٢٤قل كما يقولون فإذا انتهيت فسل تعطه٥٢٥من قال حين يسمع المؤذن وأنا أشهد أن لا إله إلا الله وحده لا شريك له…٥٢٦حدثنا إبراهيم بن مهدي، حدثنا علي بن مسهر، عن هشام بن عروة، عن أبيه،…٥٢٧إذا قال المؤذن الله أكبر الله أكبر فقال أحدكم الله أكبر الله أكبر…٥٢٨أقامها الله وأدامها٥٢٩من قال حين يسمع النداء اللهم رب هذه الدعوة التامة والصلاة القائمة آت…٥٣٠اللهم إن هذا إقبال ليلك وإدبار نهارك وأصوات دعاتك فاغفر لي٥٣١أنت إمامهم واقتد بأضعفهم واتخذ مؤذنا لا يأخذ على أذانه أجرا٥٣٢ألا إن العبد قد نام ألا إن العبد قد نام٥٣٣حدثنا أيوب بن منصور، حدثنا شعيب بن حرب، عن عبد العزيز بن أبي رواد،…٥٣٤لا تؤذن حتى يستبين لك الفجر هكذا٥٣٥وسعيد بن عبد الرحمن، عن هشام بن عروة، عن أبيه، عن عائشة، أن ابن أم…٥٣٦حدثنا محمد بن كثير، حدثنا سفيان، عن إبراهيم بن المهاجر، عن أبي…٥٣٧قال كان بلال يؤذن ثم يمهل فإذا رأى النبي صلى الله عليه وسلم قد خرج…٥٣٨حدثنا محمد بن كثير، حدثنا سفيان، حدثنا أبو يحيى القتات، عن مجاهد، قال…٥٣٩إذا أقيمت الصلاة فلا تقوموا حتى تروني٥٤٠حتى تروني قد خرجت
