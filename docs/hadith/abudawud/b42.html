سنن أبي داود

كتاب السنة

١٧٧ حديثًا•صفحة ١ من ٢
٤٥٩٦افترقت اليهود على إحدى أو ثنتين وسبعين فرقة وتفرقت النصارى على إحدى…٤٥٩٧ألا إن من قبلكم من أهل الكتاب افترقوا على ثنتين وسبعين ملة وإن هذه…٤٥٩٨فإذا رأيتم الذين يتبعون ما تشابه منه فأولئك الذين سمى الله فاحذروهم٤٥٩٩أفضل الأعمال الحب في الله والبغض في الله٤٦٠٠أن عبد الله بن كعب، - وكان قائد كعب من بنيه حين عمي - قال سمعت كعب بن…٤٦٠١اذهب فاغسل هذا عنك٤٦٠٢حدثنا موسى بن إسماعيل، حدثنا حماد، عن ثابت البناني، عن سمية، عن…٤٦٠٣المراء في القرآن كفر٤٦٠٤ألا إني أوتيت الكتاب ومثله معه ألا يوشك رجل شبعان على أريكته يقول…٤٦٠٥لا ألفين أحدكم متكئا على أريكته يأتيه الأمر من أمري مما أمرت به أو…٤٦٠٦من أحدث في أمرنا هذا ما ليس فيه فهو رد٤٦٠٧أوصيكم بتقوى الله والسمع والطاعة وإن عبدا حبشيا فإنه من يعش منكم بعدي…٤٦٠٨ألا هلك المتنطعون٤٦٠٩من دعا إلى هدى كان له من الأجر مثل أجور من تبعه لا ينقص ذلك من أجورهم…٤٦١٠إن أعظم المسلمين في المسلمين جرما من سأل عن أمر لم يحرم فحرم على…٤٦١١أن أبا إدريس الخولاني، عائذ الله أخبره أن يزيد بن عميرة وكان من أصحاب…٤٦١٢حدثنا محمد بن كثير، قال حدثنا سفيان، قال كتب رجل إلى عمر بن عبد…٤٦١٣إنه سيكون في أمتي أقوام يكذبون بالقدر٤٦١٤حدثنا عبد الله بن الجراح، قال حدثنا حماد بن زيد، عن خالد الحذاء، قال…٤٦١٥في قوله تعالى { ولذلك خلقهم } قال خلق هؤلاء لهذه وهؤلاء لهذه .٤٦١٦قال قلت للحسن { ما أنتم عليه بفاتنين * إلا من هو صال الجحيم } قال إلا…٤٦١٧قال كان الحسن يقول لأن يسقط من السماء إلى الأرض أحب إليه من أن يقول…٤٦١٨حدثنا موسى بن إسماعيل، قال حدثنا حماد، حدثنا حميد، قال قدم علينا…٤٦١٩{ كذلك نسلكه في قلوب المجرمين } قال الشرك .٤٦٢٠في قول الله عز وجل { وحيل بينهم وبين ما يشتهون } قال بينهم وبين…٤٦٢١حدثنا محمد بن عبيد، حدثنا سليم، عن ابن عون، قال كنت أسير بالشام…٤٦٢٢يقول كذب على الحسن ضربان من الناس قوم القدر رأيهم وهم يريدون أن…٤٦٢٣أن يحيى بن كثير العنبري، حدثهم قال كان قرة بن خالد يقول لنا يا فتيان…٤٦٢٤وابن، بشار قالا حدثنا مؤمل بن إسماعيل، حدثنا حماد بن زيد، عن ابن عون،…٤٦٢٥إلى شىء منه أبدا .٤٦٢٦قال ما فسر الحسن آية قط إلا على الإثبات .٤٦٢٧حدثنا عثمان بن أبي شيبة، حدثنا أسود بن عامر، حدثنا عبد العزيز بن أبي…٤٦٢٨حدثنا أحمد بن صالح، حدثنا عنبسة، حدثنا يونس، عن ابن شهاب، قال قال…٤٦٢٩حدثنا محمد بن كثير، حدثنا سفيان، حدثنا جامع بن أبي راشد، حدثنا أبو…٤٦٣٠يقول من زعم أن عليا، عليه السلام كان أحق بالولاية منهما فقد خطأ أبا…٤٦٣١يقول الخلفاء خمسة أبو بكر وعمر وعثمان وعلي وعمر بن عبد العزيز رضى…٤٦٣٢حدثنا محمد بن يحيى بن فارس، حدثنا عبد الرزاق، - قال محمد كتبته من…٤٦٣٣حدثنا محمد بن يحيى بن فارس، حدثنا محمد بن كثير، حدثنا سليمان بن كثير،…٤٦٣٤من رأى منكم رؤيا٤٦٣٥أن النبي صلى الله عليه وسلم قال ذات يوم " أيكم رأى رؤيا " . فذكر…٤٦٣٦أري الليلة رجل صالح أن أبا بكر نيط برسول الله صلى الله عليه وسلم ونيط…٤٦٣٧أن رجلا، قال يا رسول الله إني رأيت كأن دلوا دلي من السماء فجاء أبو…٤٦٣٨قال لتمخرن الروم الشام أربعين صباحا لا يمتنع منها إلا دمشق وعمان .٤٦٣٩أنه سمع أبا الأعيس عبد الرحمن بن سلمان، يقول سيأتي ملك من ملوك العجم…٤٦٤٠موضع فسطاط المسلمين في الملاحم أرض يقال لها الغوطة٤٦٤١حدثنا أبو ظفر عبد السلام، حدثنا جعفر، عن عوف، قال سمعت الحجاج، يخطب…٤٦٤٢يخطب فقال في خطبته رسول أحدكم في حاجته أكرم عليه أم خليفته في أهله…٤٦٤٣حدثنا محمد بن العلاء، حدثنا أبو بكر، عن عاصم، قال سمعت الحجاج، وهو…٤٦٤٤يقول على المنبر هذه الحمراء هبر هبر أما والله لقد قرعت عصا بعصا…٤٦٤٥قال جمعت مع الحجاج فخطب فذكر حديث أبي بكر بن عياش قال فيها فاسمعوا…٤٦٤٦خلافة النبوة ثلاثون سنة ثم يؤتي الله الملك - أو ملكه - من يشاء٤٦٤٧خلافة النبوة ثلاثون سنة ثم يؤتي الله الملك من يشاء - أو ملكه من يشاء -٤٦٤٨اثبت حراء إنه ليس عليك إلا نبي أو صديق أو شهيد٤٦٤٩عشرة في الجنة النبي في الجنة وأبو بكر في الجنة وعمر في الجنة وعثمان…٤٦٥٠أبو بكر في الجنة وعمر في الجنة٤٦٥١اثبت أحد نبي وصديق وشهيدان٤٦٥٢أتاني جبريل فأخذ بيدي فأراني باب الجنة الذي تدخل منه أمتي٤٦٥٣لا يدخل النار أحد ممن بايع تحت الشجرة٤٦٥٤قال قال رسول الله صلى الله عليه وسلم قال موسى " فلعل الله " . وقال…٤٦٥٥أن محمد بن ثور، حدثهم عن معمر، عن الزهري، عن عروة بن الزبير، عن…٤٦٥٦أن سعيد بن إياس الجريري، أخبرهم عن عبد الله بن شقيق العقيلي، عن…٤٦٥٧خير أمتي القرن الذين بعثت فيهم ثم الذين يلونهم ثم الذين يلونهم٤٦٥٨لا تسبوا أصحابي فوالذي نفسي بيده لو أنفق أحدكم مثل أحد ذهبا ما بلغ مد…٤٦٥٩أيما رجل من أمتي سببته سبة أو لعنته لعنة في غضبي - فإنما أنا من ولد…٤٦٦٠فأين أبو بكر يأبى الله ذلك والمسلمون يأبى الله ذلك والمسلمون٤٦٦١لا لا لا ليصل للناس ابن أبي قحافة٤٦٦٢إن ابني هذا سيد وإني أرجو أن يصلح الله به بين فئتين من أمتي٤٦٦٣لا تضرك الفتنة٤٦٦٤حدثنا عمرو بن مرزوق، أخبرنا شعبة، عن الأشعث بن سليم، عن أبي بردة، عن…٤٦٦٥بمعناه .٤٦٦٦حدثنا إسماعيل بن إبراهيم الهذلي، حدثنا ابن علية، عن يونس، عن الحسن،…٤٦٦٧تمرق مارقة عند فرقة من المسلمين يقتلها أولى الطائفتين بالحق٤٦٦٨لا تخيروا بين الأنبياء٤٦٦٩ما ينبغي لعبد أن يقول إني خير من يونس بن متى٤٦٧٠ما ينبغي لنبي أن يقول إني خير من يونس بن متى٤٦٧١لا تخيروني على موسى فإن الناس يصعقون فأكون أول من يفيق فإذا موسى باطش…٤٦٧٢قال قال رجل لرسول الله صلى الله عليه وسلم يا خير البرية . فقال رسول…٤٦٧٣أنا سيد ولد آدم وأول من تنشق عنه الأرض وأول شافع وأول مشفع٤٦٧٤ما أدري أتبع لعين هو أم لا وما أدري أعزير نبي هو أم لا٤٦٧٥أنا أولى الناس بابن مريم الأنبياء أولاد علات وليس بيني وبينه نبي٤٦٧٦الإيمان بضع وسبعون أفضلها قول لا إله إلا الله وأدناها إماطة العظم عن…٤٦٧٧أتدرون ما الإيمان بالله٤٦٧٨بين العبد وبين الكفر ترك الصلاة٤٦٧٩ما رأيت من ناقصات عقل ولا دين أغلب لذي لب منكن٤٦٨٠وعثمان بن أبي شيبة، - المعنى - قالا حدثنا وكيع، عن سفيان، عن سماك، عن…٤٦٨١من أحب لله وأبغض لله وأعطى لله ومنع لله فقد استكمل الإيمان٤٦٨٢أكمل المؤمنين إيمانا أحسنهم خلقا٤٦٨٣حدثنا محمد بن عبيد، حدثنا محمد بن ثور، عن معمر، قال وأخبرني الزهري،…٤٦٨٤حدثنا محمد بن عبيد، حدثنا ابن ثور، عن معمر، قال وقال الزهري { قل لم…٤٦٨٥أو مسلم إني لأعطي الرجل العطاء وغيره أحب إلى منه مخافة أن يكب على وجهه٤٦٨٦لا ترجعوا بعدي كفارا يضرب بعضكم رقاب بعض٤٦٨٧أيما رجل مسلم أكفر رجلا مسلما فإن كان كافرا وإلا كان هو الكافر٤٦٨٨أربع من كن فيه فهو منافق خالص ومن كانت فيه خلة منهن كان فيه خلة من…٤٦٨٩لا يزني الزاني حين يزني وهو مؤمن ولا يسرق حين يسرق وهو مؤمن ولا يشرب…٤٦٩٠إذا زنى الرجل خرج منه الإيمان كان عليه كالظلة فإذا انقطع رجع إليه…٤٦٩١القدرية مجوس هذه الأمة إن مرضوا فلا تعودوهم وإن ماتوا فلا تشهدوهم٤٦٩٢لكل أمة مجوس ومجوس هذه الأمة الذين يقولون لا قدر من مات منهم فلا…٤٦٩٣إن الله خلق آدم من قبضة قبضها من جميع الأرض فجاء بنو آدم على قدر…٤٦٩٤ما منكم من أحد ما من نفس منفوسة إلا قد كتب الله مكانها من النار أو من…٤٦٩٥الإسلام أن تشهد أن لا إله إلا الله وأن محمدا رسول الله وتقيم الصلاة…٤٦٩٦في شىء قد خلا ومضى٤٦٩٧إقام الصلاة وإيتاء الزكاة وحج البيت وصوم شهر رمضان والاغتسال من…٤٦٩٨وأبي، هريرة قالا كان رسول الله صلى الله عليه وسلم يجلس بين ظهرى…٤٦٩٩حدثنا محمد بن كثير، أخبرنا سفيان، عن أبي سنان، عن وهب بن خالد الحمصي،…٤٧٠٠إن أول ما خلق الله القلم فقال له اكتب . قال رب وماذا أكتب قال اكتب…٤٧٠١احتج آدم وموسى فقال موسى يا آدم أنت أبونا خيبتنا وأخرجتنا من الجنة .…٤٧٠٢إن موسى قال يا رب أرنا آدم الذي أخرجنا ونفسه من الجنة فأراه الله آدم…٤٧٠٣إن الله عز وجل خلق آدم ثم مسح ظهره بيمينه فاستخرج منه ذرية فقال خلقت…٤٧٠٤قال كنت عند عمر بن الخطاب بهذا الحديث وحديث مالك أتم .٤٧٠٥الغلام الذي قتله الخضر طبع كافرا ولو عاش لأرهق أبويه طغيانا وكفرا٤٧٠٦وكان طبع يوم طبع كافرا٤٧٠٧أبصر الخضر غلاما يلعب مع الصبيان فتناول رأسه فقلعه فقال موسى { أقتلت…٤٧٠٨إن خلق أحدكم يجمع في بطن أمه أربعين يوما ثم يكون علقة مثل ذلك ثم يكون…٤٧٠٩قال قيل لرسول الله صلى الله عليه وسلم يا رسول الله أعلم أهل الجنة من…٤٧١٠لا تجالسوا أهل القدر ولا تفاتحوهم٤٧١١الله أعلم بما كانوا عاملين٤٧١٢وكثير بن عبيد المذحجي، قالا حدثنا محمد بن حرب، - المعنى - عن محمد بن…٤٧١٣أوغير ذلك يا عائشة إن الله خلق الجنة وخلق لها أهلا وخلقها لهم وهم في…٤٧١٤كل مولود يولد على الفطرة فأبواه يهودانه وينصرانه كما تناتج الإبل من…٤٧١٥الله أعلم بما كانوا عاملين٤٧١٦كل مولود يولد على الفطرة٤٧١٧الوائدة والموءودة في النار٤٧١٨أن رجلا، قال يا رسول الله أين أبي قال " أبوك في النار " . فلما قفى…٤٧١٩إن الشيطان يجري من ابن آدم مجرى الدم٤٧٢٠لا تجالسوا أهل القدر ولا تفاتحوهم الحديث٤٧٢١لا يزال الناس يتساءلون حتى يقال هذا خلق الله الخلق فمن خلق الله فمن…٤٧٢٢فإذا قالوا ذلك فقولوا { الله أحد * الله الصمد * لم يلد ولم يولد * ولم…٤٧٢٣قال كنت في البطحاء في عصابة فيهم رسول الله صلى الله عليه وسلم فمرت…٤٧٢٤ومحمد بن سعيد، قالا أخبرنا عمرو بن أبي قيس، عن سماك، بإسناده ومعناه .٤٧٢٥بإسناده ومعنى هذا الحديث الطويل .٤٧٢٦ويحك أتدري ما تقول٤٧٢٧أذن لي أن أحدث عن ملك من ملائكة الله من حملة العرش إن ما بين شحمة…٤٧٢٨ومحمد بن يونس النسائي، - المعنى - قالا حدثنا عبد الله بن يزيد المقرئ،…٤٧٢٩إنكم سترون ربكم كما ترون هذا لا تضامون في رؤيته فإن استطعتم أن لا…٤٧٣٠هل تضارون في رؤية الشمس في الظهيرة ليست في سحابة٤٧٣١يا أبا رزين أليس كلكم يرى القمر٤٧٣٢يطوي الله السموات يوم القيامة ثم يأخذهن بيده اليمنى ثم يقول أنا الملك…٤٧٣٣ينزل ربنا كل ليلة إلى سماء الدنيا حين يبقى ثلث الليل الآخر فيقول من…٤٧٣٤ألا رجل يحملني إلى قومه فإن قريشا قد منعوني أن أبلغ كلام ربي٤٧٣٥وسعيد بن المسيب، وعلقمة بن وقاص، وعبيد الله بن عبد الله، عن حديث،…٤٧٣٦قال : كنت عند النجاشي فقرأ ابن له آية من الإنجيل فضحكت فقال : أتضحك…٤٧٣٧أعيذكما بكلمات الله التامة من كل شيطان وهامة ومن كل عين لامة٤٧٣٨إذا تكلم الله بالوحى سمع أهل السماء للسماء صلصلة كجر السلسلة على…٤٧٣٩شفاعتي لأهل الكبائر من أمتي٤٧٤٠يخرج قوم من النار بشفاعة محمد فيدخلون الجنة ويسمون الجهنميين٤٧٤١إن أهل الجنة يأكلون فيها ويشربون٤٧٤٢الصور قرن ينفخ فيه٤٧٤٣كل ابن آدم تأكل الأرض إلا عجب الذنب، منه خلق وفيه يركب٤٧٤٤لما خلق الله الجنة قال لجبريل : اذهب فانظر إليها . فذهب فنظر إليها ثم…٤٧٤٥إن أمامكم حوضا ما بين ناحيتيه كما بين جرباء وأذرح
