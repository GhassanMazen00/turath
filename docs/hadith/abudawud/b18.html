سنن أبي داود

كتاب الوصايا

٢٣ حديثًا
٢٨٦٢ما حق امرئ مسلم له شىء يوصي فيه يبيت ليلتين إلا ووصيته مكتوبة عنده٢٨٦٣ومحمد بن العلاء، قالا حدثنا أبو معاوية، عن الأعمش، عن أبي وائل، عن…٢٨٦٤وابن أبي خلف، قالا حدثنا سفيان، عن الزهري، عن عامر بن سعد، عن أبيه،…٢٨٦٥أن تصدق وأنت صحيح حريص تأمل البقاء وتخشى الفقر ولا تمهل حتى إذا بلغت…٢٨٦٦لأن يتصدق المرء في حياته بدرهم خير له من أن يتصدق بمائة عند موته٢٨٦٧إن الرجل ليعمل والمرأة بطاعة الله ستين سنة ثم يحضرهما الموت فيضاران…٢٨٦٨يا أبا ذر إني أراك ضعيفا وإني أحب لك ما أحب لنفسي فلا تأمرن على اثنين…٢٨٦٩{ إن ترك خيرا الوصية للوالدين والأقربين } فكانت الوصية كذلك حتى…٢٨٧٠إن الله قد أعطى كل ذي حق حقه فلا وصية لوارث٢٨٧١حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن عطاء، عن سعيد بن جبير، عن ابن…٢٨٧٢كل من مال يتيمك غير مسرف ولا مبادر ولا متأثل٢٨٧٣لا يتم بعد احتلام ولا صمات يوم إلى الليل٢٨٧٤اجتنبوا السبع الموبقات٢٨٧٥أنه حدثه - وكانت، له صحبة - أن رجلا، سأله فقال يا رسول الله ما…٢٨٧٦غطوا بها رأسه واجعلوا على رجليه من الإذخر٢٨٧٧قد وجب أجرك ورجعت إليك في الميراث٢٨٧٨إن شئت حبست أصلها وتصدقت بها٢٨٧٩عمر بن الخطاب رضى الله عنه قال نسخها لي عبد الحميد بن عبد الله بن عبد…٢٨٨٠إذا مات الإنسان انقطع عنه عمله إلا من ثلاثة أشياء من صدقة جارية أو…٢٨٨١نعم فتصدقي عنها٢٨٨٢أن رجلا، قال يا رسول الله إن أمي توفيت أفينفعها إن تصدقت عنها فقال "…٢٨٨٣إنه لو كان مسلما فأعتقتم عنه أو تصدقتم عنه أو حججتم عنه بلغه ذلك٢٨٨٤أن شعيب بن إسحاق، حدثهم عن هشام بن عروة، عن وهب بن كيسان، عن جابر بن…
