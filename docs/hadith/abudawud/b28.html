سنن أبي داود

كتاب الأطعمة

١١٩ حديثًا
٣٧٣٦إذا دعي أحدكم إلى الوليمة فليأتها٣٧٣٧فإن كان مفطرا فليطعم وإن كان صائما فليدع٣٧٣٨إذا دعا أحدكم أخاه فليجب عرسا كان أو نحوه٣٧٣٩بإسناد أيوب ومعناه .٣٧٤٠من دعي فليجب فإن شاء طعم وإن شاء ترك٣٧٤١من دعي فلم يجب فقد عصى الله ورسوله ومن دخل على غير دعوة دخل سارقا…٣٧٤٢أنه كان يقول شر الطعام طعام الوليمة يدعى لها الأغنياء ويترك المساكين…٣٧٤٣وقتيبة بن سعيد، قالا حدثنا حماد، عن ثابت، قال ذكر تزويج زينب بنت جحش…٣٧٤٤أن النبي صلى الله عليه وسلم أولم على صفية بسويق وتمر .٣٧٤٥الوليمة أول يوم حق والثاني معروف واليوم الثالث سمعة ورياء٣٧٤٦بهذه القصة قال فدعي اليوم الثالث فلم يجب وحصب الرسول .٣٧٤٧قال لما قدم النبي صلى الله عليه وسلم المدينة نحر جزورا أو بقرة .٣٧٤٨من كان يؤمن بالله واليوم الآخر فليكرم ضيفه جائزته يومه وليلته الضيافة…٣٧٤٩الضيافة ثلاثة أيام فما سوى ذلك فهو صدقة٣٧٥٠ليلة الضيف حق على كل مسلم فمن أصبح بفنائه فهو عليه دين إن شاء اقتضى…٣٧٥١أيما رجل أضاف قوما فأصبح الضيف محروما فإن نصره حق على كل مسلم حتى…٣٧٥٢إن نزلتم بقوم فأمروا لكم بما ينبغي للضيف فاقبلوا فإن لم يفعلوا فخذوا…٣٧٥٣حدثنا أحمد بن محمد المروزي، حدثني علي بن الحسين بن واقد، عن أبيه، عن…٣٧٥٤حدثنا هارون بن زيد بن أبي الزرقاء، حدثنا أبي، حدثنا جرير بن حازم، عن…٣٧٥٥إنه ليس لي أو لنبي أن يدخل بيتا مزوقا٣٧٥٦إذا اجتمع الداعيان فأجب أقربهما بابا فإن أقربهما بابا أقربهما جوارا…٣٧٥٧إذا وضع عشاء أحدكم وأقيمت الصلاة فلا يقوم حتى يفرغ٣٧٥٨لا تؤخر الصلاة لطعام ولا لغيره٣٧٥٩يبدأ بالعشاء قبل الصلاة . فقال عبد الله بن عمر ويحك ما كان عشاؤهم…٣٧٦٠إنما أمرت بالوضوء إذا قمت إلى الصلاة٣٧٦١بركة الطعام الوضوء قبله والوضوء بعده٣٧٦٢حدثنا أحمد بن أبي مريم، حدثنا عمي، - يعني سعيد بن الحكم - حدثنا الليث…٣٧٦٣قال ما عاب رسول الله صلى الله عليه وسلم طعاما قط إن اشتهاه أكله وإن…٣٧٦٤فلعلكم تفترقون٣٧٦٥إذا دخل الرجل بيته فذكر الله عند دخوله وعند طعامه قال الشيطان لا مبيت…٣٧٦٦إن الشيطان ليستحل الطعام الذي لم يذكر اسم الله عليه وإنه جاء بهذا…٣٧٦٧إذا أكل أحدكم فليذكر اسم الله تعالى فإن نسي أن يذكر اسم الله تعالى في…٣٧٦٨ما زال الشيطان يأكل معه فلما ذكر اسم الله عز وجل استقاء ما في بطنه٣٧٦٩قال قال رسول الله صلى الله عليه وسلم " لا آكل متكئا " .٣٧٧٠قال ما رئي رسول الله صلى الله عليه وسلم يأكل متكئا قط ولا يطأ عقبه…٣٧٧١يقول بعثني النبي صلى الله عليه وسلم فرجعت إليه فوجدته يأكل تمرا وهو…٣٧٧٢إذا أكل أحدكم طعاما فلا يأكل من أعلى الصحفة ولكن ليأكل من أسفلها فإن…٣٧٧٣إن الله جعلني عبدا كريما ولم يجعلني جبارا عنيدا٣٧٧٤حدثنا عثمان بن أبي شيبة، حدثنا كثير بن هشام، عن جعفر بن برقان، عن…٣٧٧٥بهذا الحديث .٣٧٧٦إذا أكل أحدكم فليأكل بيمينه وإذا شرب فليشرب بيمينه فإن الشيطان يأكل…٣٧٧٧ادن بنى فسم الله وكل بيمينك وكل مما يليك٣٧٧٨لا تقطعوا اللحم بالسكين فإنه من صنيع الأعاجم وانهسوه فإنه أهنأ وأمرأ٣٧٧٩أدن العظم من فيك فإنه أهنأ وأمرأ٣٧٨٠قال كان أحب العراق إلى رسول الله صلى الله عليه وسلم عراق الشاة .٣٧٨١بهذا الإسناد قال كان النبي صلى الله عليه وسلم يعجبه الذراع . قال وسم…٣٧٨٢أنه سمع أنس بن مالك، يقول إن خياطا دعا رسول الله صلى الله عليه وسلم…٣٧٨٣قال كان أحب الطعام إلى رسول الله صلى الله عليه وسلم الثريد من الخبز…٣٧٨٤لا يتخلجن في صدرك شىء ضارعت فيه النصرانية٣٧٨٥حدثنا عثمان بن أبي شيبة، حدثنا عبدة، عن محمد بن إسحاق، عن ابن أبي…٣٧٨٦حدثنا ابن المثنى، حدثني أبو عامر، حدثنا هشام، عن قتادة، عن عكرمة، عن…٣٧٨٧حدثنا أحمد بن أبي سريج، أخبرني عبد الله بن جهم، حدثنا عمرو بن أبي…٣٧٨٨حدثنا سليمان بن حرب، حدثنا حماد، عن عمرو بن دينار، عن محمد بن علي، عن…٣٧٨٩حدثنا موسى بن إسماعيل، حدثنا حماد، عن أبي الزبير، عن جابر بن عبد…٣٧٩٠وحيوة بن شريح الحمصي، قال حيوة حدثنا بقية، عن ثور بن يزيد، عن صالح بن…٣٧٩١قال كنت غلاما حزورا فصدت أرنبا فشويتها فبعث معي أبو طلحة بعجزها إلى…٣٧٩٢حدثنا يحيى بن خلف، حدثنا روح بن عبادة، حدثنا محمد بن خالد، قال سمعت…٣٧٩٣أن خالته، أهدت إلى رسول الله صلى الله عليه وسلم سمنا وأضبا وأقطا فأكل…٣٧٩٤لا ولكنه لم يكن بأرض قومي فأجدني أعافه٣٧٩٥إن أمة من بني إسرائيل مسخت دواب في الأرض وإني لا أدري أى الدواب هي٣٧٩٦أن الحكم بن نافع، حدثهم حدثنا ابن عياش، عن ضمضم بن زرعة، عن شريح بن…٣٧٩٧قال أكلت مع رسول الله صلى الله عليه وسلم لحم حبارى .٣٧٩٨قال صحبت النبي صلى الله عليه وسلم فلم أسمع لحشرة الأرض تحريما .٣٧٩٩خبيثة من الخبائث٣٨٠٠قال كان أهل الجاهلية يأكلون أشياء ويتركون أشياء تقذرا فبعث الله تعالى…٣٨٠١هو صيد ويجعل فيه كبش إذا صاده المحرم٣٨٠٢حدثنا القعنبي، عن مالك، عن ابن شهاب، عن أبي إدريس الخولاني، عن أبي…٣٨٠٣حدثنا مسدد، حدثنا أبو عوانة، عن أبي بشر، عن ميمون بن مهران، عن ابن…٣٨٠٤ألا لا يحل ذو ناب من السباع ولا الحمار الأهلي ولا اللقطة من مال معاهد…٣٨٠٥حدثنا محمد بن بشار، عن ابن أبي عدي، عن ابن أبي عروبة، عن علي بن…٣٨٠٦ألا لا تحل أموال المعاهدين إلا بحقها وحرام عليكم حمر الأهلية وخيلها…٣٨٠٧ومحمد بن عبد الملك، قالا حدثنا عبد الرزاق، عن عمر بن زيد الصنعاني،…٣٨٠٨حدثنا إبراهيم بن حسن المصيصي، حدثنا حجاج، عن ابن جريج، أخبرني عمرو بن…٣٨٠٩أطعم أهلك من سمين حمرك فإنما حرمتها من أجل جوال القرية٣٨١٠أحدهما عبد الله بن عمرو بن عويم والآخر غالب بن الأبجر . قال مسعر أرى…٣٨١١حدثنا سهل بن بكار، حدثنا وهيب، عن ابن طاوس، عن عمرو بن شعيب، عن أبيه،…٣٨١٢وسألته، عن الجراد، فقال غزوت مع رسول الله صلى الله عليه وسلم ست أو…٣٨١٣أكثر جنود الله لا آكله ولا أحرمه٣٨١٤وعلي بن عبد الله، قالا حدثنا زكرياء بن يحيى بن عمارة، عن أبي العوام…٣٨١٥ما ألقى البحر أو جزر عنه فكلوه وما مات فيه وطفا فلا تأكلوه٣٨١٦هل عندك غنى يغنيك٣٨١٧حدثنا هارون بن عبد الله، حدثنا الفضل بن دكين، حدثنا عقبة بن وهب بن…٣٨١٨وددت أن عندي خبزة بيضاء من برة سمراء ملبقة بسمن ولبن٣٨١٩قال أتي النبي صلى الله عليه وسلم بجبنة في تبوك فدعا بسكين فسمى وقطع .٣٨٢٠نعم الإدام الخل٣٨٢١نعم الإدام الخل٣٨٢٢من أكل ثوما أو بصلا فليعتزلنا - أو ليعتزل مسجدنا - وليقعد في بيته٣٨٢٣كلوه ومن أكله منكم فلا يقرب هذا المسجد حتى يذهب ريحه منه٣٨٢٤من تفل تجاه القبلة جاء يوم القيامة تفله بين عينيه ومن أكل من هذه…٣٨٢٥من أكل من هذه الشجرة فلا يقربن المساجد٣٨٢٦من أكل من هذه الشجرة فلا يقربنا حتى يذهب ريحها٣٨٢٧من أكلهما فلا يقربن مسجدنا٣٨٢٨إلا مطبوخا . قال أبو داود شريك بن حنبل .٣٨٢٩فقالت إن آخر طعام أكله رسول الله صلى الله عليه وسلم طعام فيه بصل .٣٨٣٠قال رأيت النبي صلى الله عليه وسلم أخذ كسرة من خبز شعير فوضع عليها…٣٨٣١بيت لا تمر فيه جياع أهله٣٨٣٢قال أتي النبي صلى الله عليه وسلم بتمر عتيق فجعل يفتشه يخرج السوس منه .٣٨٣٣أن النبي صلى الله عليه وسلم كان يؤتى بالتمر فيه دود فذكر معناه .٣٨٣٤حدثنا واصل بن عبد الأعلى، حدثنا ابن فضيل، عن أبي إسحاق، عن جبلة بن…٣٨٣٥أن النبي صلى الله عليه وسلم كان يأكل القثاء بالرطب .٣٨٣٦نكسر حر هذا ببرد هذا وبرد هذا بحر هذا٣٨٣٧حدثنا محمد بن الوزير، حدثنا الوليد بن مزيد، قال سمعت ابن جابر، قال…٣٨٣٨وإسماعيل، عن برد بن سنان، عن عطاء، عن جابر، قال كنا نغزو مع رسول الله…٣٨٣٩إن وجدتم غيرها فكلوا فيها واشربوا وإن لم تجدوا غيرها فارحضوها بالماء…٣٨٤٠هو رزق أخرجه الله لكم فهل معكم من لحمه شىء فتطعمونا منه٣٨٤١ألقوا ما حولها وكلوا٣٨٤٢إذا وقعت الفأرة في السمن فإن كان جامدا فألقوها وما حولها وإن كان…٣٨٤٣حدثنا أحمد بن صالح، حدثنا عبد الرزاق، أخبرنا عبد الرحمن بن بوذويه، عن…٣٨٤٤إذا وقع الذباب في إناء أحدكم فامقلوه فإن في أحد جناحيه داء وفي الآخر…٣٨٤٥إذا سقطت لقمة أحدكم فليمط عنها الأذى وليأكلها ولا يدعها للشيطان٣٨٤٦إذا صنع لأحدكم خادمه طعاما ثم جاءه به وقد ولي حره ودخانه فليقعده معه…٣٨٤٧إذا أكل أحدكم فلا يمسحن يده بالمنديل حتى يلعقها أو يلعقها٣٨٤٨أن النبي صلى الله عليه وسلم كان يأكل بثلاث أصابع ولا يمسح يده حتى…٣٨٤٩الحمد لله كثيرا طيبا مباركا فيه غير مكفي ولا مودع ولا مستغنى عنه ربنا٣٨٥٠الحمد لله الذي أطعمنا وسقانا وجعلنا مسلمين٣٨٥١الحمد لله الذي أطعم وسقى وسوغه وجعل له مخرجا٣٨٥٢من نام وفي يده غمر ولم يغسله فأصابه شىء فلا يلومن إلا نفسه٣٨٥٣قال صنع أبو الهيثم بن التيهان للنبي صلى الله عليه وسلم طعاما فدعا…٣٨٥٤أفطر عندكم الصائمون وأكل طعامكم الأبرار وصلت عليكم الملائكة
