سنن أبي داود

كتاب الطلاق

١٣٨ حديثًا
٢١٧٥ليس منا من خبب امرأة على زوجها أو عبدا على سيده٢١٧٦لا تسأل المرأة طلاق أختها لتستفرغ صحفتها ولتنكح فإنما لها ما قدر لها٢١٧٧ما أحل الله شيئا أبغض إليه من الطلاق٢١٧٨أبغض الحلال إلى الله تعالى الطلاق٢١٧٩مره فليراجعها ثم ليمسكها حتى تطهر ثم تحيض ثم تطهر ثم إن شاء أمسك بعد…٢١٨٠أن ابن عمر، طلق امرأة له وهي حائض تطليقة بمعنى حديث مالك .٢١٨١مره فليراجعها ثم ليطلقها إذا طهرت أو وهي حامل٢١٨٢مره فليراجعها ثم ليمسكها حتى تطهر ثم تحيض فتطهر ثم إن شاء طلقها طاهرا…٢١٨٣أنه سأل ابن عمر فقال كم طلقت امرأتك فقال واحدة .٢١٨٤مره فليراجعها ثم ليطلقها في قبل عدتها٢١٨٥إذا طهرت فليطلق أو ليمسك٢١٨٦أن جعفر بن سليمان، حدثهم عن يزيد الرشك، عن مطرف بن عبد الله، أن عمران…٢١٨٧أن عمر بن معتب، أخبره أن أبا حسن مولى بني نوفل أخبره أنه، استفتى ابن…٢١٨٨حدثنا محمد بن المثنى، حدثنا عثمان بن عمر، أخبرنا علي، بإسناده ومعناه…٢١٨٩طلاق الأمة تطليقتان وقرؤها حيضتان٢١٩٠لا طلاق إلا فيما تملك ولا عتق إلا فيما تملك ولا بيع إلا فيما تملك٢١٩١من حلف على معصية فلا يمين له ومن حلف على قطيعة رحم فلا يمين له٢١٩٢ولا نذر إلا فيما ابتغي به وجه الله تعالى ذكره٢١٩٣لا طلاق ولا عتاق في غلاق٢١٩٤ثلاث جدهن جد وهزلهن جد النكاح والطلاق والرجعة٢١٩٥قال { والمطلقات يتربصن بأنفسهن ثلاثة قروء ولا يحل لهن أن يكتمن ما خلق…٢١٩٦أترون فلانا يشبه منه كذا وكذا من عبد يزيد وفلانا يشبه منه - كذا وكذا٢١٩٧حدثنا حميد بن مسعدة، حدثنا إسماعيل، أخبرنا أيوب، عن عبد الله بن كثير،…٢١٩٨ومحمد بن يحيى، - وهذا حديث أحمد - قالا حدثنا عبد الرزاق، عن معمر، عن…٢١٩٩واحد، عن طاوس، أن رجلا، يقال له أبو الصهباء كان كثير السؤال لابن عباس…٢٢٠٠أن أبا الصهباء، قال لابن عباس أتعلم إنما كانت الثلاث تجعل واحدة على…٢٢٠١إنما الأعمال بالنيات وإنما لكل امرئ ما نوى فمن كانت هجرته إلى الله…٢٢٠٢وسليمان بن داود، قالا أخبرنا ابن وهب، أخبرني يونس، عن ابن شهاب،…٢٢٠٣حدثنا مسدد، حدثنا أبو عوانة، عن الأعمش، عن أبي الضحى، عن مسروق، عن…٢٢٠٤حدثنا الحسن بن علي، حدثنا سليمان بن حرب، عن حماد بن زيد، قال قلت…٢٢٠٥في أمرك بيدك . قال ثلاث .٢٢٠٦والله ما أردت إلا واحدة٢٢٠٧أن عبد الله بن الزبير، حدثهم عن محمد بن إدريس، حدثني عمي، محمد بن علي…٢٢٠٨حدثنا سليمان بن داود العتكي، حدثنا جرير بن حازم، عن الزبير بن سعيد،…٢٢٠٩إن الله تجاوز لأمتي عما لم تتكلم به أو تعمل به وبما حدثت به أنفسها٢٢١٠وخالد الطحان، - المعنى - كلهم عن خالد، عن أبي تميمة الهجيمي، أن رجلا،…٢٢١١من قومه أنه سمع النبي صلى الله عليه وسلم سمع رجلا، يقول لامرأته " يا…٢٢١٢أن إبراهيم صلى الله عليه وسلم لم يكذب قط إلا ثلاثا ثنتان في ذات الله…٢٢١٣أنت بذاك يا سلمة٢٢١٤اتقي الله فإنه ابن عمك٢٢١٥بهذا الإسناد نحوه إلا أنه قال والعرق مكتل يسع ثلاثين صاعا . قال أبو…٢٢١٦قال يعني بالعرق زنبيلا يأخذ خمسة عشر صاعا .٢٢١٧وعمرو بن الحارث، عن بكير بن الأشج، عن سليمان بن يسار، بهذا الخبر قال…٢٢١٨قال أبو داود قرأت على محمد بن وزير المصري قلت له حدثكم بشر بن بكر،…٢٢١٩أن جميلة، كانت تحت أوس بن الصامت وكان رجلا به لمم فكان إذا اشتد لممه…٢٢٢٠مثله .٢٢٢١ما حملك على ما صنعت٢٢٢٢أن رجلا، ظاهر من امرأته فرأى بريق ساقها في القمر فوقع عليها فأتى…٢٢٢٣حدثنا زياد بن أيوب، حدثنا إسماعيل، حدثنا الحكم بن أبان، عن عكرمة، عن…٢٢٢٤أن عبد العزيز بن المختار، حدثهم حدثنا خالد، حدثني محدث، عن عكرمة، عن…٢٢٢٥قال أبو داود سمعت محمد بن عيسى، يحدث به حدثنا المعتمر، قال سمعت الحكم…٢٢٢٦أيما امرأة سألت زوجها طلاقا في غير ما بأس فحرام عليها رائحة الجنة٢٢٢٧حدثنا القعنبي، عن مالك، عن يحيى بن سعيد، عن عمرة بنت عبد الرحمن بن…٢٢٢٨خذ بعض مالها وفارقها٢٢٢٩أن امرأة، ثابت بن قيس اختلعت منه فجعل النبي صلى الله عليه وسلم عدتها…٢٢٣٠قال عدة المختلعة حيضة .٢٢٣١يا بريرة اتقي الله فإنه زوجك وأبو ولدك٢٢٣٢أن زوج، بريرة كان عبدا أسود يسمى مغيثا فخيرها - يعني النبي صلى الله…٢٢٣٣في قصة بريرة قالت كان زوجها عبدا فخيرها رسول الله صلى الله عليه وسلم…٢٢٣٤والوليد بن عقبة، عن زائدة، عن سماك، عن عبد الرحمن بن القاسم، عن أبيه،…٢٢٣٥أن زوج، بريرة كان حرا حين أعتقت وأنها خيرت فقالت ما أحب أن أكون معه…٢٢٣٦إن قربك فلا خيار لك٢٢٣٧ونصر بن علي، قال زهير حدثنا عبيد الله بن عبد المجيد، حدثنا عبيد الله…٢٢٣٨أن رجلا، جاء مسلما على عهد رسول الله صلى الله عليه وسلم ثم جاءت…٢٢٣٩قال أسلمت امرأة على عهد رسول الله صلى الله عليه وسلم فتزوجت فجاء…٢٢٤٠قال رد رسول الله صلى الله عليه وسلم ابنته زينب على أبي العاصي بالنكاح…٢٢٤١اختر منهن أربعا٢٢٤٢بمعناه .٢٢٤٣طلق أيتهما شئت٢٢٤٤رافع بن سنان أنه أسلم وأبت امرأته أن تسلم فأتت النبي صلى الله عليه…٢٢٤٥قد أنزل فيك وفي صاحبتك قرآن فاذهب فأت بها٢٢٤٦أمسك المرأة عندك حتى تلد٢٢٤٧حدثنا أحمد بن صالح، حدثنا ابن وهب، قال أخبرني يونس، عن ابن شهاب، عن…٢٢٤٨أبصروها فإن جاءت به أدعج العينين عظيم الأليتين فلا أراه إلا قد صدق…٢٢٤٩بهذا الخبر قال فكان يدعى - يعني الولد - لأمه .٢٢٥٠وغيره، عن ابن شهاب، عن سهل بن سعد، في هذا الخبر قال فطلقها ثلاث…٢٢٥١ووهب بن بيان، وأحمد بن عمرو بن السرح، وعمرو بن عثمان، قالوا حدثنا…٢٢٥٢في هذا الحديث وكانت حاملا فأنكر حملها فكان ابنها يدعى إليها ثم جرت…٢٢٥٣حدثنا عثمان بن أبي شيبة، حدثنا جرير، عن الأعمش، عن إبراهيم، عن علقمة،…٢٢٥٤البينة أو حد في ظهرك٢٢٥٥حدثنا مخلد بن خالد الشعيري، حدثنا سفيان، عن عاصم بن كليب، عن أبيه، عن…٢٢٥٦أبشر يا هلال قد جعل الله عز وجل لك فرجا ومخرجا٢٢٥٧حسابكما على الله أحدكما كاذب لا سبيل لك عليها٢٢٥٨الله يعلم أن أحدكما كاذب . فهل منكما تائب٢٢٥٩وألحق الولد بالمرأة٢٢٦٠قال جاء رجل إلى النبي صلى الله عليه وسلم من بني فزارة فقال إن امرأتي…٢٢٦١بإسناده ومعناه قال وهو حينئذ يعرض بأن ينفيه .٢٢٦٢أن أعرابيا، أتى النبي صلى الله عليه وسلم فقال إن امرأتي ولدت غلاما…٢٢٦٣أيما امرأة أدخلت على قوم من ليس منهم فليست من الله في شىء ولن يدخلها…٢٢٦٤لا مساعاة في الإسلام من ساعى في الجاهلية فقد لحق بعصبته ومن ادعى ولدا…٢٢٦٥قال إن النبي صلى الله عليه وسلم قضى أن كل مستلحق استلحق بعد أبيه الذي…٢٢٦٦حدثنا محمود بن خالد، حدثنا أبي، عن محمد بن راشد، بإسناده ومعناه زاد…٢٢٦٧أى عائشة ألم ترى أن مجززا المدلجي رأى زيدا وأسامة قد غطيا رءوسهما…٢٢٦٨حدثنا قتيبة، حدثنا الليث، عن ابن شهاب، بإسناده ومعناه قال قالت دخل…٢٢٦٩قال كنت جالسا عند النبي صلى الله عليه وسلم فجاء رجل من اليمن فقال إن…٢٢٧٠قال أتي علي - رضى الله عنه - بثلاثة وهو باليمن وقعوا على امرأة في طهر…٢٢٧١سمع الشعبي، عن الخليل، أو ابن الخليل قال أتي علي بن أبي طالب - رضى…٢٢٧٢أن عائشة، - رضى الله عنها - زوج النبي صلى الله عليه وسلم أخبرته أن…٢٢٧٣الولد للفراش وللعاهر الحجر واحتجبي عنه يا سودة٢٢٧٤لا دعوة في الإسلام ذهب أمر الجاهلية الولد للفراش وللعاهر الحجر٢٢٧٥حدثنا موسى بن إسماعيل، حدثنا مهدي بن ميمون أبو يحيى، حدثنا محمد بن…٢٢٧٦أنت أحق به ما لم تنكحي٢٢٧٧أن أبا ميمونة، سلمى - مولى من أهل المدينة رجل صدق - قال بينما أنا…٢٢٧٨وأما الجارية فأقضي بها لجعفر تكون مع خالتها وإنما الخالة أم٢٢٧٩إن خالتها عنده٢٢٨٠الخالة بمنزلة الأم٢٢٨١أنها طلقت على عهد رسول الله صلى الله عليه وسلم ولم يكن للمطلقة عدة…٢٢٨٢قال { والمطلقات يتربصن بأنفسهن ثلاثة قروء } . وقال { واللائي يئسن من…٢٢٨٣أن رسول الله صلى الله عليه وسلم طلق حفصة ثم راجعها .٢٢٨٤ليس لك عليه نفقة٢٢٨٥أن فاطمة بنت قيس، حدثته أن أبا حفص بن المغيرة طلقها ثلاثا وساق الحديث…٢٢٨٦ليست لها نفقة ولا مسكن٢٢٨٧ولا تفوتيني بنفسك٢٢٨٨أن زوجها، طلقها ثلاثا فلم يجعل لها النبي صلى الله عليه وسلم نفقة ولا…٢٢٨٩أنها أخبرته أنها، كانت عند أبي حفص بن المغيرة وأن أبا حفص بن المغيرة…٢٢٩٠لا نفقة لك إلا أن تكوني حاملا٢٢٩١حدثنا نصر بن علي، أخبرني أبو أحمد، حدثنا عمار بن رزيق، عن أبي إسحاق،…٢٢٩٢قال لقد عابت ذلك عائشة - رضى الله عنها - أشد العيب يعني حديث فاطمة…٢٢٩٣أنه قيل لعائشة ألم ترى إلى قول فاطمة قالت أما إنه لا خير لها في ذكر…٢٢٩٤في خروج فاطمة قال إنما كان ذلك من سوء الخلق .٢٢٩٥وسليمان بن يسار، أنه سمعهما يذكران، أن يحيى بن سعيد بن العاص، طلق بنت…٢٢٩٦قال قدمت المدينة فدفعت إلى سعيد بن المسيب فقلت فاطمة بنت قيس طلقت…٢٢٩٧اخرجي فجدي نخلك لعلك أن تصدقي منه أو تفعلي خيرا٢٢٩٨{ والذين يتوفون منكم ويذرون أزواجا وصية لأزواجهم متاعا إلى الحول غير…٢٢٩٩لا يحل لامرأة تؤمن بالله واليوم الآخر أن تحد على ميت فوق ثلاث ليال…٢٣٠٠زينب بنت كعب بن عجرة أن الفريعة بنت مالك بن سنان، - وهي أخت أبي سعيد…٢٣٠١قال قال عطاء قال ابن عباس نسخت هذه الآية عدتها عند أهله فتعتد حيث…٢٣٠٢لا تحد المرأة فوق ثلاث إلا على زوج فإنها تحد عليه أربعة أشهر وعشرا لا…٢٣٠٣ومالك بن عبد الواحد المسمعي، قالا حدثنا يزيد بن هارون، عن هشام، عن…٢٣٠٤حدثنا زهير بن حرب، حدثنا يحيى بن أبي بكير، حدثنا إبراهيم بن طهمان،…٢٣٠٥ما هذا يا أم سلمة٢٣٠٦أن أباه، كتب إلى عمر بن عبد الله بن الأرقم الزهري يأمره أن يدخل، على…٢٣٠٧ومحمد بن العلاء، - قال عثمان حدثنا وقال ابن العلاء، أخبرنا - أبو…٢٣٠٨أن محمد بن جعفر، حدثهم ح، وحدثنا ابن المثنى، حدثنا عبد الأعلى، عن…٢٣٠٩لا تحل للأول حتى تذوق عسيلة الآخر ويذوق عسيلتها٢٣١٠أن تجعل لله ندا وهو خلقك٢٣١١أنه سمع جابر بن عبد الله، يقول جاءت مسيكة لبعض الأنصار فقالت إن سيدي…٢٣١٢{ ومن يكرههن فإن الله من بعد إكراههن غفور رحيم } قال قال سعيد بن أبي…
