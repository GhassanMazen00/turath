سنن أبي داود

كتاب الأدب

٥٠٢ حديثًا•صفحة ٢ من ٤
٤٩٢٣قال لما قدم رسول الله صلى الله عليه وسلم المدينة لعبت الحبشة لقدومه…٤٩٢٤قال سمع ابن عمر، مزمارا - قال - فوضع أصبعيه على أذنيه ونأى عن الطريق…٤٩٢٥قال كنت ردف ابن عمر إذ مر براع يزمر فذكر نحوه . قال أبو داود أدخل بين…٤٩٢٦زامر فذكر نحوه . قال أبو داود وهذا أنكرها .٤٩٢٧الغناء ينبت النفاق في القلب٤٩٢٨ومحمد بن العلاء، أن أبا أسامة، أخبرهم عن مفضل بن يونس، عن الأوزاعي،…٤٩٢٩أخرجوهم من بيوتكم٤٩٣٠أخرجوهم من بيوتكم وأخرجوا فلانا وفلانا٤٩٣١قالت كنت ألعب بالبنات فربما دخل على رسول الله صلى الله عليه وسلم…٤٩٣٢أن محمد بن إبراهيم، حدثه عن أبي سلمة بن عبد الرحمن، عن عائشة، رضى…٤٩٣٣حدثنا موسى بن إسماعيل، حدثنا حماد، ح وحدثنا بشر بن خالد، حدثنا أبو…٤٩٣٤مثله قال على خير طائر فسلمتني إليهن فغسلن رأسي وأصلحنني فلم يرعني إلا…٤٩٣٥رضى الله عنها، قالت فلما قدمنا المدينة جاءني نسوة وأنا ألعب على…٤٩٣٦ومعي، صواحباتي فأدخلنني بيتا فإذا نسوة من الأنصار فقلن على الخير…٤٩٣٧حدثنا عبيد الله بن معاذ، حدثنا أبي، حدثنا محمد، - يعني ابن عمرو - عن…٤٩٣٨من لعب بالنرد فقد عصى الله ورسوله٤٩٣٩من لعب بالنردشير فكأنما غمس يده في لحم خنزير ودمه٤٩٤٠شيطان يتبع شيطانة٤٩٤١الراحمون يرحمهم الرحمن ارحموا أهل الأرض يرحمكم من في السماء٤٩٤٢لا تنزع الرحمة إلا من شقي٤٩٤٣من لم يرحم صغيرنا ويعرف حق كبيرنا فليس منا٤٩٤٤إن الدين النصيحة إن الدين النصيحة إن الدين النصيحة٤٩٤٥أما إن الذي أخذنا منك أحب إلينا مما أعطيناك فاختر٤٩٤٦من نفس عن مسلم كربة من كرب الدنيا نفس الله عنه كربة من كرب يوم…٤٩٤٧قال قال نبيكم صلى الله عليه وسلم " كل معروف صدقة " .٤٩٤٨إنكم تدعون يوم القيامة بأسمائكم وأسماء آبائكم فأحسنوا أسماءكم٤٩٤٩أحب الأسماء إلى الله تعالى عبد الله وعبد الرحمن٤٩٥٠تسموا بأسماء الأنبياء وأحب الأسماء إلى الله عبد الله وعبد الرحمن…٤٩٥١قال ذهبت بعبد الله بن أبي طلحة إلى النبي صلى الله عليه وسلم حين ولد…٤٩٥٢ومسدد، قالا حدثنا يحيى، عن عبيد الله، عن نافع، عن ابن عمر، أن رسول…٤٩٥٣لا تزكوا أنفسكم الله أعلم بأهل البر منكم٤٩٥٤أسامة بن أخدري أن رجلا، يقال له أصرم كان في النفر الذين أتوا رسول…٤٩٥٥إن الله هو الحكم وإليه الحكم فلم تكنى أبا الحكم٤٩٥٦حدثنا أحمد بن صالح، حدثنا عبد الرزاق، عن معمر، عن الزهري، عن سعيد بن…٤٩٥٧حدثنا أبو بكر بن أبي شيبة، حدثنا هاشم بن القاسم، حدثنا أبو عقيل،…٤٩٥٨لا تسمين غلامك يسارا ولا رباحا ولا نجيحا ولا أفلح فإنك تقول أثم هو…٤٩٥٩حدثنا أحمد بن حنبل، حدثنا المعتمر، قال سمعت الركين، يحدث عن أبيه، عن…٤٩٦٠إن عشت إن شاء الله أنهى أمتي أن يسموا نافعا وأفلح وبركة٤٩٦١أخنع اسم عند الله تبارك وتعالى يوم القيامة رجل تسمى ملك الأملاك٤٩٦٢حدثنا موسى بن إسماعيل، حدثنا وهيب، عن داود، عن عامر، قال حدثني أبو…٤٩٦٣أن عمر بن الخطاب، رضى الله عنه ضرب ابنا له تكنى أبا عيسى وأن المغيرة…٤٩٦٤ومحمد بن محبوب، قالوا حدثنا أبو عوانة، عن أبي عثمان، - وسماه ابن…٤٩٦٥تسموا باسمي ولا تكتنوا بكنيتي٤٩٦٦من تسمى باسمي فلا يكتني بكنيتي ومن تكنى بكنيتي فلا يتسمى باسمي٤٩٦٧قال قال علي رحمه الله قلت يا رسول الله إن ولد لي من بعدك ولد أسميه…٤٩٦٨ما الذي أحل اسمي وحرم كنيتي٤٩٦٩حدثنا موسى بن إسماعيل، حدثنا حماد، حدثنا ثابت، عن أنس بن مالك، قال…٤٩٧٠فاكتني بابنك عبد الله٤٩٧١كبرت خيانة أن تحدث أخاك حديثا هو لك به مصدق وأنت له به كاذب٤٩٧٢حدثنا أبو بكر بن أبي شيبة، حدثنا وكيع، عن الأوزاعي، عن يحيى، عن أبي…٤٩٧٣أن النبي صلى الله عليه وسلم خطبهم فقال " أما بعد " .٤٩٧٤لا يقولن أحدكم الكرم فإن الكرم الرجل المسلم ولكن قولوا حدائق الأعناب٤٩٧٥لا يقولن أحدكم عبدي وأمتي ولا يقولن المملوك ربي وربتي وليقل المالك…٤٩٧٦وليقل سيدي ومولاى٤٩٧٧لا تقولوا للمنافق سيد فإنه إن يك سيدا فقد أسخطتم ربكم عز وجل٤٩٧٨لا يقولن أحدكم خبثت نفسي وليقل لقست نفسي٤٩٧٩لا يقولن أحدكم جاشت نفسي ولكن ليقل لقست نفسي٤٩٨٠لا تقولوا ما شاء الله وشاء فلان ولكن قولوا ما شاء الله ثم شاء فلان٤٩٨١أن خطيبا، خطب عند النبي صلى الله عليه وسلم فقال من يطع الله ورسوله…٤٩٨٢لا تقل تعس الشيطان فإنك إذا قلت ذلك تعاظم حتى يكون مثل البيت ويقول…٤٩٨٣حدثنا القعنبي، عن مالك، ح وحدثنا موسى بن إسماعيل، حدثنا حماد، عن سهيل…٤٩٨٤لا تغلبنكم الأعراب على اسم صلاتكم ألا وإنها العشاء ولكنهم يعتمون…٤٩٨٥يا بلال أقم الصلاة أرحنا بها٤٩٨٦قم يا بلال أقم فأرحنا بالصلاة٤٩٨٧رضى الله عنها، قالت ما سمعت رسول الله صلى الله عليه وسلم ينسب أحدا…٤٩٨٨حدثنا عمرو بن مرزوق، أخبرنا شعبة، عن قتادة، عن أنس، قال كان فزع…٤٩٨٩إياكم والكذب فإن الكذب يهدي إلى الفجور وإن الفجور يهدي إلى النار وإن…٤٩٩٠ويل للذي يحدث فيكذب ليضحك به القوم ويل له ويل له٤٩٩١وما أردت أن تعطيه٤٩٩٢كفى بالمرء إثما أن يحدث بكل ما سمع٤٩٩٣حسن الظن من حسن العبادة٤٩٩٤على رسلكما إنها صفية بنت حيى٤٩٩٥إذا وعد الرجل أخاه - ومن نيته أن يفي له - فلم يف ولم يجئ للميعاد فلا…٤٩٩٦يا فتى لقد شققت على أنا ها هنا منذ ثلاث أنتظرك٤٩٩٧المتشبع بما لم يعط كلابس ثوبى زور٤٩٩٨إنا حاملوك على ولد ناقة٤٩٩٩كيف رأيتني أنقذتك من الرجل٥٠٠٠قال أتيت رسول الله صلى الله عليه وسلم في غزوة تبوك وهو في قبة من أدم…٥٠٠١قال إنما قال أدخل كلي . من صغر القبة .٥٠٠٢قال قال لي رسول الله صلى الله عليه وسلم " يا ذا الأذنين " .٥٠٠٣لا يأخذن أحدكم متاع أخيه لاعبا ولا جادا٥٠٠٤لا يحل لمسلم أن يروع مسلما٥٠٠٥إن الله عز وجل يبغض البليغ من الرجال الذي يتخلل بلسانه تخلل الباقرة…٥٠٠٦من تعلم صرف الكلام ليسبي به قلوب الرجال أو الناس لم يقبل الله منه يوم…٥٠٠٧إن من البيان لسحرا٥٠٠٨لقد رأيت أو أمرت أن أتجوز في القول فإن الجواز هو خير٥٠٠٩لأن يمتلئ جوف أحدكم قيحا خير له من أن يمتلئ شعرا٥٠١٠إن من الشعر حكمة٥٠١١إن من البيان سحرا وإن من الشعر حكما٥٠١٢إن من البيان سحرا وإن من العلم جهلا وإن من الشعر حكما وإن من القول…٥٠١٣وأحمد بن عبدة، - المعنى - قالا حدثنا سفيان بن عيينة، عن الزهري، عن…٥٠١٤بمعناه زاد فخشي أن يرميه، برسول الله صلى الله عليه وسلم فأجازه .٥٠١٥إن روح القدس مع حسان ما نافح عن رسول الله صلى الله عليه وسلم٥٠١٦قال { والشعراء يتبعهم الغاوون } فنسخ من ذلك واستثنى فقال { إلا الذين…٥٠١٧هل رأى أحد منكم الليلة رؤيا ويقول٥٠١٨رؤيا المؤمن جزء من ستة وأربعين جزءا من النبوة٥٠١٩إذا اقترب الزمان لم تكد رؤيا المؤمن أن تكذب وأصدقهم رؤيا أصدقهم حديثا…٥٠٢٠الرؤيا على رجل طائر ما لم تعبر فإذا عبرت وقعت٥٠٢١الرؤيا من الله والحلم من الشيطان فإذا رأى أحدكم شيئا يكرهه فلينفث عن…٥٠٢٢إذا رأى أحدكم الرؤيا يكرهها فليبصق عن يساره وليتعوذ بالله من الشيطان…٥٠٢٣من رآني في المنام فسيراني في اليقظة٥٠٢٤من صور صورة عذبه الله بها يوم القيامة حتى ينفخ فيها وليس بنافخ ومن…٥٠٢٥رأيت الليلة كأنا في دار عقبة بن رافع وأتينا برطب من رطب ابن طاب فأولت…٥٠٢٦إذا تثاءب أحدكم فليمسك على فيه فإن الشيطان يدخل٥٠٢٧في الصلاة فليكظم ما استطاع٥٠٢٨إن الله يحب العطاس ويكره التثاؤب فإذا تثاءب أحدكم فليرده ما استطاع…٥٠٢٩قال كان رسول الله صلى الله عليه وسلم إذا عطس وضع يده أو ثوبه على فيه…٥٠٣٠خمس تجب للمسلم على أخيه رد السلام وتشميت العاطس وإجابة الدعوة وعيادة…٥٠٣١وعليك وعلى أمك٥٠٣٢ورقاء، عن منصور، عن هلال بن يساف، عن خالد بن عرفجة، عن سالم بن عبيد…٥٠٣٣إذا عطس أحدكم فليقل الحمد لله على كل حال وليقل أخوه أو صاحبه يرحمك…٥٠٣٤قال شمت أخاك ثلاثا فما زاد فهو زكام .٥٠٣٥حدثنا عيسى بن حماد المصري، أخبرنا الليث، عن ابن عجلان، عن سعيد بن أبي…٥٠٣٦تشمت العاطس ثلاثا فإن شئت أن تشمته فشمته وإن شئت فكف٥٠٣٧أن رجلا، عطس عند النبي صلى الله عليه وسلم فقال له " يرحمك الله " . ثم…٥٠٣٨يهديكم الله ويصلح بالكم٥٠٣٩إن هذا حمد الله وإن هذا لم يحمد الله٥٠٤٠انطلقوا بنا إلى بيت عائشة٥٠٤١من بات على ظهر بيت ليس له حجار فقد برئت منه الذمة٥٠٤٢ما من مسلم يبيت على ذكر طاهرا فيتعار من الليل فيسأل الله خيرا من…٥٠٤٣أن رسول الله صلى الله عليه وسلم قام من الليل فقضى حاجته فغسل وجهه…٥٠٤٤آل أم سلمة قال كان فراش النبي صلى الله عليه وسلم نحوا مما يوضع…٥٠٤٥اللهم قني عذابك يوم تبعث عبادك٥٠٤٦إذا أتيت مضجعك فتوضأ وضوءك للصلاة ثم اضطجع على شقك الأيمن وقل اللهم…٥٠٤٧إذا أويت إلى فراشك وأنت طاهر فتوسد يمينك٥٠٤٨إذا أتيت فراشك طاهرا٥٠٤٩اللهم باسمك أحيا وأموت٥٠٥٠إذا أوى أحدكم إلى فراشه فلينفض فراشه بداخلة إزاره فإنه لا يدري ما…٥٠٥١اللهم رب السموات ورب الأرض ورب كل شىء فالق الحب والنوى منزل التوراة…٥٠٥٢اللهم إني أعوذ بوجهك الكريم وكلماتك التامة من شر ما أنت آخذ بناصيته…٥٠٥٣الحمد لله الذي أطعمنا وسقانا وكفانا وآوانا فكم ممن لا كافي له ولا مئوي٥٠٥٤بسم الله وضعت جنبي اللهم اغفر لي ذنبي وأخسئ شيطاني وفك رهاني واجعلني…٥٠٥٥اقرأ { قل يا أيها الكافرون } ثم نم على خاتمتها فإنها براءة من الشرك٥٠٥٦ويزيد بن خالد بن موهب الهمداني، قالا حدثنا المفضل، - يعنيان ابن فضالة…٥٠٥٧إن فيهن آية أفضل من ألف آية٥٠٥٨الحمد لله الذي كفاني وآواني وأطعمني وسقاني والذي من على فأفضل والذي…٥٠٥٩من اضطجع مضجعا لم يذكر الله تعالى فيه إلا كان عليه ترة يوم القيامة…٥٠٦٠من تعار من الليل فقال حين يستيقظ لا إله إلا الله وحده لا شريك له له…٥٠٦١لا إله إلا أنت سبحانك اللهم أستغفرك لذنبي وأسألك رحمتك اللهم زدني…٥٠٦٢حدثنا حفص بن عمر، حدثنا شعبة، ح وحدثنا مسدد، حدثنا يحيى، عن شعبة، -…٥٠٦٣ما كان حاجتك أمس إلى آل محمد٥٠٦٤حدثنا عباس العنبري، حدثنا عبد الملك بن عمرو، حدثنا عبد العزيز بن…٥٠٦٥خصلتان أو خلتان لا يحافظ عليهما عبد مسلم إلا دخل الجنة هما يسير ومن…٥٠٦٦سبقكن يتامى بدر٥٠٦٧قل اللهم فاطر السموات والأرض عالم الغيب والشهادة رب كل شىء ومليكه…٥٠٦٨اللهم بك أصبحنا وبك أمسينا وبك نحيا وبك نموت وإليك النشور٥٠٦٩من قال حين يصبح أو يمسي اللهم إني أصبحت أشهدك وأشهد حملة عرشك…٥٠٧٠من قال حين يصبح أو حين يمسي اللهم أنت ربي لا إله إلا أنت خلقتني وأنا…٥٠٧١أمسينا وأمسى الملك لله والحمد لله لا إله إلا الله وحده لا شريك له٥٠٧٢من قال إذا أصبح وإذا أمسى رضينا بالله ربا وبالإسلام دينا وبمحمد رسولا…
