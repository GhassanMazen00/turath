كتاب الأشربة

٦٧ حديثًا
٣٦٦٩حدثنا أحمد بن حنبل، حدثنا إسماعيل بن إبراهيم، حدثنا أبو حيان، حدثني…٣٦٧٠حدثنا عباد بن موسى الختلي، أخبرنا إسماعيل، - يعني ابن جعفر - عن…٣٦٧١عليه السلام أن رجلا، من الأنصار دعاه وعبد الرحمن بن عوف فسقاهما قبل…٣٦٧٢حدثنا أحمد بن محمد المروزي، حدثنا علي بن حسين، عن أبيه، عن يزيد…٣٦٧٣حدثنا سليمان بن حرب، حدثنا حماد بن زيد، عن ثابت، عن أنس، قال كنت ساقي…٣٦٧٤لعن الله الخمر وشاربها وساقيها وبائعها ومبتاعها وعاصرها ومعتصرها…٣٦٧٥أن أبا طلحة، سأل النبي صلى الله عليه وسلم عن أيتام ورثوا خمرا قال "…٣٦٧٦إن من العنب خمرا وإن من التمر خمرا وإن من العسل خمرا وإن من البر خمرا…٣٦٧٧إن الخمر من العصير والزبيب والتمر والحنطة والشعير والذرة وإني أنهاكم…٣٦٧٨الخمر من هاتين الشجرتين النخلة والعنبة٣٦٧٩كل مسكر خمر وكل مسكر حرام ومن مات وهو يشرب الخمر يدمنها لم يشربها في…٣٦٨٠كل مخمر خمر وكل مسكر حرام ومن شرب مسكرا بخست صلاته أربعين صباحا فإن…٣٦٨١ما أسكر كثيره فقليله حرام٣٦٨٢كل شراب أسكر فهو حرام٣٦٨٣حدثنا هناد بن السري، حدثنا عبدة، عن محمد، - يعني ابن إسحاق - عن يزيد…٣٦٨٤حدثنا وهب بن بقية، عن خالد، عن عاصم بن كليب، عن أبي بردة، عن أبي…٣٦٨٥حدثنا موسى بن إسماعيل، حدثنا حماد، عن محمد بن إسحاق، عن يزيد بن أبي…٣٦٨٦حدثنا سعيد بن منصور، حدثنا أبو شهاب عبد ربه بن نافع، عن الحسن بن عمرو…٣٦٨٧كل مسكر حرام وما أسكر منه الفرق فملء الكف منه حرام٣٦٨٨ليشربن ناس من أمتي الخمر يسمونها بغير اسمها٣٦٨٩ليشربن ناس من أمتي الخمر يسمونها بغير اسمها٣٦٩٠وابن، عباس قالا نشهد أن رسول الله صلى الله عليه وسلم نهى عن الدباء…٣٦٩١ومسلم بن إبراهيم، - المعنى - قالا حدثنا جرير، عن يعلى، - يعني ابن…٣٦٩٢آمركم بأربع وأنهاكم عن أربع الإيمان بالله وشهادة أن لا إله إلا الله٣٦٩٣أنهاكم عن النقير والمقير والحنتم والدباء والمزادة المجبوبة ولكن اشرب…٣٦٩٤عليكم بأسقية الأدم التي يلاث على أفواهها٣٦٩٥لا تشربوا في نقير ولا مزفت ولا دباء ولا حنتم واشربوا في الجلد الموكإ…٣٦٩٦لا تشربوا في الدباء ولا في المزفت ولا في النقير وانتبذوا في الأسقية٣٦٩٧حدثنا مسدد، حدثنا عبد الواحد، حدثنا إسماعيل بن سميع، حدثنا مالك بن…٣٦٩٨نهيتكم عن ثلاث وأنا آمركم بهن نهيتكم عن زيارة القبور فزوروها فإن في…٣٦٩٩حدثنا مسدد، حدثنا يحيى، عن سفيان، حدثني منصور، عن سالم بن أبي الجعد،…٣٧٠٠حدثنا محمد بن جعفر بن زياد، حدثنا شريك، عن زياد بن فياض، عن أبي عياض،…٣٧٠١بإسناده قال " اجتنبوا ما أسكر " .٣٧٠٢قال كان ينبذ لرسول الله صلى الله عليه وسلم في سقاء فإذا لم يجدوا سقاء…٣٧٠٣حدثنا قتيبة بن سعيد، حدثنا الليث، عن عطاء بن أبي رباح، عن جابر بن عبد…٣٧٠٤انتبذوا كل واحد على حدة٣٧٠٥وحفص بن عمر النمري، قالا حدثنا شعبة، عن الحكم، عن ابن أبي ليلى، - عن…٣٧٠٦حدثتني ريطة، عن كبشة بنت أبي مريم، قالت سألت أم سلمة ما كان النبي صلى…٣٧٠٧رضى الله عنها أن رسول الله صلى الله عليه وسلم كان ينبذ له زبيب فيلقي…٣٧٠٨حدثتني صفية بنت عطية، قالت دخلت مع نسوة من عبد القيس على عائشة…٣٧٠٩وعكرمة، أنهما كانا يكرهان البسر وحده ويأخذان ذلك عن ابن عباس . وقال…٣٧١٠إلى الله وإلى رسوله٣٧١١- رضى الله عنها - قالت كان ينبذ لرسول الله صلى الله عليه وسلم في سقاء…٣٧١٢قال حدثتني عمتي، عمرة عن عائشة، رضى الله عنها أنها كانت تنبذ للنبي…٣٧١٣قال كان ينبذ للنبي صلى الله عليه وسلم الزبيب فيشربه اليوم والغد وبعد…٣٧١٤بل شربت عسلا عند زينب بنت جحش ولن أعود له٣٧١٥بل شربت عسلا سقتني حفصة٣٧١٦اضرب بهذا الحائط فإن هذا شراب من لا يؤمن بالله واليوم الآخر٣٧١٧أن رسول الله صلى الله عليه وسلم نهى أن يشرب الرجل قائما .٣٧١٨أن عليا، دعا بماء فشربه وهو قائم ثم قال إن رجالا يكره أحدهم أن يفعل…٣٧١٩حدثنا موسى بن إسماعيل، حدثنا حماد، أخبرنا قتادة، عن عكرمة، عن ابن…٣٧٢٠أنه سمع عبيد الله بن عبد الله، عن أبي سعيد الخدري، أن رسول الله صلى…٣٧٢١أن رسول الله صلى الله عليه وسلم دعا بإداوة يوم أحد فقال " اخنث فم…٣٧٢٢حدثنا أحمد بن صالح، حدثنا عبد الله بن وهب، أخبرني قرة بن عبد الرحمن،…٣٧٢٣هي لهم في الدنيا ولكم في الآخرة٣٧٢٤إن كان عندك ماء بات هذه الليلة في شن وإلا كرعنا٣٧٢٥ساقي القوم آخرهم شربا٣٧٢٦حدثنا القعنبي عبد الله بن مسلمة، عن مالك، عن ابن شهاب، عن أنس بن…٣٧٢٧هو أهنأ وأمرأ وأبرأ٣٧٢٨قال نهى رسول الله صلى الله عليه وسلم أن يتنفس في الإناء أو ينفخ فيه .٣٧٢٩اللهم بارك لهم فيما رزقتهم واغفر لهم وارحمهم٣٧٣٠حدثنا مسدد، حدثنا حماد، - يعني ابن زيد - ح وحدثنا موسى بن إسماعيل،…٣٧٣١أغلق بابك واذكر اسم الله فإن الشيطان لا يفتح بابا مغلقا وأطف مصباحك…٣٧٣٢فإن الشيطان لا يفتح بابا غلقا ولا يحل وكاء ولا يكشف إناء وإن الفويسقة…٣٧٣٣واكفتوا صبيانكم عند العشاء٣٧٣٤حدثنا عثمان بن أبي شيبة، حدثنا أبو معاوية، حدثنا الأعمش، عن أبي صالح،…٣٧٣٥وعبد الله بن محمد النفيلي، وقتيبة بن سعيد، قالوا حدثنا عبد العزيز بن…
