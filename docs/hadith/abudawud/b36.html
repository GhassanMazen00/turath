سنن أبي داود

كتاب الخاتم

٢٦ حديثًا
٤٢١٤محمد رسول الله٤٢١٥بمعنى حديث عيسى بن يونس زاد فكان في يده حتى قبض وفي يد أبي بكر حتى…٤٢١٦وأحمد بن صالح، قالا حدثنا ابن وهب، أخبرني يونس بن يزيد، عن ابن شهاب،…٤٢١٧قال كان خاتم النبي صلى الله عليه وسلم من فضة كله فصه منه .٤٢١٨محمد رسول الله٤٢١٩محمد رسول الله٤٢٢٠محمد رسول الله٤٢٢١حدثنا محمد بن سليمان، لوين عن إبراهيم بن سعد، عن ابن شهاب، عن أنس بن…٤٢٢٢أن ابن مسعود، كان يقول كان نبي الله صلى الله عليه وسلم يكره عشر خلال…٤٢٢٣ما لي أجد منك ريح الأصنام٤٢٢٤وزياد بن يحيى، والحسن بن علي، قالوا حدثنا سهل بن حماد أبو عتاب، حدثنا…٤٢٢٥قل اللهم اهدني وسددني واذكر بالهداية هداية الطريق واذكر بالسداد…٤٢٢٦حدثنا أحمد بن صالح، حدثنا ابن وهب، أخبرني سليمان بن بلال، عن شريك بن…٤٢٢٧حدثنا نصر بن علي، حدثني أبي، حدثنا عبد العزيز بن أبي رواد، عن نافع،…٤٢٢٨أن ابن عمر، كان يلبس خاتمه في يده اليسرى .٤٢٢٩قال رأيت على الصلت بن عبد الله بن نوفل بن عبد المطلب خاتما في خنصره…٤٢٣٠إن مع كل جرس شيطانا٤٢٣١لا تدخل الملائكة بيتا فيه جرس٤٢٣٢ومحمد بن عبد الله الخزاعي، - المعنى - قالا حدثنا أبو الأشهب، عن عبد…٤٢٣٣بمعناه . قال يزيد قلت لأبي الأشهب أدرك عبد الرحمن بن طرفة جده عرفجة…٤٢٣٤أن عرفجة، بمعناه .٤٢٣٥تحلى بهذا يا بنية٤٢٣٦من أحب أن يحلق حبيبه حلقة من نار فليحلقه حلقة من ذهب ومن أحب أن يطوق…٤٢٣٧يا معشر النساء أما لكن في الفضة ما تحلين به أما إنه ليس منكن امرأة…٤٢٣٨أيما امرأة تقلدت قلادة من ذهب قلدت في عنقها مثله من النار يوم القيامة…٤٢٣٩حدثنا حميد بن مسعدة، حدثنا إسماعيل، حدثنا خالد، عن ميمون القناد، عن…
