سنن أبي داود

كتاب الاستسقاء

٣٧ حديثًا
١١٦١أن رسول الله صلى الله عليه وسلم خرج بالناس ليستسقي فصلى بهم ركعتين…١١٦٢وسليمان بن داود، قالا أخبرنا ابن وهب، قال أخبرني ابن أبي ذئب، ويونس،…١١٦٣بهذا الحديث بإسناده لم يذكر الصلاة قال وحول رداءه فجعل عطافه الأيمن…١١٦٤أن عبد الله بن زيد، قال استسقى رسول الله صلى الله عليه وسلم وعليه…١١٦٥وعثمان بن أبي شيبة، نحوه قالا حدثنا حاتم بن إسماعيل، حدثنا هشام بن…١١٦٦أن عبد الله بن زيد، أخبره أن رسول الله صلى الله عليه وسلم خرج إلى…١١٦٧أنه سمع عباد بن تميم، يقول سمعت عبد الله بن زيد المازني، يقول خرج…١١٦٨وعمر بن مالك، عن ابن الهاد، عن محمد بن إبراهيم، عن عمير، مولى بني آبي…١١٦٩اللهم اسقنا غيثا مغيثا مريئا مريعا نافعا غير ضار عاجلا غير آجل١١٧٠أن النبي صلى الله عليه وسلم كان لا يرفع يديه في شىء من الدعاء إلا في…١١٧١أن النبي صلى الله عليه وسلم كان يستسقي هكذا يعني ومد يديه وجعل…١١٧٢رأى النبي صلى الله عليه وسلم يدعو عند أحجار الزيت باسطا كفيه .١١٧٣إنكم شكوتم جدب دياركم واستئخار المطر عن إبان زمانه عنكم وقد أمركم…١١٧٤حوالينا ولا علينا١١٧٥حدثنا عيسى بن حماد، أخبرنا الليث، عن سعيد المقبري، عن شريك بن عبد…١١٧٦اللهم اسق عبادك وبهائمك وانشر رحمتك وأحى بلدك الميت١١٧٧أصدق وظننت أنه يريد عائشة قال كسفت الشمس على عهد النبي صلى الله عليه…١١٧٨يا أيها الناس إن الشمس والقمر آيتان من آيات الله عز وجل لا ينكسفان…١١٧٩قال كسفت الشمس على عهد رسول الله صلى الله عليه وسلم في يوم شديد الحر…١١٨٠سمع الله لمن حمده ربنا ولك الحمد١١٨١قال كان كثير بن عباس يحدث أن عبد الله بن عباس، كان يحدث أن رسول الله…١١٨٢قال انكسفت الشمس على عهد رسول الله صلى الله عليه وسلم وإن النبي صلى…١١٨٣حدثنا مسدد، حدثنا يحيى، عن سفيان، حدثنا حبيب بن أبي ثابت، عن طاوس، عن…١١٨٤حدثنا أحمد بن يونس، حدثنا زهير، حدثنا الأسود بن قيس، حدثني ثعلبة بن…١١٨٥إنما هذه الآيات يخوف الله بها فإذا رأيتموها فصلوا كأحدث صلاة صليتموها…١١٨٦أن قبيصة الهلالي، حدثه أن الشمس كسفت بمعنى حديث موسى قال حتى بدت…١١٨٧وعبد الله بن أبي سلمة، وسليمان بن يسار، كلهم قد حدثني عن عروة، عن…١١٨٨أن رسول الله صلى الله عليه وسلم قرأ قراءة طويلة فجهر بها يعني في صلاة…١١٨٩- قال خسفت الشمس فصلى رسول الله صلى الله عليه وسلم والناس معه فقام…١١٩٠قالت كسفت الشمس فأمر رسول الله صلى الله عليه وسلم رجلا فنادى أن…١١٩١الشمس والقمر لا يخسفان لموت أحد ولا لحياته فإذا رأيتم ذلك فادعوا الله…١١٩٢قالت كان النبي صلى الله عليه وسلم يأمر بالعتاقة في صلاة الكسوف .١١٩٣قال كسفت الشمس على عهد رسول الله صلى الله عليه وسلم فجعل يصلي ركعتين…١١٩٤حدثنا موسى بن إسماعيل، حدثنا حماد، عن عطاء بن السائب، عن أبيه، عن عبد…١١٩٥حدثنا مسدد، حدثنا بشر بن المفضل، حدثنا الجريري، عن حيان بن عمير، عن…١١٩٦كانت ظلمة على عهد أنس بن مالك - قال - فأتيت أنسا فقلت يا أبا حمزة هل…١١٩٧إذا رأيتم آية فاسجدوا
