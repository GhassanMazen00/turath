سنن أبي داود

كتاب الزكاة

١٤٥ حديثًا
١٥٥٦أمرت أن أقاتل الناس حتى يقولوا لا إله إلا الله فمن قال لا إله إلا…١٥٥٧وسليمان بن داود، قالا أخبرنا ابن وهب، أخبرني يونس، عن الزهري، قال قال…١٥٥٨ليس فيما دون خمس ذود صدقة وليس فيما دون خمس أواق صدقة وليس فيما دون…١٥٥٩ليس فيما دون خمسة أوسق زكاة١٥٦٠قال الوسق ستون صاعا مختوما بالحجاجي .١٥٦١حدثنا محمد بن بشار، حدثني محمد بن عبد الله الأنصاري، حدثنا صرد بن أبي…١٥٦٢حدثنا محمد بن داود بن سفيان، حدثنا يحيى بن حسان، حدثنا سليمان بن موسى…١٥٦٣أتعطين زكاة هذا١٥٦٤ما بلغ أن تؤدى زكاته فزكي فليس بكنز١٥٦٥أن محمد بن عمرو بن عطاء، أخبره عن عبد الله بن شداد بن الهاد، أنه قال…١٥٦٦فذكر الحديث نحو حديث الخاتم . قيل لسفيان كيف تزكيه قال تضمه إلى غيره .١٥٦٧هذه فريضة الصدقة التي فرضها رسول الله صلى الله عليه وسلم على المسلمين…١٥٦٨في خمس من الإبل شاة وفي عشر شاتان وفي خمس عشرة ثلاث شياه وفي عشرين…١٥٦٩فإن لم تكن ابنة مخاض فابن لبون١٥٧٠فإذا كانت إحدى وعشرين ومائة ففيها ثلاث بنات لبون حتى تبلغ تسعا وعشرين…١٥٧١حدثنا عبد الله بن مسلمة، قال قال مالك وقول عمر بن الخطاب - رضى الله…١٥٧٢هاتوا ربع العشور من كل أربعين درهما درهم وليس عليكم شىء حتى تتم مائتى…١٥٧٣فإذا كانت لك مائتا درهم وحال عليها الحول ففيها خمسة دراهم وليس عليك…١٥٧٤قد عفوت عن الخيل والرقيق فهاتوا صدقة الرقة من كل أربعين درهما درهم…١٥٧٥في كل سائمة إبل في أربعين بنت لبون ولا يفرق إبل عن حسابها من أعطاها…١٥٧٦أن النبي صلى الله عليه وسلم لما وجهه إلى اليمن أمره أن يأخذ من البقر…١٥٧٧والنفيلي، وابن المثنى، قالوا حدثنا أبو معاوية، حدثنا الأعمش، عن…١٥٧٨حدثنا هارون بن زيد بن أبي الزرقاء، حدثنا أبي، عن سفيان، عن الأعمش، عن…١٥٧٩أن لا تأخذ من راضع لبن ولا تجمع بين مفترق ولا تفرق بين مجتمع١٥٨٠لا يجمع بين مفترق ولا يفرق بين مجتمع خشية الصدقة١٥٨١حدثنا الحسن بن علي، حدثنا وكيع، عن زكريا بن إسحاق المكي، عن عمرو بن…١٥٨٢ثلاث من فعلهن فقد طعم طعم الإيمان من عبد الله وحده وأنه لا إله إلا…١٥٨٣ذاك الذي عليك فإن تطوعت بخير آجرك الله فيه وقبلناه منك١٥٨٤إنك تأتي قوما أهل كتاب فادعهم إلى شهادة أن لا إله إلا الله وأني رسول…١٥٨٥المعتدي في الصدقة كمانعها١٥٨٦ومحمد بن عبيد، - المعنى - قالا حدثنا حماد، عن أيوب، عن رجل، يقال له…١٥٨٧ويحيى بن موسى، قالا حدثنا عبد الرزاق، عن معمر، عن أيوب، بإسناده…١٥٨٨سيأتيكم ركب مبغضون فإذا جاءوكم فرحبوا بهم وخلوا بينهم وبين ما يبتغون…١٥٨٩حدثنا أبو كامل، حدثنا عبد الواحد يعني ابن زياد، ح وحدثنا عثمان بن أبي…١٥٩٠اللهم صل على آل فلان١٥٩١لا جلب ولا جنب ولا تؤخذ صدقاتهم إلا في دورهم١٥٩٢حدثنا الحسن بن علي، حدثنا يعقوب بن إبراهيم، قال سمعت أبي يقول، عن…١٥٩٣لا تبتعه ولا تعد في صدقتك١٥٩٤ليس في الخيل والرقيق زكاة إلا زكاة الفطر في الرقيق١٥٩٥ليس على المسلم في عبده ولا في فرسه صدقة١٥٩٦فيما سقت السماء والأنهار والعيون أو كان بعلا العشر وفيما سقي بالسواني…١٥٩٧فيما سقت الأنهار والعيون العشر وما سقي بالسواني ففيه نصف العشر١٥٩٨وحسين بن الأسود العجلي، قالا قال وكيع البعل الكبوس الذي ينبت من ماء…١٥٩٩خذ الحب من الحب والشاة من الغنم والبعير من الإبل والبقرة من البقر١٦٠٠حدثنا أحمد بن أبي شعيب الحراني، حدثنا موسى بن أعين، عن عمرو بن الحارث…١٦٠١أن شبابة، - بطن من فهم - فذكر نحوه قال من كل عشر قرب قربة وقال سفيان…١٦٠٢أن بطنا، من فهم بمعنى المغيرة قال من عشر قرب قربة . وقال واديين لهم .١٦٠٣قال أمر رسول الله صلى الله عليه وسلم أن يخرص العنب كما يخرص النخل…١٦٠٤بإسناده ومعناه . قال أبو داود وسعيد لم يسمع من عتاب شيئا .١٦٠٥إذا خرصتم فخذوا ودعوا الثلث فإن لم تدعوا أو تجدوا الثلث فدعوا الربع١٦٠٦- رضى الله عنها - أنها قالت وهي تذكر شأن خيبر كان النبي صلى الله عليه…١٦٠٧حدثنا محمد بن يحيى بن فارس، حدثنا سعيد بن سليمان، حدثنا عباد، عن…١٦٠٨لو شاء رب هذه الصدقة تصدق بأطيب منها١٦٠٩وعبد الله بن عبد الرحمن السمرقندي، قالا حدثنا مروان، قال عبد الله…١٦١٠حدثنا عبد الله بن محمد النفيلي، حدثنا زهير، حدثنا موسى بن عقبة، عن…١٦١١أن رسول الله صلى الله عليه وسلم فرض زكاة الفطر - قال فيه فيما قرأه…١٦١٢حدثنا يحيى بن محمد بن السكن، حدثنا محمد بن جهضم، حدثنا إسماعيل بن…١٦١٣أن يحيى بن سعيد، وبشر بن المفضل، حدثاهم عن عبيد الله، ح وحدثنا موسى…١٦١٤قال كان الناس يخرجون صدقة الفطر على عهد رسول الله صلى الله عليه وسلم…١٦١٥وسليمان بن داود العتكي، قالا حدثنا حماد، عن أيوب، عن نافع، قال قال…١٦١٦حدثنا عبد الله بن مسلمة، حدثنا داود، - يعني ابن قيس - عن عياض بن عبد…١٦١٧" نصف صاع من بر " . وهو وهم من معاوية بن هشام أو ممن رواه عنه .١٦١٨سمع عياضا، قال سمعت أبا سعيد الخدري، يقول لا أخرج أبدا إلا صاعا إنا…١٦١٩صاع من بر أو قمح على كل اثنين صغير أو كبير حر أو عبد ذكر أو أنثى أما…١٦٢٠قال محمد بن يحيى هو بكر بن وائل بن داود أن الزهري، حدثهم عن عبد الله…١٦٢١قال وقال ابن شهاب قال عبد الله بن ثعلبة قال ابن صالح قال العدوي وإنما…١٦٢٢حدثنا محمد بن المثنى، حدثنا سهل بن يوسف، قال حميد أخبرنا عن الحسن،…١٦٢٣ما ينقم ابن جميل إلا أن كان فقيرا فأغناه الله وأما خالد بن الوليد…١٦٢٤أن العباس، سأل النبي صلى الله عليه وسلم في تعجيل صدقته قبل أن تحل…١٦٢٥أن زيادا، أو بعض الأمراء بعث عمران بن حصين على الصدقة فلما رجع قال…١٦٢٦من سأل وله ما يغنيه جاءت يوم القيامة خموش - أو خدوش - أو كدوح - في…١٦٢٧حدثنا عبد الله بن مسلمة، عن مالك، عن زيد بن أسلم، عن عطاء بن يسار، عن…١٦٢٨من سأل وله قيمة أوقية فقد ألحف١٦٢٩من سأل وعنده ما يغنيه فإنما يستكثر من النار١٦٣٠إن الله تعالى لم يرض بحكم نبي ولا غيره في الصدقات حتى حكم فيها هو…١٦٣١ليس المسكين الذي ترده التمرة والتمرتان والأكلة والأكلتان ولكن المسكين…١٦٣٢ولكن المسكين المتعفف١٦٣٣إن شئتما أعطيتكما ولا حظ فيها لغني ولا لقوي مكتسب١٦٣٤لا تحل الصدقة لغني ولا لذي مرة سوي١٦٣٥لا تحل الصدقة لغني إلا لخمسة لغاز في سبيل الله أو لعامل عليها أو…١٦٣٦حدثنا الحسن بن علي، حدثنا عبد الرزاق، أخبرنا معمر، عن زيد بن أسلم، عن…١٦٣٧لا تحل الصدقة لغني إلا في سبيل الله أو ابن السبيل أو جار فقير يتصدق…١٦٣٨زعم أن رجلا، من الأنصار يقال له سهل بن أبي حثمة أخبره أن النبي صلى…١٦٣٩المسائل كدوح يكدح بها الرجل وجهه فمن شاء أبقى على وجهه ومن شاء ترك…١٦٤٠أقم يا قبيصة حتى تأتينا الصدقة فنأمر لك بها١٦٤١أما في بيتك شىء١٦٤٢ألا تبايعون رسول الله صلى الله عليه وسلم١٦٤٣من تكفل لي أن لا يسأل الناس شيئا وأتكفل له بالجنة١٦٤٤ما يكون عندي من خير فلن أدخره عنكم ومن يستعفف يعفه الله ومن يستغن…١٦٤٥من أصابته فاقة فأنزلها بالناس لم تسد فاقته ومن أنزلها بالله أوشك الله…١٦٤٦لا وإن كنت سائلا لا بد فاسأل الصالحين١٦٤٧إذا أعطيت شيئا من غير أن تسأله فكل وتصدق١٦٤٨اليد العليا خير من اليد السفلى واليد العليا المنفقة والسفلى السائلة١٦٤٩الأيدي ثلاثة فيد الله العليا ويد المعطي التي تليها ويد السائل السفلى…١٦٥٠مولى القوم من أنفسهم وإنا لا تحل لنا الصدقة١٦٥١ومسلم بن إبراهيم، - المعنى - قالا حدثنا حماد، عن قتادة، عن أنس، أن…١٦٥٢لولا أني أخاف أن تكون صدقة لأكلتها١٦٥٣قال بعثني أبي إلى النبي صلى الله عليه وسلم في إبل أعطاها إياه من…١٦٥٤وعثمان بن أبي شيبة، قالا حدثنا محمد، - هو ابن أبي عبيدة - عن أبيه، عن…١٦٥٥حدثنا عمرو بن مرزوق، قال أخبرنا شعبة، عن قتادة، عن أنس، أن النبي صلى…١٦٥٦قد وجب أجرك ورجعت إليك في الميراث١٦٥٧حدثنا قتيبة بن سعيد، حدثنا أبو عوانة، عن عاصم بن أبي النجود، عن شقيق،…١٦٥٨ما من صاحب كنز لا يؤدي حقه إلا جعله الله يوم القيامة يحمى عليها في…١٦٥٩حدثنا جعفر بن مسافر، حدثنا ابن أبي فديك، عن هشام بن سعد، عن زيد بن…١٦٦٠حدثنا الحسن بن علي، حدثنا يزيد بن هارون، أخبرنا شعبة، عن قتادة، عن…١٦٦١قال قال رجل يا رسول الله ما حق الإبل فذكر نحوه زاد " وإعارة دلوها " .١٦٦٢واسع بن حبان، عن جابر بن عبد الله، أن النبي صلى الله عليه وسلم أمر من…١٦٦٣من كان عنده فضل ظهر فليعد به على من لا ظهر له ومن كان عنده فضل زاد…١٦٦٤إن الله لم يفرض الزكاة إلا ليطيب ما بقي من أموالكم وإنما فرض المواريث…١٦٦٥للسائل حق وإن جاء على فرس١٦٦٦حدثنا محمد بن رافع، حدثنا يحيى بن آدم، حدثنا زهير، عن شيخ، قال رأيت…١٦٦٧إن لم تجدي له شيئا تعطينه إياه إلا ظلفا محرقا فادفعيه إليه في يده١٦٦٨قالت قدمت على أمي راغبة في عهد قريش وهي راغمة مشركة فقلت يا رسول الله…١٦٦٩قالت استأذن أبي النبي صلى الله عليه وسلم فدخل بينه وبين قميصه فجعل…١٦٧٠هل منكم أحد أطعم اليوم مسكينا١٦٧١لا يسأل بوجه الله إلا الجنة١٦٧٢من استعاذ بالله فأعيذوه ومن سأل بالله فأعطوه ومن دعاكم فأجيبوه ومن…١٦٧٣يأتي أحدكم بما يملك فيقول هذه صدقة ثم يقعد يستكف الناس خير الصدقة ما…١٦٧٤خذ عنا مالك لا حاجة لنا به١٦٧٥سمع أبا سعيد الخدري، يقول دخل رجل المسجد فأمر النبي صلى الله عليه…١٦٧٦إن خير الصدقة ما ترك غنى أو تصدق به عن ظهر غنى وابدأ بمن تعول١٦٧٧جهد المقل وابدأ بمن تعول١٦٧٨وعثمان بن أبي شيبة، - وهذا حديثه - قالا حدثنا الفضل بن دكين، حدثنا…١٦٧٩أن سعدا، أتى النبي صلى الله عليه وسلم فقال أى الصدقة أعجب إليك قال "…١٦٨٠والحسن، عن سعد بن عبادة، عن النبي صلى الله عليه وسلم نحوه .١٦٨١أنه قال يا رسول الله إن أم سعد ماتت فأى الصدقة أفضل قال " الماء " .…١٦٨٢أيما مسلم كسا مسلما ثوبا على عرى كساه الله من خضر الجنة وأيما مسلم…١٦٨٣أربعون خصلة أعلاهن منيحة العنز ما يعمل رجل بخصلة منها رجاء ثوابها…١٦٨٤إن الخازن الأمين - الذي يعطي ما أمر به كاملا موفرا طيبة به نفسه حتى…١٦٨٥إذا أنفقت المرأة من بيت زوجها غير مفسدة كان لها أجر ما أنفقت ولزوجها…١٦٨٦الرطب تأكلنه وتهدينه١٦٨٧إذا أنفقت المرأة من كسب زوجها من غير أمره فلها نصف أجره١٦٨٨في المرأة تصدق من بيت زوجها قال لا إلا من قوتها والأجر بينهما ولا يحل…١٦٨٩اجعلها في قرابتك١٦٩٠آجرك الله أما إنك لو كنت أعطيتها أخوالك كان أعظم لأجرك١٦٩١تصدق به على نفسك١٦٩٢كفى بالمرء إثما أن يضيع من يقوت١٦٩٣من سره أن يبسط له في رزقه وينسأ له في أثره فليصل رحمه١٦٩٤قال الله أنا الرحمن وهي الرحم شققت لها اسما من اسمي من وصلها وصلته…١٦٩٥أن الرداد الليثي، أخبره عن عبد الرحمن بن عوف، أنه سمع رسول الله صلى…١٦٩٦لا يدخل الجنة قاطع رحم١٦٩٧ليس الواصل بالمكافئ ولكن الواصل هو الذي إذا قطعت رحمه وصلها١٦٩٨إياكم والشح فإنما هلك من كان قبلكم بالشح أمرهم بالبخل فبخلوا وأمرهم…١٦٩٩أعطي ولا توكي فيوكي عليك١٧٠٠أعطي ولا تحصي فيحصي عليك
